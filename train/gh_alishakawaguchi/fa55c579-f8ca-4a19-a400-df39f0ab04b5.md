    50→func (a *Agent) ChunkTranscript(content []byte, maxSize int) ([][]byte, error) {
    51→	if maxSize <= 0 {
    52→		return nil, errors.New("max-size must be greater than zero")
    53→	}
    54→	if len(content) == 0 {
    55→		return [][]byte{[]byte{}}, nil
    56→	}
    57→
    58→	var chunks [][]byte
    59→	for start := 0; start < len(content); start += maxSize {
    60→		end := start + maxSize
    61→		if end > len(content) {
    62→			end = len(content)
    63→		}
    64→		chunk := make([]byte, end-start)
    65→		copy(chunk, content[start:end])
    66→		chunks = append(chunks, chunk)
    67→	}
    68→	return chunks, nil
    69→}

   198→// trimTranscriptHistory trims already-checkpointed entries from a cumulative
   199→// kiro transcript using a stored offset. Returns the re-serialized trimmed
   200→// JSON and true if trimming was applied, or (nil, false) if the full
   201→// transcript should be used as-is.
   202→func (a *Agent) trimTranscriptHistory(raw []byte) ([]byte, bool) {
   203→	parsed, err := parseTranscript(raw)
   204→	if err != nil || len(parsed.History) == 0 {
   205→		return nil, false
   206→	}
   207→
   208→	offsetPath := a.transcriptOffsetPath()
   209→	prev := readTranscriptOffset(offsetPath)
   210→	totalLen := len(parsed.History)
   211→
   212→	trimFrom := 0
   213→	if prev.ConversationID == parsed.ConversationID && prev.Position > 0 && prev.Position <= totalLen {
   214→		trimFrom = prev.Position
   215→	}
   216→
   217→	// Only update offset when there are new entries to capture.
   218→	if trimFrom < totalLen {
   219→		_ = writeTranscriptOffset(offsetPath, transcriptOffset{
   220→			ConversationID: parsed.ConversationID,
   221→			Position:       totalLen,
   222→		})
   223→	}
   224→
   225→	if trimFrom == 0 || trimFrom >= totalLen {
   226→		// No trimming needed (first capture) or no new entries since last
   227→		// capture. Return full transcript so the caller always has content —
   228→		// an empty transcript would cause the CLI finalize step to fail.
   229→		return nil, false
   230→	}
   231→
   232→	parsed.History = parsed.History[trimFrom:]
   233→	filtered, err := json.Marshal(parsed)
   234→	if err != nil {
   235→		return nil, false
   236→	}
   237→	return filtered, true
   238→}

   240→func (a *Agent) ensureIDETranscript(cwd string, sessionID string) (string, error) {
   241→	sessionsDir, err := ideWorkspaceSessionsDir(cwd)
   242→	if err != nil {
   243→		return "", err
   244→	}
   245→
   246→	indexPath := filepath.Join(sessionsDir, "sessions.json")
   247→	indexData, err := os.ReadFile(indexPath)
   248→	if err != nil {
   249→		return "", fmt.Errorf("IDE sessions.json not found: %w", err)
   250→	}
   251→
   252→	var sessions []ideSessionIndexEntry
   253→	if err := json.Unmarshal(indexData, &sessions); err != nil {
   254→		return "", fmt.Errorf("failed to parse IDE sessions.json: %w", err)
   255→	}
   256→	if len(sessions) == 0 {
   257→		return "", errors.New("no IDE sessions found")
   258→	}
   259→
   260→	sort.Slice(sessions, func(i, j int) bool {
   261→		return sessions[i].DateCreated > sessions[j].DateCreated
   262→	})
   263→
   264→	ideSessionID := sessions[0].SessionID
   265→	if !filepath.IsLocal(ideSessionID + ".json") {
   266→		return "", fmt.Errorf("invalid session ID: %q", ideSessionID)
   267→	}
   268→
   269→	transcriptPath := filepath.Join(sessionsDir, ideSessionID+".json")
   270→	transcriptData, err := os.ReadFile(transcriptPath)
   271→	if err != nil {
   272→		return "", fmt.Errorf("failed to read IDE transcript %s: %w", transcriptPath, err)
   273→	}
   274→
   275→	// Enrich the IDE transcript with actual agent responses and tool calls.
   276→	// The IDE session format only stores "On it." as assistant content.
   277→	//
   278→	// Priority 1: Read Kiro IDE execution logs — these contain the full
   279→	// action trace (tool calls, responses, file modifications).
   280→	// Priority 2: Merge tool calls from post-tool-use hook JSONL file.
   281→	data := transcriptData
   282→	enriched := false
   283→	if chatSessionID := extractIDESessionID(transcriptData); chatSessionID != "" {
   284→		if execLogs, err := findExecutionLogsForSession(chatSessionID); err == nil && len(execLogs) > 0 {
   285→			if result := enrichIDETranscriptWithExecutionLogs(transcriptData, execLogs); result != nil {
   286→				data = result
   287→				enriched = true
   288→			}
   289→		}
   290→	}
   291→	if !enriched {
   292→		if toolCalls := a.readAndClearToolCalls(); len(toolCalls) > 0 {
   293→			if result := enrichIDETranscriptWithToolCalls(transcriptData, toolCalls); result != nil {
   294→				data = result
   295→			}
   296→		}
   297→	}
   298→
   299→	cachePath, err := a.cacheTranscriptPath(cwd, sessionID)
   300→	if err != nil {
   301→		return "", err
   302→	}
   303→	if err := os.WriteFile(cachePath, data, 0o600); err != nil {
   304→		return "", fmt.Errorf("failed to write cached IDE transcript: %w", err)
   305→	}
   306→	return cachePath, nil
   307→}

   309→// enrichIDETranscriptWithToolCalls converts an IDE transcript to CLI format
   310→// and injects the captured tool calls into the last assistant entry. Returns
   311→// the re-serialized CLI-format JSON, or nil if conversion fails.
   312→func enrichIDETranscriptWithToolCalls(ideData []byte, toolCalls []kiroToolCall) []byte {
   313→	converted := tryParseIDETranscript(ideData)
   314→	if converted == nil || len(converted.History) == 0 {
   315→		return nil
   316→	}
   317→
   318→	// Assign all tool calls to the last assistant entry, since post-tool-use
   319→	// hooks fire between user-prompt-submit and stop for the current turn.
   320→	toolUse := kiroToolUseContent{
   321→		ToolUse: kiroToolUsePayload{ToolUses: toolCalls},
   322→	}
   323→	toolUseJSON, err := json.Marshal(toolUse)
   324→	if err != nil {
   325→		return nil
   326→	}
   327→
   328→	lastIdx := len(converted.History) - 1
   329→	converted.History[lastIdx].Assistant = toolUseJSON
   330→
   331→	result, err := json.Marshal(converted)
   332→	if err != nil {
   333→		return nil
   334→	}
   335→	return result
   336→}

   532→func convertIDETranscript(ide *kiroIDETranscript) *kiroTranscript {
   533→	transcript := &kiroTranscript{}
   534→
   535→	var pendingUser *kiroIDEHistoryEntry
   536→	for i := range ide.History {
   537→		entry := &ide.History[i]
   538→		switch entry.Message.Role {
   539→		case "user":
   540→			if pendingUser != nil {
   541→				transcript.History = append(transcript.History, ideEntryToPaired(pendingUser, nil))
   542→			}
   543→			pendingUser = entry
   544→		case "assistant":
   545→			transcript.History = append(transcript.History, ideEntryToPaired(pendingUser, entry))
   546→			pendingUser = nil
   547→		}
   548→	}
   549→
   550→	if pendingUser != nil {
   551→		transcript.History = append(transcript.History, ideEntryToPaired(pendingUser, nil))
   552→	}
   553→
   554→	return transcript
   555→}

   557→func ideEntryToPaired(user, assistant *kiroIDEHistoryEntry) kiroHistoryEntry {
   558→	entry := kiroHistoryEntry{}
   559→
   560→	if user != nil {
   561→		prompt := extractIDEText(user.Message.Content, true)
   562→		if prompt != "" {
   563→			content, err := json.Marshal(kiroPromptContent{
   564→				Prompt: struct {
   565→					Prompt string `json:"prompt"`
   566→				}{Prompt: prompt},
   567→			})
   568→			if err == nil {
   569→				entry.User.Content = content
   570→			}
   571→		} else {
   572→			entry.User.Content = user.Message.Content
   573→		}
   574→	}
   575→
   576→	if assistant != nil {
   577→		text := extractIDEText(assistant.Message.Content, false)
   578→		if text != "" {
   579→			content, err := json.Marshal(kiroResponseContent{
   580→				Response: kiroResponsePayload{Content: text},
   581→			})
   582→			if err == nil {
   583→				entry.Assistant = content
   584→			}
   585→		} else {
   586→			entry.Assistant = assistant.Message.Content
   587→		}
   588→	}
   589→
   590→	return entry
   591→}

   593→// extractIDEText extracts text from a json.RawMessage that may be either a
   594→// plain string or an array of content blocks. When blocksFirst is true, it
   595→// tries to parse as blocks before falling back to a plain string (user
   596→// messages); otherwise it tries plain string first (assistant messages).
   597→func extractIDEText(content json.RawMessage, blocksFirst bool) string {
   598→	if len(content) == 0 {
   599→		return ""
   600→	}
   601→
   602→	tryBlocks := func() string {
   603→		var blocks []kiroIDEContentBlock
   604→		if err := json.Unmarshal(content, &blocks); err == nil && len(blocks) > 0 {
   605→			for _, block := range blocks {
   606→				if block.Type == "text" && block.Text != "" {
   607→					return block.Text
   608→				}
   609→			}
   610→		}
   611→		return ""
   612→	}
   613→
   614→	tryString := func() string {
   615→		var text string
   616→		if err := json.Unmarshal(content, &text); err == nil {
   617→			return text
   618→		}
   619→		return ""
   620→	}
   621→
   622→	if blocksFirst {
   623→		if s := tryBlocks(); s != "" {
   624→			return s
   625→		}
   626→		return tryString()
   627→	}
   628→	if s := tryString(); s != "" {
   629→		return s
   630→	}

   787→func scanExecutionLogsDir(dir string, chatSessionID string) map[string]*kiroExecutionLog {

   794→	// struct (avoids deserializing the large Actions array), then fully

   797→	for _, f := range files {
