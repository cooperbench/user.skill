# Fix Droid "(no prompt)" after commit

## Context

When a Factory AI Droid session creates checkpoints and the user commits, `entire explain` shows `(no prompt)` for the committed checkpoint. Before the commit, the temporary (shadow branch) checkpoint correctly shows the prompt.

**Root cause**: Three functions parse transcripts in an agent-aware way (Gemini vs Claude Code) but lack a Droid-specific path. Since Droid uses a different JSONL envelope (`{"type":"message","message":{"role":"user",...}}`) vs Claude Code (`{"type":"user","message":{"content":...}}`), the generic Claude Code parser fails to find user messages. The top-level `type` is always `"message"`, which doesn't match the `"user"`/`"human"` check.

Temporary checkpoints work because `SaveStep()` uses the agent-specific `ExtractPrompts()` method (in `lifecycle.go`) which correctly handles the Droid envelope. Committed checkpoints fail because condensation re-extracts prompts from the raw transcript using the generic parser.

   564→// extractUserPromptsFromLines extracts user prompts from JSONL transcript lines.
   565→// IDE-injected context tags (like <ide_opened_file>) are stripped from the results.
   566→func extractUserPromptsFromLines(lines []string) []string {
   567→	var prompts []string
   568→	for _, line := range lines {
   569→		line = strings.TrimSpace(line)
   570→		if line == "" {
   571→			continue
   572→		}
   573→
   574→		var entry map[string]interface{}
   575→		if err := json.Unmarshal([]byte(line), &entry); err != nil {
   576→			continue
   577→		}
   578→
   579→		// Check for user message (supports both "human" and "user" types)
   580→		msgType, ok := entry["type"].(string)
   581→		if !ok || (msgType != "human" && msgType != "user") {
   582→			continue
   583→		}
   584→
   585→		// Extract message content
   586→		message, ok := entry["message"].(map[string]interface{})
   587→		if !ok {
   588→			continue
   589→		}
   590→
   591→		// Handle string content
   592→		if content, ok := message["content"].(string); ok && content != "" {
   593→			cleaned := textutil.StripIDEContextTags(content)
   594→			if cleaned != "" {
   595→				prompts = append(prompts, cleaned)
   596→			}
   597→			continue
   598→		}
   599→
   600→		// Handle array content (e.g., multiple text blocks from VSCode)
   601→		if arr, ok := message["content"].([]interface{}); ok {
   602→			var texts []string
   603→			for _, item := range arr {
   604→				if m, ok := item.(map[string]interface{}); ok {

   606→						if text, ok := m["text"].(string); ok {
   607→							texts = append(texts, text)
   608→						}
   609→					}
   610→				}
   611→			}
   612→			if len(texts) > 0 {
   613→				cleaned := textutil.StripIDEContextTags(strings.Join(texts, "\n\n"))
   614→				if cleaned != "" {
   615→					prompts = append(prompts, cleaned)
   616→				}
   617→			}
   618→		}
   619→	}
   620→	return prompts
   621→}

   754→		}
