## Test Patterns

**Hook test** (renderHookWithProviders from `src/test/test-utils.tsx`):
```typescript
vi.mock('@/integrations/supabase/client', () => ({ supabase: { from: vi.fn() } }));
import { useMyHook } from '../useMyHook';
import { renderHookWithProviders } from '@/test/test-utils';

describe('useMyHook', () => {
  it('returns expected shape', () => {
    const { result } = renderHookWithProviders(() => useMyHook('id'));
    expect(result.current).toHaveProperty('data');
    expect(typeof result.current.someAction).toBe('function');
  });
});
```

**Component test** (mock child components, assert structure):
```typescript
vi.mock('./ChildComponent', () => ({ ChildComponent: () => <div data-testid="child" /> }));
import { render, screen } from '@testing-library/react';
import { MyComponent } from '../MyComponent';

describe('MyComponent', () => {
  it('renders key elements', () => {
    render(<MyComponent requiredProp="value" />);
    expect(screen.getByTestId('child')).toBeInTheDocument();
  });
});
```

**Pure function test** (direct import, test inputs/outputs):
```typescript
import { myFunction } from '../myModule';

describe('myFunction', () => {
  it('handles basic case', () => {
    expect(myFunction('input')).toBe('expected');
  });
});
```

**UI primitive test** (render + assert classes, following `button.test.tsx`):
```typescript
import { render, screen } from '@testing-library/react';
import { Badge } from '../badge';

describe('Badge', () => {
  it('renders with default classes', () => {
    render(<Badge>Test</Badge>);
    expect(screen.getByText('Test')).toBeInTheDocument();
  });
});
```
