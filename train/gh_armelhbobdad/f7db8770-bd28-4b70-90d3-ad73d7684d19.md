+function resolveFragmentLink(baseDir, href) {

- `/docs/path/file.md` → `/path/file/` (strips /docs prefix)
- `./path/index.md` → `./path/` (index becomes directory root)
- Preserves query strings and anchors: `file.md#section?param=value` → `file/?param=value#section`

**rehype-base-paths.js** (handles absolute path prefixing):
- Transforms `src="/img/foo.png"` → `src="/base/img/foo.png"` (for subdirectory deployments)
- Only affects absolute paths (`/`), preserves external URLs
- Skips `.md` links (handled by rehype-markdown-links)

**fix-doc-links.js** (auto-converter for local links):
- Converts relative paths to repo-relative with `.md` extension
- `./file.md` → `/docs/path/file.md`
- `/path/file/` → `/docs/path/file.md` or `/docs/path/file/index.md`
- Dry-run mode by default: `node tools/fix-doc-links.js --write` to apply

**validate-doc-links.js** (link checker):
- Validates site-relative links exist
- Validates anchor slugs point to real headings
- Auto-fixable links identified and reported
- Used in build pipeline to block broken builds
- Dry-run: `node tools/validate-doc-links.js`
- Fix: `node tools/validate-doc-links.js --write` (auto-fixes single matches)

    50→          }
    51→        }
    52→      }

    56→        const href = node.properties.href;
    57→
    58→        if (typeof href !== 'string') {
    59→          return;
    60→        }
    61→
    62→        // Only transform absolute paths starting with / (but not //)
    63→        if (!href.startsWith('/') || href.startsWith('//')) {
    64→          return;
    65→        }
    66→
    67→        // Skip if already has the base path
    68→        if (normalizedBase !== '/' && href.startsWith(normalizedBase)) {
    69→          return;
    70→        }
    71→
    72→        // Skip .md links - those are handled by rehype-markdown-links
    73→        // Extract path portion (before ? and #)
    74→        const firstDelimiter = Math.min(
    75→          href.indexOf('?') === -1 ? Infinity : href.indexOf('?'),
    76→          href.indexOf('#') === -1 ? Infinity : href.indexOf('#'),
    77→        );
    78→        const pathPortion = firstDelimiter === Infinity ? href : href.substring(0, firstDelimiter);
    79→

    81→          return; // Let rehype-markdown-links handle this
    82→        }
    83→
    84→        // Prepend base path
    85→        node.properties.href = normalizedBase + href.slice(1);
    86→      }
    87→    });
    88→  };
