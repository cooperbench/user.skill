def validate_frontmatter(
    content: str,
    skill_dir_name: str | None = None,
) -> dict:
    """Validate SKILL.md frontmatter against agentskills.io spec.

    """

        # Name validation

        issues.extend(_validate_name(name, skill_dir_name))

                "severity": "high",
                "field": "description",
                "message": "description field missing or empty",
