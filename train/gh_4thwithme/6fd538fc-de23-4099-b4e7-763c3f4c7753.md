const rule = {
	meta: {
		type: 'suggestion',
		docs: {
			description: 'Require custom functions to use single object parameter',
			category: 'Best Practices',
		},
		schema: [],
		messages: {
			requireObjectParam:
				'Custom functions must use a single object destructuring parameter',
		},
	},
	create(context) {
		const lifecycleHooks = [
			'onModuleInit',
			'onApplicationBootstrap',
			'onModuleDestroy',
			'onApplicationShutdown',
			'configure',
			'use',
			'catch',
			'intercept',
			'canActivate',
			'validate',
			'transform',
			'run',
		];

		const testingCallbacks = [
			'it',
			'test',
			'describe',
			'beforeEach',
			'afterEach',
			'beforeAll',
			'afterAll',
		];

		const nestjsDecorators = [
			'Get',
			'Post',
			'Put',
			'Delete',
			'Patch',
			'Options',
			'Head',
			'All',
			'UseGuards',
			'UseInterceptors',
			'UsePipes',
			'SetMetadata',
			'Option',
			'CronJobExecution',
		];

		const explicitSkipFunctions = ['transform'];

		function hasNestJSDecorator(node) {
			const parent = node.parent;
			if (parent?.type === 'MethodDefinition' && parent.decorators) {
				return parent.decorators.some((decorator) => {
					const dec = decorator;
					if (dec.expression.type === 'CallExpression') {
						const callExpr = dec.expression;
						const callee = callExpr.callee;
						if (callee.type === 'Identifier') {
							return nestjsDecorators.includes(callee.name);
						}
					}
					return false;
				});
			}
			return false;
		}

		function isFrameworkMethod(node) {
			const parent = node.parent;

			if (parent?.type === 'MethodDefinition') {
				if (parent.kind === 'constructor') {
					return true;
				}

				const key = parent.key;
				if (key?.type === 'Identifier' && lifecycleHooks.includes(key.name)) {
					return true;
				}

				if (hasNestJSDecorator(node)) {
					return true;
				}
			}

			return false;
		}

		function isTestingCallback(node) {
			const parent = node.parent;

			if (parent?.type === 'CallExpression') {
				const callee = parent.callee;
				if (callee.type === 'Identifier' && testingCallbacks.includes(callee.name)) {
					return true;
				}
			}

			return false;
		}

		function isCallbackFunction(node) {
			const parent = node.parent;

			if (!parent) return false;

			if (isTestingCallback(node)) {
				return true;
			}

			if (parent.type === 'CallExpression' || parent.type === 'NewExpression') {
				const callExpr = parent;
				if (callExpr.arguments.some((arg) => arg === node)) {
					return true;
				}
			}

			if (parent.type === 'ArrayExpression') {
				return true;
			}

			if (parent.type === 'Property') {
				const prop = parent;
				if (prop.value === node) {
					const grandParent = prop.parent;
					if (grandParent?.type === 'ObjectExpression') {
						const greatGrandParent = grandParent.parent;
						if (greatGrandParent?.type === 'CallExpression') {
							return true;
						}
					}
				}
			}

			return false;
		}

		function checkFunction(node) {
			if (isCallbackFunction(node) || isFrameworkMethod(node)) {
				return;
			}

			const funcName =
				node.id?.name ??
				(node.parent?.type === 'MethodDefinition' ? node.parent.key?.name : undefined);

			if (funcName && explicitSkipFunctions.includes(funcName)) {
				return;
			}

			const params = node.params;

			if (params.length === 0) {
				return;
			}

			if (params.length === 1 && params[0]?.type === 'RestElement') {
				return;
			}

			if (params.length === 1 && params[0]?.type === 'AssignmentPattern') {
				return;
			}

			const hasValidObjectParam =
				params.length === 1 && params[0]?.type === 'ObjectPattern';

			if (!hasValidObjectParam) {
				context.report({
					node: node,
					messageId: 'requireObjectParam',
				});
			}
		}

		return {
			FunctionDeclaration(node) {
				checkFunction(node);
			},
			FunctionExpression(node) {
				const funcNode = node;
				if (funcNode.parent?.type === 'MethodDefinition') {
					return;
				}
				checkFunction(funcNode);
			},
			ArrowFunctionExpression(node) {
				checkFunction(node);
			},
			MethodDefinition(node) {
				const methodNode = node;
				checkFunction(methodNode.value);
			},
		};
	},
};

module.exports = rule;

import { getEnv } from '../src/utils/get-env-variable.util';
import { DATABASE_URL } from '../src/constants/env-variables.constant';

const excludedMigrations = ['20250609221918_add_condensed_features_content.ts'];

async function resetTestDatabase() {
	console.log('🗑️  Resetting test database...');

	try {
		const connectionString = getEnv<string>({ key: DATABASE_URL, converter: String });

		console.log('📡 Connecting to database...');
		console.log('🔗 Connection string:', connectionString.replace(/:[^:@]*@/, ':***@'));

		const url = new URL(connectionString);
		const dbName = url.pathname.slice(1);

		const db = knex({
			client: 'mysql2',
			connection: {
				host: url.hostname,
				port: parseInt(url.port) || 3336,
				user: url.username,
				password: url.password,
				database: dbName,
			},
		});

		console.log(`✅ Connected to database: ${dbName} at ${url.hostname}:${url.port}`);

		const tablesResult = await db.raw('SHOW TABLES');
		const tables = tablesResult[0] || [];

		if (tables.length === 0) {
			console.log('📭 No tables found to drop');
		} else {
			console.log(`🗂️  Found ${tables.length} tables to drop`);

			await db.raw('SET FOREIGN_KEY_CHECKS = 0');

			for (const tableRow of tables) {
				const tableName = Object.values(tableRow)[0] as string;
				console.log(`  🗑️  Dropping table: ${tableName}`);
				await db.raw(`DROP TABLE IF EXISTS \`${tableName}\``);
			}

			await db.raw('SET FOREIGN_KEY_CHECKS = 1');

			console.log('✅ All tables dropped successfully');
		}

		await db.destroy();
		console.log('📡 Database connection closed');

		console.log('🚀 Running migrations...');

		const migrationDb = knex({
			client: 'mysql2',
			connection: connectionString,
			migrations: {
				migrationSource: {
					getMigrations: async () => {
						const fs = require('fs');
						const files = await fs.promises.readdir(
							path.join(__dirname, '../src/migrations'),
						);

						console.log('All migration files found:', files);

						const filteredFiles = files
							.filter((file: string) => file.endsWith('.ts'))
							.filter((file: string) => {
								const shouldExclude = excludedMigrations.includes(file);
								if (shouldExclude) {
									console.log(`🚫 Excluding migration: ${file}`);
								}
								return !shouldExclude;
							})
							.sort();

						return filteredFiles;
					},
					getMigrationName: (migration: string) => migration,
					getMigration: async (migration: string) => {
						return require(path.join(__dirname, '../src/migrations', migration));
					},
				},
			},
		});

		await migrationDb.migrate.latest();
		console.log('✅ Migrations completed successfully');

		await migrationDb.destroy();
		console.log('📡 Migration connection closed');

		console.log('🎉 Test database reset complete!');
	} catch (error: any) {
		console.error('❌ Error resetting test database:', error);

		if (error?.code === 'ENOTFOUND') {
			console.error('');
			console.error('💡 Troubleshooting:');
			console.error(
				'   - Make sure MySQL containers are running on localhost:3336 and localhost:3337',
			);
			console.error('   - Try running: docker ps to see running containers');
		} else if (error?.code === 'ENOENT') {
			console.error('');
			console.error('💡 Migration directory issue:');
			console.error(`   - Looking for migrations in: ${error.path}`);
			console.error('   - Actual location: src/migrations/');
		}

		process.exit(1);
	}
}

   316→### LogExecution Decorator (CRITICAL)
   317→
   318→**ALWAYS wrap important methods and 3rd party requests with `@LogExecution` decorator.**
   319→
   320→The `@LogExecution` decorator provides comprehensive logging and monitoring for method execution:
   321→
   322→- **Performance tracking**: Logs execution time with warnings if exceeds threshold
   323→- **Parameter logging**: Logs method input parameters
   324→- **Response logging**: Logs method output
   325→- **Error tracking**: Logs errors and reports to Rollbar
   326→- **Sensitive data masking**: Automatically masks passwords, tokens, auth keys
   327→- **Custom formatters**: Optional custom formatters for params/responses
   328→
   329→**Usage:**
   330→
   331→```typescript
   332→import { LogExecution } from '@decorators/log-execution.decorator';
   333→
   334→@Injectable()
   335→export class MyService {
   336→	// Basic usage - logs everything with defaults
   337→	@LogExecution()
   338→	async myMethod(param: string): Promise<Result> {
   339→		// method implementation
   340→	}
   341→
   342→	// With custom formatters (for large payloads or 3rd party responses)
   343→	@LogExecution({
   344→		paramFormatter: (params: { styleId: string }) => ({ styleId: params.styleId }),
   345→		responseFormatter: () => '', // Don't log large response
   346→	})
   347→	async fetchFromThirdParty({ styleId }: { styleId: string }): Promise<LargeData> {
   348→		// 3rd party API call
   349→	}
   350→
   351→	// With custom options
   352→	@LogExecution({
   353→		logParams: true,
   354→		logResponse: false,
   355→		logExecutionTime: true,
   356→		logErrToRollbar: true,
   357→		maskSensitiveData: true,
   358→	})
   359→	async sensitiveOperation(data: SensitiveData): Promise<void> {
   360→		// sensitive operation
   361→	}
   362→}
   363→```
   364→
   365→**When to use:**
   366→
   367→- ✅ All 3rd party API requests (Algolia, MMS, AWS, OpenAI, etc.)
   368→- ✅ Important business logic methods
   369→- ✅ Database operations that may be slow
   370→- ✅ Methods that handle sensitive data
   371→- ✅ Methods with complex error scenarios
   372→- ❌ Simple getters/setters
   373→- ❌ Private utility methods
   374→- ❌ Methods called in tight loops
   375→
   376→**Configuration options:**
   377→
   378→- `logParams`: Log method parameters (default: `true`)
   379→- `logResponse`: Log method response (default: `true`)
   380→- `logExecutionTime`: Log execution time (default: `true`)
   381→- `logErrToRollbar`: Log errors to Rollbar (default: `true`)
   382→- `logLongExecution`: Log long execution to Rollbar (default: `true`)
   383→- `rollbarLevel`: Rollbar log level (default: `'debug'`)
   384→- `maskSensitiveData`: Mask sensitive keys (default: `true`)
   385→- `sensitiveKeys`: Array of sensitive key names to mask
   386→- `customContext`: Additional context to include in logs
   387→- `paramFormatter`: Custom formatter function for parameters
   388→- `responseFormatter`: Custom formatter function for response
   389→
   390→**Default sensitive keys masked:**
   391→`password`, `pass`, `token`, `secret`, `auth`, `authorization`, `cookie`, `session`, `credentials`, `apiKey`, `_cookie`, `csrf`, `_csrf`

import { Injectable, Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';

@Injectable()
export class EnvConfigService {
	private readonly logger = new Logger(EnvConfigService.name);

	constructor(private readonly configService: ConfigService) {}

	get<T = unknown>({
		key,
		converter,
	}: {
		key: string;
		converter?: (param: unknown) => T;
	}): T {
		this.logger.log(
			`Fetching environment variable: ${key}, converter: ${converter ? converter.name : 'none'}`,
		);
		const envVariable = this.configService.get<T>(key);

		if (envVariable === undefined) {
			throw new Error(`Environment variable ${key} is not defined`);
		}

		return converter ? converter(envVariable) : envVariable;
	}
}

		key: string;

			await this.cacheService.set({

				ttl: 2000,
			});

			const retrievedValue = await this.cacheService.get<string>({
				namespace: Namespaces.HEALTH_CHECK,
				key: testKey,
			});

			}

			const latency = Date.now() - start;
