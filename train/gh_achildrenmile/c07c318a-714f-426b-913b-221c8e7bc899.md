   414→        if quiet_mask.any():
   415→            self.noise_floor[quiet_mask] = (
   416→                self.noise_floor[quiet_mask] * (1 - self.noise_alpha)
   417→                + fft_data[quiet_mask] * self.noise_alpha
   418→            )
   419→
   420→        # Update peak hold
   421→        self.peak_hold = np.maximum(
   422→            fft_data, self.peak_hold * self.peak_decay
   423→        )
   424→
   425→        for target in targets:
   426→            # Find bin range for this target
   427→            low_bin = max(0, self.freq_to_bin(target.low_hz))
   428→            high_bin = min(len(fft_data) - 1, self.freq_to_bin(target.high_hz))
   429→
   430→            if low_bin >= high_bin:
   431→                continue
   432→
   433→            # Get power in target range
   434→            target_fft = fft_data[low_bin:high_bin + 1]
   435→            peak_db = float(np.max(target_fft))
   436→            peak_bin = low_bin + int(np.argmax(target_fft))
   437→            peak_freq = self.bin_to_freq(peak_bin)
   438→
   439→            # Local noise floor in target range
   440→            local_noise = float(np.median(self.noise_floor[low_bin:high_bin + 1]))
   441→
   442→            # Effective threshold = max(absolute threshold, noise + margin)
   443→            effective_threshold = max(
   444→                target.threshold_db,
   445→                local_noise + 15  # at least 15 dB above noise floor
   446→            )
   447→
   448→            # Hysteresis: use lower threshold for "still active" state
   449→            if target.active:
   450→                release_threshold = effective_threshold - self.hysteresis_db
   451→                if peak_db < release_threshold:
   452→                    # Signal dropped below release threshold
   453→                    duration = now - target.active_since if target.active_since else 0
   454→                    target.active = False
   455→                    logger.info(
   456→                        f"⚪ Signal lost: {target.label or f'{peak_freq/1e6:.4f} MHz'} "
   457→                        f"(duration: {duration:.1f}s)"
   458→                    )
   459→                else:
   460→                    target.peak_db = max(target.peak_db, peak_db)
   461→                    target.last_seen = now
   462→            else:
   463→                if peak_db >= effective_threshold:
   464→                    # Check hold time
   465→                    if target.last_seen is None or (now - target.last_seen) > self.cooldown_s:
   466→                        target.active = True
   467→                        target.active_since = now
   468→                        target.peak_db = peak_db
   469→                        target.last_seen = now
   470→
   471→                        event = DetectionEvent(
   472→                            timestamp=datetime.now(timezone.utc),
   473→                            freq_hz=peak_freq,
   474→                            peak_db=peak_db,
   475→                            bandwidth_hz=target.bandwidth_hz,
   476→                            target_label=target.label,
   477→                        )
   478→                        events.append(event)
   479→
   480→        return events
   481→
   482→    def scan_full_band(
   483→        self, fft_data: np.ndarray, threshold_db: float = -50.0,
   484→        min_width_hz: float = 500, max_width_hz: float = 50000
   485→    ) -> list[dict]:
   486→        """
   487→        Scan the full band for any carrier (no predefined targets).
   488→        Returns list of detected signals with freq, power, estimated bandwidth.
   489→        """
   490→        freq_per_bin = self.config.bandwidth / self.config.fft_size
   491→        start_freq = self.config.center_freq - self.config.bandwidth / 2
   492→
   493→        # Find bins above threshold
   494→        above = fft_data > threshold_db
   495→        if not above.any():
   496→            return []
   497→
   498→        # Group consecutive bins into signals
   499→        signals = []
   500→        in_signal = False
   501→        sig_start = 0
   502→
   503→        for i in range(len(fft_data)):
   504→            if above[i] and not in_signal:
   505→                sig_start = i
   506→                in_signal = True
   507→            elif not above[i] and in_signal:
   508→                in_signal = False
   509→                width_hz = (i - sig_start) * freq_per_bin
   510→                if min_width_hz <= width_hz <= max_width_hz:
   511→                    peak_bin = sig_start + np.argmax(fft_data[sig_start:i])
   512→                    signals.append({
   513→                        "freq_hz": start_freq + peak_bin * freq_per_bin,
   514→                        "peak_db": float(fft_data[peak_bin]),
   515→                        "bandwidth_hz": width_hz,
   516→                        "start_bin": sig_start,
   517→                        "end_bin": i,
   518→                    })
   519→
   520→        # Handle signal at edge
   521→        if in_signal:
   522→            i = len(fft_data)
   523→            width_hz = (i - sig_start) * freq_per_bin
   524→            if min_width_hz <= width_hz <= max_width_hz:
   525→                peak_bin = sig_start + np.argmax(fft_data[sig_start:i])
   526→                signals.append({
   527→                    "freq_hz": start_freq + peak_bin * freq_per_bin,
   528→                    "peak_db": float(fft_data[peak_bin]),
   529→                    "bandwidth_hz": width_hz,
   530→                    "start_bin": sig_start,
   531→                    "end_bin": i,
   532→                })
   533→
   534→        return signals
   535→

   539→class AudioRecorder:
   540→    """Records audio data from WebSocket to WAV files."""
   541→
   542→    def __init__(self, output_dir: str = "./recordings", sample_rate: int = 12000):
   543→        self.output_dir = Path(output_dir)
   544→        self.output_dir.mkdir(parents=True, exist_ok=True)
   545→        self.sample_rate = sample_rate
   546→        self.active_recordings: dict[str, dict] = {}
   547→        self.max_duration_s = 300  # max 5 min per recording
   548→
   549→    def start_recording(self, target_label: str, freq_hz: float) -> str:
   550→        """Start a new recording. Returns filename."""
   551→        freq_mhz = freq_hz / 1e6
   552→        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
   553→        safe_label = target_label.replace(" ", "_").replace("/", "-") if target_label else "unknown"
   554→        filename = f"{timestamp}_{safe_label}_{freq_mhz:.3f}MHz.wav"
   555→        filepath = self.output_dir / filename
   556→
   557→        self.active_recordings[target_label] = {
   558→            "filepath": filepath,
   559→            "frames": [],
   560→            "start_time": time.time(),
   561→            "freq_hz": freq_hz,
   562→        }
   563→
   564→        logger.info(f"🎙️ Recording started: {filename}")
   565→        return str(filepath)
   566→
   567→    def add_audio_data(self, target_label: str, audio_data: bytes):
   568→        """Add audio data to an active recording."""
   569→        if target_label not in self.active_recordings:
   570→            return
   571→
   572→        rec = self.active_recordings[target_label]
   573→        rec["frames"].append(audio_data)
   574→
   575→        # Check max duration
   576→        if time.time() - rec["start_time"] > self.max_duration_s:
   577→            self.stop_recording(target_label)
   578→
   579→    def stop_recording(self, target_label: str) -> Optional[str]:
   580→        """Stop recording and save WAV file. Returns filepath."""
   581→        if target_label not in self.active_recordings:
   582→            return None
   583→
   584→        rec = self.active_recordings.pop(target_label)
   585→
   586→        if not rec["frames"]:
   587→            logger.warning(f"No audio data for recording: {target_label}")
   588→            return None
   589→
   590→        # Write WAV
   591→        filepath = rec["filepath"]
   592→        try:
   593→            all_audio = b"".join(rec["frames"])
   594→            with wave.open(str(filepath), "wb") as wf:
   595→                wf.setnchannels(1)
   596→                wf.setsampwidth(2)  # 16-bit
   597→                wf.setframerate(self.sample_rate)
   598→                wf.writeframes(all_audio)
   599→
   600→            duration = time.time() - rec["start_time"]
   601→            logger.info(f"💾 Recording saved: {filepath.name} ({duration:.1f}s)")
   602→            return str(filepath)
   603→        except Exception as e:
   604→            logger.error(f"Failed to save recording: {e}")
   605→            return None
   606→
   607→    def stop_all(self):
   608→        """Stop all active recordings."""
   609→        for label in list(self.active_recordings.keys()):
   610→            self.stop_recording(label)

   684→    def _parse_server_message(self, msg: str):
   685→        """Parse text messages from the OpenWebRX server."""
   686→        # OpenWebRX sends config as "MSG key=value key=value ... setup"
   687→        # or as JSON in newer versions (v1.2+):
   688→        #   {"type": "config", "value": {"center_freq": ..., "samp_rate": ...}}
   689→        if msg.startswith("MSG "):
   690→            parts = msg[4:].split()
   691→            for part in parts:
   692→                if "=" in part:
   693→                    key, value = part.split("=", 1)
   694→                    self._handle_config(key, value)
   695→        elif msg.startswith("{"):
   696→            try:
   697→                data = json.loads(msg)
   698→                if isinstance(data, dict):
   699→                    msg_type = data.get("type")
   700→                    value = data.get("value")
   701→                    if msg_type in ("config", "secondary_config") and isinstance(value, dict):
   702→                        for key, val in value.items():
   703→                            self._handle_config(key, str(val))
   704→                    elif msg_type == "receiver_details" and isinstance(value, dict):
   705→                        for key, val in value.items():
   706→                            self._handle_config(key, str(val))
   707→                    elif msg_type == "profiles" and isinstance(value, list):
   708→                        self.available_profiles = value
   709→                        logger.info(f"Received {len(value)} profiles from server")
   710→                    else:
   711→                        # Flat JSON fallback
   712→                        for key, val in data.items():
   713→                            self._handle_config(key, str(val))
   714→            except json.JSONDecodeError:
   715→                pass
   716→
   717→    def _handle_config(self, key: str, value: str):
   718→        """Handle a configuration key-value pair."""
   719→        try:
   720→            if key == "center_freq":
   721→                self.receiver_config.center_freq = float(value)
   722→                logger.info(f"  Center freq: {float(value)/1e6:.4f} MHz")
   723→            elif key == "bandwidth" or key == "samp_rate":
   724→                self.receiver_config.bandwidth = float(value)
   725→                logger.info(f"  Bandwidth: {float(value)/1e3:.1f} kHz")
   726→            elif key == "fft_size":
   727→                self.receiver_config.fft_size = int(value)
   728→                logger.info(f"  FFT size: {value}")
   729→            elif key == "fft_fps":
   730→                self.receiver_config.fft_fps = int(value)
   731→            elif key == "fft_compression":
   732→                self.receiver_config.fft_compression = value
   733→                logger.info(f"  FFT compression: {value}")
   734→            elif key == "audio_compression":
   735→                self.receiver_config.audio_compression = value
   736→            elif key == "max_clients":
   737→                self.receiver_config.max_clients = int(value)
   738→            elif key == "start_freq":
   739→                self.receiver_config.start_freq = float(value)
   740→            elif key == "end_freq":
   741→                self.receiver_config.end_freq = float(value)
   742→        except (ValueError, TypeError):
   743→            pass
   744→
   745→    def _process_fft_data(self, raw_data: bytes) -> Optional[np.ndarray]:
   746→        """
   747→        Process raw FFT data from WebSocket.
   748→        Returns dB power array or None.
   749→        """
   750→        if len(raw_data) < 2:
   751→            return None
   752→
   753→        if self.receiver_config.fft_compression == "adpcm":
   754→            decoded = self.adpcm_decoder.decode(raw_data)
   755→            self.adpcm_decoder.reset()
   756→            # Skip first 10 ADPCM warmup samples (COMPRESS_FFT_PAD_N)
   757→            # and convert from centidecibels (×100) to dB (÷100)
   758→            # See: OpenWebRX csdr compress_fft_adpcm_f_u8 / openwebrx.js
   759→            pad_n = 10
   760→            decoded = decoded[pad_n:]
   761→            fft_size = self.receiver_config.fft_size
   762→            if fft_size > 0 and len(decoded) > fft_size:
   763→                decoded = decoded[:fft_size]
   764→            return decoded / 100.0
   765→        else:
   766→            # Uncompressed: raw float32 or uint8
   767→            try:
   768→                # Try float32 first
   769→                return np.frombuffer(raw_data, dtype=np.float32)
   770→            except ValueError:
   771→                try:
   772→                    # Fall back to uint8 (mapped to dB range)
   773→                    data = np.frombuffer(raw_data, dtype=np.uint8).astype(np.float32)
   774→                    # Map 0-255 to typical dB range (-120 to 0)
   775→                    return data * (120.0 / 255.0) - 120.0
   776→                except Exception:
   777→                    return None

   804→    async def connect_and_monitor(self):
   805→        """Main connection and monitoring loop."""
   806→        logger.info(f"🔌 Connecting to {self.ws_url}...")
   807→
   808→        retry_delay = 2
   809→        max_retry = 60
   810→
   811→        while self.running:
   812→            try:
   813→                async with websockets.connect(
   814→                    self.ws_url,
   815→                    ping_interval=30,
   816→                    ping_timeout=10,
   817→                    max_size=2**20,  # 1MB max message
   818→                    additional_headers={
   819→                        "User-Agent": "BandWacht/1.0 (FunkPilot/OE8YML)"
   820→                    }
   821→                ) as ws:
   822→                    self._ws = ws
   823→                    self.connected_since = time.time()
   824→                    logger.info(f"✅ Connected to {self.base_url}")
   825→                    retry_delay = 2  # Reset retry delay on successful connection
   826→
   827→                    # Send initial configuration request
   828→                    try:
   829→                        await ws.send("SERVER DE CLIENT client=bandwacht type=receiver")
   830→                    except Exception:
   831→                        pass
   832→
   833→                    # Switch to desired profile (once per connection, avoid bot detection)
   834→                    if self.desired_profile and not self._profile_set:
   835→                        try:
   836→                            await ws.send(json.dumps({
   837→                                "type": "selectprofile",
   838→                                "params": {"profile": self.desired_profile}
   839→                            }))
   840→                            self._profile_set = True
   841→                            logger.info(f"Switched to profile: {self.desired_profile}")
   842→                        except Exception as e:
   843→                            logger.warning(f"Failed to switch profile: {e}")
   844→
   845→                    # Listen for messages
   846→                    async for message in ws:
   847→                        if not self.running:
   848→                            break
   849→
   850→                        if isinstance(message, str):
   851→                            # Text message: configuration data
   852→                            self._parse_server_message(message)
   853→
   854→                            # Initialize analyzer once we have config
   855→                            if (
   856→                                self.analyzer is None
   857→                                and self.receiver_config.center_freq > 0
   858→                                and self.receiver_config.bandwidth > 0
   859→                            ):
   860→                                self._init_analyzer()
   861→
   862→                        elif isinstance(message, bytes) and len(message) > 1:
   863→                            # Binary message: first byte = type
   864→                            msg_type = message[0]
   865→
   866→                            if msg_type == MSG_TYPE_FFT:
   867→                                await self._handle_fft(message[1:])
   868→                            elif msg_type == MSG_TYPE_AUDIO and self.recorder:
   869→                                self._handle_audio(message[1:])
   870→                            elif msg_type == MSG_TYPE_SMETER:
   871→                                pass  # Could log S-meter data
   872→
   873→            except websockets.exceptions.ConnectionClosed as e:
   874→                logger.warning(f"🔌 Connection closed: {e}")
   875→            except ConnectionRefusedError:
   876→                logger.warning(f"❌ Connection refused: {self.base_url}")
   877→            except OSError as e:
   878→                logger.warning(f"❌ Connection error: {e}")
   879→            except Exception as e:
   880→                logger.error(f"❌ Unexpected error: {type(e).__name__}: {e}")
   881→
   882→            if self.running:
   883→                logger.info(f"🔄 Reconnecting in {retry_delay}s...")
   884→                await asyncio.sleep(retry_delay)
   885→                retry_delay = min(retry_delay * 2, max_retry)
   886→
   887→    def _init_analyzer(self):
   888→        """Initialize the spectrum analyzer with receiver config."""
   889→        self.analyzer = SpectrumAnalyzer(
   890→            self.receiver_config,
   891→            threshold_db=self.threshold_db,
   892→            hysteresis_db=self.hysteresis_db,
   893→            hold_time_s=self.hold_time_s,
   894→            cooldown_s=self.cooldown_s,
   895→        )
   896→
   897→        start_hz, end_hz = self.receiver_config.freq_range()
   898→        logger.info(
   899→            f"📊 Analyzer initialized: "
   900→            f"{start_hz/1e6:.4f} - {end_hz/1e6:.4f} MHz "
   901→            f"({self.receiver_config.bandwidth/1e3:.0f} kHz BW, "
   902→            f"{self.receiver_config.fft_size} bins)"
   903→        )
   904→
   905→        # Log targets
   906→        if self.targets:
   907→            for t in self.targets:
   908→                in_range = start_hz <= t.freq_hz <= end_hz
   909→                status = "✅" if in_range else "⚠️  OUT OF RANGE"
   910→                logger.info(
   911→                    f"  🎯 Target: {t.label or 'unnamed'} "
   912→                    f"{t.freq_hz/1e6:.4f} MHz "
   913→                    f"(±{t.bandwidth_hz/2e3:.1f} kHz) "
   914→                    f"threshold={t.threshold_db:.0f} dB "
   915→                    f"{status}"
   916→                )
   917→        if self.scan_full_band:
   918→            logger.info(f"  🔍 Full band scan mode enabled (threshold: {self.threshold_db:.0f} dB)")
   919→
   920→    async def _handle_fft(self, data: bytes):
   921→        """Handle incoming FFT data."""
   922→        if self.analyzer is None:
   923→            return
   924→
   925→        fft_data = self._process_fft_data(data)
   926→        if fft_data is None:
   927→            return
   928→
   929→        self.fft_count += 1
   930→
   931→        # Specific frequency targets
   932→        if self.targets:
   933→            events = self.analyzer.analyze_band(fft_data, self.targets)
   934→            for event in events:
   935→                self.event_count += 1
   936→                await self._notify_all(event)
   937→
   938→                # Start recording if enabled
   939→                recording_file = ""
   940→                if self.recorder:
   941→                    recording_file = self.recorder.start_recording(
   942→                        event.target_label, event.freq_hz
   943→                    )
