   391→        """Convert FFT bin index to frequency in Hz."""
   392→        start_freq, end_freq = self.config.freq_range()
   393→        freq_per_bin = self.config.bandwidth / self.config.fft_size
   394→        return start_freq + bin_index * freq_per_bin
   395→
   396→    def freq_to_bin(self, freq_hz: float) -> int:
   397→        """Convert frequency in Hz to FFT bin index."""
   398→        start_freq, _ = self.config.freq_range()
   399→        freq_per_bin = self.config.bandwidth / self.config.fft_size
   400→        return int((freq_hz - start_freq) / freq_per_bin)
   401→
   402→    def analyze_band(
   403→        self, fft_data: np.ndarray, targets: list[WatchTarget]
   404→    ) -> list[DetectionEvent]:
   405→        """
   406→        Analyze FFT data against watch targets.
   407→        Returns list of new detection events.
   408→        """
   409→        now = time.time()
   410→        events = []
   411→
   412→        # Update noise floor estimate (using bins below threshold)
   413→        quiet_mask = fft_data < (self.threshold_db - 10)
   414→        if quiet_mask.any():
   415→            self.noise_floor[quiet_mask] = (
   416→                self.noise_floor[quiet_mask] * (1 - self.noise_alpha)
   417→                + fft_data[quiet_mask] * self.noise_alpha
   418→            )
   419→
   420→        # Update peak hold
   421→        self.peak_hold = np.maximum(
   422→            fft_data, self.peak_hold * self.peak_decay
   423→        )
   424→
   425→        for target in targets:
   426→            # Find bin range for this target
   427→            low_bin = max(0, self.freq_to_bin(target.low_hz))
   428→            high_bin = min(len(fft_data) - 1, self.freq_to_bin(target.high_hz))
   429→
   430→            if low_bin >= high_bin:
   431→                continue
   432→
   433→            # Get power in target range
   434→            target_fft = fft_data[low_bin:high_bin + 1]
   435→            peak_db = float(np.max(target_fft))
   436→            peak_bin = low_bin + int(np.argmax(target_fft))
   437→            peak_freq = self.bin_to_freq(peak_bin)
   438→
   439→            # Local noise floor in target range
   440→            local_noise = float(np.median(self.noise_floor[low_bin:high_bin + 1]))
   441→
   442→            # Effective threshold = max(absolute threshold, noise + margin)
   443→            effective_threshold = max(
   444→                target.threshold_db,
   445→                local_noise + 15  # at least 15 dB above noise floor
   446→            )
   447→
   448→            # Hysteresis: use lower threshold for "still active" state
   449→            if target.active:
   450→                release_threshold = effective_threshold - self.hysteresis_db
   451→                if peak_db < release_threshold:
   452→                    # Signal dropped below release threshold
   453→                    duration = now - target.active_since if target.active_since else 0
   454→                    target.active = False
   455→                    logger.info(
   456→                        f"⚪ Signal lost: {target.label or f'{peak_freq/1e6:.4f} MHz'} "
   457→                        f"(duration: {duration:.1f}s)"
   458→                    )
   459→                else:
   460→                    target.peak_db = max(target.peak_db, peak_db)
   461→                    target.last_seen = now
   462→            else:
   463→                if peak_db >= effective_threshold:
   464→                    # Check hold time
   465→                    if target.last_seen is None or (now - target.last_seen) > self.cooldown_s:
   466→                        target.active = True
   467→                        target.active_since = now
   468→                        target.peak_db = peak_db
   469→                        target.last_seen = now
   470→
   471→                        event = DetectionEvent(
   472→                            timestamp=datetime.now(timezone.utc),
   473→                            freq_hz=peak_freq,
   474→                            peak_db=peak_db,
   475→                            bandwidth_hz=target.bandwidth_hz,
   476→                            target_label=target.label,
   477→                        )
   478→                        events.append(event)
   479→
   480→        return events
   481→
   482→    def scan_full_band(
   483→        self, fft_data: np.ndarray, threshold_db: float = -50.0,
   484→        min_width_hz: float = 500, max_width_hz: float = 50000
   485→    ) -> list[dict]:
   486→        """
   487→        Scan the full band for any carrier (no predefined targets).
   488→        Returns list of detected signals with freq, power, estimated bandwidth.
   489→        """
   490→        freq_per_bin = self.config.bandwidth / self.config.fft_size
   491→        start_freq = self.config.center_freq - self.config.bandwidth / 2
   492→
   493→        # Find bins above threshold
   494→        above = fft_data > threshold_db
   495→        if not above.any():
   496→            return []
   497→
   498→        # Group consecutive bins into signals
   499→        signals = []
   500→        in_signal = False
   501→        sig_start = 0
   502→
   503→        for i in range(len(fft_data)):
   504→            if above[i] and not in_signal:
   505→                sig_start = i
   506→                in_signal = True
   507→            elif not above[i] and in_signal:
   508→                in_signal = False
   509→                width_hz = (i - sig_start) * freq_per_bin
   510→                if min_width_hz <= width_hz <= max_width_hz:
   511→                    peak_bin = sig_start + np.argmax(fft_data[sig_start:i])
   512→                    signals.append({
   513→                        "freq_hz": start_freq + peak_bin * freq_per_bin,
   514→                        "peak_db": float(fft_data[peak_bin]),
   515→                        "bandwidth_hz": width_hz,
   516→                        "start_bin": sig_start,
   517→                        "end_bin": i,
   518→                    })
   519→
   520→        # Handle signal at edge
   521→        if in_signal:
   522→            i = len(fft_data)
   523→            width_hz = (i - sig_start) * freq_per_bin
   524→            if min_width_hz <= width_hz <= max_width_hz:
   525→                peak_bin = sig_start + np.argmax(fft_data[sig_start:i])
   526→                signals.append({
   527→                    "freq_hz": start_freq + peak_bin * freq_per_bin,
   528→                    "peak_db": float(fft_data[peak_bin]),
   529→                    "bandwidth_hz": width_hz,
   530→                    "start_bin": sig_start,
   531→                    "end_bin": i,
   532→                })
   533→
   534→        return signals
   535→

   537→# ─── Audio Recorder ─────────────────────────────────────────────────────────
   538→
   539→class AudioRecorder:
   540→    """Records audio data from WebSocket to WAV files."""
   541→
   542→    def __init__(self, output_dir: str = "./recordings", sample_rate: int = 12000):
   543→        self.output_dir = Path(output_dir)
   544→        self.output_dir.mkdir(parents=True, exist_ok=True)
   545→        self.sample_rate = sample_rate
   546→        self.active_recordings: dict[str, dict] = {}
   547→        self.max_duration_s = 300  # max 5 min per recording
   548→
   549→    def start_recording(self, target_label: str, freq_hz: float) -> str:
   550→        """Start a new recording. Returns filename."""
   551→        freq_mhz = freq_hz / 1e6
   552→        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
   553→        safe_label = target_label.replace(" ", "_").replace("/", "-") if target_label else "unknown"
   554→        filename = f"{timestamp}_{safe_label}_{freq_mhz:.3f}MHz.wav"
   555→        filepath = self.output_dir / filename
   556→
   557→        self.active_recordings[target_label] = {
   558→            "filepath": filepath,
   559→            "frames": [],
   560→            "start_time": time.time(),
   561→            "freq_hz": freq_hz,
   562→        }
   563→
   564→        logger.info(f"🎙️ Recording started: {filename}")
   565→        return str(filepath)
   566→
   567→    def add_audio_data(self, target_label: str, audio_data: bytes):
   568→        """Add audio data to an active recording."""
   569→        if target_label not in self.active_recordings:
   570→            return
   571→
   572→        rec = self.active_recordings[target_label]
   573→        rec["frames"].append(audio_data)
   574→
   575→        # Check max duration
   576→        if time.time() - rec["start_time"] > self.max_duration_s:
   577→            self.stop_recording(target_label)
   578→
   579→    def stop_recording(self, target_label: str) -> Optional[str]:
   580→        """Stop recording and save WAV file. Returns filepath."""
   581→        if target_label not in self.active_recordings:
   582→            return None
   583→
   584→        rec = self.active_recordings.pop(target_label)
   585→
   586→        if not rec["frames"]:
   587→            logger.warning(f"No audio data for recording: {target_label}")
   588→            return None
   589→
   590→        # Write WAV
   591→        filepath = rec["filepath"]
   592→        try:
   593→            all_audio = b"".join(rec["frames"])
   594→            with wave.open(str(filepath), "wb") as wf:
   595→                wf.setnchannels(1)
   596→                wf.setsampwidth(2)  # 16-bit
   597→                wf.setframerate(self.sample_rate)
   598→                wf.writeframes(all_audio)
   599→
   600→            duration = time.time() - rec["start_time"]
   601→            logger.info(f"💾 Recording saved: {filepath.name} ({duration:.1f}s)")
   602→            return str(filepath)
   603→        except Exception as e:
   604→            logger.error(f"Failed to save recording: {e}")
   605→            return None
   606→
   607→    def stop_all(self):
   608→        """Stop all active recordings."""
   609→        for label in list(self.active_recordings.keys()):
   610→            self.stop_recording(label)

   646→
   647→        self.targets = targets or []
   648→        self.scan_full_band = scan_full_band
   649→        self.threshold_db = threshold_db

   664→
   665→        # Stats
   666→        self.fft_count = 0

   671→        # CSV log

   677→    def _parse_server_message(self, msg: str):
   678→        """Parse text messages from the OpenWebRX server."""
   679→        # OpenWebRX sends config as "MSG key=value key=value ... setup"
   680→        # or as JSON in newer versions
   681→        if msg.startswith("MSG "):
   682→            parts = msg[4:].split()
   683→            for part in parts:
   684→                if "=" in part:
   685→                    key, value = part.split("=", 1)
   686→                    self._handle_config(key, value)
   687→        elif msg.startswith("{"):
   688→            try:
   689→                data = json.loads(msg)
   690→                if isinstance(data, dict):
   691→                    for key, value in data.items():
   692→                        self._handle_config(key, str(value))
   693→            except json.JSONDecodeError:
   694→                pass
   695→
   696→    def _handle_config(self, key: str, value: str):
   697→        """Handle a configuration key-value pair."""
   698→        try:
   699→            if key == "center_freq":
   700→                self.receiver_config.center_freq = float(value)
   701→                logger.info(f"  Center freq: {float(value)/1e6:.4f} MHz")
   702→            elif key == "bandwidth" or key == "samp_rate":
   703→                self.receiver_config.bandwidth = float(value)
   704→                logger.info(f"  Bandwidth: {float(value)/1e3:.1f} kHz")
   705→            elif key == "fft_size":
   706→                self.receiver_config.fft_size = int(value)
   707→                logger.info(f"  FFT size: {value}")
   708→            elif key == "fft_fps":
   709→                self.receiver_config.fft_fps = int(value)
   710→            elif key == "fft_compression":
   711→                self.receiver_config.fft_compression = value
   712→                logger.info(f"  FFT compression: {value}")
   713→            elif key == "audio_compression":
   714→                self.receiver_config.audio_compression = value
   715→            elif key == "max_clients":
   716→                self.receiver_config.max_clients = int(value)
   717→            elif key == "start_freq":
   718→                self.receiver_config.start_freq = float(value)
   719→            elif key == "end_freq":
   720→                self.receiver_config.end_freq = float(value)
   721→        except (ValueError, TypeError):
   722→            pass
   723→
   724→    def _process_fft_data(self, raw_data: bytes) -> Optional[np.ndarray]:
   725→        """
   726→        Process raw FFT data from WebSocket.
   727→        Returns dB power array or None.
   728→        """
   729→        if len(raw_data) < 2:
   730→            return None
   731→
   732→        if self.receiver_config.fft_compression == "adpcm":
   733→            # ADPCM compressed: first byte is type, rest is data
   734→            decoded = self.adpcm_decoder.decode(raw_data)
   735→            # Reset decoder for each new FFT frame
   736→            self.adpcm_decoder.reset()
   737→            return decoded
   738→        else:
   739→            # Uncompressed: raw float32 or uint8
   740→            try:
   741→                # Try float32 first
   742→                return np.frombuffer(raw_data, dtype=np.float32)
   743→            except ValueError:
   744→                try:
   745→                    # Fall back to uint8 (mapped to dB range)
   746→                    data = np.frombuffer(raw_data, dtype=np.uint8).astype(np.float32)
   747→                    # Map 0-255 to typical dB range (-120 to 0)
   748→                    return data * (120.0 / 255.0) - 120.0
   749→                except Exception:
   750→                    return None
   751→
   752→    async def _notify_all(self, event: DetectionEvent):
   753→        """Send event to all notification backends."""
   754→        for notifier in self.notifiers:
   755→            try:
   756→                await notifier.send(event)
   757→            except Exception as e:
   758→                logger.error(f"Notification error ({type(notifier).__name__}): {e}")
   759→
   760→    def _log_csv(self, event: DetectionEvent, recording_file: str = ""):
   761→        """Log detection event to CSV file."""
   762→        if not self.log_csv:
   763→            return
   764→        try:
   765→            with open(self.csv_file, "a") as f:
   766→                f.write(
   767→                    f"{event.timestamp.isoformat()},"
   768→                    f"{event.freq_hz / 1e6:.4f},"
   769→                    f"{event.peak_db:.1f},"
   770→                    f"{event.bandwidth_hz:.0f},"
   771→                    f"\"{event.target_label}\","
   772→                    f"\"{recording_file}\"\n"
   773→                )
   774→        except Exception as e:
   775→            logger.error(f"CSV logging error: {e}")
   776→
   777→    async def connect_and_monitor(self):
   778→        """Main connection and monitoring loop."""
   779→        logger.info(f"🔌 Connecting to {self.ws_url}...")
   780→
   781→        retry_delay = 2
   782→        max_retry = 60
   783→
   784→        while self.running:
   785→            try:
   786→                async with websockets.connect(
   787→                    self.ws_url,
   788→                    ping_interval=30,
   789→                    ping_timeout=10,
   790→                    max_size=2**20,  # 1MB max message
   791→                    additional_headers={
   792→                        "User-Agent": "BandWacht/1.0 (FunkPilot/OE8YML)"
   793→                    }
   794→                ) as ws:
   795→                    self.connected_since = time.time()
   796→                    logger.info(f"✅ Connected to {self.base_url}")
   797→                    retry_delay = 2  # Reset retry delay on successful connection
   798→
   799→                    # Send initial configuration request
   800→                    try:
   801→                        await ws.send("SERVER DE CLIENT client=bandwacht type=receiver")
   802→                    except Exception:
   803→                        pass
   804→
   805→                    # Listen for messages
   806→                    async for message in ws:
   807→                        if not self.running:
   808→                            break
   809→
   810→                        if isinstance(message, str):
   811→                            # Text message: configuration data
   812→                            self._parse_server_message(message)
   813→
   814→                            # Initialize analyzer once we have config
   815→                            if (
   816→                                self.analyzer is None
   817→                                and self.receiver_config.center_freq > 0
   818→                                and self.receiver_config.bandwidth > 0
   819→                            ):
   820→                                self._init_analyzer()
   821→
   822→                        elif isinstance(message, bytes) and len(message) > 1:
   823→                            # Binary message: first byte = type
   824→                            msg_type = message[0]
   825→
   826→                            if msg_type == MSG_TYPE_FFT:
   827→                                await self._handle_fft(message[1:])
   828→                            elif msg_type == MSG_TYPE_AUDIO and self.recorder:
   829→                                self._handle_audio(message[1:])
   830→                            elif msg_type == MSG_TYPE_SMETER:
   831→                                pass  # Could log S-meter data
   832→
   833→            except websockets.exceptions.ConnectionClosed as e:
   834→                logger.warning(f"🔌 Connection closed: {e}")
   835→            except ConnectionRefusedError:
   836→                logger.warning(f"❌ Connection refused: {self.base_url}")
   837→            except OSError as e:
   838→                logger.warning(f"❌ Connection error: {e}")
   839→            except Exception as e:
   840→                logger.error(f"❌ Unexpected error: {type(e).__name__}: {e}")
   841→
   842→            if self.running:
   843→                logger.info(f"🔄 Reconnecting in {retry_delay}s...")
   844→                await asyncio.sleep(retry_delay)
   845→                retry_delay = min(retry_delay * 2, max_retry)
   846→
   847→    def _init_analyzer(self):
   848→        """Initialize the spectrum analyzer with receiver config."""
   849→        self.analyzer = SpectrumAnalyzer(
   850→            self.receiver_config,
   851→            threshold_db=self.threshold_db,
   852→            hysteresis_db=self.hysteresis_db,
   853→            hold_time_s=self.hold_time_s,
   854→            cooldown_s=self.cooldown_s,
   855→        )
   856→
   857→        start_hz, end_hz = self.receiver_config.freq_range()
   858→        logger.info(
   859→            f"📊 Analyzer initialized: "
   860→            f"{start_hz/1e6:.4f} - {end_hz/1e6:.4f} MHz "
   861→            f"({self.receiver_config.bandwidth/1e3:.0f} kHz BW, "
   862→            f"{self.receiver_config.fft_size} bins)"
   863→        )
   864→
   865→        # Log targets
   866→        if self.targets:
   867→            for t in self.targets:
   868→                in_range = start_hz <= t.freq_hz <= end_hz
   869→                status = "✅" if in_range else "⚠️  OUT OF RANGE"
   870→                logger.info(
   871→                    f"  🎯 Target: {t.label or 'unnamed'} "
   872→                    f"{t.freq_hz/1e6:.4f} MHz "
   873→                    f"(±{t.bandwidth_hz/2e3:.1f} kHz) "
   874→                    f"threshold={t.threshold_db:.0f} dB "
   875→                    f"{status}"
   876→                )
   877→        if self.scan_full_band:
   878→            logger.info(f"  🔍 Full band scan mode enabled (threshold: {self.threshold_db:.0f} dB)")
   879→
   880→    async def _handle_fft(self, data: bytes):
   881→        """Handle incoming FFT data."""
   882→        if self.analyzer is None:
   883→            return
   884→
   885→        fft_data = self._process_fft_data(data)
   886→        if fft_data is None:
   887→            return
   888→
   889→        self.fft_count += 1
   890→
   891→        # Specific frequency targets
   892→        if self.targets:
   893→            events = self.analyzer.analyze_band(fft_data, self.targets)
   894→            for event in events:
   895→                self.event_count += 1
   896→                await self._notify_all(event)
   897→
   898→                # Start recording if enabled
   899→                recording_file = ""
   900→                if self.recorder:
   901→                    recording_file = self.recorder.start_recording(
   902→                        event.target_label, event.freq_hz
   903→                    )
   904→                    event.recording_file = recording_file
   905→
   906→                self._log_csv(event, recording_file)
   907→
   908→        # Full band scan
   909→        if self.scan_full_band and self.fft_count % 10 == 0:  # every ~1s at 9fps
   910→            signals = self.analyzer.scan_full_band(
   911→                fft_data, threshold_db=self.threshold_db
   912→            )
   913→            for sig in signals:
   914→                freq_mhz = sig["freq_hz"] / 1e6
   915→                logger.debug(
   916→                    f"  📶 Signal: {freq_mhz:.4f} MHz "
   917→                    f"({sig['peak_db']:.1f} dB, "
   918→                    f"BW: {sig['bandwidth_hz']/1e3:.1f} kHz)"
   919→                )
   920→
   921→        # Periodic stats
   922→        if self.fft_count % 100 == 0:
   923→            uptime = time.time() - (self.connected_since or time.time())
   924→            logger.debug(

   147→## Wie es funktioniert
   148→
   149→1. **WebSocket-Verbindung**: BandWacht verbindet sich zum OpenWebRX-Server über den gleichen WebSocket-Endpunkt, den auch der Browser nutzt (`ws://host:port/ws/`)
   150→
   151→2. **FFT-Daten empfangen**: Der Server sendet komprimierte FFT-Spektrumdaten (ADPCM) die das gesamte sichtbare Band darstellen
   152→
   153→3. **Spektrumanalyse**: Die FFT-Daten werden dekomprimiert und auf definierte Frequenzbereiche analysiert. Ein adaptiver Noise-Floor-Schätzer vermeidet Fehlalarme.
   154→
   155→4. **Carrier-Erkennung**: Signale über dem Schwellwert (absolut oder relativ zum Noise Floor) lösen nach einer Haltezeit Events aus
   156→
   157→5. **Benachrichtigung & Aufnahme**: Events werden an konfigurierte Backends gesendet und optional wird Audio aufgenommen

   159→## Einschränkungen
   160→
   161→- **Ein Slot pro Verbindung**: Jede WebSocket-Verbindung belegt einen Client-Slot am Server

   163→- **Audio-Aufnahme limitiert**: Audio wird nur für die aktuell eingestellte Demodulator-Frequenz empfangen (Limitation von OpenWebRX)
