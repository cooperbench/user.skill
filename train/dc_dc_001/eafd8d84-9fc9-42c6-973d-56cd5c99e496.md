```python
"""
实验：awq_lite with/without lm_head Identity patch
测试 pre_quant_scale 一致性 + 量化速度 + 峰值显存。
"""
from __future__ import annotations

import gc
import json
import os
import time
from pathlib import Path

import torch
import modelopt.torch.quantization as mtq
from transformers import AutoConfig, AutoModelForCausalLM, AutoTokenizer

SRC        = Path("/root/shared-nvme/openbmb/models/MiniCPM-SALA")
CALIB_DATA = Path("/root/shared-nvme/openbmb/demo-sala/data/calib_mcq_niah_qa.jsonl")
MAX_LENGTH = 4096
N_SAMPLES  = 5
DTYPE      = torch.bfloat16
DEVICE     = "cuda"

os.environ.setdefault("PYTORCH_ALLOC_CONF", "expandable_segments:True")


def load_texts(n: int) -> list[str]:
    texts = []
    with open(CALIB_DATA) as f:
        for line in f:
            texts.append(json.loads(line)["question"])
    tokenizer = AutoTokenizer.from_pretrained(str(SRC), trust_remote_code=True)
    tokenizer.padding_side = "left"
    by_len = sorted(texts, key=lambda t: len(tokenizer.encode(t)), reverse=True)
    chosen = by_len[:n]
    lens = [len(tokenizer.encode(t)) for t in chosen]
    print(f"  Calibration sample token lengths: {lens}")
    return chosen


def load_model():
    torch.cuda.reset_peak_memory_stats()
    cfg = AutoConfig.from_pretrained(str(SRC), trust_remote_code=True)
    if hasattr(cfg, "sparse_config"):
        cfg.sparse_config = None
    cfg.max_position_embeddings = MAX_LENGTH * 2
    model = AutoModelForCausalLM.from_pretrained(
        str(SRC), config=cfg, torch_dtype=DTYPE,
        device_map="auto", trust_remote_code=True,
        attn_implementation="sdpa", low_cpu_mem_usage=True,
    )
    tokenizer = AutoTokenizer.from_pretrained(str(SRC), trust_remote_code=True)
    tokenizer.padding_side = "left"
    mem_loaded = torch.cuda.memory_allocated() / 1024**3
    peak_loaded = torch.cuda.max_memory_allocated() / 1024**3
    print(f"  model loaded: alloc={mem_loaded:.1f} GB  peak={peak_loaded:.1f} GB")
    return model, tokenizer


def make_fwd(tokenizer, texts, verbose=True):
    def fwd(model):
        for i, t in enumerate(texts):
            inp = tokenizer(t, return_tensors="pt", truncation=True,
                            max_length=MAX_LENGTH, padding=False)
            inp = {k: v.to(DEVICE) for k, v in inp.items()}
            actual = inp["input_ids"].shape[1]
            with torch.no_grad():
                model(**inp)
            if verbose:
                alloc = torch.cuda.memory_allocated() / 1024**3
                peak  = torch.cuda.max_memory_allocated() / 1024**3
                print(f"    calib [{i+1}/{len(texts)}] len={actual}  "
                      f"alloc={alloc:.1f}GB  peak={peak:.1f}GB")
    return fwd


def collect_scales(model) -> dict:
    scales = {}
    for name, module in model.named_modules():
        if hasattr(module, "pre_quant_scale") and module.pre_quant_scale is not None:
            scales[name] = module.pre_quant_scale.detach().cpu().clone()
    return scales


def compare_scales(s1: dict, s2: dict) -> None:
    keys1, keys2 = set(s1.keys()), set(s2.keys())
    if keys1 != keys2:
        only1 = keys1 - keys2
        only2 = keys2 - keys1
        if only1:
            print(f"  [diff] Keys only in baseline: {len(only1)}")
        if only2:
            print(f"  [diff] Keys only in patch: {len(only2)}")

    common = keys1 & keys2
    if not common:
        print("  [diff] No common keys to compare!")
        return

    max_abs_diff = 0.0
    max_rel_diff = 0.0
    total_elems = 0
    worst_key = ""
    for k in common:
        a, b = s1[k].float(), s2[k].float()
        diff = (a - b).abs()
        abs_d = diff.max().item()
        rel_d = (diff / (a.abs() + 1e-8)).max().item()
        if abs_d > max_abs_diff:
            max_abs_diff = abs_d
            worst_key = k
        max_rel_diff = max(max_rel_diff, rel_d)
        total_elems += a.numel()

    print(f"  Common scale tensors: {len(common)}, total elements: {total_elems}")
    print(f"  Max absolute diff:    {max_abs_diff:.8f}  (worst: {worst_key})")
    print(f"  Max relative diff:    {max_rel_diff:.6f} ({max_rel_diff*100:.4f}%)")
    if max_abs_diff < 1e-6:
        print("  ==> RESULT: IDENTICAL — patch has ZERO effect on AWQ scales")
    elif max_rel_diff < 0.001:
        print("  ==> RESULT: NEARLY IDENTICAL — relative diff < 0.1%")
    elif max_rel_diff < 0.01:
        print("  ==> RESULT: CLOSE — relative diff < 1%")
    else:
        print("  ==> RESULT: SIGNIFICANT DIFFERENCE — patch affects AWQ calibration")


def run_experiment():
    print("=" * 60)
    print("Experiment: awq_lite consistency with/without lm_head patch")
    print(f"  MAX_LENGTH={MAX_LENGTH}, N_SAMPLES={N_SAMPLES}, DTYPE={DTYPE}")
    print("=" * 60)

    print("\nLoading calibration texts...")
    texts = load_texts(N_SAMPLES)

    results = {}

    for patch_enabled in [False, True]:
        label = "awq_lite+patch" if patch_enabled else "awq_lite_baseline"
        print(f"\n{'─'*60}")
        print(f"Run: {label}")
        print(f"{'─'*60}")

        torch.cuda.empty_cache()
        gc.collect()
        torch.cuda.reset_peak_memory_stats()

        model, tokenizer = load_model()

        orig_lm_head = None
        if patch_enabled:
            orig_lm_head = model.lm_head
            model.lm_head = torch.nn.Identity()
            vocab_size = orig_lm_head.weight.shape[0]
            print(f"  lm_head patched to Identity "
                  f"(removes seq_len x {vocab_size} x 4B float32 tensor)")

        torch.cuda.reset_peak_memory_stats()
        t0 = time.perf_counter()

        fwd = make_fwd(tokenizer, texts, verbose=True)
        model = mtq.quantize(model, mtq.NVFP4_AWQ_LITE_CFG, forward_loop=fwd)

        elapsed = time.perf_counter() - t0
        peak_quant = torch.cuda.max_memory_allocated() / 1024**3
        mem_after  = torch.cuda.memory_allocated() / 1024**3

        print(f"\n  Quantization done in {elapsed:.1f}s")
        print(f"  Peak VRAM during quant = {peak_quant:.1f} GB")
        print(f"  VRAM after quant       = {mem_after:.1f} GB")

        if patch_enabled and orig_lm_head is not None:
            model.lm_head = orig_lm_head
            print("  lm_head restored")

        scales = collect_scales(model)
        print(f"  Collected {len(scales)} pre_quant_scale tensors")

        results[label] = {
            "elapsed": elapsed,
            "peak_quant": peak_quant,
            "mem_after": mem_after,
            "scales": scales,
        }

        del model
        gc.collect()
        torch.cuda.empty_cache()
        print("  Model deleted, cache cleared")

    print(f"\n{'='*60}")
    print("COMPARISON RESULTS")
    print(f"{'='*60}")

    r_base  = results["awq_lite_baseline"]
    r_patch = results["awq_lite+patch"]

    t_delta = r_patch['elapsed'] - r_base['elapsed']
    m_delta = r_patch['peak_quant'] - r_base['peak_quant']
    print(f"\n  {'Metric':32s} {'baseline':>12} {'patch':>12} {'delta':>10}")
    print(f"  {'─'*68}")
    print(f"  {'Quantization time (s)':32s} "
          f"{r_base['elapsed']:>12.1f} {r_patch['elapsed']:>12.1f} {t_delta:>+10.1f}")
    print(f"  {'Peak VRAM (GB)':32s} "
          f"{r_base['peak_quant']:>12.1f} {r_patch['peak_quant']:>12.1f} {m_delta:>+10.1f}")
    print(f"  {'VRAM after quant (GB)':32s} "
          f"{r_base['mem_after']:>12.1f} {r_patch['mem_after']:>12.1f}")

    speedup = r_base['elapsed'] / r_patch['elapsed'] if r_patch['elapsed'] > 0 else float('inf')
    print(f"\n  Patch speedup: {speedup:.2f}x  ({abs(t_delta):.1f}s {'saved' if t_delta < 0 else 'added'})")
    print(f"  Peak VRAM saving: {-m_delta:+.1f} GB")

    print(f"\n  Pre-quant scale consistency (AWQ smooth factor):")
    compare_scales(r_base["scales"], r_patch["scales"])

    print(f"\n{'='*60}")
    print("DONE")
    print(f"{'='*60}")


if __name__ == "__main__":
    run_experiment()
```
