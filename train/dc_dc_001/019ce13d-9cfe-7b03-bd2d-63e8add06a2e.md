```js
const http = require("http");
const { WebSocketServer } = require("ws");
const { setupRelay, getRelayStats } = require("./relay");

const port = Number(process.env.PORT || 8080);

const server = http.createServer((req, res) => {
  if (req.url === "/health") {
    const body = JSON.stringify({
      ok: true,
      ...getRelayStats(),
      now: new Date().toISOString(),
    });
    res.writeHead(200, { "content-type": "application/json; charset=utf-8" });
    res.end(body);
    return;
  }

  res.writeHead(200, { "content-type": "text/plain; charset=utf-8" });
  res.end("remodex relay alive\n");
});

const wss = new WebSocketServer({
  server,
  perMessageDeflate: false,
});

setupRelay(wss);

server.listen(port, "0.0.0.0", () => {
  console.log(`[relay] listening on :${port}`);
});
```

**为什么我不建议宿主机直跑**

不是因为做不到，而是因为你这台机器已经有明确约束：

- `80/443` 是受保护入口，不能抢
- iPhone 侧最好走 `wss://域名/relay`，而不是 `ws://IP:高端口`
- 如果你走 `ws://` 明文，中间链路就不是 TLS 了
- 如果你想自己在宿主机上搞 `443 + TLS`，那就一定会和当前 ingress 边界打架

所以从长期维护和风险控制看，宿主机 systemd/pm2 直跑不是好路。

**我建议的结构**

1. 做一个很小的 `remodex-relay` 镜像
2. 在 k3s 里建一个独立 namespace，比如 `remodex`
3. 起一个 `Deployment`
   单副本 `replicas: 1`
4. 起一个 `Service`
   Pod 内监听 `8080`
5. 起一个 `Ingress`
   独立 host，比如 `relay.zhenghandong.shop`
6. 本地 bridge 改成：
   `REMODEX_RELAY=wss://relay.zhenghandong.shop/relay`
7. 重启 bridge，手机重新扫码

- 不碰宿主机 `80/443`
- 不新增宿主机反代
- 单实例正好符合 relay 的内存态会话模型
- 跟你现有 k3s 习惯一致，后面维护最省心
- iPhone 侧仍然是标准 `wss://域名/relay`

因为它非常轻：

- 无数据库
- 无持久化要求
- 无本地磁盘依赖

- 只需要 WebSocket + 一个 `/health`
