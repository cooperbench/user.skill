```1:200:/root/shared-nvme/openbmb/quantize_nvfp4.py
"""
NVFP4 quantization for MiniCPM-SALA using nvidia-modelopt.
Calibration with awq_lite on the evaluation dataset.

Usage:
    python quantize_nvfp4.py \
        --input /root/shared-nvme/openbmb/models/MiniCPM-SALA \
        --output /root/shared-nvme/openbmb/models/MiniCPM-SALA-NVFP4 \
        --algorithm awq_lite
"""
from __future__ import annotations

import argparse
import json
import time
from pathlib import Path

import torch
import modelopt.torch.quantization as mtq
from modelopt.torch.export import export_hf_checkpoint
from transformers import AutoModelForCausalLM, AutoTokenizer


ALGO_MAP = {
    "max": mtq.NVFP4_DEFAULT_CFG,
    "awq_lite": mtq.NVFP4_AWQ_LITE_CFG,
    "awq_full": mtq.NVFP4_AWQ_FULL_CFG,
    "awq_clip": mtq.NVFP4_AWQ_CLIP_CFG,
}

CALIB_DATASET = "toolkit/eval_dataset/perf_public_set.jsonl"


def load_calibration_data(dataset_path: str) -> list[str]:
    texts = []
    with open(dataset_path) as f:
        for line in f:
            item = json.loads(line)
            texts.append(item["question"])
    return texts


def make_forward_loop(
    tokenizer: AutoTokenizer,
    calib_texts: list[str],
    max_length: int = 512,
    device: str = "cuda",
):
    def forward_loop(model):
        total = len(calib_texts)
        for i, text in enumerate(calib_texts):
            inputs = tokenizer(
                text,
                return_tensors="pt",
                truncation=True,
                max_length=max_length,
                padding=False,
            )
            inputs = {k: v.to(device) for k, v in inputs.items()}
            with torch.no_grad():
                model(**inputs)
            if (i + 1) % 10 == 0 or i == 0:
                print(f"  [calib] {i+1}/{total}")

    return forward_loop


def main():
    parser = argparse.ArgumentParser(description="NVFP4 quantization for MiniCPM-SALA")
    parser.add_argument("--input", required=True, help="Original model path")
    parser.add_argument("--output", required=True, help="Quantized model output path")
    parser.add_argument(
        "--algorithm",
        choices=list(ALGO_MAP.keys()),
        default="awq_lite",
        help="Calibration algorithm (default: awq_lite)",
    )
    parser.add_argument(
        "--calib-dataset",
        default=CALIB_DATASET,
        help="Calibration dataset (JSONL with 'question' field)",
    )
    parser.add_argument("--max-length", type=int, default=512)
    parser.add_argument(
        "--dtype",
        choices=["bfloat16", "float16"],
        default="bfloat16",
    )
    args = parser.parse_args()

    src = Path(args.input).resolve()
    dst = Path(args.output).resolve()
    dst.mkdir(parents=True, exist_ok=True)

    model_dtype = torch.bfloat16 if args.dtype == "bfloat16" else torch.float16

    # --- Step 1: Load tokenizer ---
    print(f"[1/5] Loading tokenizer from {src}")
    tokenizer = AutoTokenizer.from_pretrained(str(src), trust_remote_code=True)
    tokenizer.padding_side = "left"
    print(f"  vocab_size={tokenizer.vocab_size}, pad={tokenizer.pad_token}")

    # --- Step 2: Load model ---
    # Disable sparse_config so standard attention layers use SDPA instead of
    # InfLLMv2Attention (which requires flash-attn package).  For calibration
    # forward passes this is fine — we only need representative activations.
    from transformers import AutoConfig

    cfg = AutoConfig.from_pretrained(str(src), trust_remote_code=True)
    orig_sparse_config = getattr(cfg, "sparse_config", None)
    if hasattr(cfg, "sparse_config"):
        cfg.sparse_config = None
        print(f"  Disabled sparse_config for calibration (will restore before export)")

    print(f"[2/5] Loading model ({args.dtype}, device_map=auto, attn=sdpa)")
    # Shrink RoPE cache for calibration: max_position_embeddings drives cos/sin cache size
    # (524288 × 128 × 64 buffers = 17 GB float32). Set to max_length for calibration only.
    orig_max_pos = cfg.max_position_embeddings
    cfg.max_position_embeddings = args.max_length * 2  # plenty for calibration
    print(f"  Shrinking max_position_embeddings: {orig_max_pos} → {cfg.max_position_embeddings} (calibration only)")
    t0 = time.time()
    model = AutoModelForCausalLM.from_pretrained(
        str(src),
        config=cfg,
        dtype=model_dtype,
        device_map="auto",
        trust_remote_code=True,
        attn_implementation="sdpa",
        low_cpu_mem_usage=True,
    )
    print(f"  Model loaded in {time.time()-t0:.1f}s")
    print(f"  Class: {model.__class__.__name__}")
    mem_gb = torch.cuda.memory_allocated() / 1024**3
    print(f"  GPU memory used: {mem_gb:.1f} GB")

    # --- Step 3: Prepare calibration ---
    print(f"[3/5] Loading calibration data from {args.calib_dataset}")
    calib_texts = load_calibration_data(args.calib_dataset)
    print(f"  {len(calib_texts)} samples loaded")

    forward_loop = make_forward_loop(
        tokenizer, calib_texts, max_length=args.max_length
    )

    # --- Step 4: Quantize ---
    quant_cfg = ALGO_MAP[args.algorithm]
    print(f"[4/5] Quantizing with {args.algorithm} (NVFP4)")
    print(f"  Config: group_size=16, SE2M1 weights, FP8-E4M3 block scales")
    t0 = time.time()
    model = mtq.quantize(model, quant_cfg, forward_loop=forward_loop)
    quant_time = time.time() - t0
    print(f"  Quantization completed in {quant_time:.1f}s ({quant_time/60:.1f} min)")

    print("\n  === Quantization Summary ===")
    mtq.print_quant_summary(model)

    # --- Step 5: Export ---
    # Restore original max_position_embeddings before export so the checkpoint
    # is compatible with full-length inference (we only shrunk it for calibration).
    if hasattr(cfg, "max_position_embeddings"):
        cfg.max_position_embeddings = orig_max_pos
        model.config.max_position_embeddings = orig_max_pos
        print(f"  Restored max_position_embeddings → {orig_max_pos}")
    if orig_sparse_config is not None:
        cfg.sparse_config = orig_sparse_config
        model.config.sparse_config = orig_sparse_config
        print(f"  Restored sparse_config → {orig_sparse_config}")
    print(f"\n[5/5] Exporting to {dst}")
    t0 = time.time()
    with torch.inference_mode():
        export_hf_checkpoint(
            model,
            dtype=model_dtype,
            export_dir=str(dst),
        )
    print(f"  Export completed in {time.time()-t0:.1f}s")

    # Verify output files
    output_files = sorted(dst.iterdir())
    total_size = sum(f.stat().st_size for f in output_files if f.is_file())
    print(f"\n  Output directory: {dst}")
    print(f"  Total files: {len(output_files)}")
    print(f"  Total size: {total_size/1024**3:.2f} GB")
    for f in output_files:
        size = f.stat().st_size
        if size > 1024 * 1024:
            print(f"    {f.name}: {size/1024**2:.1f} MB")
        else:
            print(f"    {f.name}: {size/1024:.1f} KB")

    print("\n[DONE] Quantization complete!")
    print(f"  Algorithm: {args.algorithm}")
    print(f"  Time: {quant_time/60:.1f} min")
    print(f"  Output: {dst}")
    print(f"  Load with: --quantization modelopt_fp4 --model-path {dst}")


if __name__ == "__main__":
    main()
```
