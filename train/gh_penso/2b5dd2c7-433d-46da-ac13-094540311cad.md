   414→    Ok(())

   417→fn validate_sidecar_files(files: &[Value]) -> anyhow::Result<Vec<ValidatedSkillFile>> {
   418→    if files.is_empty() {
   419→        return Err(Error::message("at least one file is required").into());
   420→    }
   421→    if files.len() > MAX_SIDECAR_FILES_PER_CALL {
   422→        return Err(Error::message(format!(
   423→            "too many files: maximum is {MAX_SIDECAR_FILES_PER_CALL}"
   424→        ))
   425→        .into());
   426→    }
   427→
   428→    let mut total_bytes = 0usize;
   429→    let mut seen_paths = HashSet::new();
   430→    let mut validated = Vec::with_capacity(files.len());
   431→
   432→    for file in files {
   433→        let path = file
   434→            .get("path")
   435→            .and_then(|v| v.as_str())
   436→            .ok_or_else(|| Error::message("each file needs a string 'path'"))?;
   437→        let content = file
   438→            .get("content")
   439→            .and_then(|v| v.as_str())
   440→            .ok_or_else(|| Error::message("each file needs a string 'content'"))?;
   441→
   442→        let relative_path = normalize_relative_skill_file_path(path)?;
   443→        if !seen_paths.insert(relative_path.clone()) {
   444→            return Err(Error::message(format!(
   445→                "duplicate file path '{}'",
   446→                relative_path.display()
   447→            ))
   448→            .into());
   449→        }
   450→
   451→        let file_bytes = content.len();
   452→        if file_bytes > MAX_SIDECAR_FILE_BYTES {
   453→            return Err(Error::message(format!(
   454→                "file '{}' exceeds maximum size of {MAX_SIDECAR_FILE_BYTES} bytes",
   455→                relative_path.display()
   456→            ))
   457→            .into());
   458→        }
   459→
   460→        total_bytes += file_bytes;
   461→        if total_bytes > MAX_SIDECAR_TOTAL_BYTES {
   462→            return Err(Error::message(format!(
   463→                "total file content exceeds maximum size of {MAX_SIDECAR_TOTAL_BYTES} bytes"
   464→            ))
   465→            .into());
   466→        }
   467→
   468→        validated.push(ValidatedSkillFile {
   469→            relative_path,
   470→            content: content.to_string(),
   471→        });
   472→    }
   473→
   474→    Ok(validated)
   475→}

   477→fn normalize_relative_skill_file_path(path: &str) -> anyhow::Result<PathBuf> {
   478→    let trimmed = path.trim();
   479→    if trimmed.is_empty() {
   480→        return Err(Error::message("file path must not be empty").into());
   481→    }
   482→
   483→    let candidate = Path::new(trimmed);
   484→    if candidate.is_absolute() {
   485→        return Err(Error::message("file path must be relative").into());
   486→    }
   487→
   488→    let mut normalized = PathBuf::new();
   489→    for component in candidate.components() {
   490→        match component {
   491→            Component::Normal(segment) => {
   492→                let Some(segment_str) = segment.to_str() else {
   493→                    return Err(Error::message("file path must be valid UTF-8").into());
   494→                };
   495→                if segment_str.starts_with('.') {
   496→                    return Err(Error::message(format!(
   497→                        "hidden path components are not allowed: '{trimmed}'"
   498→                    ))
   499→                    .into());
   500→                }
   501→                normalized.push(segment);
   502→            },
   503→            Component::CurDir => {},
   504→            Component::ParentDir | Component::RootDir | Component::Prefix(_) => {
   505→                return Err(Error::message("path traversal is not allowed").into());
   506→            },
   507→        }
   508→    }
   509→
   510→    let Some(file_name) = normalized.file_name().and_then(|name| name.to_str()) else {
   511→        return Err(Error::message("file path must name a file").into());
   512→    };
   513→
   514→    if file_name.eq_ignore_ascii_case("SKILL.md") {
   515→        return Err(
   516→            Error::message("SKILL.md must be managed with create_skill/update_skill").into(),
   517→        );
   518→    }
   519→
   520→    Ok(normalized)
   521→}

   523→async fn write_sidecar_files(skill_dir: &Path, files: &[ValidatedSkillFile]) -> crate::Result<()> {
   524→    let canonical_base = tokio::fs::canonicalize(skill_dir).await?;
   525→
   526→    for file in files {
   527→        let target = skill_dir.join(&file.relative_path);
   528→        let parent = target
   529→            .parent()
   530→            .ok_or_else(|| Error::message("invalid file path"))?;
   531→        tokio::fs::create_dir_all(parent).await?;
   532→
   533→        let canonical_parent = tokio::fs::canonicalize(parent).await?;
   534→        if !canonical_parent.starts_with(&canonical_base) {
   535→            return Err(Error::message(
   536→                "can only write inside the personal skill directory",
   537→            ));
   538→        }
   539→
   540→        if let Ok(metadata) = tokio::fs::symlink_metadata(&target).await {
   541→            if metadata.file_type().is_symlink() {
   542→                return Err(Error::message(format!(
   543→                    "refusing to write through symlink '{}'",
   544→                    file.relative_path.display()
   545→                )));
   546→            }
   547→            if metadata.is_dir() {
   548→                return Err(Error::message(format!(
   549→                    "target '{}' is a directory",
   550→                    file.relative_path.display()
   551→                )));
   552→            }
   553→        }
   554→
   555→        let Some(file_name) = file
   556→            .relative_path
   557→            .file_name()
   558→            .and_then(|value| value.to_str())
   559→        else {
   560→            return Err(Error::message("invalid file name"));
   561→        };
   562→        let temp_name = format!(".{file_name}.moltis-tmp-{}", uuid::Uuid::new_v4());
   563→        let temp_path = parent.join(temp_name);
   564→
   565→        tokio::fs::write(&temp_path, &file.content).await?;
   566→        if let Err(error) = tokio::fs::rename(&temp_path, &target).await {
   567→            let _ = tokio::fs::remove_file(&temp_path).await;
   568→            return Err(error.into());
   569→        }
   570→    }
   571→
   572→    Ok(())
   573→}

   575→fn audit_sidecar_file_write(data_dir: &Path, skill_name: &str, files: &[ValidatedSkillFile]) {
   576→    let dir = data_dir.join("logs");
   577→    let path = dir.join("security-audit.jsonl");
   578→    let now_ms = std::time::SystemTime::now()
   579→        .duration_since(std::time::UNIX_EPOCH)
   580→        .unwrap_or_default()
   581→        .as_millis() as u64;
   582→    let line = serde_json::json!({
   583→        "ts": now_ms,
   584→        "event": "skills.sidecar_files.write",
   585→        "details": {
   586→            "skill": skill_name,
   587→            "files": files.iter().map(|file| {
   588→                serde_json::json!({
   589→                    "path": file.relative_path.display().to_string(),
   590→                    "bytes": file.content.len(),
   591→                })
   592→            }).collect::<Vec<_>>(),
   593→        },
   594→    })
   595→    .to_string();
   596→
   597→    let _ = (|| -> std::io::Result<()> {
   598→        std::fs::create_dir_all(&dir)?;
   599→        let mut file = std::fs::OpenOptions::new()
   600→            .create(true)
   601→            .append(true)
   602→            .open(path)?;
   603→        use std::io::Write as _;
   604→        writeln!(file, "{line}")?;
   605→        Ok(())
   606→    })();
   607→}
