   546→        | "$dynamicRef" | "$dynamicAnchor" | "$ref" => {
   547→            canonicalize_string_keyword(key, value, pointer)
   548→        }
   549→        "uniqueItems" => canonicalize_boolean_keyword("uniqueItems", value, pointer),
   550→        "$schema" => canonicalize_schema_uri(value, pointer),
   551→        "const" | "default" | JSONCOMPAT_METADATA_KEY => Ok(canonicalize_json(value)),
   552→        "examples" => {
   553→            let items = value
   554→                .as_array()
   555→                .ok_or_else(|| invalid_keyword_type(pointer, "examples", "an array", value))?;
   556→            Ok(Value::Array(items.iter().map(canonicalize_json).collect()))
   557→        }
   558→        _ => Ok(canonicalize_json(value)),
   559→    }
   560→}
   561→
   562→fn canonicalize_unknown_keyword_value(
   563→    value: &Value,
   564→    pointer: &str,
   565→    options: CanonicalizationOptions<'_>,
   566→) -> Result<Value> {
   567→    match value {
   568→        Value::Object(_) | Value::Bool(_) => canonicalize_schema_value(value, pointer, options),
   569→        Value::Array(items) => items
   570→            .iter()
   571→            .enumerate()
   572→            .map(|(index, item)| {
   573→                canonicalize_unknown_keyword_value(
   574→                    item,
   575→                    &join_pointer(pointer, &index.to_string()),
   576→                    options,
   577→                )
   578→            })
   579→            .collect::<Result<Vec<_>>>()
   580→            .map(Value::Array),
   581→        _ => Ok(canonicalize_json(value)),
   582→    }
   583→}

   585→fn rewrite_schema_object(
   586→    schema: &Value,
   587→    pointer: &str,
   588→    options: CanonicalizationOptions<'_>,
   589→) -> Result<Value> {
   590→    let Value::Object(source) = schema else {
   591→        return Ok(schema.clone());
   592→    };
   593→    let mut obj = source.clone();
   594→
   595→    normalize_schema_uri(&mut obj);
   596→    remove_not_false(&mut obj, pointer, options.local_ref_targets);
   597→    remove_empty_conditionals(&mut obj, pointer, options.local_ref_targets);
   598→    remove_without_dependencies(&mut obj, pointer, options.local_ref_targets);
   599→    dedupe_required(&mut obj, pointer)?;
   600→    dedupe_enum(&mut obj);
   601→    dedupe_schema_array(&mut obj, "allOf", pointer, options.local_ref_targets);
   602→    dedupe_schema_array(&mut obj, "anyOf", pointer, options.local_ref_targets);
   603→    simplify_allof(&mut obj, pointer, options.local_ref_targets);
   604→    if let Some(result) = simplify_anyof(&mut obj, pointer, options.local_ref_targets) {
   605→        return Ok(result);
   606→    }
   607→    if let Some(result) = simplify_oneof(&mut obj, pointer, options.local_ref_targets) {
   608→        return Ok(result);
   609→    }
   610→    fold_required_dependencies(&mut obj, pointer)?;
   611→    infer_required_properties(&mut obj, pointer)?;
   612→    if let Some(result) = normalize_numeric_bounds(&mut obj, pointer)? {
   613→        return Ok(result);
   614→    }
   615→    normalize_single_type_arrays(&mut obj, pointer)?;
   616→    normalize_type_specific_keywords(&mut obj, pointer, options.local_ref_targets);
   617→    lower_const_with_type(&mut obj);
   618→    lower_enum_with_type(&mut obj);
   619→    if let Some(result) = lower_equal_bounds_to_enum(&mut obj, pointer)? {
   620→        return Ok(result);
   621→    }
   622→    if let Some(result) = lower_const_to_enum(&mut obj) {
   623→        return Ok(result);
   624→    }
   625→    lower_boolean_and_null_types(&mut obj);
   626→    insert_implicit_type_union(&mut obj, pointer, options.local_ref_targets);
   627→    lower_type_array_to_any_of(&mut obj, pointer)?;
   628→    if let Some(result) = rewrite_unsatisfiable_object(&obj, pointer, options.local_ref_targets) {
   629→        return Ok(result);
   630→    }
   631→    fill_implicit_constraints(&mut obj);
   632→
   633→    let schema = Value::Object(sorted_object(obj));
   634→    if let Value::Object(obj) = &schema
   635→        && obj.is_empty()
   636→    {
   637→        return Ok(Value::Bool(true));
   638→    }
   639→
   640→    Ok(schema)
   641→}

   643→fn normalize_schema_uri(schema: &mut Map<String, Value>) {
   644→    let Some(Value::String(uri)) = schema.get_mut("$schema") else {
   645→        return;
   646→    };
   647→
   648→    if uri == JSON_SCHEMA_DRAFT_2020_12_WITH_FRAGMENT {
   649→        uri.pop();
   650→    }
   651→}

   653→fn canonicalize_schema_uri(value: &Value, pointer: &str) -> Result<Value> {
   654→    let Some(uri) = value.as_str() else {
   655→        return Err(invalid_keyword_type(
   656→            pointer,
   657→            "$schema",
   658→            "a URI string",
   659→            value,
   660→        ));
   661→    };
   662→
   663→    match uri {
   664→        JSON_SCHEMA_DRAFT_2020_12 | JSON_SCHEMA_DRAFT_2020_12_WITH_FRAGMENT => {
   665→            Ok(Value::String(JSON_SCHEMA_DRAFT_2020_12.to_owned()))
   666→        }
   667→        _ => Err(CanonicalizeError::UnsupportedSchemaDialect {
   668→            pointer: pointer.to_owned(),
   669→            expected_uri: JSON_SCHEMA_DRAFT_2020_12,
   670→            actual_uri: uri.to_owned(),
   671→        }),
   672→    }
   673→}

   675→fn canonicalize_string_keyword(key: &str, value: &Value, pointer: &str) -> Result<Value> {
   676→    value
   677→        .as_str()
   678→        .ok_or_else(|| invalid_keyword_type(pointer, key, "a string", value))?;
   679→    Ok(value.clone())
   680→}

   682→fn canonicalize_unsigned_integer_keyword(key: &str, value: &Value, pointer: &str) -> Result<Value> {
   683→    let Some(integer) =
   684→        semantic_integer_value_from_json(value).and_then(|integer| u64::try_from(integer).ok())
   685→    else {
   686→        return Err(invalid_keyword_type(
   687→            pointer,
   688→            key,
   689→            "a non-negative integer",
   690→            value,
   691→        ));
   692→    };
   693→    Ok(Value::Number(Number::from(integer)))
   694→}

   696→fn canonicalize_boolean_keyword(key: &str, value: &Value, pointer: &str) -> Result<Value> {
   697→    value
   698→        .as_bool()
   699→        .ok_or_else(|| invalid_keyword_type(pointer, key, "a boolean", value))?;
   700→    Ok(value.clone())
   701→}

   703→fn canonicalize_numeric_keyword(key: &str, value: &Value, pointer: &str) -> Result<Value> {
   704→    let Some(number) = value.as_f64() else {
   705→        return Err(invalid_keyword_type(pointer, key, "a finite number", value));
   706→    };
   707→    if !number.is_finite() {
   708→        return Err(CanonicalizeError::NonFiniteNumericKeyword {
   709→            pointer: pointer.to_owned(),
   710→            keyword: key.to_owned(),
   711→        });
   712→    }
   713→    if key == "multipleOf" && number <= 0.0 {
   714→        return Err(invalid_keyword_type(
   715→            pointer,
   716→            key,
   717→            "a positive number",
   718→            value,
   719→        ));
   720→    }
   721→
   722→    Ok(value.clone())
   723→}

   725→fn remove_not_false(
   726→    schema: &mut Map<String, Value>,
   727→    pointer: &str,
   728→    local_ref_targets: &BTreeSet<String>,
   729→) {
   730→    if schema.get("not") == Some(&Value::Bool(false))
   731→        && !preserve_unknown_keyword_at_pointer(&join_pointer(pointer, "not"), local_ref_targets)
   732→    {
   733→        schema.remove("not");
   734→    }
   735→}

   737→fn remove_empty_conditionals(
   738→    schema: &mut Map<String, Value>,
   739→    pointer: &str,
   740→    local_ref_targets: &BTreeSet<String>,
   741→) {
   742→    if matches!(schema.get("then"), Some(Value::Bool(true))) {
   743→        schema.remove("then");
   744→    }
   745→    if matches!(schema.get("else"), Some(Value::Bool(true))) {
   746→        schema.remove("else");
   747→    }
   748→    if let Some(if_schema) = schema.get("if")
   749→        && !schema.contains_key("then")
   750→        && !schema.contains_key("else")
   751→        && !schema_contains_resource_identifier(if_schema)
   752→        && !preserve_unknown_keyword_at_pointer(&join_pointer(pointer, "if"), local_ref_targets)
   753→    {
   754→        schema.remove("if");
   755→    }
   756→}

   758→fn remove_without_dependencies(
   759→    schema: &mut Map<String, Value>,
   760→    pointer: &str,
   761→    local_ref_targets: &BTreeSet<String>,
   762→) {
   763→    if !schema.contains_key("if") {
   764→        if schema.get("then").is_some_and(|then_schema| {
   765→            !schema_contains_resource_identifier(then_schema)
   766→                && !preserve_unknown_keyword_at_pointer(
   767→                    &join_pointer(pointer, "then"),
   768→                    local_ref_targets,
   769→                )
   770→        }) {
   771→            schema.remove("then");
   772→        }
   773→        if schema.get("else").is_some_and(|else_schema| {
   774→            !schema_contains_resource_identifier(else_schema)
   775→                && !preserve_unknown_keyword_at_pointer(
   776→                    &join_pointer(pointer, "else"),
   777→                    local_ref_targets,
   778→                )
   779→        }) {
   780→            schema.remove("else");
   781→        }
   782→    }
   783→    if !schema.contains_key("contains") {
   784→        schema.remove("minContains");
   785→        schema.remove("maxContains");
   786→    }
   787→    if !schema.contains_key("contentEncoding") {
   788→        schema.remove("contentMediaType");
   789→    }
   790→    if !schema.contains_key("contentMediaType") {
   791→        schema.remove("contentSchema");
   792→    }
   793→    if !matches!(schema.get("items"), Some(Value::Array(_))) {
   794→        schema.remove("additionalItems");
   795→    }
   796→}

   798→fn dedupe_required(schema: &mut Map<String, Value>, pointer: &str) -> Result<()> {
   799→    let Some(Value::Array(required)) = schema.get("required") else {
   800→        return Ok(());
   801→    };
   802→    let mut names = BTreeSet::new();
   803→    let required_pointer = join_pointer(pointer, "required");
   804→    for (index, name) in required.iter().enumerate() {
   805→        let Some(name) = name.as_str() else {
   806→            return Err(invalid_keyword_entry_type(
   807→                &required_pointer,
   808→                "required",
   809→                index,
   810→                "a string",
   811→                name,
   812→            ));
   813→        };
   814→        names.insert(name.to_owned());
   815→    }
   816→    schema.insert(
   817→        "required".to_owned(),
   818→        Value::Array(names.into_iter().map(Value::String).collect()),
   819→    );
   820→    Ok(())
   821→}

   823→fn dedupe_enum(schema: &mut Map<String, Value>) {
   824→    let Some(Value::Array(values)) = schema.get("enum") else {
   825→        return;
   826→    };
   827→    schema.insert("enum".to_owned(), Value::Array(sorted_unique_json(values)));
   828→}

   830→fn dedupe_schema_array(
   831→    schema: &mut Map<String, Value>,
   832→    keyword: &str,
   833→    pointer: &str,
   834→    local_ref_targets: &BTreeSet<String>,
   835→) {
   836→    let Some(Value::Array(values)) = schema.get(keyword) else {
   837→        return;
   838→    };
   839→
   840→    if preserve_unknown_keyword_at_pointer(&join_pointer(pointer, keyword), local_ref_targets) {
   841→        return;
   842→    }
   843→
   844→    let mut seen = BTreeSet::new();
   845→    let mut unique = Vec::new();
   846→    for value in values {
   847→        let key = serde_json::to_string(&canonicalize_json(value))
   848→            .expect("serializing canonical JSON value cannot fail");
   849→        if seen.insert(key) {
   850→            unique.push(value.clone());
   851→        }
   852→    }
   853→    schema.insert(keyword.to_owned(), Value::Array(unique));
   854→}

   856→fn simplify_allof(
   857→    schema: &mut Map<String, Value>,
   858→    pointer: &str,
   859→    local_ref_targets: &BTreeSet<String>,
   860→) {
   861→    let Some(Value::Array(branches)) = schema.get_mut("allOf") else {
   862→        return;
   863→    };
   864→
   865→    if preserve_unknown_keyword_at_pointer(&join_pointer(pointer, "allOf"), local_ref_targets) {
   866→        return;
   867→    }
   868→
   869→    if branches.iter().any(|branch| branch == &Value::Bool(false)) {
   870→        let preserved = if preserve_unknown_keyword_at_pointer(pointer, local_ref_targets) {
   871→            preserved_meta(schema)
   872→        } else {
   873→            preserved_terminal_meta(schema)
   874→        };
   875→        schema.clear();
   876→        schema.extend(preserved);
   877→        schema.insert("not".to_owned(), Value::Bool(true));
   878→        return;
   879→    }
   880→
   881→    branches.retain(|branch| branch != &Value::Bool(true));
   882→    if branches.is_empty() {
   883→        schema.remove("allOf");
   884→    }
   885→}

   887→fn simplify_anyof(
   888→    schema: &mut Map<String, Value>,
   889→    pointer: &str,
   890→    local_ref_targets: &BTreeSet<String>,
   891→) -> Option<Value> {
   892→    let Some(Value::Array(branches)) = schema.get_mut("anyOf") else {
   893→        return None;
   894→    };
   895→
   896→    if preserve_unknown_keyword_at_pointer(&join_pointer(pointer, "anyOf"), local_ref_targets) {
   897→        return None;
   898→    }
   899→
   900→    if branches.iter().any(|branch| branch == &Value::Bool(true)) {
   901→        schema.remove("anyOf");
   902→        return None;
   903→    }
   904→
   905→    branches.retain(|branch| branch != &Value::Bool(false));
   906→    if branches.is_empty() || (branches.len() == 1 && branches[0] == Value::Bool(false)) {
   907→        let mut obj = if preserve_unknown_keyword_at_pointer(pointer, local_ref_targets) {
   908→            preserved_meta(schema)
   909→        } else {
   910→            preserved_terminal_meta(schema)
   911→        };
   912→        obj.insert("not".to_owned(), Value::Bool(true));
   913→        return Some(Value::Object(sorted_object(obj)));
   914→    }
   915→    if branches.is_empty() {
   916→        schema.remove("anyOf");
   917→    }
   918→    None
   919→}
   920→
   921→fn simplify_oneof(
   922→    schema: &mut Map<String, Value>,
   923→    pointer: &str,
   924→    local_ref_targets: &BTreeSet<String>,
   925→) -> Option<Value> {
   926→    let Some(Value::Array(branches)) = schema.get("oneOf") else {
   927→        return None;
   928→    };
   929→
   930→    if preserve_unknown_keyword_at_pointer(&join_pointer(pointer, "oneOf"), local_ref_targets) {
   931→        return None;
   932→    }
   933→
   934→    if branches.len() == 1 && branches[0] == Value::Bool(false) {
   935→        let mut obj = if preserve_unknown_keyword_at_pointer(pointer, local_ref_targets) {
   936→            preserved_meta(schema)
   937→        } else {
   938→            preserved_terminal_meta(schema)
   939→        };
   940→        obj.insert("not".to_owned(), Value::Bool(true));
   941→        return Some(Value::Object(sorted_object(obj)));
   942→    }
   943→
   944→    if can_rewrite_oneof_to_anyof(branches) {
   945→        let branches = branches.clone();
   946→        schema.remove("oneOf");
   947→        schema.insert("anyOf".to_owned(), Value::Array(branches));
   948→    }
   949→    None
   950→}
   951→
   952→#[must_use]
   953→fn can_rewrite_oneof_to_anyof(branches: &[Value]) -> bool {
   954→    let mut seen_types = PrimitiveTypeSet::empty();
   955→    if branches.len() < 2 {
   956→        return false;
   957→    }
   958→
   959→    for branch in branches {
   960→        let Some(types) = branch_type_set(branch) else {
   961→            return false;
   962→        };
   963→        if types.is_empty() {
   964→            return false;
   965→        }
   966→        if seen_types.intersects(types) {
   967→            return false;
   968→        }
   969→        seen_types |= types;
   970→    }
   971→
   972→    true
   973→}
   974→
   975→#[must_use]
   976→fn branch_type_set(branch: &Value) -> Option<PrimitiveTypeSet> {
   977→    match branch {
   978→        Value::Bool(true) => None,
   979→        Value::Bool(false) => Some(PrimitiveTypeSet::empty()),
   980→        Value::Object(obj) => {
   981→            if let Some(Value::String(type_name)) = obj.get("type") {
   982→                return primitive_type_set(type_name.as_str());
   983→            }
   984→            if let Some(Value::Array(values)) = obj.get("enum") {
   985→                let mut types = PrimitiveTypeSet::empty();
   986→                for value in values {
   987→                    types |= value_type_name(value)?;
   988→                }
   989→                return Some(types);
   990→            }
   991→            None
   992→        }
   993→        _ => None,
   994→    }
   995→}
   996→
   997→#[must_use]
   998→fn primitive_type_set(type_name: &str) -> Option<PrimitiveTypeSet> {
   999→    match type_name {
  1000→        "null" => Some(PrimitiveType::Null.into()),
  1001→        "boolean" => Some(PrimitiveType::Boolean.into()),
  1002→        "object" => Some(PrimitiveType::Object.into()),
  1003→        "array" => Some(PrimitiveType::Array.into()),
  1004→        "string" => Some(PrimitiveType::String.into()),
  1005→        "number" => Some(
  1006→            PrimitiveTypeSet::from(PrimitiveType::Number)
  1007→                | PrimitiveTypeSet::from(PrimitiveType::Integer),
  1008→        ),
  1009→        "integer" => Some(PrimitiveType::Integer.into()),
  1010→        _ => None,
  1011→    }
  1012→}
  1013→
  1014→#[must_use]
  1015→fn value_type_name(value: &Value) -> Option<PrimitiveType> {
  1016→    match value {
  1017→        Value::Null => Some(PrimitiveType::Null),
  1018→        Value::Bool(_) => Some(PrimitiveType::Boolean),
  1019→        Value::Object(_) => Some(PrimitiveType::Object),
  1020→        Value::Array(_) => Some(PrimitiveType::Array),
  1021→        Value::String(_) => Some(PrimitiveType::String),
  1022→        Value::Number(number) if number.is_i64() || number.is_u64() => Some(PrimitiveType::Integer),
  1023→        // This is a bit unfortunate because it relies on f64 semantics, but JSON
  1024→        // Schema's "integer" type intentionally accepts values like 1.0.
  1025→        Value::Number(number) if number.as_f64().is_some_and(|value| value.fract() == 0.0) => {
  1026→            Some(PrimitiveType::Integer)
  1027→        }
  1028→        Value::Number(_) => Some(PrimitiveType::Number),
  1029→    }
  1030→}

  1077→fn fold_required_dependencies(schema: &mut Map<String, Value>, pointer: &str) -> Result<()> {
  1078→    let Some(Value::Object(dependent_required)) = schema.get("dependentRequired") else {
  1079→        return Ok(());
  1080→    };
  1081→    let Some(Value::Array(required_values)) = schema.get("required") else {
  1082→        return Ok(());
  1083→    };
  1084→
  1085→    let required_pointer = join_pointer(pointer, "required");
  1086→    let mut required = required_values
  1087→        .iter()
  1088→        .enumerate()
  1089→        .map(|(index, value)| {
  1090→            value.as_str().map(str::to_owned).ok_or_else(|| {
  1091→                invalid_keyword_entry_type(&required_pointer, "required", index, "a string", value)
  1092→            })
  1093→        })
  1094→        .collect::<Result<BTreeSet<_>>>()?;
  1095→
  1096→    let mut remaining = dependent_required.clone();
  1097→    loop {
  1098→        let mut changed = false;
  1099→        let triggers = required.iter().cloned().collect::<Vec<_>>();
  1100→        for trigger in triggers {
  1101→            let Some(Value::Array(deps)) = remaining.remove(&trigger) else {
  1102→                continue;
  1103→            };
  1104→            let trigger_pointer = join_pointer(
  1105→                &join_pointer(pointer, "dependentRequired"),
  1106→                &escape_pointer_token(&trigger),
  1107→            );
  1108→            for (index, dep) in deps.iter().enumerate() {
  1109→                let dep = dep.as_str().ok_or_else(|| {
  1110→                    invalid_keyword_entry_type(
  1111→                        &trigger_pointer,
  1112→                        "dependentRequired",
  1113→                        index,
  1114→                        "a string",
  1115→                        dep,
  1116→                    )
  1117→                })?;
  1118→                changed |= required.insert(dep.to_owned());
  1119→            }
  1120→        }
  1121→        if !changed {
  1122→            break;
  1123→        }
  1124→    }
  1125→
  1126→    schema.insert(
  1127→        "required".to_owned(),

  1184→            (maximum.as_f64(), exclusive_maximum.as_f64())

  1293→fn integer_exclusive_maximum(
  1294→    value: &Value,
  1295→    pointer: &str,
  1296→) -> Result<Option<NormalizedIntegerBound>> {
  1297→    let Some(number) = value.as_number() else {
  1298→        return Ok(None);
  1299→    };
  1300→    if let Some(value) = number.as_i64() {
  1301→        return Ok(Some(
  1302→            value
  1303→                .checked_sub(1)
  1304→                .map(|value| NormalizedIntegerBound::Inclusive(Value::Number(Number::from(value))))
  1305→                .unwrap_or(NormalizedIntegerBound::Unsatisfiable),
  1306→        ));
  1307→    }
  1308→    if let Some(value) = number.as_u64() {
  1309→        return if value == 0 {
  1310→            Ok(Some(NormalizedIntegerBound::Inclusive(Value::Number(
  1311→                Number::from(-1_i64),
  1312→            ))))
  1313→        } else {
  1314→            let maximum = (value - 1).min(i64::MAX as u64);
  1315→            Ok(Some(NormalizedIntegerBound::Inclusive(Value::Number(
  1316→                Number::from(maximum),
  1317→            ))))
  1318→        };
  1319→    }
  1320→    let Some(value) = number.as_f64() else {
  1321→        return Err(CanonicalizeError::NonFiniteNumericKeyword {
  1322→            pointer: pointer.to_owned(),
  1323→            keyword: last_pointer_token(pointer),
  1324→        });
  1325→    };
  1326→    let mut normalized = value.floor();
  1327→    if value.fract() == 0.0 {
  1328→        normalized -= 1.0;
  1329→    }
  1330→    if normalized < i64::MIN as f64 {
  1331→        return Ok(Some(NormalizedIntegerBound::Unsatisfiable));
  1332→    }
  1333→    if normalized > i64::MAX as f64 {
  1334→        return Ok(Some(NormalizedIntegerBound::Inclusive(Value::Number(
  1335→            Number::from(i64::MAX),
  1336→        ))));
  1337→    }
  1338→    Ok(Some(NormalizedIntegerBound::Inclusive(number_from_f64(
  1339→        normalized, pointer,
  1340→    )?)))
