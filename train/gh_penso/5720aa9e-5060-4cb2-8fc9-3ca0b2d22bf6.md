    39→/// Sanitize an OpenClaw agent ID into a valid Moltis agent ID.
    40→///
    41→/// - Lowercase, replace non-alphanumeric with `-`, collapse `--`, trim `-`
    42→/// - Truncate to 50 characters
    43→/// - Append `-2`, `-3` etc. if the ID collides with an existing one or `"main"`
    44→pub fn sanitize_agent_id(raw: &str, existing_ids: &HashSet<String>) -> String {
    45→    let lowered: String = raw
    46→        .chars()
    47→        .map(|c| {
    48→            if c.is_ascii_alphanumeric() {
    49→                c.to_ascii_lowercase()
    50→            } else {
    51→                '-'
    52→            }
    53→        })
    54→        .collect();
    55→
    56→    // Collapse repeated dashes, trim leading/trailing dashes
    57→    let mut collapsed = String::with_capacity(lowered.len());
    58→    let mut prev_dash = true; // treat start as dash to trim leading
    59→    for c in lowered.chars() {
    60→        if c == '-' {
    61→            if !prev_dash {
    62→                collapsed.push('-');
    63→            }
    64→            prev_dash = true;
    65→        } else {
    66→            collapsed.push(c);
    67→            prev_dash = false;
    68→        }
    69→    }
    70→    let trimmed = collapsed.trim_end_matches('-');
    71→    let truncated = if trimmed.len() > 50 {
    72→        &trimmed[..50]
    73→    } else {
    74→        trimmed
    75→    };
    76→    let base = if truncated.is_empty() {
    77→        "agent".to_string()
    78→    } else {
    79→        truncated.to_string()
    80→    };
    81→
    82→    // Avoid collision with "main" or existing IDs
    83→    if base != "main" && !existing_ids.contains(&base) {
    84→        return base;
    85→    }
    86→
    87→    for suffix in 2..=999 {
    88→        let candidate = format!("{base}-{suffix}");
    89→        if !existing_ids.contains(&candidate) {
    90→            return candidate;
    91→        }
    92→    }
    93→
    94→    // Extremely unlikely fallback
    95→    format!("{base}-overflow")
    96→}

   220→/// Resolve the workspace directory for a specific agent.
   221→///
   222→/// Priority: configured workspace path → remap basename under `~/` →
   223→/// agent's `agent/` subdir → detection workspace
   224→fn resolve_agent_workspace(
   225→    configured_workspace: Option<&str>,
   226→    agent_id: &str,
   227→    detection: &OpenClawDetection,

   230→    if let Some(ws) = configured_workspace {

   232→            PathBuf::from(ws)
   233→        } else {
   234→            detection.home_dir.join(ws)
   235→        };
   236→        if path.is_dir() {
   237→            return Some(path);
   238→        }
   239→
   240→        // 1b. Remap: basename under ~/ (cross-machine, e.g. /root/clawd → ~/clawd)
   241→        let abs_path = PathBuf::from(ws);
   242→        if let Some(basename) = abs_path.file_name()

   252→                );
   253→                return Some(remapped);
   254→            }
   255→        }
   256→    }
   257→

   259→    let agent_dir = detection.home_dir.join("agents").join(agent_id);

   284→/// Extract theme from an agent's workspace IDENTITY.md or config.
   285→fn extract_agent_identity(workspace: &Option<PathBuf>, config: &OpenClawConfig) -> Option<String> {
   286→    // Try IDENTITY.md in the workspace

   294→            if theme.is_some() {

   296→            }

   303→            identity::compose_theme(assistant.creature.clone(), assistant.vibe.clone())

   318→fn normalize_epoch(value: f64) -> Option<u64> {
   319→    if !value.is_finite() || value < 0.0 {
   320→        return None;
   321→    }
   322→    // Treat very large values as milliseconds, otherwise seconds.
   323→    let ms = if value >= 1_000_000_000_000.0 {
   324→        value
   325→    } else {
   326→        value * 1000.0
   327→    };

   334→    let nanos = dt.unix_timestamp_nanos();

   336→        return None;
   337→    }

   159→## Incremental Session Import
   160→
   161→If you continue using OpenClaw after the initial import, Moltis will detect new messages when you re-import. Sessions are compared by source file line count — if the source JSONL has grown since the last import, Moltis re-converts the full session and updates the metadata.

   171→## Automatic Background Syncing
   172→
   173→When the `file-watcher` feature is enabled (default), Moltis automatically watches the OpenClaw sessions directory for changes. Any new or appended session files are synced incrementally within seconds, without requiring a manual re-import.
   174→
   175→**How it works:**
   176→
   177→- Moltis uses OS-level file notifications (FSEvents on macOS, inotify on Linux) to detect `.jsonl` file changes in the OpenClaw sessions directory
   178→- Events are debounced with a 5-second window to batch rapid writes during active conversations
   179→- A 60-second periodic fallback ensures changes are caught even if file notifications are missed
   180→- Only sessions are synced automatically — provider keys, memory, skills, and other categories are handled by the manual import or their own dedicated watchers
   181→
   182→**What gets synced:**
   183→
   184→- New session files are imported with `oc:` prefixed keys
   185→- Existing sessions that have grown (new messages appended) are re-converted and updated
   186→- Markdown transcripts are regenerated for updated sessions so they remain searchable
   187→- Session metadata (`id`, `created_at`) is preserved across updates
   188→

   190→
   191→```
   192→openclaw: session watcher started
   193→```
   194→

   197→## Idempotency
   198→
   199→Running the import multiple times is safe:
   200→

   203→- **MCP servers** skip entries with matching names
   204→- **Sessions** use `oc:` prefixed keys that won't collide with native Moltis sessions. Unchanged sessions (same line count) are skipped; grown sessions are re-converted
   205→- **Provider keys** merge with existing keys without overwriting
   206→- **Workspace files** skip if the destination already has user content; replace only auto-seeded defaults
