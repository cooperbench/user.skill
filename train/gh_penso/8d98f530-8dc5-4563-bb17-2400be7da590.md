   322→fn compact_tool_results_newest_first_in_place(
   323→    messages: &mut [ChatMessage],
   324→    tokens_needed: usize,
   325→) -> usize {
   326→    if tokens_needed == 0 {
   327→        return 0;
   328→    }
   329→
   330→    let mut reduced = 0;
   331→    for message in messages.iter_mut().rev() {
   332→        if reduced >= tokens_needed {
   333→            break;
   334→        }
   335→
   336→        let ChatMessage::Tool {
   337→            tool_call_id,
   338→            content,
   339→        } = message
   340→        else {
   341→            continue;
   342→        };
   343→        if content == TOOL_RESULT_COMPACTION_PLACEHOLDER
   344→            || content.len() < TOOL_RESULT_COMPACTION_MIN_BYTES
   345→        {
   346→            continue;
   347→        }
   348→
   349→        let tool_call_id = tool_call_id.clone();
   350→        let original = content.clone();
   351→        let before = estimate_message_tokens(&ChatMessage::tool(&tool_call_id, &original));
   352→        *content = TOOL_RESULT_COMPACTION_PLACEHOLDER.to_string();
   353→        let after = estimate_message_tokens(&ChatMessage::tool(
   354→            &tool_call_id,
   355→            TOOL_RESULT_COMPACTION_PLACEHOLDER,
   356→        ));
   357→        let saved = before.saturating_sub(after);
   358→        if saved == 0 {
   359→            *content = original;
   360→            continue;
   361→        }
   362→
   363→        reduced = reduced.saturating_add(saved);
   364→    }
   365→
   366→    reduced
   367→}

   369→fn enforce_tool_result_context_budget(
   370→    messages: &mut [ChatMessage],
   371→    tool_schemas: &[serde_json::Value],
   372→    context_window: u32,
   373→) -> Result<(), AgentRunError> {
   374→    let context_window = context_window as usize;
   375→    if context_window == 0 || !has_tool_result_messages(messages) {
   376→        return Ok(());
   377→    }
   378→
   379→    let compaction_budget =
   380→        context_window.saturating_mul(TOOL_RESULT_COMPACTION_RATIO_PERCENT) / 100;
   381→    let overflow_budget = context_window.saturating_mul(PREEMPTIVE_OVERFLOW_RATIO_PERCENT) / 100;
   382→    let current_tokens = estimate_prompt_tokens(messages, tool_schemas);
   383→
   384→    if current_tokens > compaction_budget {
   385→        let needed = current_tokens.saturating_sub(compaction_budget);
   386→        let reduced = compact_tool_results_newest_first_in_place(messages, needed);
   387→        tracing::debug!(
   388→            current_tokens,
   389→            compaction_budget,
   390→            overflow_budget,
   391→            needed,
   392→            reduced,
   393→            "compacted newest tool results to preserve prompt budget"
   394→        );
   395→    }
   396→
   397→    let post_compaction_tokens = estimate_prompt_tokens(messages, tool_schemas);
   398→    if post_compaction_tokens > overflow_budget {
   399→        return Err(AgentRunError::ContextWindowExceeded(format!(
   400→            "preemptive context overflow: estimated prompt size {post_compaction_tokens} tokens exceeds {overflow_budget} token budget after tool-result compaction"
   401→        )));
   402→    }
   403→
   404→    Ok(())
   405→}
