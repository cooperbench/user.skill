   393→    async fn exec(&self, id: &SandboxId, command: &str, opts: &ExecOpts) -> Result<ExecResult> {
   394→        let name = self.container_name(id);
   395→
   396→        let mut args = vec!["exec".to_string()];
   397→
   398→        if let Some(ref dir) = opts.working_dir {
   399→            args.extend(["-w".to_string(), dir.display().to_string()]);
   400→        }
   401→
   402→        // Inject proxy env vars so traffic routes through the trusted-network
   403→        // proxy running on the host.
   404→        args.extend(self.proxy_exec_env_args());
   405→
   406→        for (k, v) in &opts.env {
   407→            args.extend(["-e".to_string(), format!("{}={}", k, v)]);
   408→        }
   409→
   410→        args.push(name);
   411→        args.extend(["sh".to_string(), "-c".to_string(), command.to_string()]);
   412→
   413→        let child = tokio::process::Command::new(self.cli)
   414→            .args(&args)
   415→            .stdout(std::process::Stdio::piped())
   416→            .stderr(std::process::Stdio::piped())
   417→            .stdin(std::process::Stdio::null())
   418→            .spawn()?;
   419→
   420→        let result = tokio::time::timeout(opts.timeout, child.wait_with_output()).await;
   421→
   422→        match result {
   423→            Ok(Ok(output)) => {
   424→                let mut stdout = String::from_utf8_lossy(&output.stdout).into_owned();
   425→                let mut stderr = String::from_utf8_lossy(&output.stderr).into_owned();
   426→
   427→                truncate_output_for_display(&mut stdout, opts.max_output_bytes);
   428→                truncate_output_for_display(&mut stderr, opts.max_output_bytes);
   429→
   430→                Ok(ExecResult {
   431→                    stdout,
   432→                    stderr,
   433→                    exit_code: output.status.code().unwrap_or(-1),
   434→                })
   435→            },
   436→            Ok(Err(e)) => {
   437→                return Err(Error::message(format!("{} exec failed: {e}", self.cli)));
   438→            },
   439→            Err(_) => {
   440→                return Err(Error::message(format!(
   441→                    "{} exec timed out after {}s",
   442→                    self.cli,
   443→                    opts.timeout.as_secs()
   444→                )));
   445→            },
   446→        }
   447→    }

   564→pub(crate) fn podman_supports_host_gateway() -> bool {
   565→    let Ok(output) = std::process::Command::new("podman")

   581→/// Resolve the host IP that a Podman container (< 5.0) can use to reach the
   582→/// host.  Rootless Podman defaults to slirp4netns where the host is always
   583→/// `10.0.2.2`.  Rootful Podman uses a bridge network whose gateway we query
   584→/// with `podman network inspect`.
   585→pub(crate) fn podman_resolve_host_ip() -> Option<String> {
   586→    let rootless = std::process::Command::new("podman")
   587→        .args(["info", "--format", "{{.Host.Security.Rootless}}"])
   588→        .output()
   589→        .ok()
   590→        .map(|o| String::from_utf8_lossy(&o.stdout).trim().to_string());
   591→
   592→    if rootless.as_deref() == Some("true") {
   593→        // slirp4netns (default rootless network before Podman 5.0) maps the
   594→        // host to 10.0.2.2.
   595→        return Some("10.0.2.2".to_string());
   596→    }
   597→
   598→    // Rootful — ask for the gateway of the default "podman" network.
   599→    let output = std::process::Command::new("podman")
   600→        .args([
   601→            "network",
   602→            "inspect",
   603→            "podman",
   604→            "--format",
   605→            "{{(index .Subnets 0).Gateway}}",
   606→        ])
   607→        .output()
   608→        .ok()?;
   609→    let gateway = String::from_utf8_lossy(&output.stdout).trim().to_string();
   610→    if gateway.is_empty() {
   611→        None
   612→    } else {
   613→        Some(gateway)
   614→    }
   615→}
