# Fix #796: Sandbox image not exported to Podman store after BuildKit build

## Context

**Issue:** [#796](https://github.com/moltis-org/moltis/issues/796)
**Environment:** Podman 5.4.2, Debian 13, `backend = "auto"`

When Podman delegates `podman build` to BuildKit (via a `buildx_buildkit_default` container),
the build exits 0 but the image lands in BuildKit's internal cache — not in the Podman store.
Subsequent `podman image inspect` / `podman run` fail with "image not found".

The code at `docker.rs:build_image` only checks the exit code, never verifies the image
actually exists in the local store after building.

## Approach

**Post-build verification + export retry** in `DockerSandbox::build_image()`.

1. After `podman build` succeeds, verify the image is in the store via `sandbox_image_exists()`
2. If missing (BuildKit scenario), re-run the build with `--output type=docker,dest=<tmpfile>`
   and pipe to `podman load -i <tmpfile>` to force the image into the Podman store
3. BuildKit's layer cache makes the re-build essentially free (same Dockerfile = cache hit)
4. Normal Podman+buildah flows are unaffected — the verify check passes immediately

Why this over adding `--load` to the initial build: Podman's `build` command doesn't support
`--load` (that's Docker buildx–only). The `--output type=docker,dest=<file>` + `podman load`
pattern is portable across all Podman 4.x+ versions.

   473→    async fn exec(&self, id: &SandboxId, command: &str, opts: &ExecOpts) -> Result<ExecResult> {
   474→        let name = self.container_name(id);
   475→
   476→        let mut args = vec!["exec".to_string()];
   477→
   478→        if let Some(ref dir) = opts.working_dir {
   479→            args.extend(["-w".to_string(), dir.display().to_string()]);
   480→        }
   481→
   482→        // Inject proxy env vars so traffic routes through the trusted-network
   483→        // proxy running on the host.
   484→        args.extend(self.proxy_exec_env_args());
   485→
   486→        for (k, v) in &opts.env {
   487→            args.extend(["-e".to_string(), format!("{}={}", k, v)]);
   488→        }
   489→
   490→        args.push(name);
   491→        args.extend(["sh".to_string(), "-c".to_string(), command.to_string()]);
   492→
   493→        let child = tokio::process::Command::new(self.cli)
   494→            .args(&args)
   495→            .stdout(std::process::Stdio::piped())
   496→            .stderr(std::process::Stdio::piped())
   497→            .stdin(std::process::Stdio::null())
   498→            .spawn()?;
   499→
   500→        let result = tokio::time::timeout(opts.timeout, child.wait_with_output()).await;
   501→
   502→        match result {
   503→            Ok(Ok(output)) => {
   504→                let mut stdout = String::from_utf8_lossy(&output.stdout).into_owned();
   505→                let mut stderr = String::from_utf8_lossy(&output.stderr).into_owned();
   506→
   507→                truncate_output_for_display(&mut stdout, opts.max_output_bytes);
   508→                truncate_output_for_display(&mut stderr, opts.max_output_bytes);
   509→
   510→                Ok(ExecResult {
   511→                    stdout,
   512→                    stderr,
   513→                    exit_code: output.status.code().unwrap_or(-1),
   514→                })
   515→            },
   516→            Ok(Err(e)) => {
   517→                return Err(Error::message(format!("{} exec failed: {e}", self.cli)));
   518→            },
   519→            Err(_) => {
   520→                return Err(Error::message(format!(
   521→                    "{} exec timed out after {}s",
   522→                    self.cli,
   523→                    opts.timeout.as_secs()
   524→                )));
   525→            },
   526→        }
   527→    }

   643→/// Return `true` when the installed Podman version supports `host-gateway`
   644→/// in `--add-host` (added in Podman 5.0).
   645→pub(crate) fn podman_supports_host_gateway() -> bool {
   646→    let Ok(output) = std::process::Command::new("podman")
   647→        .args(["version", "--format", "{{.Client.Version}}"])
   648→        .output()
   649→    else {
   650→        return false;
   651→    };
   652→    let version_str = String::from_utf8_lossy(&output.stdout);
   653→    let major: u32 = version_str
   654→        .trim()
   655→        .split('.')
   656→        .next()
   657→        .and_then(|s| s.parse().ok())
   658→        .unwrap_or(0);
   659→    major >= 5
   660→}

   662→/// Resolve the host IP that a Podman container (< 5.0) can use to reach the
   663→/// host.  Rootless Podman defaults to slirp4netns where the host is always
   664→/// `10.0.2.2`.  Rootful Podman uses a bridge network whose gateway we query
   665→/// with `podman network inspect`.
   666→pub(crate) fn podman_resolve_host_ip() -> Option<String> {
   667→    let rootless = std::process::Command::new("podman")
   668→        .args(["info", "--format", "{{.Host.Security.Rootless}}"])
   669→        .output()
   670→        .ok()
   671→        .map(|o| String::from_utf8_lossy(&o.stdout).trim().to_string());
   672→
   673→    if rootless.as_deref() == Some("true") {
   674→        // slirp4netns (default rootless network before Podman 5.0) maps the
   675→        // host to 10.0.2.2.
   676→        return Some("10.0.2.2".to_string());
   677→    }
   678→
   679→    // Rootful — ask for the gateway of the default "podman" network.
   680→    let output = std::process::Command::new("podman")
   681→        .args([
   682→            "network",
   683→            "inspect",
   684→            "podman",
   685→            "--format",
   686→            "{{(index .Subnets 0).Gateway}}",
   687→        ])
   688→        .output()
   689→        .ok()?;
   690→    let gateway = String::from_utf8_lossy(&output.stdout).trim().to_string();
   691→    if gateway.is_empty() {
   692→        None
   693→    } else {
   694→        Some(gateway)
   695→    }
   696→}

   124→    let dockerfile = sandbox_image_dockerfile(base, packages);

   180→/// List all local `<instance>-sandbox:*` images across available container CLIs.
   181→pub async fn list_sandbox_images() -> Result<Vec<SandboxImage>> {
   182→    let mut images = Vec::new();
   183→    let mut seen = HashSet::new();
   184→
   185→    // Docker/Podman: both support --format with Go templates.
   186→    for cli in OCI_COMPATIBLE_CLIS {
   187→        if !is_cli_available(cli) {
   188→            continue;
   189→        }
   190→        let output = tokio::process::Command::new(cli)
   191→            .args([
   192→                "image",
   193→                "ls",
   194→                "--format",
   195→                "{{.Repository}}:{{.Tag}}\t{{.Size}}\t{{.CreatedSince}}",
   196→            ])
   197→            .output()
   198→            .await?;
   199→        if output.status.success() {
   200→            let stdout = String::from_utf8_lossy(&output.stdout);
   201→            for line in stdout.lines() {
   202→                let parts: Vec<&str> = line.splitn(3, '\t').collect();
   203→                if parts.len() != 3 {
   204→                    continue;
   205→                }
   206→                // Podman prefixes local repos with "localhost/"; normalize.
   207→                let tag = parts[0].strip_prefix("localhost/").unwrap_or(parts[0]);
   208→                if is_sandbox_image_tag(tag) && seen.insert(tag.to_string()) {
   209→                    images.push(SandboxImage {
   210→                        tag: tag.to_string(),
   211→                        size: parts[1].to_string(),
   212→                        created: parts[2].to_string(),
   213→                    });
   214→                }
   215→            }
   216→        }
   217→    }
   218→
   219→    // Apple Container: fixed table output (NAME  TAG  DIGEST), no --format.
   220→    // Parse the table, then use `image inspect` JSON for metadata.
   221→    if is_cli_available("container") {
   222→        let output = tokio::process::Command::new("container")
   223→            .args(["image", "ls"])
   224→            .output()
   225→            .await?;
   226→        if output.status.success() {
   227→            let stdout = String::from_utf8_lossy(&output.stdout);
   228→            for line in stdout.lines().skip(1) {
   229→                // Columns are whitespace-separated: NAME TAG DIGEST
   230→                let cols: Vec<&str> = line.split_whitespace().collect();
   231→                if cols.len() >= 2 && cols[0].ends_with("-sandbox") {
   232→                    let tag = format!("{}:{}", cols[0], cols[1]);
   233→                    if !seen.insert(tag.clone()) {
   234→                        continue;
   235→                    }
   236→                    // Fetch size and created from inspect JSON.
   237→                    let (size, created) = inspect_apple_container_image(&tag).await;
   238→                    images.push(SandboxImage { tag, size, created });
   239→                }
   240→            }
   241→        }
   242→    }
   243→
   244→    Ok(images)
   245→}

   247→/// Extract size and created timestamp from Apple Container `image inspect` JSON.
   248→async fn inspect_apple_container_image(tag: &str) -> (String, String) {
   249→    let output = tokio::process::Command::new("container")
   250→        .args(["image", "inspect", tag])
   251→        .output()
   252→        .await;
   253→    let fallback = ("—".to_string(), "—".to_string());
   254→    let Ok(output) = output else {
   255→        return fallback;
   256→    };
   257→    if !output.status.success() {
   258→        return fallback;
   259→    }
   260→    let Ok(json): std::result::Result<serde_json::Value, _> =
   261→        serde_json::from_slice(&output.stdout)
   262→    else {
   263→        return fallback;
   264→    };
   265→    let entry = json.as_array().and_then(|a| a.first());
   266→    let Some(entry) = entry else {
   267→        return fallback;
   268→    };
   269→    let created = entry
   270→        .pointer("/index/annotations/org.opencontainers.image.created")
   271→        .and_then(|v| v.as_str())
   272→        .unwrap_or("—")
   273→        .to_string();
   274→    let size = entry
   275→        .pointer("/variants/0/size")
   276→        .and_then(|v| v.as_u64())
   277→        .map(format_bytes)
   278→        .unwrap_or_else(|| "—".to_string());
   279→    (size, created)
   280→}

   282→/// Format a byte count as a human-readable string (e.g. "361 MB").
   283→pub(crate) fn format_bytes(bytes: u64) -> String {
   284→    const KB: u64 = 1024;
   285→    const MB: u64 = 1024 * KB;
   286→    const GB: u64 = 1024 * MB;
   287→    if bytes >= GB {
   288→        format!("{:.1} GB", bytes as f64 / GB as f64)
   289→    } else if bytes >= MB {
   290→        format!("{:.0} MB", bytes as f64 / MB as f64)
   291→    } else if bytes >= KB {
   292→        format!("{:.0} KB", bytes as f64 / KB as f64)
   293→    } else {
   294→        format!("{bytes} B")
   295→    }
   296→}

   298→/// Remove a specific `<instance>-sandbox:*` image.
   299→pub async fn remove_sandbox_image(tag: &str) -> Result<()> {
   300→    if !is_sandbox_image_tag(tag) {
   301→        return Err(Error::message(format!(
   302→            "refusing to remove non-sandbox image: {tag}"
   303→        )));
   304→    }
   305→    // OCI-compatible CLIs (docker, podman) + Apple Container.
   306→    let all_clis: Vec<&str> = OCI_COMPATIBLE_CLIS
   307→        .iter()
   308→        .copied()
   309→        .chain(std::iter::once("container"))
   310→        .collect();
   311→    for cli in &all_clis {
   312→        if !is_cli_available(cli) {
   313→            continue;
   314→        }
   315→        if sandbox_image_exists(cli, tag).await {
   316→            // Apple Container uses `image delete`, Docker/Podman use `image rm`.
   317→            let subcmd = if *cli == "container" {
   318→                "delete"
   319→            } else {
   320→                "rm"
   321→            };
   322→            let output = tokio::process::Command::new(cli)
   323→                .args(["image", subcmd, tag])
   324→                .output()
   325→                .await?;
   326→            if !output.status.success() {
   327→                let stderr = String::from_utf8_lossy(&output.stderr);
   328→                return Err(Error::message(format!(
   329→                    "{cli} image {subcmd} failed for {tag}: {}",
   330→                    stderr.trim()
   331→                )));
   332→            }
   333→        }
   334→    }
   335→    Ok(())
   336→}

   338→/// Remove all local `<instance>-sandbox:*` images.
   339→pub async fn clean_sandbox_images() -> Result<usize> {
   340→    let images = list_sandbox_images().await?;

   342→    for img in &images {
   343→        remove_sandbox_image(&img.tag).await?;
   344→    }
   345→    Ok(count)

   399→}

   414→/// List all containers whose name starts with `container_prefix`.
   415→///
   416→/// Queries both Apple Container and Docker backends when available,
   417→/// merging results with the appropriate backend label.
   418→pub async fn list_running_containers(container_prefix: &str) -> Result<Vec<RunningContainer>> {
   419→    let mut containers = Vec::new();
   420→    let mut seen = HashSet::new();
   421→
   422→    // Apple Container: `container list --format json` outputs a JSON array.
   423→    // Each element has nested fields: configuration.id, status,
   424→    // configuration.image.reference, configuration.resources, networks[].
   425→    if is_cli_available("container") {
   426→        let output = tokio::process::Command::new("container")
   427→            .args(["list", "--format", "json"])
   428→            .output()
   429→            .await;
   430→        if let Ok(output) = output
   431→            && output.status.success()
   432→        {
   433→            let stdout = String::from_utf8_lossy(&output.stdout);
   434→            let entries: Vec<serde_json::Value> = serde_json::from_str(&stdout).unwrap_or_default();
   435→            for entry in entries {
   436→                let name = entry
   437→                    .pointer("/configuration/id")
   438→                    .and_then(|v| v.as_str())
   439→                    .unwrap_or_default();
   440→                if !name.starts_with(container_prefix) || !seen.insert(name.to_string()) {
   441→                    continue;
   442→                }
   443→                let state_str = entry
   444→                    .get("status")
   445→                    .and_then(|v| v.as_str())
   446→                    .unwrap_or_default();
   447→                let state = match state_str {
   448→                    "running" => ContainerRunState::Running,
   449→                    "stopped" => ContainerRunState::Stopped,
   450→                    "exited" => ContainerRunState::Exited,
   451→                    _ => ContainerRunState::Unknown,
   452→                };
   453→                let image = entry
   454→                    .pointer("/configuration/image/reference")
   455→                    .and_then(|v| v.as_str())
   456→                    .unwrap_or_default()
   457→                    .to_string();
   458→                let cpus = entry
   459→                    .pointer("/configuration/resources/cpus")
   460→                    .and_then(|v| v.as_u64())
   461→                    .map(|v| v as u32);
   462→                let memory_mb = entry
   463→                    .pointer("/configuration/resources/memoryInBytes")
   464→                    .and_then(|v| v.as_u64())
   465→                    .map(|v| v / (1024 * 1024));
   466→                // startedDate is a Core Foundation absolute time (seconds since 2001-01-01).
   467→                // Store as unix timestamp string; the frontend formats for display.
   468→                let started =
   469→                    entry
   470→                        .get("startedDate")
   471→                        .and_then(|v| v.as_f64())
   472→                        .map(|cf_timestamp| {
   473→                            // CF absolute time epoch: 2001-01-01T00:00:00Z = 978307200 unix seconds.
   474→                            let unix_secs = cf_timestamp as i64 + 978_307_200;
   475→                            unix_secs.to_string()
   476→                        });
   477→                let addr = entry
   478→                    .pointer("/networks/0/ipv4Address")
   479→                    .and_then(|v| v.as_str())
   480→                    .map(String::from);
   481→
   482→                containers.push(RunningContainer {
   483→                    name: name.to_string(),
   484→                    image,
   485→                    state,
   486→                    backend: ContainerBackend::AppleContainer,
   487→                    cpus,
   488→                    memory_mb,
   489→                    started,
   490→                    addr,
   491→                });
   492→            }
   493→        }
   494→    }
   495→
   496→    // Docker/Podman: `<cli> ps -a --filter name=<prefix> --format json` outputs one JSON per line.
   497→    for (cli, backend) in OCI_COMPATIBLE_CLIS
   498→        .iter()
   499→        .zip([ContainerBackend::Docker, ContainerBackend::Podman])
   500→    {
   501→        if !is_cli_available(cli) {
   502→            continue;
   503→        }
   504→        let output = tokio::process::Command::new(cli)
   505→            .args([
   506→                "ps",
   507→                "-a",
   508→                "--filter",
   509→                &format!("name={container_prefix}"),
   510→                "--format",
   511→                "{{json .}}",
   512→            ])
   513→            .output()
   514→            .await;
   515→        if let Ok(output) = output
   516→            && output.status.success()
   517→        {
   518→            let stdout = String::from_utf8_lossy(&output.stdout);
   519→            for line in stdout.lines() {
   520→                let line = line.trim();
   521→                if line.is_empty() {
   522→                    continue;
   523→                }
   524→                let Ok(json): std::result::Result<serde_json::Value, _> =
   525→                    serde_json::from_str(line)
   526→                else {
   527→                    continue;
   528→                };
   529→                let name = json
   530→                    .get("Names")
   531→                    .and_then(|v| v.as_str())
   532→                    .unwrap_or_default();
   533→                if !name.starts_with(container_prefix) || !seen.insert(name.to_string()) {
   534→                    continue;
   535→                }
   536→                let state_str = json
   537→                    .get("State")
   538→                    .and_then(|v| v.as_str())
   539→                    .unwrap_or_default();
   540→                let state = match state_str {
   541→                    "running" => ContainerRunState::Running,
   542→                    "exited" => ContainerRunState::Exited,
   543→                    _ => ContainerRunState::Stopped,
   544→                };
   545→                let image = json
   546→                    .get("Image")
   547→                    .and_then(|v| v.as_str())
   548→                    .unwrap_or_default()
   549→                    .to_string();
   550→                let started = json
   551→                    .get("CreatedAt")
   552→                    .and_then(|v| v.as_str())
   553→                    .map(String::from);
   554→
   555→                containers.push(RunningContainer {
   556→                    name: name.to_string(),
   557→                    image,
   558→                    state,
   559→                    backend,
   560→                    cpus: None,
   561→                    memory_mb: None,
   562→                    started,
   563→                    addr: None,
   564→                });
   565→            }
   566→        }
   567→    }
   568→
   569→    containers.retain(|c| !is_zombie(&c.name));
   570→    Ok(containers)
   571→}

   585→/// Query container runtime disk usage.
   586→///
   587→/// Uses `container system df --format json` for Apple Container,
   588→/// falls back to `docker system df --format json` for Docker.
   589→pub async fn container_disk_usage() -> Result<ContainerDiskUsage> {
   590→    // Try Apple Container first.
   591→    if is_cli_available("container") {
   592→        let output = tokio::process::Command::new("container")
   593→            .args(["system", "df", "--format", "json"])
   594→            .output()
   595→            .await;
   596→        if let Ok(output) = output
   597→            && output.status.success()
   598→        {
   599→            let stdout = String::from_utf8_lossy(&output.stdout);
   600→            if let Ok(json) = serde_json::from_str::<serde_json::Value>(&stdout) {
   601→                let c = json.get("containers").cloned().unwrap_or_default();
   602→                let i = json.get("images").cloned().unwrap_or_default();
   603→                return Ok(ContainerDiskUsage {
   604→                    containers_total: c.get("total").and_then(|v| v.as_u64()).unwrap_or(0),
