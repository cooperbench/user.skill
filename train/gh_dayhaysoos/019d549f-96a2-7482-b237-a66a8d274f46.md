      if (typeof chunk === 'string') {
        chunks.push(Buffer.from(chunk));
      } else {
        chunks.push(chunk);
      }
    });
    req.once('end', () => {
      resolvePromise(Buffer.concat(chunks));
    });
    req.once('error', (error) => {
      rejectPromise(error);
    });
  });
}
