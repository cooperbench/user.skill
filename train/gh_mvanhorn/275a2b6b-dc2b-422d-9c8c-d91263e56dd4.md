    97→	var resp HookResponse
    98→	if len(output.Stdout) > 0 {
    99→		if err := json.Unmarshal(output.Stdout, &resp); err != nil {
   100→			return nil, fmt.Errorf("failed to parse hook response: %w\nStdout: %s",
   101→				err, output.Stdout)
   102→		}
   103→	}

   105→	return &resp, nil

   257→// CreateTranscript builds and writes a transcript for the session.
   258→func (s *Session) CreateTranscript(prompt string, changes []FileChange) string {
   259→	s.TranscriptBuilder.AddUserMessage(prompt)
   260→	s.TranscriptBuilder.AddAssistantMessage("I'll help you with that.")
   261→
   262→	for _, change := range changes {
   263→		toolID := s.TranscriptBuilder.AddToolUse("mcp__acp__Write", change.Path, change.Content)
   264→		s.TranscriptBuilder.AddToolResult(toolID)
   265→	}
   266→
   267→	s.TranscriptBuilder.AddAssistantMessage("Done!")
   268→
   269→	if err := s.TranscriptBuilder.WriteToFile(s.TranscriptPath); err != nil {
   270→		s.env.T.Fatalf("failed to write transcript: %v", err)
   271→	}
   272→
   273→	return s.TranscriptPath
   274→}

   478→// WriteSessionState writes the session state for the given session ID.
   479→// This is useful for tests that need to manipulate session state directly.
   480→func (env *TestEnv) WriteSessionState(sessionID string, state *strategy.SessionState) error {
   481→	env.T.Helper()
   482→
   483→	stateDir := filepath.Join(env.RepoDir, ".git", "entire-sessions")
   484→	if err := os.MkdirAll(stateDir, 0o755); err != nil {
   485→		return fmt.Errorf("failed to create session state dir: %w", err)
   486→	}
   487→
   488→	stateFile := filepath.Join(stateDir, sessionID+".json")
   489→
   490→	data, err := json.MarshalIndent(state, "", "  ")
   491→	if err != nil {
   492→		return fmt.Errorf("failed to marshal session state: %w", err)
   493→	}
   494→
   495→	if err := os.WriteFile(stateFile, data, 0o644); err != nil {
   496→		return fmt.Errorf("failed to write session state: %w", err)
   497→	}
   498→	return nil
   499→}

## `e2e_test/` vs `integration_test/` — Key Differences

### 1. Real Agents vs Simulated Hooks

The fundamental difference is **what drives the session**.

**`e2e_test/`** launches a **real AI agent** (Claude Code, Gemini CLI, etc.) as a subprocess. The `AgentRunner` interface (`agent_runner.go`) actually invokes `claude -p "Create a file..."` and waits for it to finish. The agent creates real files, makes real API calls, and fires real hooks.

**`integration_test/`** **simulates** the agent's hook lifecycle by calling the CLI hook commands directly. The `HookRunner` (`hooks.go`) manually invokes `entire hooks claude-code user-prompt-submit`, `entire hooks claude-code stop`, etc. with crafted JSON input. Transcripts are fabricated with `TranscriptBuilder` — no actual LLM is called.

### 2. Build Tags & Cost

| | `e2e_test/` | `integration_test/` |
|---|---|---|
| Build tag | `//go:build e2e` | `//go:build integration` |
| Run command | `E2E_AGENT=claude-code go test -tags=e2e` | `mise run test:integration` |

### 3. TestEnv Design

Both suites have a `TestEnv` but they're designed differently:

- **Integration `TestEnv`** imports internal packages (`strategy`, `checkpoint`, `paths`, `agent`, `trailers`) and has rich hook simulation — `SimulateUserPromptSubmit`, `SimulateStop`, `SimulatePreTask`, `SimulatePostTask`, etc. for Claude, Gemini, and Factory Droid. - **E2E `TestEnv`** is a thinner wrapper. It has `RunAgent(prompt)` which delegates to the real agent binary and `RunEntireEnable(strategy)` which calls the actual `entire enable` CLI command. It doesn't import internal packages — it exercises the system purely through the CLI binary and agent subprocesses.

### 4. What They Test

**Integration tests** verify the **internal machinery** — hook dispatch, checkpoint creation, shadow branch management, condensation, rewind, session state transitions, etc. — with full control over inputs and deterministic behavior. They cover edge cases like mid-session commits, trailer removal, amend workflows, multi-session interleaving, and concurrent session conflicts.

**E2E tests** verify the **end-to-end contract** — "if I give Claude a prompt, does the right file get created, does the checkpoint get saved, does rewind work?" They validate that the real agent + real hooks + real CLI all wire together correctly. The test names (`scenario_checkpoint_workflows_test.go`, `scenario_rewind_test.go`, `scenario_basic_workflow_test.go`) reflect this scenario-level scope.

`★ Insight ─────────────────────────────────────`
- This is a classic **test pyramid** structure. Integration tests form the wide middle layer (fast, deterministic, thorough coverage of internal logic), while E2E tests form the narrow top (slow, expensive, but validate the full system wiring). The build tag separation (`integration` vs `e2e`) lets CI run the cheap tests on every push while keeping expensive agent calls opt-in.
- The `HookRunner` pattern in integration tests is a form of **test double at the process boundary** — rather than mocking Go interfaces, it feeds crafted JSON to the real CLI binary via stdin. This tests the actual binary's hook parsing without needing a live agent, catching serialization bugs that pure unit tests would miss.
