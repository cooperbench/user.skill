   356→// FormatCondensedTranscript formats an Input into a human-readable string for LLM.
   357→// The format is:
   358→//
   359→//	[User] user prompt here
   360→//
   361→//	[Assistant] assistant response here
   362→//
   363→//	[Tool] ToolName: description or file path
   364→func FormatCondensedTranscript(input Input) string {
   365→	var sb strings.Builder
   366→
   367→	for i, entry := range input.Transcript {
   368→		if i > 0 {
   369→			sb.WriteString("\n")
   370→		}
   371→
   372→		switch entry.Type {
   373→		case EntryTypeUser:
   374→			sb.WriteString("[User] ")
   375→			sb.WriteString(entry.Content)
   376→			sb.WriteString("\n")
   377→		case EntryTypeAssistant:
   378→			sb.WriteString("[Assistant] ")
   379→			sb.WriteString(entry.Content)
   380→			sb.WriteString("\n")
   381→		case EntryTypeTool:
   382→			sb.WriteString("[Tool] ")
   383→			sb.WriteString(entry.ToolName)
   384→			if entry.ToolDetail != "" {
   385→				sb.WriteString(": ")
   386→				sb.WriteString(entry.ToolDetail)
   387→			}
   388→			sb.WriteString("\n")
   389→		}
   390→	}
   391→
   392→	if len(input.FilesTouched) > 0 {
   393→		sb.WriteString("\n[Files Modified]\n")
   394→		for _, file := range input.FilesTouched {
   395→			fmt.Fprintf(&sb, "- %s\n", file)
   396→		}
   397→	}
   398→
   399→	return sb.String()
   400→}
