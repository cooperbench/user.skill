   558→// The function first uses 'git rev-parse --show-toplevel' to find the repository
   559→// root, which works correctly even when called from a subdirectory within the repo.
   560→func OpenRepository() (*git.Repository, error) {
   561→	// First, find the repository root using git rev-parse --show-toplevel
   562→	// This works correctly from any subdirectory within the repository
   563→	repoRoot, err := GetWorktreePath()

   565→		// Fallback to current directory if git command fails
   566→		// (e.g., if git is not installed or we're not in a repo)
   567→		repoRoot = "."
   568→	}
   569→
   570→	repo, err := git.PlainOpenWithOptions(repoRoot, &git.PlainOpenOptions{
   571→		EnableDotGitCommonDir: true,
   572→	})
   573→	if err != nil {
   574→		return nil, fmt.Errorf("failed to open repository: %w", err)
   575→	}
   576→	return repo, nil

   579→// IsInsideWorktree returns true if the current directory is inside a git worktree
   580→// (as opposed to the main repository). Worktrees have .git as a file pointing
   581→// to the main repo, while the main repo has .git as a directory.
   582→// This function works correctly from any subdirectory within the repository.
   583→func IsInsideWorktree() bool {
   584→	// First find the repository root
   585→	repoRoot, err := GetWorktreePath()
   586→	if err != nil {
   587→		return false
   588→	}
   589→
   590→	gitPath := filepath.Join(repoRoot, gitDir)
   591→	gitInfo, err := os.Stat(gitPath)
   592→	if err != nil {
   593→		return false
   594→	}
   595→	return !gitInfo.IsDir()

   598→// GetMainRepoRoot returns the root directory of the main repository.
   599→// In the main repo, this is the worktree path (repo root).
   600→// In a worktree, this parses the .git file to find the main repo.
   601→// This function works correctly from any subdirectory within the repository.
   602→//
   603→// Per gitrepository-layout(5), a worktree's .git file is a "gitfile" containing
   604→// "gitdir: <path>" pointing to $GIT_DIR/worktrees/<id> in the main repository.
   605→// See: https://git-scm.com/docs/gitrepository-layout
   606→func GetMainRepoRoot() (string, error) {
   607→	// First find the worktree/repo root
   608→	repoRoot, err := GetWorktreePath()
   609→	if err != nil {
   610→		return "", fmt.Errorf("failed to get worktree path: %w", err)
   611→	}
   612→
   613→	if !IsInsideWorktree() {
   614→		return repoRoot, nil
   615→	}
   616→
   617→	// Worktree .git file contains: "gitdir: /path/to/main/.git/worktrees/<id>"
   618→	gitFilePath := filepath.Join(repoRoot, gitDir)
   619→	content, err := os.ReadFile(gitFilePath) //nolint:gosec // G304: gitFilePath is constructed from repo root, not user input
   620→	if err != nil {
   621→		return "", fmt.Errorf("failed to read .git file: %w", err)
   622→	}
   623→
   624→	gitdir := strings.TrimSpace(string(content))
   625→	gitdir = strings.TrimPrefix(gitdir, "gitdir: ")
   626→
   627→	// Extract main repo root: everything before "/.git/"
   628→	idx := strings.LastIndex(gitdir, "/.git/")
   629→	if idx < 0 {
   630→		return "", fmt.Errorf("unexpected gitdir format: %s", gitdir)
   631→	}
   632→	return gitdir[:idx], nil
   633→}

   635→// GetGitCommonDir returns the path to the shared git directory.
   636→// In a regular checkout, this is .git/
   637→// In a worktree, this is the main repo's .git/ (not .git/worktrees/<name>/)
   638→// Uses git rev-parse --git-common-dir for reliable handling of worktrees.
   639→func GetGitCommonDir() (string, error) {
   640→	ctx := context.Background()
   641→	cmd := exec.CommandContext(ctx, "git", "rev-parse", "--git-common-dir")
   642→	cmd.Dir = "."
   643→	output, err := cmd.Output()
   644→	if err != nil {
   645→		return "", fmt.Errorf("failed to get git common dir: %w", err)
   646→	}

   648→	commonDir := strings.TrimSpace(string(output))
   649→
   650→	// git rev-parse --git-common-dir returns relative paths from the working directory,
   651→	// so we need to make it absolute if it isn't already
   652→	if !filepath.IsAbs(commonDir) {
   653→		commonDir = filepath.Join(".", commonDir)
   654→	}
   655→
   656→	return filepath.Clean(commonDir), nil
   657→}

   725→// checkCanRewind checks if working directory is clean enough for rewind.
   726→// Returns (canRewind, reason, error). Shared by shadow and linear-shadow strategies.
   727→func checkCanRewind() (bool, string, error) {
   728→	repo, err := OpenRepository()
   729→	if err != nil {
   730→		return false, "", fmt.Errorf("failed to open git repository: %w", err)
   731→	}
   732→
   733→	worktree, err := repo.Worktree()
   734→	if err != nil {
   735→		return false, "", fmt.Errorf("failed to get worktree: %w", err)
   736→	}
   737→
   738→	status, err := worktree.Status()
   739→	if err != nil {
   740→		return false, "", fmt.Errorf("failed to get status: %w", err)
   741→	}
   742→
   743→	if status.IsClean() {
   744→		return true, "", nil
   745→	}
   746→
   747→	var modified, added, deleted []string
   748→	for file, st := range status {
   749→		// Skip .entire directory
   750→		if paths.IsInfrastructurePath(file) {
   751→			continue
   752→		}
   753→
   754→		// Skip untracked files
   755→		if st.Worktree == git.Untracked {
   756→			continue
   757→		}
   758→
   759→		switch {
   760→		case st.Staging == git.Added || st.Worktree == git.Added:
   761→			added = append(added, file)
   762→		case st.Staging == git.Deleted || st.Worktree == git.Deleted:
   763→			deleted = append(deleted, file)
   764→		case st.Staging == git.Modified || st.Worktree == git.Modified:
   765→			modified = append(modified, file)
   766→		}
   767→	}
   768→
   769→	if len(modified) == 0 && len(added) == 0 && len(deleted) == 0 {
   770→		return true, "", nil
   771→	}
   772→
   773→	var msg strings.Builder
   774→	msg.WriteString("You have uncommitted changes:\n")
   775→	for _, f := range modified {
   776→		msg.WriteString(fmt.Sprintf("  modified:   %s\n", f))
   777→	}
   778→	for _, f := range added {
   779→		msg.WriteString(fmt.Sprintf("  added:      %s\n", f))
   780→	}
   781→	for _, f := range deleted {
   782→		msg.WriteString(fmt.Sprintf("  deleted:    %s\n", f))
   783→	}
   784→	msg.WriteString("\nPlease commit or stash your changes before rewinding.")
   785→
   786→	return false, msg.String(), nil
   787→}

   789→// checkCanRewindWithWarning checks working directory and returns a warning with diff stats.
   790→// Unlike checkCanRewind, this always returns canRewind=true but includes a warning message
   791→// with +/- line stats for uncommitted changes. Used by manual-commit strategy.
   792→func checkCanRewindWithWarning() (bool, string, error) {
   793→	repo, err := OpenRepository()
   794→	if err != nil {
   795→		// Can't open repo - still allow rewind but without stats
   796→		return true, "", nil //nolint:nilerr // Rewind allowed even if repo can't be opened
   797→	}
   798→
   799→	worktree, err := repo.Worktree()
   800→	if err != nil {
   801→		return true, "", nil //nolint:nilerr // Rewind allowed even if worktree can't be accessed
   802→	}
   803→
   804→	status, err := worktree.Status()
   805→	if err != nil {
   806→		return true, "", nil //nolint:nilerr // Rewind allowed even if status can't be retrieved
   807→	}
   808→
   809→	if status.IsClean() {
   810→		return true, "", nil
   811→	}
   812→
   813→	// Get HEAD commit tree for comparison - if we can't get it, just return without stats
   814→	head, err := repo.Head()
   815→	if err != nil {
   816→		return true, "", nil //nolint:nilerr // Rewind allowed even without HEAD (e.g., empty repo)
   817→	}
   818→
   819→	headCommit, err := repo.CommitObject(head.Hash())
   820→	if err != nil {
   821→		return true, "", nil //nolint:nilerr // Rewind allowed even if commit lookup fails
   822→	}
   823→
   824→	headTree, err := headCommit.Tree()
   825→	if err != nil {
   826→		return true, "", nil //nolint:nilerr // Rewind allowed even if tree lookup fails
   827→	}
   828→
   829→	type fileChange struct {
   830→		status   string // "modified", "added", "deleted"
   831→		added    int
   832→		removed  int
   833→		filename string
   834→	}
   835→
   836→	var changes []fileChange
   837→	// Use repo root, not cwd - git status returns paths relative to repo root
   838→	repoRoot, err := GetWorktreePath()
   839→	if err != nil {
   840→		return true, "", nil //nolint:nilerr // Rewind allowed even if repo root lookup fails
   841→	}
   842→
   843→	for file, st := range status {
   844→		// Skip .entire directory
   845→		if paths.IsInfrastructurePath(file) {
   846→			continue
   847→		}
   848→
   849→		// Skip untracked files
   850→		if st.Worktree == git.Untracked {
   851→			continue
   852→		}
   853→
   854→		var change fileChange
   855→		change.filename = file
   856→
   857→		switch {
   858→		case st.Staging == git.Added || st.Worktree == git.Added:
   859→			change.status = "added"
   860→			// New file - count all lines as added
   861→			absPath := filepath.Join(repoRoot, file)
   862→			if content, err := os.ReadFile(absPath); err == nil { //nolint:gosec // absPath is repo root + relative path from git status
   863→				change.added = countLines(content)
   864→			}
   865→		case st.Staging == git.Deleted || st.Worktree == git.Deleted:
   866→			change.status = "deleted"
   867→			// Deleted file - count lines from HEAD as removed
   868→			if entry, err := headTree.File(file); err == nil {
   869→				if content, err := entry.Contents(); err == nil {
   870→					change.removed = countLines([]byte(content))
   871→				}
   872→			}
   873→		case st.Staging == git.Modified || st.Worktree == git.Modified:
   874→			change.status = "modified"
   875→			// Modified file - compute diff stats
   876→			var headContent, workContent []byte
   877→			if entry, err := headTree.File(file); err == nil {
   878→				if content, err := entry.Contents(); err == nil {
   879→					headContent = []byte(content)
   880→				}
   881→			}
   882→			absPath := filepath.Join(repoRoot, file)
   883→			if content, err := os.ReadFile(absPath); err == nil { //nolint:gosec // absPath is repo root + relative path from git status
   884→				workContent = content
   885→			}
   886→			if headContent != nil && workContent != nil {
   887→				change.added, change.removed = computeDiffStats(headContent, workContent)
   888→			}
   889→		default:
   890→			continue
   891→		}
   892→
   893→		changes = append(changes, change)
   894→	}
   895→
   896→	if len(changes) == 0 {
   897→		return true, "", nil
   898→	}
   899→
   900→	// Sort changes by filename for consistent output
   901→	sort.Slice(changes, func(i, j int) bool {
   902→		return changes[i].filename < changes[j].filename
   903→	})
   904→
   905→	var msg strings.Builder
   906→	msg.WriteString("The following uncommitted changes will be reverted:\n")
   907→
   908→	totalAdded, totalRemoved := 0, 0
   909→	for _, c := range changes {
   910→		totalAdded += c.added
   911→		totalRemoved += c.removed
   912→
   913→		var stats string
   914→		switch {
   915→		case c.added > 0 && c.removed > 0:
   916→			stats = fmt.Sprintf("+%d/-%d", c.added, c.removed)
   917→		case c.added > 0:
   918→			stats = fmt.Sprintf("+%d", c.added)
   919→		case c.removed > 0:
   920→			stats = fmt.Sprintf("-%d", c.removed)
   921→		}
   922→
   923→		msg.WriteString(fmt.Sprintf("  %-10s %s", c.status+":", c.filename))
   924→		if stats != "" {
   925→			msg.WriteString(fmt.Sprintf(" (%s)", stats))
   926→		}
   927→		msg.WriteString("\n")
   928→	}
   929→
   930→	if totalAdded > 0 || totalRemoved > 0 {
   931→		msg.WriteString(fmt.Sprintf("\nTotal: +%d/-%d lines\n", totalAdded, totalRemoved))
   932→	}
   933→
   934→	return true, msg.String(), nil
   935→}

   937→// countLines counts the number of lines in content.
   938→func countLines(content []byte) int {
   939→	if len(content) == 0 {
   940→		return 0
   941→	}
   942→	count := 1
   943→	for _, b := range content {
   944→		if b == '\n' {
   945→			count++
   946→		}
   947→	}
   948→	// Don't count trailing newline as extra line
   949→	if len(content) > 0 && content[len(content)-1] == '\n' {
   950→		count--
   951→	}
   952→	return count
   953→}

   955→// computeDiffStats computes added and removed line counts between old and new content.
   956→// Uses a simple line-based diff algorithm.
   957→func computeDiffStats(oldContent, newContent []byte) (added, removed int) {
   958→	oldLines := splitLines(oldContent)
   959→	newLines := splitLines(newContent)
   960→
   961→	// Build a set of old lines with counts
   962→	oldSet := make(map[string]int)
   963→	for _, line := range oldLines {
   964→		oldSet[line]++
   965→	}
   966→
   967→	// Check which new lines are truly new
   968→	for _, line := range newLines {
   969→		if oldSet[line] > 0 {
   970→			oldSet[line]--
   971→		} else {
   972→			added++
   973→		}
   974→	}
   975→
   976→	// Remaining old lines are removed
   977→	for _, count := range oldSet {
   978→		removed += count
   979→	}
   980→
   981→	return added, removed
   982→}

   984→// splitLines splits content into lines, preserving empty lines.
   985→func splitLines(content []byte) []string {
   986→	if len(content) == 0 {
   987→		return nil
   988→	}
   989→	s := string(content)
   990→	// Remove trailing newline to avoid empty last element
   991→	s = strings.TrimSuffix(s, "\n")
   992→	return strings.Split(s, "\n")
   993→}

  1011→// StageFiles stages modified, new, and deleted files to the git worktree.
  1012→//
  1013→// This function handles three categories of file changes:
  1014→//  1. Modified files: existing files that have been changed
  1015→//  2. New files: files that were created during the session
  1016→//  3. Deleted files: files that were removed during the session
  1017→//
  1018→// Error Handling Strategy:
  1019→//   - Individual file staging errors are logged to stderr but don't fail the operation
  1020→//   - This ensures that partial staging succeeds even if some files have issues
  1021→//   - If a modified file no longer exists, it's treated as a deletion
  1022→//
  1023→// The stageCtx parameter is used for user-facing messages to indicate whether
  1024→// this is staging for a session checkpoint or a task checkpoint.
  1025→func StageFiles(worktree *git.Worktree, modified, newFiles, deleted []string, stageCtx StageFilesContext) {
  1026→	// Get repo root for resolving file paths
  1027→	// This is critical because fileExists() uses os.Stat() which resolves relative to CWD,
  1028→	// but worktree.Add/Remove resolve relative to repo root. If CWD != repo root,
  1029→	// fileExists() could return false for existing files, causing worktree.Remove()
  1030→	// to incorrectly delete them.
  1031→	repoRoot := worktree.Filesystem.Root()
  1032→
  1033→	// Stage modified files
  1034→	for _, file := range modified {
  1035→		// Resolve path relative to repo root for existence check
  1036→		absPath := filepath.Join(repoRoot, file)
  1037→		if fileExists(absPath) {
  1038→			if _, err := worktree.Add(file); err != nil {
  1039→				fmt.Fprintf(os.Stderr, "  Failed to stage %s: %v\n", file, err)
  1040→			} else {
  1041→				fmt.Fprintf(os.Stderr, "  Staged: %s\n", file)
  1042→			}
  1043→		} else {
  1044→			// File was deleted - stage the deletion
  1045→			if _, err := worktree.Remove(file); err != nil {
  1046→				fmt.Fprintf(os.Stderr, "  File not found or deleted: %s\n", file)
  1047→			}
  1048→		}
  1049→	}
  1050→
  1051→	// Stage new files
  1052→	if len(newFiles) > 0 {
  1053→		fmt.Fprintf(os.Stderr, "Staging %d new files created during %s:\n", len(newFiles), stageCtx)
  1054→		for _, file := range newFiles {
  1055→			if _, err := worktree.Add(file); err != nil {
  1056→				fmt.Fprintf(os.Stderr, "  Failed to stage %s: %v\n", file, err)
  1057→			} else {
  1058→				fmt.Fprintf(os.Stderr, "  Staged new file: %s\n", file)
  1059→			}
  1060→		}
  1061→	}
  1062→
  1063→	// Stage deleted files
  1064→	for _, file := range deleted {
  1065→		if _, err := worktree.Remove(file); err != nil {
  1066→			fmt.Fprintf(os.Stderr, "  Failed to stage deleted file %s: %v\n", file, err)
  1067→		} else {
  1068→			fmt.Fprintf(os.Stderr, "  Staged deleted file: %s\n", file)
  1069→		}
  1070→	}
  1071→}

  1073→// getTaskCheckpointFromTree retrieves a task checkpoint from a commit tree.

  1116→// getTaskTranscriptFromTree retrieves a task transcript from a commit tree.

  1119→	if !point.IsTaskCheckpoint {
