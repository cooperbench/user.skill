   202→		// For Gemini/OpenCode (JSON), CheckpointTranscriptStart is a message index.
   203→		var scopedTranscript []byte
   204→		switch state.AgentType {
   205→		case agent.AgentTypeGemini:
   206→			scopedTranscript = geminicli.SliceFromMessage(sessionData.Transcript, state.CheckpointTranscriptStart)
   207→		case agent.AgentTypeOpenCode:
   208→			scopedTranscript = opencode.SliceFromMessage(sessionData.Transcript, state.CheckpointTranscriptStart)
   209→		case agent.AgentTypeClaudeCode, agent.AgentTypeUnknown:
   210→			scopedTranscript = transcript.SliceFromLine(sessionData.Transcript, state.CheckpointTranscriptStart)
   211→		}
   212→		if len(scopedTranscript) > 0 {
   213→			var err error
   214→			summary, err = summarize.GenerateFromTranscript(summarizeCtx, scopedTranscript, sessionData.FilesTouched, state.AgentType, nil)
   215→			if err != nil {
   216→				logging.Warn(summarizeCtx, "summary generation failed",
   217→					slog.String("session_id", state.SessionID),
   218→					slog.String("error", err.Error()))
   219→				// Continue without summary - non-blocking
   220→			} else {
   221→				logging.Info(summarizeCtx, "summary generated",
   222→					slog.String("session_id", state.SessionID))
   223→			}
   224→		}
   225→	}
   226→
   227→	// Write checkpoint metadata using the checkpoint store
   228→	if err := store.WriteCommitted(context.Background(), cpkg.WriteCommittedOptions{
   229→		CheckpointID:                checkpointID,
   230→		SessionID:                   state.SessionID,
   231→		Strategy:                    StrategyNameManualCommit,
   232→		Branch:                      branchName,
   233→		Transcript:                  sessionData.Transcript,
   234→		Prompts:                     sessionData.Prompts,
   235→		Context:                     sessionData.Context,
   236→		FilesTouched:                sessionData.FilesTouched,
   237→		CheckpointsCount:            state.StepCount,
   238→		EphemeralBranch:             shadowBranchName,
   239→		AuthorName:                  authorName,
   240→		AuthorEmail:                 authorEmail,
   241→		Agent:                       state.AgentType,
   242→		TurnID:                      state.TurnID,
   243→		TranscriptIdentifierAtStart: state.TranscriptIdentifierAtStart,
   244→		CheckpointTranscriptStart:   state.CheckpointTranscriptStart,
   245→		TokenUsage:                  sessionData.TokenUsage,
   246→		InitialAttribution:          attribution,
   247→		Summary:                     summary,
   248→	}); err != nil {
   249→		return nil, fmt.Errorf("failed to write checkpoint metadata: %w", err)
   250→	}
   251→
   252→	return &CondenseResult{
   253→		CheckpointID:         checkpointID,
   254→		SessionID:            state.SessionID,
   255→		CheckpointsCount:     state.StepCount,
   256→		FilesTouched:         sessionData.FilesTouched,
   257→		TotalTranscriptLines: sessionData.FullTranscriptLines,
   258→	}, nil
   259→}

   261→func calculateSessionAttributions(repo *git.Repository, shadowRef *plumbing.Reference, sessionData *ExtractedSessionData, state *SessionState) *cpkg.InitialAttribution {
   262→	// Calculate initial attribution using accumulated prompt attribution data.
   263→	// This uses user edits captured at each prompt start (before agent works),
   264→	// plus any user edits after the final checkpoint (shadow → head).
   265→	//
   266→	// When shadowRef is nil (agent committed mid-turn before SaveStep),
   267→	// HEAD is used as the shadow tree. This is correct because the agent's
   268→	// commit IS HEAD — there are no user edits between agent work and commit.
   269→	logCtx := logging.WithComponent(context.Background(), "attribution")
   270→
   271→	headRef, headErr := repo.Head()
   272→	if headErr != nil {
   273→		logging.Debug(logCtx, "attribution skipped: failed to get HEAD",
   274→			slog.String("error", headErr.Error()))
   275→		return nil
   276→	}
   277→
   278→	headCommit, commitErr := repo.CommitObject(headRef.Hash())
   279→	if commitErr != nil {
   280→		logging.Debug(logCtx, "attribution skipped: failed to get HEAD commit",
   281→			slog.String("error", commitErr.Error()))
   282→		return nil
   283→	}
   284→
   285→	headTree, treeErr := headCommit.Tree()
   286→	if treeErr != nil {
   287→		logging.Debug(logCtx, "attribution skipped: failed to get HEAD tree",
   288→			slog.String("error", treeErr.Error()))
   289→		return nil
   290→	}
   291→
   292→	// Get shadow tree: from shadow branch if available, otherwise HEAD (agent committed directly).
   293→	var shadowTree *object.Tree
   294→	if shadowRef != nil {
   295→		shadowCommit, shadowErr := repo.CommitObject(shadowRef.Hash())
   296→		if shadowErr != nil {
   297→			logging.Debug(logCtx, "attribution skipped: failed to get shadow commit",
   298→				slog.String("error", shadowErr.Error()),
   299→				slog.String("shadow_ref", shadowRef.Hash().String()))
   300→			return nil
   301→		}
   302→		var shadowTreeErr error
   303→		shadowTree, shadowTreeErr = shadowCommit.Tree()
   304→		if shadowTreeErr != nil {
   305→			logging.Debug(logCtx, "attribution skipped: failed to get shadow tree",
   306→				slog.String("error", shadowTreeErr.Error()))
   307→			return nil
   308→		}
   309→	} else {
   310→		// No shadow branch: agent committed mid-turn. Use HEAD as shadow
   311→		// because the agent's work is the commit itself.
   312→		logging.Debug(logCtx, "attribution: using HEAD as shadow (no shadow branch)")
   313→		shadowTree = headTree
   314→	}
   315→
   316→	// Get base tree (state before session started)
   317→	var baseTree *object.Tree
   318→	attrBase := state.AttributionBaseCommit
   319→	if attrBase == "" {
   320→		attrBase = state.BaseCommit // backward compat
   321→	}
   322→	if baseCommit, baseErr := repo.CommitObject(plumbing.NewHash(attrBase)); baseErr == nil {
   323→		if tree, baseTErr := baseCommit.Tree(); baseTErr == nil {
   324→			baseTree = tree
   325→		} else {
   326→			logging.Debug(logCtx, "attribution: base tree unavailable",
   327→				slog.String("error", baseTErr.Error()))
   328→		}
   329→	} else {
   330→		logging.Debug(logCtx, "attribution: base commit unavailable",
   331→			slog.String("error", baseErr.Error()),
   332→			slog.String("attribution_base", attrBase))
   333→	}
   334→
   335→	// Log accumulated prompt attributions for debugging
   336→	var totalUserAdded, totalUserRemoved int
   337→	for i, pa := range state.PromptAttributions {
   338→		totalUserAdded += pa.UserLinesAdded
   339→		totalUserRemoved += pa.UserLinesRemoved
   340→		logging.Debug(logCtx, "prompt attribution data",
   341→			slog.Int("checkpoint", pa.CheckpointNumber),
   342→			slog.Int("user_added", pa.UserLinesAdded),
   343→			slog.Int("user_removed", pa.UserLinesRemoved),
   344→			slog.Int("agent_added", pa.AgentLinesAdded),
   345→			slog.Int("agent_removed", pa.AgentLinesRemoved),
   346→			slog.Int("index", i))
   347→	}
   348→
   349→	attribution := CalculateAttributionWithAccumulated(
   350→		baseTree,
   351→		shadowTree,
   352→		headTree,
   353→		sessionData.FilesTouched,
   354→		state.PromptAttributions,
   355→	)
   356→
   357→	if attribution != nil {
   358→		logging.Info(logCtx, "attribution calculated",
   359→			slog.Int("agent_lines", attribution.AgentLines),
   360→			slog.Int("human_added", attribution.HumanAdded),
   361→			slog.Int("human_modified", attribution.HumanModified),
   362→			slog.Int("human_removed", attribution.HumanRemoved),
   363→			slog.Int("total_committed", attribution.TotalCommitted),
   364→			slog.Float64("agent_percentage", attribution.AgentPercentage),
   365→			slog.Int("accumulated_user_added", totalUserAdded),
   366→			slog.Int("accumulated_user_removed", totalUserRemoved),
   367→			slog.Int("files_touched", len(sessionData.FilesTouched)))
   368→	}
   369→
   370→	return attribution
   371→}

   373→// extractSessionData extracts session data from the shadow branch.
   374→// filesTouched is the list of files tracked during the session (from SessionState.FilesTouched).
   375→// agentType identifies the agent (e.g., "Gemini CLI", "Claude Code") to determine transcript format.
   376→// liveTranscriptPath, when non-empty and readable, is preferred over the shadow branch copy.
   377→// This handles the case where SaveStep was skipped (no code changes) but the transcript
   378→// continued growing — the shadow branch copy would be stale.
   379→// checkpointTranscriptStart is the line offset (Claude) or message index (Gemini) where the current checkpoint began.
   380→func (s *ManualCommitStrategy) extractSessionData(repo *git.Repository, shadowRef plumbing.Hash, sessionID string, filesTouched []string, agentType agent.AgentType, liveTranscriptPath string, checkpointTranscriptStart int) (*ExtractedSessionData, error) {
   381→	commit, err := repo.CommitObject(shadowRef)
   382→	if err != nil {
   383→		return nil, fmt.Errorf("failed to get commit object: %w", err)
   384→	}
   385→
   386→	tree, err := commit.Tree()
   387→	if err != nil {
   388→		return nil, fmt.Errorf("failed to get commit tree: %w", err)
   389→	}
   390→
   391→	data := &ExtractedSessionData{}
   392→	// sessionID is already an "entire session ID" (with date prefix)
   393→	metadataDir := paths.SessionMetadataDirFromSessionID(sessionID)
   394→
   395→	// Extract transcript — prefer the live file when available, fall back to shadow branch.
   396→	// The shadow branch copy may be stale if the last turn ended without code changes
   397→	// (SaveStep is only called when there are file modifications).
   398→	var fullTranscript string
   399→	if liveTranscriptPath != "" {
   400→		if liveData, readErr := os.ReadFile(liveTranscriptPath); readErr == nil && len(liveData) > 0 { //nolint:gosec // path from session state
   401→			fullTranscript = string(liveData)
   402→		}
   403→	}
   404→	if fullTranscript == "" {
   405→		// Fall back to shadow branch copy
   406→		if file, fileErr := tree.File(metadataDir + "/" + paths.TranscriptFileName); fileErr == nil {
   407→			if content, contentErr := file.Contents(); contentErr == nil {
   408→				fullTranscript = content
   409→			}
   410→		} else if file, fileErr := tree.File(metadataDir + "/" + paths.TranscriptFileNameLegacy); fileErr == nil {
   411→			if content, contentErr := file.Contents(); contentErr == nil {
   412→				fullTranscript = content
   413→			}
   414→		}
   415→	}
   416→
   417→	// Process transcript based on agent type
   418→	if fullTranscript != "" {
   419→		data.Transcript = []byte(fullTranscript)
   420→		data.FullTranscriptLines = countTranscriptItems(agentType, fullTranscript)
   421→		data.Prompts = extractUserPrompts(agentType, fullTranscript)
   422→		data.Context = generateContextFromPrompts(data.Prompts)
   423→	}
   424→
   425→	// Use tracked files from session state (not all files in tree)
   426→	data.FilesTouched = filesTouched
   427→
   428→	// Calculate token usage from the extracted transcript portion
   429→	if len(data.Transcript) > 0 {
   430→		data.TokenUsage = calculateTokenUsage(agentType, data.Transcript, checkpointTranscriptStart)
   431→	}
   432→
   433→	return data, nil
   434→}

   436→// extractSessionDataFromLiveTranscript extracts session data directly from the live transcript file.
   437→// This is used for mid-session commits where no shadow branch exists yet.
   438→func (s *ManualCommitStrategy) extractSessionDataFromLiveTranscript(state *SessionState) (*ExtractedSessionData, error) {
   439→	data := &ExtractedSessionData{}
   440→
   441→	// Read the live transcript
   442→	if state.TranscriptPath == "" {
   443→		return nil, errors.New("no transcript path in session state")
   444→	}
   445→
   446→	liveData, err := os.ReadFile(state.TranscriptPath)
   447→	if err != nil {
   448→		return nil, fmt.Errorf("failed to read live transcript: %w", err)
   449→	}
   450→
   451→	if len(liveData) == 0 {
   452→		return nil, errors.New("live transcript is empty")
   453→	}
   454→
   455→	fullTranscript := string(liveData)
   456→	data.Transcript = liveData
   457→	data.FullTranscriptLines = countTranscriptItems(state.AgentType, fullTranscript)
   458→	data.Prompts = extractUserPrompts(state.AgentType, fullTranscript)
   459→	data.Context = generateContextFromPrompts(data.Prompts)
   460→
   461→	// Extract files from transcript since state.FilesTouched may be empty for mid-session commits
   462→	// (no SaveStep/Stop has been called yet to populate it)
   463→	if len(state.FilesTouched) > 0 {
   464→		data.FilesTouched = state.FilesTouched
   465→	} else {
   466→		// Use the shared helper which includes subagent transcripts
   467→		data.FilesTouched = s.extractModifiedFilesFromLiveTranscript(state, state.CheckpointTranscriptStart)
   468→	}
   469→
   470→	// Calculate token usage from the extracted transcript portion
   471→	if len(data.Transcript) > 0 {
   472→		data.TokenUsage = calculateTokenUsage(state.AgentType, data.Transcript, state.CheckpointTranscriptStart)
   473→	}
   474→
   475→	return data, nil
   476→}

   478→// countTranscriptItems counts lines (JSONL) or messages (JSON) in a transcript.
   479→// For Claude Code and JSONL-based agents, this counts lines.
   480→// For Gemini CLI, OpenCode, and JSON-based agents, this counts messages.
   481→// Returns 0 if the content is empty or malformed.
   482→func countTranscriptItems(agentType agent.AgentType, content string) int {
   483→	if content == "" {
   484→		return 0
   485→	}
   486→
   487→	// OpenCode uses export JSON format with {"info": {...}, "messages": [...]}
   488→	if agentType == agent.AgentTypeOpenCode {
   489→		session, err := opencode.ParseExportSession([]byte(content))
   490→		if err == nil && session != nil {
   491→			return len(session.Messages)
   492→		}
   493→		return 0
   494→	}
   495→
   496→	// Try Gemini format first if agentType is Gemini, or as fallback if Unknown
   497→	if agentType == agent.AgentTypeGemini || agentType == agent.AgentTypeUnknown {
   498→		transcript, err := geminicli.ParseTranscript([]byte(content))
   499→		if err == nil && transcript != nil && len(transcript.Messages) > 0 {
   500→			return len(transcript.Messages)
   501→		}
   502→		// If agentType is explicitly Gemini but parsing failed, return 0
   503→		if agentType == agent.AgentTypeGemini {
   504→			return 0
   505→		}
   506→		// Otherwise fall through to JSONL parsing for Unknown type
   507→	}
   508→
   509→	// Claude Code and other JSONL-based agents
   510→	allLines := strings.Split(content, "\n")
   511→	// Trim trailing empty lines (from final \n in JSONL)
   512→	for len(allLines) > 0 && strings.TrimSpace(allLines[len(allLines)-1]) == "" {
   513→		allLines = allLines[:len(allLines)-1]
   514→	}
   515→	return len(allLines)
   516→}

   518→// extractUserPrompts extracts all user prompts from transcript content.
   519→// Returns prompts with IDE context tags stripped (e.g., <ide_opened_file>).
   520→func extractUserPrompts(agentType agent.AgentType, content string) []string {
   521→	if content == "" {
   522→		return nil
   523→	}
   524→
   525→	// OpenCode uses JSONL with a different per-line schema than Claude Code
   526→	if agentType == agent.AgentTypeOpenCode {
   527→		prompts, err := opencode.ExtractAllUserPrompts([]byte(content))
   528→		if err == nil && len(prompts) > 0 {
   529→			cleaned := make([]string, 0, len(prompts))
   530→			for _, prompt := range prompts {
   531→				if stripped := textutil.StripIDEContextTags(prompt); stripped != "" {
   532→					cleaned = append(cleaned, stripped)
   533→				}
   534→			}
   535→			return cleaned
   536→		}
   537→		return nil
   538→	}
   539→
   540→	// Try Gemini format first if agentType is Gemini, or as fallback if Unknown
   541→	if agentType == agent.AgentTypeGemini || agentType == agent.AgentTypeUnknown {
   542→		prompts, err := geminicli.ExtractAllUserPrompts([]byte(content))
   543→		if err == nil && len(prompts) > 0 {
   544→			// Strip IDE context tags for consistency with Claude Code handling
   545→			cleaned := make([]string, 0, len(prompts))
   546→			for _, prompt := range prompts {
   547→				if stripped := textutil.StripIDEContextTags(prompt); stripped != "" {
   548→					cleaned = append(cleaned, stripped)
   549→				}
   550→			}
   551→			return cleaned
   552→		}
   553→		// If agentType is explicitly Gemini but parsing failed, return nil
   554→		if agentType == agent.AgentTypeGemini {
   555→			return nil
   556→		}
   557→		// Otherwise fall through to JSONL parsing for Unknown type
   558→	}
   559→
   560→	// Claude Code and other JSONL-based agents
   561→	return extractUserPromptsFromLines(strings.Split(content, "\n"))
   562→}

   564→// calculateTokenUsage calculates token usage from raw transcript data.
   565→// startOffset is the line number (Claude Code) or message index (Gemini CLI)
   566→// where the current checkpoint began, allowing calculation for only the portion
   567→// of the transcript since the last checkpoint.
   568→func calculateTokenUsage(agentType agent.AgentType, data []byte, startOffset int) *agent.TokenUsage {
   569→	if len(data) == 0 {
   570→		return &agent.TokenUsage{}
   571→	}
   572→
   573→	// OpenCode uses JSONL with token info on assistant messages (different schema from Claude Code)
   574→	if agentType == agent.AgentTypeOpenCode {
   575→		return opencode.CalculateTokenUsageFromBytes(data, startOffset)
   576→	}
   577→
   578→	// Try Gemini format first if agentType is Gemini, or as fallback if Unknown
   579→	if agentType == agent.AgentTypeGemini || agentType == agent.AgentTypeUnknown {
   580→		// Attempt to parse as Gemini JSON
   581→		transcript, err := geminicli.ParseTranscript(data)
   582→		if err == nil && transcript != nil && len(transcript.Messages) > 0 {
   583→			return geminicli.CalculateTokenUsage(data, startOffset)
   584→		}
   585→		// If agentType is explicitly Gemini but parsing failed, return empty usage
   586→		if agentType == agent.AgentTypeGemini {
   587→			return &agent.TokenUsage{}
   588→		}
   589→		// Otherwise fall through to JSONL parsing for Unknown type
   590→	}
   591→
   592→	// Claude Code and other JSONL-based agents
   593→	lines, err := transcript.ParseFromBytes(data)
   594→	if err != nil || len(lines) == 0 {
   595→		return &agent.TokenUsage{}
   596→	}
   597→	// Slice transcript lines to only include checkpoint portion
   598→	if startOffset > 0 && startOffset < len(lines) {
   599→		lines = lines[startOffset:]
   600→	}
   601→	return claudecode.CalculateTokenUsage(lines)
   602→}

   604→// extractUserPromptsFromLines extracts user prompts from JSONL transcript lines.
   605→// IDE-injected context tags (like <ide_opened_file>) are stripped from the results.
   606→func extractUserPromptsFromLines(lines []string) []string {
   607→	var prompts []string
   608→	for _, line := range lines {
   609→		line = strings.TrimSpace(line)
   610→		if line == "" {
   611→			continue
   612→		}
   613→
   614→		var entry map[string]interface{}
   615→		if err := json.Unmarshal([]byte(line), &entry); err != nil {
   616→			continue
   617→		}
   618→
   619→		// Check for user message (supports both "human" and "user" types)
   620→		msgType, ok := entry["type"].(string)
   621→		if !ok || (msgType != "human" && msgType != "user") {
   622→			continue
   623→		}
   624→
   625→		// Extract message content
   626→		message, ok := entry["message"].(map[string]interface{})
   627→		if !ok {
   628→			continue
   629→		}
   630→
   631→		// Handle string content
   632→		if content, ok := message["content"].(string); ok && content != "" {
   633→			cleaned := textutil.StripIDEContextTags(content)
   634→			if cleaned != "" {
   635→				prompts = append(prompts, cleaned)
   636→			}
   637→			continue
   638→		}
   639→
   640→		// Handle array content (e.g., multiple text blocks from VSCode)
   641→		if arr, ok := message["content"].([]interface{}); ok {
   642→			var texts []string
   643→			for _, item := range arr {
   644→				if m, ok := item.(map[string]interface{}); ok {
   645→					if m["type"] == "text" {
   646→						if text, ok := m["text"].(string); ok {
   647→							texts = append(texts, text)
   648→						}
   649→					}
   650→				}
   651→			}
   652→			if len(texts) > 0 {
   653→				cleaned := textutil.StripIDEContextTags(strings.Join(texts, "\n\n"))
   654→				if cleaned != "" {
   655→					prompts = append(prompts, cleaned)
   656→				}
   657→			}
   658→		}
   659→	}
   660→	return prompts
   661→}

   663→// generateContextFromPrompts generates context.md content from a list of prompts.
   664→func generateContextFromPrompts(prompts []string) []byte {
   665→	if len(prompts) == 0 {
   666→		return nil
   667→	}
   668→
   669→	var buf strings.Builder
   670→	buf.WriteString("# Session Context\n\n")
   671→	buf.WriteString("## User Prompts\n\n")
   672→
   673→	for i, prompt := range prompts {
   674→		// Truncate very long prompts for readability
   675→		displayPrompt := prompt
   676→		if len(displayPrompt) > 500 {
   677→			displayPrompt = displayPrompt[:500] + "..."
   678→		}
   679→		buf.WriteString(fmt.Sprintf("### Prompt %d\n\n", i+1))
   680→		buf.WriteString(displayPrompt)
   681→		buf.WriteString("\n\n")
   682→	}
   683→
   684→	return []byte(buf.String())
   685→}

   687→// CondenseSessionByID force-condenses a session by its ID and cleans up.
   688→// This is used by "entire doctor" to salvage stuck sessions.
   689→func (s *ManualCommitStrategy) CondenseSessionByID(sessionID string) error {
   690→	ctx := logging.WithComponent(context.Background(), "condense-by-id")
   691→
   692→	// Load session state
   693→	state, err := s.loadSessionState(sessionID)
   694→	if err != nil {
   695→		return fmt.Errorf("failed to load session state: %w", err)
   696→	}
   697→	if state == nil {
   698→		return fmt.Errorf("session not found: %s", sessionID)
   699→	}
   700→
   701→	// Open repository
   702→	repo, err := OpenRepository()
   703→	if err != nil {
   704→		return fmt.Errorf("failed to open repository: %w", err)
   705→	}
   706→
   707→	// Generate a checkpoint ID
   708→	checkpointID, err := id.Generate()
   709→	if err != nil {
   710→		return fmt.Errorf("failed to generate checkpoint ID: %w", err)
   711→	}
   712→
   713→	// Check if shadow branch exists (required for condensation)
   714→	shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   715→	refName := plumbing.NewBranchReferenceName(shadowBranchName)
   716→	_, refErr := repo.Reference(refName, true)
   717→	hasShadowBranch := refErr == nil
   718→
   719→	if !hasShadowBranch {
   720→		// No shadow branch means no checkpoint data to condense.
   721→		// Just clean up the state file.
   722→		logging.Info(ctx, "no shadow branch for session, clearing state only",
   723→			slog.String("session_id", sessionID),
   724→			slog.String("shadow_branch", shadowBranchName),
   725→		)
   726→		if err := s.clearSessionState(sessionID); err != nil {
   727→			return fmt.Errorf("failed to clear session state: %w", err)
   728→		}
   729→		return nil
   730→	}

   671→// CreateGeminiTranscript creates a Gemini JSON transcript file for the session.
   672→func (s *GeminiSession) CreateGeminiTranscript(prompt string, changes []FileChange) string {
   673→	// Build Gemini-format transcript (JSON, not JSONL)
   674→	messages := []map[string]interface{}{
   675→		{
   676→			"type":    "user",
   677→			"content": prompt,
   678→		},
   679→		{
   680→			"type":    "assistant",
   681→			"content": "I'll help you with that.",
   682→		},
   683→	}
   684→
   685→	for _, change := range changes {
   686→		messages = append(messages, map[string]interface{}{
   687→			"type": "tool_use",
   688→			"name": "write_file",
   689→			"input": map[string]string{
   690→				"path":    change.Path,
   691→				"content": change.Content,

   693→		})
   694→		messages = append(messages, map[string]interface{}{
   695→			"type":   "tool_result",
   696→			"output": "File written successfully",
   697→		})
   698→	}
   699→
   700→	messages = append(messages, map[string]interface{}{
   701→		"type":    "assistant",
   702→		"content": "Done!",
   703→	})
   704→
   705→	transcript := map[string]interface{}{
   706→		"sessionId": s.ID,
   707→		"messages":  messages,
   708→	}
   709→
   710→	// Ensure directory exists
   711→	if err := os.MkdirAll(filepath.Dir(s.TranscriptPath), 0o755); err != nil {
   712→		s.env.T.Fatalf("failed to create transcript dir: %v", err)
   713→	}
   714→
   715→	// Write transcript
   716→	data, err := json.MarshalIndent(transcript, "", "  ")
   717→	if err != nil {
   718→		s.env.T.Fatalf("failed to marshal transcript: %v", err)
   719→	}
   720→	if err := os.WriteFile(s.TranscriptPath, data, 0o644); err != nil {
   721→		s.env.T.Fatalf("failed to write transcript: %v", err)
   722→	}
   723→
   724→	return s.TranscriptPath
   725→}
