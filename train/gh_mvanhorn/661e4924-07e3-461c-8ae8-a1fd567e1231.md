   289→// addTaskMetadataToTree adds task checkpoint metadata to a git tree using incremental updates.
   290→// Only the metadata paths are modified; unchanged subtrees are reused by hash reference.
   291→// When IsIncremental is true, only adds the incremental checkpoint file.
   292→func (s *GitStore) addTaskMetadataToTree(baseTreeHash plumbing.Hash, opts WriteTemporaryTaskOptions) (plumbing.Hash, error) {
   293→	// Compute metadata paths
   294→	sessionMetadataDir := paths.EntireMetadataDir + "/" + opts.SessionID
   295→	taskMetadataDir := sessionMetadataDir + "/tasks/" + opts.ToolUseID
   296→
   297→	// Build change tree with metadata additions
   298→	changes := newChangeTree()
   299→
   300→	if opts.IsIncremental {
   301→		// Incremental checkpoint: only add the checkpoint file
   302→		// Use proper JSON marshaling to handle nil/empty IncrementalData correctly
   303→		var incData []byte
   304→		var err error
   305→		if opts.IncrementalData != nil {
   306→			incData, err = redact.JSONLBytes(opts.IncrementalData)
   307→			if err != nil {
   308→				return plumbing.ZeroHash, fmt.Errorf("failed to redact incremental checkpoint: %w", err)
   309→			}
   310→		}
   311→		incrementalCheckpoint := struct {
   312→			Type      string          `json:"type"`
   313→			ToolUseID string          `json:"tool_use_id"`
   314→			Timestamp time.Time       `json:"timestamp"`
   315→			Data      json.RawMessage `json:"data"`
   316→		}{
   317→			Type:      opts.IncrementalType,
   318→			ToolUseID: opts.ToolUseID,
   319→			Timestamp: time.Now().UTC(),
   320→			Data:      incData,
   321→		}
   322→		cpData, err := jsonutil.MarshalIndentWithNewline(incrementalCheckpoint, "", "  ")
   323→		if err != nil {
   324→			return plumbing.ZeroHash, fmt.Errorf("failed to marshal incremental checkpoint: %w", err)
   325→		}
   326→
   327→		cpBlobHash, err := CreateBlobFromContent(s.repo, cpData)
   328→		if err != nil {
   329→			return plumbing.ZeroHash, fmt.Errorf("failed to create incremental checkpoint blob: %w", err)
   330→		}
   331→		cpFilename := fmt.Sprintf("%03d-%s.json", opts.IncrementalSequence, opts.ToolUseID)
   332→		cpPath := taskMetadataDir + "/checkpoints/" + cpFilename
   333→		changes.addFile(cpPath, cpBlobHash, filemode.Regular)
   334→	} else {
   335→		// Final checkpoint: add transcripts and checkpoint.json
   336→
   337→		// Add session transcript (with chunking support for large transcripts)
   338→		if opts.TranscriptPath != "" {
   339→			if transcriptContent, readErr := os.ReadFile(opts.TranscriptPath); readErr == nil {
   340→				// Detect agent type from content for proper chunking
   341→				agentType := agent.DetectAgentTypeFromContent(transcriptContent)
   342→
   343→				// Chunk if necessary
   344→				chunks, chunkErr := agent.ChunkTranscript(transcriptContent, agentType)
   345→				if chunkErr != nil {
   346→					logging.Warn(context.Background(), "failed to chunk transcript, checkpoint will be saved without transcript",
   347→						slog.String("error", chunkErr.Error()),
   348→						slog.String("session_id", opts.SessionID),
   349→					)
   350→				} else {
   351→					for i, chunk := range chunks {
   352→						chunkPath := sessionMetadataDir + "/" + agent.ChunkFileName(paths.TranscriptFileName, i)
   353→						blobHash, blobErr := CreateBlobFromContent(s.repo, chunk)
   354→						if blobErr != nil {
   355→							logging.Warn(context.Background(), "failed to create blob for transcript chunk",
   356→								slog.String("error", blobErr.Error()),
   357→								slog.String("session_id", opts.SessionID),
   358→								slog.Int("chunk_index", i),
   359→							)
   360→							continue
   361→						}
   362→						changes.addFile(chunkPath, blobHash, filemode.Regular)
   363→					}
   364→				}
   365→			}
   366→		}
   367→
   368→		// Add subagent transcript if available
   369→		if opts.SubagentTranscriptPath != "" && opts.AgentID != "" {
   370→			if agentContent, readErr := os.ReadFile(opts.SubagentTranscriptPath); readErr == nil {
   371→				redacted, jsonlErr := redact.JSONLBytes(agentContent)
   372→				if jsonlErr != nil {
   373→					logging.Warn(context.Background(), "subagent transcript is not valid JSONL, falling back to plain redaction",
   374→						slog.String("path", opts.SubagentTranscriptPath),
   375→						slog.String("error", jsonlErr.Error()),
   376→					)
   377→					redacted = redact.Bytes(agentContent)
   378→				}
   379→				agentContent = redacted
   380→				if blobHash, blobErr := CreateBlobFromContent(s.repo, agentContent); blobErr == nil {
   381→					agentPath := taskMetadataDir + "/agent-" + opts.AgentID + ".jsonl"
   382→					changes.addFile(agentPath, blobHash, filemode.Regular)
   383→				}
   384→			}
   385→		}
   386→
   387→		// Add checkpoint.json
   388→		checkpointJSON := fmt.Sprintf(`{
   389→  "session_id": %q,
   390→  "tool_use_id": %q,
   391→  "checkpoint_uuid": %q,
   392→  "agent_id": %q
   393→}`, opts.SessionID, opts.ToolUseID, opts.CheckpointUUID, opts.AgentID)
   394→
   395→		blobHash, err := CreateBlobFromContent(s.repo, []byte(checkpointJSON))
   396→		if err != nil {
   397→			return plumbing.ZeroHash, fmt.Errorf("failed to create checkpoint blob: %w", err)
   398→		}
   399→		checkpointPath := taskMetadataDir + "/checkpoint.json"
   400→		changes.addFile(checkpointPath, blobHash, filemode.Regular)
   401→	}
   402→
   403→	// Apply changes incrementally to the base tree
   404→	return applyChangesToTree(s.repo, baseTreeHash, changes)
   405→}

   423→// listCheckpointsForBranch lists checkpoint commits for a specific shadow branch name.
   424→// This is an internal helper used by ListTemporaryCheckpoints, ListCheckpointsForBranch, and ListAllTemporaryCheckpoints.
   425→func (s *GitStore) listCheckpointsForBranch(ctx context.Context, shadowBranchName, sessionID string, limit int) ([]TemporaryCheckpointInfo, error) {
   426→	_ = ctx // Reserved for future use
   427→
   428→	refName := plumbing.NewBranchReferenceName(shadowBranchName)
   429→
   430→	ref, err := s.repo.Reference(refName, true)
   431→	if err != nil {
   432→		return nil, nil //nolint:nilerr // No shadow branch is expected case
   433→	}
   434→
   435→	iter, err := s.repo.Log(&git.LogOptions{From: ref.Hash()})
   436→	if err != nil {
   437→		return nil, fmt.Errorf("failed to get commit log: %w", err)
   438→	}

   441→	count := 0
   442→

   445→			return errStop
   446→		}

   450→		commitSessionID, hasTrailer := trailers.ParseSession(c.Message)
   451→		if !hasTrailer {
   452→			return nil // Skip commits without session trailer
   453→		}
   454→		if sessionID != "" && commitSessionID != sessionID {
   455→			return nil // Skip commits from other sessions
   456→		}
   457→
   458→		// Get first line of message
   459→		message := c.Message
   460→		if idx := strings.Index(message, "\n"); idx > 0 {
   461→			message = message[:idx]
   462→		}
   463→

   465→			CommitHash: c.Hash,
   466→			Message:    message,
   467→			SessionID:  commitSessionID,
   468→			Timestamp:  c.Author.When,
   469→		}
   470→
   471→		// Check for task checkpoint first
   472→		taskMetadataDir, foundTask := trailers.ParseTaskMetadata(c.Message)
   473→		if foundTask {
   474→			info.IsTaskCheckpoint = true
   475→			info.MetadataDir = taskMetadataDir
   476→			info.ToolUseID = extractToolUseIDFromPath(taskMetadataDir)

   478→			metadataDir, found := trailers.ParseMetadata(c.Message)
   479→			if found {
   480→				info.MetadataDir = metadataDir
   481→			}
   482→		}
   483→
   484→		results = append(results, info)
   485→
   486→		if len(results) >= limit {
   487→			return errStop
   488→		}
   489→		return nil
   490→	})
   491→
   492→	if err != nil && !errors.Is(err, errStop) {
   493→		return nil, fmt.Errorf("error iterating commits: %w", err)
   494→	}
   495→
   496→	return results, nil
   497→}

   544→// This is used for shadow branch checkpoints where the transcript is stored in the commit tree

   563→		// Use the helper function that handles chunking

   570→	// Fall back to direct file access (for backwards compatibility)
   571→	transcriptPath := metadataDir + "/" + paths.TranscriptFileName
   572→	if file, fileErr := tree.File(transcriptPath); fileErr == nil {

   574→		if contentErr == nil {

   612→// ShadowBranchNameForCommit returns the shadow branch name for a base commit hash
   613→// and worktree identifier. The worktree ID should be empty for the main worktree
   614→// or the internal git worktree name for linked worktrees.
   615→// Format: entire/<commit[:7]>-<hash(worktreeID)[:6]>
   616→func ShadowBranchNameForCommit(baseCommit, worktreeID string) string {
   617→	commitPart := baseCommit
   618→	if len(baseCommit) >= ShadowBranchHashLength {
   619→		commitPart = baseCommit[:ShadowBranchHashLength]
   620→	}
   621→	worktreeHash := HashWorktreeID(worktreeID)
   622→	return ShadowBranchPrefix + commitPart + "-" + worktreeHash
   623→}

   625→// ParseShadowBranchName extracts the commit prefix and worktree hash from a shadow branch name.
   626→// Input format: "entire/<commit[:7]>-<worktreeHash[:6]>"
   627→// Returns (commitPrefix, worktreeHash, ok). Returns ("", "", false) if not a valid shadow branch.
   628→func ParseShadowBranchName(branchName string) (commitPrefix, worktreeHash string, ok bool) {
   629→	if !strings.HasPrefix(branchName, ShadowBranchPrefix) {
   630→		return "", "", false
   631→	}
   632→	suffix := strings.TrimPrefix(branchName, ShadowBranchPrefix)
   633→
   634→	// Find the last dash - everything before is commit prefix, after is worktree hash
   635→	lastDash := strings.LastIndex(suffix, "-")
   636→	if lastDash == -1 || lastDash == 0 || lastDash == len(suffix)-1 {

   801→// createBlobFromFile creates a blob object from a file in the working directory.
   802→func createBlobFromFile(repo *git.Repository, filePath string) (plumbing.Hash, filemode.FileMode, error) {
   803→	info, err := os.Stat(filePath)
   804→	if err != nil {
   805→		return plumbing.ZeroHash, 0, fmt.Errorf("failed to stat file: %w", err)
   806→	}
   807→
   808→	// Determine file mode
   809→	mode := filemode.Regular
   810→	if info.Mode()&0o111 != 0 {
   811→		mode = filemode.Executable
   812→	}
   813→	if info.Mode()&os.ModeSymlink != 0 {
   814→		mode = filemode.Symlink
   815→	}
   816→
   817→	// Read file contents
   818→	content, err := os.ReadFile(filePath) //nolint:gosec // filePath comes from walking the repository
   819→	if err != nil {
   820→		return plumbing.ZeroHash, 0, fmt.Errorf("failed to read file: %w", err)
   821→	}
   822→
   823→	// Create blob object
   824→	obj := repo.Storer.NewEncodedObject()
   825→	obj.SetType(plumbing.BlobObject)
   826→	obj.SetSize(int64(len(content)))
   827→
   828→	writer, err := obj.Writer()
   829→	if err != nil {
   830→		return plumbing.ZeroHash, 0, fmt.Errorf("failed to get object writer: %w", err)
   831→	}
   832→
   833→	_, err = writer.Write(content)
   834→	if err != nil {
   835→		_ = writer.Close()
   836→		return plumbing.ZeroHash, 0, fmt.Errorf("failed to write blob content: %w", err)
   837→	}
   838→	if err := writer.Close(); err != nil {
   839→		return plumbing.ZeroHash, 0, fmt.Errorf("failed to close blob writer: %w", err)
   840→	}
   841→
   842→	hash, err := repo.Storer.SetEncodedObject(obj)
   843→	if err != nil {
   844→		return plumbing.ZeroHash, 0, fmt.Errorf("failed to store blob object: %w", err)
   845→	}
   846→
   847→	return hash, mode, nil
   848→}

   850→// addDirectoryToEntriesWithAbsPath recursively adds all files in a directory to the entries map.
   851→func addDirectoryToEntriesWithAbsPath(repo *git.Repository, dirPathAbs, dirPathRel string, entries map[string]object.TreeEntry) error {
   852→	err := filepath.Walk(dirPathAbs, func(path string, info os.FileInfo, err error) error {
   853→		if err != nil {
   854→			return err
   855→		}
   856→
   857→		// Skip symlinks to prevent reading files outside the metadata directory.
   858→		// A symlink could point to sensitive files (e.g., /etc/passwd) which would
   859→		// then be captured in the checkpoint and stored in git history.
   860→		// NOTE: filepath.Walk uses os.Stat (follows symlinks), so info.Mode() never
   861→		// reports ModeSymlink. We use os.Lstat to check the entry itself.
   862→		// This check MUST come before IsDir() because Walk follows symlinked
   863→		// directories and would recurse into them otherwise.
   864→		linfo, lstatErr := os.Lstat(path)

   866→			return fmt.Errorf("failed to lstat %s: %w", path, lstatErr)
   867→		}

   869→			if info.IsDir() {
   870→				return filepath.SkipDir
   871→			}
   872→			return nil
   873→		}
   874→
   875→		if info.IsDir() {
   876→			return nil
   877→		}
   878→

   880→		relWithinDir, err := filepath.Rel(dirPathAbs, path)
   881→		if err != nil {
   882→			return fmt.Errorf("failed to get relative path for %s: %w", path, err)
   883→		}
   884→

   886→		if strings.HasPrefix(relWithinDir, "..") {
   887→			return fmt.Errorf("path traversal detected: %s", relWithinDir)
   888→		}
   889→
   890→		treePath := filepath.ToSlash(filepath.Join(dirPathRel, relWithinDir))
   891→
   892→		blobHash, mode, err := createRedactedBlobFromFile(repo, path, treePath)
   893→		if err != nil {
   894→			return fmt.Errorf("failed to create blob for %s: %w", path, err)
   895→		}

   897→			Name: treePath,
   898→			Mode: mode,
   899→			Hash: blobHash,
   900→		}
   901→		return nil
   902→	})
   903→	if err != nil {
   904→		return fmt.Errorf("failed to walk directory %s: %w", dirPathAbs, err)
   905→	}

   957→func buildTreeObject(repo *git.Repository, node *treeNode) (plumbing.Hash, error) {
   958→	var treeEntries []object.TreeEntry
   959→
   960→	// Add files
   961→	treeEntries = append(treeEntries, node.files...)
   962→
   963→	// Recursively build subtrees
   964→	for name, subnode := range node.entries {
   965→		subHash, err := buildTreeObject(repo, subnode)
   966→		if err != nil {
   967→			return plumbing.ZeroHash, err
   968→		}
   969→		treeEntries = append(treeEntries, object.TreeEntry{
   970→			Name: name,
   971→			Mode: filemode.Dir,
   972→			Hash: subHash,
   973→		})
   974→	}
   975→
   976→	// Sort entries (git requires sorted entries)
   977→	sortTreeEntries(treeEntries)
   978→
   979→	// Create tree object
   980→	tree := &object.Tree{Entries: treeEntries}
   981→
   982→	obj := repo.Storer.NewEncodedObject()
   983→	if err := tree.Encode(obj); err != nil {
   984→		return plumbing.ZeroHash, fmt.Errorf("failed to encode tree: %w", err)
   985→	}

   988→	if err != nil {
   989→		return plumbing.ZeroHash, fmt.Errorf("failed to store tree: %w", err)
   990→	}
   991→
   992→	return hash, nil
   993→}

   995→// sortTreeEntries sorts tree entries in git's required order.
   996→// Git sorts tree entries by name, with directories having a trailing /
   997→func sortTreeEntries(entries []object.TreeEntry) {
   998→	sort.Slice(entries, func(i, j int) bool {
   999→		nameI := entries[i].Name
  1000→		nameJ := entries[j].Name
  1001→		if entries[i].Mode == filemode.Dir {
  1002→			nameI += "/"
  1003→		}
  1004→		if entries[j].Mode == filemode.Dir {
  1005→			nameJ += "/"
  1006→		}
  1007→		return nameI < nameJ
  1008→	})
  1009→}

  1011→// collectChangedFiles collects all changed files (modified tracked + untracked non-ignored)
  1012→// using git CLI. This is much faster than filesystem walk and respects all gitignore sources
  1013→// including global gitignore (core.excludesfile).
  1014→//
  1015→// Uses git CLI instead of go-git because go-git's worktree.Status() does not respect
  1016→// global gitignore, which can cause globally ignored files to appear as untracked.
  1017→// See: https://github.com/entireio/cli/pull/129
  1018→//
  1019→// changedFilesResult contains both changed and deleted files from git status.
  1020→type changedFilesResult struct {
  1021→	Changed []string // Files to include (modified, added, untracked, renamed, etc.)
  1022→	Deleted []string // Files that were deleted (need to be excluded from checkpoint tree)
  1023→}
  1024→
  1025→// collectChangedFiles returns all changed files from git status for the first checkpoint.
  1026→//
  1027→// For the first checkpoint, we need to capture:
  1028→// - Modified tracked files (user's uncommitted changes)
  1029→// - Untracked non-ignored files (new files not yet added to git)
  1030→// - Renamed/copied files (both source removal and destination)
  1031→// - Deleted files (to exclude from checkpoint tree)
  1032→//
  1033→// The base tree from HEAD already contains all unchanged tracked files.
  1034→//
  1035→// Uses `git status --porcelain -z` for reliable parsing of filenames with special characters.
  1036→func collectChangedFiles(ctx context.Context, repo *git.Repository) (changedFilesResult, error) {
  1037→	// Get repo root directory for running git command
  1038→	wt, err := repo.Worktree()
  1039→	if err != nil {
  1040→		return changedFilesResult{}, fmt.Errorf("failed to get worktree: %w", err)
  1041→	}
  1042→	repoRoot := wt.Filesystem.Root()
  1043→
  1044→	// Use -z for NUL-separated output (handles quoted filenames with spaces/special chars)
  1045→	// Use -uall to list individual untracked files instead of collapsed directories.
  1046→	// Note: CLAUDE.md warns against -uall for user-facing display, but we need the full list
  1047→	// for checkpointing.
  1048→	cmd := exec.CommandContext(ctx, "git", "status", "--porcelain", "-z", "-uall")
  1049→	cmd.Dir = repoRoot
  1050→	output, err := cmd.Output()
  1051→	if err != nil {
  1052→		return changedFilesResult{}, fmt.Errorf("failed to get git status in %s: %w", repoRoot, err)
  1053→	}
  1054→
  1055→	changedSeen := make(map[string]struct{})
  1056→	deletedSeen := make(map[string]struct{})
  1057→
  1058→	// Parse NUL-separated output
  1059→	// Format: XY filename\0 (for most entries)
  1060→	// For renames/copies: XY newname\0oldname\0
  1061→	entries := strings.Split(string(output), "\x00")
  1062→
  1063→	for i := 0; i < len(entries); i++ {
  1064→		entry := entries[i]
  1065→		if len(entry) < 3 {
  1066→			continue
  1067→		}
  1068→
  1069→		// git status --porcelain format: XY filename
  1070→		// X = staging status, Y = worktree status
  1071→		staging := entry[0]
  1072→		wtStatus := entry[1]
  1073→		filename := entry[3:] // No TrimSpace needed with -z format
  1074→
  1075→		// Handle R/C (rename/copy) first - they have a second entry we must skip
  1076→		// even if the new filename is an infrastructure path
  1077→		if staging == 'R' || staging == 'C' {
  1078→			// Renamed or copied: current entry is new name, next entry is old name
  1079→			if !paths.IsInfrastructurePath(filename) {
  1080→				changedSeen[filename] = struct{}{}
  1081→			}
  1082→			// The old name follows as the next NUL-separated entry - must always skip it
  1083→			if i+1 < len(entries) && entries[i+1] != "" {
  1084→				oldName := entries[i+1]
  1085→				if staging == 'R' && !paths.IsInfrastructurePath(oldName) {
  1086→					// For renames, old file is effectively deleted
  1087→					deletedSeen[oldName] = struct{}{}
  1088→				}
  1089→				i++ // Skip the old name entry
  1090→			}
  1091→			continue
  1092→		}
  1093→
  1094→		// Skip .entire directory for non-R/C entries
  1095→		if paths.IsInfrastructurePath(filename) {
  1096→			continue
  1097→		}
  1098→
  1099→		// Handle different status codes
  1100→		switch {
  1101→		case staging == 'D' || wtStatus == 'D':
  1102→			// Deleted file - track separately
  1103→			deletedSeen[filename] = struct{}{}
  1104→
  1105→		case wtStatus == 'M' || wtStatus == 'A':
  1106→			// Modified or added in worktree
  1107→			changedSeen[filename] = struct{}{}
  1108→
  1109→		case staging == '?' && wtStatus == '?':
  1110→			// Untracked file
  1111→			changedSeen[filename] = struct{}{}
  1112→
  1113→		case staging == 'A' || staging == 'M':
  1114→			// Staged add or modify
  1115→			changedSeen[filename] = struct{}{}
  1116→
  1117→		case staging == 'T' || wtStatus == 'T':
  1118→			// Type change (e.g., file to symlink)
  1119→			changedSeen[filename] = struct{}{}
  1120→
  1121→		case staging == 'U' || wtStatus == 'U':
  1122→			// Unmerged (conflict) - include current file state
  1123→			changedSeen[filename] = struct{}{}
  1124→		}
  1125→	}
  1126→
  1127→	changed := make([]string, 0, len(changedSeen))
  1128→	for file := range changedSeen {
  1129→		changed = append(changed, file)
  1130→	}
  1131→
  1132→	deleted := make([]string, 0, len(deletedSeen))
  1133→	for file := range deletedSeen {
  1134→		deleted = append(deleted, file)
  1135→	}
  1136→
  1137→	return changedFilesResult{Changed: changed, Deleted: deleted}, nil

  1062→// checkResetSafety checks for potential issues before a git reset --hard.
  1063→// Returns a list of warning messages (empty if safe to proceed without warnings).
  1064→// If uncommittedChangesWarning is provided, it will be used instead of a generic warning.
  1065→func checkResetSafety(targetCommitHash string, uncommittedChangesWarning string) ([]string, error) {
  1066→	var warnings []string
  1067→
  1068→	repo, err := openRepository()
  1069→	if err != nil {
  1070→		return nil, err
  1071→	}
  1072→
  1073→	// Check for uncommitted changes
  1074→	if uncommittedChangesWarning != "" {
  1075→		// Use the detailed warning from strategy's CanRewind()
  1076→		warnings = append(warnings, uncommittedChangesWarning)
  1077→	} else {
  1078→		// Fall back to generic check
  1079→		worktree, err := repo.Worktree()
  1080→		if err != nil {
  1081→			return nil, fmt.Errorf("failed to get worktree: %w", err)
  1082→		}
  1083→
  1084→		status, err := worktree.Status()
  1085→		if err != nil {
  1086→			return nil, fmt.Errorf("failed to get status: %w", err)
  1087→		}

## PR #407 Review: perf: optimize stop hook - transcript parsing and tree updates

### Summary

Three optimization themes plus several quality improvements:
1. **Incremental tree updates** - O(changed_files * tree_depth) vs O(total_files)
2. **`FromLines` variants** - avoid re-parsing transcript 3x
3. **Misc improvements** - `UserPrompt` field, pre-existing untracked fix, rewind fallback

---

### Blocking Issues

#### 1. Missing symlink protection in `addDirectoryToChangeTree` (security)

`tree_incremental.go:239-267` - The new `addDirectoryToChangeTree` replaces `addDirectoryToEntriesWithAbsPath` as the caller in `buildTreeWithChanges`. But the original function (`temporary.go:850-907`) has critical symlink protections that are **missing** from the new one:

The original `addDirectoryToEntriesWithAbsPath` uses `createRedactedBlobFromFile` which applies secrets redaction (JSONL-aware for transcript files, plain string for others). The new function uses `createBlobFromFile` which stores content **unredacted**.
