   459→	// Run prepare-commit-msg hook

   518→
   519→	// Run post-commit hook - since trailer was removed, no condensation should happen
   520→	//nolint:gosec,noctx // test code, args are from trusted test setup, no context needed

   714→	env.T.Helper()
