   683→// shellCompletionTarget returns the rc file path and completion lines for the
   684→// user's current shell.
   685→func shellCompletionTarget() (shellName, rcFile, completionLine string, err error) {
   686→	home, err := os.UserHomeDir()
   687→	if err != nil {
   688→		return "", "", "", fmt.Errorf("cannot determine home directory: %w", err)
   689→	}
   690→
   691→	shell := os.Getenv("SHELL")
   692→	switch {
   693→	case strings.Contains(shell, "zsh"):
   694→		return "Zsh",
   695→			filepath.Join(home, ".zshrc"),
   696→			"source <(entire completion zsh)",
   697→			nil
   698→	case strings.Contains(shell, "bash"):
   699→		bashRC := filepath.Join(home, ".bashrc")
   700→		if _, err := os.Stat(filepath.Join(home, ".bash_profile")); err == nil {
   701→			bashRC = filepath.Join(home, ".bash_profile")
   702→		}
   703→		return "Bash",
   704→			bashRC,
   705→			"source <(entire completion bash)",
   706→			nil
   707→	default:
   708→		return "", "", "", errUnsupportedShell
   709→	}
   710→}

   712→// promptShellCompletion offers to add shell completion to the user's rc file.
   713→// Only prompts if completion is not already configured.
   714→func promptShellCompletion(w io.Writer) error {
   715→	shellName, rcFile, completionLine, err := shellCompletionTarget()
   716→	if err != nil {
   717→		if errors.Is(err, errUnsupportedShell) {
   718→			fmt.Fprintf(w, "Note: Shell completion not available for your shell. Supported: zsh, bash.\n")
   719→			return nil
   720→		}
   721→		return fmt.Errorf("shell completion: %w", err)
   722→	}
   723→
   724→	if isCompletionConfigured(rcFile) {
   725→		fmt.Fprintf(w, "✓ Shell completion already configured in %s\n", rcFile)
   726→		return nil
   727→	}
   728→
   729→	var selected string
   730→	form := NewAccessibleForm(
   731→		huh.NewGroup(
   732→			huh.NewSelect[string]().
   733→				Title(fmt.Sprintf("Enable shell completion? (detected: %s)", shellName)).
   734→				Options(
   735→					huh.NewOption("Yes", "yes"),
   736→					huh.NewOption("No", "no"),
   737→				).
   738→				Value(&selected),
   739→		),
   740→	)
   741→
   742→	if err := form.Run(); err != nil {
   743→		//nolint:nilerr // User cancelled - not a fatal error, just skip

   745→	}
   746→
   747→	if selected != "yes" {
   748→		return nil
   749→	}
   750→
   751→	if err := appendShellCompletion(rcFile, completionLine); err != nil {
   752→		return fmt.Errorf("failed to update %s: %w", rcFile, err)
   753→	}
   754→
   755→	fmt.Fprintf(w, "✓ Shell completion added to %s\n", rcFile)
   756→	fmt.Fprintln(w, "  Restart your shell to activate")
   757→
   758→	return nil
   759→}

   830→// runUninstall completely removes Entire from the repository.
   831→func runUninstall(w, errW io.Writer, force bool) error {

   833→	if _, err := paths.RepoRoot(); err != nil {
   834→		fmt.Fprintln(errW, "Not a git repository. Nothing to uninstall.")
   835→		return NewSilentError(errors.New("not a git repository"))

   842→	claudeHooksInstalled := checkClaudeCodeHooksInstalled()
   843→	geminiHooksInstalled := checkGeminiCLIHooksInstalled()
   844→	entireDirExists := checkEntireDirExists()

   848→		shadowBranchCount == 0 && !claudeHooksInstalled && !geminiHooksInstalled {
   849→		fmt.Fprintln(w, "Entire is not installed in this repository.")

   861→		}
   862→		if sessionStateCount > 0 {
   863→			fmt.Fprintf(w, "  - Session state files (%d)\n", sessionStateCount)
   864→		}
   865→		if shadowBranchCount > 0 {
   866→			fmt.Fprintf(w, "  - Shadow branches (%d)\n", shadowBranchCount)
   867→		}
   868→		switch {
   869→		case claudeHooksInstalled && geminiHooksInstalled:
   870→			fmt.Fprintln(w, "  - Agent hooks (Claude Code, Gemini CLI)")
   871→		case claudeHooksInstalled:
   872→			fmt.Fprintln(w, "  - Agent hooks (Claude Code)")
   873→		case geminiHooksInstalled:
   874→			fmt.Fprintln(w, "  - Agent hooks (Gemini CLI)")
   875→		}
   876→		fmt.Fprintln(w)

   878→		var confirmed bool
   879→		form := NewAccessibleForm(
   880→			huh.NewGroup(
   881→				huh.NewConfirm().
   882→					Title("Are you sure you want to uninstall Entire?").

   885→					Value(&confirmed),

   890→			return fmt.Errorf("confirmation cancelled: %w", err)
   891→		}

   894→			fmt.Fprintln(w, "Uninstall cancelled.")
   895→			return nil
   896→		}
   897→	}

   900→

   904→	}

## Changes Made

### Bug 1: Git hook installation order with `--local-dev` (introduced by this PR)

**Root cause:** `strategy.InstallGitHook()` calls `isLocalDev()` which reads `local_dev` from settings. The PR moved hook installation to *before* settings were saved, causing hooks to be installed with wrong command prefix when `--local-dev` flag is used.

**Fix:** Moved `strategy.InstallGitHook()` calls to *after* settings are saved in all three functions:
- `runEnableWithStrategy()` 
- `runEnableInteractive()`
- `setupAgentHooksNonInteractive()`

### Bug 2: Nil-pointer risk in `setupAgentHooksNonInteractive` (pre-existing)

**Root cause:** The code ignored the error from `LoadEntireSettings()`:
```go
settings, _ := LoadEntireSettings() //nolint:errcheck // settings defaults are fine
```
If the settings file exists but contains invalid JSON, `Load()` returns `(nil, err)` - not defaults. The subsequent `settings.Enabled = true` would panic.

**Fix:** Changed to handle the error properly, matching the pattern used in the other two functions:
```go
settings, err := LoadEntireSettings()
if err != nil {
    // If we can't load, start with defaults
    settings = &EntireSettings{}
}
```
