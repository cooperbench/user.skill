   251→// findCheckpointInHistory walks commit history from start looking for a checkpoint trailer.
   252→// If stopAt is provided, stops when reaching that commit (exclusive).
   253→// Returns the first checkpoint found and info about commits between HEAD and the checkpoint.
   254→// It distinguishes between merge commits (bringing in other branches) and regular commits
   255→// (actual branch work) to avoid false warnings after merging main.
   256→func findCheckpointInHistory(start *object.Commit, stopAt *plumbing.Hash) *branchCheckpointResult {
   257→	result := &branchCheckpointResult{}
   258→	branchWorkCommits := 0 // Regular commits without checkpoints (actual work)
   259→	const maxCommits = 100 // Limit search depth
   260→	totalChecked := 0
   261→
   262→	current := start
   263→	for current != nil && totalChecked < maxCommits {
   264→		// Stop if we've reached the boundary
   265→		if stopAt != nil && current.Hash == *stopAt {
   266→			break
   267→		}
   268→
   269→		// Check for checkpoint trailer
   270→		if cpID, found := trailers.ParseCheckpoint(current.Message); found {
   271→			result.checkpointID = cpID
   272→			result.commitHash = current.Hash.String()
   273→			result.commitMessage = current.Message
   274→			// Only warn about branch work commits, not merge commits
   275→			result.newerCommitsExist = branchWorkCommits > 0
   276→			result.newerCommitCount = branchWorkCommits
   277→			return result
   278→		}
   279→
   280→		// Only count regular commits (not merge commits) as "branch work"
   281→		if current.NumParents() <= 1 {
   282→			branchWorkCommits++
   283→		}
   284→
   285→		totalChecked++
   286→
   287→		// Move to parent (first parent for merge commits - follows the main line)
   288→		if current.NumParents() == 0 {
   289→			break
   290→		}
   291→		parent, err := current.Parent(0)
   292→		if err != nil {
   293→			// Can't get parent, treat as end of history
   294→			break
   295→		}
   296→		current = parent
   297→	}
   298→
   299→	// No checkpoint found
   300→	return result
   301→}

   468→// RunResume executes the resume command and returns the combined output.
   469→// The subprocess is detached from the controlling terminal (via Setsid) to prevent
   470→// interactive prompts from hanging tests. This simulates non-interactive environments like CI.
   471→func (env *TestEnv) RunResume(branchName string) (string, error) {
   472→	env.T.Helper()
   473→
   474→	ctx := env.T.Context()
   475→	cmd := exec.CommandContext(ctx, getTestBinary(), "resume", branchName)
   476→	cmd.Dir = env.RepoDir
   477→	cmd.Env = append(os.Environ(),
   478→		"ENTIRE_TEST_CLAUDE_PROJECT_DIR="+env.ClaudeProjectDir,
   479→	)
   480→	// Detach from controlling terminal so huh can't open /dev/tty for interactive prompts
   481→	cmd.SysProcAttr = &syscall.SysProcAttr{Setsid: true}
   482→
   483→	output, err := cmd.CombinedOutput()
   484→	return string(output), err
   485→}

All resume tests were failing because they created sessions (user-prompt-submit + stop) without committing. With auto-commit gone, manual-commit requires the user to commit manually. Added `env.GitCommitWithShadowHooks(...)` after each session's stop to simulate the user committing, which:
1. Creates the `Entire-Checkpoint` trailer on the commit (so resume can find it)
2. Condenses session metadata to `entire/checkpoints/v1`
3. Commits the files (so branch switching doesn't fail with "uncommitted changes")

Also fixed `TestResume_MultipleSessionsOnBranch`'s assertion — with 2 sessions condensed into one commit, the output says "Restored 2 sessions" instead of "Session:".
