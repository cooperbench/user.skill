package movement_and_rotation_test

// --- Feature JSON loader ---

type gherkinFeature struct {
	Scenarios []gherkinScenario `json:"scenarios"`
}

type gherkinScenario struct {
	Examples []map[string]string `json:"examples"`
}

var loadedFeature *gherkinFeature

func featureExamples(scenarioIndex, exampleIndex int) map[string]string {
	if loadedFeature == nil {
		path := os.Getenv("GHERKIN_FEATURE_JSON")
		if path == "" {
			return nil
		}
		f, err := os.Open(path)
		if err != nil {
			return nil
		}
		defer f.Close()
		var feat gherkinFeature
		if err := json.NewDecoder(f).Decode(&feat); err != nil {
			return nil
		}
		loadedFeature = &feat
	}
	if scenarioIndex >= len(loadedFeature.Scenarios) {
		return nil
	}
	s := loadedFeature.Scenarios[scenarioIndex]
	if exampleIndex >= len(s.Examples) {
		return nil
	}
	return s.Examples[exampleIndex]
