// PlayerSoftDrops implements PlayerSoftDrops rule.
func (s *Session) PlayerSoftDrops() bool {
	if s.Phase != PhasePlaying || s.ActivePiece == nil {
		return false
	}
	if s.ActivePiece.AtRest(s.Board) {
		return false
	}
	s.ActivePiece.Origin.Row++
	s.Score += SoftDropScore
	return true
}

// PlayerHardDrops implements HardDropLocks rule.
func (s *Session) PlayerHardDrops() bool {
	if s.Phase != PhasePlaying || s.ActivePiece == nil {
		return false
	}
	ghost := s.ActivePiece.GhostOrigin(s.Board)
	rowsDropped := ghost.Row - s.ActivePiece.Origin.Row
	s.ActivePiece.Origin = ghost
	s.Score += rowsDropped * HardDropScore
	s.lockPiece()
	return true
}
