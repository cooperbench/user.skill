// Session holds all mutable game state. It is the single source of truth that
// maps to tetris.allium's GameSession entity.
type Session struct {
	Phase             Phase
	Board             *Board
	ActivePiece       *ActivePiece // nil when phase != playing|paused
	Bag               *Bag         // nil when phase != playing|paused
	NextKind          Kind         // KindNone on first spawn; set to bag.Peek() after first draw
	HeldKind          Kind         // KindNone = nothing held
	HoldUsed          bool
	Score             int
	Level             int
	LinesClearedTotal int
	LockTicksRemaining int
	rng               *rand.Rand
}

// NewSession creates a session in the waiting phase with a seeded RNG.
func NewSession(rng *rand.Rand) *Session {
	return &Session{
		Phase:    PhaseWaiting,
		HeldKind: KindNone,
		NextKind: KindNone,
		rng:      rng,
	}
}

// --- Commands (actions that mutate state) ---

// PlayerStartsGame implements the GameStarts rule.
// Returns false and leaves state unchanged if phase != waiting.
func (s *Session) PlayerStartsGame() bool {
	if s.Phase != PhaseWaiting {
		return false
	}
	s.Phase = PhasePlaying
	s.Score = 0
	s.Level = 1
	s.LinesClearedTotal = 0
	s.HoldUsed = false
	s.HeldKind = KindNone
	s.Board = NewBoard()
	s.Bag = NewBag(s.rng)
	s.NextKind = KindNone

	s.bagRefilled()
	s.pieceSpawned()
	return true
}

// Returns false if phase != game_over.
func (s *Session) PlayerRestartsGame() bool {
	if s.Phase != PhaseGameOver {
		return false
	}
	s.Phase = PhaseWaiting
	s.PlayerStartsGame()
	return true
}

	s.pieceSpawned()

	if s.Phase != PhasePlaying || s.ActivePiece == nil {
		return

	if s.ActivePiece.AtRest(s.Board) {
		if s.LockTicksRemaining > 1 {

		return false

		if s.Bag.Empty() {

	s.NextKind = s.Bag.Peek()

	s.ActivePiece = &ActivePiece{
		Kind:     kind,
		Origin:   Position{Col: SpawnCol, Row: SpawnRow},
		Rotation: RotationNorth,
	}
	s.HoldUsed = false
	s.LockTicksRemaining = LockTicks

	if !s.ActivePiece.ValidPlacement(s.Board) {
		s.gameOver()
	}
}

	s.Phase = PhaseGameOver

func (s *Session) lockPiece() {
	if s.ActivePiece == nil {
		return
	}
	for _, p := range s.ActivePiece.Footprint() {
		s.Board.SetCell(p.Col, p.Row, s.ActivePiece.Kind)
	}
	s.ActivePiece = nil
	s.linesCleared()
}

// linesCleared implements LinesCleared() internal event.
func (s *Session) linesCleared() {
	full := s.Board.FullRows()
	if len(full) > 0 {
		s.Board.ClearRows(full)
		s.scoreAwarded(len(full))
		s.LinesClearedTotal += len(full)
		s.levelProgressed()
	}
	s.pieceSpawned()
}

// scoreAwarded implements ScoreAwarded(count) rule.
func (s *Session) scoreAwarded(count int) {
	multiplier := map[int]int{1: 100, 2: 300, 3: 500, 4: 800}[count]
	s.Score += multiplier * s.Level
}

func (s *Session) levelProgressed() {
	earned := (s.LinesClearedTotal/LinesPerLevel) + 1
	if earned > s.Level {
		s.Level = earned
	}
}

package game

// SpawnBufferRows is the number of hidden rows above the visible board.
// Pieces spawn at row SpawnRow = -2, so the buffer must cover rows -2..-1.
const SpawnBufferRows = 2

// totalRows is the full internal storage height: buffer + visible.
const totalRows = SpawnBufferRows + BoardRows

// Board is the 10×20 visible play field plus a 2-row hidden spawn buffer.
// Internal storage uses rows 0..totalRows-1; public API uses row indices
// where 0 is the topmost visible row and negative indices address the buffer.
type Board struct {
	// cells[internalRow][col]; internalRow = row + SpawnBufferRows
	cells [totalRows][BoardCols]Kind
}

// NewBoard returns a blank board.
func NewBoard() *Board {
	return &Board{}
}

func (b *Board) toInternal(row int) int {
	return row + SpawnBufferRows
}

// Cell returns the Kind at (col, row). Returns KindNone for out-of-range rows.
func (b *Board) Cell(col, row int) Kind {
	ir := b.toInternal(row)
	if ir < 0 || ir >= totalRows {
		return KindNone
	}
	return b.cells[ir][col]
}

// SetCell places a kind at (col, row). No-op for out-of-range rows.
func (b *Board) SetCell(col, row int, k Kind) {
	ir := b.toInternal(row)
	if ir < 0 || ir >= totalRows {
		return
	}
	b.cells[ir][col] = k
}

// CellOccupied reports whether (col, row) contains a locked cell.
// Returns true for column out-of-bounds or row below the bottom.
// Returns false for rows above the buffer top (shouldn't occur in play).
func (b *Board) CellOccupied(col, row int) bool {
	if col < 0 || col >= BoardCols {
		return true
	}
	ir := b.toInternal(row)
	if ir < 0 {
		return false
	}
	if ir >= totalRows {
		return true
	}
	return b.cells[ir][col] != KindNone
}

// FullRows returns visible row indices (0-indexed) where every cell is filled.
func (b *Board) FullRows() []int {
	var full []int
	for r := 0; r < BoardRows; r++ {
		if b.isRowFull(r) {
			full = append(full, r)
		}
	}
	return full
}

func (b *Board) isRowFull(row int) bool {
	ir := b.toInternal(row)
	for c := 0; c < BoardCols; c++ {
		if b.cells[ir][c] == KindNone {
			return false
		}
	}
	return true
}

// ClearRows removes the given visible rows and drops everything above them.
// Rows must be provided in ascending order.
func (b *Board) ClearRows(rows []int) {
	if len(rows) == 0 {
		return
	}
	cleared := make(map[int]bool, len(rows))
	for _, r := range rows {
		cleared[r] = true
	}

	// Rebuild internal storage from bottom up, skipping cleared rows.
	var next [totalRows][BoardCols]Kind
	dst := totalRows - 1
	for src := totalRows - 1; src >= 0; src-- {
		visibleRow := src - SpawnBufferRows
		if cleared[visibleRow] {
			continue
		}
		next[dst] = b.cells[src]
		dst--
	}
	b.cells = next
}

// FilledCells returns all visible (col, row, kind) tuples where kind != KindNone.
func (b *Board) FilledCells() []struct {
	Col, Row int
	Kind     Kind
} {
	var out []struct {
		Col, Row int
		Kind     Kind
	}
	for r := 0; r < BoardRows; r++ {
		ir := b.toInternal(r)
		for c := 0; c < BoardCols; c++ {
			if b.cells[ir][c] != KindNone {
				out = append(out, struct {
					Col, Row int
					Kind     Kind
				}{c, r, b.cells[ir][c]})
			}
		}
	}
	return out
}
