// acceptance-runner is a persistent runner adapter for gherkin-mutator.
// It reads WorkerRequest JSON lines from stdin and writes WorkerResponse JSON lines to stdout.
// Each request provides a path to a (possibly mutated) feature JSON; the runner executes
// the acceptance tests with GHERKIN_FEATURE_JSON pointing to that file.
package main

import (
	"bufio"
	"bytes"
	"encoding/json"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"time"
)

type workerRequest struct {
	ID           string `json:"id"`
	FeatureJSON  string `json:"feature_json"`
	GeneratedDir string `json:"generated_dir"`
	WorkDir      string `json:"work_dir"`
	Timeout      string `json:"timeout,omitempty"`
}

type workerResponse struct {
	ID       string `json:"id"`
	Outcome  string `json:"outcome"`
	Output   string `json:"output"`
	Error    string `json:"error"`
	Duration int64  `json:"duration"`
}

func main() {
	projectRoot := findProjectRoot()

	scanner := bufio.NewScanner(os.Stdin)
	encoder := json.NewEncoder(os.Stdout)

	for scanner.Scan() {
		var req workerRequest
		if err := json.Unmarshal(scanner.Bytes(), &req); err != nil {
			resp := workerResponse{Outcome: "infrastructure_error", Error: "invalid request JSON: " + err.Error()}
			_ = encoder.Encode(resp)
			continue
		}

		start := time.Now()
		featureJSON := req.FeatureJSON
		if !filepath.IsAbs(featureJSON) {
			featureJSON = filepath.Join(projectRoot, featureJSON)
		}

		cmd := exec.Command("go", "test", "./acceptance/game_lifecycle/...")
		cmd.Dir = projectRoot
		cmd.Env = append(os.Environ(), "GHERKIN_FEATURE_JSON="+featureJSON)

		var out bytes.Buffer
		cmd.Stdout = &out
		cmd.Stderr = &out

		err := cmd.Run()
		duration := time.Since(start).Nanoseconds()

		var outcome string
		if err != nil {
			if _, ok := err.(*exec.ExitError); ok {
				outcome = "test_failure"
			} else {
				outcome = "infrastructure_error"
			}
		} else {
			outcome = "test_success"
		}

		resp := workerResponse{
			ID:       req.ID,
			Outcome:  outcome,
			Output:   out.String(),
			Duration: duration,
		}
		if err != nil {
			if _, ok := err.(*exec.ExitError); !ok {
				resp.Error = err.Error()
			}
		}
		if encErr := encoder.Encode(resp); encErr != nil {
			fmt.Fprintf(os.Stderr, "encode error: %v\n", encErr)
			return
		}
	}
	if err := scanner.Err(); err != nil {
		fmt.Fprintf(os.Stderr, "stdin error: %v\n", err)
	}
}

func findProjectRoot() string {
	dir, err := os.Getwd()
	if err != nil {
		return "."
	}
	for {
		if _, err := os.Stat(filepath.Join(dir, "go.mod")); err == nil {
			return dir
		}
		parent := filepath.Dir(dir)
		if parent == dir {
			return "."
		}
		dir = parent
	}
}

// RawTerminal saves and restores terminal state for raw input.
type RawTerminal struct {
	orig syscall.Termios
	fd   int
}

// EnableRaw puts the terminal into raw mode.
func EnableRaw() (*RawTerminal, error) {
	fd := int(os.Stdin.Fd())
	var orig syscall.Termios
	if _, _, errno := syscall.Syscall(syscall.SYS_IOCTL, uintptr(fd), syscall.TIOCGETA, uintptr(unsafe.Pointer(&orig))); errno != 0 {
		return nil, errno
	}
	raw := orig
	raw.Iflag &^= syscall.ICRNL | syscall.IXON
	raw.Lflag &^= syscall.ECHO | syscall.ICANON | syscall.IEXTEN | syscall.ISIG
	raw.Cc[syscall.VMIN] = 1
	raw.Cc[syscall.VTIME] = 0
	if _, _, errno := syscall.Syscall(syscall.SYS_IOCTL, uintptr(fd), syscall.TIOCSETA, uintptr(unsafe.Pointer(&raw))); errno != 0 {
		return nil, errno
	}
	return &RawTerminal{orig: orig, fd: fd}, nil
}

// Restore puts the terminal back to its original state.
func (rt *RawTerminal) Restore() {
	syscall.Syscall(syscall.SYS_IOCTL, uintptr(rt.fd), syscall.TIOCSETA, uintptr(unsafe.Pointer(&rt.orig)))
}

// TermSize returns the current terminal dimensions.
func TermSize() (width, height int) {
	type winsize struct {
		Row, Col, Xpixel, Ypixel uint16
	}
	fd := int(os.Stdout.Fd())
	var ws winsize
	syscall.Syscall(syscall.SYS_IOCTL, uintptr(fd), syscall.TIOCGWINSZ, uintptr(unsafe.Pointer(&ws)))
	return int(ws.Col), int(ws.Row)
}

// ReadKey reads one key event from stdin (blocking).
func ReadKey() Key {
	buf := make([]byte, 4)
	n, err := os.Stdin.Read(buf)
	if err != nil || n == 0 {
		return KeyUnknown
	}
	switch {
	case n == 1 && buf[0] == 3:
		return KeyCtrlC
	case n == 1 && buf[0] == 27:
		return KeyEsc
	case n == 1 && (buf[0] == 13 || buf[0] == 10):
		return KeyEnter
	case n == 1 && buf[0] == 32:
		return KeySpace
	case n == 1 && (buf[0] == 'r' || buf[0] == 'R'):
		return KeyR
	case n == 1 && (buf[0] == 'x' || buf[0] == 'X'):
		return KeyX
	case n == 1 && (buf[0] == 'z' || buf[0] == 'Z'):
		return KeyZ
	case n == 1 && (buf[0] == 'c' || buf[0] == 'C'):
		return KeyC
	case n == 1 && (buf[0] == 'p' || buf[0] == 'P'):
		return KeyP
	case n >= 3 && buf[0] == 27 && buf[1] == '[':
		switch buf[2] {
		case 'A':
			return KeyUp
		case 'B':
			return KeyDown
		case 'C':
			return KeyRight
		case 'D':
			return KeyLeft
		}
	}
	return KeyUnknown
}

// NewBag creates an empty bag with the given random source.
func NewBag(rng *rand.Rand) *Bag {
	return &Bag{rng: rng}
}

// Empty reports whether the bag has no remaining kinds.
func (bg *Bag) Empty() bool {
	return len(bg.remaining) == 0
}

// Refill fills the bag with all 7 kinds in a random order.
func (bg *Bag) Refill() {
	bg.remaining = make([]Kind, 7)
	copy(bg.remaining, AllKinds[:])
	bg.rng.Shuffle(len(bg.remaining), func(i, j int) {
		bg.remaining[i], bg.remaining[j] = bg.remaining[j], bg.remaining[i]
	})
}

	return k
}

	return bg.remaining[0]
