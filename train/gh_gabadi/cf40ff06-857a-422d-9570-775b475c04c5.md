493	func compareSnapshot(t *testing.T, name string, rows []string, expectedLines []string) {

497				continue
498			}
499			if i >= len(rows) {

503			got := rows[i]
504			if !strings.Contains(got, want) {
505				t.Errorf("%s snapshot row %d:\n  want (contains): %q\n  got:             %q", name, i, want, got)
506			}
507		}
508	}
