import "math/rand"

// Session holds all mutable game state. It is the single source of truth that
// maps to tetris.allium's GameSession entity.
type Session struct {
	Phase              Phase
	Board              *Board
	ActivePiece        *ActivePiece // nil when phase != playing|paused
	Bag                *Bag         // nil when phase != playing|paused
	NextKind           Kind         // KindNone on first spawn; set to bag.Peek() after first draw
	HeldKind           Kind         // KindNone = nothing held
	HoldUsed           bool
	Score              int
	Level              int
	LinesClearedTotal  int
	LockTicksRemaining int
	FlashRows          []int // non-nil during the one flash tick before row removal
	rng                *rand.Rand
}

// NewSession creates a session in the waiting phase with a seeded RNG.
func NewSession(rng *rand.Rand) *Session {
	return &Session{
		Phase:    PhaseWaiting,
		HeldKind: KindNone,
		NextKind: KindNone,
		rng:      rng,
	}
}

// --- Commands (actions that mutate state) ---

// PlayerStartsGame implements the GameStarts rule.
// Returns false and leaves state unchanged if phase != waiting.
func (s *Session) PlayerStartsGame() bool {
	if s.Phase != PhaseWaiting {
		return false
	}
	s.Phase = PhasePlaying
	s.Score = 0
	s.Level = 1
	s.LinesClearedTotal = 0
	s.HoldUsed = false
	s.HeldKind = KindNone
	s.Board = NewBoard()
	s.Bag = NewBag(s.rng)
	s.NextKind = KindNone

	s.bagRefilled()
	s.pieceSpawned()
	return true
}

// PlayerRestartsGame implements the GameRestarted rule.
// Returns false if phase != game_over.
func (s *Session) PlayerRestartsGame() bool {
	if s.Phase != PhaseGameOver {
		return false
	}
	s.Phase = PhaseWaiting
	s.PlayerStartsGame()
	return true
}

// SpawnPiece triggers PieceSpawned() — called by the game loop after a piece locks.
func (s *Session) SpawnPiece() {
	s.pieceSpawned()
}

// Tick advances the game by one frame.
func (s *Session) Tick() {
	if s.Phase != PhasePlaying {
		return
	}
	// Commit tick: FlashRows pending means rows are ready to be cleared and a new
	// piece must spawn. This guard must precede the ActivePiece == nil return
	// because ActivePiece is nil during the flash tick.
	if len(s.FlashRows) > 0 {
		s.Board.ClearRows(s.FlashRows)
		s.FlashRows = nil
		s.pieceSpawned()
		return
	}
	if s.ActivePiece == nil {
		return
	}
	if s.ActivePiece.AtRest(s.Board) {
		if s.LockTicksRemaining > 1 {
			s.LockTicksRemaining--
		} else {
			s.lockPiece()
		}
	} else {
		s.ActivePiece.Origin.Row++
	}
}

// PlayerMovesLeft implements PlayerMovesLeft rule.
func (s *Session) PlayerMovesLeft() bool {
	if s.Phase != PhasePlaying || s.ActivePiece == nil {
		return false
	}
	if !s.ActivePiece.CanMoveLeft(s.Board) {
		return false
	}
	s.ActivePiece.Origin.Col--
	if s.ActivePiece.AtRest(s.Board) {
		s.LockTicksRemaining = LockTicks
	}
	return true
}

// PlayerMovesRight implements PlayerMovesRight rule.
func (s *Session) PlayerMovesRight() bool {
	if s.Phase != PhasePlaying || s.ActivePiece == nil {
		return false
	}
	if !s.ActivePiece.CanMoveRight(s.Board) {
		return false
	}
	s.ActivePiece.Origin.Col++
	if s.ActivePiece.AtRest(s.Board) {
		s.LockTicksRemaining = LockTicks
	}
	return true
}

// PlayerSoftDrops implements PlayerSoftDrops rule.

		return false
	}
	if s.ActivePiece.AtRest(s.Board) {
		return false
	}
	s.ActivePiece.Origin.Row++
	s.Score += SoftDropScore
	return true

// PlayerHardDrops implements HardDropLocks rule.
func (s *Session) PlayerHardDrops() bool {
	if s.Phase != PhasePlaying || s.ActivePiece == nil {
		return false
	}
	ghost := s.ActivePiece.GhostOrigin(s.Board)
	rowsDropped := ghost.Row - s.ActivePiece.Origin.Row
	s.ActivePiece.Origin = ghost
	s.Score += rowsDropped * HardDropScore
	s.lockPiece()
	return true
}

// PlayerHolds implements PlayerHoldsNoHeld and PlayerHoldsSwap rules.
func (s *Session) PlayerHolds() bool {
	if s.Phase != PhasePlaying || s.HoldUsed || s.ActivePiece == nil {
		return false
	}
	currentKind := s.ActivePiece.Kind
	s.ActivePiece = nil

	if s.HeldKind == KindNone {
		s.HeldKind = currentKind
		s.pieceSpawned()
		s.HoldUsed = true // override reset from pieceSpawned
	} else {
		returning := s.HeldKind
		s.HeldKind = currentKind
		s.ActivePiece = &ActivePiece{
			Kind:     returning,
			Origin:   Position{Col: SpawnCol, Row: SpawnRow},
			Rotation: RotationNorth,
		}
		s.HoldUsed = true
	}
	return true
}

// Snapshot returns a pure-value RenderState for the renderer.
func (s *Session) Snapshot() RenderState {
	rs := RenderState{
		Phase:             s.Phase,
		NextKind:          s.NextKind,
		HeldKind:          s.HeldKind,
		HoldUsed:          s.HoldUsed,
		Score:             s.Score,
		Level:             s.Level,
		LinesClearedTotal: s.LinesClearedTotal,
	}
	if len(s.FlashRows) > 0 {
		rs.FlashRows = append([]int(nil), s.FlashRows...)
	}
	if s.Board != nil {
		snapshotBoard(s.Board, &rs.Board)
	}
	if s.ActivePiece != nil && s.Phase == PhasePlaying {
		rs.Active = snapshotActivePiece(s.ActivePiece, s.Board)
	}
	return rs
}

func snapshotBoard(b *Board, grid *CellGrid) {
	for r := 0; r < BoardRows; r++ {
		for c := 0; c < BoardCols; c++ {
			grid[r][c] = b.Cell(c, r)
		}
	}
}

func snapshotActivePiece(ap *ActivePiece, b *Board) *ActivePieceView {
	fp := ap.Footprint()
	fpSet := make(map[Position]bool, len(fp))
	for _, p := range fp {
		fpSet[p] = true
	}
	ghostOrigin := ap.GhostOrigin(b)
	ghostPiece := &ActivePiece{Kind: ap.Kind, Origin: ghostOrigin, Rotation: ap.Rotation}
	ghost := ghostFPExcluding(ghostPiece.Footprint(), fpSet)
	return &ActivePieceView{Kind: ap.Kind, Footprint: fp, Ghost: ghost}
}

func ghostFPExcluding(ghostFP []Position, exclude map[Position]bool) []Position {
	var ghost []Position
	for _, p := range ghostFP {
		if !exclude[p] {
			ghost = append(ghost, p)
		}
	}
	return ghost
}

// bagRefilled implements BagRefilled() internal event.
func (s *Session) bagRefilled() {
	s.Bag.Refill()
}

// pieceSpawned implements PieceSpawned() internal event per the allium rule.
func (s *Session) pieceSpawned() {
	var kind Kind
	if s.NextKind == KindNone {
		// First spawn: draw directly from bag.
		if s.Bag.Empty() {
			s.bagRefilled()
		}
		kind = s.Bag.Draw()
	} else {
		kind = s.NextKind
		if !s.Bag.Empty() {
			s.Bag.Draw() // consume the peeked element
		}
	}

	// Advance bag for next preview; refill if needed.
	if s.Bag.Empty() {
		s.bagRefilled()
	}
	s.NextKind = s.Bag.Peek()

	s.ActivePiece = &ActivePiece{
		Kind:     kind,
		Origin:   Position{Col: SpawnCol, Row: SpawnRow},
		Rotation: RotationNorth,
	}
	s.HoldUsed = false
	s.LockTicksRemaining = LockTicks

	if !s.ActivePiece.ValidPlacement(s.Board) {
		s.gameOver()
	}
}

// gameOver implements GameOver() internal event.
func (s *Session) gameOver() {
	s.Phase = PhaseGameOver
}

// lockPiece implements PieceLockedOnBoard() internal event.
func (s *Session) lockPiece() {
	if s.ActivePiece == nil {
		return
	}
	for _, p := range s.ActivePiece.Footprint() {
		s.Board.SetCell(p.Col, p.Row, s.ActivePiece.Kind)
	}
	s.ActivePiece = nil
	s.linesCleared()
}

// linesCleared implements LinesCleared() internal event.
// When rows are cleared: stats update immediately; rows stay on the board for one
// flash tick (stored in FlashRows); ClearRows and spawn defer to the next Tick.
// When no rows are cleared: spawn happens synchronously.
func (s *Session) linesCleared() {
	full := s.Board.FullRows()
	if len(full) > 0 {
		s.scoreAwarded(len(full))
		s.LinesClearedTotal += len(full)
		s.levelProgressed()
		s.FlashRows = full // commit tick will call ClearRows + pieceSpawned
		return
	}
	s.pieceSpawned()
}

// scoreAwarded implements ScoreAwarded(count) rule.
func (s *Session) scoreAwarded(count int) {
	multiplier := map[int]int{1: 100, 2: 300, 3: 500, 4: 800}[count]
	s.Score += multiplier * s.Level
}

// levelProgressed implements LevelProgressed() rule.
func (s *Session) levelProgressed() {
	earned := (s.LinesClearedTotal/LinesPerLevel) + 1
	if earned > s.Level {
		s.Level = earned
	}
}

			return nil

			ok := ls.onKey(k)
			mu.Unlock()

				return nil
			}

			mu.Unlock()

// Render computes a complete Frame for the given render state.
func Render(rs game.RenderState, termW, termH int) Frame {
	const (
		assemblyW = 40 // 8 + 1 + 22 + 1 + 8
		assemblyH = 22 // 20 cells + 2 border rows
	)

	lines := make([]string, assemblyH)
	buildAssembly(rs, lines)

	leftPad := 0
	if termW > assemblyW {
		leftPad = (termW - assemblyW) / 2
	}
	topPad := 0
	if termH > assemblyH {
		topPad = (termH - assemblyH) / 2
	}

	padStr := strings.Repeat(" ", leftPad)
	out := make([]string, termH)
	for i := range out {
		row := i - topPad
		if row >= 0 && row < assemblyH {
			out[i] = padStr + lines[row]
		} else {
			out[i] = padStr + strings.Repeat(" ", assemblyW)
		}
	}
	return Frame{rows: out}
}

// RenderTooSmall returns a frame for terminals smaller than 40×24.
func RenderTooSmall(termW, termH int) Frame {
	msg := "RESIZE TERMINAL"
	out := make([]string, termH)
	for i := range out {
		if i == termH/2 {
			pad := (termW - len(msg)) / 2
			if pad < 0 {
				pad = 0
			}
			out[i] = strings.Repeat(" ", pad) + colFgPrim + msg + reset
		} else {
			out[i] = ""
		}
	}
	return Frame{rows: out}
}

// stripANSI removes ANSI CSI escape sequences (ESC [ ... letter) from s.
func stripANSI(s string) string {
	var out strings.Builder
	for i := 0; i < len(s); {
		if isANSIEscapeStart(s, i) {
			i = skipANSIEscape(s, i+2)
		} else {
			out.WriteByte(s[i])
			i++
		}
	}
	return out.String()
}

func isANSIEscapeStart(s string, i int) bool {
	return s[i] == '\x1b' && i+1 < len(s) && s[i+1] == '['
}

func skipANSIEscape(s string, i int) int {
	for i < len(s) && !isANSITerminator(s[i]) {
		i++
	}
	if i < len(s) {
		i++
	}
	return i
}

func isANSITerminator(b byte) bool {
	return b >= 'A' && b <= 'z'
}

	}

func boardRow(rs game.RenderState, row int, activeFP, ghostFP map[game.Position]bool, activeKind game.Kind) string {
	if row == 0 {
		return colChrome + "┌" + strings.Repeat("─", 20) + "┐" + reset
	}
	if row == 21 {
		return colChrome + "└" + strings.Repeat("─", 20) + "┘" + reset
	}

	cellRow := row - 1 // board row 0..19
	var sb strings.Builder
	sb.WriteString(colChrome + "│" + reset)
	for col := 0; col < game.BoardCols; col++ {
		pos := game.Position{Col: col, Row: cellRow}
		sb.WriteString(cellStr(rs, pos, activeFP, ghostFP, activeKind))
	}
	sb.WriteString(colChrome + "│" + reset)
	return sb.String()
}

// cellStr returns the 2-char styled string for one board cell.
func cellStr(rs game.RenderState, pos game.Position, activeFP, ghostFP map[game.Position]bool, activeKind game.Kind) string {
	switch rs.Phase {
	case game.PhasePaused:
		return pausedCell(pos)
	case game.PhaseWaiting:
		return overlayCell(pos, waitingOverlayGrid())
	case game.PhaseGameOver:
		return gameOverCell(rs, pos)
	default:
		return playingCell(rs, pos, activeFP, ghostFP, activeKind)
	}
}

	fp := game.Footprint(kind, game.Position{Col: 0, Row: 0}, game.RotationNorth)
	minC, minR := fp[0].Col, fp[0].Row

// SpawnBufferRows is the number of hidden rows above the visible board.
// Pieces spawn at row SpawnRow = -2, so the buffer must cover rows -2..-1.
const SpawnBufferRows = 2

// totalRows is the full internal storage height: buffer + visible.
const totalRows = SpawnBufferRows + BoardRows

// Board is the 10×20 visible play field plus a 2-row hidden spawn buffer.
// Internal storage uses rows 0..totalRows-1; public API uses row indices
// where 0 is the topmost visible row and negative indices address the buffer.
type Board struct {
	// cells[internalRow][col]; internalRow = row + SpawnBufferRows
	cells [totalRows][BoardCols]Kind
}

// NewBoard returns a blank board.
func NewBoard() *Board {
	return &Board{}
}

func (b *Board) toInternal(row int) int {
	return row + SpawnBufferRows
}

// Cell returns the Kind at (col, row). Returns KindNone for out-of-range rows.
func (b *Board) Cell(col, row int) Kind {
	ir := b.toInternal(row)
	if ir < 0 || ir >= totalRows {
		return KindNone
	}
	return b.cells[ir][col]
}

func (b *Board) SetCell(col, row int, k Kind) {
	ir := b.toInternal(row)
	if ir < 0 || ir >= totalRows {
		return
	}
	b.cells[ir][col] = k
}

// CellOccupied reports whether (col, row) contains a locked cell.
// Returns true for column out-of-bounds or row below the bottom.
// Returns false for rows above the buffer top (shouldn't occur in play).
func (b *Board) CellOccupied(col, row int) bool {
	if col < 0 || col >= BoardCols {
		return true
	}
	ir := b.toInternal(row)
	if ir < 0 {
		return false
	}
	if ir >= totalRows {
		return true
	}
	return b.cells[ir][col] != KindNone
}

			full = append(full, r)
		}
	}

	ir := b.toInternal(row)
	for c := 0; c < BoardCols; c++ {
		if b.cells[ir][c] == KindNone {
			return false
		}
	}
	return true
}

// ClearRows removes the given visible rows and drops everything above them.
// Rows must be provided in ascending order.
func (b *Board) ClearRows(rows []int) {
	if len(rows) == 0 {
		return
	}
	cleared := make(map[int]bool, len(rows))
	for _, r := range rows {
		cleared[r] = true
	}

	// Rebuild internal storage from bottom up, skipping cleared rows.
	var next [totalRows][BoardCols]Kind
	dst := totalRows - 1
	for src := totalRows - 1; src >= 0; src-- {
		visibleRow := src - SpawnBufferRows
		if cleared[visibleRow] {
			continue
		}
		next[dst] = b.cells[src]
		dst--
	}
	b.cells = next
}

func (b *Board) FilledCells() []struct {
	Col, Row int
	Kind     Kind
} {
	var out []struct {

// Bag is the 7-bag randomizer. It holds a set of piece kinds; when empty it

type Bag struct {

// NewBag creates an empty bag with the given random source.

	return &Bag{rng: rng}

// Empty reports whether the bag has no remaining kinds.
func (bg *Bag) Empty() bool {
	return len(bg.remaining) == 0
}

// Refill fills the bag with all 7 kinds in a random order.
func (bg *Bag) Refill() {
	bg.remaining = make([]Kind, 7)
	copy(bg.remaining, AllKinds[:])
	bg.rng.Shuffle(len(bg.remaining), func(i, j int) {
		bg.remaining[i], bg.remaining[j] = bg.remaining[j], bg.remaining[i]
	})
}

	k := bg.remaining[0]

}

// TestPropertyHoldInteriorCharsetBounded verifies that the interior cells of holdRow
// are bounded to the allowed vocabulary: ██ or "  " (blank).
func TestPropertyHoldInteriorCharsetBounded(t *testing.T) {
	f := func(kindIdx uint8, holdUsed bool, previewRow uint8) bool {
		kindCount := len(game.AllKinds) + 1
		idx := int(kindIdx) % kindCount
		var kind game.Kind
		if idx < len(game.AllKinds) {
			kind = game.AllKinds[idx]
		} else {
			kind = game.KindNone
		}
		row := int(previewRow%3) + 2 // interior rows 2..4

		rs := game.RenderState{HeldKind: kind, HoldUsed: holdUsed}
		plain := stripANSI(holdRow(rs, row))

		// Strip leading/trailing border runes: │...│ → interior runes.
		// Use rune slicing since │ is a multi-byte UTF-8 character.
		runes := []rune(plain)
		if len(runes) < 2 {
			return false
		}
		interior := string(runes[1 : len(runes)-1])

		// Each 2-char glyph cell must be in holdVocab.
		// The interior is 3 cells × 2 chars = 6 rune pairs.
		interiorRunes := []rune(interior)
		for i := 0; i+1 < len(interiorRunes); i += 2 {
			cell := string(interiorRunes[i : i+2])
			if !holdVocab[cell] {
				return false
			}
		}
		return true
	}
	if err := quick.Check(f, nil); err != nil {
		t.Error(err)
	}
}

// TestPropertyHoldFullAndDimNeverCoRendered verifies that a hold panel never shows
// both full-color (no dim) and dimmed glyphs in the same frame — the states are
// mutually exclusive: either hold_used=false (full color) or hold_used=true (dim).
func TestPropertyHoldFullAndDimNeverCoRendered(t *testing.T) {
	f := func(kindIdx uint8) bool {
		kindCount := len(game.AllKinds)
		if kindCount == 0 {
			return true
		}
		kind := game.AllKinds[int(kindIdx)%kindCount]

		// Full-color state (hold_used=false).
		rsFull := game.RenderState{HeldKind: kind, HoldUsed: false}
		// Dim state (hold_used=true).
		rsDim := game.RenderState{HeldKind: kind, HoldUsed: true}

		for row := 2; row <= 4; row++ {

				if !strings.Contains(dimRaw, colFgDim) {
					return false
				}
			}
		}
		return true
	}
	if err := quick.Check(f, nil); err != nil {
		t.Error(err)
	}
}
