// acceptance-runner is a persistent runner adapter for gherkin-mutator.
// It reads WorkerRequest JSON lines from stdin and writes WorkerResponse JSON lines to stdout.
// Each request provides a path to a (possibly mutated) feature JSON; the runner executes
// the acceptance tests with GHERKIN_FEATURE_JSON pointing to that file.
package main

import (
	"bufio"
	"bytes"
	"encoding/json"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"time"
)

type workerRequest struct {
	ID           string `json:"id"`
	FeatureJSON  string `json:"feature_json"`
	GeneratedDir string `json:"generated_dir"`
	WorkDir      string `json:"work_dir"`
	Timeout      string `json:"timeout,omitempty"`
}

type workerResponse struct {
	ID       string `json:"id"`
	Outcome  string `json:"outcome"`
	Output   string `json:"output"`
	Error    string `json:"error"`
	Duration int64  `json:"duration"`
}

func main() {
	projectRoot := findProjectRoot()

	scanner := bufio.NewScanner(os.Stdin)
	encoder := json.NewEncoder(os.Stdout)

	for scanner.Scan() {
		var req workerRequest
		if err := json.Unmarshal(scanner.Bytes(), &req); err != nil {
			resp := workerResponse{Outcome: "infrastructure_error", Error: "invalid request JSON: " + err.Error()}
			_ = encoder.Encode(resp)
			continue
		}

		start := time.Now()
		featureJSON := req.FeatureJSON
		if !filepath.IsAbs(featureJSON) {
			featureJSON = filepath.Join(projectRoot, featureJSON)
		}

		testPkg := acceptancePackage(req.GeneratedDir, featureJSON)
		cmd := exec.Command("go", "test", "-count=1", testPkg)
		cmd.Dir = projectRoot
		cmd.Env = append(os.Environ(), "GHERKIN_FEATURE_JSON="+featureJSON)

		var out bytes.Buffer
		cmd.Stdout = &out
		cmd.Stderr = &out

		err := cmd.Run()
		duration := time.Since(start).Nanoseconds()

		var outcome string
		if err != nil {
			if _, ok := err.(*exec.ExitError); ok {
				outcome = "test_failure"
			} else {
				outcome = "infrastructure_error"
			}
		} else {
			outcome = "test_success"
		}

		resp := workerResponse{
			ID:       req.ID,
			Outcome:  outcome,
			Output:   out.String(),
			Duration: duration,
		}
		if err != nil {
			if _, ok := err.(*exec.ExitError); !ok {
				resp.Error = err.Error()
			}
		}
		if encErr := encoder.Encode(resp); encErr != nil {
			fmt.Fprintf(os.Stderr, "encode error: %v\n", encErr)
			return
		}
	}
	if err := scanner.Err(); err != nil {
		fmt.Fprintf(os.Stderr, "stdin error: %v\n", err)
	}
}

// acceptancePackage returns the go test package path for a given generated-dir or feature JSON path.
func acceptancePackage(generatedDir, featureJSON string) string {
	for _, name := range []string{"gravity_and_lock", "game_lifecycle", "lines_and_score", "movement_and_rotation", "soft_drop", "hold", "pause"} {
		if strings.Contains(generatedDir, name) || strings.Contains(featureJSON, name) {
			return "./acceptance/" + name + "/..."
		}
	}

}

func findProjectRoot() string {
	dir, err := os.Getwd()
	if err != nil {
		return "."
	}
	for {
		if _, err := os.Stat(filepath.Join(dir, "go.mod")); err == nil {
			return dir
		}
		parent := filepath.Dir(dir)
		if parent == dir {
			return "."
		}
		dir = parent
	}
}
