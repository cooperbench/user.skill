  +// --- Overlay grid ---
  +
  +// overlayRow is one row of the board interior: 10 cells, each 2 chars.
  +// Empty string means the cell renders as the default empty cell glyph.
  +type overlayRow [10]string
  +
  +// overlayGrid is 20 rows of the board interior.
  +type overlayGrid [20]overlayRow
  +
  +func overlayCell(pos game.Position, g overlayGrid) string {
  +	if pos.Row < 0 || pos.Row >= 20 || pos.Col < 0 || pos.Col >= 10 {
  +		return colFgDim + colBg + "· " + reset
  +	}
  +	cell := g[pos.Row][pos.Col]
  +	if cell == "" {
  +		return colFgDim + colBg + "· " + reset
  +	}
  +	return cell
  +}
  +
  +// writeOverlayLine writes text centered in a 20-char (10-cell) overlay row.
  +// color is the ANSI color prefix to use.
  +func writeOverlayLine(g *overlayGrid, row int, text, color string) {
  +	if row < 0 || row >= 20 {
  +		return
  +	}
  +	// Pad text to 20 chars if shorter; truncate if longer.
  +	if len(text) > 20 {
  +		text = text[:20]
  +	}
  +	pad := (20 - len(text)) / 2
  +	// Build a 20-char string padded on both sides.
  +	full := strings.Repeat(" ", pad) + text
  +	// Right-pad to 20.
  +	for len(full) < 20 {
  +		full += " "
  +	}
  +	// Split into cells of 2 chars each. All cells (including spaces) suppress the dot.
  +	for col := 0; col < 10; col++ {
  +		ch := full[col*2 : col*2+2]
  +		if ch == "  " {
  +			g[row][col] = colFgDim + colBg + "  " + reset // blank, no dot
  +		} else {
  +			g[row][col] = color + ch + reset
  +		}
  +	}
  +}
