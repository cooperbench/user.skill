// Run starts the game loop. It blocks until the user exits (Ctrl+C).
func Run() error {
	rng := rand.New(rand.NewSource(time.Now().UnixNano()))
	s := game.NewSession(rng)

	rt, err := EnableRaw()
	if err != nil {
		return fmt.Errorf("raw terminal: %w", err)
	}
	defer rt.Restore()

	// Hide cursor, clear screen.
	fmt.Fprint(os.Stdout, "\x1b[?25l\x1b[2J")
	defer fmt.Fprint(os.Stdout, "\x1b[?25h\x1b[2J") // show cursor, clear on exit

	// Signal handler for clean exit.
	sigCh := make(chan os.Signal, 1)
	signal.Notify(sigCh, syscall.SIGTERM, syscall.SIGINT)

	// Key channel.
	keyCh := make(chan Key, 16)
	go func() {
		for {
			k := ReadKey()
			keyCh <- k
		}
	}()

	ticker := time.NewTicker(time.Second / tickHz)
	defer ticker.Stop()

	var mu sync.Mutex
	lastW, lastH := TermSize()
	tooSmall := lastW < minCols || lastH < minRows

	redraw := func() {
		w, h := TermSize()
		lastW, lastH = w, h
		tooSmall = w < minCols || h < minRows
		var f Frame
		if tooSmall {
			f = RenderTooSmall(w, h)
		} else {
			f = Render(s, w, h)
		}
		fmt.Fprint(os.Stdout, f.String())
	}

	redraw()

	for {
		select {
		case <-sigCh:
			return nil

		case k := <-keyCh:
			if k == KeyCtrlC {
				return nil
			}
			mu.Lock()
			handleKey(s, k, tooSmall)
			redraw()
			mu.Unlock()

		case <-ticker.C:
			mu.Lock()
			w, h := TermSize()
			if w != lastW || h != lastH {
				// Terminal resized.
				if w < minCols || h < minRows {
					if s.Phase == game.PhasePlaying {
						s.PlayerPauses()
					}
				}
			}
			if s.Phase == game.PhasePlaying {
				s.Tick()
			}
			redraw()
			mu.Unlock()
		}
	}
}

// handleKey dispatches a key event to the session.
func handleKey(s *game.Session, k Key, tooSmall bool) {
	if tooSmall {
		return
	}
	switch s.Phase {
	case game.PhaseWaiting:
		if k == KeyEnter {
			s.PlayerStartsGame()
		}
	case game.PhasePlaying:
		switch k {
		case KeyLeft:
			s.PlayerMovesLeft()
		case KeyRight:
			s.PlayerMovesRight()
		case KeyUp, KeyX:
			// PlayerRotatesCW — deferred (rotation not yet in session)
		case KeyZ:
			// PlayerRotatesCCW — deferred
		case KeyDown:
			s.PlayerSoftDrops()
		case KeySpace:
			s.PlayerHardDrops()
		case KeyC:
			s.PlayerHolds()
		case KeyP, KeyEsc:
			s.PlayerPauses()
		}
	case game.PhasePaused:
		switch k {
		case KeyP, KeyEsc:
			s.PlayerResumes()
		}
	case game.PhaseGameOver:
		if k == KeyR {
			s.PlayerRestartsGame()
		}
	}
}

// Frame is a rendered terminal frame (one string per terminal row).
type Frame struct {
	rows []string
}

// Render computes a complete Frame for the given session.
func Render(s *game.Session, termW, termH int) Frame {
	const (
		assemblyW = 40 // 8 + 1 + 22 + 1 + 8
		assemblyH = 22 // 20 cells + 2 border rows
	)

	lines := make([]string, assemblyH)
	buildAssembly(s, lines)

	leftPad := 0
	if termW > assemblyW {
		leftPad = (termW - assemblyW) / 2
	}
	topPad := 0
	if termH > assemblyH {
		topPad = (termH - assemblyH) / 2
	}

	padStr := strings.Repeat(" ", leftPad)
	out := make([]string, termH)
	for i := range out {
		row := i - topPad
		if row >= 0 && row < assemblyH {
			out[i] = padStr + lines[row]
		} else {
			out[i] = padStr + strings.Repeat(" ", assemblyW)
		}
	}
	return Frame{rows: out}
}

// RenderTooSmall returns a frame for terminals smaller than 40×24.
func RenderTooSmall(termW, termH int) Frame {
	msg := "RESIZE TERMINAL"
	out := make([]string, termH)
	for i := range out {
		if i == termH/2 {
			pad := (termW - len(msg)) / 2
			if pad < 0 {
				pad = 0
			}
			out[i] = strings.Repeat(" ", pad) + colFgPrim + msg + reset
		} else {
			out[i] = ""
		}
	}
	return Frame{rows: out}
}

// String serializes the frame for terminal output.
func (f Frame) String() string {
	var sb strings.Builder
	sb.WriteString("\x1b[H")
	sb.WriteString(colBg)
	for _, row := range f.rows {
		sb.WriteString(row)
		sb.WriteString(reset + "\r\n")
	}
	return sb.String()
}

// PlainRows returns frame rows stripped of ANSI escapes, for snapshot tests.
func (f Frame) PlainRows() []string {
	out := make([]string, len(f.rows))
	for i, r := range f.rows {
		out[i] = stripANSI(r)
	}
	return out
}

// stripANSI removes ANSI CSI escape sequences (ESC [ ... letter) from s.
func stripANSI(s string) string {
	var out strings.Builder
	i := 0
	for i < len(s) {
		if s[i] == '\x1b' && i+1 < len(s) && s[i+1] == '[' {
			i += 2
			for i < len(s) && !isANSITerminator(s[i]) {
				i++
			}
			if i < len(s) {
				i++
			}
			continue
		}
		out.WriteByte(s[i])
		i++
	}
	return out.String()
}

func isANSITerminator(b byte) bool {
	return b >= 'A' && b <= 'z'
}

// --- Assembly builder ---

// Column layout: hold(8) + gap(1) + board(22) + gap(1) + right(8) = 40
func buildAssembly(s *game.Session, lines []string) {
	// Compute active piece and ghost footprints (only when playing).
	activeFP := map[game.Position]bool{}
	ghostFP := map[game.Position]bool{}
	var activePieceKind game.Kind
	if s.ActivePiece != nil && s.Phase == game.PhasePlaying {
		activePieceKind = s.ActivePiece.Kind
		for _, p := range s.ActivePiece.Footprint() {
			activeFP[p] = true
		}
		ghostOrigin := s.ActivePiece.GhostOrigin(s.Board)
		ghostPiece := &game.ActivePiece{Kind: s.ActivePiece.Kind, Origin: ghostOrigin, Rotation: s.ActivePiece.Rotation}
		for _, p := range ghostPiece.Footprint() {
			if !activeFP[p] {
				ghostFP[p] = true
			}
		}
	}

	const boardH = 22
	for row := 0; row < boardH; row++ {
		left := holdRow(s, row)
		center := boardRow(s, row, activeFP, ghostFP, activePieceKind)
		right := rightRow(s, row)
		lines[row] = left + " " + center + " " + right
	}
}

// --- Hold panel (8 cols wide, rows 0..5 used; rows 6..21 blank) ---

func holdRow(s *game.Session, row int) string {
	switch row {
	case 0:
		return colChrome + "┌──────┐" + reset
	case 1:
		return colChrome + "│ HOLD │" + reset
	case 2, 3, 4:
		return previewPanelRow(s.HeldKind, s.HoldUsed, row-2)
	case 5:
		return colChrome + "└──────┘" + reset
	default:
		return "        " // 8 spaces
	}
}

// previewPanelRow renders one of the 3 interior rows of an 8-col panel,
// showing the piece preview in rows 0..1 (rows 2 is just padding).
// Panel interior is 6 chars (border │ + 6 + │ = 8).
func previewPanelRow(kind game.Kind, dimmed bool, previewRow int) string {
	if kind == game.KindNone {
		return colChrome + "│      │" + reset
	}
	cells := normalizedFootprint(kind)

	// Place the 3-col-wide (6-char) piece in a 6-char interior.
	// Each cell is 2 chars. Piece bounding box is at most 4 cells wide.
	// Center horizontally in the 3-cell (6-char) space.
	maxCol := 0
	for _, p := range cells {
		if p.Col > maxCol {
			maxCol = p.Col
		}
	}
	pieceW := maxCol + 1
	// Available interior cells = 3. Center offset.
	offsetCells := (3 - pieceW) / 2
	if offsetCells < 0 {
		offsetCells = 0
	}

	// Build 3 "cells" worth of 2-char content.
	type cell struct{ filled bool }
	row3 := [3]cell{}
	for _, p := range cells {
		if p.Row == previewRow {
			c := p.Col + offsetCells
			if c >= 0 && c < 3 {
				row3[c].filled = true
			}
		}
	}

	var sb strings.Builder
	sb.WriteString(colChrome + "│" + reset)
	for _, c := range row3 {
		if c.filled {
			if dimmed {
				sb.WriteString(colFgDim + colBg + "░░" + reset)
			} else {
				sb.WriteString(pieceColorActive(int(kind)) + "██" + reset)
			}
		} else {
			sb.WriteString("  ")
		}
	}
	sb.WriteString(colChrome + "│" + reset)
	return sb.String()
}

// normalizedFootprint returns piece cells in north rotation, 0-indexed from (0,0).
func normalizedFootprint(kind game.Kind) []game.Position {
	fp := game.Footprint(kind, game.Position{Col: 0, Row: 0}, game.RotationNorth)
	if len(fp) == 0 {
		return nil
	}
	minC, minR := fp[0].Col, fp[0].Row
	for _, p := range fp {
		if p.Col < minC {
			minC = p.Col
		}
		if p.Row < minR {
			minR = p.Row
		}
	}
	norm := make([]game.Position, len(fp))
	for i, p := range fp {
		norm[i] = game.Position{Col: p.Col - minC, Row: p.Row - minR}
	}
	return norm
}

// cellStr returns the 2-char styled string for one board cell.

// writeOverlayLine writes text centered in a 20-char (10-cell) overlay row.
// color is the ANSI color prefix to use.
func writeOverlayLine(g *overlayGrid, row int, text, color string) {
	if row < 0 || row >= 20 {
		return
	}
	// Pad text to 20 chars if shorter; truncate if longer.
	if len(text) > 20 {
		text = text[:20]
	}
	pad := (20 - len(text)) / 2
	// Build a 20-char string padded on both sides.
	full := strings.Repeat(" ", pad) + text
	// Right-pad to 20.
	for len(full) < 20 {
		full += " "
	}
	// Split into cells of 2 chars each. All cells (including spaces) suppress the dot.
	for col := 0; col < 10; col++ {
		ch := full[col*2 : col*2+2]
		if ch == "  " {
			g[row][col] = colFgDim + colBg + "  " + reset // blank, no dot
		} else {
			g[row][col] = color + ch + reset
		}
	}
}

// SpawnBufferRows is the number of hidden rows above the visible board.
// Pieces spawn at row SpawnRow = -2, so the buffer must cover rows -2..-1.
const SpawnBufferRows = 2

// totalRows is the full internal storage height: buffer + visible.
const totalRows = SpawnBufferRows + BoardRows

// Board is the 10×20 visible play field plus a 2-row hidden spawn buffer.
// Internal storage uses rows 0..totalRows-1; public API uses row indices
// where 0 is the topmost visible row and negative indices address the buffer.
type Board struct {
	// cells[internalRow][col]; internalRow = row + SpawnBufferRows
	cells [totalRows][BoardCols]Kind
}

// NewBoard returns a blank board.
func NewBoard() *Board {
	return &Board{}
}

func (b *Board) toInternal(row int) int {
	return row + SpawnBufferRows
}

// Cell returns the Kind at (col, row). Returns KindNone for out-of-range rows.
func (b *Board) Cell(col, row int) Kind {
	ir := b.toInternal(row)
	if ir < 0 || ir >= totalRows {
		return KindNone
	}
	return b.cells[ir][col]
}

// SetCell places a kind at (col, row). No-op for out-of-range rows.
func (b *Board) SetCell(col, row int, k Kind) {
	ir := b.toInternal(row)
	if ir < 0 || ir >= totalRows {
		return
	}
	b.cells[ir][col] = k
}

// CellOccupied reports whether (col, row) contains a locked cell.
// Returns true for column out-of-bounds or row below the bottom.
// Returns false for rows above the buffer top (shouldn't occur in play).
func (b *Board) CellOccupied(col, row int) bool {
	if col < 0 || col >= BoardCols {
		return true
	}
	ir := b.toInternal(row)
	if ir < 0 {
		return false
	}
	if ir >= totalRows {
		return true
	}
	return b.cells[ir][col] != KindNone
}

// FullRows returns visible row indices (0-indexed) where every cell is filled.
func (b *Board) FullRows() []int {
	var full []int
	for r := 0; r < BoardRows; r++ {
		if b.isRowFull(r) {
			full = append(full, r)
		}
	}
	return full
}

func (b *Board) isRowFull(row int) bool {
	ir := b.toInternal(row)
	for c := 0; c < BoardCols; c++ {
		if b.cells[ir][c] == KindNone {
			return false
		}
	}
	return true
}

// ClearRows removes the given visible rows and drops everything above them.
// Rows must be provided in ascending order.
func (b *Board) ClearRows(rows []int) {
	if len(rows) == 0 {
		return
	}
	cleared := make(map[int]bool, len(rows))
	for _, r := range rows {
		cleared[r] = true
	}

	// Rebuild internal storage from bottom up, skipping cleared rows.
	var next [totalRows][BoardCols]Kind
	dst := totalRows - 1
	for src := totalRows - 1; src >= 0; src-- {
		visibleRow := src - SpawnBufferRows
		if cleared[visibleRow] {
			continue
		}
		next[dst] = b.cells[src]
		dst--
	}
	b.cells = next
}

// FilledCells returns all visible (col, row, kind) tuples where kind != KindNone.
func (b *Board) FilledCells() []struct {
	Col, Row int
	Kind     Kind
} {
	var out []struct {
		Col, Row int
		Kind     Kind
	}
	for r := 0; r < BoardRows; r++ {
		ir := b.toInternal(r)
		for c := 0; c < BoardCols; c++ {
			if b.cells[ir][c] != KindNone {
				out = append(out, struct {
					Col, Row int
					Kind     Kind
				}{c, r, b.cells[ir][c]})
			}
		}
	}
	return out

// Footprint returns the absolute board positions for kind at origin+rotation.
// k must be KindI..KindZ (not KindNone).
func Footprint(k Kind, origin Position, rot Rotation) []Position {
	offsets := pieceFootprints[k-1][rot]
	out := make([]Position, len(offsets))
	for i, o := range offsets {
		out[i] = Position{Col: origin.Col + o.Col, Row: origin.Row + o.Row}
	}
	return out
}

// ValidPlacement reports whether all footprint cells are in-column-bounds and
// not occupied on the board, including the hidden spawn buffer rows.
func (ap *ActivePiece) ValidPlacement(b *Board) bool {
	for _, p := range ap.Footprint() {
		if b.CellOccupied(p.Col, p.Row) {
			return false
		}
		if p.Row >= BoardRows {
			return false
		}
	}
	return true
}

// AtRest reports whether shifting origin.row by +1 would make ValidPlacement false.
func (ap *ActivePiece) AtRest(b *Board) bool {
	below := &ActivePiece{Kind: ap.Kind, Origin: Position{Col: ap.Origin.Col, Row: ap.Origin.Row + 1}, Rotation: ap.Rotation}
	return !below.ValidPlacement(b)
}

// GhostOrigin returns the lowest row at which the piece still has a valid placement.
func (ap *ActivePiece) GhostOrigin(b *Board) Position {
	ghost := &ActivePiece{Kind: ap.Kind, Origin: ap.Origin, Rotation: ap.Rotation}
	for {
		next := &ActivePiece{Kind: ap.Kind, Origin: Position{Col: ghost.Origin.Col, Row: ghost.Origin.Row + 1}, Rotation: ap.Rotation}
		if !next.ValidPlacement(b) {
			break
		}
		ghost.Origin = next.Origin
	}
	return ghost.Origin
}

// EnableRaw puts the terminal into raw mode.
func EnableRaw() (*RawTerminal, error) {
	fd := int(os.Stdin.Fd())
	var orig syscall.Termios
	if _, _, errno := syscall.Syscall(syscall.SYS_IOCTL, uintptr(fd), syscall.TIOCGETA, uintptr(unsafe.Pointer(&orig))); errno != 0 {
		return nil, errno
	}
	raw := orig
	raw.Iflag &^= syscall.ICRNL | syscall.IXON
	raw.Lflag &^= syscall.ECHO | syscall.ICANON | syscall.IEXTEN | syscall.ISIG
	raw.Cc[syscall.VMIN] = 1
	raw.Cc[syscall.VTIME] = 0
	if _, _, errno := syscall.Syscall(syscall.SYS_IOCTL, uintptr(fd), syscall.TIOCSETA, uintptr(unsafe.Pointer(&raw))); errno != 0 {
		return nil, errno
	}
	return &RawTerminal{orig: orig, fd: fd}, nil
}

	syscall.Syscall(syscall.SYS_IOCTL, uintptr(rt.fd), syscall.TIOCSETA, uintptr(unsafe.Pointer(&rt.orig)))
}
