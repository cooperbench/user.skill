package tui

import "github.com/gamaleds/tetris/pkg/game"

// handleResize pauses the game if the terminal shrinks below the minimum dimensions
// while the game is playing.
func handleResize(s *game.Session, w, h int) {
	if w < minCols || h < minRows {
		if s.CurrentPhase() == game.PhasePlaying {
			s.PlayerPauses()
		}
	}
}

// handleKey dispatches a key event to the session.
func handleKey(s *game.Session, k Key, tooSmall bool) {
	if tooSmall {
		return
	}
	switch s.CurrentPhase() {
	case game.PhaseWaiting:
		if k == KeyEnter {
			s.PlayerStartsGame()
		}
	case game.PhasePlaying:
		switch k {
		case KeyLeft:
			s.PlayerMovesLeft()
		case KeyRight:
			s.PlayerMovesRight()
		case KeyUp, KeyX:
			s.PlayerRotatesCW()
		case KeyZ:
			s.PlayerRotatesCCW()
		case KeyDown:
			s.PlayerSoftDrops()
		case KeySpace:
			s.PlayerHardDrops()
		case KeyC:
			s.PlayerHolds()
		case KeyP, KeyEsc:
			s.PlayerPauses()
		}
	case game.PhasePaused:
		switch k {
		case KeyP, KeyEsc:
			s.PlayerResumes()
		}
	case game.PhaseGameOver:
		if k == KeyR {
			s.PlayerRestartsGame()
		}
	}
}

const (
	minCols = 40
	minRows = 24
	tickHz  = 10 // gravity ticks per second (simplified; full speed curve deferred)

	// Spec DAS timing (EXPERIENCE.md). Raw-mode terminal reads carry no
	// key-release events, so an internal repeat timer cannot know when the
	// key is let go — it runs away and glides the piece into the wall.
	// Hold-repeat is therefore delegated to the terminal's own key
	// autorepeat (one move per key event) until release detection exists
	// (e.g. kitty keyboard protocol).
	dasDelay    = 167 * time.Millisecond
	dasInterval = 33 * time.Millisecond
)

// Run starts the game loop. It blocks until the user exits (Ctrl+C).
func Run() error {
	rng := rand.New(rand.NewSource(time.Now().UnixNano()))
	s := game.NewSession(rng)

	rt, err := EnableRaw()
	if err != nil {
		return fmt.Errorf("raw terminal: %w", err)
	}
	defer rt.Restore()

	// Hide cursor, clear screen.
	fmt.Fprint(os.Stdout, "\x1b[?25l\x1b[2J")
	defer fmt.Fprint(os.Stdout, "\x1b[?25h\x1b[2J") // show cursor, clear on exit

	// Signal handler for clean exit.
	sigCh := make(chan os.Signal, 1)
	signal.Notify(sigCh, syscall.SIGTERM, syscall.SIGINT)

	// Key channel.
	keyCh := make(chan Key, 16)
	go func() {
		for {
			k := ReadKey()
			keyCh <- k
		}
	}()

	ticker := time.NewTicker(time.Second / tickHz)
	defer ticker.Stop()

	var mu sync.Mutex
	lastW, lastH := TermSize()
	tooSmall := lastW < minCols || lastH < minRows

	redraw := func() {
		w, h := TermSize()
		lastW, lastH = w, h
		tooSmall = w < minCols || h < minRows
		var f Frame
		if tooSmall {
			f = RenderTooSmall(w, h)
		} else {
			f = Render(s.Snapshot(), w, h)
		}
		fmt.Fprint(os.Stdout, f.String())
	}

	redraw()

	for {
		select {
		case <-sigCh:
			return nil

		case k := <-keyCh:
			if k == KeyCtrlC {
				return nil
			}
			mu.Lock()
			handleKey(s, k, tooSmall)
			redraw()
			mu.Unlock()

		case <-ticker.C:
			mu.Lock()
			w, h := TermSize()
			if w != lastW || h != lastH {
				handleResize(s, w, h)
			}
			if s.CurrentPhase() == game.PhasePlaying {
				s.Tick()
			}
			redraw()
			mu.Unlock()
		}
	}
}

// --- Board (22 cols wide, 22 rows tall) ---

func boardRow(rs game.RenderState, row int, activeFP, ghostFP map[game.Position]bool, activeKind game.Kind) string {
	if row == 0 {
		return colChrome + "┌" + strings.Repeat("─", 20) + "┐" + reset
	}
	if row == 21 {
		return colChrome + "└" + strings.Repeat("─", 20) + "┘" + reset
	}

	cellRow := row - 1 // board row 0..19
	var sb strings.Builder
	sb.WriteString(colChrome + "│" + reset)
	for col := 0; col < game.BoardCols; col++ {
		pos := game.Position{Col: col, Row: cellRow}
		sb.WriteString(cellStr(rs, pos, activeFP, ghostFP, activeKind))
	}
	sb.WriteString(colChrome + "│" + reset)
	return sb.String()

// Frame is a rendered terminal frame (one string per terminal row).
type Frame struct {
	rows []string
}

// Render computes a complete Frame for the given render state.
func Render(rs game.RenderState, termW, termH int) Frame {
	const (
		assemblyW = 40 // 8 + 1 + 22 + 1 + 8
		assemblyH = 22 // 20 cells + 2 border rows
	)

	lines := make([]string, assemblyH)
	buildAssembly(rs, lines)

	leftPad := 0
	if termW > assemblyW {
		leftPad = (termW - assemblyW) / 2
	}
	topPad := 0
	if termH > assemblyH {
		topPad = (termH - assemblyH) / 2
	}

	padStr := strings.Repeat(" ", leftPad)
	out := make([]string, termH)
	for i := range out {
		row := i - topPad
		if row >= 0 && row < assemblyH {
			out[i] = padStr + lines[row]
		} else {
			out[i] = padStr + strings.Repeat(" ", assemblyW)
		}
	}
	return Frame{rows: out}
}

// String serializes the frame for terminal output.
func (f Frame) String() string {

// stripANSI removes ANSI CSI escape sequences (ESC [ ... letter) from s.
func stripANSI(s string) string {
	var out strings.Builder
	for i := 0; i < len(s); {
		if isANSIEscapeStart(s, i) {
			i = skipANSIEscape(s, i+2)
		} else {
			out.WriteByte(s[i])
			i++
		}
	}
	return out.String()
}

func isANSIEscapeStart(s string, i int) bool {
	return s[i] == '\x1b' && i+1 < len(s) && s[i+1] == '['
}

func skipANSIEscape(s string, i int) int {
	for i < len(s) && !isANSITerminator(s[i]) {
		i++
	}
	if i < len(s) {
		i++
	}
	return i
}

func isANSITerminator(b byte) bool {
	return b >= 'A' && b <= 'z'
}

func ReadKey() Key {
	buf := make([]byte, 4)
	n, err := os.Stdin.Read(buf)

	}

// parseKey converts raw bytes from a single terminal read into a Key.
// n is the number of bytes read into buf.
func parseKey(buf []byte, n int) Key {
	if n >= 3 && buf[0] == 27 && buf[1] == '[' {
		return parseArrowKey(buf[2])
	}
	if n == 1 {
		return parseSingleByte(buf[0])
	}
	return KeyUnknown
}

func parseArrowKey(code byte) Key {
	switch code {
	case 'A':
		return KeyUp
	case 'B':
		return KeyDown
	case 'C':
		return KeyRight
	case 'D':
		return KeyLeft
	}
	return KeyUnknown
}

func parseSingleByte(b byte) Key {
	if k := parseControlByte(b); k != KeyUnknown {
		return k
	}
	return parseLetterByte(b)
}

func parseControlByte(b byte) Key {
	switch b {
	case 3:
		return KeyCtrlC
	case 27:
		return KeyEsc
	case 13, 10:
		return KeyEnter
	case 32:
		return KeySpace
	}
	return KeyUnknown
}

func parseLetterByte(b byte) Key {
	switch b {
	case 'r', 'R':
		return KeyR
	case 'x', 'X':
		return KeyX
	case 'z', 'Z':
		return KeyZ
	case 'c', 'C':
		return KeyC
	case 'p', 'P':
		return KeyP
	}
