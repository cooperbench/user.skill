func boardRow(rs game.RenderState, row int, activeFP, ghostFP map[game.Position]bool, activeKind game.Kind) string {
	if row == 0 {
		return colChrome + "┌" + strings.Repeat("─", 20) + "┐" + reset
	}
	if row == 21 {
		return colChrome + "└" + strings.Repeat("─", 20) + "┘" + reset
	}

	cellRow := row - 1 // board row 0..19
	var sb strings.Builder
	sb.WriteString(colChrome + "│" + reset)
	for col := 0; col < game.BoardCols; col++ {
		pos := game.Position{Col: col, Row: cellRow}
		sb.WriteString(cellStr(rs, pos, activeFP, ghostFP, activeKind))
	}
	sb.WriteString(colChrome + "│" + reset)
	return sb.String()
}

// cellStr returns the 2-char styled string for one board cell.
func cellStr(rs game.RenderState, pos game.Position, activeFP, ghostFP map[game.Position]bool, activeKind game.Kind) string {
	switch rs.Phase {
	case game.PhasePaused:
		return pausedCell(pos)
	case game.PhaseWaiting:
		return overlayCell(pos, waitingOverlayGrid())
	case game.PhaseGameOver:
		return gameOverCell(rs, pos)
	default:
		return playingCell(rs, pos, activeFP, ghostFP, activeKind)
	}
}

func pausedCell(pos game.Position) string {
	if pos.Row == 10 {
		pausedOverlay := pausedOverlayGrid()
		cell := pausedOverlay[pos.Row][pos.Col]
		if cell != "" {
			return cell
		}
	}
	return colFgDim + colBg + "░ " + reset
}

func gameOverCell(rs game.RenderState, pos game.Position) string {
	ov := gameOverOverlayGrid(rs.Score)
	if cell := overlayLookup(ov, pos); cell != "" {
		return cell
	}
	return frozenBoardCell(rs, pos)
}

func overlayLookup(g overlayGrid, pos game.Position) string {
	if pos.Row < 0 || pos.Row >= 20 || pos.Col < 0 || pos.Col >= 10 {
		return ""

}

	k := rs.Board[pos.Row][pos.Col]

	}
	return colFgDim + colBg + "· " + reset
}

// Frame is a rendered terminal frame (one string per terminal row).
type Frame struct {
	rows []string
}

// Render computes a complete Frame for the given render state.
func Render(rs game.RenderState, termW, termH int) Frame {
	const (
		assemblyW = 40 // 8 + 1 + 22 + 1 + 8
		assemblyH = 22 // 20 cells + 2 border rows
	)

	lines := make([]string, assemblyH)
	buildAssembly(rs, lines)

	leftPad := 0
	if termW > assemblyW {
		leftPad = (termW - assemblyW) / 2
	}
	topPad := 0
	if termH > assemblyH {
		topPad = (termH - assemblyH) / 2
	}

	padStr := strings.Repeat(" ", leftPad)
	out := make([]string, termH)
	for i := range out {
		row := i - topPad
		if row >= 0 && row < assemblyH {
			out[i] = padStr + lines[row]
		} else {
			out[i] = padStr + strings.Repeat(" ", assemblyW)
		}
	}
	return Frame{rows: out}
}

// RenderTooSmall returns a frame for terminals smaller than 40×24.
func RenderTooSmall(termW, termH int) Frame {
	msg := "RESIZE TERMINAL"
	out := make([]string, termH)
	for i := range out {
		if i == termH/2 {
			pad := (termW - len(msg)) / 2
			if pad < 0 {
				pad = 0
			}
			out[i] = strings.Repeat(" ", pad) + colFgPrim + msg + reset
		} else {
			out[i] = ""
		}
	}
	return Frame{rows: out}
}

// String serializes the frame for terminal output.
func (f Frame) String() string {
	var sb strings.Builder
	sb.WriteString("\x1b[H")
	sb.WriteString(colBg)
	for _, row := range f.rows {
		sb.WriteString(row)
		sb.WriteString(reset + "\r\n")
	}
	return sb.String()
}

// PlainRows returns frame rows stripped of ANSI escapes, for snapshot tests.
func (f Frame) PlainRows() []string {
	out := make([]string, len(f.rows))
	for i, r := range f.rows {
		out[i] = stripANSI(r)
	}
	return out
}

// stripANSI removes ANSI CSI escape sequences (ESC [ ... letter) from s.
func stripANSI(s string) string {
	var out strings.Builder
	for i := 0; i < len(s); {
		if isANSIEscapeStart(s, i) {
			i = skipANSIEscape(s, i+2)
		} else {
			out.WriteByte(s[i])
			i++
		}
	}
	return out.String()
}

func isANSIEscapeStart(s string, i int) bool {
	return s[i] == '\x1b' && i+1 < len(s) && s[i+1] == '['
}

func skipANSIEscape(s string, i int) int {
	for i < len(s) && !isANSITerminator(s[i]) {
		i++
	}
	if i < len(s) {
		i++
	}
	return i
}

func isANSITerminator(b byte) bool {
	return b >= 'A' && b <= 'z'
}

// Column layout: hold(8) + gap(1) + board(22) + gap(1) + right(8) = 40
func buildAssembly(rs game.RenderState, lines []string) {
	activeFP := map[game.Position]bool{}
	ghostFP := map[game.Position]bool{}
	var activePieceKind game.Kind
	if rs.Active != nil {
		activePieceKind = rs.Active.Kind
		for _, p := range rs.Active.Footprint {
			activeFP[p] = true
		}
		for _, p := range rs.Active.Ghost {
			ghostFP[p] = true
		}

		left := holdRow(rs, row)
		center := boardRow(rs, row, activeFP, ghostFP, activePieceKind)
		right := rightRow(rs, row)
		lines[row] = left + " " + center + " " + right
	}
}

import "os"

// Key represents a single parsed input event.
type Key int

const (
	KeyUnknown Key = iota
	KeyEnter
	KeyLeft
	KeyRight
	KeyUp
	KeyDown
	KeySpace
	KeyR
	KeyX
	KeyZ
	KeyC
	KeyP
	KeyEsc
	KeyCtrlC
)

// ReadKey reads one key event from stdin (blocking).
func ReadKey() Key {
	buf := make([]byte, 4)
	n, err := os.Stdin.Read(buf)
	if err != nil || n == 0 {
		return KeyUnknown
	}
	return parseKey(buf, n)
}

// parseKey converts raw bytes from a single terminal read into a Key.
// n is the number of bytes read into buf.
func parseKey(buf []byte, n int) Key {
	if n >= 3 && buf[0] == 27 && buf[1] == '[' {
		return parseArrowKey(buf[2])
	}
	if n == 1 {
		return parseSingleByte(buf[0])
	}
	return KeyUnknown
}

func parseArrowKey(code byte) Key {
	switch code {
	case 'A':
		return KeyUp
	case 'B':
		return KeyDown
	case 'C':
		return KeyRight
	case 'D':
		return KeyLeft
	}
	return KeyUnknown
}

func parseSingleByte(b byte) Key {
	if k := parseControlByte(b); k != KeyUnknown {
		return k
	}
	return parseLetterByte(b)
}

func parseControlByte(b byte) Key {
	switch b {
	case 3:
		return KeyCtrlC
	case 27:
		return KeyEsc
	case 13, 10:
		return KeyEnter
	case 32:
		return KeySpace
	}
	return KeyUnknown
}

func parseLetterByte(b byte) Key {
	switch b {
	case 'r', 'R':
		return KeyR
	case 'x', 'X':
		return KeyX
	case 'z', 'Z':
		return KeyZ
