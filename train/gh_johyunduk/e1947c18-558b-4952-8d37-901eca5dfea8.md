## RLS 정책 추가 경로 (2025 최신)

CREATE POLICY "Users can insert own row"
ON user_skor

   418→### 가챠 로직 (`pullTen` 상세)
   419→
   420→```typescript
   421→function pullTen(): GachaResult[] {
   422→  const results: GachaResult[] = [];
   423→  let hasRareOrAbove = false;
   424→
   425→  for (let i = 0; i < 10; i++) {
   426→    const result = pullOnce();
   427→    results.push(result);
   428→    if (result.rarity !== 'normal') hasRareOrAbove = true;
   429→  }
   430→
   431→  // 10연차 Rare 이상 보장: 10개 모두 Normal이면 마지막을 Rare로 교체
   432→  if (!hasRareOrAbove) {
   433→    results[9] = pullWithMinRarity('rare');
   434→  }
   435→
   436→  return results;
   437→}
   438→
   439→function pullOnce(): GachaResult {
   440→  const srPity = getSRPity();
   441→
   442→  // 천장 체크: 100회 도달 시 SR 이상 확정
   443→  let rarity: 'normal' | 'rare' | 'super_rare' | 'ultra_rare';
   444→  if (srPity >= 99) {
   445→    // SR 이상 확률: SR 80%, UR 20%
   446→    rarity = Math.random() < 0.2 ? 'ultra_rare' : 'super_rare';
   447→  } else {
   448→    const rand = Math.random() * 100;
   449→    if (rand < 3) rarity = 'ultra_rare';       // 3%
   450→    else if (rand < 15) rarity = 'super_rare';  // 12%
   451→    else if (rand < 40) rarity = 'rare';         // 25%
   452→    else rarity = 'normal';                      // 60%
   453→  }
   454→
   455→  // 해당 등급 캐릭터 중 랜덤 선택
   456→  const pool = CHARACTERS.filter(c => c.rarity === rarity);
   457→  const selected = pool[Math.floor(Math.random() * pool.length)];
   458→
   459→  // 천장 카운터 갱신
   460→  if (rarity === 'super_rare' || rarity === 'ultra_rare') {
   461→    resetSRPity();
   462→  } else {
   463→    incrementSRPity();
   464→  }
   465→
   466→  // 중복 처리
   467→  const isDuplicate = hasCharacter(selected.type);
   468→  if (!isDuplicate) {
   469→    unlockCharacter(selected.type);
   470→  }
   471→  const refundCoins = isDuplicate ? REFUND_TABLE[rarity] : 0;
   472→  if (refundCoins > 0) addCoins(refundCoins);
   473→
   474→  return { character: selected.type, rarity, isDuplicate, refundCoins };
   475→}
