377	    const visible = api.poops.getChildren().filter(
378	      c => c.active && (c as Phaser.Physics.Arcade.Sprite).y > -10,
379	    ) as Phaser.Physics.Arcade.Sprite[];
380	
381	    if (visible.length === 0) return;
382	
383	    // Partial Fisher-Yates: count개만 앞으로 뽑아내고 나머지 순회 생략
384	    const pool = visible.slice();
385	    const take = Math.min(count, pool.length);
386	    for (let i = 0; i < take; i++) {
387	      const j = i + Math.floor(Math.random() * (pool.length - i));
388	      [pool[i], pool[j]] = [pool[j], pool[i]];
389	    }
390	    const targets = pool.slice(0, take);
391	
392	    this.drawFireTrail(
393	      api.scene,
394	      api.player.x, api.player.y,
395	      targets.map(p => ({ x: p.x, y: p.y })),
396	    );
397	
398	    for (const poop of targets) {
399	      if (!poop.active) continue;
400	      this.playBurnEffect(api.scene, poop.x, poop.y);
401	      (poop as PoolablePoopBase).recycle();
402	      api.addAbilityBonus(10);
403	    }
