476	# shorter than the full crown-to-chin head height the specs measure. This factor

485	def detect_face_and_eyes(image_path: Path) -> dict | None:
486	    """Run the frontal-face and eye cascades once on an image.
487	
488	    Returns the geometric facts the per-profile checks need, or None if the
489	    image could not be read. The face box height and centre are measured against
490	    the detected-face bounding box (brow-to-chin), which understates the full
491	    crown-to-chin head height; the head band is the target after cropping, so a
492	    raw, uncropped portrait is expected to fall short of it here.
493	    """
494	    import cv2
495	
496	    img = cv2.imread(str(image_path))
497	    if img is None:
498	        return None
499	
500	    h, w = img.shape[:2]
501	    scale = DETECT_MAX_DIM / max(h, w)
502	    work = cv2.resize(img, (round(w * scale), round(h * scale))) if scale < 1 else img
503	    gray_plain = cv2.cvtColor(work, cv2.COLOR_BGR2GRAY)
504	    # Histogram equalisation makes frontal-face detection robust to lighting, but
505	    # it suppresses the eye cascade — so faces use the equalised image, eyes the plain one.
506	    gray = cv2.equalizeHist(gray_plain)
507	    wh, ww = gray.shape[:2]
508	
509	    face_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + "haarcascade_frontalface_default.xml")
510	    min_side = int(min(ww, wh) * FACE_MIN_FRAC)
511	    faces = face_cascade.detectMultiScale(
512	        gray, scaleFactor=1.1, minNeighbors=5, minSize=(min_side, min_side)
513	    )
514	    faces = sorted(faces, key=lambda f: f[2] * f[3], reverse=True)
515	
516	    facts: dict = {"n_faces": len(faces)}
517	    if not faces:
518	        return facts
519	
520	    fx, fy, fw, fh = faces[0]
521	    facts["height_frac"] = fh / wh
522	    facts["centre_x_frac"] = (fx + fw / 2) / ww
523	    facts["face_box_frac"] = (fx / ww, fy / wh, fw / ww, fh / wh)
524	
525	    # Eyes are expected in the upper half of an upright frontal face; restricting
526	    # the search there rejects nostril/mouth false positives and confirms orientation.
527	    eye_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + "haarcascade_eye.xml")
528	    roi = gray_plain[fy:fy + fh // 2, fx:fx + fw]
529	    eyes = eye_cascade.detectMultiScale(roi, scaleFactor=1.1, minNeighbors=5)
530	    facts["n_eyes_upper"] = len(eyes)
531	    return facts

585	def crop_to_profile(image_path: Path, profile: dict, face_facts: dict, out_path: Path) -> bool:
586	    """Crop and scale source image to profile fix dimensions with head in the target band.
587	
588	    Returns True on success. The crop centres the face horizontally and places the
589	    estimated crown with balanced top/bottom margins. Regions outside the source
590	    image are padded with white.
591	    """
592	    import cv2
593	    import numpy as np
594	
595	    if "face_box_frac" not in face_facts:
596	        return False

605	    # Face box in source pixels

624	    crop_h = head_h_src / head_frac
625	    crop_w = crop_h * out_w / out_h  # maintain output aspect ratio
626	
627	    # Balanced vertical margin: crown at (1 - head_frac) / 2 from top of crop
628	    top_margin = (1.0 - head_frac) / 2.0 * crop_h

632	    # Convert to integer pixel coordinates

634	    cy1 = int(round(crop_y1))

641	    sy1 = max(0, cy1)

646	        dy1 = sy1 - cy1

109	def make_contrast_jpeg(tmp_path: Path, width: int, height: int,
110	                       border_gray: int, face_gray: int,
111	                       name: str | None = None) -> Path:
112	    """Create JPEG with uniform border (border_gray) and a center rectangle (face_gray)."""
113	    path = tmp_path / (name or f"contrast_b{border_gray}_f{face_gray}.jpg")
114	    arr = np.full((height, width, 3), border_gray, dtype=np.uint8)
115	    fy1, fy2 = int(height * 0.3), int(height * 0.7)
116	    fx1, fx2 = int(width * 0.3), int(width * 0.7)
117	    arr[fy1:fy2, fx1:fx2] = face_gray
118	    cv2.imwrite(str(path), arr, [cv2.IMWRITE_JPEG_QUALITY, 95])
119	    return path

### Context
- F2: JPEG quality iteration so visa fixed file lands in 70 KB–3.5 MB
- F3: Re-run all checks on the produced fixed file

### Correctness ✓

**`_save_jpeg_within_size`:** Binary search correctly finds the *highest* quality ≤ max_kb by updating `best_data` on every in-bound step and moving `lo` up. The fast path at quality 95 avoids 7 extra encodes in the common case. One edge case to verify: when `size_kb > max_kb` after the search but `best_data` was found (e.g. quality 1 gives 60 KB — below min_kb 70 KB for visa) — the function falls back to quality 95 (which is >max_kb). A2 will then fail and report the actual size. This is correct behavior — the file is saved and the check reports the truth.

**F3 re-check:** The recursive `run_checks(fixed_path, [profile_name], fix=False, out=None)` is sound. `fix=False` prevents infinite recursion. Face detection re-runs on the fixed file — correct, since dimensions changed.

**B1/B2/B3 guidance:** Added to fail messages only, not to the contrast check (which is also B3). The contrast check already has descriptive text ("face blends into background") — no guidance needed there since it's a composition issue, not a camera defect.

**Issue:** The `test_caps_at_max_kb` test sets `min_kb=0` and `max_kb = q95_kb * 0.5`. The binary search will find a quality whose size ≤ max_kb — correct. But the assertion `size_kb <= max_kb` passes even if `best_data` was not written (fallback to quality 95 > max_kb path). Let me check: since `min_kb=0`, the condition `best_data and best_size >= min_kb` is `best_data and best_size >= 0` which is true whenever `best_data` is non-empty. Since 200×200 random noise at quality 1 gives a non-trivial file, `best_data` will be set. So the assertion is valid.

**Nit:** `_enc` returns `(b"", 0.0)` on encode failure. If `ok=False`, `best_data = b""` and `best_data` is falsy — the fallback path fires. Correct.

### Readability ✓

The `_enc` closure is concise and named well. The binary search comment ("highest quality whose file size is ≤ max_kb") matches the implementation exactly. The guidance strings are actionable and human-readable.

**Nit:** `_save_jpeg_within_size` docstring says "Falls back to quality 95 if no quality in [1, 95] lands in the band" but quality 95 was already tried before the search and found to be > max_kb. A subtle reader might wonder whether the fallback could be a different quality. It's correct (quality 95 was `data`) but could be slightly clearer. Not blocking.

### Architecture ✓

The helper is placed between `_fix_output_path` and `crop_to_profile` — logical ordering. Profile-driven (uses `file_size_min_kb`/`file_size_max_kb`) so it works correctly for both profiles without a visa-specific branch. Clean.

The F3 re-check as a tail call on `run_checks` is elegant. No new state, no new abstraction, just reuse of the existing public function. The `fixed: list[tuple[str, Path]]` collection is minimal.

### Security ✓

No user input, no shell commands, no network calls. File paths come from `_fix_output_path` which is derived from the user-provided image path (already validated to exist before `run_checks` is called). No new surface.

### Performance ✓

Binary search over [1, 94] is at most 7 `cv2.imencode` calls. For the common case (quality 95 in band), it's 1 call. Each call allocates ~size_kb bytes — totaling ~7 × average_file_size for the binary search path. For a 1200×1600 visa image, that's at most ~10–15 MB of allocations across the search. Acceptable for a CLI tool.
