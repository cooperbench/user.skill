421	# resulting Haar box at ~58% of frame → 0.75 / 0.58 ≈ 1.30. Derived from a single
422	# reference face; crown position varies with hairstyle, so re-validate against
423	# more portraits if outputs frame the head too high or low.
424	HEAD_FACE_RATIO = 1.30
425

427	def detect_face_and_eyes(image_path: Path) -> dict | None:
428	    """Run the frontal-face and eye cascades once on an image.
429	
430	    Returns the geometric facts the per-profile checks need, or None if the
431	    image could not be read. The face box height and centre are measured against
432	    the detected-face bounding box (brow-to-chin), which understates the full
433	    crown-to-chin head height; the head band is the target after cropping, so a
434	    raw, uncropped portrait is expected to fall short of it here.
435	    """
436	    import cv2
437	
438	    img = cv2.imread(str(image_path))
439	    if img is None:
440	        return None
441	
442	    h, w = img.shape[:2]
443	    scale = DETECT_MAX_DIM / max(h, w)
444	    work = cv2.resize(img, (round(w * scale), round(h * scale))) if scale < 1 else img
445	    gray_plain = cv2.cvtColor(work, cv2.COLOR_BGR2GRAY)
446	    # Histogram equalisation makes frontal-face detection robust to lighting, but
447	    # it suppresses the eye cascade — so faces use the equalised image, eyes the plain one.
448	    gray = cv2.equalizeHist(gray_plain)
449	    wh, ww = gray.shape[:2]
450	
451	    face_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + "haarcascade_frontalface_default.xml")
452	    min_side = int(min(ww, wh) * FACE_MIN_FRAC)
453	    faces = face_cascade.detectMultiScale(
454	        gray, scaleFactor=1.1, minNeighbors=5, minSize=(min_side, min_side)
455	    )
456	    faces = sorted(faces, key=lambda f: f[2] * f[3], reverse=True)
457	
458	    facts: dict = {"n_faces": len(faces)}
459	    if not faces:
460	        return facts
461	
462	    fx, fy, fw, fh = faces[0]
463	    facts["height_frac"] = fh / wh
464	    facts["centre_x_frac"] = (fx + fw / 2) / ww
465	    facts["face_box_frac"] = (fx / ww, fy / wh, fw / ww, fh / wh)
466	
467	    # Eyes are expected in the upper half of an upright frontal face; restricting
468	    # the search there rejects nostril/mouth false positives and confirms orientation.
469	    eye_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + "haarcascade_eye.xml")
470	    roi = gray_plain[fy:fy + fh // 2, fx:fx + fw]
471	    eyes = eye_cascade.detectMultiScale(roi, scaleFactor=1.1, minNeighbors=5)
472	    facts["n_eyes_upper"] = len(eyes)
473	    return facts
474

476	def append_face_geometry(result: CheckResult, facts: dict | None, profile: dict) -> None:
477	    """Append the four Tier C face-geometry findings to a profile's result."""
478	    if facts is None:
479	        result.fail("C1 could not read image for face detection")
480	        return
481	
482	    n = facts["n_faces"]
483	    if n == 0:
484	        result.fail("C1 no frontal face detected — not a portrait (document scan, or face too small/turned)")
485	        return
486	    if n > 1:
487	        result.fail(f"C1 {n} faces detected — a passport photo must contain exactly one")
488	        return
489	    result.ok("C1 exactly one frontal face detected")
490	
491	    hf = facts["height_frac"]  # Haar face-box (brow-to-chin) as fraction of frame
492	    estimated_head_frac = hf * HEAD_FACE_RATIO  # crown-to-chin estimate
493	    lo, hi = profile["head_height_min_frac"], profile["head_height_max_frac"]
494	    if lo <= estimated_head_frac <= hi:
495	        result.ok(f"C2 estimated head height {estimated_head_frac:.0%} of frame (target {lo:.0%}–{hi:.0%})")
496	    else:
497	        result.fail(
498	            f"C2 estimated head height {estimated_head_frac:.0%} of frame, "
499	            f"outside target {lo:.0%}–{hi:.0%} — crop/scale needed"
500	        )
501	
502	    cx = facts["centre_x_frac"]
503	    offset = abs(cx - 0.5)
504	    if offset <= CENTRE_TOLERANCE:
505	        result.ok(f"C3 face horizontally centred (centre at {cx:.0%} of width)")
506	    else:
507	        result.fail(f"C3 face off-centre (centre at {cx:.0%} of width, must be 50% ±{CENTRE_TOLERANCE:.0%})")
508	
509	    eyes = facts["n_eyes_upper"]
510	    if eyes >= 2:
511	        result.ok(f"C4 {eyes} eyes detected in upper half of face")
512	    elif eyes == 1:
513	        result.warn("C4 only one eye detected in upper half of face — head may be tilted or partly obscured")
514	    else:
515	        result.fail("C4 no eyes detected in upper half of face — face may be obscured, tilted, or a false detection")

527	def crop_to_profile(image_path: Path, profile: dict, face_facts: dict, out_path: Path) -> bool:
528	    """Crop and scale source image to profile fix dimensions with head in the target band.
529	
530	    Returns True on success. The crop centres the face horizontally and places the
531	    estimated crown with balanced top/bottom margins. Regions outside the source
532	    image are padded with white.
533	    """
534	    import cv2
535	    import numpy as np
536	
537	    if "face_box_frac" not in face_facts:
538	        return False
539	
540	    img = cv2.imread(str(image_path))
541	    if img is None:
542	        return False
543	
544	    src_h, src_w = img.shape[:2]
545	    xf, yf, wf, hf = face_facts["face_box_frac"]
546	
547	    # Face box in source pixels
548	    fx = xf * src_w
549	    fy = yf * src_h
550	    fw = wf * src_w
551	    fh = hf * src_h
552	
553	    # Estimated head bounds using the calibration factor
554	    crown_y = fy - (HEAD_FACE_RATIO - 1.0) * fh   # crown above the face-box top
555	    chin_y = fy + fh                                 # chin at the face-box bottom
556	    head_h_src = chin_y - crown_y                    # = fh * HEAD_FACE_RATIO
557	    face_cx = fx + fw / 2.0
558	
559	    # Target: head at the centre of the 70–80% band
560	    head_frac = (profile["head_height_min_frac"] + profile["head_height_max_frac"]) / 2.0
561	    out_w = profile["fix_width_px"]
562	    out_h = profile["fix_height_px"]
563	
564	    # Crop region in source pixels: scale so head_h_src maps to head_frac * out_h
565	    # crop_h / out_h = head_h_src / (head_frac * out_h)  →  crop_h = head_h_src / head_frac
566	    crop_h = head_h_src / head_frac
567	    crop_w = crop_h * out_w / out_h  # maintain output aspect ratio
568	
569	    # Balanced vertical margin: crown at (1 - head_frac) / 2 from top of crop
570	    top_margin = (1.0 - head_frac) / 2.0 * crop_h
571	    crop_y1 = crown_y - top_margin
572	    crop_x1 = face_cx - crop_w / 2.0
573	
574	    # Convert to integer pixel coordinates
575	    cx1 = int(round(crop_x1))
576	    cy1 = int(round(crop_y1))
577	    cw = int(round(crop_w))
578	    ch = int(round(crop_h))
579	
580	    # White canvas; copy the source region into the correct position
581	    canvas = np.full((ch, cw, 3), 255, dtype=np.uint8)
582	    sx1 = max(0, cx1)
583	    sy1 = max(0, cy1)
584	    sx2 = min(src_w, cx1 + cw)
585	    sy2 = min(src_h, cy1 + ch)
586	    if sx2 > sx1 and sy2 > sy1:
587	        dx1 = sx1 - cx1
588	        dy1 = sy1 - cy1
589	        canvas[dy1:dy1 + (sy2 - sy1), dx1:dx1 + (sx2 - sx1)] = img[sy1:sy2, sx1:sx2]
590	
591	    out_img = cv2.resize(canvas, (out_w, out_h), interpolation=cv2.INTER_LANCZOS4)
592	
593	    out_path.parent.mkdir(parents=True, exist_ok=True)
594	    ext = out_path.suffix.lower()
595	    if ext in (".jpg", ".jpeg"):
596	        cv2.imwrite(str(out_path), out_img, [cv2.IMWRITE_JPEG_QUALITY, 95])
597	    else:
598	        cv2.imwrite(str(out_path), out_img)
599	    return True
