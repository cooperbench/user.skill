117	async function compressImage(file: File, maxPx = 800, quality = 0.7): Promise<File> {
118	  return new Promise((resolve) => {
119	    const img = new Image()
120	    const blobUrl = URL.createObjectURL(file)
121	    img.onload = () => {
122	      URL.revokeObjectURL(blobUrl)
123	      const scale = Math.min(1, maxPx / Math.max(img.naturalWidth, img.naturalHeight))
124	      const canvas = document.createElement('canvas')
125	      canvas.width = Math.round(img.naturalWidth * scale)
126	      canvas.height = Math.round(img.naturalHeight * scale)
127	      const ctx = canvas.getContext('2d')
128	      if (!ctx) { resolve(file); return }
129	      ctx.drawImage(img, 0, 0, canvas.width, canvas.height)
130	      canvas.toBlob(
131	        (blob) => {
132	          if (!blob) { resolve(file); return }
133	          const name = file.name.replace(/\.[^.]+$/, '.jpg')
134	          resolve(new File([blob], name, { type: 'image/jpeg' }))
135	        },
136	        'image/jpeg',
137	        quality,
138	      )
139	    }
140	    img.onerror = () => { URL.revokeObjectURL(blobUrl); resolve(file) }
141	    img.src = blobUrl

849	function NotificationPermissionBanner() {

851	  const [dismissed, setDismissed] = useState(false)
852	
853	  useEffect(() => {
854	    if (typeof window === 'undefined' || !('Notification' in window)) {
855	      setPerm('unsupported')
856	      return
857	    }
858	    setPerm(Notification.permission as 'default' | 'granted' | 'denied')
859	    setDismissed(localStorage.getItem('yff_notif_banner_dismissed') === '1')
860	  }, [])
861	
862	  const handleEnable = async () => {
863	    if (typeof window === 'undefined' || !('Notification' in window)) return
864	    const result = await Notification.requestPermission()
865	    setPerm(result)
866	    if (result === 'granted') {
867	      try { new Notification('YourFamilyFarmer', { body: 'You will be alerted on every new order.' }) } catch {}
868	    }
869	  }
870	
871	  const handleDismiss = () => {
872	    localStorage.setItem('yff_notif_banner_dismissed', '1')
873	    setDismissed(true)
874	  }
875	
876	  if (perm === 'unsupported' || perm === 'granted') return null
877	  if (dismissed) return null
878	
879	  if (perm === 'denied') {
880	    return (
881	      <div className="bg-amber-50 border border-amber-200 rounded-2xl p-3 flex items-start gap-3">
882	        <span className="text-xl flex-shrink-0">🔕</span>

887	          <p className="text-[11px] text-amber-700 mt-0.5 leading-snug">
888	            Enable notifications in your browser settings to get alerted on new orders.
889	          </p>
890	        </div>
891	        <button onClick={handleDismiss} className="text-amber-700 text-xl leading-none px-1">×</button>
892	      </div>
893	    )
894	  }
895	
896	  return (
897	    <div className="bg-blue-50 border border-blue-200 rounded-2xl p-3 flex items-start gap-3">
898	      <span className="text-xl flex-shrink-0">🔔</span>
899	      <div className="flex-1 min-w-0">

911	            Enable / ఆన్ చేయండి

914	            onClick={handleDismiss}

> AGENT

**GMV = Gross Merchandise Value** — the total don-platform sales value of orders over a period, before subtracting anything (no deductions for refunds, delivery fees, commissions, or costs). It's the headline "how much business flowed through the marketplace" number.

In your app it's literally `SUM(orders.total_price)` over a time window — that's exactly what the moderator stats route computes (`gmvThisWeek` in `src/app/api/moderator/stats/route.ts:49`).

A few things worth knowing about how it's measured here:
- **It's gross.** Your current query sums *all* orders in the window, including `declined`/`cancelled` ones. So GMV ≠ money actually collected or earned. (If you ever want "net," I'd filter those out — I checked earlier and excluding declined/cancelled made no difference to your numbers, but it would once you have declines.)
- **It's not revenue.** Revenue would be your commission/margin on that GMV, not the full order value.
- **It's zoned** on the moderator dashboard — each moderator sees GMV only for farmers in their region.
