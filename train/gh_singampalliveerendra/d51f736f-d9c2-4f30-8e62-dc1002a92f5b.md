117	async function compressImage(file: File, maxPx = 800, quality = 0.7): Promise<File> {
118	  return new Promise((resolve) => {
119	    const img = new Image()
120	    const blobUrl = URL.createObjectURL(file)
121	    img.onload = () => {
122	      URL.revokeObjectURL(blobUrl)
123	      const scale = Math.min(1, maxPx / Math.max(img.naturalWidth, img.naturalHeight))
124	      const canvas = document.createElement('canvas')
125	      canvas.width = Math.round(img.naturalWidth * scale)
126	      canvas.height = Math.round(img.naturalHeight * scale)
127	      const ctx = canvas.getContext('2d')
128	      if (!ctx) { resolve(file); return }
129	      ctx.drawImage(img, 0, 0, canvas.width, canvas.height)
130	      canvas.toBlob(
131	        (blob) => {
132	          if (!blob) { resolve(file); return }
133	          const name = file.name.replace(/\.[^.]+$/, '.jpg')
134	          resolve(new File([blob], name, { type: 'image/jpeg' }))
135	        },
136	        'image/jpeg',
137	        quality,
138	      )
139	    }
140	    img.onerror = () => { URL.revokeObjectURL(blobUrl); resolve(file) }
141	    img.src = blobUrl
142	  })
143	}
