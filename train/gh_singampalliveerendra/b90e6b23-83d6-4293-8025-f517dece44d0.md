683	function NotificationPermissionBanner() {

685	  const [perm, setPerm] = useState<'unsupported' | 'default' | 'granted' | 'denied'>('unsupported')
686	  const [dismissed, setDismissed] = useState(false)
687

690	      setPerm('unsupported')
691	      return
692	    }
693	    setPerm(Notification.permission as 'default' | 'granted' | 'denied')
694	    setDismissed(localStorage.getItem('yff_notif_banner_dismissed') === '1')
695	  }, [])
696	
697	  const handleEnable = async () => {
698	    if (typeof window === 'undefined' || !('Notification' in window)) return
699	    const result = await Notification.requestPermission()
700	    setPerm(result)
701	    if (result === 'granted') {
702	      try { new Notification('YourFamilyFarmer', { body: 'You will be alerted on every new order.' }) } catch {}
703	    }
704	  }
705	
706	  const handleDismiss = () => {
707	    localStorage.setItem('yff_notif_banner_dismissed', '1')
708	    setDismissed(true)
709	  }
710	
711	  if (perm === 'unsupported' || perm === 'granted') return null
712	  if (dismissed) return null
713	
714	  if (perm === 'denied') {
715	    return (
716	      <div className="bg-amber-50 border border-amber-200 rounded-2xl p-3 flex items-start gap-3">
