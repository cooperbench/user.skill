935	function FarmImagesCard({ farmerId }: { farmerId: string }) {
936	  const [photos, setPhotos] = useState<MediaPhoto[]>([])
937	  const [loading, setLoading] = useState(true)
938	  const [uploading, setUploading] = useState(false)
939	  const [error, setError] = useState('')
940	  const [preview, setPreview] = useState<string | null>(null)
941	  const [pendingFile, setPendingFile] = useState<File | null>(null)
942	  const [caption, setCaption] = useState('')
943	
944	  const loadPhotos = useCallback(async () => {
945	    setLoading(true)
946	    const { data } = await supabase
947	      .from('media')
948	      .select('id, url, caption')
949	      .eq('farmer_id', farmerId)
950	      .eq('type', 'photo')
951	      .order('sort_order', { ascending: true })
952	    setPhotos((data ?? []) as MediaPhoto[])
953	    setLoading(false)
954	  }, [farmerId])
955	
956	  useEffect(() => { loadPhotos() }, [loadPhotos])
957	
958	  const handleFile = (file: File | null) => {
959	    setError('')
960	    if (!file) return
961	    if (!file.type.startsWith('image/')) {
962	      setError('Please pick an image file / చిత్రం ఫైల్ ఎంచుకోండి')
963	      return
964	    }
965	    if (file.size > 8 * 1024 * 1024) {
966	      setError('Image is too large (max 8 MB) / చిత్రం 8 MB కంటే ఎక్కువ')
967	      return
968	    }
969	    setPendingFile(file)
970	    const reader = new FileReader()
971	    reader.onload = (e) => setPreview(e.target?.result as string)
972	    reader.readAsDataURL(file)
973	  }
974	
975	  const handleUpload = async () => {
976	    if (!pendingFile) return
977	    setUploading(true)
978	    setError('')
979	    try {
980	      const ext = pendingFile.name.split('.').pop()?.toLowerCase() || 'jpg'
981	      const path = `${farmerId}/${Date.now()}-${Math.random().toString(36).slice(2, 8)}.${ext}`
982	      const { error: upErr } = await supabase.storage
983	        .from('farm-images')
984	        .upload(path, pendingFile, {
985	          cacheControl: '3600',
986	          contentType: pendingFile.type,
987	          upsert: false,
988	        })
989	      if (upErr) throw upErr
990	
991	      const { data: pub } = supabase.storage.from('farm-images').getPublicUrl(path)
992	      const publicUrl = pub.publicUrl
993	
994	      const { error: insErr } = await supabase.from('media').insert({
995	        farmer_id: farmerId,
996	        type: 'photo',
997	        url: publicUrl,
998	        caption: caption.trim() || null,
999	        sort_order: photos.length,
1000	      })
1001	      if (insErr) throw insErr
1002	
1003	      setPreview(null)
1004	      setPendingFile(null)
1005	      setCaption('')
1006	      await loadPhotos()
1007	    } catch (e) {
1008	      const msg = e instanceof Error ? e.message : 'Upload failed'
1009	      setError(msg)
1010	    } finally {
1011	      setUploading(false)
1012	    }
1013	  }
1014	
1015	  const handleCancel = () => {
1016	    setPreview(null)
1017	    setPendingFile(null)
1018	    setCaption('')
1019	    setError('')
1020	  }
1021	
1022	  return (
1023	    <div className="bg-white rounded-2xl border border-gray-100 p-4 space-y-3">
1024	      <div>
1025	        <h2 className="font-extrabold text-gray-900 text-base leading-tight">
1026	          Farm photos / పొలం ఫోటోలు
1027	        </h2>
1028	        <p className="text-xs text-gray-500 mt-0.5">
1029	          Show buyers your farm / కొనుగోలుదారులకు మీ పొలం చూపించండి
1030	        </p>
1031	      </div>
1032	
1033	      {!preview && (
1034	        <label className="block w-full border-2 border-dashed border-green-300 rounded-xl py-5 px-4 text-center cursor-pointer active:bg-green-50">
1035	          <div className="text-3xl mb-1">📸</div>
1036	          <p className="text-sm font-bold text-green-800">
1037	            Upload photo / ఫోటో అప్‌లోడ్
1038	          </p>
1039	          <p className="text-[11px] text-gray-500 mt-0.5">
1040	            Gallery or camera / గ్యాలరీ లేదా కెమెరా
1041	          </p>
1042	          <input
1043	            type="file"
1044	            accept="image/*"
1045	            capture="environment"
1046	            className="hidden"
1047	            onChange={(e) => handleFile(e.target.files?.[0] ?? null)}
1048	          />
1049	        </label>
1050	      )}
1051	
1052	      {preview && (
1053	        <div className="space-y-3">
1054	          <div className="rounded-xl overflow-hidden bg-gray-50 border border-gray-200">
1055	            {/* eslint-disable-next-line @next/next/no-img-element */}
1056	            <img src={preview} alt="Preview" className="w-full max-h-64 object-cover" />
1057	          </div>
1058	          <input
1059	            type="text"
1060	            placeholder="Caption (optional) / శీర్షిక"
1061	            value={caption}
1062	            onChange={(e) => setCaption(e.target.value)}
1063	            className="w-full border border-gray-200 rounded-xl px-4 py-3 text-sm focus:border-green-500 focus:outline-none"
1064	          />
1065	          {error && (
