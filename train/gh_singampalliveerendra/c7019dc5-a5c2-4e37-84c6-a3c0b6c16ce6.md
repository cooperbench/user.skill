474	  onClose,
475	  onSaved,
476	}: {

479	  onSaved: (updated: Farmer) => void

484	  const [district, setDistrict] = useState(farmer.district ?? '')

498

503	
504	  // Pesticide cert
505	  const [certFile, setCertFile] = useState<File | null>(null)

609	      .select('*')

629	              {tx.profileModalTitle}
