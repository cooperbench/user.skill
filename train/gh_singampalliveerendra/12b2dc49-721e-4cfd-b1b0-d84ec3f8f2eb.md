const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i

function safeEqual(a: string, b: string): boolean {
  if (a.length !== b.length) return false
  let diff = 0
  for (let i = 0; i < a.length; i++) diff |= a.charCodeAt(i) ^ b.charCodeAt(i)
  return diff === 0
}

// Farmer confirms they personally handed a home-delivery / courier order to the
// buyer at the door. Just like self-pickup (confirm-pickup) and the rider
// door-handover, the buyer reads the 4-digit code off their order page and the
// farmer types it here — the code is never sent to the farmer's browser, so it
// must come from the real customer. A correct code stamps received_at (and a
// shipped_at if it was never set), which resolves the order immediately; the
// buyer does NOT need to separately confirm "Received".
//
// This is only for orders the FARMER delivers himself. A rider-driven home
// delivery is closed by the rider at the door, not here.
export async function POST(req: NextRequest, ctx: { params: Promise<{ id: string }> }) {
  const session = getFarmerSessionFromRequest(req)
  if (!session) return NextResponse.json({ error: 'Please log in.' }, { status: 401 })

  const { id } = await ctx.params
  if (!id || !UUID_RE.test(id)) return NextResponse.json({ error: 'Invalid order id.' }, { status: 400 })

  // Anti-guessing: 6 attempts per farmer+order per 10 min (4-digit code).
  if (!rateLimit(`farmer-deliver-otp:${session.farmerId}:${id}`, 6, 10 * 60 * 1000)) {
    return NextResponse.json({ error: 'Too many wrong codes. Please check with the customer.' }, { status: 429 })
  }

  const body = await req.json().catch(() => null)
  const otp = String((body && (body as { otp?: unknown }).otp) ?? '').trim()
  if (!/^\d{4}$/.test(otp)) {
    return NextResponse.json({ error: 'Enter the 4-digit code from the customer.' }, { status: 400 })
  }

  const supabase = createClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.SUPABASE_SERVICE_ROLE_KEY!,
  )

  const { data: order, error: loadErr } = await supabase
    .from('orders')
    .select('id, farmer_id, status, delivery_type, delivery_status, delivery_boy_id, handover_otp, shipped_at, received_at')
    .eq('id', id)
    .maybeSingle()

  if (loadErr) return NextResponse.json({ error: loadErr.message }, { status: 500 })
  if (!order) return NextResponse.json({ error: 'Order not found.' }, { status: 404 })
  if (order.farmer_id !== session.farmerId) {
    return NextResponse.json({ error: 'Not your order.' }, { status: 403 })
  }
  // Self-pickup is closed via confirm-pickup, not here.
  if (order.delivery_type === 'self_pickup' || !order.delivery_type) {
    return NextResponse.json({ error: 'This is a pickup order — mark it Picked Up instead.' }, { status: 409 })
  }
  // A rider-driven home delivery is closed by the rider at the door.
  const riderAssigned =
    order.delivery_type === 'home_delivery'
    && (!!order.delivery_boy_id || (order.delivery_status != null && order.delivery_status !== 'unassigned'))
  if (riderAssigned) {
    return NextResponse.json({ error: 'A delivery agent is assigned — they confirm this at the door.' }, { status: 409 })
  }
  if (order.status !== 'approved') {
    return NextResponse.json({ error: 'Only an approved order can be marked delivered.' }, { status: 409 })
  }
  if (order.received_at) {
    return NextResponse.json({ error: 'This order is already marked delivered.' }, { status: 409 })
  }
  if (!order.handover_otp || !safeEqual(order.handover_otp, otp)) {
    return NextResponse.json({ error: 'Wrong code. Ask the customer to read it again.' }, { status: 401 })
  }

  const now = new Date().toISOString()
  const { data: updated, error: updErr } = await supabase
    .from('orders')
    .update({ received_at: now, ...(order.shipped_at ? {} : { shipped_at: now }) })
    .eq('id', id)
    .eq('farmer_id', session.farmerId)
    .eq('status', 'approved')
    .is('received_at', null)
    .select('id, received_at, shipped_at')

  if (updErr) {
    console.error('[YFF farmer/deliver] update failed:', updErr.message)
    return NextResponse.json({ error: 'Could not confirm delivery. Please try again.' }, { status: 500 })
  }
  if (!updated || updated.length === 0) {
    return NextResponse.json({ error: 'Could not confirm delivery. Refresh and try again.' }, { status: 409 })
  }

  return NextResponse.json({ ok: true, received_at: updated[0].received_at, shipped_at: updated[0].shipped_at })
}

Here's what's going on, in code terms.

## The current state machine

For a farmer-fulfilled order (self-pickup, courier, or home-delivery *without* a rider), the action button lives in `src/components/farmer/OrderCard.tsx:448-558`. The branches are:

```
!isApproved            → "Confirm pickup/delivery date" + Decline
isDelivery && rider    → Decline only (rider closes it at the door)
isShipped              → read-only "🚚 Shipped · Awaiting buyer confirmation"
else (approved, unshipped) → "✓ Delivered" → 4-digit code box → /deliver
```

So once the farmer taps **Confirm date**, the order becomes `approved` and the very next button is **✓ Delivered**. There is no "Shipped" action in between — exactly the client's complaint.

## Why "Shipped" is effectively missing

`isShipped` is just `!!order.shipped_at` (`OrderCard.tsx:104`). The only thing that ever sets `shipped_at` is the **deliver** route (`src/app/api/farmer/orders/[id]/deliver/route.ts`), and it sets `received_at` **and** `shipped_at` in the *same* update:

```ts
.update({ received_at: now, ...(order.shipped_at ? {} : { shipped_at: now }) })
```

Meanwhile the standalone **ship** route (`.../ship/route.ts`) is deliberately **disabled** — it returns `410 Gone` (that was part of the recent handover-code work; see the `[[project_handover_code_verification]]` note).

Net effect: `shipped_at` is never set *on its own*. By the time it's set, `received_at` is set too and the order is already complete and filed into history. So the `isShipped` branch (lines 485-497) is **dead code** — unreachable — and the machine really only has `Confirm date → Delivered`.

## What the fix involves (status-machine change)

Insert a real intermediate step, reconciled with the mandatory-code rule:

- **Shipped = trust-based dispatch marker** (no code). Farmer taps "🚚 Mark Shipped" → stamps `shipped_at` only, order stays `approved`.
- **Delivered = still requires the buyer's 4-digit code** (unchanged), stamps `received_at` and closes it.

That keeps the security model intact: dispatch is informational, but the order can still only be *closed* with the real customer's code.

Concrete edits:

1. **`.../ship/route.ts`** — re-enable it to stamp `shipped_at` only, guarded to `status='approved'`, non-rider, `shipped_at IS NULL`. (Keep the "this doesn't close the order" semantics.)
2. **`OrderCard.tsx:498-520`** — split the single "✓ Delivered" branch into two: when `!isShipped`, show **"🚚 Mark Shipped"** (calls `/ship`); when `isShipped && !received`, show the existing **"✓ Delivered"** + code box. The current dead `isShipped` "awaiting buyer" branch (485-497) gets replaced by the Delivered+code UI.
3. **Handlers** in `src/app/farmer/dashboard/orders/page.tsx` and `src/app/farmer/dashboard/page.tsx` — add an `onMarkShipped` handler with optimistic `shipped_at` update (mirrors the existing `handleMarkDelivered`).
