90	    const newQty = item.stockQty != null ? Math.min(rawQty, item.stockQty) : rawQty
91	    const merged = { ...item, qty: newQty }
92	    const { price } = getActiveTier(newQty, merged)
93	    next[item.listingId] = { ...merged, pricePerKg: price ?? item.pricePerKg }
94	    writeCart(next)
95	  }, [])
96	
97	  const setQty = useCallback((listingId: string, qty: number) => {
98	    const next = { ...readCart() }
99	    if (qty <= 0) delete next[listingId]
100	    else if (next[listingId]) {
101	      const existing = next[listingId]
102	      const cappedQty = existing.stockQty != null ? Math.min(qty, existing.stockQty) : qty
103	      const updated = { ...existing, qty: cappedQty }
104	      const { price } = getActiveTier(cappedQty, updated)
105	      next[listingId] = { ...updated, pricePerKg: price ?? updated.pricePerKg }
106	    }
107	    writeCart(next)
108	  }, [])
109	
110	  const removeItem = useCallback((listingId: string) => {
111	    const next = { ...readCart() }
112	    delete next[listingId]
113	    writeCart(next)
114	  }, [])
115	
116	  const clear = useCallback(() => writeCart({}), [])
117	
118	  const clearFarmer = useCallback((farmerId: string) => {
119	    const next = { ...readCart() }
120	    for (const key of Object.keys(next)) {
121	      if (next[key].farmerId === farmerId) delete next[key]
122	    }
123	    writeCart(next)
124	  }, [])
125	
126	  const items = Object.values(cart)
127	  const count = items.reduce((sum, i) => sum + i.qty, 0)
128	
129	  return { cart, items, count, addItem, setQty, removeItem, clear, clearFarmer }
130	}
