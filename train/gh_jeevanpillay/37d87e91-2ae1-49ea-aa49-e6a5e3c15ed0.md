   492→    // Resolve workspace: use arg if provided, otherwise auto-detect from cwd
   493→    let workspace;
   494→    let entry = if let Some(ws) = workspace_arg {
   495→        workspace = ws.to_string();
   496→        match st.resolve_workspace(ws) {
   497→            Some(e) => e.clone(),
   498→            None => {
   499→                error!("unknown workspace '{ws}'");
   500→                return 1;
   501→            }
   502→        }
   503→    } else {
   504→        match detect_workspace(&st) {
   505→            Some(e) => {
   506→                workspace = config::workspace_id(&e.repo, &e.branch);
   507→                info!("Auto-detected workspace: {workspace}");
   508→                e
   509→            }
   510→            None => {
   511→                error!("could not detect workspace from current directory");
   512→                info!("Usage: dual destroy [workspace]");
   513→                return 1;
   514→            }
   515→        }
   516→    };
   517→
   518→    let workspace_root = st.workspace_root();
   519→    let container_name = config::container_name(&entry.repo, &entry.branch);
   520→    let session_name = config::session_name(&entry.repo, &entry.branch);
   521→
   522→    // Destroy tmux session
   523→    if backend.is_alive(&session_name) {
   524→        info!("Destroying session {session_name}...");
   525→        if let Err(e) = backend.destroy(&session_name) {
   526→            warn!("session destroy failed: {e}");
   527→        }
   528→    }
   529→
   530→    // Stop and remove container
   531→    match container::status(&container_name) {
   532→        container::ContainerStatus::Running => {
   533→            info!("Stopping container {container_name}...");
   534→            if let Err(e) = container::stop(&container_name) {
   535→                warn!("container stop failed: {e}");
   536→            }
   537→            info!("Removing container {container_name}...");
   538→            if let Err(e) = container::destroy(&container_name) {
   539→                warn!("container remove failed: {e}");
   540→            }
   541→        }
   542→        container::ContainerStatus::Stopped => {
   543→            info!("Removing container {container_name}...");
   544→            if let Err(e) = container::destroy(&container_name) {
   545→                warn!("container remove failed: {e}");
   546→            }
   547→        }
   548→        container::ContainerStatus::Missing => {}
   549→    }
   550→
   551→    // Remove clone (only for non-explicit-path workspaces)
   552→    if entry.path.is_none() && clone::workspace_exists(&workspace_root, &entry.repo, &entry.branch)
   553→    {
   554→        info!("Removing clone...");
   555→        if let Err(e) = clone::remove_workspace(&workspace_root, &entry.repo, &entry.branch) {
   556→            error!("failed to remove clone: {e}");
   557→            return 1;
   558→        }
   559→    }
   560→
   561→    // Remove from state
   562→    st.remove_workspace(&entry.repo, &entry.branch);
   563→    if let Err(e) = state::save(&st) {
   564→        warn!("failed to save state: {e}");
   565→    }
   566→
   567→    info!("Workspace '{workspace}' destroyed.");
   568→    0
   569→}

   681→/// Sync shared config files for a workspace.
   682→fn cmd_sync(workspace_arg: Option<String>) -> i32 {
   683→    let st = match state::load() {
   684→        Ok(s) => s,
   685→        Err(e) => {
   686→            error!("{e}");
   687→            return 1;
   688→        }
   689→    };
   690→
   691→    // Resolve which workspace we're syncing
   692→    let entry = if let Some(ws) = workspace_arg {
   693→        match st.resolve_workspace(&ws) {
   694→            Some(e) => e.clone(),
   695→            None => {
   696→                error!("unknown workspace '{ws}'");
   697→                return 1;
   698→            }
   699→        }
   700→    } else {
   701→        match detect_workspace(&st) {
   702→            Some(e) => e,
   703→            None => {
   704→                error!("not inside a dual workspace");
   705→                info!("Usage: dual sync [workspace]");
   706→                return 1;
   707→            }
   708→        }
   709→    };
   710→
   711→    // Load hints
   712→    let workspace_dir = st.workspace_dir(&entry);
   713→    let hints = config::load_hints(&workspace_dir).unwrap_or_default();
   714→    let shared_config = match &hints.shared {
   715→        Some(s) if !s.files.is_empty() => s,
   716→        _ => {
   717→            error!("no [shared] section in .dual.toml (or files list is empty)");
   718→            return 1;
   719→        }
   720→    };
   721→
   722→    let shared_dir = match shared::ensure_shared_dir(&entry.repo) {
   723→        Ok(d) => d,
   724→        Err(e) => {
   725→            error!("{e}");
   726→            return 1;
   727→        }
   728→    };
   729→
   730→    let is_main = entry.path.is_some();
   731→
   732→    if is_main {
   733→        // Main workspace: init shared dir, then prompt to sync all branches
   734→        match shared::init_from_main(&workspace_dir, &shared_dir, &shared_config.files) {
   735→            Ok(moved) => {
   736→                for f in &moved {
   737→                    info!("  moved {f} → shared/");
   738→                }
   739→            }
   740→            Err(e) => {
   741→                error!("{e}");
   742→                return 1;
   743→            }
   744→        }
   745→
   746→        // Prompt to sync all branches
   747→        let branches: Vec<_> = st
   748→            .workspaces_for_repo(&entry.repo)
   749→            .into_iter()
   750→            .filter(|ws| ws.path.is_none())
   751→            .collect();
   752→
   753→        if branches.is_empty() {
   754→            info!("No branch workspaces to sync.");
   755→            return 0;
   756→        }
   757→
   758→        // Interactive prompt — use println! directly since this is user interaction
   759→        println!(
   760→            "\nSync shared files to ALL {} branch workspace(s)? [y/N]",
   761→            branches.len()
   762→        );
   763→        let mut input = String::new();
   764→        std::io::stdin().read_line(&mut input).unwrap_or(0);
   765→        if !input.trim().eq_ignore_ascii_case("y") {
   766→            info!("Cancelled.");
   767→            return 0;
   768→        }
   769→
   770→        for branch_entry in &branches {
   771→            let branch_dir = st.workspace_dir(branch_entry);
   772→            if !branch_dir.exists() {
   773→                continue; // Not yet cloned
   774→            }
   775→            let ws_id = config::workspace_id(&branch_entry.repo, &branch_entry.branch);
   776→            match shared::copy_to_branch(&branch_dir, &shared_dir, &shared_config.files) {
   777→                Ok(copied) => {
   778→                    info!("{ws_id}: synced {} file(s)", copied.len());
   779→                }
   780→                Err(e) => error!("{ws_id}: {e}"),
   781→            }
   782→        }
   783→    } else {
   784→        // Branch workspace: copy from shared dir
   785→        match shared::copy_to_branch(&workspace_dir, &shared_dir, &shared_config.files) {
   786→            Ok(copied) => {
   787→                if copied.is_empty() {
   788→                    info!(
   789→                        "No shared files available yet. Run `dual sync` in the main workspace first."
   790→                    );
   791→                } else {
   792→                    for f in &copied {
   793→                        info!("  synced {f}");
   794→                    }
   795→                }
   796→            }
   797→            Err(e) => {
   798→                error!("{e}");
   799→                return 1;
   800→            }
   801→        }
   802→    }
   803→
   804→    0
   805→}

   807→/// Detect the repo name from the current working directory.
   808→///
   809→/// Matches the git remote URL of the cwd against known workspace URLs in state.
   810→fn detect_repo_from_cwd(st: &state::WorkspaceState) -> Option<String> {
   811→    let (_, url, _) = detect_git_repo().ok()?;
   812→
   813→    // Match against known workspace URLs
   814→    for ws in st.all_workspaces() {
   815→        if ws.url == url {
   816→            return Some(ws.repo.clone());
   817→        }
   818→    }
   819→
   820→    // Also try matching by git root path against workspace paths
   821→    let cwd = std::env::current_dir().ok()?;
   822→    let root = std::process::Command::new("git")
   823→        .args(["rev-parse", "--show-toplevel"])
   824→        .output()
   825→        .ok()
   826→        .filter(|o| o.status.success())
   827→        .map(|o| PathBuf::from(String::from_utf8_lossy(&o.stdout).trim().to_string()))?;
   828→
   829→    for ws in st.all_workspaces() {
   830→        let ws_dir = st.workspace_dir(ws);
   831→        if ws_dir == root || ws_dir == cwd {
   832→            return Some(ws.repo.clone());
   833→        }
   834→    }
   835→
   836→    None
   837→}

   839→/// Detect which workspace the current directory belongs to.
   840→fn detect_workspace(st: &state::WorkspaceState) -> Option<state::WorkspaceEntry> {
   841→    let cwd = std::env::current_dir().ok()?;
   842→
   843→    // Try git root first (handles being in subdirectories)
   844→    let root = std::process::Command::new("git")
   845→        .args(["rev-parse", "--show-toplevel"])
   846→        .output()
   847→        .ok()
   848→        .filter(|o| o.status.success())
   849→        .map(|o| PathBuf::from(String::from_utf8_lossy(&o.stdout).trim().to_string()))
   850→        .unwrap_or(cwd);
   851→
   852→    for ws in st.all_workspaces() {
   853→        let ws_dir = st.workspace_dir(ws);
   854→        if ws_dir == root {
   855→            return Some(ws.clone());
   856→        }
   857→    }
   858→    None
   859→}

   861→/// Print workspace status grouped by repo.
   862→fn print_workspace_status(st: &state::WorkspaceState, backend: &dyn MultiplexerBackend) {
   863→    let workspace_root = st.workspace_root();
   864→
   865→    // Collect unique repo names in order of first appearance
   866→    let mut repos: Vec<String> = Vec::new();
   867→    for ws in st.all_workspaces() {
   868→        if !repos.contains(&ws.repo) {
   869→            repos.push(ws.repo.clone());
   870→        }
   871→    }
   872→
   873→    for repo in &repos {
   874→        println!("{repo}");
   875→        for ws in st.workspaces_for_repo(repo) {
   876→            let container_name = config::container_name(&ws.repo, &ws.branch);
   877→            let session_name = config::session_name(&ws.repo, &ws.branch);
   878→
   879→            let clone_exists = if ws.path.is_some() {
   880→                ws.path
   881→                    .as_ref()
   882→                    .map(|p| PathBuf::from(p).join(".git").exists())
   883→                    .unwrap_or(false)
   884→            } else {
   885→                clone::workspace_exists(&workspace_root, &ws.repo, &ws.branch)
   886→            };
   887→            let container_st = container::status(&container_name);
   888→            let tmux_alive = backend.is_alive(&session_name);
   889→
   890→            let (icon, status_text) = match (&container_st, tmux_alive) {
   891→                (container::ContainerStatus::Running, true) => {
   892→                    ("\u{25cf}", "running  (container: up, tmux: attached)")
   893→                }
   894→                (container::ContainerStatus::Running, false) => {
   895→                    ("\u{25cf}", "running  (container: up, tmux: none)")
   896→                }
   897→                (container::ContainerStatus::Stopped, true) => (
   898→                    "\u{25cb}",
   899→                    "stopped  (container: stopped, tmux: background)",
   900→                ),
   901→                (container::ContainerStatus::Stopped, false) => {
   902→                    ("\u{25cb}", "stopped  (container: stopped, tmux: none)")
   903→                }
   904→                (container::ContainerStatus::Missing, _) if clone_exists => {
   905→                    ("\u{25cb}", "stopped  (not launched)")
   906→                }
   907→                (container::ContainerStatus::Missing, _) => {
   908→                    ("\u{25cc}", "lazy     (not cloned yet)")
   909→                }
   910→            };
   911→
   912→            let branch_display = config::decode_branch(&config::encode_branch(&ws.branch));
   913→            println!("  {branch_display:<24} {icon} {status_text}");
   914→        }
   915→        println!();
   916→    }
   917→}

   919→/// Detect git repo info from the current directory.
   920→fn detect_git_repo() -> Result<(PathBuf, String, String), String> {
   921→    // Get repo root
   922→    let root_output = std::process::Command::new("git")
   923→        .args(["rev-parse", "--show-toplevel"])
   924→        .output()
   925→        .map_err(|_| "git not found".to_string())?;
   926→    if !root_output.status.success() {
   927→        return Err("not inside a git repository".to_string());
   928→    }
   929→    let root = PathBuf::from(String::from_utf8_lossy(&root_output.stdout).trim());
   930→
   931→    // Get remote URL
   932→    let url_output = std::process::Command::new("git")
   933→        .args(["remote", "get-url", "origin"])
   934→        .output()
   935→        .map_err(|_| "git not found".to_string())?;
   936→    let url = if url_output.status.success() {
   937→        String::from_utf8_lossy(&url_output.stdout)
   938→            .trim()
   939→            .to_string()
   940→    } else {
   941→        root.to_string_lossy().to_string() // local-only repo, use path as URL
   942→    };
   943→
   944→    // Get current branch
   945→    let branch_output = std::process::Command::new("git")
   946→        .args(["rev-parse", "--abbrev-ref", "HEAD"])
   947→        .output()
   948→        .map_err(|_| "git not found".to_string())?;
   949→    let branch = String::from_utf8_lossy(&branch_output.stdout)
   950→        .trim()
   951→        .to_string();
   952→
   953→    Ok((root, url, branch))
   954→}
