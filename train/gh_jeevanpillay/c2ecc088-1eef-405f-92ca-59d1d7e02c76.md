# devcontainer.json as Primary Config Source

## Context

The current implementation treats devcontainer.json as a fallback when `.dual.toml` is missing. The user wants the opposite model: devcontainer.json is the **primary source** for container configuration, and `.dual.toml` contains only Dual-specific orchestration fields. This aligns with the devcontainer ecosystem — users define their container in devcontainer.json, and `.dual.toml` handles only what devcontainer can't (command routing, shared files, volume isolation).

**Breaking change**: Fields `image`, `ports`, `setup`, `env` are removed from `.dual.toml`. Users must move container config to `devcontainer.json`.

## New Config Model

**`.dual.toml`** (Dual-specific only):
```toml
# devcontainer = ".devcontainer/devcontainer.json"  # auto-detects if omitted
extra_commands = ["cargo", "go"]
anonymous_volumes = ["node_modules", ".next"]
[shared]
files = [".env.local"]
```

**`devcontainer.json`** (container config):
```json
{
    "image": "node:20",
    "forwardPorts": [3000],
    "postCreateCommand": "pnpm install",
    "containerEnv": { "NODE_ENV": "development" }
}
```

**Loading flow**: `.dual.toml` (DualConfig) + `devcontainer.json` (DevcontainerJson) → merged `RepoHints`

   526→/// Destroy a workspace: tmux → container → clone.
   527→fn cmd_destroy(workspace_arg: Option<&str>, backend: &dyn MultiplexerBackend) -> i32 {
   528→    let mut st = match state::load() {
   529→        Ok(s) => s,
   530→        Err(e) => {
   531→            error!("{e}");
   532→            return 1;
   533→        }
   534→    };
   535→
   536→    // Resolve workspace: use arg if provided, otherwise auto-detect from cwd
   537→    let workspace;
   538→    let entry = if let Some(ws) = workspace_arg {
   539→        workspace = ws.to_string();
   540→        match st.resolve_workspace(ws) {
   541→            Some(e) => e.clone(),
   542→            None => {
   543→                error!("unknown workspace '{ws}'");
   544→                return 1;
   545→            }
   546→        }
   547→    } else {
   548→        match detect_workspace(&st) {
   549→            Some(e) => {
   550→                workspace = config::workspace_id(&e.repo, &e.branch);
   551→                info!("Auto-detected workspace: {workspace}");
   552→                e
   553→            }
   554→            None => {
   555→                error!("could not detect workspace from current directory");
   556→                info!("Usage: dual destroy [workspace]");
   557→                return 1;
   558→            }
   559→        }
   560→    };
   561→
   562→    let workspace_root = st.workspace_root();
   563→    let container_name = config::container_name(&entry.repo, &entry.branch);
   564→    let session_name = config::session_name(&entry.repo, &entry.branch);
   565→
   566→    // Destroy tmux session
   567→    if backend.is_alive(&session_name) {
   568→        info!("Destroying session {session_name}...");
   569→        if let Err(e) = backend.destroy(&session_name) {
   570→            warn!("session destroy failed: {e}");
   571→        }
   572→    }
   573→
   574→    // Stop and remove container
   575→    match container::status(&container_name) {
   576→        container::ContainerStatus::Running => {
   577→            info!("Stopping container {container_name}...");
   578→            if let Err(e) = container::stop(&container_name) {
   579→                warn!("container stop failed: {e}");
   580→            }
   581→            info!("Removing container {container_name}...");
   582→            if let Err(e) = container::destroy(&container_name) {
   583→                warn!("container remove failed: {e}");
   584→            }
   585→        }
   586→        container::ContainerStatus::Stopped => {
   587→            info!("Removing container {container_name}...");
   588→            if let Err(e) = container::destroy(&container_name) {
   589→                warn!("container remove failed: {e}");
   590→            }
   591→        }
   592→        container::ContainerStatus::Missing => {}
   593→    }
   594→
   595→    // Remove clone (only for non-explicit-path workspaces)
   596→    if entry.path.is_none() && clone::workspace_exists(&workspace_root, &entry.repo, &entry.branch)
   597→    {
   598→        info!("Removing clone...");
   599→        if let Err(e) = clone::remove_workspace(&workspace_root, &entry.repo, &entry.branch) {
   600→            error!("failed to remove clone: {e}");
   601→            return 1;
   602→        }
   603→    }
   604→
   605→    // Remove from state
   606→    st.remove_workspace(&entry.repo, &entry.branch);
   607→    if let Err(e) = state::save(&st) {
   608→        warn!("failed to save state: {e}");
   609→    }
   610→
   611→    info!("Workspace '{workspace}' destroyed.");
   612→    0
   613→}

   725→/// Sync shared config files for a workspace.
   726→fn cmd_sync(workspace_arg: Option<String>) -> i32 {
   727→    let st = match state::load() {
   728→        Ok(s) => s,
   729→        Err(e) => {
   730→            error!("{e}");
   731→            return 1;
   732→        }
   733→    };
   734→
   735→    // Resolve which workspace we're syncing
   736→    let entry = if let Some(ws) = workspace_arg {
   737→        match st.resolve_workspace(&ws) {
   738→            Some(e) => e.clone(),
   739→            None => {
   740→                error!("unknown workspace '{ws}'");
   741→                return 1;
   742→            }
   743→        }
   744→    } else {
   745→        match detect_workspace(&st) {
   746→            Some(e) => e,
   747→            None => {
   748→                error!("not inside a dual workspace");
   749→                info!("Usage: dual sync [workspace]");
   750→                return 1;
   751→            }
   752→        }
   753→    };
   754→
   755→    // Load hints
   756→    let workspace_dir = st.workspace_dir(&entry);
   757→    let hints = config::load_hints(&workspace_dir).unwrap_or_default();
   758→    let shared_config = match &hints.shared {
   759→        Some(s) if !s.files.is_empty() => s,
   760→        _ => {
   761→            error!("no [shared] section in .dual.toml (or files list is empty)");
   762→            return 1;
   763→        }
   764→    };
   765→
   766→    let shared_dir = match shared::ensure_shared_dir(&entry.repo) {
   767→        Ok(d) => d,
   768→        Err(e) => {
   769→            error!("{e}");
   770→            return 1;
   771→        }
   772→    };
   773→
   774→    let is_main = entry.path.is_some();
   775→
   776→    if is_main {
   777→        // Main workspace: init shared dir, then prompt to sync all branches
   778→        match shared::init_from_main(&workspace_dir, &shared_dir, &shared_config.files) {
   779→            Ok(moved) => {
   780→                for f in &moved {
   781→                    info!("  moved {f} → shared/");
   782→                }
   783→            }
   784→            Err(e) => {
   785→                error!("{e}");
   786→                return 1;
   787→            }
   788→        }
   789→
   790→        // Prompt to sync all branches
   791→        let branches: Vec<_> = st
   792→            .workspaces_for_repo(&entry.repo)
   793→            .into_iter()
   794→            .filter(|ws| ws.path.is_none())
   795→            .collect();
   796→
   797→        if branches.is_empty() {
   798→            info!("No branch workspaces to sync.");
   799→            return 0;
   800→        }
   801→
   802→        // Interactive prompt — use println! directly since this is user interaction
   803→        println!(
   804→            "\nSync shared files to ALL {} branch workspace(s)? [y/N]",
   805→            branches.len()
   806→        );
   807→        let mut input = String::new();
   808→        std::io::stdin().read_line(&mut input).unwrap_or(0);
   809→        if !input.trim().eq_ignore_ascii_case("y") {
   810→            info!("Cancelled.");
   811→            return 0;
   812→        }
   813→
   814→        for branch_entry in &branches {
   815→            let branch_dir = st.workspace_dir(branch_entry);
   816→            if !branch_dir.exists() {
   817→                continue; // Not yet cloned
   818→            }
   819→            let ws_id = config::workspace_id(&branch_entry.repo, &branch_entry.branch);
   820→            match shared::copy_to_branch(&branch_dir, &shared_dir, &shared_config.files) {
   821→                Ok(copied) => {
   822→                    info!("{ws_id}: synced {} file(s)", copied.len());
   823→                }
   824→                Err(e) => error!("{ws_id}: {e}"),
   825→            }
   826→        }
   827→    } else {
   828→        // Branch workspace: copy from shared dir
   829→        match shared::copy_to_branch(&workspace_dir, &shared_dir, &shared_config.files) {
   830→            Ok(copied) => {
   831→                if copied.is_empty() {
   832→                    info!(
   833→                        "No shared files available yet. Run `dual sync` in the main workspace first."
   834→                    );
   835→                } else {
   836→                    for f in &copied {
   837→                        info!("  synced {f}");
   838→                    }
   839→                }
   840→            }
   841→            Err(e) => {
   842→                error!("{e}");
   843→                return 1;
   844→            }
   845→        }
   846→    }
   847→
   848→    0
   849→}

   851→/// Detect the repo name from the current working directory.
   852→///
   853→/// Matches the git remote URL of the cwd against known workspace URLs in state.
   854→fn detect_repo_from_cwd(st: &state::WorkspaceState) -> Option<String> {
   855→    let (_, url, _) = detect_git_repo().ok()?;
   856→
   857→    // Match against known workspace URLs
   858→    for ws in st.all_workspaces() {
   859→        if ws.url == url {
   860→            return Some(ws.repo.clone());
   861→        }
   862→    }
   863→
   864→    // Also try matching by git root path against workspace paths
   865→    let cwd = std::env::current_dir().ok()?;
   866→    let root = std::process::Command::new("git")
   867→        .args(["rev-parse", "--show-toplevel"])
   868→        .output()
   869→        .ok()
   870→        .filter(|o| o.status.success())
   871→        .map(|o| PathBuf::from(String::from_utf8_lossy(&o.stdout).trim().to_string()))?;
   872→
   873→    for ws in st.all_workspaces() {
   874→        let ws_dir = st.workspace_dir(ws);
   875→        if ws_dir == root || ws_dir == cwd {
   876→            return Some(ws.repo.clone());
   877→        }
   878→    }
   879→
   880→    None
   881→}

   883→/// Detect which workspace the current directory belongs to.
   884→fn detect_workspace(st: &state::WorkspaceState) -> Option<state::WorkspaceEntry> {
   885→    let cwd = std::env::current_dir().ok()?;
   886→
   887→    // Try git root first (handles being in subdirectories)
   888→    let root = std::process::Command::new("git")
   889→        .args(["rev-parse", "--show-toplevel"])
   890→        .output()
   891→        .ok()
   892→        .filter(|o| o.status.success())
   893→        .map(|o| PathBuf::from(String::from_utf8_lossy(&o.stdout).trim().to_string()))
   894→        .unwrap_or(cwd);
   895→
   896→    for ws in st.all_workspaces() {
   897→        let ws_dir = st.workspace_dir(ws);
   898→        if ws_dir == root {
   899→            return Some(ws.clone());
   900→        }
   901→    }
   902→    None
   903→}

   905→/// Print workspace status grouped by repo.
   906→fn print_workspace_status(st: &state::WorkspaceState, backend: &dyn MultiplexerBackend) {
   907→    let workspace_root = st.workspace_root();
   908→
   909→    // Collect unique repo names in order of first appearance
   910→    let mut repos: Vec<String> = Vec::new();
   911→    for ws in st.all_workspaces() {
   912→        if !repos.contains(&ws.repo) {
   913→            repos.push(ws.repo.clone());
   914→        }
   915→    }
   916→
   917→    for repo in &repos {
   918→        println!("{repo}");
   919→        for ws in st.workspaces_for_repo(repo) {
   920→            let container_name = config::container_name(&ws.repo, &ws.branch);
   921→            let session_name = config::session_name(&ws.repo, &ws.branch);
   922→
   923→            let clone_exists = if ws.path.is_some() {
   924→                ws.path
   925→                    .as_ref()
   926→                    .map(|p| PathBuf::from(p).join(".git").exists())
   927→                    .unwrap_or(false)
   928→            } else {
   929→                clone::workspace_exists(&workspace_root, &ws.repo, &ws.branch)
   930→            };
   931→            let container_st = container::status(&container_name);
   932→            let tmux_alive = backend.is_alive(&session_name);
   933→
   934→            let (icon, status_text) = match (&container_st, tmux_alive) {
   935→                (container::ContainerStatus::Running, true) => {
   936→                    ("\u{25cf}", "running  (container: up, tmux: attached)")
   937→                }
   938→                (container::ContainerStatus::Running, false) => {
   939→                    ("\u{25cf}", "running  (container: up, tmux: none)")
   940→                }
   941→                (container::ContainerStatus::Stopped, true) => (
   942→                    "\u{25cb}",
   943→                    "stopped  (container: stopped, tmux: background)",
   944→                ),
   945→                (container::ContainerStatus::Stopped, false) => {
   946→                    ("\u{25cb}", "stopped  (container: stopped, tmux: none)")
   947→                }
   948→                (container::ContainerStatus::Missing, _) if clone_exists => {
   949→                    ("\u{25cb}", "stopped  (not launched)")
   950→                }
   951→                (container::ContainerStatus::Missing, _) => {
   952→                    ("\u{25cc}", "lazy     (not cloned yet)")
   953→                }
   954→            };
   955→
   956→            let branch_display = config::decode_branch(&config::encode_branch(&ws.branch));
   957→            println!("  {branch_display:<24} {icon} {status_text}");
   958→        }
   959→        println!();
   960→    }
   961→}

   963→/// Detect git repo info from the current directory.
   964→fn detect_git_repo() -> Result<(PathBuf, String, String), String> {
   965→    // Get repo root
   966→    let root_output = std::process::Command::new("git")
   967→        .args(["rev-parse", "--show-toplevel"])
   968→        .output()
   969→        .map_err(|_| "git not found".to_string())?;
   970→    if !root_output.status.success() {
   971→        return Err("not inside a git repository".to_string());
   972→    }
   973→    let root = PathBuf::from(String::from_utf8_lossy(&root_output.stdout).trim());
   974→
   975→    // Get remote URL
   976→    let url_output = std::process::Command::new("git")
   977→        .args(["remote", "get-url", "origin"])
   978→        .output()
   979→        .map_err(|_| "git not found".to_string())?;
   980→    let url = if url_output.status.success() {
   981→        String::from_utf8_lossy(&url_output.stdout)
   982→            .trim()
   983→            .to_string()
   984→    } else {
   985→        root.to_string_lossy().to_string() // local-only repo, use path as URL
   986→    };
   987→
   988→    // Get current branch
   989→    let branch_output = std::process::Command::new("git")
   990→        .args(["rev-parse", "--abbrev-ref", "HEAD"])
   991→        .output()
   992→        .map_err(|_| "git not found".to_string())?;
   993→    let branch = String::from_utf8_lossy(&branch_output.stdout)
   994→        .trim()
   995→        .to_string();
   996→
   997→    Ok((root, url, branch))
   998→}

   201→/// Convert DevcontainerJson fields to RepoHints.
   202→///
   203→/// Maps:
   204→/// - `image` → `hints.image`
   205→/// - `build` → `hints.dockerfile`
   206→/// - `forwardPorts` → `hints.ports`
   207→/// - `containerEnv` → `hints.env`
   208→/// - `postCreateCommand` → `hints.setup`
   209→/// - `mounts` (volume type, /workspace/* target) → `hints.anonymous_volumes`
   210→pub fn to_repo_hints(dc: &DevcontainerJson, devcontainer_dir: &Path) -> RepoHints {
   211→    let mut hints = RepoHints::default();
   212→
   213→    // Image (mutually exclusive with build)
   214→    if let Some(ref image) = dc.image {
   215→        hints.image = image.clone();
   216→    }
   217→
   218→    // Build → DockerfileBuild
   219→    if let Some(ref build) = dc.build
   220→        && build.dockerfile.is_some()
   221→    {
   222→        hints.dockerfile = Some(crate::config::DockerfileBuild {
   223→            path: build
   224→                .dockerfile
   225→                .clone()
   226→                .unwrap_or_else(|| "Dockerfile".to_string()),
   227→            context: build.context.clone().unwrap_or_else(|| ".".to_string()),
   228→            args: build.args.clone().unwrap_or_default(),
   229→            target: build.target.clone(),
   230→            base_dir: Some(devcontainer_dir.to_path_buf()),
   231→        });
   232→    }
   233→
   234→    // Ports
   235→    if let Some(ref ports) = dc.forward_ports {
   236→        hints.ports = ports.iter().filter_map(|p| p.to_port()).collect();
   237→    }
   238→
   239→    // Environment variables
   240→    if let Some(ref env) = dc.container_env {
   241→        hints.env = env.clone();
   242→    }
   243→
   244→    // Setup command
   245→    if let Some(ref cmd) = dc.post_create_command {
   246→        let shell_cmd = cmd.to_shell_command();
   247→        if !shell_cmd.is_empty() {
   248→            hints.setup = Some(shell_cmd);
   249→        }
   250→    }
   251→
   252→    // Anonymous volumes from mounts
   253→    if let Some(ref mounts) = dc.mounts {
   254→        let extra_volumes: Vec<String> = mounts
   255→            .iter()
   256→            .filter_map(|m| m.as_anonymous_volume())
   257→            .collect();
   258→        if !extra_volumes.is_empty() {
   259→            for vol in extra_volumes {
   260→                if !hints.anonymous_volumes.contains(&vol) {
   261→                    hints.anonymous_volumes.push(vol);
   262→                }
   263→            }
   264→        }
   265→    }
   266→
   267→    hints
   268→}

   179→## How It Works
   180→
   181→When you select a workspace (via `dual` or `dual launch`):
   182→
   183→1. **Clone** — Clones the repo into `{workspace_root}/{repo}/{branch}/` (uses `git clone --local` from main workspace for speed)
   184→2. **Shared files** — Copies shared config files (`.env.local`, `.vercel`, etc.) from `~/.dual/shared/{repo}/`
   185→3. **Container** — Creates and starts a Docker container with the clone bind-mounted
   186→4. **Setup** — Runs `setup` command on first launch (e.g., `pnpm install`)
   187→5. **Shell RC** — Generates transparent command routing that intercepts runtime commands and routes them to the container via `docker exec`
   188→6. **Tmux** — Creates a tmux session in the workspace directory and attaches
   189→
   190→Your editor, git, and credentials stay on the host. The container handles all runtime processes. Claude Code never knows it's running inside a container.
   191→
   192→### Shell Hook (Pane Propagation)
   193→
   194→`dual add` automatically appends a small snippet to your `~/.zshrc` or `~/.bashrc`:
   195→
   196→```bash
   197→# dual: shell interception (auto-generated)
   198→if [ -n "$DUAL_ACTIVE" ] && [ -n "$DUAL_RC_PATH" ] && [ -f "$DUAL_RC_PATH" ]; then
   199→    source "$DUAL_RC_PATH"
   200→fi
   201→```
   202→
   203→This ensures that when you split a pane (`Ctrl+b %`) or create a new window (`Ctrl+b c`) inside a Dual tmux session, the new shell automatically loads command interception. Without this, new panes would run commands on the host instead of in the container.
   204→
   205→The snippet is a no-op outside Dual sessions — it only activates when `DUAL_ACTIVE` is set (which Dual configures via `tmux set-environment`).
   206→
   207→## Architecture
   208→
   209→```
   210→Terminal
   211→├── State A: Dual TUI (ratatui)
   212→│   └── Select workspace → suspend TUI → launch pipeline → tmux attach
   213→└── State B: tmux session
   214→    └── Detach (Ctrl+b d) → resume TUI
   215→
   216→Host                          Container
   217→+--------------------------+  +--------------------------+
   218→| nvim, git, claude, ssh   |  | pnpm, node, python       |
   219→| file reads/writes        |  | curl localhost, tests    |
   220→| credentials, SSH keys    |  | port-binding processes   |

   222→        |    bind mount    |

   227→        --> container

   200→        // Check for the field as a top-level key (not in a comment)
   201→        for line in contents.lines() {
   202→            let trimmed = line.trim();
   203→            if trimmed.starts_with('#') {
   204→                continue;
   205→            }
   206→            let matches = if *old_field == "[env]" {
   207→                trimmed == "[env]"
   208→            } else {
   209→                trimmed.starts_with(&format!("{old_field} "))
   210→                    || trimmed.starts_with(&format!("{old_field}="))
   211→            };
   212→            if matches {
   213→                warn!(
   214→                    "{}: '{}' should be moved to devcontainer.json as '{}'. \
   215→                     Container config fields in .dual.toml are deprecated.",
   216→                    path.display(),
   217→                    old_field,
   218→                    new_field
   219→                );
   220→                break;
   221→            }
   222→        }
   223→    }
   224→}

   226→/// Resolve the path to devcontainer.json.
   227→///
   228→/// Uses explicit path from DualConfig if set, otherwise auto-detects.
   229→fn resolve_devcontainer_path(workspace_dir: &Path, config: &DualConfig) -> Option<PathBuf> {
   230→    if let Some(ref explicit) = config.devcontainer {
   231→        let path = workspace_dir.join(explicit);
   232→        if path.exists() {
   233→            return Some(path);
   234→        }
   235→        warn!(
   236→            "devcontainer path '{}' specified in .dual.toml does not exist",
   237→            path.display()
   238→        );
   239→        return None;
   240→    }
   241→
   242→    crate::devcontainer::find_devcontainer_json(workspace_dir)
   243→}

   245→/// Merge DualConfig + devcontainer RepoHints into final RepoHints.
   246→///
   247→/// devcontainer.json provides: image, ports, setup, env, dockerfile
   248→/// DualConfig provides: extra_commands, anonymous_volumes, shared
   249→fn merge_config(dual: &DualConfig, dc_hints: Option<&RepoHints>) -> RepoHints {
   250→    let base = dc_hints.cloned().unwrap_or_default();
   251→
   252→    RepoHints {
   253→        // Container fields come from devcontainer.json (or defaults)
   254→        image: base.image,
   255→        ports: base.ports,
   256→        setup: base.setup,
   257→        env: base.env,
   258→        dockerfile: base.dockerfile,
   259→
   260→        // Dual-specific fields come from .dual.toml
   261→        extra_commands: dual.extra_commands.clone(),
   262→        anonymous_volumes: dual.anonymous_volumes.clone(),
   263→        shared: dual.shared.clone(),
   264→    }
   265→}

   282→# extra_commands = []

   353→/// Encode a branch name for filesystem use.
   354→/// Replaces `/` with `__` (double underscore).
   355→/// e.g. "feat/auth" → "feat__auth"
   356→pub fn encode_branch(branch: &str) -> String {
   357→    branch.replace('/', "__")
   358→}
   359→
   360→/// Decode an encoded branch name back to the original.
   361→/// Replaces `__` with `/`.
   362→/// e.g. "feat__auth" → "feat/auth"
   363→pub fn decode_branch(encoded: &str) -> String {
   364→    encoded.replace("__", "/")
   365→}
