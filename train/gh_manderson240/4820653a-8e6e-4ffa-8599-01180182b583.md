**Location**: `.claude/skills/python-enum-auto-string-lookup/SKILL.md`

**Why worth extracting**: Non-obvious Python footgun — `auto()` enum members look like they should be addressable by name string, but `EnumClass("name")` does *value* lookup (integers for `auto()`), not name lookup. Affects any code that builds an enum value from a dict key or user input string. No existing skill covers this.

**Content plan**:
- **Trigger conditions**: `ValueError: 'X' is not a valid SomeEnum`, enum uses `auto()`, caller has a string key from a dict/config/user input
- **Root cause**: `auto()` assigns integer values (1, 2, 3...). `EnumClass(value)` is value-based lookup. `EnumClass["NAME"]` is name-based lookup (case-sensitive).
- **Fix**: `EnumClass[key.upper()]` or a case-insensitive lookup dict:
  ```python
  _INTENT_MAP = {name.lower(): member for name, member in IntentType.__members__.items()}

  ```
- **Alternative**: Use `str`-valued enum (`class IntentType(str, Enum): GENERATE = "generate"`) so value lookup works with strings — but this requires changing the enum definition.
- **Where this bit us**: `request_alignment_analyzer.py:356` and `:412` — `IntentType(best_intent)` where `best_intent` is a string key like `"generate"` from `_INTENT_KEYWORDS` dict.

   117→_INTENT_DEFAULTS = {
   118→    IntentType.GENERATE: [
   119→        SuccessCriterion("Output is coherent", "coherence", 0.7, False),
   120→    ],
   121→    IntentType.ANALYZE: [
   122→        SuccessCriterion("Analysis is correct", "correctness", 0.8, False),
   123→    ],
   124→    IntentType.SEARCH: [
   125→        SuccessCriterion("Results are relevant", "relevance", 0.7, False),
   126→    ],
   127→    IntentType.TRANSFORM: [
   128→        SuccessCriterion("Output format is correct", "format_correctness", 0.8, False),

   133→class RequestAlignmentAnalyzer:
   134→    """Analyze alignment between human requests and task execution.
   135→
   136→    Bridges the gap between human intent and compound execution by:
   137→    1. Parsing requests into structured intent/constraints/criteria
   138→    2. Analyzing how well execution matches the request
   139→    3. Scoring misalignment for experience-guided improvement
   140→    4. Persisting alignment patterns to vault
   141→    """
   142→
   143→    def __init__(
   144→        self,
   145→        mcp_client: MCPClient,
   146→        intent_confidence_threshold: float = 0.1,
   147→        constraint_tolerance: float = 0.1,
   148→    ):
   149→        """Initialize request alignment analyzer.
   150→
   151→        Args:
   152→            mcp_client: Connected MCPClient for vault operations
   153→            intent_confidence_threshold: Minimum confidence for intent classification
   154→            constraint_tolerance: % tolerance for constraint satisfaction
   155→        """
   156→        self.mcp_client = mcp_client
   157→        self.intent_confidence_threshold = intent_confidence_threshold
   158→        self.constraint_tolerance = constraint_tolerance
   159→
   160→        # Lazy-loaded semantic encoder
   161→        self._text_encoder = None
   162→
   163→    @property
   164→    def text_encoder(self) -> Any | None:
   165→        """Lazy-load text encoder for semantic intent classification.
   166→
   167→        Returns:
   168→            SemanticTextEncoder or None if unavailable
   169→        """
   170→        if self._text_encoder is None:
   171→            try:
   172→                from cohezion.cache.text_encoder import get_text_encoder
   173→
   174→                self._text_encoder = get_text_encoder()
   175→                logger.debug("Initialized semantic text encoder")
   176→            except Exception as e:
   177→                logger.debug("Failed to load semantic encoder (using keyword fallback): %s", e)
   178→                self._text_encoder = False  # Flag: tried and failed
   179→        return self._text_encoder if self._text_encoder else None

   181→    def parse_request(self, request_text: str) -> HumanRequest:
   182→        """Parse human request into structured HumanRequest.
   183→
   184→        Args:
   185→            request_text: Raw human request text
   186→
   187→        Returns:
   188→            HumanRequest with parsed intent, constraints, criteria
   189→        """
   190→        request = HumanRequest(raw_text=request_text)
   191→
   192→        # Step 1: Classify intent (keywords first, then semantic)
   193→        intent_type, intent_confidence = self._classify_intent(request_text)
   194→        request.intent = intent_type
   195→        request.intent_confidence = intent_confidence
   196→        logger.debug(
   197→            "Classified intent: %s (confidence=%.2f)",
   198→            intent_type.value,
   199→            intent_confidence,
   200→        )
   201→
   202→        # Step 2: Extract constraints
   203→        request.constraints = self._extract_constraints(request_text)
   204→        logger.debug("Extracted %d constraints", len(request.constraints))
   205→
   206→        # Step 3: Extract success criteria
   207→        request.criteria = self._extract_criteria(request_text, intent_type)
   208→        logger.debug("Extracted %d success criteria", len(request.criteria))
   209→
   210→        # Step 4: Extract scope
   211→        request.scope_includes, request.scope_excludes = self._extract_scope(request_text)
   212→
   213→        return request

   215→    def analyze_alignment(
   216→        self,
   217→        request: HumanRequest,
   218→        execution_result: ExecutionResult,
   219→        operation_type: str,
   220→        anomaly_analysis: AnomalyDetection | None = None,
   221→    ) -> ExecutionAlignment:
   222→        """Analyze alignment between request and execution result.
   223→
   224→        Args:
   225→            request: Parsed HumanRequest
   226→            execution_result: ExecutionResult from compound execution
   227→            operation_type: Type of operation that was executed
   228→            anomaly_analysis: Optional AnomalyDetection from InflectionDetector
   229→
   230→        Returns:
   231→            ExecutionAlignment with scores, violations, failures, recommendations
   232→        """
   233→        # Compute intent match score
   234→        intent_match_score = self._compute_intent_match(
   235→            request.intent, operation_type, execution_result
   236→        )
   237→
   238→        # Check constraint satisfaction
   239→        violations = self._check_constraints(request.constraints or [], execution_result.metrics)
   240→        constraint_satisfaction = max(0.0, 1.0 - (len(violations) * 0.3))  # Penalty per violation
   241→
   242→        # Check success criteria
   243→        failures = self._check_criteria(request.criteria or [], execution_result.metrics)
   244→        criteria_satisfaction = max(0.0, 1.0 - (len(failures) * 0.2))  # Penalty per failure
   245→
   246→        # Detect drift signals
   247→        drift_signals = self._detect_drift_signals(execution_result, anomaly_analysis)
   248→        drift_penalty = (
   249→            sum(s.severity for s in drift_signals) / len(drift_signals) if drift_signals else 0.0
   250→        )
   251→
   252→        # Compute composite misalignment score
   253→        alignment_score = (
   254→            0.4 * intent_match_score + 0.3 * constraint_satisfaction + 0.3 * criteria_satisfaction
   255→        )
   256→        misalignment_score = (1.0 - alignment_score) + (drift_penalty * 0.2)
   257→        misalignment_score = min(1.0, max(0.0, misalignment_score))
   258→
   259→        # Generate issues and recommendations
   260→        issues = self._generate_issues(violations, failures, drift_signals, intent_match_score)
   261→        recommendations = self._generate_recommendations(
   262→            violations, failures, drift_signals, request.intent
   263→        )
   264→
   265→        return ExecutionAlignment(
   266→            intent_match_score=intent_match_score,
   267→            constraint_satisfaction=constraint_satisfaction,
   268→            criteria_satisfaction=criteria_satisfaction,
   269→            misalignment_score=misalignment_score,
   270→            violations=violations,
   271→            failures=failures,
   272→            drift_signals=drift_signals,
   273→            issues=issues,
   274→            recommendations=recommendations,
   275→            should_retry=misalignment_score > 0.5,
   276→        )

   278→    def log_alignment_to_vault(
   279→        self, request: HumanRequest, alignment: ExecutionAlignment, project: str
   280→    ) -> str:
   281→        """Log alignment analysis to vault for experience guidance.

   291→        if alignment.misalignment_score > 0.5:

   327→    def _classify_intent(self, request_text: str) -> tuple[IntentType, float]:
   328→        """Classify request intent using keywords then semantic fallback.
   329→
   330→        Args:
   331→            request_text: Request text to classify
   332→
   333→        Returns:
   334→            Tuple of (IntentType, confidence_score)
   335→        """
   336→        # Phase 1: Keyword matching (fast)
   337→        scores: dict[str, int] = {}
   338→        lower_text = request_text.lower()
   339→
   340→        for intent_str, keywords in _INTENT_KEYWORDS.items():
   341→            score = 0
   342→            for keyword in keywords:
   343→                if re.search(rf"\b{keyword}\b", lower_text):
   344→                    score += 1
   345→            scores[intent_str] = score
   346→
   347→        best_intent = max(scores, key=scores.get)
   348→        best_score = scores[best_intent]
   349→        # Normalize by matched category's keywords (not all keywords) so a
   350→        # single match out of 7 gives confidence ≈ 0.14, not 0.03.
   351→        intent_keyword_count = len(_INTENT_KEYWORDS.get(best_intent, [1]))
   352→        confidence = best_score / max(1, intent_keyword_count)
   353→
   354→        if confidence >= self.intent_confidence_threshold:
   355→            return (
   356→                IntentType[best_intent.upper()],
   357→                confidence,
   358→            )
   359→
   360→        # Phase 2: Semantic fallback (if encoder available)
   361→        if self.text_encoder:
   362→            return self._classify_intent_semantic(request_text)
   363→
   364→        # Phase 3: Default to UNKNOWN
   365→        logger.debug("Could not classify intent, defaulting to UNKNOWN")
   366→        return IntentType.UNKNOWN, 0.0
   367→
   368→    def _classify_intent_semantic(self, request_text: str) -> tuple[IntentType, float]:
   369→        """Semantic intent classification (fallback).
   370→
   371→        Args:
   372→            request_text: Request text to classify
   373→
   374→        Returns:
   375→            Tuple of (IntentType, confidence_score)
   376→        """
   377→        try:
   378→            import numpy as np
   379→
   380→            encoder = self.text_encoder
   381→            if not encoder:
   382→                return IntentType.UNKNOWN, 0.0
   383→
   384→            # Encode request
   385→            request_embedding = encoder.encode(request_text)
   386→
   387→            # Compute intent prototypes (lazy)
   388→            intent_prototypes = {
   389→                "generate": encoder.encode(
   390→                    "Create, write, compose, draft, and produce new content"
   391→                ),
   392→                "analyze": encoder.encode("Evaluate, assess, review, and analyze existing content"),
   393→                "search": encoder.encode("Find, locate, discover, and search for items"),
   394→                "transform": encoder.encode("Convert, reformat, extract, and transform data"),
   395→                "persist": encoder.encode("Store, save, record, and persist information"),
   396→            }
   397→
   398→            # Compute similarities
   399→            best_intent = None
   400→            best_similarity = 0.0
   401→
   402→            for intent_str, prototype in intent_prototypes.items():
   403→                similarity = float(
   404→                    np.dot(request_embedding, prototype)
   405→                    / (np.linalg.norm(request_embedding) * np.linalg.norm(prototype))
   406→                )
   407→                if similarity > best_similarity:
   408→                    best_similarity = similarity
   409→                    best_intent = intent_str
   410→
   411→            if best_similarity >= 0.3:  # Semantic threshold
   412→                return IntentType[best_intent.upper()], best_similarity
   413→
   414→            return IntentType.UNKNOWN, 0.0
   415→        except Exception as e:
   416→            logger.debug("Semantic intent classification failed: %s", e)
   417→            return IntentType.UNKNOWN, 0.0

   419→    def _extract_constraints(self, request_text: str) -> list[ExecutionConstraint]:
   420→        """Extract constraints from request text using regex patterns.
   421→
   422→        Args:
   423→            request_text: Request text to extract from
   424→
   425→        Returns:
   426→            List of ExecutionConstraints
   427→        """
   428→        constraints = []
   429→
   430→        # Check TOKENS constraint
   431→        match = _CONSTRAINT_PATTERNS[ConstraintType.TOKENS].search(request_text)
   432→        if match:
   433→            value = float(match.group(1))
   434→            constraints.append(
   435→                ExecutionConstraint(
   436→                    type=ConstraintType.TOKENS, value=value, unit="tokens", is_hard=True
   437→                )
   438→            )
   439→
   440→        # Check LATENCY constraint
   441→        match = _CONSTRAINT_PATTERNS[ConstraintType.LATENCY].search(request_text)
   442→        if match:
   443→            value = float(match.group(1))
   444→            unit = match.group(2).lower()
   445→            # Normalize to milliseconds
   446→            multipliers = {
   447→                "ms": 1,
   448→                "millisecond": 1,
   449→                "sec": 1000,
   450→                "second": 1000,
   451→                "min": 60000,
   452→                "minute": 60000,
   453→            }
   454→            value_ms = value * multipliers.get(unit, 1)
   455→            constraints.append(
   456→                ExecutionConstraint(
   457→                    type=ConstraintType.LATENCY, value=value_ms, unit="ms", is_hard=True
   458→                )
   459→            )
   460→
   461→        # Check QUALITY constraint
   462→        match = _CONSTRAINT_PATTERNS[ConstraintType.QUALITY].search(request_text)
   463→        if match:
   464→            quality_level = match.group(1).lower()
   465→            quality_map = {"high": 0.7, "max": 0.9, "low": 0.3, "min": 0.1}
   466→            value = quality_map.get(quality_level, 0.5)
   467→            constraints.append(
   468→                ExecutionConstraint(
   469→                    type=ConstraintType.QUALITY,
   470→                    value=value,
   471→                    unit="score",
   472→                    is_hard=False,  # Soft preference
   473→                )
   474→            )
   475→
   476→        # Check SCOPE constraint
   477→        match = _CONSTRAINT_PATTERNS[ConstraintType.SCOPE].search(request_text)
   478→        if match:
   479→            scope = match.group(1)
   480→            constraints.append(
   481→                ExecutionConstraint(type=ConstraintType.SCOPE, value=1.0, unit=scope, is_hard=True)
   482→            )
   483→
   484→        return constraints

   486→    def _extract_criteria(
   487→        self, request_text: str, intent_type: IntentType
   488→    ) -> list[SuccessCriterion]:
   489→        """Extract success criteria from request text.
   490→
   491→        Args:
   492→            request_text: Request text to extract from
   493→            intent_type: Inferred intent type for default criteria
   494→
   495→        Returns:
   496→            List of SuccessCriteria
   497→        """
   498→        criteria = []
   499→
   500→        # Look for explicit criteria ("must be...", "should be...")
   501→        explicit_pattern = re.compile(r"(?:must|should|needs?)\s+(?:be\s+)?(\w+)", re.IGNORECASE)
   502→        for match in explicit_pattern.finditer(request_text):
   503→            criterion_word = match.group(1).lower()
   504→            if criterion_word in _CRITERION_METRIC_MAP:
   505→                metric_name, threshold = _CRITERION_METRIC_MAP[criterion_word]
   506→                criteria.append(
   507→                    SuccessCriterion(
   508→                        description=f"Output must be {criterion_word}",
   509→                        metric_name=metric_name,
   510→                        threshold=threshold,
   511→                        is_explicit=True,
   512→                    )
   513→                )
   514→
   515→        # Add intent-specific default criteria if none extracted
   516→        if not criteria and intent_type in _INTENT_DEFAULTS:
   517→            criteria = _INTENT_DEFAULTS[intent_type].copy()
   518→
   519→        return criteria

   521→    def _extract_scope(self, request_text: str) -> tuple[list[str], list[str]]:
   522→        """Extract scope inclusions/exclusions from request.
   523→
   524→        Args:
   525→            request_text: Request text to extract from
   526→
   527→        Returns:
   528→            Tuple of (scope_includes, scope_excludes)
   529→        """
   530→        includes = []
   531→        excludes = []
   532→
   533→        # Look for "only", "just" (inclusions)
   534→        only_pattern = re.compile(r"(?:only|just)\s+(\w+)", re.IGNORECASE)
   535→        for match in only_pattern.finditer(request_text):
   536→            includes.append(match.group(1))
   537→
   538→        # Look for "except", "excluding", "not", "without" (exclusions)
   539→        except_pattern = re.compile(r"(?:except|excluding|not|without)\s+(\w+)", re.IGNORECASE)
   540→        for match in except_pattern.finditer(request_text):
   541→            excludes.append(match.group(1))
   542→
   543→        return includes, excludes

   545→    def _compute_intent_match(
   546→        self, request_intent: IntentType, operation_type: str, result: ExecutionResult
   547→    ) -> float:
   548→        """Compute how well executed operation matched request intent.
   549→
   550→        Args:
   551→            request_intent: Intended operation type
   552→            operation_type: Actual operation type executed
   553→            result: ExecutionResult to analyze
   554→
   555→        Returns:
   556→            0.0-1.0 intent match score
   557→        """
   558→        # Direct intent match (compare lowercase name to operation_type string)
   559→        if request_intent.name.lower() == operation_type.lower():
   560→            return 1.0
   561→
   562→        # Semantic similarity (if encoder available)
   563→        if self.text_encoder and result.output:
   564→            try:
   565→                import numpy as np
   566→
   567→                encoder = self.text_encoder
   568→                intent_prototype = encoder.encode(f"This is a {request_intent.name.lower()} task")
   569→                output_embedding = encoder.encode(result.output[:500])  # First 500 chars
   570→
   571→                similarity = float(
   572→                    np.dot(intent_prototype, output_embedding)
   573→                    / (np.linalg.norm(intent_prototype) * np.linalg.norm(output_embedding))
   574→                )
   575→                return max(0.0, min(1.0, similarity))
   576→            except Exception as e:
   577→                logger.debug("Intent alignment computation failed: %s", e)
   578→
   579→        # Multi-step request handling
   580→        if request_intent == IntentType.MULTI_STEP:
   581→            return 0.7  # Partial credit
   582→
   583→        # Mismatch
   584→        return 0.0
   585→
   586→    def _check_constraints(
   587→        self, constraints: list[ExecutionConstraint], metrics: dict[str, Any]
   588→    ) -> list[ConstraintViolation]:
   589→        """Check which constraints were violated.
   590→
   591→        Args:
   592→            constraints: Constraints to check
   593→            metrics: Execution metrics dict
   594→
   595→        Returns:
   596→            List of ConstraintViolations
   597→        """
   598→        violations = []
   599→
   600→        for constraint in constraints:
   601→            if constraint.type == ConstraintType.TOKENS:
   602→                actual = metrics.get("tokens_used", 0)
   603→                allowed = constraint.value * (1 + self.constraint_tolerance)
   604→                if actual > allowed:
   605→                    severity = min(1.0, (actual - allowed) / allowed)
   606→                    violations.append(
   607→                        ConstraintViolation(
   608→                            constraint=constraint,
   609→                            requested_value=constraint.value,
   610→                            actual_value=actual,
   611→                            severity=severity,
   612→                        )
   613→                    )
   614→
   615→            elif constraint.type == ConstraintType.LATENCY:
   616→                actual_ms = metrics.get("duration_seconds", 0) * 1000
   617→                allowed_ms = constraint.value * (1 + self.constraint_tolerance)
   618→                if actual_ms > allowed_ms:
   619→                    severity = min(1.0, (actual_ms - allowed_ms) / allowed_ms)
   620→                    violations.append(
   621→                        ConstraintViolation(
   622→                            constraint=constraint,
   623→                            requested_value=constraint.value,
   624→                            actual_value=actual_ms,
   625→                            severity=severity,
   626→                        )
   627→                    )
   628→
   629→            elif constraint.type == ConstraintType.QUALITY:
   630→                # Check quality metrics (coherence, accuracy, etc.)
   631→                quality_metrics = ["coherence", "accuracy", "correctness"]
   632→                actual_quality = max((metrics.get(m, 0) for m in quality_metrics), default=0.0)
   633→                if actual_quality < constraint.value:
   634→                    severity = min(1.0, (constraint.value - actual_quality) / constraint.value)
   635→                    violations.append(
   636→                        ConstraintViolation(
   637→                            constraint=constraint,
   638→                            requested_value=constraint.value,
   639→                            actual_value=actual_quality,
   640→                            severity=severity,
   641→                        )
   642→                    )
   643→
   644→        return violations
   645→
   646→    def _check_criteria(
   647→        self, criteria: list[SuccessCriterion], metrics: dict[str, Any]
   648→    ) -> list[CriterionFailure]:
   649→        """Check which success criteria were not met.
   650→
   651→        Args:
   652→            criteria: Success criteria to check
   653→            metrics: Execution metrics dict
   654→
   655→        Returns:
   656→            List of CriterionFailures
   657→        """
   658→        failures = []
   659→
   660→        for criterion in criteria:
   661→            actual = metrics.get(criterion.metric_name, 0.0)
   662→            if actual < criterion.threshold:
   663→                gap = criterion.threshold - actual
   664→                failures.append(
   665→                    CriterionFailure(
   666→                        criterion=criterion,
   667→                        expected_value=criterion.threshold,
   668→                        actual_value=actual,
   669→                        gap=gap,
   670→                    )
   671→                )
   672→
   673→        return failures

   675→    def _detect_drift_signals(
   676→        self, result: ExecutionResult, anomaly: AnomalyDetection | None
   677→    ) -> list[DriftSignal]:
   678→        """Detect signals indicating execution divergence.
   679→
   680→        Args:
   681→            result: ExecutionResult to analyze
   682→            anomaly: Optional AnomalyDetection from InflectionDetector
   683→
   684→        Returns:
   685→            List of DriftSignals
   686→        """
   687→        signals = []
   688→
   689→        # Execution failure
   690→        if not result.success:
   691→            signals.append(
   692→                DriftSignal(
   693→                    signal_type="execution_failed",
   694→                    severity=1.0,
   695→                    description="Task execution failed",
   696→                    metadata={"error": result.metrics.get("error", "unknown")},
   697→                )
   698→            )
   699→
   700→        # Anomaly detection signals
   701→        if anomaly:
   702→            if anomaly.severity.value == "critical":
   703→                signals.append(
   704→                    DriftSignal(
   705→                        signal_type="anomaly_critical",
   706→                        severity=0.9,
   707→                        description=f"Critical anomaly detected: {anomaly.issues}",
   708→                        metadata={"anomaly_score": anomaly.score},
   709→                    )
   710→                )
   711→            elif anomaly.severity.value == "warning":
   712→                signals.append(
   713→                    DriftSignal(
   714→                        signal_type="anomaly_warning",
   715→                        severity=0.5,
   716→                        description=f"Anomaly warning: {anomaly.issues}",
   717→                        metadata={"anomaly_score": anomaly.score},
   718→                    )
   719→                )
   720→
   721→        # Coherence drop
   722→        if result.metrics.get("coherence", 1.0) < 0.3:
   723→            signals.append(
   724→                DriftSignal(
   725→                    signal_type="coherence_drop",
   726→                    severity=0.7,
   727→                    description="Output coherence below threshold",
   728→                    metadata={"coherence": result.metrics.get("coherence")},
   729→                )
   730→            )
   731→
