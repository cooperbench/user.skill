#!/bin/bash
# Monitor tunnel health every 5 minutes

mkdir -p "$(dirname "$LOG_FILE")"

check_health() {
    local response=$(curl -s -w "\n%{http_code}" "$HEALTH_URL" 2>&1)
    local http_code=$(echo "$response" | tail -1)
    local timestamp=$(date -u +"%Y-%m-%dT%H:%M:%SZ")
    
    echo "$timestamp | HTTP $http_code | $(echo "$response" | head -1)" >> "$LOG_FILE"

        # Non-blocking: log only, don't crash

    fi
}

check_health
