   118→    def analyze(self, query: str) -> QueryComplexity:
   119→        """Analyze query complexity.
   120→
   121→        Args:
   122→            query: User query string
   123→
   124→        Returns:
   125→            QueryComplexity tier (SIMPLE, MEDIUM, or COMPLEX)
   126→        """
   127→        # Token estimation (rough)
   128→        token_count = self._estimate_tokens(query)
   129→
   130→        # Keyword analysis (word-boundary matching to avoid substring matches)
   131→        query_lower = query.lower()
   132→        simple_matches = sum(1 for kw in self.SIMPLE_KEYWORDS if f" {kw} " in f" {query_lower} " or query_lower.startswith(f"{kw} ") or query_lower.endswith(f" {kw}"))
   133→        complex_matches = sum(1 for kw in self.COMPLEX_KEYWORDS if f" {kw} " in f" {query_lower} " or query_lower.startswith(f"{kw} ") or query_lower.endswith(f" {kw}"))
   134→
   135→        # Heuristics
   136→        has_code = any(pattern in query for pattern in ["```", "def ", "class ", "import", "function"])
   137→        has_data_processing = any(
   138→            word in query_lower for word in ["process", "analyze", "transform", "pipeline"]
   139→        )
   140→        has_logic = " and " in query_lower or " or " in query_lower or "if " in query_lower
   141→        is_short = token_count < 30
   142→        is_long = token_count > 200
   143→
   144→        # Determine complexity tier
   145→        # SIMPLE: very short (< 10 tokens) without complex keywords, or has simple keywords + no complex keywords + short
   146→        if (token_count < 10 and complex_matches == 0 and not has_code) or (simple_matches > 0 and complex_matches == 0 and not has_code and token_count < 50):
   147→            complexity = QueryComplexity.SIMPLE
   148→        # COMPLEX: has multiple complex keywords, code, or is long with logic
   149→        elif (complex_matches >= 2) or (has_code and has_logic) or (is_long and has_data_processing):
   150→            complexity = QueryComplexity.COMPLEX
   151→        # MEDIUM: everything else
   152→        else:
   153→            complexity = QueryComplexity.MEDIUM
   154→
   155→        # Record for analytics
   156→        self.history.append(
   157→            {
   158→                "timestamp": time.time(),
   159→                "query_len": len(query),
   160→                "token_count": token_count,
   161→                "simple_matches": simple_matches,
   162→                "complex_matches": complex_matches,
   163→                "complexity": complexity.value,
   164→            }
   165→        )
   166→
   167→        return complexity

   179→        clean = re.sub(r"```[\s\S]*?```", "", query)

   323→    def select_model(
   324→        self, query: str, max_cost_usd: Optional[float] = None
   325→    ) -> Tuple[ModelRoutingDecision, bool]:
   326→        """Select optimal model for query with cost/token optimization.
   327→
   328→        Args:
   329→            query: User query
   330→            max_cost_usd: Maximum allowed cost (optional)
   331→
   332→        Returns:
   333→            Tuple of (decision, can_proceed)
   334→            - decision: ModelRoutingDecision with selected model
   335→            - can_proceed: False if budget enforcer blocks (decision still returned)
   336→        """
   337→        # Analyze complexity
   338→        complexity = self.complexity_analyzer.analyze(query)
   339→        estimated_tokens = self.EXPECTED_TOKENS[complexity]
   340→
   341→        # Select model by complexity
   342→        if complexity == QueryComplexity.SIMPLE:
   343→            primary_model = "phi3:mini"
   344→        elif complexity == QueryComplexity.MEDIUM:
   345→            primary_model = "qwen3-coder:32b"
   346→        else:
   347→            primary_model = "deepseek-r1:8b"
   348→
   349→        # Optimize model selection based on cost/token ratio if enabled
   350→        model = self._optimize_model_selection(primary_model, complexity, estimated_tokens)
   351→
   352→        # Calculate estimated cost
   353→        cost_per_1k = self.MODEL_COSTS[model]
   354→        estimated_cost = (estimated_tokens / 1000.0) * cost_per_1k
   355→
   356→        # Check budget constraint
   357→        can_proceed = True
   358→        if max_cost_usd and estimated_cost > max_cost_usd:
   359→            can_proceed = False
   360→
   361→        # Check budget enforcer (if available)
   362→        if self.budget_enforcer and self.cost_tracker:
   363→            enforcer_ok, enforcer_msg = self.budget_enforcer.check_budget(
   364→                self.cost_tracker.total_cost_usd + estimated_cost
   365→            )
   366→            if not enforcer_ok:
   367→                can_proceed = False
   368→
   369→        # Build decision
   370→        reason = f"Routed {complexity.value} query to {model}"
   371→        if model != primary_model:
   372→            reason += f" (optimized from {primary_model})"
   373→
   374→        decision = ModelRoutingDecision(
   375→            model=model,
   376→            complexity=complexity,
   377→            estimated_tokens=estimated_tokens,
   378→            estimated_cost_usd=estimated_cost,
   379→            reason=reason,
   380→            quality_score=self.MODEL_QUALITY[model],
   381→        )
   382→
   383→        # Record decision
   384→        self.routing_decisions.append(decision)
   385→        self.query_count_per_model[model] += 1
   386→
   387→        logger.info(
   388→            f"Cost router: {complexity.value} query → {model} "
   389→            f"(est. {estimated_tokens} tokens, ${estimated_cost:.6f})"
   390→        )
   391→
   392→        return decision, can_proceed

   394→    def _optimize_model_selection(
   395→        self, primary_model: str, complexity: QueryComplexity, estimated_tokens: int
   396→    ) -> str:
   397→        """Optimize model selection based on cost/token ratio with aggressive cost reduction.
   398→
   399→        Args:
   400→            primary_model: Model selected by complexity analysis
   401→            complexity: Query complexity tier
   402→            estimated_tokens: Estimated tokens for the query
   403→
   404→        Returns:
   405→            Optimized model name (may be different from primary)
   406→        """
   407→        if not self.prefer_longer_models_if_cheaper_per_token:
   408→            return primary_model
   409→
   410→        # For complex queries, check if a faster model is good enough
   411→        if complexity == QueryComplexity.COMPLEX:
   412→            # Check if medium model (qwen) has acceptable cost/token ratio
   413→            primary_cost_per_token = self._get_cost_per_token(primary_model, estimated_tokens)
   414→            qwen_cost_per_token = self._get_cost_per_token("qwen3-coder:32b", estimated_tokens)
   415→
   416→            # If qwen is cheaper per token AND latency is acceptable, use it
   417→            if self._is_cheaper_with_acceptable_latency(
   418→                "qwen3-coder:32b", primary_model, qwen_cost_per_token, primary_cost_per_token
   419→            ):
   420→                self.token_optimization_swaps += 1
   421→                return "qwen3-coder:32b"
   422→
   423→            # If aggressive cost reduction enabled, also check phi3 for complex queries
   424→            if self.aggressive_cost_reduction:
   425→                phi3_cost_per_token = self._get_cost_per_token("phi3:mini", estimated_tokens)
   426→                # Check if phi3 can handle complex queries with acceptable latency/quality tradeoff
   427→                if self._is_cheaper_with_acceptable_latency(
   428→                    "phi3:mini", primary_model, phi3_cost_per_token, primary_cost_per_token,
   429→                    aggressive=True
   430→                ):
   431→                    self.token_optimization_swaps += 1
   432→                    return "phi3:mini"
   433→
   434→        # For medium queries, check if simple model (phi3) is good enough
   435→        if complexity == QueryComplexity.MEDIUM:
   436→            primary_cost_per_token = self._get_cost_per_token(primary_model, estimated_tokens)
   437→            phi3_cost_per_token = self._get_cost_per_token("phi3:mini", estimated_tokens)
   438→
   439→            # Standard check
   440→            if self._is_cheaper_with_acceptable_latency(
   441→                "phi3:mini", primary_model, phi3_cost_per_token, primary_cost_per_token
   442→            ):
   443→                self.token_optimization_swaps += 1
   444→                return "phi3:mini"
   445→
   446→            # If aggressive cost reduction, be more lenient with phi3 for medium queries
   447→            if self.aggressive_cost_reduction:
   448→                latency_diff = self.MODEL_LATENCY.get("phi3:mini", 50.0) - self.MODEL_LATENCY.get(primary_model, 100.0)
   449→                # Allow phi3 even if latency is slightly higher (up to 200ms for 50% cost savings)
   450→                if latency_diff <= 200.0:
   451→                    self.token_optimization_swaps += 1
   452→                    return "phi3:mini"
   453→
   454→        # For simple queries, strongly prefer phi3 for cost savings
   455→        if complexity == QueryComplexity.SIMPLE:
   456→            # Always prefer phi3 for simple queries unless latency is critical
   457→            if primary_model != "phi3:mini":
   458→                latency_diff = self.MODEL_LATENCY.get("phi3:mini", 50.0) - self.MODEL_LATENCY.get(primary_model, 50.0)
   459→                if latency_diff <= 150.0:  # phi3 is still acceptable for simple queries
   460→                    self.token_optimization_swaps += 1
   461→                    return "phi3:mini"
   462→
   463→        return primary_model

   465→    def _get_cost_per_token(self, model: str, tokens: int) -> float:
   466→        """Calculate cost per token for a model.
   467→
   468→        Args:
   469→            model: Model name
   470→            tokens: Token count
   471→
   472→        Returns:
   473→            Cost per token (in USD)
   474→        """
   475→        if tokens == 0:
   476→            return 0.0
   477→        cost_per_1k = self.MODEL_COSTS.get(model, 0.0)
   478→        return cost_per_1k / 1000.0

   480→    def _is_cheaper_with_acceptable_latency(
   481→        self,
   482→        candidate_model: str,
   483→        primary_model: str,
   484→        candidate_cost_per_token: float,
   485→        primary_cost_per_token: float,
   486→        aggressive: bool = False,
   487→    ) -> bool:
   488→        """Check if candidate model is cheaper AND has acceptable latency trade-off.
   489→
   490→        Args:
   491→            candidate_model: Model to consider
   492→            primary_model: Primary model for comparison
   493→            candidate_cost_per_token: Cost per token for candidate
   494→            primary_cost_per_token: Cost per token for primary
   495→            aggressive: If True, use relaxed thresholds for cost reduction
   496→
   497→        Returns:
   498→            True if candidate is cheaper AND latency trade-off is acceptable
   499→        """
   500→        # Check cost difference
   501→        if primary_cost_per_token == 0.0:
   502→            # For local models, use TPS as proxy for cost efficiency
   503→            candidate_tps = self.MODEL_TPS.get(candidate_model, 1.0)
   504→            primary_tps = self.MODEL_TPS.get(primary_model, 1.0)
   505→            # Use aggressive threshold if enabled (more lenient)
   506→            threshold = self.cost_threshold if not aggressive else (self.cost_threshold + 0.15)
   507→            # If candidate is at least X% as fast as primary, it's acceptable
   508→            if candidate_tps < primary_tps * (1.0 - threshold):
   509→                return False
   510→        else:
   511→            # For API models, check cost ratio
   512→            cost_ratio = candidate_cost_per_token / primary_cost_per_token if primary_cost_per_token > 0 else 1.0
   513→            threshold = self.cost_threshold if not aggressive else (self.cost_threshold + 0.15)
   514→            if cost_ratio > (1.0 - threshold):
   515→                return False
   516→
   517→        # Check latency trade-off
   518→        candidate_latency = self.MODEL_LATENCY.get(candidate_model, 100.0)
   519→        primary_latency = self.MODEL_LATENCY.get(primary_model, 100.0)
   520→        latency_increase = candidate_latency - primary_latency
   521→
   522→        # Use relaxed latency threshold if aggressive mode
   523→        threshold = self.latency_threshold if not aggressive else (self.latency_threshold + 100.0)
   524→        # If latency increase is within threshold, accept the optimization
   525→        return latency_increase <= threshold

   527→    def record_execution(
   528→        self, model: str, actual_tokens: int, duration_ms: float, success: bool = True
   529→    ) -> float:
   530→        """Record execution and track costs with success metrics.
   531→
   532→        Args:
   533→            model: Model name
   534→            actual_tokens: Actual tokens used
   535→            duration_ms: Execution duration in milliseconds
   536→            success: Whether execution was successful (for dynamic tuning)
   537→
   538→        Returns:
   539→            Cost in USD
   540→        """
   541→        # Track with cost tracker
   542→        cost_usd = 0.0
   543→        if self.cost_tracker:
   544→            cost_usd = self.cost_tracker.track_usage_fast(
   545→                model=model, tokens=actual_tokens, duration_ms=duration_ms
   546→            )
   547→        else:
   548→            # Fallback calculation
   549→            cost_per_1k = self.MODEL_COSTS.get(model, 0.0)
   550→            cost_usd = (actual_tokens / 1000.0) * cost_per_1k
   551→
   552→        # Update aggregated costs
   553→        self.cost_per_model[model] += cost_usd
   554→
   555→        # Update success tracking for dynamic threshold tuning
   556→        if success:
   557→            if model == "phi3:mini":
   558→                self._phi3_success_count += 1
   559→            elif model == "qwen3-coder:32b":
   560→                self._qwen_success_count += 1
   561→
   562→        # Track latency for threshold adjustment
   563→        self._cumulative_latency_ms += duration_ms
   564→
   565→        # Auto-tune thresholds if enabled
   566→        if self.dynamic_threshold_tuning:
   567→            self._tune_thresholds_based_on_success()
   568→
   569→        logger.debug(
   570→            f"Recorded execution: {model} {actual_tokens} tokens, "
   571→            f"${cost_usd:.6f}, {duration_ms:.1f}ms, success={success}"
   572→        )
   573→
   574→        return cost_usd

   576→    def _tune_thresholds_based_on_success(self) -> None:
   577→        """Dynamically tune cost/latency thresholds based on success patterns.
   578→
   579→        If phi3 has high success rate, gradually relax thresholds to route more queries to it.
   580→        """
   581→        # Calculate success rates
   582→        phi3_total = self._phi3_success_count
   583→        qwen_total = self._qwen_success_count
   584→        total_tracked = phi3_total + qwen_total
   585→
   586→        if total_tracked < 10:  # Need minimum sample size
   587→            return
   588→
   589→        phi3_rate = phi3_total / total_tracked if total_tracked > 0 else 0.0
   590→
   591→        # If phi3 has >85% success rate, we can be more aggressive
   592→        if phi3_rate >= 0.85:
   593→            # Increase cost threshold (more aggressive cost cutting)
   594→            if self.cost_threshold < 0.25:
   595→                self.cost_threshold = min(0.25, self.cost_threshold + 0.01)
   596→            # Increase latency threshold tolerance
   597→            if self.latency_threshold < 250.0:
   598→                self.latency_threshold = min(250.0, self.latency_threshold + 5.0)
   599→
   600→        # If phi3 success rate is too low (<60%), be more conservative
   601→        elif phi3_rate < 0.60:
   602→            # Reduce cost threshold
   603→            if self.cost_threshold > 0.05:
   604→                self.cost_threshold = max(0.05, self.cost_threshold - 0.01)
   605→            # Reduce latency threshold tolerance
   606→            if self.latency_threshold > 100.0:
   607→                self.latency_threshold = max(100.0, self.latency_threshold - 5.0)

1. **Three disconnected model registries** — CostAwareRouter, DynamicModelRouter, and TierConfig each hardcode different model sets. Production alignment requires a shared registry.
2. **No journey tracking** — pool transitions (load/evict/promote) don't emit 12D events
3. **No circuit breakers** — bare try/except on Ollama calls instead of `get_circuit()`
4. **Sequential health checks** — `asyncio.gather()` would cut worst-case from 20s to 5s

**Phase A — Registry Unification** (HIGH value, blocks real pool→router integration):
- Create single `ModelRegistry` that CostAwareRouter, DynamicModelRouter, and TierConfig all read from
- Without this, pool fallback always hits the `.get(model, 0.0)` default path — works but loses cost/quality optimization

**Phase B — Pool Observability** (MEDIUM):
- Wire journey tracking into load/evict/promote/demote
- Add circuit breaker around Ollama calls
- Parallel health checks via `asyncio.gather()`

**Phase C — Auto-Promotion** (NICE-TO-HAVE):
- Track call frequency per model, auto-promote COLD→WARM after `promotion_threshold_calls`
