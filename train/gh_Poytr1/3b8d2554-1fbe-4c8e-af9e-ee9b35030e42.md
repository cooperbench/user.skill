# Rename Subscription Types: source→source_release, project→semantic_release

## Context

The current subscription type values `"source"` and `"project"` are confusing — they describe what entity you're subscribing to, not what you receive. Users expect to choose between "Source Releases" (raw release notifications) and "Semantic Releases" (AI-analyzed reports). Renaming to `source_release` / `semantic_release` makes the intent clear.

   396→
   397→### Endpoint
   398→
   399→| Method | Path               | Description                           |
   400→|--------|--------------------|---------------------------------------|
   401→| `GET`  | `/api/v1/events`   | SSE stream of real-time events        |
   402→
   403→### Architecture
   404→
   405→```
   406→PostgreSQL LISTEN/NOTIFY  →  Go listener goroutine  →  SSE broadcaster  →  connected clients
   407→```
   408→
   409→1. When a release is ingested (transactional outbox commits), a PostgreSQL trigger fires `NOTIFY release_events, '{...}'`
   410→2. A Go goroutine holds a persistent `LISTEN release_events` connection
   411→3. On notification, it broadcasts to all connected SSE clients via the `Broadcaster` (64-buffered channels, non-blocking send skips slow clients)
   412→
   413→### Event Types
   414→

   738→// EnqueueAgentRun creates an agent_run row and enqueues a River AgentJobArgs.
   739→// This follows the transactional outbox pattern for zero-loss guarantee.
   740→func (s *PgStore) EnqueueAgentRun(ctx context.Context, projectID, trigger, version string) error {
   741→	tx, err := s.pool.Begin(ctx)
   742→	if err != nil {
   743→		return fmt.Errorf("begin tx: %w", err)
   744→	}
   745→	defer tx.Rollback(ctx)
   746→
   747→	var runID string
   748→	err = tx.QueryRow(ctx,
   749→		`INSERT INTO agent_runs (project_id, trigger, version, status)
   750→		 VALUES ($1, $2, $3, 'pending')
   751→		 RETURNING id`, projectID, trigger, version,
   752→	).Scan(&runID)
   753→	if err != nil {
   754→		return fmt.Errorf("insert agent run: %w", err)
   755→	}
   756→
   757→	if s.river != nil {
   758→		_, err = s.river.InsertTx(ctx, tx, queue.AgentJobArgs{
   759→			AgentRunID: runID,
   760→			ProjectID:  projectID,
   761→			Version:    version,
   762→		}, nil)
   763→		if err != nil {
   764→			return fmt.Errorf("enqueue agent job: %w", err)
   765→		}
   766→	}
   767→
   768→	return tx.Commit(ctx)
   769→}

   799→	if err != nil {
   800→		return fmt.Errorf("begin tx: %w", err)
   801→	}
   802→	defer tx.Rollback(ctx)
   803→
   804→	err = tx.QueryRow(ctx,
   805→		`INSERT INTO semantic_releases (project_id, version, report, status, completed_at)
   806→		 VALUES ($1, $2, $3, $4, $5)
   807→		 RETURNING id, created_at`,
   808→		sr.ProjectID, sr.Version, sr.Report, sr.Status, sr.CompletedAt,
   809→	).Scan(&sr.ID, &sr.CreatedAt)
   810→	if err != nil {
   811→		return fmt.Errorf("insert semantic release: %w", err)
   812→	}
   813→
   814→	for _, relID := range releaseIDs {
   815→		_, err = tx.Exec(ctx,
   816→			`INSERT INTO semantic_release_sources (semantic_release_id, release_id)
   817→			 VALUES ($1, $2) ON CONFLICT DO NOTHING`,
   818→			sr.ID, relID,
   819→		)
   820→		if err != nil {
   821→			return fmt.Errorf("insert semantic release source: %w", err)
   822→		}
   823→	}
   824→
   825→	return tx.Commit(ctx)

func (s *PgStore) CreateSubscriptionBatch(ctx context.Context, subs []models.Subscription) ([]models.Subscription, error) {
	tx, err := s.pool.Begin(ctx)
	if err != nil {
		return nil, fmt.Errorf("begin tx: %w", err)
	}
	defer tx.Rollback(ctx)
	
	var result []models.Subscription
	for i := range subs {
		sub := subs[i]
		err := tx.QueryRow(ctx, `INSERT INTO...`).Scan(&sub.ID, &sub.CreatedAt)
		if err != nil {
			return nil, fmt.Errorf("insert subscription %d: %w", i, err)
		}
		result = append(result, sub)
	}
	
	if err := tx.Commit(ctx); err != nil {
		return nil, fmt.Errorf("commit tx: %w", err)
	}
	return result, nil
}
