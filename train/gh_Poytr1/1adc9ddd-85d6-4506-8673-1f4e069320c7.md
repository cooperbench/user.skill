```go
// CheckAgentRules evaluates agent rules against new/previous versions
// Returns true if ANY rule matches and an agent run should be triggered
func CheckAgentRules(rules *models.AgentRules, newVersion, previousVersion string) bool {
	if rules == nil {
		return false
	}

	newMajor, newMinor, _ := parseVersion(newVersion)
	oldMajor, oldMinor, _ := parseVersion(previousVersion)

	if rules.OnMajorRelease && newMajor > oldMajor {
		return true
	}
	if rules.OnMinorRelease && (newMajor > oldMajor || newMinor > oldMinor) {
		return true
	}
	if rules.OnSecurityPatch && isSecurityPatch(newVersion) {
		return true
	}
	if rules.VersionPattern != "" {
		if matched, _ := regexp.MatchString(rules.VersionPattern, newVersion); matched {
			return true
		}
	}
	return false
}

// isSecurityPatch detects version strings containing: "security", "cve", or "vuln"
func isSecurityPatch(version string) bool {
	lower := strings.ToLower(version)
	keywords := []string{"security", "cve", "vuln"}
	for _, kw := range keywords {
		if strings.Contains(lower, kw) {
			return true
		}
	}
	return false
}
```

```go
func (w *NotifyWorker) checkAgentRules(ctx context.Context, release *models.Release, sourceID string) {
	source, err := w.store.GetSource(ctx, sourceID)
	project, err := w.store.GetProject(ctx, source.ProjectID)
	
	// Parse project's agent_rules JSON

	if len(project.AgentRules) > 0 {

			slog.Error("unmarshal agent rules", "project_id", project.ID, "err", err)
```

```go
func (s *PgStore) EnqueueAgentRun(ctx context.Context, projectID, trigger string) error {
	tx, err := s.pool.Begin(ctx)
	defer tx.Rollback(ctx)
	
	// Insert agent_run row
	var runID string
	err = tx.QueryRow(ctx,
		`INSERT INTO agent_runs (project_id, trigger, status)
		 VALUES ($1, $2, 'pending')
		 RETURNING id`, projectID, trigger,
	).Scan(&runID)
	
	// Enqueue River job in same transaction (zero-loss guarantee)
	if s.river != nil {
		_, err = s.river.InsertTx(ctx, tx, queue.AgentJobArgs{
			AgentRunID: runID,
			ProjectID:  projectID,
		}, nil)
	}
	
	return tx.Commit(ctx)
}
```

#### **Trigger conditions:**

1. **OnMajorRelease**: Triggered when `newMajor > oldMajor`

2. **OnMinorRelease**: Triggered when `newMajor > oldMajor OR newMinor > oldMinor`

4. **VersionPattern**: Custom regex pattern matches the version string

#### **Special cases:**
- **Nil or empty rules**: Returns `false` (no trigger)
- **Empty previous version**: Compared against version 0.0.0, so v1.0.0 with `OnMajorRelease=true` will trigger
- **Version parsing**: Strips leading "v" and tolerates pre-release suffixes

```go
func (s *PgStore) EnqueueAgentRun(ctx context.Context, projectID, trigger string) error {
    // 1. Start a transaction
    tx, err := s.pool.Begin(ctx)
    if err != nil {
        return fmt.Errorf("begin tx: %w", err)
    }
    defer tx.Rollback(ctx)

    // 2. INSERT into agent_runs table, get the generated ID
    var runID string
    err = tx.QueryRow(ctx,
        `INSERT INTO agent_runs (project_id, trigger, status)
         VALUES ($1, $2, 'pending')
         RETURNING id`, projectID, trigger,
    ).Scan(&runID)
    if err != nil {
        return fmt.Errorf("insert agent run: %w", err)
    }

    // 3. Enqueue River job with the generated runID
    if s.river != nil {
        _, err = s.river.InsertTx(ctx, tx, queue.AgentJobArgs{
            AgentRunID: runID,
            ProjectID:  projectID,
        }, nil)
        if err != nil {
            return fmt.Errorf("enqueue agent job: %w", err)
        }
    }

    // 4. Commit transaction (transactional outbox pattern)
    return tx.Commit(ctx)
}
```

**Key points:**
- **Transactional atomicity**: Both INSERT and job enqueue happen in same transaction
- **Trigger string format**: `"auto:version:<version>"` (e.g., `"auto:version:v1.5.0"`)
- **AgentRun status**: Creates record with status `'pending'`
- **River job**: Only enqueued if `s.river` is configured (gracefully skips if nil)

```go
func (o *Orchestrator) RunAgent(ctx context.Context, run *models.AgentRun) error {
    // 1. Mark run as "running"
    if err := o.store.UpdateAgentRunStatus(ctx, run.ID, "running"); err != nil {
        return fmt.Errorf("update agent run status to running: %w", err)
    }

    // 2. Execute the agent (LLM interaction)
    result, err := o.executeAgent(ctx, run)
    if err != nil {
        slog.Error("agent run failed", "run_id", run.ID, "err", err)
        _ = o.store.UpdateAgentRunStatus(ctx, run.ID, "failed")
        return err
    }

    // 3. Link semantic release to agent run
    if err := o.store.UpdateAgentRunResult(ctx, run.ID, result.semanticReleaseID); err != nil {
        return fmt.Errorf("update agent run result: %w", err)
    }

    // 4. Mark run as "completed"
    if err := o.store.UpdateAgentRunStatus(ctx, run.ID, "completed"); err != nil {
        return fmt.Errorf("update agent run status to completed: %w", err)
    }

    // 5. Send project-level notifications (best-effort)
    o.sendProjectNotifications(ctx, run, result)

    return nil
}
```
