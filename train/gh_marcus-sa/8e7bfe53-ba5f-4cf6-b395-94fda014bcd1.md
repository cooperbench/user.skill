   241→export async function collectSseEvents<T extends { type: string }>(streamUrl: string, timeoutMs: number): Promise<T[]> {
   242→  const response = await fetch(streamUrl, { headers: { Accept: "text/event-stream" } });
   243→  if (!response.ok || !response.body) {
   244→    throw new Error(`Failed to open SSE stream (${response.status})`);
   245→  }
   246→
   247→  const reader = response.body.getReader();
   248→  const decoder = new TextDecoder();
   249→  const events: T[] = [];
   250→  let buffer = "";
   251→
   252→  const timeout = setTimeout(() => {
   253→    void reader.cancel();
   254→  }, timeoutMs);
   255→
   256→  try {
   257→    while (true) {
   258→      const { value, done } = await reader.read();
   259→      if (done) {
   260→        break;
   261→      }
   262→
   263→      buffer += decoder.decode(value, { stream: true });
   264→      const segments = buffer.split("\n\n");
   265→      buffer = segments.pop() ?? "";
   266→
   267→      for (const segment of segments) {
   268→        const dataLine = segment.split("\n").find((line) => line.startsWith("data: "));
   269→        if (!dataLine) {
   270→          continue;
   271→        }
   272→
   273→        const event = JSON.parse(dataLine.slice("data: ".length)) as T;
   274→        events.push(event);
   275→
   276→        if (event.type === "error") {
   277→          const errorText = (event as { error?: string }).error;
   278→          throw new Error(`SSE error event: ${errorText ?? "unknown error"}`);
   279→        }
   280→
   281→        if (event.type === "done") {
   282→          return events;
   283→        }
   284→      }
   285→    }
   286→  } finally {
   287→    clearTimeout(timeout);
   288→    reader.releaseLock();
   289→  }
   290→
   291→  return events;
   292→}

    42→### Dependency Justification
    43→
    44→| Need | Solution | Why No New Dependency |
    45→|---|---|---|
    46→| ES256 key pair generation | `crypto.subtle.generateKey()` (Web Crypto) | Built into Bun runtime |
    47→| JWK thumbprint (RFC 7638) | `jose.calculateJwkThumbprint()` | Already in jose |
    48→| DPoP proof creation/validation | `jose.SignJWT` / `jose.jwtVerify` | Already in jose |

    54→## Technology Decisions
    55→
    56→### Why NOT a Separate DPoP Library
    57→
    58→Several npm packages implement DPoP proof handling (e.g., `dpop`, `oauth4webapi`). These are rejected because:
    59→
    60→1. `jose` already covers all JWT/JWK operations needed for DPoP
    61→2. DPoP proof is a JWT with specific claims -- `jose.SignJWT` + `jose.jwtVerify` handle this directly
    62→3. Adding a DPoP-specific library creates an unnecessary abstraction layer over `jose`
    63→4. The Brain's DPoP implementation has custom requirements (brain_action integration, intent binding) that generic libraries do not cover
    64→
    65→### Why NOT Separate JWKS for Custom AS
    66→
    67→The Custom AS needs its own signing key (separate from Better Auth's JWKS). Options:
    68→
    69→1. **Separate JWKS endpoint** (`/api/auth/brain/.well-known/jwks`) -- **Selected**. Clean separation. Resource server fetches AS public key via standard JWKS protocol.
    70→2. **Shared Better Auth JWKS** -- Rejected. Custom AS tokens have different claims and lifetime from Better Auth tokens. Sharing keys conflates trust boundaries.
    71→3. **Hardcoded public key in resource server** -- Rejected. No key rotation capability. Violates security best practices.
    72→
    73→### Nonce Cache Implementation
    74→
    75→Options for replay protection nonce storage:
    76→
    77→1. **In-memory time-windowed Map** -- **Selected**. Simple. Dependency-injected (not module singleton per project conventions). Entries auto-expire. Sufficient for single-process deployment.
    78→2. **Redis/external store** -- Rejected. No Redis in the stack. Over-engineered for single-process monolith.
    79→3. **SurrealDB table** -- Rejected. Nonce lookups must be < 1ms (NFR-1). DB round-trip too slow for per-request check.

   241→export async function collectSseEvents<T extends { type: string }>(streamUrl: string, timeoutMs: number): Promise<T[]> {
   242→  const response = await fetch(streamUrl, { headers: { Accept: "text/event-stream" } });
   243→  if (!response.ok || !response.body) {
   244→    throw new Error(`Failed to open SSE stream (${response.status})`);

   252→  const timeout = setTimeout(() => {
   253→    void reader.cancel();
   254→  }, timeoutMs);

   259→      if (done) {
   260→        break;
   261→      }
   262→
   263→      buffer += decoder.decode(value, { stream: true });
   264→      const segments = buffer.split("\n\n");
   265→      buffer = segments.pop() ?? "";
   266→
   267→      for (const segment of segments) {
   268→        const dataLine = segment.split("\n").find((line) => line.startsWith("data: "));
   269→        if (!dataLine) {
   270→          continue;
   271→        }
   272→
   273→        const event = JSON.parse(dataLine.slice("data: ".length)) as T;
   274→        events.push(event);

   276→        if (event.type === "error") {
   277→          const errorText = (event as { error?: string }).error;

     5→## US-001: DPoP Key Pair Lifecycle for All Actors
     6→
     7→### Problem
     8→
     9→Agent "Kira" (code_agent in E2B sandbox, workspace "Lusaka") and human operator Marcus Santos (dashboard user) currently have no sender-constraining mechanism. Kira authenticates with a long-lived Bearer token that any interceptor can replay. Marcus authenticates with a session cookie that any XSS attack can steal. Neither can prove that a token presentation came from the same entity the token was issued to.
    10→
    11→### Who
    12→
    13→- AI agent runtime | Running in isolated sandbox (E2B) | Needs cryptographic identity for proof-of-possession
    14→- Human operator | Using dashboard in browser | Needs browser-based cryptographic identity for Bridge tokens
    15→
    16→### Solution
    17→
    18→All actor runtimes (agent sandboxes, browser dashboard, CLI tools) generate and manage an ES256 key pair per session, computing a JWK thumbprint that serves as the binding identifier for all DPoP-bound tokens in that session.
    19→
    20→### Domain Examples
    21→
    22→#### 1: Agent Session Start Key Generation
    23→Agent "Kira" starts a new session in workspace "Lusaka" for task T-4821 (Q1 invoicing for Acme Corp). During session initialization, the runtime generates an ES256 key pair. The private key is held in memory. The JWK thumbprint `REDACTED` is computed from the public key. Generation completes in under 50ms.
    24→
    25→#### 2: Browser Session Key Generation (Bridge)
    26→Marcus Santos opens the Brain dashboard in Chrome. The dashboard client library generates an ES256 key pair via Web Crypto API. The private key is held in browser memory (CryptoKey object, non-extractable). The thumbprint `BrowserKey-marcus-abc123` is computed. This key pair is used for all Bridge token exchanges during this browser session.
    27→
    28→#### 3: Key Destruction on Runtime Termination
    29→Agent "Kira" completes task T-4821 and the E2B sandbox shuts down. The in-memory private key is destroyed with the process. Marcus closes the dashboard tab -- the browser CryptoKey is garbage collected. Any tokens previously issued with `cnf.jkt` bound to these key pairs become unusable.
    30→

    38→And the JWK thumbprint is computed from the public key via RFC 7638
    39→
    40→#### Scenario: Browser key pair generated for dashboard session
