52	  suiteName: string | null;

55	  pollMs?: number;

73	function LiveRunProgressBody({
74	  runId,
75	  canManage,
76	  pollMs,
77	}: {
78	  runId: string;
79	  canManage: boolean;
80	  pollMs: number;
81	}) {

84	  const [error, setError] = useState<string | null>(null);
85	  const [cancelling, setCancelling] = useState(false);
86	  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
87	  // `stopped` latches once the run reaches a terminal state (poll-observed or
88	  // cancel-forced). It guards against a poll that was already in flight before a
89	  // cancel resolving afterwards and clobbering the terminal status back to

91	  // pre-cancel state). Terminal is sticky.

94	  // Poll until terminal via a self-scheduling timeout (not setInterval, so a slow
95	  // request can't pile up overlapping fetches). `active` guards a late resolution
96	  // after unmount. A transient fetch error keeps polling (the live view
97	  // self-heals when the endpoint recovers) — only a terminal status stops it.
98	  useEffect(() => {
99	    let active = true;
100	    stoppedRef.current = false;
101	    const tick = async () => {
102	      try {
103	        const next = await getRunProgress(runId);
104	        if (!active || stoppedRef.current) return;
105	        setProgress(next);
106	        setError(null);
107	        if (isTerminal(next.status)) {
108	          stoppedRef.current = true;
109	        } else {

111	        }
112	      } catch (err) {
113	        if (!active || stoppedRef.current) return;
114	        setError(err instanceof Error ? err.message : String(err));
115	        // Keep polling through a transient error rather than freezing the live
116	        // view; the cadence is bounded by pollMs, and a terminal status / unmount
117	        // still stops it.
118	        timerRef.current = setTimeout(tick, pollMs);
119	      }
120	    };
121	    void tick();

123	      active = false;
124	      if (timerRef.current) clearTimeout(timerRef.current);
125	    };

197	          back to this run until the recent-runs table lands. */}
198	      <Link to="/results">View full results →</Link>
199	    </Flex>

206	 * produced a result — e.g. a cancelled run) it shows a neutral "not run" rather
