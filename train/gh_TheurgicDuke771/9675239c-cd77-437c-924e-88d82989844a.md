148	def _poll_orchestration_runs(
149	    session: Session,
150	    *,
151	    secret_store: SecretStore,
152	    now: datetime | None = None,
153	    lookback: timedelta = _POLL_LOOKBACK,
154	) -> dict[str, int]:
155	    """Poll every orchestrator connection for recent succeeded runs (#171, ADR 0004).
156	
157	    The polling fallback for runs that never produced a webhook: iterate each
158	    ADF / Airflow connection, ask the provider's `list_recent_runs` for runs
159	    updated within the ``lookback`` window, and hand them to `ingest_polled_runs`
160	    (upsert + trigger-on-success). Goes through the `OrchestrationProvider` seam —
161	    no per-provider branching. Each connection is isolated: a transport/auth
162	    failure logs + continues so one bad connection can't starve the rest.
163	
164	    ``lookback`` widens for gap recovery (B2): the same sweep over a 1-hour window
165	    re-ingests runs missed during downtime. ``skip_updated_since`` rides the same
166	    window, so a run we already recorded inside it is skipped while a genuinely
167	    missed one (no row) is upserted.
168	    """
169	    since = (now or datetime.now(UTC)) - lookback
170	    summary = {"connections": 0, "recorded": 0, "triggered": 0, "skipped": 0, "errors": 0}
171	    connections = list(
172	        session.scalars(
173	            select(Connection).where(
174	                Connection.type.in_(ORCHESTRATION_PROVIDERS),
175	                Connection.secret_ref.isnot(None),
176	            )
177	        )
178	    )
179	    for connection in connections:
180	        if not connection.secret_ref:  # defensive; the query already filters
181	            continue

183	            provider_impl = get_orchestration_provider(connection.type)
184	            secret = secret_store.get(connection.secret_ref)
185	            updates = provider_impl.list_recent_runs(dict(connection.config), secret, since)
186	            result = orchestration_service.ingest_polled_runs(
187	                session,
188	                provider_impl=provider_impl,
189	                connection=connection,
190	                updates=updates,

195	            summary["triggered"] += len(result.triggered_runs)

242	def _advance_schedule(schedule: Schedule, *, now: datetime) -> bool:
243	    """Roll ``schedule`` forward to its next future fire and stamp ``last_run_at``.
244	
245	    **No-backfill semantics**: ``cron.next_fire`` returns the next occurrence
246	    strictly after ``now``, so a gap (worker/beat down across several slots) is
247	    collapsed to a single fire rather than backfilled. Returns True if advanced;
248	    False (and disables the schedule) if the stored cron/tz is somehow invalid —
249	    validated on write, so this only guards against direct DB tampering and stops
250	    an un-advanceable row from hot-looping the dispatcher every tick.
251	    """
252	    schedule.last_run_at = now
253	    try:
254	        schedule.next_run_at = cron.next_fire(schedule.cron, schedule.timezone, after=now)
255	    except DataQError:
256	        schedule.enabled = False
257	        log.error(
258	            "schedule_disabled_invalid_cron",
259	            schedule_id=str(schedule.id),
260	            cron=schedule.cron,
261	            timezone=schedule.timezone,
262	        )
263	        return False
264	    return True

267	def _fire_schedule(session: Session, schedule: Schedule, *, now: datetime) -> str:
268	    """Fire one due schedule: advance it, then queue + dispatch a suite run.
269	
270	    Advancing ``next_run_at`` happens **before** the run is created and is
271	    committed in every branch, so the schedule leaves the due window for this
272	    tick whatever the run's fate — a misconfigured suite never hot-loops. The run
273	    is created with the canonical ``schedule:<id>`` ``triggered_by`` marker and
274	    handed to the worker exactly like the manual / pipeline-trigger paths; a
275	    targetless suite is skipped (not queued-then-failed), and a broker outage
276	    marks the run ``failed`` rather than leaving it stuck ``queued`` (#227).
277	    """
278	    if not _advance_schedule(schedule, now=now):
279	        session.commit()
280	        return "disabled"
281	
282	    suite = session.get(Suite, schedule.suite_id)
283	    assert suite is not None  # schedule cascade-deletes with its suite
284	    connection = session.get(Connection, suite.connection_id)
285	    assert connection is not None  # suite.connection_id FK is RESTRICT
286	    try:
287	        run_target.resolve_target(connection.type, suite.target)
288	    except DataQError:
289	        session.commit()  # persist the advance; skip the doomed run
290	        log.warning(
291	            "schedule_skipped_invalid_target",
292	            schedule_id=str(schedule.id),
293	            suite_id=str(suite.id),
294	        )
295	        return "skipped_target"
296	
297	    run = Run(suite_id=suite.id, status="queued", triggered_by=f"schedule:{schedule.id}")
298	    session.add(run)
299	    session.commit()
300	    session.refresh(run)
301	    # Shared dispatch+broker-failure block (#227): on failure the run is marked
302	    # terminal-`failed` and logged (with schedule_id kept on the event); the
303	    # advance is already committed, so the schedule has left the due window.
304	    if not run_dispatch.dispatch_or_fail(session, run, schedule_id=str(schedule.id)):
305	        return "dispatch_failed"
306	    log.info("schedule_fired", schedule_id=str(schedule.id), run_id=str(run.id))
307	    return "dispatched"

310	def _dispatch_due_schedules(session: Session, *, now: datetime | None = None) -> dict[str, int]:
311	    """Fire every enabled schedule whose ``next_run_at`` has passed (A7).
312	
313	    Pulls due schedules one at a time with ``FOR UPDATE SKIP LOCKED`` so two
314	    overlapping dispatcher ticks can't double-fire the same schedule: the second
315	    skips a row the first holds, and once fired the row's ``next_run_at`` is past
316	    ``now`` so it drops out of the due set. ``now`` is fixed at entry, so the loop
317	    is finite (each iteration advances one row out of the window).
318	    """
319	    now = now or datetime.now(UTC)
320	    summary = {"due": 0, "dispatched": 0, "skipped_target": 0, "dispatch_failed": 0, "disabled": 0}
321	    while True:
322	        schedule = session.scalars(
323	            select(Schedule)
324	            .where(Schedule.enabled.is_(True), Schedule.next_run_at <= now)
325	            .order_by(Schedule.next_run_at)
326	            .with_for_update(skip_locked=True)
327	            .limit(1)
328	        ).first()
329	        if schedule is None:
330	            break
331	        summary["due"] += 1
332	        outcome = _fire_schedule(session, schedule, now=now)
333	        summary[outcome] = summary.get(outcome, 0) + 1
334	    log.info("schedules_dispatch_completed", **summary)
335	    return summary
