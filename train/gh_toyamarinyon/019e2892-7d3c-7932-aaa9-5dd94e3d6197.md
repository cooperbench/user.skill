```
			const normalized = answer.trim().toLowerCase();
			const result =
				normalized === "" || normalized === "y" || normalized === "yes";
			if (process.stdout.isTTY) {
				process.stdout.write("\x1b[1A\x1b[2K");
			} else if (!process.stdin.isTTY) {
				process.stdout.write("\n");
			}

		},
		close(): void {
			rl.close();
		},

async function confirmWithKeypress(
	message: string,
	defaultValue = true,
): Promise<boolean | "cancel"> {
	let selected = defaultValue;
	const render = () => {
		process.stdout.write(
			`${CLEAR_LINE}${color.cyan("?")} ${message} ${color.cyan("›")} ${formatChoices(selected)}`,
		);
	};
	const renderDone = () => {
		process.stdout.write(
			`${CLEAR_LINE}${color.green("✔")} ${message} ${color.cyan("›")} ${formatChoices(selected)}\n`,
		);
	};

	emitKeypressEvents(process.stdin);
	process.stdin.setRawMode(true);
	process.stdin.resume();
	process.stdout.write(HIDE_CURSOR);
	render();

	return new Promise((resolve) => {
		const cleanup = () => {
			process.stdin.off("keypress", onKeypress);
			process.stdin.setRawMode(false);
			process.stdout.write(SHOW_CURSOR);
			rlResumeIfNeeded();
		};
		const finish = (value: boolean | "cancel") => {
			cleanup();
			if (value === "cancel") {
				process.stdout.write(`${CLEAR_LINE}`);
			} else {
				renderDone();
			}
			resolve(value);
		};
		const onKeypress = (
			_chunk: string,
			key: { name?: string; ctrl?: boolean },
		) => {
			if (key.ctrl && key.name === "c") {
				finish("cancel");
				return;
			}
			if (key.name === "left" || key.name === "right") {
				selected = !selected;
				render();
				return;
			}
			if (key.name === "tab") {
				selected = !selected;
				render();
				return;
			}
			if (key.name === "return" || key.name === "enter") {
				finish(selected);
			}
		};

		process.stdin.on("keypress", onKeypress);
	});
}

function rlResumeIfNeeded(): void {
	if (!process.stdin.isPaused()) {
		return;
	}
	process.stdin.resume();
```

原因はほぼこれで良さそうです:

```diff
"bin": {

}
```

```text
bin/eslint-plugin-raula.js
```

検証状況:

```text
npm pack --dry-run --cache /private/tmp/eslint-plugin-raula-npm-cache
```

これは成功して、CLI entrypoint も含まれていました。

```text
npm publish --dry-run --ignore-scripts --cache /private/tmp/eslint-plugin-raula-npm-cache
```

これも成功し、警告は出ませんでした。
