   423→// calculateTokenUsage calculates token usage from raw transcript data.
   424→// startOffset is the line number (Claude Code) or message index (Gemini CLI)

   426→// of the transcript since the last checkpoint.
   427→func calculateTokenUsage(agentType agent.AgentType, data []byte, startOffset int) *agent.TokenUsage {
   428→	if len(data) == 0 {
   429→		return &agent.TokenUsage{}
   430→	}
   431→
   432→	// Try Gemini format first if agentType is Gemini, or as fallback if Unknown
   433→	if agentType == agent.AgentTypeGemini || agentType == agent.AgentTypeUnknown {
   434→		// Attempt to parse as Gemini JSON
   435→		transcript, err := geminicli.ParseTranscript(data)
   436→		if err == nil && transcript != nil && len(transcript.Messages) > 0 {
   437→			return geminicli.CalculateTokenUsage(data, startOffset)
   438→		}
   439→		// If agentType is explicitly Gemini but parsing failed, return empty usage
   440→		if agentType == agent.AgentTypeGemini {
   441→			return &agent.TokenUsage{}
   442→		}
   443→		// Otherwise fall through to JSONL parsing for Unknown type
   444→	}
   445→
   446→	// Claude Code and other JSONL-based agents
   447→	lines, err := claudecode.ParseTranscript(data)
   448→	if err != nil || len(lines) == 0 {
   449→		return &agent.TokenUsage{}
   450→	}

   452→	if startOffset > 0 && startOffset < len(lines) {
   453→		lines = lines[startOffset:]
   454→	}
   455→	return claudecode.CalculateTokenUsage(lines)

   458→// extractUserPromptsFromLines extracts user prompts from JSONL transcript lines.

   476→			continue

   489→				prompts = append(prompts, cleaned)
   490→			}
   491→			continue

   493→
   494→		// Handle array content (e.g., multiple text blocks from VSCode)
   495→		if arr, ok := message["content"].([]interface{}); ok {
   496→			var texts []string
   497→			for _, item := range arr {

   106→// CondenseSession condenses a session's shadow branch to permanent storage.
   107→// checkpointID is the 12-hex-char value from the Entire-Checkpoint trailer.
   108→// Metadata is stored at sharded path: <checkpoint_id[:2]>/<checkpoint_id[2:]>/
   109→// Uses checkpoint.GitStore.WriteCommitted for the git operations.
   110→func (s *ManualCommitStrategy) CondenseSession(repo *git.Repository, checkpointID id.CheckpointID, state *SessionState) (*CondenseResult, error) {
   111→	// Get shadow branch
   112→	shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   113→	refName := plumbing.NewBranchReferenceName(shadowBranchName)
   114→	ref, err := repo.Reference(refName, true)
   115→	if err != nil {
   116→		return nil, fmt.Errorf("shadow branch not found: %w", err)
   117→	}
   118→
   119→	// Extract session data from the shadow branch (with live transcript fallback).
   120→	// Use tracked files from session state instead of collecting all files from tree.
   121→	// Pass agent type to handle different transcript formats (JSONL for Claude, JSON for Gemini).
   122→	// Pass live transcript path so condensation reads the current file rather than a
   123→	// potentially stale shadow branch copy (SaveChanges may have been skipped if the
   124→	// last turn had no code changes).
   125→	// Pass CheckpointTranscriptStart for accurate token calculation (line offset for Claude, message index for Gemini).
   126→	sessionData, err := s.extractSessionData(repo, ref.Hash(), state.SessionID, state.FilesTouched, state.AgentType, state.TranscriptPath, state.CheckpointTranscriptStart)
   127→	if err != nil {
   128→		return nil, fmt.Errorf("failed to extract session data: %w", err)
   129→	}
   130→
   131→	// Get checkpoint store
   132→	store, err := s.getCheckpointStore()
   133→	if err != nil {
   134→		return nil, fmt.Errorf("failed to get checkpoint store: %w", err)
   135→	}
   136→
   137→	// Get author info
   138→	authorName, authorEmail := GetGitAuthorFromRepo(repo)
   139→	attribution := calculateSessionAttributions(repo, ref, sessionData, state)
   140→	// Get current branch name
   141→	branchName := GetCurrentBranchName(repo)
   142→
   143→	// Generate summary if enabled
   144→	var summary *cpkg.Summary
   145→	if settings.IsSummarizeEnabled() && len(sessionData.Transcript) > 0 {
   146→		logCtx := logging.WithComponent(context.Background(), "attribution")
   147→		summarizeCtx := logging.WithComponent(logCtx, "summarize")
   148→
   149→		// Scope transcript to this checkpoint's portion
   150→		scopedTranscript := transcript.SliceFromLine(sessionData.Transcript, state.CheckpointTranscriptStart)
   151→		if len(scopedTranscript) > 0 {
   152→			var err error
   153→			summary, err = summarize.GenerateFromTranscript(summarizeCtx, scopedTranscript, sessionData.FilesTouched, nil)
   154→			if err != nil {
   155→				logging.Warn(summarizeCtx, "summary generation failed",
   156→					slog.String("session_id", state.SessionID),
   157→					slog.String("error", err.Error()))
   158→				// Continue without summary - non-blocking
   159→			} else {
   160→				logging.Info(summarizeCtx, "summary generated",
   161→					slog.String("session_id", state.SessionID))
   162→			}
   163→		}
   164→	}
   165→
   166→	// Write checkpoint metadata using the checkpoint store
   167→	if err := store.WriteCommitted(context.Background(), cpkg.WriteCommittedOptions{
   168→		CheckpointID:                checkpointID,
   169→		SessionID:                   state.SessionID,
   170→		Strategy:                    StrategyNameManualCommit,
   171→		Branch:                      branchName,
   172→		Transcript:                  sessionData.Transcript,
   173→		Prompts:                     sessionData.Prompts,
   174→		Context:                     sessionData.Context,
   175→		FilesTouched:                sessionData.FilesTouched,
   176→		CheckpointsCount:            state.StepCount,
   177→		EphemeralBranch:             shadowBranchName,
   178→		AuthorName:                  authorName,
   179→		AuthorEmail:                 authorEmail,
   180→		Agent:                       state.AgentType,
   181→		TranscriptIdentifierAtStart: state.TranscriptIdentifierAtStart,
   182→		CheckpointTranscriptStart:   state.CheckpointTranscriptStart,
   183→		TokenUsage:                  sessionData.TokenUsage,
   184→		InitialAttribution:          attribution,
   185→		Summary:                     summary,
   186→	}); err != nil {
   187→		return nil, fmt.Errorf("failed to write checkpoint metadata: %w", err)
   188→	}
   189→
   190→	return &CondenseResult{
   191→		CheckpointID:         checkpointID,
   192→		SessionID:            state.SessionID,
   193→		CheckpointsCount:     state.StepCount,
   194→		FilesTouched:         sessionData.FilesTouched,
   195→		TotalTranscriptLines: sessionData.FullTranscriptLines,
   196→	}, nil
   197→}

   199→func calculateSessionAttributions(repo *git.Repository, shadowRef *plumbing.Reference, sessionData *ExtractedSessionData, state *SessionState) *cpkg.InitialAttribution {
   200→	// Calculate initial attribution using accumulated prompt attribution data.
   201→	// This uses user edits captured at each prompt start (before agent works),
   202→	// plus any user edits after the final checkpoint (shadow → head).
   203→	logCtx := logging.WithComponent(context.Background(), "attribution")
   204→	var attribution *cpkg.InitialAttribution
   205→	headRef, headErr := repo.Head()
   206→	if headErr != nil {
   207→		logging.Debug(logCtx, "attribution skipped: failed to get HEAD",
   208→			slog.String("error", headErr.Error()))
   209→	} else {
   210→		headCommit, commitErr := repo.CommitObject(headRef.Hash())
   211→		if commitErr != nil {
   212→			logging.Debug(logCtx, "attribution skipped: failed to get HEAD commit",
   213→				slog.String("error", commitErr.Error()))
   214→		} else {
   215→			headTree, treeErr := headCommit.Tree()
   216→			if treeErr != nil {
   217→				logging.Debug(logCtx, "attribution skipped: failed to get HEAD tree",
   218→					slog.String("error", treeErr.Error()))
   219→			} else {
   220→				// Get shadow branch tree (checkpoint tree - what the agent wrote)
   221→				shadowCommit, shadowErr := repo.CommitObject(shadowRef.Hash())
   222→				if shadowErr != nil {
   223→					logging.Debug(logCtx, "attribution skipped: failed to get shadow commit",
   224→						slog.String("error", shadowErr.Error()),
   225→						slog.String("shadow_ref", shadowRef.Hash().String()))
   226→				} else {
   227→					shadowTree, shadowTreeErr := shadowCommit.Tree()
   228→					if shadowTreeErr != nil {
   229→						logging.Debug(logCtx, "attribution skipped: failed to get shadow tree",
   230→							slog.String("error", shadowTreeErr.Error()))
   231→					} else {
   232→						// Get base tree (state before session started)
   233→						var baseTree *object.Tree
   234→						attrBase := state.AttributionBaseCommit
   235→						if attrBase == "" {
   236→							attrBase = state.BaseCommit // backward compat
   237→						}
   238→						if baseCommit, baseErr := repo.CommitObject(plumbing.NewHash(attrBase)); baseErr == nil {
   239→							if tree, baseTErr := baseCommit.Tree(); baseTErr == nil {
   240→								baseTree = tree
   241→							} else {
   242→								logging.Debug(logCtx, "attribution: base tree unavailable",
   243→									slog.String("error", baseTErr.Error()))
   244→							}
   245→						} else {
   246→							logging.Debug(logCtx, "attribution: base commit unavailable",
   247→								slog.String("error", baseErr.Error()),
   248→								slog.String("attribution_base", attrBase))
   249→						}
   250→
   251→						// Log accumulated prompt attributions for debugging
   252→						var totalUserAdded, totalUserRemoved int
   253→						for i, pa := range state.PromptAttributions {
   254→							totalUserAdded += pa.UserLinesAdded
   255→							totalUserRemoved += pa.UserLinesRemoved
   256→							logging.Debug(logCtx, "prompt attribution data",
   257→								slog.Int("checkpoint", pa.CheckpointNumber),
   258→								slog.Int("user_added", pa.UserLinesAdded),
   259→								slog.Int("user_removed", pa.UserLinesRemoved),
   260→								slog.Int("agent_added", pa.AgentLinesAdded),
   261→								slog.Int("agent_removed", pa.AgentLinesRemoved),
   262→								slog.Int("index", i))
   263→						}
   264→
   265→						attribution = CalculateAttributionWithAccumulated(
   266→							baseTree,
   267→							shadowTree,
   268→							headTree,
   269→							sessionData.FilesTouched,
   270→							state.PromptAttributions,
   271→						)
   272→
   273→						if attribution != nil {
   274→							logging.Info(logCtx, "attribution calculated",
   275→								slog.Int("agent_lines", attribution.AgentLines),
   276→								slog.Int("human_added", attribution.HumanAdded),
   277→								slog.Int("human_modified", attribution.HumanModified),
   278→								slog.Int("human_removed", attribution.HumanRemoved),
   279→								slog.Int("total_committed", attribution.TotalCommitted),
   280→								slog.Float64("agent_percentage", attribution.AgentPercentage),
   281→								slog.Int("accumulated_user_added", totalUserAdded),
   282→								slog.Int("accumulated_user_removed", totalUserRemoved),
   283→								slog.Int("files_touched", len(sessionData.FilesTouched)))
   284→						}
   285→					}
   286→				}
   287→			}
   288→		}
   289→	}
   290→	return attribution
   291→}

   403→// isGeminiJSONTranscript detects if the transcript is in Gemini's JSON format.
   404→// Gemini transcripts are JSON objects containing a "messages" array.
   405→// Returns true if the content has a valid "messages" field, even if empty.
   406→func isGeminiJSONTranscript(content string) bool {
   407→	content = strings.TrimSpace(content)
   408→	// Quick check: Gemini JSON starts with {
   409→	if !strings.HasPrefix(content, "{") {
   410→		return false
   411→	}
   412→	// Try to parse as Gemini format - check for messages field existence
   413→	var transcript struct {
   414→		Messages []json.RawMessage `json:"messages"`
   415→	}
   416→	if err := json.Unmarshal([]byte(content), &transcript); err != nil {
   417→		return false
   418→	}
   419→	return transcript.Messages != nil
   420→}

   434→// extractUserPromptsFromGeminiJSON extracts user prompts from Gemini's JSON transcript format.
   435→// Gemini transcripts are structured as: {"messages": [{"type": "user", "content": "..."}, ...]}
   436→func extractUserPromptsFromGeminiJSON(content string) []string {
   437→	var transcript struct {
   438→		Messages []struct {
   439→			Type    string `json:"type"`
   440→			Content string `json:"content"`
   441→		} `json:"messages"`
   442→	}
   443→
   444→	if err := json.Unmarshal([]byte(content), &transcript); err != nil {
   445→		return nil
   446→	}
   447→
   448→	var prompts []string
   449→	for _, msg := range transcript.Messages {
   450→		if msg.Type == "user" && msg.Content != "" {
   451→			// Strip IDE context tags for consistency with Claude Code handling
   452→			cleaned := textutil.StripIDEContextTags(msg.Content)
   453→			if cleaned != "" {
   454→				prompts = append(prompts, cleaned)
   455→			}
   456→		}
   457→	}
   458→
   459→	return prompts
   460→}

   462→// extractUserPromptsFromLines extracts user prompts from JSONL transcript lines.
   463→// IDE-injected context tags (like <ide_opened_file>) are stripped from the results.
   464→func extractUserPromptsFromLines(lines []string) []string {
   465→	var prompts []string
   466→	for _, line := range lines {
   467→		line = strings.TrimSpace(line)
   468→		if line == "" {
   469→			continue
   470→		}
   471→
   472→		var entry map[string]interface{}
   473→		if err := json.Unmarshal([]byte(line), &entry); err != nil {
   474→			continue
   475→		}
   476→
   477→		// Check for user message (supports both "human" and "user" types)
   478→		msgType, ok := entry["type"].(string)
   479→		if !ok || (msgType != "human" && msgType != "user") {
   480→			continue
   481→		}
   482→
   483→		// Extract message content
   484→		message, ok := entry["message"].(map[string]interface{})
   485→		if !ok {
   486→			continue
   487→		}
   488→
   489→		// Handle string content
   490→		if content, ok := message["content"].(string); ok && content != "" {
   491→			cleaned := textutil.StripIDEContextTags(content)
   492→			if cleaned != "" {
   493→				prompts = append(prompts, cleaned)
   494→			}
   495→			continue
   496→		}
   497→
   498→		// Handle array content (e.g., multiple text blocks from VSCode)
   499→		if arr, ok := message["content"].([]interface{}); ok {
   500→			var texts []string
   501→			for _, item := range arr {
   502→				if m, ok := item.(map[string]interface{}); ok {
   503→					if m["type"] == "text" {
   504→						if text, ok := m["text"].(string); ok {
   505→							texts = append(texts, text)
   506→						}
   507→					}
   508→				}
   509→			}
   510→			if len(texts) > 0 {
   511→				cleaned := textutil.StripIDEContextTags(strings.Join(texts, "\n\n"))
   512→				if cleaned != "" {
   513→					prompts = append(prompts, cleaned)
   514→				}
   515→			}
   516→		}
   517→	}
   518→	return prompts
   519→}
