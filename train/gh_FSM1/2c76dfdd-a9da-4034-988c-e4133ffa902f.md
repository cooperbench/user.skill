   147→      } catch (error) {
   148→        // Re-throw HttpException immediately (e.g., parsing errors) - don't retry
   149→        if (error instanceof HttpException) {
   150→          throw error;
   151→        }
   152→
   153→        lastError = error instanceof Error ? error : new Error(String(error));
   154→

   157→          lastError.message.includes('Delegated routing returned') &&

   188→  private async fetchWithTimeout<T>(
   189→    url: string,
   190→    init: RequestInit,
   191→    consume: (response: Response) => Promise<T>
   192→  ): Promise<T> {
   193→    const controller = new AbortController();

   195→    try {
   196→      const response = await fetch(url, { ...init, signal: controller.signal });
   197→      return await consume(response);
   198→    } finally {
   199→      clearTimeout(timeout);
   200→    }
   201→  }
   202→
   203→  private getRetryDelayMs(retryAfter: string | null, attempt: number): number {
   204→    const fallback = this.baseDelayMs * Math.pow(2, attempt);
   205→    if (!retryAfter) return Math.min(fallback, this.maxRetryDelayMs);
   206→
   207→    const seconds = Number(retryAfter);
   208→    if (Number.isFinite(seconds) && seconds >= 0) {
   209→      return Math.min(seconds * 1000, this.maxRetryDelayMs);
   210→    }
   211→
   212→    const retryAt = Date.parse(retryAfter);
   213→    if (!Number.isNaN(retryAt)) {
   214→      return Math.min(Math.max(0, retryAt - Date.now()), this.maxRetryDelayMs);
   215→    }
   216→
   217→    return Math.min(fallback, this.maxRetryDelayMs);
   218→  }

   221→    return new Promise((resolve) => setTimeout(resolve, ms));

   211→    try {
   212→      const response = await fetch(url, { ...init, signal: controller.signal });
   213→      return await consume(response);
   214→    } finally {
   215→      clearTimeout(timeout);
   216→    }
   217→  }
   218→
   219→  private getRetryDelayMs(retryAfter: string | null, attempt: number): number {
   220→    const fallback = this.baseDelayMs * Math.pow(2, attempt);
   221→    if (!retryAfter) return Math.min(fallback, this.maxRetryDelayMs);
   222→
   223→    const seconds = Number(retryAfter);
   224→    if (Number.isFinite(seconds) && seconds >= 0) {
   225→      return Math.min(seconds * 1000, this.maxRetryDelayMs);
   226→    }
   227→
   228→    const retryAt = Date.parse(retryAfter);
   229→    if (!Number.isNaN(retryAt)) {
   230→      return Math.min(Math.max(0, retryAt - Date.now()), this.maxRetryDelayMs);
   231→    }
   232→
   233→    return Math.min(fallback, this.maxRetryDelayMs);
   234→  }
   235→
   236→  private delay(ms: number): Promise<void> {
   237→    return new Promise((resolve) => setTimeout(resolve, ms));
   238→  }
