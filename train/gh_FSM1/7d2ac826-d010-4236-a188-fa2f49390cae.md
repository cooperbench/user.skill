    14→@Injectable()
    15→export class TokenService {
    16→  private readonly REFRESH_TOKEN_EXPIRY_DAYS = 7;
    17→
    18→  constructor(
    19→    private jwtService: JwtService,
    20→    @InjectRepository(RefreshToken)
    21→    private refreshTokenRepo: Repository<RefreshToken>
    22→  ) {}
    23→
    24→  async createTokens(userId: string, publicKey: string): Promise<TokenPair> {
    25→    // Generate access token
    26→    const accessToken = this.jwtService.sign({ sub: userId, publicKey }, { expiresIn: '15m' });
    27→
    28→    // Generate refresh token
    29→    const refreshToken = randomBytes(32).toString('hex');
    30→    const tokenHash = await argon2.hash(refreshToken);
    31→
    32→    // Calculate expiry date
    33→    const expiresAt = new Date();
    34→    expiresAt.setDate(expiresAt.getDate() + this.REFRESH_TOKEN_EXPIRY_DAYS);
    35→
    36→    // Save to database with prefix for O(1) lookup
    37→    const tokenPrefix = refreshToken.substring(0, 16);
    38→    await this.refreshTokenRepo.save({
    39→      userId,
    40→      tokenHash,
    41→      tokenPrefix,
    42→      expiresAt,
    43→    });
    44→
    45→    return { accessToken, refreshToken };
    46→  }
    47→
    48→  async rotateRefreshToken(
    49→    oldRefreshToken: string,
    50→    userId: string,
    51→    publicKey: string
    52→  ): Promise<TokenPair> {
    53→    // Find candidate tokens by prefix for O(1) lookup instead of O(N) Argon2 scan
    54→    const prefix = oldRefreshToken.substring(0, 16);
    55→    const tokens = await this.refreshTokenRepo.find({
    56→      where: {
    57→        userId,
    58→        tokenPrefix: prefix,
    59→        revokedAt: IsNull(),
    60→      },
    61→    });
    62→
    63→    // Find matching token
    64→    let validToken: RefreshToken | null = null;
    65→    for (const token of tokens) {
    66→      try {
    67→        if (await argon2.verify(token.tokenHash, oldRefreshToken)) {
    68→          validToken = token;
    69→          break;
    70→        }
    71→      } catch {
    72→        // argon2.verify throws on invalid hash format, continue checking
    73→        continue;
    74→      }
    75→    }
    76→
    77→    if (!validToken) {
    78→      throw new UnauthorizedException('Invalid refresh token');
    79→    }
    80→
    81→    if (validToken.expiresAt < new Date()) {
    82→      // Revoke expired token
    83→      validToken.revokedAt = new Date();
    84→      await this.refreshTokenRepo.save(validToken);
    85→      throw new UnauthorizedException('Refresh token expired');
    86→    }
    87→
    88→    // Revoke old token
    89→    validToken.revokedAt = new Date();
    90→    await this.refreshTokenRepo.save(validToken);
    91→
    92→    // Create new tokens
    93→    return this.createTokens(userId, publicKey);
    94→  }
    95→
    96→  async revokeAllUserTokens(userId: string): Promise<void> {
    97→    await this.refreshTokenRepo.update({ userId, revokedAt: IsNull() }, { revokedAt: new Date() });
    98→  }

   102→      { id: tokenId, revokedAt: IsNull() },
   103→      { revokedAt: new Date() }
   104→    );

## 3. ECIES ENCRYPTION FORMAT (`packages/crypto/src/ecies/`)

### **ECIES Binary Format**

The `eciesjs` library produces ciphertext with this structure:

```
[ephemeral_public_key (65 bytes)] || [ciphertext] || [auth_tag (16 bytes)]
```

**Breakdown:**
- **Ephemeral public key**: 65 bytes (uncompressed secp256k1, 0x04 prefix)
- **Ciphertext**: Variable length (encrypted key or data)
- **Auth tag**: 16 bytes (AES-GCM authentication tag)
- **Minimum size**: 65 + 16 = 81 bytes (for 0-byte plaintext)

**Encryption Process** (`wrapKey()`):
1. Validates recipient public key (65 bytes, 0x04 prefix)
2. Validates public key is valid secp256k1 point
3. Calls `eciesjs.encrypt()` which:
   - Generates ephemeral keypair
   - Performs ECDH key agreement
   - Derives symmetric key using HKDF
   - Encrypts plaintext with AES-256-GCM
   - Returns ephemeral_pubkey || ciphertext || auth_tag

**Decryption Process** (`unwrapKey()`):
1. Validates private key (32 bytes)
2. Validates ciphertext length ≥ 81 bytes
3. Calls `eciesjs.decrypt()` which:
   - Extracts ephemeral public key from first 65 bytes
   - Performs ECDH with user's private key
   - Derives same symmetric key using HKDF
   - Verifies and decrypts AES-GCM ciphertext
   - Returns plaintext or throws on auth failure

**Error Handling:**
- All errors throw generic `CryptoError('Key wrapping/unwrapping failed', ...)` 
- Does NOT reveal whether private key was wrong, ciphertext modified, or auth tag failed (oracle attack prevention)
