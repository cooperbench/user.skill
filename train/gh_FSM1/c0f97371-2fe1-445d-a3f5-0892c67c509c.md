   389→  const updatedChildren = [...targetFolder.children, finalChild];
   390→  const { newSequenceNumber } = await updateFolderMetadata({
   391→    folderId: targetFolder.id,
   392→    children: updatedChildren,
   393→    folderKey: targetFolder.folderKey,
   394→    ipnsPrivateKey: targetFolder.ipnsPrivateKey,
   395→    ipnsName: targetFolder.ipnsName,
   396→    sequenceNumber: targetFolder.sequenceNumber,
   397→  });
   398→
   399→  // Update folder store
   400→  const folderStore = useFolderStore.getState();
   401→  folderStore.updateFolderChildren(targetFolder.id, updatedChildren);
   402→  folderStore.updateFolderSequence(targetFolder.id, newSequenceNumber);
   403→
   404→  // If restoring a folder, re-add the folder node to the store
   405→  if (entry.itemType === 'folder' && entry.folderEntry) {
   406→    const existingNode = folderStore.folders[entry.folderEntry.id];
   407→    if (!existingNode) {
   408→      // Create a placeholder node -- it will be fully loaded on next navigation
   409→      folderStore.setFolder({
   410→        id: entry.folderEntry.id,
   411→        name: finalChild.name,
   412→        ipnsName: entry.folderEntry.ipnsName,
   413→        parentId: targetFolder.id,
   414→        children: [],
   415→        isLoaded: false,
   416→        isLoading: false,
   417→        sequenceNumber: 0n,
   418→        // These will be populated when the folder is loaded
   419→        folderKey: new Uint8Array(32),
   420→        ipnsPrivateKey: new Uint8Array(64),
   421→      });
   422→    }
   423→  }
   424→
   425→  // Remove entry from bin metadata and publish
   426→  await removeEntriesFromBin([entryId], { userPublicKey, userPrivateKey });
   427→}

   433→/**
   434→ * Restore multiple bin entries, publishing to the bin IPNS only once at the end.
   435→ *
   436→ * Each item is restored to its target folder (which may each require a folder
   437→ * publish), but the bin metadata is updated in a single publish.
   438→ */
   439→export async function restoreFromBinBatch(params: {
   440→  entryIds: string[];
   441→  userPublicKey: Uint8Array;
   442→  userPrivateKey: Uint8Array;
   443→}): Promise<void> {
   444→  const { entryIds, userPublicKey, userPrivateKey } = params;
   445→  if (entryIds.length === 0) return;
   446→
   447→  const { updateFolderMetadata } = await import('./folder.service');
   448→
   449→  const restoredIds: string[] = [];
   450→
   451→  for (const entryId of entryIds) {
   452→    // Read fresh state each iteration — recursive restores may mutate the store
   453→    const entry = useBinStore.getState().entries.find((e) => e.id === entryId);
   454→    if (!entry) continue;
   455→
   456→    try {
   457→      const targetFolder = await resolveTargetFolder(entry.originalParentIpnsName, 0, {
   458→        userPublicKey,
   459→        userPrivateKey,
   460→      });
   461→
   462→      const child = getRestoredChild(entry);
   463→      const finalChild = resolveNameCollision(child, targetFolder.children);
   464→
   465→      const updatedChildren = [...targetFolder.children, finalChild];
   466→      const { newSequenceNumber } = await updateFolderMetadata({
   467→        folderId: targetFolder.id,
   468→        children: updatedChildren,
   469→        folderKey: targetFolder.folderKey,
   470→        ipnsPrivateKey: targetFolder.ipnsPrivateKey,
   471→        ipnsName: targetFolder.ipnsName,
   472→        sequenceNumber: targetFolder.sequenceNumber,
   473→      });
   474→
   475→      const folderStore = useFolderStore.getState();
   476→      folderStore.updateFolderChildren(targetFolder.id, updatedChildren);
   477→      folderStore.updateFolderSequence(targetFolder.id, newSequenceNumber);
   478→
   479→      if (entry.itemType === 'folder' && entry.folderEntry) {
   480→        const existingNode = folderStore.folders[entry.folderEntry.id];
   481→        if (!existingNode) {
   482→          folderStore.setFolder({
   483→            id: entry.folderEntry.id,
   484→            name: finalChild.name,
   485→            ipnsName: entry.folderEntry.ipnsName,
   486→            parentId: targetFolder.id,
   487→            children: [],
   488→            isLoaded: false,
   489→            isLoading: false,
   490→            sequenceNumber: 0n,
   491→            folderKey: new Uint8Array(32),
   492→            ipnsPrivateKey: new Uint8Array(64),
   493→          });
   494→        }
   495→      }
   496→
   497→      restoredIds.push(entryId);
   498→    } catch (err) {
   499→      console.error(`[Bin] Failed to restore entry ${entry.id}:`, err);
   500→    }
   501→  }
   502→
   503→  // Single bin IPNS publish for all restored entries
   504→  if (restoredIds.length > 0) {
   505→    await removeEntriesFromBin(restoredIds, { userPublicKey, userPrivateKey });
   506→  }
   507→}

   684→/**
   685→ * Cleanup CIDs for a bin entry (unpin from IPFS + update quota).
   686→ *
   687→ * Files: use stored contentCid if available, else fall back to IPNS resolution.
   688→ * Folders: unwrap folderKey, decrypt folder metadata, decrypt each nested file's
   689→ * metadata to get CID for unpinning. Recurses for subfolders.
   690→ */
   691→async function cleanupEntryCids(entry: BinEntry, userPrivateKey: Uint8Array): Promise<void> {
   692→  if (entry.itemType === 'file') {
   693→    await unpinFileCids(entry);
   694→  } else if (entry.itemType === 'folder' && entry.folderEntry) {
   695→    await cleanupFolderCids(entry.folderEntry, userPrivateKey);
   696→  }
   697→}
   698→
   699→/**
   700→ * Unpin a single file's content and metadata CIDs, and update quota.
   701→ *
   702→ * Prefers stored contentCid (captured at soft-delete time) to avoid
   703→ * re-decrypting encrypted file metadata (which was the GAP-1 bug).
   704→ * Falls back to legacy IPNS resolution for old entries without contentCid.
   705→ */
   706→async function unpinFileCids(entry: BinEntry): Promise<void> {
   707→  try {
   708→    // Preferred path: use stored contentCid from soft-delete time
   709→    if (entry.contentCid) {
   710→      await unpinFromIpfs(entry.contentCid);
   711→      if (Number.isFinite(entry.contentSize)) {
   712→        useQuotaStore.getState().removeUsage(entry.contentSize!);
   713→      }
   714→      // Unpin version CIDs captured at soft-delete time
   715→      if (entry.versionCids?.length) {
   716→        for (const v of entry.versionCids) {
   717→          await unpinFromIpfs(v.cid).catch(() => {});
   718→          if (Number.isFinite(v.size)) {
   719→            useQuotaStore.getState().removeUsage(v.size);
   720→          }
   721→        }
   722→      }
   723→      // Also unpin the file metadata IPNS record CID
   724→      if (entry.filePointer) {
   725→        const resolved = await resolveIpnsRecord(entry.filePointer.fileMetaIpnsName);
   726→        if (resolved?.cid) {
   727→          await unpinFromIpfs(resolved.cid).catch(() => {});
   728→        }
   729→      }
   730→      return;
   731→    }
   732→
   733→    // Fallback for legacy entries without contentCid:
   734→    // Resolve IPNS and attempt to parse file metadata as plaintext JSON.
   735→    // NOTE: This will fail for encrypted file metadata (the original GAP-1 bug).
   736→    // Legacy entries that hit this path will log an error but not crash.
   737→    if (!entry.filePointer) return;
   738→    const fileIpnsName = entry.filePointer.fileMetaIpnsName;
   739→    if (!fileIpnsName) return;
   740→
   741→    const resolved = await resolveIpnsRecord(fileIpnsName);
   742→    if (!resolved?.cid) return;
   743→
   744→    const metaBytes = await fetchFromIpfs(resolved.cid);
   745→    const metaJson = new TextDecoder().decode(metaBytes);
   746→    const fileMeta = JSON.parse(metaJson);
   747→
   748→    if (fileMeta.cid) {
   749→      await unpinFromIpfs(fileMeta.cid);
   750→      if (Number.isFinite(fileMeta.size)) {
   751→        useQuotaStore.getState().removeUsage(fileMeta.size);
   752→      }
   753→    }
   754→    await unpinVersionCids(fileMeta);
   755→
   756→    await unpinFromIpfs(resolved.cid).catch(() => {});
   757→  } catch (err) {
   758→    console.error(`[Bin] CID cleanup failed for file ${entry.id}:`, err);
   759→  }
   760→}
   761→
   762→/**
   763→ * Unpin version CIDs from a decrypted/parsed file metadata object and reclaim quota.
   764→ * Best-effort: errors on individual versions are caught and logged.
   765→ */
   766→async function unpinVersionCids(fileMeta: {
   767→  versions?: Array<{ cid?: string; size?: number }>;
   768→}): Promise<void> {
   769→  if (!Array.isArray(fileMeta.versions)) return;
   770→  for (const version of fileMeta.versions) {
   771→    try {
   772→      if (version.cid) {
   773→        await unpinFromIpfs(version.cid);
   774→      }
   775→      if (Number.isFinite(version.size)) {
   776→        useQuotaStore.getState().removeUsage(version.size!);
   777→      }
   778→    } catch {
   779→      // Best-effort for individual version cleanup
   780→    }
   781→  }
   782→}

   784→/**
   785→ * Recursively unpin all CIDs in a folder tree.
   786→ *
   787→ * Unwraps the folder's AES key using the user's private key, then decrypts
   788→ * the folder metadata to enumerate children. For each file child, decrypts
   789→ * the file metadata to extract the content CID for unpinning.
   790→ * For subfolder children, recurses with the user's private key.
   791→ */
   792→async function cleanupFolderCids(
   793→  folderEntry: FolderEntry,
   794→  userPrivateKey: Uint8Array
   795→): Promise<void> {
   796→  let folderMetadataCid: string | null = null;
   797→  try {
   798→    // Step 1: Unwrap the folder's AES key using user's private key (ECIES)
   799→    const wrappedFolderKey = hexToBytes(folderEntry.folderKeyEncrypted);
   800→    const folderKey = await unwrapKey(wrappedFolderKey, userPrivateKey);
   801→
   802→    // Step 2: Resolve folder's IPNS to get its metadata CID
   803→    const resolved = await resolveIpnsRecord(folderEntry.ipnsName);
   804→    if (!resolved?.cid) return;
   805→    folderMetadataCid = resolved.cid;
   806→
   807→    // Step 3: Fetch and decrypt the folder metadata
   808→    const metaBytes = await fetchFromIpfs(resolved.cid);
   809→    const metaJson = new TextDecoder().decode(metaBytes);
   810→    const encrypted = JSON.parse(metaJson) as EncryptedFolderMetadata;
   811→    const folderMeta = await decryptFolderMetadata(encrypted, folderKey);
   812→
   813→    // Step 4: For each child, unpin CIDs
   814→    for (const child of folderMeta.children) {
   815→      try {
   816→        if (child.type === 'file') {
   817→          const fp = child as FilePointer;
   818→          // Decrypt file metadata to get content CID
   819→          const fileResolved = await resolveIpnsRecord(fp.fileMetaIpnsName);
   820→          if (fileResolved?.cid) {
   821→            const fileMetaBytes = await fetchFromIpfs(fileResolved.cid);
   822→            const fileMetaJson = new TextDecoder().decode(fileMetaBytes);
   823→            const encryptedFileMeta = JSON.parse(fileMetaJson) as EncryptedFileMetadata;
   824→            const fileMeta = await decryptFileMetadata(encryptedFileMeta, folderKey);
   825→            if (fileMeta.cid) {
   826→              await unpinFromIpfs(fileMeta.cid);
   827→              if (Number.isFinite(fileMeta.size)) {
   828→                useQuotaStore.getState().removeUsage(fileMeta.size);
   829→              }
   830→            }
   831→            // Unpin any historical version CIDs
   832→            await unpinVersionCids(fileMeta);
   833→            // Also unpin the file metadata CID
   834→            await unpinFromIpfs(fileResolved.cid).catch(() => {});
   835→          }
   836→        } else if (child.type === 'folder') {
   837→          // Subfolder: folderKeyEncrypted is ECIES-wrapped with the user's publicKey
   838→          const subFe = child as FolderEntry;
   839→          await cleanupFolderCids(subFe, userPrivateKey);
   840→        }
   841→      } catch (childErr) {
   842→        console.error(`[Bin] Failed to cleanup CIDs for child ${child.name}:`, childErr);
   843→      }
   844→    }
   845→  } catch (err) {
   846→    console.error(`[Bin] Folder CID cleanup failed for ${folderEntry.ipnsName}:`, err);
   847→  } finally {
   848→    // Always unpin the folder metadata CID, even if child cleanup threw
   849→    if (folderMetadataCid) {
   850→      await unpinFromIpfs(folderMetadataCid).catch(() => {});
   851→    }
   852→  }
   853→}

   855→/**
   856→ * Resolve the target folder for a restore operation.
   857→ *
   858→ * Walks the folder store to find a folder with the given IPNS name.
   859→ * If not found, checks if the parent is in the bin (recursive restore).
   860→ * Falls back to root folder.
   861→ */
   862→async function resolveTargetFolder(
   863→  parentIpnsName: string,
   864→  depth: number,
   865→  params: { userPublicKey: Uint8Array; userPrivateKey: Uint8Array }
   866→): Promise<FolderNode> {
   867→  // Max recursion depth to prevent infinite loops
   868→  if (depth > 5) {
   869→    console.warn('[Bin] Max restore depth reached, falling back to root');
   870→    return getRootFolder();
   871→  }
   872→
   873→  // Look up folder by IPNS name in folder store
   874→  const folders = useFolderStore.getState().folders;
   875→  const targetFolder = Object.values(folders).find((f) => f.ipnsName === parentIpnsName);
   876→
   877→  if (targetFolder) {
   878→    return targetFolder;
   879→  }
   880→
   881→  // Check if parent is in the bin
   882→  const binStore = useBinStore.getState();
   883→  const parentBinEntry = binStore.entries.find(
   884→    (e) => e.itemType === 'folder' && e.folderEntry && e.folderEntry.ipnsName === parentIpnsName
   885→  );
   886→
   887→  if (parentBinEntry) {
   888→    // Restore parent first (recursive, depth incremented)
   889→    console.log(`[Bin] Restoring parent folder (id: ${parentBinEntry.id}) first`);
   890→    await restoreFromBinInternal(
   891→      {
   892→        entryId: parentBinEntry.id,
   893→        userPublicKey: params.userPublicKey,
   894→        userPrivateKey: params.userPrivateKey,
   895→      },
   896→      depth + 1
   897→    );
   898→
   899→    // After restoring parent, try to find it again
   900→    const updatedFolders = useFolderStore.getState().folders;
   901→    const restoredParent = Object.values(updatedFolders).find((f) => f.ipnsName === parentIpnsName);
   902→    if (restoredParent) {
   903→      return restoredParent;
   904→    }
   905→  }
   906→
   907→  // Parent truly gone -- restore to root
   908→  console.warn(`[Bin] Original parent not found (${parentIpnsName}), restoring to root`);
   909→  return getRootFolder();
   910→}

   912→/**
   913→ * Get the root folder from the folder store.
   914→ */
   915→function getRootFolder(): FolderNode {
   916→  const folders = useFolderStore.getState().folders;
   917→  const root = folders['root'];
   918→  if (root) return root;
   919→
   920→  // Fallback: construct from vault store
   921→  const vault = useVaultStore.getState();
   922→  if (!vault.rootFolderKey || !vault.rootIpnsKeypair || !vault.rootIpnsName) {
   923→    throw new Error('Vault not initialized - cannot restore to root');
   924→  }
   925→
   926→  return {
   927→    id: 'root',
   928→    name: 'My Vault',
   929→    ipnsName: vault.rootIpnsName,
   930→    parentId: null,
   931→    children: folders['root']?.children ?? [],
   932→    isLoaded: true,
   933→    isLoading: false,
   934→    sequenceNumber: 0n,
   935→    folderKey: vault.rootFolderKey,
   936→    ipnsPrivateKey: vault.rootIpnsKeypair.privateKey,
   937→  };
   938→}

   940→/**
   941→ * Get the FolderChild to restore from a bin entry.
   942→ */
   943→function getRestoredChild(entry: BinEntry): FolderChild {
   944→  if (entry.itemType === 'file' && entry.filePointer) {
   945→    return { ...entry.filePointer, modifiedAt: Date.now() };
   946→  } else if (entry.itemType === 'folder' && entry.folderEntry) {
   947→    return { ...entry.folderEntry, modifiedAt: Date.now() };
   948→  }
   949→  throw new Error(`Bin entry ${entry.id} has no stored item reference`);
   950→}

   952→/**
   953→ * Check for name collisions and rename if needed.
   954→ */
   955→function resolveNameCollision(child: FolderChild, existingChildren: FolderChild[]): FolderChild {
   956→  const existingNames = new Set(existingChildren.map((c) => c.name));
   957→  if (!existingNames.has(child.name)) {
   958→    return child;
   959→  }
   960→
   961→  // Append " (restored)" suffix, incrementing if needed
   962→  let newName = `${child.name} (restored)`;
   963→  let counter = 2;
   964→  while (existingNames.has(newName)) {
   965→    newName = `${child.name} (restored ${counter})`;
   966→    counter++;
   967→  }
   968→
