   223→      const resolved = await resolveIpnsRecord(fp.fileMetaIpnsName);
   224→      if (resolved?.cid) {
   225→        const metaBytes = await fetchFromIpfs(resolved.cid);
   226→        const metaJson = new TextDecoder().decode(metaBytes);
   227→        const encrypted = JSON.parse(metaJson) as EncryptedFileMetadata;
   228→        const fileMeta = await decryptFileMetadata(encrypted, folderKey);
   229→        entry.contentCid = fileMeta.cid;
   230→        entry.contentSize = fileMeta.size;
   231→        if (Number.isFinite(fileMeta.size)) {
   232→          entry.size = fileMeta.size;
   233→        }
   234→        if (fileMeta.mimeType) {
   235→          entry.mimeType = fileMeta.mimeType;
   236→        }
   237→        if (Array.isArray(fileMeta.versions) && fileMeta.versions.length > 0) {
   238→          entry.versionCids = fileMeta.versions
   239→            .filter((v) => v.cid)
   240→            .map((v) => ({ cid: v.cid, size: v.size ?? 0 }));
   241→        }
   242→      }
   243→    } catch (err) {
   244→      console.warn('[Bin] Failed to resolve file CID for bin entry (non-blocking):', err);
   245→    }
   246→  }
   247→
   248→  // Read current bin state and build updated metadata
   249→  const store = useBinStore.getState();
   250→  const currentEntries = [...store.entries, entry];
   251→  const newSeq = store.sequenceNumber + 1;
   252→
   253→  const metadata: RecycleBinMetadata = {
   254→    version: BIN_METADATA_VERSION,
   255→    sequenceNumber: newSeq,
   256→    entries: currentEntries,
   257→  };
   258→
   259→  // Encrypt and publish (non-blocking: folder delete already committed, bin is best-effort)
   260→  try {
   261→    await saveBinMetadata({ metadata, userPublicKey, userPrivateKey });
   262→    store.setEntries(currentEntries, newSeq);
   263→  } catch {
   264→    console.error('[Bin] addToBin publish failed (non-blocking)');
   265→  }
   266→}

   272→/**
   273→ * Add multiple deleted items to the recycle bin in a single IPNS publish.
   274→ *
   275→ * Unlike calling addToBin in a loop (which publishes per-item), this batches
   276→ * all entries into one metadata update.
   277→ */
   278→export async function addManyToBin(params: {
   279→  items: FolderChild[];
   280→  parentIpnsName: string;
   281→  parentPath: string;
   282→  userPublicKey: Uint8Array;
   283→  userPrivateKey: Uint8Array;
   284→  folderKey?: Uint8Array;
   285→}): Promise<void> {
   286→  const { items, parentIpnsName, parentPath, userPublicKey, userPrivateKey, folderKey } = params;
   287→  if (items.length === 0) return;
   288→
   289→  const now = Date.now();
   290→  const newEntries: BinEntry[] = [];
   291→  for (const item of items) {
   292→    const isFile = item.type === 'file';
   293→    const entry: BinEntry = {
   294→      id: crypto.randomUUID(),
   295→      itemType: isFile ? 'file' : 'folder',
   296→      name: item.name,
   297→      originalParentIpnsName: parentIpnsName,
   298→      originalPath: parentPath,
   299→      deletedAt: now,
   300→      size: 0,
   301→      mimeType: '',
   302→      filePointer: isFile ? (item as FilePointer) : undefined,
   303→      folderEntry: !isFile ? (item as FolderEntry) : undefined,
   304→    };
   305→
   306→    // Resolve file CID+size at soft-delete time (when folderKey is available)
   307→    if (isFile && folderKey) {
   308→      try {
   309→        const fp = item as FilePointer;
   310→        const resolved = await resolveIpnsRecord(fp.fileMetaIpnsName);
   311→        if (resolved?.cid) {
   312→          const metaBytes = await fetchFromIpfs(resolved.cid);
   313→          const metaJson = new TextDecoder().decode(metaBytes);
   314→          const encrypted = JSON.parse(metaJson) as EncryptedFileMetadata;
   315→          const fileMeta = await decryptFileMetadata(encrypted, folderKey);
   316→          entry.contentCid = fileMeta.cid;
   317→          entry.contentSize = fileMeta.size;
   318→          if (Number.isFinite(fileMeta.size)) {
   319→            entry.size = fileMeta.size;
   320→          }
   321→          if (fileMeta.mimeType) {
   322→            entry.mimeType = fileMeta.mimeType;
   323→          }
   324→          if (Array.isArray(fileMeta.versions) && fileMeta.versions.length > 0) {
   325→            entry.versionCids = fileMeta.versions
   326→              .filter((v) => v.cid)
   327→              .map((v) => ({ cid: v.cid, size: v.size ?? 0 }));
   328→          }
   329→        }
   330→      } catch (err) {
   331→        console.warn('[Bin] Failed to resolve file CID for bin entry (non-blocking):', err);
   332→      }
   333→    }
   334→
   335→    newEntries.push(entry);
   336→  }
   337→
   338→  const store = useBinStore.getState();
   339→  const currentEntries = [...store.entries, ...newEntries];
   340→  const newSeq = store.sequenceNumber + 1;
   341→
   342→  const metadata: RecycleBinMetadata = {
   343→    version: BIN_METADATA_VERSION,
   344→    sequenceNumber: newSeq,
   345→    entries: currentEntries,
   346→  };
   347→
   348→  // Non-blocking: folder delete already committed, bin is best-effort
   349→  try {
   350→    await saveBinMetadata({ metadata, userPublicKey, userPrivateKey });
   351→    store.setEntries(currentEntries, newSeq);
   352→  } catch {
   353→    console.error('[Bin] addManyToBin publish failed (non-blocking)');
   354→  }
   355→}

   377→async function restoreFromBinInternal(
   378→  params: { entryId: string; userPublicKey: Uint8Array; userPrivateKey: Uint8Array },
   379→  depth: number
   380→): Promise<void> {
   381→  const { entryId, userPublicKey, userPrivateKey } = params;
   382→
   383→  const binStore = useBinStore.getState();
   384→  const entry = binStore.entries.find((e) => e.id === entryId);
   385→  if (!entry) throw new Error('Bin entry not found');
   386→
   387→  // Resolve original parent folder
   388→  const targetFolder = await resolveTargetFolder(entry.originalParentIpnsName, depth, {
   389→    userPublicKey,
   390→    userPrivateKey,
   391→  });
   392→
   393→  // Get the FolderChild to restore
   394→  const child = getRestoredChild(entry);
   395→
   396→  // Check for name collisions in target folder and rename if needed
   397→  const finalChild = resolveNameCollision(child, targetFolder.children);
   398→
   399→  // Add child back to target folder
   400→  const { updateFolderMetadata } = await import('./folder.service');
   401→  const updatedChildren = [...targetFolder.children, finalChild];
   402→  const { newSequenceNumber } = await updateFolderMetadata({
   403→    folderId: targetFolder.id,
   404→    children: updatedChildren,
   405→    folderKey: targetFolder.folderKey,
   406→    ipnsPrivateKey: targetFolder.ipnsPrivateKey,
   407→    ipnsName: targetFolder.ipnsName,
   408→    sequenceNumber: targetFolder.sequenceNumber,
   409→  });
   410→
   411→  // Update folder store
   412→  const folderStore = useFolderStore.getState();
   413→  folderStore.updateFolderChildren(targetFolder.id, updatedChildren);
   414→  folderStore.updateFolderSequence(targetFolder.id, newSequenceNumber);
   415→
   416→  // If restoring a folder, re-add the folder node to the store
   417→  if (entry.itemType === 'folder' && entry.folderEntry) {
   418→    const existingNode = folderStore.folders[entry.folderEntry.id];
   419→    if (!existingNode) {
   420→      // Create a placeholder node -- it will be fully loaded on next navigation
   421→      folderStore.setFolder({
   422→        id: entry.folderEntry.id,
   423→        name: finalChild.name,
   424→        ipnsName: entry.folderEntry.ipnsName,
   425→        parentId: targetFolder.id,
   426→        children: [],
   427→        isLoaded: false,
   428→        isLoading: false,
   429→        sequenceNumber: 0n,
   430→        // These will be populated when the folder is loaded
   431→        folderKey: new Uint8Array(32),
   432→        ipnsPrivateKey: new Uint8Array(64),
   433→      });
   434→    }
   435→  }
   436→
   437→  // Remove entry from bin metadata and publish
   438→  await removeEntriesFromBin([entryId], { userPublicKey, userPrivateKey });
   439→}

   445→/**
   446→ * Restore multiple bin entries, publishing to the bin IPNS only once at the end.
   447→ *
   448→ * Each item is restored to its target folder (which may each require a folder
   449→ * publish), but the bin metadata is updated in a single publish.
   450→ */
   451→export async function restoreFromBinBatch(params: {
   452→  entryIds: string[];
   453→  userPublicKey: Uint8Array;
   454→  userPrivateKey: Uint8Array;
   455→}): Promise<void> {
   456→  const { entryIds, userPublicKey, userPrivateKey } = params;
   457→  if (entryIds.length === 0) return;
   458→
   459→  const { updateFolderMetadata } = await import('./folder.service');
   460→
   461→  const restoredIds: string[] = [];
   462→
   463→  for (const entryId of entryIds) {
   464→    // Read fresh state each iteration — recursive restores may mutate the store
   465→    const entry = useBinStore.getState().entries.find((e) => e.id === entryId);
   466→    if (!entry) continue;
   467→
   468→    try {
   469→      const targetFolder = await resolveTargetFolder(entry.originalParentIpnsName, 0, {
   470→        userPublicKey,
   471→        userPrivateKey,
   472→      });
   473→
   474→      const child = getRestoredChild(entry);
   475→      const finalChild = resolveNameCollision(child, targetFolder.children);
   476→
   477→      const updatedChildren = [...targetFolder.children, finalChild];
   478→      const { newSequenceNumber } = await updateFolderMetadata({
   479→        folderId: targetFolder.id,
   480→        children: updatedChildren,
   481→        folderKey: targetFolder.folderKey,
   482→        ipnsPrivateKey: targetFolder.ipnsPrivateKey,
   483→        ipnsName: targetFolder.ipnsName,
   484→        sequenceNumber: targetFolder.sequenceNumber,
   485→      });
   486→
   487→      const folderStore = useFolderStore.getState();
   488→      folderStore.updateFolderChildren(targetFolder.id, updatedChildren);
   489→      folderStore.updateFolderSequence(targetFolder.id, newSequenceNumber);
   490→
   491→      if (entry.itemType === 'folder' && entry.folderEntry) {
   492→        const existingNode = folderStore.folders[entry.folderEntry.id];
   493→        if (!existingNode) {
   494→          folderStore.setFolder({
   495→            id: entry.folderEntry.id,
   496→            name: finalChild.name,
   497→            ipnsName: entry.folderEntry.ipnsName,
   498→            parentId: targetFolder.id,
   499→            children: [],
   500→            isLoaded: false,
   501→            isLoading: false,
   502→            sequenceNumber: 0n,
   503→            folderKey: new Uint8Array(32),
   504→            ipnsPrivateKey: new Uint8Array(64),
   505→          });
   506→        }
   507→      }
   508→
   509→      restoredIds.push(entryId);
   510→    } catch (err) {
   511→      console.error(`[Bin] Failed to restore entry ${entry.id}:`, err);
   512→    }
   513→  }
   514→
   515→  // Single bin IPNS publish for all restored entries
   516→  if (restoredIds.length > 0) {

   696→/**
   697→ * Cleanup CIDs for a bin entry (unpin from IPFS + update quota).
   698→ *
   699→ * Files: use stored contentCid if available, else fall back to IPNS resolution.
   700→ * Folders: unwrap folderKey, decrypt folder metadata, decrypt each nested file's
   701→ * metadata to get CID for unpinning. Recurses for subfolders.
   702→ */
   703→async function cleanupEntryCids(entry: BinEntry, userPrivateKey: Uint8Array): Promise<void> {
   704→  if (entry.itemType === 'file') {
   705→    await unpinFileCids(entry);
   706→  } else if (entry.itemType === 'folder' && entry.folderEntry) {
   707→    await cleanupFolderCids(entry.folderEntry, userPrivateKey);
   708→  }
   709→}
   710→
   711→/**
   712→ * Unpin a single file's content and metadata CIDs, and update quota.
   713→ *
   714→ * Prefers stored contentCid (captured at soft-delete time) to avoid
   715→ * re-decrypting encrypted file metadata (which was the GAP-1 bug).
   716→ * Falls back to legacy IPNS resolution for old entries without contentCid.
   717→ */
   718→async function unpinFileCids(entry: BinEntry): Promise<void> {
   719→  try {
   720→    // Preferred path: use stored contentCid from soft-delete time
   721→    if (entry.contentCid) {
   722→      await unpinFromIpfs(entry.contentCid);
   723→      if (Number.isFinite(entry.contentSize)) {
   724→        useQuotaStore.getState().removeUsage(entry.contentSize!);
   725→      }
   726→      // Unpin version CIDs captured at soft-delete time
   727→      if (entry.versionCids?.length) {
   728→        for (const v of entry.versionCids) {
   729→          await unpinFromIpfs(v.cid).catch(() => {});
   730→          if (Number.isFinite(v.size)) {
   731→            useQuotaStore.getState().removeUsage(v.size);
   732→          }
   733→        }
   734→      }
   735→      // Also unpin the file metadata IPNS record CID
   736→      if (entry.filePointer) {
   737→        const resolved = await resolveIpnsRecord(entry.filePointer.fileMetaIpnsName);
   738→        if (resolved?.cid) {
   739→          await unpinFromIpfs(resolved.cid).catch(() => {});
   740→        }
   741→      }
   742→      return;
   743→    }
   744→
   745→    // Fallback for legacy entries without contentCid:
   746→    // Resolve IPNS and attempt to interpret file metadata as plaintext JSON.
   747→    // NOTE: For encrypted file metadata ({ iv, data }), JSON.parse will usually
   748→    // succeed, but the parsed object will not have cid/size, so no content unpin
   749→    // or quota reclaim happens (the original GAP-1 bug). Entries that hit this
   750→    // path may log an error but will not crash the delete flow.
   751→    if (!entry.filePointer) return;
   752→    const fileIpnsName = entry.filePointer.fileMetaIpnsName;
   753→    if (!fileIpnsName) return;
   754→
   755→    const resolved = await resolveIpnsRecord(fileIpnsName);
   756→    if (!resolved?.cid) return;
   757→
   758→    const metaBytes = await fetchFromIpfs(resolved.cid);
   759→    const metaJson = new TextDecoder().decode(metaBytes);
   760→    const fileMeta = JSON.parse(metaJson);
   761→
   762→    if (fileMeta.cid) {
   763→      await unpinFromIpfs(fileMeta.cid);
   764→      if (Number.isFinite(fileMeta.size)) {
   765→        useQuotaStore.getState().removeUsage(fileMeta.size);
   766→      }
   767→    }
   768→    await unpinVersionCids(fileMeta);
   769→
   770→    await unpinFromIpfs(resolved.cid).catch(() => {});
   771→  } catch (err) {
   772→    console.error(`[Bin] CID cleanup failed for file ${entry.id}:`, err);
   773→  }
   774→}
   775→
   776→/**
   777→ * Unpin version CIDs from a decrypted/parsed file metadata object and reclaim quota.
   778→ * Best-effort: errors on individual versions are caught and logged.
   779→ */
   780→async function unpinVersionCids(fileMeta: {
   781→  versions?: Array<{ cid?: string; size?: number }>;
   782→}): Promise<void> {
   783→  if (!Array.isArray(fileMeta.versions)) return;
   784→  for (const version of fileMeta.versions) {
   785→    try {
   786→      if (version.cid) {
   787→        await unpinFromIpfs(version.cid);
   788→      }
   789→      if (Number.isFinite(version.size)) {
   790→        useQuotaStore.getState().removeUsage(version.size!);
   791→      }
   792→    } catch {
   793→      // Best-effort for individual version cleanup
   794→    }
   795→  }
   796→}
   797→
   798→/**
   799→ * Recursively unpin all CIDs in a folder tree.
   800→ *
   801→ * Unwraps the folder's AES key using the user's private key, then decrypts
   802→ * the folder metadata to enumerate children. For each file child, decrypts
   803→ * the file metadata to extract the content CID for unpinning.
   804→ * For subfolder children, recurses with the user's private key.
   805→ */
   806→async function cleanupFolderCids(
   807→  folderEntry: FolderEntry,
   808→  userPrivateKey: Uint8Array
   809→): Promise<void> {
   810→  let folderMetadataCid: string | null = null;
   811→  try {
   812→    // Step 1: Unwrap the folder's AES key using user's private key (ECIES)

   379→                    item_type: crate::crypto::bin::BinItemType::File,
   380→                    name: item_name.clone(),
   381→                    original_parent_ipns_name: parent_ipns_name,
   382→                    original_path: parent_path,
   383→                    deleted_at: now
   384→                        .duration_since(std::time::UNIX_EPOCH)
   385→                        .unwrap_or_default()
   386→                        .as_millis() as u64,
   387→                    size: file_size,
   388→                    mime_type: crate::crypto::utils::mime_from_extension(&item_name).to_string(),
   389→                    content_cid: if content_cid.is_empty() { None } else { Some(content_cid) },
   390→                    content_size: Some(file_size),
   391→                    version_cids: ver_cids,
   392→                    file_pointer: Some(file_pointer),
   393→                    folder_entry: None,
   394→                };
   395→
   396→                crate::fuse::spawn_bin_entry_publish(
   397→                    fs.api.clone(),
   398→                    fs.rt.clone(),
   399→                    bin_entry,
   400→                    fs.private_key.clone(),
   401→                    fs.public_key.to_vec(),
   402→                    fs.publish_coordinator.clone(),
   403→                );
   404→            }
   405→        }
   406→
   407→        reply.ok();
   408→    }

   410→    /// Create a new directory.
   411→    pub fn handle_mkdir(
   412→        fs: &mut CipherBoxFS,
   413→        parent: u64,
   414→        name: &OsStr,
   415→        reply: ReplyEntry,
   416→    ) {
   417→        let name_str = match name.to_str() {
   418→            Some(n) => n,
   419→            None => {
   420→                reply.error(libc::EINVAL);
   421→                return;
   422→            }
   423→        };
   424→
   425→        if is_platform_special(name_str) {
   426→            reply.error(libc::EACCES);
   427→            return;
   428→        }
   429→
   430→        let parent_exists = fs.inodes.get(parent).map(|inode| {
   431→            matches!(inode.kind, InodeKind::Root { .. } | InodeKind::Folder { .. })
   432→        });
   433→        if parent_exists != Some(true) {
   434→            reply.error(libc::ENOENT);
   435→            return;
   436→        }
   437→
   438→        log::debug!("mkdir: {} in parent {}", name_str, parent);
   439→
   440→        let result = (|| -> Result<fuser::FileAttr, String> {
   441→            let folder_key = crate::crypto::utils::generate_file_key();
   442→
   443→            let (ipns_public_key, ipns_private_key) =
   444→                crate::crypto::ed25519::generate_ed25519_keypair();
   445→
   446→            let ipns_pub_arr: [u8; 32] = ipns_public_key.clone().try_into()
   447→                .map_err(|_| "Invalid IPNS public key length".to_string())?;
   448→            let ipns_name = crate::crypto::ipns::derive_ipns_name(&ipns_pub_arr)
   449→                .map_err(|e| format!("Failed to derive IPNS name: {}", e))?;
   450→
   451→            let wrapped_folder_key = crate::crypto::ecies::wrap_key(
   452→                &folder_key, &fs.public_key,
   453→            )
   454→            .map_err(|e| format!("Folder key wrapping failed: {}", e))?;
   455→            let encrypted_folder_key_hex = hex::encode(&wrapped_folder_key);
   456→
   457→            let ino = fs.inodes.allocate_ino();
   458→            let now = SystemTime::now();
   459→
   460→            let attr = FileAttrs {
   461→                ino,
   462→                size: 0,
   463→                blocks: 0,
   464→                atime: now,
   465→                mtime: now,
   466→                ctime: now,
   467→                crtime: now,
   468→                is_dir: true,
   469→                perm: 0o755,
   470→                nlink: 2,
   471→            };
   472→            let fuse_attr = attr.to_fuse_attr(current_uid(), current_gid());
   473→
   474→            let inode = InodeData {
   475→                ino,
   476→                parent_ino: parent,
   477→                name: name_str.to_string(),
   478→                kind: InodeKind::Folder {
   479→                    ipns_name: ipns_name.clone(),
   480→                    encrypted_folder_key: encrypted_folder_key_hex,
   481→                    folder_key: zeroize::Zeroizing::new(folder_key.to_vec()),
   482→                    ipns_private_key: Some(zeroize::Zeroizing::new(ipns_private_key.clone())),
   483→                    children_loaded: true,
   484→                },
   485→                attr,
   486→                children: Some(vec![]),
   487→                write_generation: 0,
   488→            };
   489→
   490→            fs.inodes.insert(inode);
   491→
   492→            if let Some(parent_inode) = fs.inodes.get_mut(parent) {
   493→                if let Some(ref mut children) = parent_inode.children {
   494→                    children.push(ino);
   495→                }
   496→                parent_inode.attr.mtime = SystemTime::now();
   497→                parent_inode.attr.ctime = SystemTime::now();
   498→            }
   499→
   500→            let metadata = crate::crypto::folder::FolderMetadata {
   501→                version: "v2".to_string(),
   502→                children: vec![],
   503→            };
   504→
   505→            let json_bytes = crate::fuse::encrypt_metadata_to_json(
   506→                &metadata, &folder_key,
   507→            )?;
   508→
   509→            let encrypted_ipns_for_tee = if let Some(ref tee_key) = fs.tee_public_key {
   510→                let wrapped = crate::crypto::ecies::wrap_key(&ipns_private_key, tee_key)
   511→                    .map_err(|e| format!("TEE key wrapping failed: {}", e))?;
   512→                Some(hex::encode(&wrapped))
   513→            } else {
   514→                None
   515→            };
   516→            let tee_key_epoch = fs.tee_key_epoch;
   517→
   518→            let (parent_metadata, parent_folder_key, parent_ipns_key, parent_ipns_name, parent_old_cid) =
   519→                fs.build_folder_metadata(parent)?;
   520→
   521→            let api = fs.api.clone();
   522→            let rt = fs.rt.clone();
   523→            let ipns_name_clone = ipns_name.clone();
   524→            let coordinator = fs.publish_coordinator.clone();
   525→
   526→            std::thread::spawn(move || {
   527→                let result = rt.block_on(async {
   528→                    let initial_cid = crate::api::ipfs::upload_content(
   529→                        &api, &json_bytes,
   530→                    ).await?;
   531→
   532→                    let ipns_key_arr: [u8; 32] = ipns_private_key.try_into()
   533→                        .map_err(|_| "Invalid IPNS key length".to_string())?;
   534→                    let value = format!("/ipfs/{}", initial_cid);
   535→                    let record = crate::crypto::ipns::create_ipns_record(
   536→                        &ipns_key_arr, &value, 0, 86_400_000,
   537→                    ).map_err(|e| format!("IPNS record creation failed: {}", e))?;
   538→                    let marshaled = crate::crypto::ipns::marshal_ipns_record(&record)
   539→                        .map_err(|e| format!("IPNS marshal failed: {}", e))?;
   540→
   541→                    use base64::Engine;
   542→                    let record_b64 = base64::engine::general_purpose::STANDARD
   543→                        .encode(&marshaled);
   544→
   545→                    // New folder initial publish: sequence 0, no conflict check needed
   546→                    let req = crate::api::ipns::IpnsPublishRequest {
   547→                        ipns_name: ipns_name_clone.clone(),
   548→                        record: record_b64,
   549→                        metadata_cid: initial_cid,
   550→                        encrypted_ipns_private_key: encrypted_ipns_for_tee,
   551→                        key_epoch: tee_key_epoch,
   552→                        expected_sequence_number: None,
   553→                    };
   554→                    match crate::api::ipns::publish_ipns(&api, &req).await? {
   555→                        crate::api::ipns::PublishResult::Success => {
   556→                            coordinator.record_publish(&ipns_name_clone, 0);
   557→                            log::info!("New folder IPNS published: {}", ipns_name_clone);
   558→                        }
   559→                        crate::api::ipns::PublishResult::Conflict { .. } => {
   560→                            // Sequence 0 should never conflict -- log and continue
   561→                            log::warn!("Unexpected conflict on new folder IPNS publish for {}", ipns_name_clone);
   562→                        }
   563→                    }
   564→
   565→                    let lock = coordinator.get_lock(&parent_ipns_name);
   566→                    let _guard = lock.lock().await;
   567→
   568→                    let parent_json = crate::fuse::encrypt_metadata_to_json(
   569→                        &parent_metadata, &parent_folder_key,
   570→                    )?;
   571→
   572→                    let seq = coordinator.resolve_sequence(&api, &parent_ipns_name).await?;
   573→
   574→                    let parent_meta_cid = crate::api::ipfs::upload_content(
   575→                        &api, &parent_json,
   576→                    ).await?;
   577→
   578→                    let parent_key_arr: [u8; 32] = parent_ipns_key.try_into()
   579→                        .map_err(|_| "Invalid parent IPNS key length".to_string())?;
   580→                    let new_seq = seq + 1;
   581→                    let parent_value = format!("/ipfs/{}", parent_meta_cid);
   582→                    let parent_record = crate::crypto::ipns::create_ipns_record(
   583→                        &parent_key_arr, &parent_value, new_seq, 86_400_000,
   584→                    ).map_err(|e| format!("Parent IPNS record failed: {}", e))?;
   585→                    let parent_marshaled = crate::crypto::ipns::marshal_ipns_record(
   586→                        &parent_record,
   587→                    ).map_err(|e| format!("Parent IPNS marshal failed: {}", e))?;
   588→                    let parent_record_b64 = base64::engine::general_purpose::STANDARD
   589→                        .encode(&parent_marshaled);
   590→
   591→                    // Parent folder publish after mkdir includes conflict detection.
   592→                    // On conflict, log a warning -- the debounced publish queue will
   593→                    // retry the parent metadata on the next cycle.
   594→                    // TODO: Add full re-fetch+merge+retry for parent mkdir publish (v2).
   595→                    let parent_req = crate::api::ipns::IpnsPublishRequest {
   596→                        ipns_name: parent_ipns_name.clone(),
   597→                        record: parent_record_b64,
   598→                        metadata_cid: parent_meta_cid,
   599→                        encrypted_ipns_private_key: None,
   600→                        key_epoch: None,
   601→                        expected_sequence_number: Some(seq.to_string()),
   602→                    };
   603→                    match crate::api::ipns::publish_ipns(&api, &parent_req).await? {
   604→                        crate::api::ipns::PublishResult::Success => {
   605→                            coordinator.record_publish(&parent_ipns_name, new_seq);
   606→                            // Only unpin old CID on successful publish
   607→                            if let Some(old) = parent_old_cid {
   608→                                let _ = crate::api::ipfs::unpin_content(&api, &old).await;
   609→                            }
   610→                            log::info!("Parent metadata published after mkdir");
   611→                        }
   612→                        crate::api::ipns::PublishResult::Conflict { current_sequence_number } => {
   613→                            log::warn!(
   614→                                "Conflict on parent publish after mkdir (expected seq {}, server has {}). \
   615→                                Debounced publish will retry.",
   616→                                seq, current_sequence_number
   617→                            );
   618→                            // Do not record_publish or unpin -- let the next debounced publish pick up fresh seq
   619→                        }
   620→                    }
   621→                    Ok::<(), String>(())
   622→                });
   623→
   624→                if let Err(e) = result {
   625→                    log::error!("Background mkdir publish failed: {}", e);
   626→                }
   627→            });
   628→
   629→            Ok(fuse_attr)
   630→        })();
   631→
   632→        match result {
   633→            Ok(fuse_attr) => {
   634→                reply.entry(&DIR_TTL, &fuse_attr, 0);
   635→            }
   636→            Err(e) => {
   637→                log::error!("mkdir failed: {}", e);
   638→                reply.error(libc::EIO);
   639→            }
   640→        }
   641→    }

   643→    /// Remove an empty directory.
   644→    pub fn handle_rmdir(
   645→        fs: &mut CipherBoxFS,
   646→        parent: u64,
   647→        name: &OsStr,
   648→        reply: ReplyEmpty,
   649→    ) {
   650→        let name_str = match name.to_str() {
   651→            Some(n) => n,
   652→            None => {
   653→                reply.error(libc::ENOENT);
   654→                return;
   655→            }
   656→        };
   657→
   658→        let child_ino = match fs.inodes.find_child(parent, name_str) {
   659→            Some(ino) => ino,
   660→            None => {
   661→                reply.error(libc::ENOENT);
   662→                return;
   663→            }
   664→        };
   665→
   666→        // Capture folder data for bin entry before inode removal
   667→        let bin_entry_data = match fs.inodes.get(child_ino) {
   668→            Some(inode) => {
   669→                match &inode.kind {
   670→                    InodeKind::Folder {
   671→                        ipns_name,
   672→                        encrypted_folder_key,
   673→                        ipns_private_key,
   674→                        ..
   675→                    } => {
   676→                        // Check for non-empty folder (POSIX requirement)
   677→                        if let Some(ref children) = inode.children {
   678→                            if !children.is_empty() {
   679→                                reply.error(libc::ENOTEMPTY);
   680→                                return;
   681→                            }
   682→                        }
   683→
   684→                        let now_ms = SystemTime::now()
   685→                            .duration_since(std::time::UNIX_EPOCH)
   686→                            .unwrap_or_default()
   687→                            .as_millis() as u64;
   688→                        let created_ms = inode.attr.crtime
   689→                            .duration_since(std::time::UNIX_EPOCH)
   690→                            .unwrap_or_default()
   691→                            .as_millis() as u64;
   692→
   693→                        // Build the ECIES-wrapped IPNS private key for the FolderEntry
   694→                        let ipns_key_encrypted = match ipns_private_key {
   695→                            Some(key) => match crate::crypto::ecies::wrap_key(key, &fs.public_key) {
   696→                                Ok(wrapped) => hex::encode(&wrapped),
   697→                                Err(e) => {
   698→                                    log::error!("rmdir: failed to wrap IPNS key for bin entry: {}", e);
   699→                                    reply.error(libc::EIO);
   700→                                    return;
   701→                                }
   702→                            },
   703→                            None => {
   704→                                log::error!("rmdir: missing folder IPNS private key for ino {}", child_ino);
   705→                                reply.error(libc::EIO);
   706→                                return;
   707→                            }
   708→                        };
   709→
   710→                        let folder_entry = crate::crypto::folder::FolderEntry {
   711→                            id: crate::crypto::utils::generate_uuid_v4(),
   712→                            name: inode.name.clone(),
   713→                            ipns_name: ipns_name.clone(),
   714→                            folder_key_encrypted: encrypted_folder_key.clone(),
   715→                            ipns_private_key_encrypted: ipns_key_encrypted,
   716→                            created_at: if created_ms > 0 { created_ms } else { now_ms },
   717→                            modified_at: now_ms,
   718→                        };
   719→
   720→                        Some((inode.name.clone(), folder_entry))
   721→                    }
   722→                    _ => {
   723→                        reply.error(libc::ENOTDIR);
   724→                        return;
   725→                    }
   726→                }
   727→            }
   728→            None => {
   729→                reply.error(libc::ENOENT);
   730→                return;
   731→            }
   732→        };
   733→
   734→        log::debug!("rmdir: {} from parent {}", name_str, parent);
   735→
   736→        let now = SystemTime::now();
   737→
   738→        fs.inodes.remove(child_ino);
   739→
   740→        if let Some(parent_inode) = fs.inodes.get_mut(parent) {
   741→            parent_inode.attr.mtime = now;
   742→            parent_inode.attr.ctime = now;
   743→        }
   744→
   745→        if let Err(e) = fs.update_folder_metadata(parent) {
   746→            log::error!("Failed to update folder metadata after rmdir: {}", e);
   747→        }
   748→
   749→        // Create bin entry and publish to bin IPNS (fire-and-forget)
   750→        if let Some((item_name, folder_entry)) = bin_entry_data {
   751→            let parent_ipns_name = fs.inodes.get(parent)
   752→                .map(|p| match &p.kind {
   753→                    InodeKind::Root { ipns_name, .. } => ipns_name.clone().unwrap_or_default(),
   754→                    InodeKind::Folder { ipns_name, .. } => ipns_name.clone(),
   755→                    _ => String::new(),
   756→                })
   757→                .unwrap_or_default();
   758→
   759→            if parent_ipns_name.is_empty() {
   760→                log::warn!(
   761→                    "rmdir: missing parent IPNS name for parent ino {}, skipping bin publish",
   762→                    parent
   763→                );
   764→            } else {
   765→                let parent_path = build_folder_path(fs, parent);
   766→
   767→                let bin_entry = crate::crypto::bin::BinEntry {
   768→                    id: crate::crypto::utils::generate_uuid_v4(),
   769→                    item_type: crate::crypto::bin::BinItemType::Folder,
   770→                    name: item_name,
   771→                    original_parent_ipns_name: parent_ipns_name,
   772→                    original_path: parent_path,
   773→                    deleted_at: now
   774→                        .duration_since(std::time::UNIX_EPOCH)
   775→                        .unwrap_or_default()
   776→                        .as_millis() as u64,
   777→                    size: 0,
   778→                    mime_type: String::new(),
   779→                    content_cid: None,
   780→                    content_size: None,
   781→                    version_cids: None,
   782→                    file_pointer: None,
   783→                    folder_entry: Some(folder_entry),
   784→                };
   785→
   786→                crate::fuse::spawn_bin_entry_publish(
   787→                    fs.api.clone(),
   788→                    fs.rt.clone(),
   789→                    bin_entry,
   790→                    fs.private_key.clone(),
   791→                    fs.public_key.to_vec(),
   792→                    fs.publish_coordinator.clone(),
   793→                );
   794→            }
   795→        }
   796→
   797→        reply.ok();
   798→    }

   800→    /// Rename or move a file/folder.
   801→    pub fn handle_rename(
   802→        fs: &mut CipherBoxFS,
   803→        parent: u64,
   804→        name: &OsStr,
   805→        newparent: u64,
   806→        newname: &OsStr,
   807→        reply: ReplyEmpty,
   808→    ) {
   809→        log::debug!(
   810→            "rename: {:?} (parent {}) -> {:?} (parent {})",
   811→            name, parent, newname, newparent,
   812→        );
   813→        let name_str = match name.to_str() {
   814→            Some(n) => n,
   815→            None => {
   816→                reply.error(libc::EINVAL);
   817→                return;
   818→            }
   819→        };
   820→        let newname_str = match newname.to_str() {
   821→            Some(n) => n,
   822→            None => {
   823→                reply.error(libc::EINVAL);
   824→                return;
   825→            }
   826→        };
   827→
   828→        // Find source inode (with FUSE-T truncated name fallback)
   829→        let (source_ino, actual_name) = match fs.inodes.find_child(parent, name_str) {
   830→            Some(ino) => (ino, name_str.to_string()),
   831→            None => {
   832→                let parent_inode = match fs.inodes.get(parent) {
   833→                    Some(i) => i,
   834→                    None => {
   835→                        reply.error(libc::ENOENT);
   836→                        return;
   837→                    }
   838→                };
   839→                let children = parent_inode.children.clone().unwrap_or_default();
   840→                let mut matches: Vec<(u64, String)> = Vec::new();
   841→                for &child_ino in &children {
   842→                    if let Some(child) = fs.inodes.get(child_ino) {
   843→                        if is_platform_special(&child.name) {
   844→                            continue;
   845→                        }
   846→                        if child.name.ends_with(name_str) && child.name.len() > name_str.len() {
   847→                            matches.push((child_ino, child.name.clone()));
   848→                        }
   849→                    }
   850→                }
   851→                if matches.len() == 1 {
   852→                    log::debug!(
   853→                        "rename suffix-match: truncated {:?} matched full name {:?}",
   854→                        name_str, matches[0].1
   855→                    );
   856→                    (matches[0].0, matches[0].1.clone())
   857→                } else {
   858→                    log::debug!(
   859→                        "rename failed: {:?} not found (suffix matches: {})",
   860→                        name_str, matches.len()
   861→                    );
   862→                    reply.error(libc::ENOENT);
   863→                    return;
   864→                }
   865→            }
   866→        };
   867→
   868→        let name_str = &actual_name;
   869→
   870→        log::debug!(
   871→            "rename: {} (ino {}) in parent {} -> {} in parent {}",
   872→            name_str, source_ino, parent, newname_str, newparent,
   873→        );
   874→
   875→        // If destination exists, handle replacement
   876→        if let Some(dest_ino) = fs.inodes.find_child(newparent, newname_str) {
   877→            // Self-replace (rename "a" to "a" in same dir): no-op
   878→            if dest_ino == source_ino {
   879→                reply.ok();
   880→                return;
   881→            }
   882→
   883→            if let Some(dest_inode) = fs.inodes.get(dest_ino) {
   884→                // Validate kind compatibility (POSIX: can't replace file with dir or vice versa)
   885→                let source_is_dir = fs.inodes.get(source_ino)
   886→                    .map(|i| matches!(i.kind, InodeKind::Root { .. } | InodeKind::Folder { .. }))
   887→                    .unwrap_or(false);
   888→                let dest_is_dir = matches!(
   889→                    dest_inode.kind,
   890→                    InodeKind::Root { .. } | InodeKind::Folder { .. }
   891→                );
   892→                if source_is_dir && !dest_is_dir {
   893→                    reply.error(libc::ENOTDIR);
   894→                    return;
   895→                }
   896→                if !source_is_dir && dest_is_dir {
   897→                    reply.error(libc::EISDIR);
   898→                    return;
   899→                }
   900→
   901→                match &dest_inode.kind {
   902→                    InodeKind::Folder { .. } => {
   903→                        if let Some(ref children) = dest_inode.children {
   904→                            if !children.is_empty() {
   905→                                reply.error(libc::ENOTEMPTY);
   906→                                return;
   907→                            }
   908→                        }
   909→                    }
   910→                    InodeKind::File { cid, .. } => {
   911→                        if !cid.is_empty() {
   912→                            let cid_clone = cid.clone();
   913→                            let api = fs.api.clone();
   914→                            fs.rt.spawn(async move {
   915→                                let _ = crate::api::ipfs::unpin_content(
   916→                                    &api, &cid_clone,
   917→                                ).await;
   918→                            });
   919→                        }
   920→                    }
   921→                    _ => {}
   922→                }
   923→            }
   924→            fs.inodes.remove(dest_ino);
   925→        }
   926→
   927→        // Remove source from old parent's name index (NFC-normalized)
   928→        {
   929→            use unicode_normalization::UnicodeNormalization;
   930→            let nfc_key: String = name_str.nfc().collect();
   931→            fs.inodes.name_to_ino.remove(&(parent, nfc_key));
   932→        }
   933→
