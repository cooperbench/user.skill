257	    if runtime.kind not in RUNTIME_KINDS:

239	    values: Mapping[str, Any],

241	    command_spec = orchestrator.runtime.command
242	    if command_spec is None:
243	        raise OrchestratorRunnerError(f"orchestrator {orchestrator.id!r} has no command runtime")
244	    placeholders = _placeholder_values(orchestrator, request, values)

514	def _validate_out_requirement(orchestrator: OrchestratorDefinition, request: OrchestratorRunRequest) -> None:
515	    if request.out is not None:
516	        return
517	    if _orchestrator_requires_output_path(orchestrator):
518	        raise OrchestratorRunnerError(f"--out is required for {orchestrator.id}")
519	    if request.dry_run:
520	        return
521	    if orchestrator.runtime.kind == "command" and _command_runtime_requires_out(orchestrator, request):
522	        raise OrchestratorRunnerError("--out is required for command runtime placeholders")
523	    raise OrchestratorRunnerError("--out is required for orchestrator execution")

526	def _command_runtime_requires_out(orchestrator: OrchestratorDefinition, request: OrchestratorRunRequest) -> bool:
527	    command = orchestrator.runtime.command
528	    if command is None:
529	        return False
530	    values = [*command.argv]
531	    if command.cwd:
532	        values.append(command.cwd)
533	    values.extend(command.env.values())
534	    if any(_uses_placeholder(value, "out") for value in values):
535	        return True
536	    for output in orchestrator.outputs:
537	        if output.name in request.outputs or (output.placeholder and output.placeholder in request.outputs):
538	            continue
539	        if output.path_template is None or _uses_placeholder(output.path_template, "out"):
540	            return True
541	    return False
