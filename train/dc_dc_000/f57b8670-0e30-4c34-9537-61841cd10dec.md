161	def new(
162	    title: str,
163	    *,
164	    body: str = "",
165	    tags: Sequence[str] | None = None,
166	    store: Store | None = None,
167	    cwd: Path | None = None,
168	) -> str:
169	    """Create a new ticket and return its ULID.
170	
171	    Parameters
172	    ----------
173	    title:
174	        Ticket title (required).
175	    body:
176	        Markdown body.  ``"-"`` means read from stdin (empty input is rejected).
177	    tags:
178	        Optional tags.
179	    store:
180	        Explicit store.  If *None*, resolved from environment.
181	    cwd:
182	        Working directory for git operations.
183	
184	    Returns
185	    -------
186	    str
187	        The ULID of the newly created ticket (printed to stdout).
188	    """
189	    # Handle stdin body
190	    if body == "-":
191	        body = sys.stdin.read()
192	        if not body.strip():
193	            raise ValueError("stdin body is empty — body is required")
194	
195	    if store is None:
196	        store = _resolve_store()
197	
198	    source, turn_id, actor_id = _derive_source()
199	    ticket_id = str(ULID())
200	    slug = slugify(title)
201	    now = datetime.now(timezone.utc)
202	
203	    # Determine codebase_id (cloud mode only)
204	    codebase_id: str | None
205	    if store is not None and is_cloud_store(store):
206	        codebase_id = _ensure_codebase(store, cwd)
207	    else:
208	        codebase_id = None
209	
210	    # Build record dict for file
211	    record: dict[str, Any] = {
212	        "id": ticket_id,
213	        "title": title,
214	        "status": "open",
215	        "source": source,
216	        "tags": list(tags or []),
217	        "filed_by_actor_id": actor_id,
218	        "filed_in_turn_id": turn_id,
219	        "codebase_id": codebase_id,
220	        "created_at": now,
221	        "last_edited_at": now,
222	        "resolution_note": None,
223	        "addressed_at": None,
224	        "epics": [],
225	        "__body__": body,
226	    }
227	
228	    # Write file (both modes)
229	    if cwd:
230	        repo_root = str(cwd)
231	    else:
232	        repo_root = os.getcwd()
233	    fpath = ticket_file_path(repo_root, ticket_id, slug)
234	    write_ticket_file(fpath, record)
235	
236	    # Cloud: write DB row
237	    if store is not None and is_cloud_store(store) and codebase_id:
238	        from megaplan.store.db import DBStore
239	
240	        db: DBStore = store  # type: ignore[assignment]
241	        db.create_ticket(
242	            codebase_id=codebase_id,
243	            title=title,
244	            body=body,
245	            source=source,
246	            tags=list(tags or []),
247	            filed_by_actor_id=actor_id,
248	            filed_in_turn_id=turn_id,
249	            slug=slug,
250	            ticket_id=ticket_id,
251	        )
252	
253	    # Print only the ULID to stdout (per spec)
254	    print(ticket_id, flush=True)
255	    return ticket_id

346	_SORT_KEYS = {"created", "edited", "length", "title"}
347	
348	
349	def search(
350	    keywords: Sequence[str] | None = None,
351	    *,
352	    keywords_all: bool = False,
353	    status: str | None = None,
354	    tags: Sequence[str] | None = None,
355	    projects: Sequence[str | Path] | None = None,
356	    all_projects: bool = False,
357	    sort: str = "created",
358	    order: str = "desc",
359	    limit: int | None = None,
360	    snippet: bool = False,
361	    snippet_width: int = 120,
362	    store: Store | None = None,
363	    cwd: Path | None = None,
364	) -> list[dict[str, Any]]:
365	    """Search tickets across local files and/or the cloud DB.
366	
367	    Scope resolution
368	    ----------------
369	    * ``projects`` given → restrict to those repos (path or known-name).
370	    * ``all_projects`` → every known repo (local) or every codebase (cloud).
371	    * neither → current repo only (default).
372	
373	    Keyword matching is case-insensitive substring across title + body +
374	    tags + resolution_note.  Default is OR (any keyword matches); set
375	    ``keywords_all=True`` for AND semantics.
376	
377	    Returns a list of result dicts.  Each result includes a ``project``
378	    field (path string or ``owner/name``) when results span multiple
379	    projects, plus a ``snippet`` field when *snippet* is true and at
380	    least one keyword matched.
381	    """
382	    if sort not in _SORT_KEYS:
383	        raise ValueError(f"sort must be one of {_SORT_KEYS!r}, got {sort!r}")
384	    if order.lower() not in {"asc", "desc"}:
385	        raise ValueError(f"order must be 'asc' or 'desc', got {order!r}")
386	
387	    if store is None:
388	        store = _resolve_store()
389	
390	    kw_list = [k for k in (keywords or []) if k]
391	
392	    results: list[dict[str, Any]] = []
393	
394	    if store is not None and is_cloud_store(store):
395	        from megaplan.store.db import DBStore
396	        from .registry import resolve_project as _resolve_project
397	
398	        db: DBStore = store  # type: ignore[assignment]
399	        codebase_ids: list[str] | None = None
400	        # Map projects → codebase_ids
401	        if projects:
402	            ids: list[str] = []
403	            for p in projects:
404	                resolved = _resolve_project_to_codebase(db, p)
405	                if resolved:
406	                    ids.append(resolved)
407	            codebase_ids = ids or [""]  # empty → guaranteed no rows
408	        elif not all_projects:
409	            cur_id = _resolve_codebase_id(store, cwd)
410	            if cur_id:
411	                codebase_ids = [cur_id]
412	            # else: leave as None (matches none) — but keep behaviour where
413	            # missing codebase falls through to "nothing" rather than "all".
414	            else:
415	                codebase_ids = [""]
416	
417	        tickets = db.list_tickets(
418	            codebase_ids=codebase_ids,
419	            status=status,
420	            tags=tags,
421	            keywords=kw_list or None,
422	            keywords_all=keywords_all,
423	            sort=sort,
424	            order=order,
425	            limit=limit,
426	        )
427	        # Pre-load codebases for project labelling
428	        cb_cache: dict[str, str] = {}
429	        for t in tickets:
430	            project_label = ""
431	            if t.codebase_id:
432	                if t.codebase_id not in cb_cache:
433	                    cb = db.load_codebase(t.codebase_id)
434	                    cb_cache[t.codebase_id] = (
435	                        f"{cb.owner}/{cb.name}" if cb and cb.owner and cb.name else t.codebase_id
436	                    )
437	                project_label = cb_cache[t.codebase_id]
438	            d = {
439	                "id": t.id,
440	                "title": t.title,
441	                "status": t.status,
442	                "source": t.source,
443	                "tags": t.tags,
444	                "slug": t.slug,
445	                "codebase_id": t.codebase_id,
446	                "project": project_label,
447	                "created_at": t.created_at.isoformat() if t.created_at else None,
448	                "last_edited_at": t.last_edited_at.isoformat() if t.last_edited_at else None,
449	                "resolution_note": t.resolution_note,
450	                "body": t.body or "",
451	            }
452	            if snippet and kw_list:
453	                d["snippet"] = _make_snippet(t.body or "", t.title or "", kw_list, snippet_width)
454	            results.append(d)
455	        return results
456	
457	    # ----- local-only mode -----
458	    scan_roots = _resolve_scan_roots(projects=projects, all_projects=all_projects, cwd=cwd)
459	
460	    for repo_root in scan_roots:
461	        project_label = _project_label_for(repo_root)
462	        for fpath, fm in iterate_ticket_files(repo_root):
463	            if status and fm.get("status") != status:
464	                continue
465	            if tags:
466	                file_tags = set(fm.get("tags") or [])
467	                if not file_tags.intersection(tags):
468	                    continue
469	            title = fm.get("title") or ""
470	            body = fm.get("__body__") or ""
471	            tag_blob = " ".join(fm.get("tags") or [])
472	            res_note = fm.get("resolution_note") or ""
473	            if kw_list:
474	                haystack = (title + "\n" + body + "\n" + tag_blob + "\n" + res_note).lower()
475	                matches = [kw.lower() in haystack for kw in kw_list]
476	                if keywords_all:
477	                    if not all(matches):
478	                        continue
479	                else:
480	                    if not any(matches):
481	                        continue
482	            d = {
483	                "id": fm.get("id"),
484	                "title": title,
485	                "status": fm.get("status"),
486	                "source": fm.get("source"),
487	                "tags": fm.get("tags", []),
488	                "slug": slugify(title),
489	                "codebase_id": fm.get("codebase_id"),
490	                "project": project_label,
491	                "created_at": _iso(fm.get("created_at")),
492	                "last_edited_at": _iso(fm.get("last_edited_at")),
493	                "resolution_note": fm.get("resolution_note"),
494	                "body": body,
495	                "epics": fm.get("epics", []),
496	            }
497	            if snippet and kw_list:
498	                d["snippet"] = _make_snippet(body, title, kw_list, snippet_width)
499	            results.append(d)
500	
501	    # In-process sort for local mode (matches DB ORDER BY semantics)
502	    sort_key = {
503	        "created": lambda d: d.get("created_at") or "",
504	        "edited": lambda d: d.get("last_edited_at") or "",
505	        "length": lambda d: len(d.get("body") or ""),
506	        "title": lambda d: (d.get("title") or "").lower(),
507	    }[sort]
508	    results.sort(key=sort_key, reverse=(order.lower() == "desc"))
509	    if limit is not None:
510	        results = results[:limit]
511	    return results

514	def _resolve_project_to_codebase(db: Any, project: str | Path) -> str | None:
515	    """Resolve a project spec to a codebase_id (cloud).
516	
517	    Accepts: an ``owner/name`` string, a bare ``name`` (must be unique),
518	    or a filesystem path (uses git root_commit_sha lookup).
519	    """
520	    from megaplan.store.db import DBStore  # noqa: F401  (typing hint only)
521	
522	    spec = str(project)
523	    # Path-like first
524	    p = Path(spec).expanduser()
525	    if p.is_dir():
526	        try:
527	            sha = repo_root_sha(p)
528	        except Exception:
529	            sha = None
530	        if sha:
531	            cb = db.resolve_codebase_by_root_sha(sha)
532	            if cb:
533	                return cb.id
534	    # owner/name
535	    if "/" in spec:
536	        owner, name = spec.split("/", 1)
537	        cb = db.find_codebase(owner, name)
538	        if cb:
539	            return cb.id
540	    # bare name — scan list_codebases
541	    matches = [c for c in db.list_codebases() if (c.name or "").lower() == spec.lower()]
542	    if len(matches) == 1:
543	        return matches[0].id
544	    return None

584	def _make_snippet(body: str, title: str, keywords: Sequence[str], width: int) -> str:
585	    """Return a single-line snippet centred on the first keyword hit."""
586	    text = (title + " — " + body) if body else title
587	    flat = " ".join(text.split())  # collapse whitespace
588	    low = flat.lower()
589	    idx = -1
590	    for kw in keywords:
591	        i = low.find(kw.lower())
592	        if i >= 0 and (idx < 0 or i < idx):
593	            idx = i
594	    if idx < 0:
595	        return flat[: width] + ("…" if len(flat) > width else "")
596	    half = width // 2
597	    start = max(0, idx - half)
598	    end = min(len(flat), start + width)
599	    snippet = flat[start:end]
600	    if start > 0:
601	        snippet = "…" + snippet
602	    if end < len(flat):
603	        snippet = snippet + "…"
604	    return snippet

2	
3	Tickets are short, repo-scoped notes on problems, bugs, or "we should look at this" observations against a codebase. They live as committed `.md` files under `<repo>/.megaplan/tickets/` and (when a cloud store is configured) mirror to `tickets` / `ticket_epics` tables. Tickets are inert until folded into an epic — they capture problems, not work.
4	
5	## When to file a ticket
6	
7	Reach for `megaplan ticket new` when:
8	- During epic or plan execution you notice an **out-of-scope problem** that should be tracked but isn't blocking the current task.
9	- The user explicitly asks you to **capture an observation** ("file a ticket for that", "make a note of this for later").
10	- You want to log a rough edge or potential bug whose fix would be its own piece of work.
11	
12	Do **not** use tickets for:
13	- Guidance/corrections to *how the tooling itself behaves* — that's `megaplan feedback` (a separate, plan-scoped concept).
