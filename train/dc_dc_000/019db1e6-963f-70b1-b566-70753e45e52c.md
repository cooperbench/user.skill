### Step 3: Add a `useHeadMeta` hook for per-route title + meta + og/twitter (`src/hooks/useHeadMeta.ts`)
**Scope:** Small
1. **Create** `src/hooks/useHeadMeta.ts`. A small imperative hook:
   ```ts
   type HeadMeta = {
     title: string;
     description: string;
     ogImage?: string;
     ogUrl?: string;
   };
   export function useHeadMeta({ title, description, ogImage, ogUrl }: HeadMeta) {
     useEffect(() => {
       const prevTitle = document.title;
       document.title = title;
       const setMeta = (selector: string, value: string) => {
         const el = document.head.querySelector<HTMLMetaElement>(selector);
         if (el) el.setAttribute('content', value);
       };
       const prev = {
         desc: document.head.querySelector<HTMLMetaElement>('meta[name="description"]')?.content,
         ogTitle: document.head.querySelector<HTMLMetaElement>('meta[property="og:title"]')?.content,
         ogDesc: document.head.querySelector<HTMLMetaElement>('meta[property="og:description"]')?.content,
         ogImg: document.head.querySelector<HTMLMetaElement>('meta[property="og:image"]')?.content,
         ogUrl: document.head.querySelector<HTMLMetaElement>('meta[property="og:url"]')?.content,
         twTitle: document.head.querySelector<HTMLMetaElement>('meta[name="twitter:title"]')?.content,
         twDesc: document.head.querySelector<HTMLMetaElement>('meta[name="twitter:description"]')?.content,
       };
       setMeta('meta[name="description"]', description);
       setMeta('meta[property="og:title"]', title);
       setMeta('meta[property="og:description"]', description);
       setMeta('meta[name="twitter:title"]', title);
       setMeta('meta[name="twitter:description"]', description);
       if (ogImage) setMeta('meta[property="og:image"]', ogImage);
       if (ogUrl)   setMeta('meta[property="og:url"]', ogUrl);
       return () => {
         document.title = prevTitle;
         if (prev.desc !== undefined)    setMeta('meta[name="description"]', prev.desc);
         if (prev.ogTitle !== undefined) setMeta('meta[property="og:title"]', prev.ogTitle);
         if (prev.ogDesc !== undefined)  setMeta('meta[property="og:description"]', prev.ogDesc);
         if (prev.ogImg !== undefined)   setMeta('meta[property="og:image"]', prev.ogImg);
         if (prev.ogUrl !== undefined)   setMeta('meta[property="og:url"]', prev.ogUrl);
         if (prev.twTitle !== undefined) setMeta('meta[name="twitter:title"]', prev.twTitle);
         if (prev.twDesc !== undefined)  setMeta('meta[name="twitter:description"]', prev.twDesc);
       };
     }, [title, description, ogImage, ogUrl]);
   }
   ```
   No new dependency. Handles title + description + og:title/description/image/url + twitter:title/description, all with restore-on-unmount so back-navigation doesn't leak.

### Step 3: `useHeadMeta` hook covering every mutable meta tag in `index.html` (`src/hooks/useHeadMeta.ts`)
**Scope:** Small
1. **Create** `src/hooks/useHeadMeta.ts`:
   ```ts
   type HeadMeta = {
     title: string;
     description: string;
     ogImage?: string;      // falls back to baseline if omitted
     ogUrl?: string;
     twitterImage?: string; // falls back to baseline if omitted
   };
   const SELECTORS = {
     desc: 'meta[name="description"]',
     ogTitle: 'meta[property="og:title"]',
     ogDesc: 'meta[property="og:description"]',
     ogImg: 'meta[property="og:image"]',
     ogUrl: 'meta[property="og:url"]',
     twTitle: 'meta[name="twitter:title"]',
     twDesc: 'meta[name="twitter:description"]',
     twImg: 'meta[name="twitter:image"]',
   } as const;
   export function useHeadMeta({ title, description, ogImage, ogUrl, twitterImage }: HeadMeta) {
     useEffect(() => {
       const prevTitle = document.title;
       const get = (sel: string) =>
         document.head.querySelector<HTMLMetaElement>(sel)?.content;
       const set = (sel: string, v: string | undefined) => {
         if (v === undefined) return;
         const el = document.head.querySelector<HTMLMetaElement>(sel);
         if (el) el.setAttribute('content', v);
       };
       const prev = Object.fromEntries(
         Object.entries(SELECTORS).map(([k, sel]) => [k, get(sel)])
       ) as Record<keyof typeof SELECTORS, string | undefined>;

       document.title = title;
       set(SELECTORS.desc, description);
       set(SELECTORS.ogTitle, title);
       set(SELECTORS.ogDesc, description);
       set(SELECTORS.ogUrl, ogUrl);
       set(SELECTORS.ogImg, ogImage);
       set(SELECTORS.twTitle, title);
       set(SELECTORS.twDesc, description);
       set(SELECTORS.twImg, twitterImage);
       return () => {
         document.title = prevTitle;
         (Object.keys(SELECTORS) as Array<keyof typeof SELECTORS>).forEach((k) => {
           if (prev[k] !== undefined) set(SELECTORS[k], prev[k]);
         });
       };
     }, [title, description, ogImage, ogUrl, twitterImage]);
   }
   ```
   Covers every mutable tag in `index.html` (title, description, og:title/description/image/url, twitter:title/description/image). Omitted fields fall through to whatever the baseline was at mount — no accidental clobbering.
