   259→            normalized_dimensions=normalized_dimensions,

   263→        # this is a first run for these dimensions, not a rerun.

   267→                blocking_dims=blocking_dims,
   268→            )

   280→                )

   287→            colorize(
   288→                f"  Cleared stale review markers: {', '.join(sorted(cleared))}",
   289→                "cyan",
   290→            )
