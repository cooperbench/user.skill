399	def detect_provider_for_model(
400	    model_name: str,
401	    current_provider: str,
402	) -> Optional[tuple[str, str]]:
403	    """Auto-detect the best provider for a model name.
404	
405	    Returns ``(provider_id, model_name)`` — the model name may be remapped
406	    (e.g. bare ``deepseek-chat`` → ``deepseek/deepseek-chat`` for OpenRouter).
407	    Returns ``None`` when no confident match is found.
408	
409	    Priority:
410	    0. Bare provider name → switch to that provider's default model
411	    1. Direct provider with credentials (highest)
412	    2. Direct provider without credentials → remap to OpenRouter slug
413	    3. OpenRouter catalog match
414	    """
415	    name = (model_name or "").strip()
416	    if not name:
417	        return None
418	
419	    name_lower = name.lower()
420	
421	    # --- Step 0: bare provider name typed as model ---
422	    # If someone types `/model nous` or `/model anthropic`, treat it as a
423	    # provider switch and pick the first model from that provider's catalog.
424	    # Skip "custom" and "openrouter" — custom has no model catalog, and
425	    # openrouter requires an explicit model name to be useful.
426	    resolved_provider = _PROVIDER_ALIASES.get(name_lower, name_lower)
427	    if resolved_provider not in {"custom", "openrouter"}:
428	        default_models = _PROVIDER_MODELS.get(resolved_provider, [])
429	        if (
430	            resolved_provider in _PROVIDER_LABELS
431	            and default_models
432	            and resolved_provider != normalize_provider(current_provider)
433	        ):
434	            return (resolved_provider, default_models[0])
435	
436	    # Aggregators list other providers' models — never auto-switch TO them
437	    _AGGREGATORS = {"nous", "openrouter"}
438	
439	    # If the model belongs to the current provider's catalog, don't suggest switching
440	    current_models = _PROVIDER_MODELS.get(current_provider, [])
441	    if any(name_lower == m.lower() for m in current_models):
442	        return None
443	
444	    # --- Step 1: check static provider catalogs for a direct match ---
445	    direct_match: Optional[str] = None
446	    for pid, models in _PROVIDER_MODELS.items():
447	        if pid == current_provider or pid in _AGGREGATORS:
448	            continue
449	        if any(name_lower == m.lower() for m in models):
450	            direct_match = pid
451	            break
452	
453	    if direct_match:
454	        # Check if we have credentials for this provider
455	        has_creds = False
456	        try:
457	            from hermes_cli.auth import PROVIDER_REGISTRY
458	            pconfig = PROVIDER_REGISTRY.get(direct_match)
459	            if pconfig:
460	                import os
461	                for env_var in pconfig.api_key_env_vars:
462	                    if os.getenv(env_var, "").strip():
463	                        has_creds = True
464	                        break
465	        except Exception:
466	            pass
467	
468	        if has_creds:
469	            return (direct_match, name)
470	
471	        # No direct creds — try to find this model on OpenRouter instead
472	        or_slug = _find_openrouter_slug(name)
473	        if or_slug:
474	            return ("openrouter", or_slug)
475	        # Still return the direct provider — credential resolution will
476	        # give a clear error rather than silently using the wrong provider
477	        return (direct_match, name)
478	
479	    # --- Step 2: check OpenRouter catalog ---
480	    # First try exact match (handles provider/model format)
481	    or_slug = _find_openrouter_slug(name)
482	    if or_slug:
483	        if current_provider != "openrouter":
484	            return ("openrouter", or_slug)
485	        # Already on openrouter, just return the resolved slug
486	        if or_slug != name:
487	            return ("openrouter", or_slug)
488	        return None  # already on openrouter with matching name
489	
490	    return None

493	def _find_openrouter_slug(model_name: str) -> Optional[str]:
494	    """Find the full OpenRouter model slug for a bare or partial model name.
495	
496	    Handles:
497	    - Exact match: ``anthropic/claude-opus-4.6`` → as-is
498	    - Bare name: ``deepseek-chat`` → ``deepseek/deepseek-chat``
499	    - Bare name: ``claude-opus-4.6`` → ``anthropic/claude-opus-4.6``
500	    """
501	    name_lower = model_name.strip().lower()
502	    if not name_lower:
503	        return None
504	
505	    # Exact match (already has provider/ prefix)
506	    for mid, _ in OPENROUTER_MODELS:
507	        if name_lower == mid.lower():
508	            return mid
509	
510	    # Try matching just the model part (after the /)
511	    for mid, _ in OPENROUTER_MODELS:
512	        if "/" in mid:
513	            _, model_part = mid.split("/", 1)
514	            if name_lower == model_part.lower():
515	                return mid
516	
517	    return None

684	def _copilot_catalog_item_is_text_model(item: dict[str, Any]) -> bool:
685	    model_id = str(item.get("id") or "").strip()
686	    if not model_id:
687	        return False
688	
689	    if item.get("model_picker_enabled") is False:
690	        return False
691	
692	    capabilities = item.get("capabilities")
693	    if isinstance(capabilities, dict):
694	        model_type = str(capabilities.get("type") or "").strip().lower()
695	        if model_type and model_type != "chat":
696	            return False
697	
698	    supported_endpoints = item.get("supported_endpoints")
699	    if isinstance(supported_endpoints, list):
700	        normalized_endpoints = {
701	            str(endpoint).strip()
702	            for endpoint in supported_endpoints

704	        }
705	        if normalized_endpoints and not normalized_endpoints.intersection(
706	            {"/chat/completions", "/responses", "/v1/messages"}
707	        ):
708	            return False
709	
710	    return True

716	    """Fetch the live GitHub Copilot model catalog for this account."""

803	    *,
804	    catalog: Optional[list[dict[str, Any]]] = None,
805	    api_key: Optional[str] = None,

808	    if not raw:

812	    alias = _COPILOT_MODEL_ALIASES.get(raw)

831	        seen.add(candidate)
832	        if candidate in _COPILOT_MODEL_ALIASES:
833	            return _COPILOT_MODEL_ALIASES[candidate]
834	        if candidate in catalog_ids:

839	    return raw

870	    model_id: Optional[str],
871	    *,
872	    catalog: Optional[list[dict[str, Any]]] = None,
873	    api_key: Optional[str] = None,
874	) -> str:
875	    """Determine the API mode for a Copilot model.
876	
877	    Uses the model ID pattern (matching opencode's approach) as the
878	    primary signal.  Falls back to the catalog's ``supported_endpoints``
879	    only for models not covered by the pattern check.
880	    """
881	    normalized = normalize_copilot_model_id(model_id, catalog=catalog, api_key=api_key)
882	    if not normalized:
883	        return "chat_completions"
884	
885	    # Primary: model ID pattern (matches opencode's shouldUseCopilotResponsesApi)
886	    if _should_use_copilot_responses_api(normalized):
887	        return "codex_responses"
888	
889	    # Secondary: check catalog for non-GPT-5 models (Claude via /v1/messages, etc.)
890	    if catalog is None and api_key:
891	        catalog = fetch_github_model_catalog(api_key=api_key)
892	
893	    if catalog:
894	        catalog_entry = next((item for item in catalog if item.get("id") == normalized), None)
895	        if isinstance(catalog_entry, dict):
896	            supported_endpoints = {
897	                str(endpoint).strip()
898	                for endpoint in (catalog_entry.get("supported_endpoints") or [])
899	                if str(endpoint).strip()
900	            }
901	            # For non-GPT-5 models, check if they only support messages API
902	            if "/v1/messages" in supported_endpoints and "/chat/completions" not in supported_endpoints:
903	                return "anthropic_messages"
904	
905	    return "chat_completions"

957	    """Probe an OpenAI-compatible ``/models`` endpoint with light URL heuristics."""

959	    if not normalized:

1006	                    "used_fallback": is_fallback,

1012	        "models": None,
1013	        "probed_url": tried[-1] if tried else normalized.rstrip("/") + "/models",

117	def generate_hermes_tools_module(enabled_tools: List[str]) -> str:
118	    """
119	    Build the source code for the hermes_tools.py stub module.
120	
121	    Only tools in both SANDBOX_ALLOWED_TOOLS and enabled_tools get stubs.
122	    """
123	    tools_to_generate = sorted(SANDBOX_ALLOWED_TOOLS & set(enabled_tools))
124	
125	    stub_functions = []
126	    export_names = []

135	        )

142	_sock = None

146	# Convenience helpers (avoid common scripting pitfalls)

151	    Use this instead of json.loads() when parsing output from terminal()
152	    or web_extract() that may contain raw tabs/newlines in strings."""
153	    return json.loads(text, strict=False)

164	def retry(fn, max_attempts=3, delay=2):
165	    """Retry a function up to max_attempts times with exponential backoff.
166	    Use for transient failures (network errors, API rate limits):
167	        result = retry(lambda: terminal("gh issue list ..."))
168	    """
169	    last_err = None
170	    for attempt in range(max_attempts):
171	        try:
172	            return fn()
173	        except Exception as e:
174	            last_err = e
175	            if attempt < max_attempts - 1:
176	                time.sleep(delay * (2 ** attempt))
177	    raise last_err

179	def _connect():
180	    global _sock
181	    if _sock is None:
182	        _sock = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
183	        _sock.connect(os.environ["HERMES_RPC_SOCKET"])
184	        _sock.settimeout(300)
185	    return _sock

187	def _call(tool_name, args):
188	    """Send a tool call to the parent process and return the parsed result."""
189	    conn = _connect()
190	    request = json.dumps({"tool": tool_name, "args": args}) + "\\n"
191	    conn.sendall(request.encode())
192	    buf = b""
193	    while True:
194	        chunk = conn.recv(65536)
195	        if not chunk:
196	            raise RuntimeError("Agent process disconnected")
197	        buf += chunk
198	        if buf.endswith(b"\\n"):
199	            break
200	    raw = buf.decode().strip()
201	    result = json.loads(raw)
202	    if isinstance(result, str):
203	        try:
204	            return json.loads(result)
205	        except (json.JSONDecodeError, TypeError):
206	            return result
207	    return result
208	
209	'''
210	
211	    return header + "\n".join(stub_functions)

222	def _rpc_server_loop(
223	    server_sock: socket.socket,
224	    task_id: str,
225	    tool_call_log: list,
226	    tool_call_counter: list,   # mutable [int] so the thread can increment
227	    max_tool_calls: int,
228	    allowed_tools: frozenset,
229	):
230	    """
231	    Accept one client connection and dispatch tool-call requests until
232	    the client disconnects or the call limit is reached.
233	    """
234	    from model_tools import handle_function_call
235	
236	    conn = None
237	    try:
238	        server_sock.settimeout(5)
239	        conn, _ = server_sock.accept()
240	        conn.settimeout(300)
241	
242	        buf = b""
243	        while True:
244	            try:
245	                chunk = conn.recv(65536)
246	            except socket.timeout:
247	                break
248	            if not chunk:
249	                break
250	            buf += chunk
251	
252	            # Process all complete newline-delimited messages in the buffer
253	            while b"\n" in buf:
254	                line, buf = buf.split(b"\n", 1)
255	                line = line.strip()
256	                if not line:
257	                    continue
258	
259	                call_start = time.monotonic()
260	                try:
261	                    request = json.loads(line.decode())
262	                except (json.JSONDecodeError, UnicodeDecodeError) as exc:
263	                    resp = json.dumps({"error": f"Invalid RPC request: {exc}"})
264	                    conn.sendall((resp + "\n").encode())
265	                    continue
266	
267	                tool_name = request.get("tool", "")
268	                tool_args = request.get("args", {})
269	
270	                # Enforce the allow-list
271	                if tool_name not in allowed_tools:
272	                    available = ", ".join(sorted(allowed_tools))
273	                    resp = json.dumps({
274	                        "error": (
275	                            f"Tool '{tool_name}' is not available in execute_code. "
276	                            f"Available: {available}"
277	                        )
278	                    })
279	                    conn.sendall((resp + "\n").encode())
280	                    continue
281	
282	                # Enforce tool call limit
283	                if tool_call_counter[0] >= max_tool_calls:
284	                    resp = json.dumps({
285	                        "error": (
286	                            f"Tool call limit reached ({max_tool_calls}). "
287	                            "No more tool calls allowed in this execution."
288	                        )
289	                    })
290	                    conn.sendall((resp + "\n").encode())
291	                    continue
292	
293	                # Strip forbidden terminal parameters
294	                if tool_name == "terminal" and isinstance(tool_args, dict):
295	                    for param in _TERMINAL_BLOCKED_PARAMS:
296	                        tool_args.pop(param, None)
297	
298	                # Dispatch through the standard tool handler.
299	                # Suppress stdout/stderr from internal tool handlers so
300	                # their status prints don't leak into the CLI spinner.
301	                try:
302	                    _real_stdout, _real_stderr = sys.stdout, sys.stderr
303	                    devnull = open(os.devnull, "w")
304	                    try:
305	                        sys.stdout = devnull
306	                        sys.stderr = devnull
307	                        result = handle_function_call(
308	                            tool_name, tool_args, task_id=task_id
309	                        )
310	                    finally:
311	                        sys.stdout, sys.stderr = _real_stdout, _real_stderr
312	                        devnull.close()
313	                except Exception as exc:
314	                    logger.error("Tool call failed in sandbox: %s", exc, exc_info=True)
315	                    result = json.dumps({"error": str(exc)})
316	
317	                tool_call_counter[0] += 1
318	                call_duration = time.monotonic() - call_start
319	
320	                # Log for observability
321	                args_preview = str(tool_args)[:80]
322	                tool_call_log.append({
323	                    "tool": tool_name,
324	                    "args_preview": args_preview,
325	                    "duration": round(call_duration, 2),
326	                })
327	
328	                conn.sendall((result + "\n").encode())
329	
330	    except socket.timeout:
331	        logger.debug("RPC listener socket timeout")
332	    except OSError as e:
333	        logger.debug("RPC listener socket error: %s", e, exc_info=True)
334	    finally:
335	        if conn:
336	            try:
337	                conn.close()
338	            except OSError as e:
339	                logger.debug("RPC conn close error: %s", e)
340

346	def execute_code(
347	    code: str,
348	    task_id: Optional[str] = None,
349	    enabled_tools: Optional[List[str]] = None,
350	) -> str:
351	    """
352	    Run a Python script in a sandboxed child process with RPC access
353	    to a subset of Hermes tools.
354	
355	    Args:
356	        code:          Python source code to execute.
357	        task_id:       Session task ID for tool isolation (terminal env, etc.).
358	        enabled_tools: Tool names enabled in the current session. The sandbox
359	                       gets the intersection with SANDBOX_ALLOWED_TOOLS.
360	
361	    Returns:
362	        JSON string with execution results.
363	    """
364	    if not SANDBOX_AVAILABLE:
365	        return json.dumps({
366	            "error": "execute_code is not available on Windows. Use normal tool calls instead."
367	        })
368	
369	    if not code or not code.strip():
370	        return json.dumps({"error": "No code provided."})
371	
372	    # Import interrupt event from terminal_tool (cooperative cancellation)
373	    from tools.terminal_tool import _interrupt_event
374	
375	    # Resolve config
376	    _cfg = _load_config()
377	    timeout = _cfg.get("timeout", DEFAULT_TIMEOUT)
378	    max_tool_calls = _cfg.get("max_tool_calls", DEFAULT_MAX_TOOL_CALLS)
379	
380	    # Determine which tools the sandbox can call
381	    session_tools = set(enabled_tools) if enabled_tools else set()
382	    sandbox_tools = frozenset(SANDBOX_ALLOWED_TOOLS & session_tools)
383	
384	    if not sandbox_tools:
385	        sandbox_tools = SANDBOX_ALLOWED_TOOLS
386	
387	    # --- Set up temp directory with hermes_tools.py and script.py ---
388	    tmpdir = tempfile.mkdtemp(prefix="hermes_sandbox_")
389	    # Use /tmp on macOS to avoid the long /var/folders/... path that pushes
390	    # Unix domain socket paths past the 104-byte macOS AF_UNIX limit.
391	    # On Linux, tempfile.gettempdir() already returns /tmp.
392	    _sock_tmpdir = "/tmp" if sys.platform == "darwin" else tempfile.gettempdir()
393	    sock_path = os.path.join(_sock_tmpdir, f"hermes_rpc_{uuid.uuid4().hex}.sock")
394	
395	    tool_call_log: list = []
396	    tool_call_counter = [0]  # mutable so the RPC thread can increment
397	    exec_start = time.monotonic()
398	    server_sock = None
399	
400	    try:
401	        # Write the auto-generated hermes_tools module
402	        # sandbox_tools is already the correct set (intersection with session
403	        # tools, or SANDBOX_ALLOWED_TOOLS as fallback — see lines above).
404	        tools_src = generate_hermes_tools_module(list(sandbox_tools))
405	        with open(os.path.join(tmpdir, "hermes_tools.py"), "w") as f:
406	            f.write(tools_src)
407	
408	        # Write the user's script
409	        with open(os.path.join(tmpdir, "script.py"), "w") as f:
410	            f.write(code)
411	
412	        # --- Start UDS server ---
413	        server_sock = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
414	        server_sock.bind(sock_path)
415	        server_sock.listen(1)
416	
417	        rpc_thread = threading.Thread(
418	            target=_rpc_server_loop,
419	            args=(
420	                server_sock, task_id, tool_call_log,
421	                tool_call_counter, max_tool_calls, sandbox_tools,
422	            ),
423	            daemon=True,
424	        )
425	        rpc_thread.start()
426	
427	        # --- Spawn child process ---
428	        # Build a minimal environment for the child. We intentionally exclude
429	        # API keys and tokens to prevent credential exfiltration from LLM-
430	        # generated scripts. The child accesses tools via RPC, not direct API.
431	        _SAFE_ENV_PREFIXES = ("PATH", "HOME", "USER", "LANG", "LC_", "TERM",
432	                              "TMPDIR", "TMP", "TEMP", "SHELL", "LOGNAME",
433	                              "XDG_", "PYTHONPATH", "VIRTUAL_ENV", "CONDA")
434	        _SECRET_SUBSTRINGS = ("KEY", "TOKEN", "SECRET", "PASSWORD", "CREDENTIAL",
435	                              "PASSWD", "AUTH")
436	        child_env = {}
437	        for k, v in os.environ.items():
438	            if any(s in k.upper() for s in _SECRET_SUBSTRINGS):
439	                continue
440	            if any(k.startswith(p) for p in _SAFE_ENV_PREFIXES):
441	                child_env[k] = v
442	        child_env["HERMES_RPC_SOCKET"] = sock_path
443	        child_env["PYTHONDONTWRITEBYTECODE"] = "1"
444	        # Ensure the hermes-agent root is importable in the sandbox so
445	        # modules like minisweagent_path are available to child scripts.
446	        _hermes_root = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
447	        _existing_pp = child_env.get("PYTHONPATH", "")
448	        child_env["PYTHONPATH"] = _hermes_root + (os.pathsep + _existing_pp if _existing_pp else "")
449	        # Inject user's configured timezone so datetime.now() in sandboxed
450	        # code reflects the correct wall-clock time.
451	        _tz_name = os.getenv("HERMES_TIMEZONE", "").strip()
452	        if _tz_name:
453	            child_env["TZ"] = _tz_name
454	
455	        proc = subprocess.Popen(
456	            [sys.executable, "script.py"],
457	            cwd=tmpdir,
458	            env=child_env,
459	            stdout=subprocess.PIPE,
460	            stderr=subprocess.PIPE,
461	            stdin=subprocess.DEVNULL,
462	            preexec_fn=None if _IS_WINDOWS else os.setsid,
463	        )
464	
465	        # --- Poll loop: watch for exit, timeout, and interrupt ---
466	        deadline = time.monotonic() + timeout
467	        stderr_chunks: list = []
468	
469	        # Background readers to avoid pipe buffer deadlocks.
470	        # For stdout we use a head+tail strategy: keep the first HEAD_BYTES
471	        # and a rolling window of the last TAIL_BYTES so the final print()
472	        # output is never lost.  Stderr keeps head-only (errors appear early).
473	        _STDOUT_HEAD_BYTES = int(MAX_STDOUT_BYTES * 0.4)   # 40% head
474	        _STDOUT_TAIL_BYTES = MAX_STDOUT_BYTES - _STDOUT_HEAD_BYTES  # 60% tail
475	
476	        def _drain(pipe, chunks, max_bytes):
477	            """Simple head-only drain (used for stderr)."""
478	            total = 0
479	            try:
480	                while True:
481	                    data = pipe.read(4096)
482	                    if not data:
483	                        break
484	                    if total < max_bytes:
485	                        keep = max_bytes - total
486	                        chunks.append(data[:keep])
487	                    total += len(data)
488	            except (ValueError, OSError) as e:
489	                logger.debug("Error reading process output: %s", e, exc_info=True)
490	
491	        stdout_total_bytes = [0]  # mutable ref for total bytes seen
492	
493	        def _drain_head_tail(pipe, head_chunks, tail_chunks, head_bytes, tail_bytes, total_ref):
494	            """Drain stdout keeping both head and tail data."""
495	            head_collected = 0
496	            from collections import deque
497	            tail_buf = deque()
498	            tail_collected = 0
499	            try:
500	                while True:
501	                    data = pipe.read(4096)
502	                    if not data:
503	                        break
504	                    total_ref[0] += len(data)
505	                    # Fill head buffer first
506	                    if head_collected < head_bytes:
507	                        keep = min(len(data), head_bytes - head_collected)
508	                        head_chunks.append(data[:keep])
509	                        head_collected += keep
510	                        data = data[keep:]  # remaining goes to tail
511	                        if not data:
512	                            continue
513	                    # Everything past head goes into rolling tail buffer
514	                    tail_buf.append(data)
515	                    tail_collected += len(data)
516	                    # Evict old tail data to stay within tail_bytes budget
517	                    while tail_collected > tail_bytes and tail_buf:
518	                        oldest = tail_buf.popleft()
519	                        tail_collected -= len(oldest)
520	            except (ValueError, OSError):
521	                pass
522	            # Transfer final tail to output list
523	            tail_chunks.extend(tail_buf)
524	
525	        stdout_head_chunks: list = []
526	        stdout_tail_chunks: list = []
527	
528	        stdout_reader = threading.Thread(
529	            target=_drain_head_tail,
530	            args=(proc.stdout, stdout_head_chunks, stdout_tail_chunks,
531	                  _STDOUT_HEAD_BYTES, _STDOUT_TAIL_BYTES, stdout_total_bytes),
532	            daemon=True
533	        )
534	        stderr_reader = threading.Thread(
535	            target=_drain, args=(proc.stderr, stderr_chunks, MAX_STDERR_BYTES), daemon=True
536	        )
537	        stdout_reader.start()
538	        stderr_reader.start()
539	
540	        status = "success"
541	        while proc.poll() is None:
542	            if _interrupt_event.is_set():
543	                _kill_process_group(proc)
544	                status = "interrupted"
545	                break
546	            if time.monotonic() > deadline:
547	                _kill_process_group(proc, escalate=True)
548	                status = "timeout"
549	                break
550	            time.sleep(0.2)
551	
552	        # Wait for readers to finish draining
553	        stdout_reader.join(timeout=3)
554	        stderr_reader.join(timeout=3)
555	
556	        stdout_head = b"".join(stdout_head_chunks).decode("utf-8", errors="replace")
557	        stdout_tail = b"".join(stdout_tail_chunks).decode("utf-8", errors="replace")
558	        stderr_text = b"".join(stderr_chunks).decode("utf-8", errors="replace")
559	
560	        # Assemble stdout with head+tail truncation
561	        total_stdout = stdout_total_bytes[0]
562	        if total_stdout > MAX_STDOUT_BYTES and stdout_tail:
563	            omitted = total_stdout - len(stdout_head) - len(stdout_tail)
564	            truncated_notice = (
565	                f"\n\n... [OUTPUT TRUNCATED - {omitted:,} chars omitted "
566	                f"out of {total_stdout:,} total] ...\n\n"
567	            )
568	            stdout_text = stdout_head + truncated_notice + stdout_tail
569	        else:
570	            stdout_text = stdout_head + stdout_tail
571	
572	        exit_code = proc.returncode if proc.returncode is not None else -1
573	        duration = round(time.monotonic() - exec_start, 2)
574	
575	        # Wait for RPC thread to finish
576	        server_sock.close()  # break accept() so thread exits promptly
577	        server_sock = None  # prevent double close in finally
578	        rpc_thread.join(timeout=3)
579	
580	        # Strip ANSI escape sequences so the model never sees terminal
581	        # formatting — prevents it from copying escapes into file writes.
582	        from tools.ansi_strip import strip_ansi
583	        stdout_text = strip_ansi(stdout_text)
584	        stderr_text = strip_ansi(stderr_text)
585	
586	        # Build response
587	        result: Dict[str, Any] = {
588	            "status": status,
589	            "output": stdout_text,
590	            "tool_calls_made": tool_call_counter[0],
591	            "duration_seconds": duration,
592	        }
593	
594	        if status == "timeout":
595	            result["error"] = f"Script timed out after {timeout}s and was killed."
596	        elif status == "interrupted":
597	            result["output"] = stdout_text + "\n[execution interrupted — user sent a new message]"
598	        elif exit_code != 0:
599	            result["status"] = "error"
600	            result["error"] = stderr_text or f"Script exited with code {exit_code}"
601	            # Include stderr in output so the LLM sees the traceback
602	            if stderr_text:
603	                result["output"] = stdout_text + "\n--- stderr ---\n" + stderr_text
604	
605	        return json.dumps(result, ensure_ascii=False)
606	
607	    except Exception as exc:
608	        duration = round(time.monotonic() - exec_start, 2)
609	        logger.error(
610	            "execute_code failed after %ss with %d tool calls: %s: %s",
611	            duration,
612	            tool_call_counter[0],
613	            type(exc).__name__,
614	            exc,
615	            exc_info=True,
616	        )
617	        return json.dumps({
618	            "status": "error",
619	            "error": str(exc),
620	            "tool_calls_made": tool_call_counter[0],
621	            "duration_seconds": duration,
622	        }, ensure_ascii=False)
623	
624	    finally:
625	        # Cleanup temp dir and socket
626	        if server_sock is not None:
627	            try:
628	                server_sock.close()
629	            except OSError as e:
630	                logger.debug("Server socket close error: %s", e)
631	        import shutil
632	        shutil.rmtree(tmpdir, ignore_errors=True)
633	        try:
634	            os.unlink(sock_path)
635	        except OSError:
636	            pass  # already cleaned up or never created

639	def _kill_process_group(proc, escalate: bool = False):
640	    """Kill the child and its entire process group."""
641	    try:
642	        if _IS_WINDOWS:
643	            proc.terminate()
644	        else:
645	            os.killpg(os.getpgid(proc.pid), signal.SIGTERM)
646	    except (ProcessLookupError, PermissionError) as e:
647	        logger.debug("Could not kill process group: %s", e, exc_info=True)
648	        try:
649	            proc.kill()
650	        except Exception as e2:
651	            logger.debug("Could not kill process: %s", e2, exc_info=True)
652	
653	    if escalate:
654	        # Give the process 5s to exit after SIGTERM, then SIGKILL
655	        try:
656	            proc.wait(timeout=5)
657	        except subprocess.TimeoutExpired:
658	            try:
659	                if _IS_WINDOWS:
660	                    proc.kill()
661	                else:
