21	def compute_flag_history(

25	    flags_by_id: dict[str, list[dict[str, Any]]] = {}
26	    for iteration in range(1, current_iteration + 1):
27	        critique_path = plan_dir / f"critique_v{iteration}.json"
28	        if not critique_path.exists():
29	            continue
30	        critique = read_json(critique_path)
31	        seen_ids: set[str] = set()
32	        for check in critique.get("checks", []):
33	            for finding in check.get("findings", []):
34	                if finding.get("flagged"):
35	                    fid = check.get("id", "")
36	                    if fid:
37	                        seen_ids.add(fid)
38	                        flags_by_id.setdefault(fid, []).append(
39	                            {"iteration": iteration, "status": "open", "concern": finding.get("detail", "")}
40	                        )
41	        for flag in critique.get("flags", []):
42	            fid = flag.get("id", "")
43	            if fid:
44	                seen_ids.add(fid)
45	                flags_by_id.setdefault(fid, []).append(
46	                    {"iteration": iteration, "status": flag.get("status", "open"), "concern": flag.get("concern", "")}
47	                )
48	
49	    faults_path = plan_dir / "faults.json"
50	    if faults_path.exists():
51	        faults = read_json(faults_path)
52	        for flag in faults.get("flags", []):
53	            fid = flag.get("id", "")
54	            if fid and fid not in flags_by_id:
55	                flags_by_id.setdefault(fid, []).append(
56	                    {"iteration": 0, "status": flag.get("status", "open"), "concern": flag.get("concern", "")}
57	                )
58	
59	    result = []
60	    for fid, entries in flags_by_id.items():
61	        concern = entries[-1].get("concern", "") if entries else ""
62	        iterations_seen = sorted({e["iteration"] for e in entries})
63	        statuses = [e["status"] for e in entries]
64	        reopen_count = 0
65	        was_addressed = False
66	        for s in statuses:
67	            if s in ("addressed", "verified"):
68	                was_addressed = True
69	            elif s == "open" and was_addressed:
70	                reopen_count += 1
71	                was_addressed = False
72	        result.append({
73	            "id": fid,
74	            "concern": concern,
75	            "iterations": iterations_seen,
76	            "iterations_open": len(iterations_seen),
77	            "addressed_then_reopened_count": reopen_count,
78	        })
79	    return result

82	def compute_fuzzy_groups(
83	    flags: list[dict[str, Any]], threshold: float = 0.6
84	) -> dict[str, list[str]]:
85	    """Group flags by Jaccard word similarity on concern text."""
86	    word_sets = {f["id"]: _concern_word_set(f.get("concern", "")) for f in flags}
87	    assigned: dict[str, str] = {}
88	    groups: dict[str, list[str]] = {}
89	    group_counter = 0
90	
91	    for flag in flags:
92	        fid = flag["id"]
93	        if fid in assigned:
94	            continue
95	        group_counter += 1
96	        gid = f"FG-{group_counter:03d}"
97	        groups[gid] = [fid]
98	        assigned[fid] = gid
99	
100	        for other in flags:
101	            oid = other["id"]
102	            if oid in assigned:
103	                continue
104	            if _jaccard_similarity(word_sets[fid], word_sets[oid]) >= threshold:
105	                groups[gid].append(oid)
106	                assigned[oid] = gid
107	
108	    return groups

111	def compute_iteration_pressure(
112	    plan_dir: Path, state: PlanState
113	) -> list[IterationPressureEntry]:
114	    current_iteration = state.get("iteration", 1)
115	    flag_history = compute_flag_history(plan_dir, current_iteration)
116	    if not flag_history:
117	        return []
118	
119	    groups = compute_fuzzy_groups(flag_history, threshold=0.6)
120	    history_by_id = {f["id"]: f for f in flag_history}
121	
122	    entries: list[IterationPressureEntry] = []
123	    for gid, member_ids in groups.items():
124	        members = [history_by_id[fid] for fid in member_ids if fid in history_by_id]
125	        if not members:
126	            continue
127	        all_iterations: set[int] = set()
128	        max_reopen = 0
129	        for m in members:
130	            all_iterations.update(m.get("iterations", []))
131	            max_reopen = max(max_reopen, m.get("addressed_then_reopened_count", 0))
132	
133	        entries.append(IterationPressureEntry(
134	            fuzzy_group_id=gid,
135	            member_flag_ids=member_ids,
136	            iterations_open=len(all_iterations),
137	            addressed_then_reopened_count=max_reopen,
138	            representative_concern=members[0].get("concern", ""),
139	        ))
140	    return entries

144	    for entry in entries:

146	            return True
147	        if entry["iterations_open"] >= 2 and len(entry["member_flag_ids"]) >= 2:
148	            return True
