   205→def cache_document_from_bytes(data: bytes, filename: str) -> str:
   206→    """
   207→    Save raw document bytes to the cache and return the absolute file path.
   208→
   209→    The cached filename preserves the original human-readable name with a
   210→    unique prefix: ``doc_{uuid12}_{original_filename}``.
   211→
   212→    Args:
   213→        data: Raw document bytes.
   214→        filename: Original filename (e.g. "report.pdf").
   215→
   216→    Returns:
   217→        Absolute path to the cached document file as a string.
   218→
   219→    Raises:
   220→        ValueError: If the sanitized path escapes the cache directory.
   221→    """
   222→    cache_dir = get_document_cache_dir()
   223→    # Sanitize: strip directory components, null bytes, and control characters
   224→    safe_name = Path(filename).name if filename else "document"
   225→    safe_name = safe_name.replace("\x00", "").strip()
   226→    if not safe_name or safe_name in (".", ".."):
   227→        safe_name = "document"
   228→    cached_name = f"doc_{uuid.uuid4().hex[:12]}_{safe_name}"
   229→    filepath = cache_dir / cached_name
   230→    # Final safety check: ensure path stays inside cache dir
   231→    if not filepath.resolve().is_relative_to(cache_dir.resolve()):
   232→        raise ValueError(f"Path traversal rejected: {filename!r}")
   233→    filepath.write_bytes(data)
   234→    return str(filepath)

   471→    @staticmethod
   472→    def extract_images(content: str) -> Tuple[List[Tuple[str, str]], str]:
   473→        """
   474→        Extract image URLs from markdown and HTML image tags in a response.
   475→        
   476→        Finds patterns like:
   477→        - ![alt text](https://example.com/image.png)
   478→        - <img src="https://example.com/image.png">
   479→        - <img src="https://example.com/image.png"></img>
   480→        
   481→        Args:
   482→            content: The response text to scan.
   483→        
   484→        Returns:
   485→            Tuple of (list of (url, alt_text) pairs, cleaned content with image tags removed).
   486→        """
   487→        images = []
   488→        cleaned = content
   489→        
   490→        # Match markdown images: ![alt](url)
   491→        md_pattern = r'!\[([^\]]*)\]\((https?://[^\s\)]+)\)'
   492→        for match in re.finditer(md_pattern, content):
   493→            alt_text = match.group(1)
   494→            url = match.group(2)
   495→            # Only extract URLs that look like actual images
   496→            if any(url.lower().endswith(ext) or ext in url.lower() for ext in
   497→                   ['.png', '.jpg', '.jpeg', '.gif', '.webp', 'fal.media', 'fal-cdn', 'replicate.delivery']):
   498→                images.append((url, alt_text))
   499→        
   500→        # Match HTML img tags: <img src="url"> or <img src="url"></img> or <img src="url"/>
   501→        html_pattern = r'<img\s+src=["\']?(https?://[^\s"\'<>]+)["\']?\s*/?>\s*(?:</img>)?'
   502→        for match in re.finditer(html_pattern, content):
   503→            url = match.group(1)
   504→            images.append((url, ""))
   505→        
   506→        # Remove only the matched image tags from content (not all markdown images)
   507→        if images:
   508→            extracted_urls = {url for url, _ in images}
   509→            def _remove_if_extracted(match):
   510→                url = match.group(2) if match.lastindex >= 2 else match.group(1)
   511→                return '' if url in extracted_urls else match.group(0)
   512→            cleaned = re.sub(md_pattern, _remove_if_extracted, cleaned)
   513→            cleaned = re.sub(html_pattern, _remove_if_extracted, cleaned)
   514→            # Clean up leftover blank lines
   515→            cleaned = re.sub(r'\n{3,}', '\n\n', cleaned).strip()
   516→        
   517→        return images, cleaned

## KEY ARCHITECTURAL INSIGHTS

| Component | Thread Safety | Queueing | Timing |
|-----------|---------------|----------|--------|
| `_control_queue` | Lock-protected (`_control_lock`) | Yes (deque) | Drained at loop start/end |
| `_interrupt_requested` | Atomic flag | No | Checked throughout loop |
| `_active_sessions` (adapter) | Async-safe | Event-based | Polled every 200ms |
| `_pending_messages` (adapter) | Not locked (single-threaded async) | Manual dict | Retrieved after interrupt |
| `_interrupt_queue` (CLI) | Queue-safe | Yes (queue.Queue) | Polled every 100ms in chat() |

**Critical Detail**: The distinction between *command queueing* (`_control_queue` for deferred operations) and *interrupt signaling* (`_interrupt_requested` for immediate loop exit) allows both fast interrupts AND safe complex operations like context compaction.
