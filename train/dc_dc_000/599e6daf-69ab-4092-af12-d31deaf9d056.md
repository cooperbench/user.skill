   122→      if (firstVideo.canny_intensity != null) guidance.canny_intensity = firstVideo.canny_intensity;
   123→      if (firstVideo.depth_contrast != null) guidance.depth_contrast = firstVideo.depth_contrast;
   124→    }
   125→
   126→    return guidance;
   127→  }
   128→
   129→  if (structureVideoConfig.structure_video_path) {
   130→    const isUni3c = structureVideoConfig.structure_video_type === 'uni3c';
   131→    const legacyUni3cEndPercent = structureVideoConfig.uni3c_end_percent ?? uni3cEndPercent ?? 1.0;
   132→
   133→    const guidance: Record<string, unknown> = {
   134→      target: isUni3c ? 'uni3c' : 'vace',
   135→      videos: [{
   136→        path: structureVideoConfig.structure_video_path,
   137→        start_frame: 0,
   138→        end_frame: totalFrames,
   139→        treatment: structureVideoConfig.structure_video_treatment ?? DEFAULT_VIDEO_STRUCTURE_PARAMS.structure_video_treatment,
   140→      }],
   141→      strength: structureVideoConfig.structure_video_motion_strength ?? 1.0,
   142→    };
   143→
   144→    if (isUni3c) {
   145→      guidance.step_window = [0, legacyUni3cEndPercent];
   146→      guidance.frame_policy = 'fit';
   147→      guidance.zero_empty_frames = true;
   148→    } else {
   149→      guidance.preprocessing = PREPROCESSING_MAP[structureVideoConfig.structure_video_type ?? 'flow'] ?? 'flow';
   150→    }
   151→
   152→    return guidance;
   153→  }
   154→
   155→  return null;
   156→}

   188→/** Shared Supabase query shape for shot_generations with joined generation data */

   190→  id,

   192→  timeline_frame,
   193→  metadata,
   194→  generation:generations!shot_generations_generation_id_generations_id_fk (
   195→    id,
   196→    location,
   197→    type
   198→  )
   199→`;

   201→type ShotGenRow = {

   203→  generation_id: string | null;
   204→  timeline_frame: number | null;

   209→/** Filter shot_generations to positioned image rows with valid locations */
   210→function filterImageShotGenerations(rows: ShotGenRow[]): ShotGenRow[] {
   211→  return rows.filter(shotGen => {
   212→    const hasValidLocation = shotGen.generation?.location && shotGen.generation.location !== '/placeholder.svg';
   213→    return shotGen.generation &&
   214→           shotGen.timeline_frame != null &&
   215→           !isVideoShotGenerations(shotGen as ShotGenerationsLike) &&
   216→           hasValidLocation;
   217→  });
   218→}

   220→/**
   221→ * Prepare and submit a video generation task.
   222→ * Handles resolution, fresh DB queries, per-pair overrides, model selection,
   223→ * structure guidance, and task creation.
   224→ */
   225→export async function generateVideo(params: GenerateVideoParams): Promise<GenerateVideoResult> {
   226→  const {
   227→    projectId,
   228→    selectedShotId,
   229→    selectedShot,
   230→    queryClient,
   231→    effectiveAspectRatio,
   232→    generationMode,
   233→    promptConfig,
   234→    motionConfig,
   235→    modelConfig,
   236→    structureVideoConfig,
   237→    batchVideoFrames,
   238→    selectedLoras,
   239→    variantNameParam,
   240→    clearAllEnhancedPrompts,
   241→    parentGenerationId,
   242→    stitchConfig,
   243→  } = params;
   244→
   245→  const {
   246→    base_prompt: batchVideoPrompt,
   247→    enhance_prompt: enhancePrompt,
   248→    text_before_prompts: textBeforePrompts,
   249→    text_after_prompts: textAfterPrompts,
   250→    default_negative_prompt: defaultNegativePrompt,
   251→  } = promptConfig;
   252→
   253→  const {
   254→    amount_of_motion: rawAmountOfMotion,
   255→    motion_mode: motionMode,
   256→    advanced_mode: advancedMode,
   257→    phase_config: phaseConfig,
   258→    selected_phase_preset_id: selectedPhasePresetId,
   259→  } = motionConfig;
   260→
   261→  const { seed, random_seed: randomSeed, turbo_mode: turboMode, debug } = modelConfig;
   262→
   263→  // Ensure amountOfMotion has a valid default (destructuring default only applies when absent, not undefined)
   264→  const amountOfMotion = rawAmountOfMotion ?? 50;
   265→
   266→  if (!projectId) {
   267→    toast.error('No project selected. Please select a project first.');
   268→    return { success: false, error: 'No project selected' };
   269→  }
   270→
   271→  // Wait for pending mutations to complete before submitting (prevents stale data race conditions)
   272→  const mutationCache = queryClient.getMutationCache();
   273→  const pendingMutations = mutationCache.getAll().filter(m => m.state.status === 'pending');
   274→  if (pendingMutations.length > 0) {
   275→    await Promise.race([
   276→      Promise.all(pendingMutations.map(m => m.state.promise?.catch(() => {}))),
   277→      new Promise(resolve => setTimeout(resolve, 1000)),
   278→    ]);
   279→  }
   280→
   281→  // ── Resolution ──────────────────────────────────────────────────────────
   282→  // Priority: shot aspect ratio > project aspect ratio > default
   283→  let resolution: string | undefined;
   284→  if (selectedShot?.aspect_ratio) {
   285→    resolution = ASPECT_RATIO_TO_RESOLUTION[selectedShot.aspect_ratio];
   286→  }
   287→  if (!resolution && effectiveAspectRatio) {
   288→    resolution = ASPECT_RATIO_TO_RESOLUTION[effectiveAspectRatio];
   289→  }
   290→  if (!resolution) {
   291→    resolution = DEFAULT_RESOLUTION;
   292→  }
   293→
   294→  // ── Fetch fresh shot generation data from DB ────────────────────────────
   295→  // Single query used for both image URL extraction and timeline override extraction.
   296→  // Query directly to avoid using stale React Query cache.
   297→  let allShotGenerations: ShotGenRow[];
   298→
   299→  try {
   300→    const { data, error } = await supabase
   301→      .from('shot_generations')
   302→      .select(SHOT_GEN_QUERY)
   303→      .eq('shot_id', selectedShotId)
   304→      .order('timeline_frame', { ascending: true });
   305→
   306→    if (error) {
   307→      handleError(error, { context: 'TaskSubmission', toastTitle: 'Failed to fetch current images' });
   308→      return { success: false, error: 'Failed to fetch shot data' };
   309→    }
   310→
   311→    allShotGenerations = (data || []) as ShotGenRow[];
   312→  } catch (err) {
   313→    handleError(err, { context: 'TaskSubmission', toastTitle: 'Failed to prepare task data' });
   314→    return { success: false, error: 'Failed to prepare task data' };
   315→  }
   316→
   317→  // ── Extract image URLs and IDs from fetched data ────────────────────────
   318→  const filteredForUrls = filterImageShotGenerations(allShotGenerations)
   319→    .sort((a, b) => (a.timeline_frame ?? 0) - (b.timeline_frame ?? 0));
   320→
   321→  const freshImagesWithIds = filteredForUrls.map(shotGen => ({
   322→    location: shotGen.generation?.location,
   323→    generationId: shotGen.generation?.id || shotGen.generation_id,
   324→    shotGenerationId: shotGen.id,
   325→  })).filter(item => Boolean(item.location));
   326→
   327→  const absoluteImageUrls = freshImagesWithIds
   328→    .map(item => getDisplayUrl(item.location))
   329→    .filter((url): url is string => Boolean(url) && url !== '/placeholder.svg');
   330→
   331→  const filteredImages = freshImagesWithIds.filter(item => {
   332→    const url = getDisplayUrl(item.location);
   333→    return Boolean(url) && url !== '/placeholder.svg';
   334→  });
   335→
   336→  const imageGenerationIds = filteredImages
   337→    .map(item => item.generationId)
   338→    .filter((id): id is string => Boolean(id));
   339→
   340→  // Each pair_shot_generation_id is the shot_generations.id of the START image of that pair
   341→  const pairShotGenerationIds = filteredImages
   342→    .slice(0, -1)
   343→    .map(item => item.shotGenerationId)
   344→    .filter((id): id is string => Boolean(id));
   345→
   346→  // ── Per-pair overrides (timeline vs batch) ──────────────────────────────
   347→  let basePrompts: string[];
   348→  let segmentFrames: number[];
   349→  let frameOverlap: number[];
   350→  let negativePrompts: string[];
   351→  let enhancedPromptsArray: string[] = [];
   352→  let pairPhaseConfigsArray: (PhaseConfig | null)[] = [];
   353→  let pairLorasArray: (Array<{ path: string; strength: number }> | null)[] = [];
   354→  let pairMotionSettingsArray: (Record<string, unknown> | null)[] = [];
   355→
   356→  if (generationMode === 'timeline') {
   357→    // ── TIMELINE MODE: Extract per-pair overrides from fetched data ──────
   358→    const pairPrompts: Record<number, { prompt: string; negativePrompt: string }> = {};
   359→    const enhancedPrompts: Record<number, string> = {};
   360→    const pairPhaseConfigsOverrides: Record<number, PhaseConfig> = {};
   361→    const pairLorasOverrides: Record<number, Array<{ path: string; strength: number }>> = {};
   362→    const pairMotionSettingsOverrides: Record<number, Record<string, unknown>> = {};
   363→    const pairNumFramesOverrides: Record<number, number> = {};
   364→
   365→    // Reuse the same data already fetched — filter to match absoluteImageUrls exactly
   366→    const filteredShotGenerations = filterImageShotGenerations(allShotGenerations);
   367→
   368→    // Build sorted positions for frame gap calculation
   369→    const sortedPositions = filteredShotGenerations
   370→      .filter(shotGen => shotGen.timeline_frame != null && shotGen.timeline_frame >= 0)
   371→      .map(shotGen => ({
   372→        id: getGenerationId(shotGen),
   373→        pos: shotGen.timeline_frame!,
   374→      }))
   375→      .sort((a, b) => a.pos - b.pos);
   376→
   377→    // Extract per-pair overrides from filtered data (indexes match actual image pairs)
   378→    for (let i = 0; i < filteredShotGenerations.length - 1; i++) {
   379→      const firstItem = filteredShotGenerations[i];
   380→      const metadata = firstItem.metadata as Record<string, unknown> | null;
   381→      const { overrides, enhancedPrompt, legacyOverrides } = extractPairOverrides(metadata);
   382→
   383→      // Prompts
   384→      if (overrides.prompt || overrides.negativePrompt) {
   385→        pairPrompts[i] = {
   386→          prompt: overrides.prompt || '',
   387→          negativePrompt: overrides.negativePrompt || '',
   388→        };
   389→      }
   390→
   391→      if (enhancedPrompt) {
   392→        enhancedPrompts[i] = enhancedPrompt;
   393→      }
   394→
   395→      // Phase config (ignore stale phase_config when motion_mode is 'basic')
   396→      const pairMotionMode = overrides.motionMode || legacyOverrides?.motion_mode;
   397→      const isBasicMode = pairMotionMode === 'basic';
   398→
   399→      if (!isBasicMode) {
   400→        if (overrides.phaseConfig) {
   401→          pairPhaseConfigsOverrides[i] = overrides.phaseConfig;
   402→        } else if (legacyOverrides?.phase_config) {
   403→          pairPhaseConfigsOverrides[i] = legacyOverrides.phase_config;
   404→        }
   405→      }
   406→
   407→      // LoRAs
   408→      if (overrides.loras && overrides.loras.length > 0) {
   409→        pairLorasOverrides[i] = overrides.loras.map(lora => ({
   410→          path: lora.path,
   411→          strength: lora.strength,
   412→        }));
   413→      } else if (legacyOverrides?.additional_loras && Object.keys(legacyOverrides.additional_loras).length > 0) {
   414→        pairLorasOverrides[i] = Object.entries(legacyOverrides.additional_loras).map(([path, strength]) => ({
   415→          path,
   416→          strength: typeof strength === 'number' ? strength : 1.0,
   417→        }));
   418→      }
   419→
   420→      // Motion and structure settings
   421→      const hasMotionOverrides = overrides.motionMode !== undefined || overrides.amountOfMotion !== undefined;
   422→      const hasStructureOverrides = overrides.structureMotionStrength !== undefined ||
   423→                                    overrides.structureTreatment !== undefined ||
   424→                                    overrides.structureUni3cEndPercent !== undefined;
   425→
   426→      if (hasMotionOverrides || hasStructureOverrides) {
   427→        pairMotionSettingsOverrides[i] = {
   428→          ...(overrides.amountOfMotion !== undefined && { amount_of_motion: overrides.amountOfMotion / 100 }),
   429→          ...(overrides.motionMode !== undefined && { motion_mode: overrides.motionMode }),
   430→          ...(overrides.structureMotionStrength !== undefined && { structure_motion_strength: overrides.structureMotionStrength }),
   431→          ...(overrides.structureTreatment !== undefined && { structure_treatment: overrides.structureTreatment }),
   432→          ...(overrides.structureUni3cEndPercent !== undefined && { uni3c_end_percent: overrides.structureUni3cEndPercent }),
   433→        };
   434→      } else if (legacyOverrides?.amount_of_motion !== undefined || legacyOverrides?.motion_mode !== undefined) {
   435→        pairMotionSettingsOverrides[i] = {
   436→          amount_of_motion: legacyOverrides.amount_of_motion,
   437→          motion_mode: legacyOverrides.motion_mode,
   438→        };
   439→      }
   440→
   441→      // Num frames
   442→      if (overrides.numFrames !== undefined) {
   443→        pairNumFramesOverrides[i] = overrides.numFrames;
   444→      }
   445→    }
   446→
   447→    // Calculate frame gaps from sorted positions
   448→    const frameGaps: number[] = [];
   449→    for (let i = 0; i < sortedPositions.length - 1; i++) {
   450→      frameGaps.push(sortedPositions[i + 1].pos - sortedPositions[i].pos);
   451→    }
   452→
   453→    // Build per-pair arrays from overrides
   454→    basePrompts = frameGaps.length > 0
   455→      ? frameGaps.map((_, index) => {
   456→          // Empty string = backend uses base_prompt (singular)
   457→          const pairPrompt = pairPrompts[index]?.prompt;
   458→          return (pairPrompt && pairPrompt.trim()) ? pairPrompt.trim() : '';
   459→        })
   460→      : [''];
   461→
   462→    segmentFrames = frameGaps.length > 0
   463→      ? frameGaps.map((gap, index) => pairNumFramesOverrides[index] ?? gap)
   464→      : [batchVideoFrames];
   465→
   466→    frameOverlap = frameGaps.length > 0 ? frameGaps.map(() => 10) : [10];
   467→
   468→    negativePrompts = frameGaps.length > 0
   469→      ? frameGaps.map((_, index) => {
   470→          const pairNeg = pairPrompts[index]?.negativePrompt;
   471→          return (pairNeg && pairNeg.trim()) ? pairNeg.trim() : defaultNegativePrompt;
   472→        })
   473→      : [defaultNegativePrompt];
   474→
   475→    enhancedPromptsArray = frameGaps.length > 0
   476→      ? frameGaps.map((_, index) => enhancedPrompts[index] || '')
   477→      : [];
   478→
   479→    pairPhaseConfigsArray = frameGaps.length > 0
   480→      ? frameGaps.map((_, index) => {
   481→          const config = pairPhaseConfigsOverrides[index];
   482→          return config ? stripModeFromPhaseConfig(config) : null;
   483→        })
   484→      : [];
   485→
   486→    pairLorasArray = frameGaps.length > 0
   487→      ? frameGaps.map((_, index) => pairLorasOverrides[index] || null)
   488→      : [];
   489→
   490→    pairMotionSettingsArray = frameGaps.length > 0
   491→      ? frameGaps.map((_, index) => pairMotionSettingsOverrides[index] || null)
   492→      : [];
   493→  } else {
   494→    // ── BATCH MODE: Uniform settings for all pairs ──────────────────────
   495→    const numPairs = Math.max(0, absoluteImageUrls.length - 1);
   496→
   497→    basePrompts = numPairs > 0 ? Array(numPairs).fill('') : [''];
   498→    negativePrompts = numPairs > 0 ? Array(numPairs).fill(defaultNegativePrompt) : [defaultNegativePrompt];
   499→    segmentFrames = numPairs > 0 ? Array(numPairs).fill(batchVideoFrames) : [batchVideoFrames];
   500→    frameOverlap = numPairs > 0 ? Array(numPairs).fill(10) : [10];
   501→    pairPhaseConfigsArray = numPairs > 0 ? Array(numPairs).fill(null) : [];
   502→    pairLorasArray = numPairs > 0 ? Array(numPairs).fill(null) : [];
   503→    pairMotionSettingsArray = numPairs > 0 ? Array(numPairs).fill(null) : [];
   504→    enhancedPromptsArray = numPairs > 0 ? Array(numPairs).fill(null) : [];
   505→  }
   506→
   507→  // ── Model and phase config determination ────────────────────────────────
   508→  let actualModelName: string;
   509→  let effectivePhaseConfig: PhaseConfig;
   510→
   511→  const userLorasForPhaseConfig = (selectedLoras || []).map(l => ({
   512→    path: l.path,
   513→    strength: parseFloat(l.strength?.toString() ?? '0') || 0.0,
   514→    lowNoisePath: (l as { lowNoisePath?: string }).lowNoisePath,
   515→    isMultiStage: (l as { isMultiStage?: boolean }).isMultiStage,
   516→  }));
   517→
   518→  // Use advanced mode only when explicitly enabled, phaseConfig exists, and motionMode isn't 'basic'
   519→  const useAdvancedMode = advancedMode && phaseConfig && motionMode !== 'basic';
   520→
   521→  if (useAdvancedMode) {
   522→    actualModelName = phaseConfig.num_phases === 2
   523→      ? 'wan_2_2_i2v_lightning_baseline_3_3'
   524→      : 'wan_2_2_i2v_lightning_baseline_2_2_2';
   525→    effectivePhaseConfig = phaseConfig;
   526→  } else {
   527→    const basicConfig = buildBasicModePhaseConfig(amountOfMotion, userLorasForPhaseConfig);
   528→    actualModelName = basicConfig.model;
   529→    effectivePhaseConfig = basicConfig.phaseConfig;
   530→  }
   531→
   532→  // Validate phase config consistency before sending
   533→  {
   534→    const phasesLength = effectivePhaseConfig.phases?.length || 0;
   535→    const stepsLength = effectivePhaseConfig.steps_per_phase?.length || 0;
   536→    const numPhases = effectivePhaseConfig.num_phases;
   537→
   538→    if (numPhases !== phasesLength || numPhases !== stepsLength) {
   539→      console.error('[Generation] Inconsistent phase config:', {
   540→        num_phases: numPhases, phases_array_length: phasesLength, steps_array_length: stepsLength,
   541→      });
   542→      toast.error(`Invalid phase configuration: num_phases (${numPhases}) doesn't match arrays (phases: ${phasesLength}, steps: ${stepsLength}). Please reset to defaults.`);
   543→      return { success: false, error: 'Invalid phase configuration' };
   544→    }
   545→  }
   546→
   547→  // Only include enhanced_prompts if there are actual non-empty values
   548→  const hasValidEnhancedPrompts = enhancedPromptsArray.some(p => p && p.trim().length > 0);
   549→
   550→  // ── Build request body ──────────────────────────────────────────────────
   551→  const requestBody: Record<string, unknown> = {
   552→    project_id: projectId,
   553→    shot_id: selectedShot.id,
   554→    image_urls: absoluteImageUrls,
   555→    ...(imageGenerationIds.length > 0 && imageGenerationIds.length === absoluteImageUrls.length
   556→      ? { image_generation_ids: imageGenerationIds }
   557→      : {}),
   558→    ...(pairShotGenerationIds.length > 0 ? { pair_shot_generation_ids: pairShotGenerationIds } : {}),
   559→    ...(parentGenerationId ? { parent_generation_id: parentGenerationId } : {}),
   560→    base_prompts: basePrompts,
   561→    base_prompt: batchVideoPrompt,
   562→    segment_frames: segmentFrames,
   563→    frame_overlap: frameOverlap,
   564→    negative_prompts: negativePrompts,
   565→    ...(hasValidEnhancedPrompts ? { enhanced_prompts: enhancedPromptsArray } : {}),
   566→    ...(pairPhaseConfigsArray.some(x => x !== null) ? { pair_phase_configs: pairPhaseConfigsArray } : {}),
   567→    ...(pairLorasArray.some(x => x !== null) ? { pair_loras: pairLorasArray } : {}),
   568→    ...(pairMotionSettingsArray.some(x => x !== null) ? { pair_motion_settings: pairMotionSettingsArray } : {}),
   569→    model_name: actualModelName,
   570→    model_type: 'i2v',
   571→    seed,
   572→    debug: debug ?? DEFAULT_STEERABLE_MOTION_SETTINGS.debug,
   573→    show_input_images: DEFAULT_STEERABLE_MOTION_SETTINGS.show_input_images,
   574→    enhance_prompt: enhancePrompt,
   575→    generation_mode: generationMode,
   576→    random_seed: randomSeed,
   577→    turbo_mode: turboMode,
   578→    amount_of_motion: amountOfMotion / 100.0,
   579→    advanced_mode: useAdvancedMode,
   580→    motion_mode: motionMode,
   581→    phase_config: stripModeFromPhaseConfig(effectivePhaseConfig),
   582→    regenerate_anchors: false,
   583→    selected_phase_preset_id: selectedPhasePresetId || undefined,
   584→    generation_name: variantNameParam.trim() || undefined,
   585→    ...(textBeforePrompts ? { text_before_prompts: textBeforePrompts } : {}),
   586→    ...(textAfterPrompts ? { text_after_prompts: textAfterPrompts } : {}),
   587→    independent_segments: true,
   588→    chain_segments: false,
   589→    use_svi: false,
   590→  };
   591→
   592→  // LoRAs are in phase_config for GPU worker processing.
   593→  // Also send as separate loras array for orchestrator_details (segment regeneration).
   594→  if (selectedLoras && selectedLoras.length > 0) {
   595→    requestBody.loras = selectedLoras.map(l => ({
   596→      path: l.path,
   597→      strength: parseFloat(l.strength?.toString() ?? '1') || 1.0,
   598→    }));
   599→  }
   600→
   601→  if (resolution) {
   602→    requestBody.resolution = resolution;
   603→  }
   604→
   605→  if (stitchConfig) {
   606→    requestBody.stitch_config = stitchConfig;
   607→  }
   608→
   609→  // Structure guidance (unified format with videos nested inside)
   610→  const totalFrames = segmentFrames.reduce((a, b) => a + b, 0);
   611→  const structureGuidance = buildStructureGuidance(
   612→    params.structureVideos,
   613→    structureVideoConfig,
   614→    params.uni3cEndPercent,
   615→    totalFrames,
   616→  );
   617→  if (structureGuidance) {
   618→    requestBody.structure_guidance = structureGuidance;
   619→  }
   620→
   621→  // ── Submit task ─────────────────────────────────────────────────────────
   622→  try {
   623→    // Clear stale enhanced prompts when enhance_prompt is disabled
   624→    if (!enhancePrompt) {
   625→      try {
   626→        await clearAllEnhancedPrompts();
   627→      } catch (clearError) {
   628→        handleError(clearError, { context: 'generateVideoService', showToast: false });
   629→      }
   630→    }
   631→
   632→    const result = await createTravelBetweenImagesTask(requestBody as TravelBetweenImagesTaskParams);
   633→    return { success: true, parentGenerationId: result.parentGenerationId };
