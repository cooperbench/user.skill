   485→        };

   496→        uploadVideoToStorage(file, selectedProjectId || '', clipId, {
   497→          maxRetries: 3,
   498→          timeoutMs: 300000,
   499→        }),
   500→        uploadBlobToStorage(posterBlob, 'poster.jpg', 'image/jpeg', {

   502→          timeoutMs: 30000,
   503→        }),
   504→        uploadBlobToStorage(finalFrameBlob, 'final-frame.jpg', 'image/jpeg', {
   505→          maxRetries: 2,
   506→          timeoutMs: 30000,
   507→        })
   508→      ]);
   509→
   510→      // Create a generation record so the video appears in the gallery
   511→      if (selectedProjectId) {
   512→        try {
   513→          await createGenerationMutation.mutateAsync({
   514→            imageUrl: videoUrl,
   515→            fileName: file.name,
   516→            fileType: 'video',
   517→            fileSize: file.size,
   518→            projectId: selectedProjectId,
   519→            prompt: 'Uploaded clip for Join',
   520→            thumbnailUrl: posterUrl,
   521→          });
   522→        } catch (genError) {
   523→          handleError(genError, { context: 'JoinClipsPage', showToast: false });
   524→        }
   525→      }
   526→
   527→      return { videoUrl, posterUrl, finalFrameUrl, durationSeconds };
   528→    } catch (error) {
   529→      handleError(error, { context: 'JoinClipsPage', toastTitle: 'Upload failed' });
   530→      return null;
   531→    } finally {
   532→      setUploadingClipId(null);
   533→    }
