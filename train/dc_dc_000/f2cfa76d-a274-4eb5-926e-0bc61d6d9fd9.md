   287→                 for f in fs]
   288→            for fp, fs in sorted(by_file.items(), key=lambda x: -len(x[1]))

   425→def cmd_fix(args):
   426→    """Auto-fix mechanical issues (unused imports)."""
   427→    fixer = args.fixer
   428→    dry_run = getattr(args, "dry_run", False)
   429→
   430→    if fixer == "unused-imports":
   431→        from .detectors.unused import detect_unused
   432→        from .fix import fix_unused_imports
   433→        from .state import load_state, save_state, resolve_findings
   434→
   435→        path = Path(args.path)
   436→        print(c("\nDetecting unused imports...", "dim"), file=sys.stderr)
   437→        entries = detect_unused(path, category="imports")
   438→        print(c(f"  Found {len(entries)} unused imports across {len(set(e['file'] for e in entries))} files\n", "dim"),
   439→              file=sys.stderr)
   440→
   441→        if not entries:
   442→            print(c("No unused imports found.", "green"))
   443→            return
   444→
   445→        results = fix_unused_imports(entries, dry_run=dry_run)
   446→
   447→        total_removed = sum(len(r["removed"]) for r in results)
   448→        total_lines = sum(r["lines_removed"] for r in results)
   449→        verb = "Would remove" if dry_run else "Removed"
   450→
   451→        print(c(f"\n  {verb} {total_removed} unused imports from {len(results)} files ({total_lines} lines)\n", "bold"))
   452→        for r in results[:30]:
   453→            syms = ", ".join(r["removed"][:5])
   454→            if len(r["removed"]) > 5:
   455→                syms += f" (+{len(r['removed']) - 5})"
   456→            print(f"  {rel(r['file'])}  →  {syms}")
   457→        if len(results) > 30:
   458→            print(f"  ... and {len(results) - 30} more files")
   459→
   460→        # Auto-resolve in state
   461→        if not dry_run:
   462→            sp = _state_path(args)
   463→            state = load_state(sp)
   464→            prev_score = state.get("score", 0)
   465→
   466→            resolved_ids = []
   467→            for r in results:
   468→                rfile = rel(r["file"])
   469→                for sym in r["removed"]:
   470→                    fid = f"unused::{rfile}::{sym}"
   471→                    if fid in state["findings"] and state["findings"][fid]["status"] == "open":
   472→                        state["findings"][fid]["status"] = "fixed"
   473→                        state["findings"][fid]["note"] = "auto-fixed by decruftify fix unused-imports"
   474→                        resolved_ids.append(fid)
   475→
   476→            save_state(state, sp)
   477→            delta = state["score"] - prev_score
   478→            delta_str = f" ({'+' if delta > 0 else ''}{delta})" if delta else ""
   479→            print(f"\n  Auto-resolved {len(resolved_ids)} findings in state")
   480→            print(f"  Score: {state['score']}/100{delta_str}")
   481→
   482→            _write_query({"command": "fix", "fixer": fixer,
   483→                          "files_fixed": len(results), "imports_removed": total_removed,
   484→                          "findings_resolved": len(resolved_ids),
   485→                          "score": state["score"], "prev_score": prev_score})
   486→        else:
   487→            _write_query({"command": "fix", "fixer": fixer, "dry_run": True,
   488→                          "files_would_fix": len(results), "imports_would_remove": total_removed})
   489→
   490→        print()
   491→    elif fixer == "debug-logs":
   492→        from .detectors.logs import detect_logs
   493→        from .fix import fix_debug_logs, fix_unused_imports
   494→        from .detectors.unused import detect_unused
   495→        from .state import load_state, save_state
   496→
   497→        path = Path(args.path)
   498→        print(c("\nDetecting tagged debug logs...", "dim"), file=sys.stderr)
   499→        entries = detect_logs(path)
   500→        print(c(f"  Found {len(entries)} tagged logs across {len(set(e['file'] for e in entries))} files\n", "dim"),
   501→              file=sys.stderr)
   502→
   503→        if not entries:
   504→            print(c("No tagged debug logs found.", "green"))
   505→            return
   506→
   507→        results = fix_debug_logs(entries, dry_run=dry_run)
   508→        total_logs = sum(r["log_count"] for r in results)
   509→        total_lines = sum(r["lines_removed"] for r in results)
   510→        verb = "Would remove" if dry_run else "Removed"
   511→
   512→        print(c(f"\n  {verb} {total_logs} debug logs from {len(results)} files ({total_lines} lines)\n", "bold"))
   513→        for r in results[:30]:
   514→            tags = ", ".join(f"[{t}]" for t in r["tags"][:4])
   515→            if len(r["tags"]) > 4:
   516→                tags += f" (+{len(r['tags']) - 4})"
   517→            print(f"  {rel(r['file'])}  ({r['log_count']} logs)  {tags}")
   518→        if len(results) > 30:
   519→            print(f"  ... and {len(results) - 30} more files")
   520→
   521→        # Auto-resolve log findings in state
   522→        log_resolved = 0
   523→        if not dry_run:
   524→            sp = _state_path(args)
   525→            state = load_state(sp)
   526→            prev_score = state.get("score", 0)
   527→
   528→            for r in results:
   529→                rfile = rel(r["file"])
   530→                for tag in r["tags"]:
   531→                    fid = f"logs::{rfile}::{tag}"
   532→                    if fid in state["findings"] and state["findings"][fid]["status"] == "open":
   533→                        state["findings"][fid]["status"] = "fixed"
   534→                        state["findings"][fid]["note"] = "auto-fixed by decruftify fix debug-logs"
   535→                        log_resolved += 1
   536→
   537→            save_state(state, sp)
   538→            delta = state["score"] - prev_score
   539→            delta_str = f" ({'+' if delta > 0 else ''}{delta})" if delta else ""
   540→            print(f"\n  Auto-resolved {log_resolved} log findings in state")
   541→            print(f"  Score: {state['score']}/100{delta_str}")
   542→
   543→            # Cascading cleanup: removing logs may leave orphaned imports
   544→            print(c("\n  Running cascading import cleanup...", "dim"), file=sys.stderr)
   545→            import_entries = detect_unused(path, category="imports")
   546→            if import_entries:
   547→                import_results = fix_unused_imports(import_entries, dry_run=dry_run)
   548→                import_removed = sum(len(r["removed"]) for r in import_results)
   549→                import_lines = sum(r["lines_removed"] for r in import_results)
   550→
   551→                if import_results:
   552→                    print(c(f"  Cascade: removed {import_removed} now-orphaned imports from {len(import_results)} files ({import_lines} lines)", "green"))
   553→
   554→                    # Resolve those too
   555→                    import_resolved = 0
   556→                    for r in import_results:
   557→                        rfile = rel(r["file"])
   558→                        for sym in r["removed"]:
   559→                            fid = f"unused::{rfile}::{sym}"
   560→                            if fid in state["findings"] and state["findings"][fid]["status"] == "open":
   561→                                state["findings"][fid]["status"] = "fixed"
   562→                                state["findings"][fid]["note"] = "cascade: orphaned after debug-log removal"
   563→                                import_resolved += 1
   564→
   565→                    save_state(state, sp)
   566→                    if import_resolved:
   567→                        print(f"  Cascade: auto-resolved {import_resolved} import findings")
   568→                        delta = state["score"] - prev_score
   569→                        delta_str = f" ({'+' if delta > 0 else ''}{delta})" if delta else ""
   570→                        print(f"  Score: {state['score']}/100{delta_str}")
   571→                else:
   572→                    print(c("  Cascade: no orphaned imports found", "dim"))
   573→            else:
   574→                print(c("  Cascade: no orphaned imports found", "dim"))
   575→
   576→            _write_query({"command": "fix", "fixer": fixer,
   577→                          "files_fixed": len(results), "logs_removed": total_logs,
   578→                          "log_findings_resolved": log_resolved,
   579→                          "score": state["score"], "prev_score": prev_score})
   580→        else:
   581→            _write_query({"command": "fix", "fixer": fixer, "dry_run": True,
   582→                          "files_would_fix": len(results), "logs_would_remove": total_logs})
   583→
   584→        print()
   585→    else:
   586→        print(c(f"Unknown fixer: {fixer}. Available: unused-imports, debug-logs", "red"))
   587→        sys.exit(1)

### 3. Why Findings Get Reopened

A finding is **reopened** when ALL of these conditions are met:

1. **Finding previously existed** in state (has a matching ID)
2. **Status was `fixed` or `auto_resolved`** (lines 184)
3. **Finding reappears** in a new scan (detected again by the same detector)

**Triggers for Reopening**:

- **Code regression**: You marked something as fixed, but the issue returned
  - Example: Unused import was removed, then someone added it back
  - Example: Debug log was deleted, then reintroduced

- **False auto-resolution**: Something was `auto_resolved` (disappeared from scan) but wasn't actually fixed
  - Example: Detector temporarily failed to detect the issue
  - Example: File was temporarily excluded, then included again

**What does NOT trigger reopening**:
- Findings marked `wontfix` or `false_positive` stay in that status even if redetected
- Brand new findings (never seen before) start as `open`, not "reopened"
- Findings that remain continuously `open` stay `open` (no status change)

### Key Invariants

1. **Stable IDs**: Findings use deterministic IDs (`detector::file::name`) so the same issue is tracked across scans
2. **Metadata refresh**: Even if status doesn't change, `tier`/`confidence`/`summary` are updated on every scan
3. **Manual overrides persist**: `wontfix` and `false_positive` are never auto-changed by scans
4. **Weighted scoring**: Score uses tier weights (T1=4x, T2=3x, T3=2x, T4=1x) to incentivize fixing high-priority items

### Example Scenario

```
Scan 1: Detects "unused::foo.ts::React" → status: open
User: marks as "fixed" after removing it
Scan 2: Finding disappears → status stays "fixed" (not auto_resolved)
Someone adds React import back
Scan 3: Detects "unused::foo.ts::React" again → status: open, note: "Reopened — reappeared in scan (was fixed)", reopened_count++
```

This reconciliation design ensures that:
- Automatic fixes are detected and tracked
- Regressions are highlighted
- Manual decisions (`wontfix`) are respected
- State evolves smoothly across multiple scans without losing history
