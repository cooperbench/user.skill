    12→    *,
    13→    status_filter: str = "open",
    14→) -> list[str]:
    15→    """Resolve one or more patterns to a deduplicated list of finding IDs."""
    16→    seen: set[str] = set()

   394→def _fixer_help_lines(langs: list[str]) -> list[str]:
   395→    fixer_help_lines: list[str] = []
   396→    for lang_name in langs:
   397→        try:
   398→            fixer_names = sorted(get_lang(lang_name).fixers.keys())
   399→        except (ImportError, ValueError, TypeError, AttributeError):
   400→            fixer_names = []
   401→        fixer_list = ", ".join(fixer_names) if fixer_names else "none yet"
   402→        fixer_help_lines.append(f"fixers ({lang_name}): {fixer_list}")
   403→    fixer_help_lines.append("special: review — prepare structured review data")
   404→    return fixer_help_lines
