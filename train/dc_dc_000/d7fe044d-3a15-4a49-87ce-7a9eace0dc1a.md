   115→def add_structural_signal(structural: dict, file: str, signal: str, detail: dict):
   116→    """Add a complexity signal to the per-file structural dict.
   117→
   118→    Accumulates signals per file so they can be merged into tiered findings.

   126→def merge_structural_signals(structural: dict, stderr_fn,
   127→                              *, complexity_only_min: int = 35) -> list[Finding]:
   128→    """Convert per-file structural signals into tiered findings.
   129→
   130→    3+ signals -> T4/high (needs decomposition).
   131→    1-2 signals -> T3/medium.
   132→    Complexity-only files (no large/god signals) need score >= complexity_only_min
   133→    to be flagged — lower complexity in small files is normal, not decomposition-worthy.
   134→    """
   135→    results = []
   136→    suppressed = 0
   137→    for filepath, data in structural.items():
   138→        if "loc" not in data["detail"]:
   139→            try:
   140→                p = Path(filepath) if Path(filepath).is_absolute() else PROJECT_ROOT / filepath
   141→                data["detail"]["loc"] = len(p.read_text().splitlines())
   142→            except (OSError, UnicodeDecodeError):
   143→                data["detail"]["loc"] = 0
   144→
   145→        # Suppress complexity-only findings below the elevated threshold
   146→        signals = data["signals"]
   147→        is_complexity_only = all(s.startswith("complexity") for s in signals)
   148→        if is_complexity_only:
   149→            score = data["detail"].get("complexity_score", 0)
   150→            if score < complexity_only_min:
   151→                suppressed += 1
   152→                continue
   153→
   154→        signal_count = len(signals)
   155→        tier = 4 if signal_count >= 3 else 3
   156→        confidence = "high" if signal_count >= 3 else "medium"
   157→        summary = "Needs decomposition: " + " / ".join(signals)
   158→        results.append(make_finding(
   159→            "structural", filepath, "",
   160→            tier=tier, confidence=confidence,
   161→            summary=summary,
   162→            detail=data["detail"],
   163→        ))
   164→    if suppressed:
   165→        stderr_fn(f"         {suppressed} complexity-only files below threshold (< {complexity_only_min})")
   166→    stderr_fn(f"         -> {len(results)} structural findings")
   167→    return results
