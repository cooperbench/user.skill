def _is_current(path: Path) -> bool:
    if not path.is_file():
        return False
    content = path.read_text(encoding="utf-8", errors="replace")
    match = SKILL_VERSION_RE.search(content)
    return match is not None and int(match.group(1)) >= SKILL_VERSION

```python
def find_stale_global_installs() -> list[SkillInstall]:
    """Return all globally-installed skill documents that are stale."""
    home = Path.home()
    stale: list[SkillInstall] = []
    for rel_path, _interface, _dedicated in GLOBAL_SEARCH_PATHS:
        full = home / rel_path
        if not full.is_file():
            continue
        try:
            content = full.read_text(encoding="utf-8", errors="replace")
        except OSError:
            continue
        version_match = SKILL_VERSION_RE.search(content)
        if not version_match:
            continue
        installed_version = int(version_match.group(1))
        if installed_version < SKILL_VERSION:
            overlay_match = SKILL_OVERLAY_RE.search(content)
            overlay = overlay_match.group(1) if overlay_match else None
            stale.append(SkillInstall(
                rel_path=f"~/{rel_path}",
                version=installed_version,
                overlay=overlay,
                stale=True,
            ))
    return stale
```

```python
def _scan_global_installs() -> list[SkillInstall]:
    """Find all globally-installed skill documents."""
    home = Path.home()
    results: list[SkillInstall] = []
    for rel_path, _interface, _dedicated in GLOBAL_SEARCH_PATHS:
        full = home / rel_path
        if not full.is_file():
            continue
        try:
            content = full.read_text(encoding="utf-8", errors="replace")
        except OSError:
            continue
        version_match = SKILL_VERSION_RE.search(content)
        if not version_match:
            continue
        installed_version = int(version_match.group(1))
        overlay_match = SKILL_OVERLAY_RE.search(content)
        overlay = overlay_match.group(1) if overlay_match else None
        results.append(SkillInstall(
            rel_path=f"~/{rel_path}",
            version=installed_version,
            overlay=overlay,
            stale=installed_version < SKILL_VERSION,
        ))
    return results
```

def find_stale_global_installs() -> list[SkillInstall]:

def find_any_global_install() -> bool:
    """Return True if any global skill install exists (stale or current)."""
