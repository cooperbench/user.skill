   416→def build_queue_snapshot(
   417→    state: StateModel,
   418→    *,
   419→    options: object | None = None,
   420→    plan: dict | None = None,
   421→    target_strict: float = DEFAULT_TARGET_STRICT_SCORE,
   422→) -> QueueSnapshot:
   423→    """Build the canonical queue snapshot for the current state."""
   424→    context = _option_value(options, "context", None)
   425→    effective_plan = context.plan if context is not None else (
   426→        plan if plan is not None else _option_value(options, "plan", None)
   427→    )
   428→    scan_path = _resolved_scan_path(options, state)
   429→    skipped_ids = set((effective_plan or {}).get("skipped", {}).keys())
   430→    scoped_issues = path_scoped_issues((state.get("work_items") or state.get("issues", {})), scan_path)
   431→    scope = _option_value(options, "scope", None)
   432→    chronic = bool(_option_value(options, "chronic", False))
   433→
   434→    all_issue_items = build_issue_items(
   435→        state,
   436→        scan_path=scan_path,
   437→        status_filter="open",
   438→        scope=scope,
   439→        chronic=chronic,
   440→        forced_ids=_live_planned_queue_ids(effective_plan),
   441→    )
   442→    objective_items = [
   443→        item for item in all_issue_items
   444→        if _is_objective_item(item, skipped_ids=skipped_ids)
   445→    ]
   446→    executable_objective_ids = _executable_objective_ids(
   447→        {item.get("id", "") for item in objective_items},
   448→        effective_plan,
   449→    )
   450→    all_clustered_ids = _all_cluster_issue_ids(effective_plan)
   451→    if (
   452→        isinstance(effective_plan, dict)
   453→        and not _live_planned_queue_ids(effective_plan)
   454→        and all_clustered_ids & executable_objective_ids
   455→    ):
   456→        executable_objective_ids -= all_clustered_ids
   457→    explicit_objective_items = [
   458→        item for item in objective_items
   459→        if item.get("id", "") in executable_objective_ids
   460→    ]
   461→    triage_snapshot = (
   462→        build_triage_snapshot(effective_plan, state)
   463→        if isinstance(effective_plan, dict)
   464→        else None
   465→    )
   466→    review_issue_items = _review_issue_items(all_issue_items)
   467→    assessment_request_items = _assessment_request_items(all_issue_items)
   468→    executable_review_items = _executable_review_issue_items(
   469→        effective_plan,
   470→        state,
   471→        review_issue_items,
   472→    )
   473→    review_issue_ids = {item.get("id", "") for item in review_issue_items}
   474→    assessment_request_ids = {item.get("id", "") for item in assessment_request_items}
   475→    explicit_queue_items, anchored_execution_items = _merge_execution_candidates(
   476→        all_issue_items=all_issue_items,
   477→        explicit_objective_items=explicit_objective_items,
   478→        plan=effective_plan,
   479→        review_issue_ids=review_issue_ids,
   480→        assessment_request_ids=assessment_request_ids,
   481→    )
   482→    initial_review_items, subjective_postflight_items = _subjective_partitions(
   483→        state,
   484→        scoped_issues=scoped_issues,
   485→        threshold=target_strict,
   486→        plan=effective_plan,
   487→    )
   488→    # Suppress subjective dimension items when review issues already cover
   489→    # the same dimension — the review issues are more actionable and will
   490→    # surface in the review-postflight phase instead.
   491→    postflight_assessment_items = [
   492→        item
   493→        for item in subjective_postflight_items
   494→        if not (
   495→            item.get("kind") == "subjective_dimension"
   496→            and int((item.get("detail") or {}).get("open_review_issues", 0)) > 0
   497→        )
   498→    ] + list(assessment_request_items)
   499→    postflight_review_items = [
   500→        *executable_review_items,
   501→    ]
   502→    scan_items, postflight_workflow_items, triage_items = _workflow_partitions(
   503→        effective_plan,
   504→        state,
   505→    )
   506→
   507→    fresh_boundary = _is_fresh_boundary(effective_plan)
   508→    phase = _phase_for_snapshot(
   509→        effective_plan,
   510→        fresh_boundary=fresh_boundary,
   511→        initial_review_items=initial_review_items,
   512→        anchored_execution_items=anchored_execution_items,
   513→        explicit_queue_items=explicit_queue_items,
   514→        scan_items=scan_items,
   515→        review_backlog_present=bool(review_issue_items),
   516→        undispositioned_review_backlog_present=bool(
   517→            triage_snapshot is not None and triage_snapshot.undispositioned_ids

   519→        postflight_assessment_items=postflight_assessment_items,
   520→        postflight_review_items=postflight_review_items,

   522→        triage_items=triage_items,
   523→    )
   524→    execution_items = _execution_items_for_phase(
   525→        phase,
   526→        explicit_queue_items=explicit_queue_items,
   527→        initial_review_items=initial_review_items,
   528→        scan_items=scan_items,
   529→        postflight_assessment_items=postflight_assessment_items,
   530→        postflight_review_items=postflight_review_items,
   531→        postflight_workflow_items=postflight_workflow_items,
   532→        triage_items=triage_items,

   534→
   535→    execution_ids = {item.get("id", "") for item in execution_items}
   536→    backlog_items = [
   537→        item for item in (
   538→            [
   539→                *objective_items,
   540→                *initial_review_items,
   541→                *postflight_assessment_items,
   542→                *review_issue_items,
   543→                *scan_items,
   544→                *postflight_workflow_items,
   545→                *triage_items,
   546→            ]
   547→        )
   548→        if item.get("id", "") not in execution_ids
   549→    ]
   550→    objective_backlog_count = sum(
   551→        1 for item in objective_items if item.get("id", "") not in execution_ids
   552→    )
   553→    has_unplanned_objective_blockers = len(explicit_objective_items) < len(objective_items)
   554→
   555→    return QueueSnapshot(
   556→        phase=phase,
   557→        all_objective_items=tuple(objective_items),
   558→        all_initial_review_items=tuple(initial_review_items),
   559→        all_postflight_assessment_items=tuple(postflight_assessment_items),
   560→        all_postflight_review_items=tuple(postflight_review_items),
