   306→        # the unscored phase and should not be visible until the objective
   307→        # backlog clears or a cycle completes.

   309→            fid for fid in order

   312→        ]

   318→        to_front = cycle_just_completed and has_real_items

   348→def sync_triage_needed(
   349→    plan: PlanModel,
   350→    state: StateModel,
   351→) -> TriageSyncResult:
   352→    """Inject triage stage IDs at front of queue when review issues change.
   353→
   354→    Only injects stages not already confirmed in ``epic_triage_meta``.
   355→
   356→    When stages are already present but all new issues have been resolved
   357→    since injection, auto-prunes the stale stages and updates the hash.
   358→
   359→    When issues are *resolved* (current IDs are a subset of previously
   360→    triaged IDs), the snapshot hash is updated silently — no re-triage
   361→    is needed since the user is working through the plan.
   362→    """
   363→    ensure_plan_defaults(plan)
   364→    result = TriageSyncResult()
   365→    order: list[str] = plan["queue_order"]
   366→    meta = plan.get("epic_triage_meta", {})
   367→    confirmed = set(meta.get("triage_stages", {}).keys())
   368→
   369→    # Check if any triage stage is already in queue
   370→    already_present = any(sid in order for sid in TRIAGE_IDS)
   371→
   372→    current_hash = review_issue_snapshot_hash(state)
   373→    last_hash = meta.get("issue_snapshot_hash", "")
   374→
   375→    if already_present:
   376→        # Stages present — check if the reason for injection still applies.
   377→        # Only auto-prune when triage was completed before (hash exists),
   378→        # all new issues have been resolved, and no triage work is in
   379→        # progress.  This avoids pruning the initial triage or a
   380→        # user-started triage session.
   381→        if last_hash and not confirmed:
   382→            new_since_triage = _new_review_ids_since_triage(state, meta)
   383→
   384→            if not new_since_triage:
   385→                # No new issues remain — prune stale stages
   386→                _prune_all_triage_stages(order)
   387→                if current_hash:
   388→                    meta["issue_snapshot_hash"] = current_hash
   389→                    plan["epic_triage_meta"] = meta
   390→                result.pruned = True
   391→        return result
   392→
   393→    if current_hash and current_hash != last_hash:
   394→        # Distinguish "new issues appeared" from "issues were resolved".
   395→        # Only re-triage when genuinely new issues exist.
   396→        new_since_triage = _new_review_ids_since_triage(state, meta)
   397→
   398→        if new_since_triage:
   399→            # New review issues appeared — re-triage needed
   400→            result.injected = _inject_pending_triage_stages(order, plan, confirmed)
   401→        else:
   402→            # Only resolved issues changed the hash — update silently
   403→            meta["issue_snapshot_hash"] = current_hash
   404→            plan["epic_triage_meta"] = meta
   405→
   406→    return result

   144→def reconcile_plan_after_scan(
   145→    plan: PlanModel,
   146→    state: StateModel,
   147→) -> ReconcileResult:
   148→    """Reconcile plan against current state after a scan.
   149→

   151→    open, moves them to superseded, and prunes old superseded entries.

   164→    for cluster in plan.get("clusters", {}).values():
   165→        referenced_ids.update(cluster.get("issue_ids", []))
   166→

   168→    already_superseded = set(plan.get("superseded", {}).keys())
   169→    referenced_ids -= already_superseded
   170→    referenced_ids = {
   171→        fid for fid in referenced_ids
   172→        if not any(fid.startswith(prefix) for prefix in SYNTHETIC_PREFIXES)
   173→    }

   175→    # Snapshot non-epic cluster sizes before superseding so we can detect
   176→    # clusters that become empty (all issues resolved by scan).
   177→    clusters = plan.get("clusters", {})
   178→    pre_sizes: dict[str, int] = {
   179→        name: len(cluster.get("issue_ids", []))
   180→        for name, cluster in clusters.items()
   181→        if not name.startswith(EPIC_PREFIX)
   182→    }
   183→
   184→    # Check each referenced ID
   185→    for fid in sorted(referenced_ids):

   187→            if _supersede_id(plan, state, fid, now):

   190→

   192→    # Log cluster_done so the plan tracks completion the same way as user resolves.
   193→    for name, prev_size in pre_sizes.items():

   208→            )
   209→            result.changes += 1
   210→
   211→    # Reconcile epic clusters: remove dead issues, delete empty epics
   212→    clusters = plan.get("clusters", {})
   213→    epic_names_to_delete: list[str] = []
   214→    for name, cluster in list(clusters.items()):
   215→        if not name.startswith(EPIC_PREFIX):
   216→            continue
   217→        issue_ids = cluster.get("issue_ids", [])
   218→        alive_ids = [fid for fid in issue_ids if _is_issue_alive(state, fid)]
   219→        if alive_ids != issue_ids:
   220→            cluster["issue_ids"] = alive_ids
   221→            result.changes += 1
   222→        if not alive_ids:
   223→            epic_names_to_delete.append(name)
   224→    for name in epic_names_to_delete:
   225→        clusters.pop(name, None)
   226→        result.changes += 1
   227→
   228→    # Resurface stale temporary skips
   229→    scan_count = state.get("scan_count", 0)

   235→
   236→    # Prune old superseded entries
   237→    pruned = _prune_old_superseded(plan, now_dt)
   238→    result.pruned = pruned
   239→    result.changes += len(pruned)
   240→
   241→    # Log reconciliation if any changes were made
   242→    if result.changes > 0:
   243→        append_log_entry(
   244→            plan,
   245→            "reconcile",
   246→            issue_ids=result.superseded,
   247→            actor="system",
   248→            detail={
   249→                "superseded_count": len(result.superseded),
   250→                "pruned_count": len(result.pruned),
   251→                "resurfaced_count": len(result.resurfaced),
   252→                "clusters_completed_count": len(result.clusters_completed),
   253→            },
   254→        )
   255→
   256→    return result

- This causes `is_triage_stale()` to return `True` EVEN WHEN NO NEW ISSUES APPEARED
