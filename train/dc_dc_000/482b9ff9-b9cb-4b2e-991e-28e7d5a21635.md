    93→def _apply_subjective_interleave_guardrail(

   104→        item for item in items if _is_subjective_queue_item(item)

   107→        item for item in items if not _is_subjective_queue_item(item)

   114→        for _ in range(objective_burst):
   115→            if not objective_items:
   116→                break

   118→        for _ in range(subjective_burst):
   119→            if not subjective_items:
   120→                break
   121→            interleaved.append(subjective_items.popleft())

   377→            # (--force-review-rerun) while objective backlog exists — the

   379→            if has_objective_backlog and item.get("backlog_gated"):

   405→    should_collapse = (
   406→        resolved_options.collapse_clusters
   407→        and resolved_options.plan
   408→        and not resolved_options.cluster
   409→        and not resolved_options.plan.get("active_cluster")
   410→    )

   412→        all_items = _collapse_clusters(all_items, resolved_options.plan)

   415→    gate_bypass = bool(resolved_options.scope or resolved_options.cluster)

   417→    plan_dirty = False

   427→        and status in {"open", "all"}

   429→        and not resolved_options.scope
   430→        and not resolved_options.cluster
