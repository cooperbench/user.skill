88	 * path in moveSelectedClipsToTrack. Pure; does not touch dataRef.
89	 */
90	function translateGroupMembers({
91	  current,
92	  group,
93	  targetRowId,
94	  deltaTime,
95	}: {
96	  current: TimelineData;
97	  group: PinnedShotGroup;
98	  targetRowId: string;
99	  deltaTime: number;
100	}): {
101	  nextRows: TimelineRow[];
102	  metaUpdates: Record<string, Partial<TimelineData['meta'][string]>>;
103	  nextClipOrder: TimelineData['clipOrder'];
104	  pinnedShotGroupsOverride: PinnedShotGroup[];
105	} {
106	  const memberSet = new Set(group.clipIds);
107	  const sourceTrackId = group.trackId;
108	  const storedGroup = findGroupForTrack(
109	    current.config.pinnedShotGroups ?? [],
110	    group.shotId,
111	    sourceTrackId,
112	    current.rows,
113	  );
114	  const storedTrackId = storedGroup?.trackId ?? sourceTrackId;
115	
116	  // Capture each member's current action so we can translate it.
117	  const memberActions = new Map<string, { start: number; end: number; effectId: string }>();
118	  for (const row of current.rows) {
119	    for (const action of row.actions) {
120	      if (memberSet.has(action.id)) {
121	        memberActions.set(action.id, { start: action.start, end: action.end, effectId: action.effectId });
122	      }
123	    }
124	  }
125	
126	  // Remove members from their current rows, then add them (translated) to the target row.
127	  let nextRows = current.rows.map((row) => ({
128	    ...row,
129	    actions: row.actions.filter((a) => !memberSet.has(a.id)),
130	  }));
131	  const translatedActions = group.clipIds
132	    .map((clipId) => {
133	      const original = memberActions.get(clipId);
134	      if (!original) return null;
135	      return {
136	        id: clipId,
137	        start: original.start + deltaTime,
138	        end: original.end + deltaTime,
139	        effectId: original.effectId,
140	      };
141	    })
142	    .filter((a): a is NonNullable<typeof a> => a !== null);
143	
144	  nextRows = nextRows.map((row) => (
145	    row.id === targetRowId
146	      ? { ...row, actions: [...row.actions, ...translatedActions] }
147	      : row
148	  ));
149	
150	  // Cross-track move: update each member's `track` meta patch and clip order.
151	  const metaUpdates: Record<string, Partial<TimelineData['meta'][string]>> = {};
152	  let nextClipOrder = current.clipOrder;
153	  if (sourceTrackId !== targetRowId) {
154	    for (const clipId of group.clipIds) {
155	      metaUpdates[clipId] = { track: targetRowId };
156	      nextClipOrder = moveClipBetweenTracks(nextClipOrder, clipId, sourceTrackId, targetRowId);
157	    }
158	  }
159	
160	  // Soft-tag override: trackId updated, clipIds re-derived from the translated
161	  // positions (order preserved for the trivial same-delta case, but reads from
162	  // nextRows so any future resolveOverlaps shuffles are honored).
163	  const orderedClipIds = orderClipIdsByAt(group.clipIds, { rows: nextRows });
164	  const pinnedShotGroupsOverride = (current.config.pinnedShotGroups ?? []).map((g) => (
165	    g.shotId === group.shotId && g.trackId === storedTrackId
166	      ? { ...g, trackId: targetRowId, clipIds: orderedClipIds }
167	      : g
168	  ));
169	
170	  return { nextRows, metaUpdates, nextClipOrder, pinnedShotGroupsOverride };
171	}
172	
173	function getLiveGroupStart(current: Pick<TimelineData, 'rows'>, group: PinnedShotGroup): number | null {
174	  const actionStarts = current.rows
175	    .flatMap((row) => row.actions)
176	    .filter((action) => group.clipIds.includes(action.id))
177	    .map((action) => action.start);
178	
179	  if (actionStarts.length === 0) {
180	    return null;
181	  }
182	
183	  return Math.min(...actionStarts);
184	}
185	
186	function getLiveGroupEnd(current: Pick<TimelineData, 'rows'>, group: PinnedShotGroup): number | null {
187	  const actionEnds = current.rows
188	    .flatMap((row) => row.actions)
189	    .filter((action) => group.clipIds.includes(action.id))
190	    .map((action) => action.end);
191	
192	  if (actionEnds.length === 0) {
193	    return null;
194	  }
195	
196	  return Math.max(...actionEnds);
197	}

199	export function useTimelineTrackManagement({

203	  setSelectedTrackId,
204	  applyEdit,
205	}: UseTimelineTrackManagementArgs): UseTimelineTrackManagementResult {
206	  const moveClipToRow = useCallback((
207	    clipId: string,
208	    targetRowId: string,
209	    newStartTime?: number,
210	    transactionId?: string,
211	  ) => {
212	    let current = dataRef.current;
213	    if (!current) {
214	      return;
215	    }
216	
217	    const enclosingGroup = findEnclosingPinnedGroup(current.config, clipId);
218	    if (enclosingGroup) {
219	      const resolvedTrackId = resolveGroupTrackId(enclosingGroup.group, current.rows);
220	      const resolvedGroup = resolvedTrackId === enclosingGroup.group.trackId
221	        ? enclosingGroup.group
222	        : { ...enclosingGroup.group, trackId: resolvedTrackId };
223	      const sourceTrack = current.tracks.find((track) => track.id === resolvedTrackId);
224	      const targetTrack = current.tracks.find((track) => track.id === targetRowId);
225	      if (!sourceTrack || !targetTrack || sourceTrack.kind !== targetTrack.kind) {
226	        return;
227	      }
228	
229	      const groupStart = getLiveGroupStart(current, resolvedGroup);
230	      const groupEnd = getLiveGroupEnd(current, resolvedGroup);
231	      if (groupStart == null || groupEnd == null) {
232	        return;
233	      }
234	      const nextStart = typeof newStartTime === 'number' ? Math.max(0, newStartTime) : groupStart;
235	      const groupDuration = groupEnd - groupStart;
236	
237	      // Find a free track for the group's bounding box, excluding the group's own clips
238	      const memberSet = new Set(resolvedGroup.clipIds);
239	      const rowsWithoutGroup = current.rows.map((row) => ({
240	        ...row,
241	        actions: row.actions.filter((a) => !memberSet.has(a.id)),
242	      }));
243	      const snapResult = trySnapToEdge(
244	        rowsWithoutGroup,
245	        targetRowId,
246	        nextStart,
247	        groupDuration,
248	      );
249	      const effectiveGroupStart = snapResult.snapped ? snapResult.time : nextStart;
250	      let finalTargetId = snapResult.snapped
251	        ? targetRowId
252	        : findNearestFreeTrack(
253	            current.tracks,
254	            rowsWithoutGroup,
255	            targetRowId,
256	            sourceTrack.kind,
257	            effectiveGroupStart,
258	            groupDuration,
259	          );
260	
261	      if (!finalTargetId) {
262	        const prefix = sourceTrack.kind === 'audio' ? 'A' : 'V';
263	        const nextNumber = getTrackIndex(current.tracks, prefix) + 1;
264	        finalTargetId = `${prefix}${nextNumber}`;
265	        current = {
266	          ...current,
267	          tracks: [...current.tracks, { id: finalTargetId, kind: sourceTrack.kind, label: finalTargetId }],
268	          rows: [...current.rows, { id: finalTargetId, actions: [] }],
269	        };
270	        dataRef.current = current;
271	      }
272	
273	      const translatedGroup = translateGroupMembers({
274	        current,
275	        group: resolvedGroup,
276	        targetRowId: finalTargetId,
277	        deltaTime: effectiveGroupStart - groupStart,
278	      });
279	      applyEdit({
280	        type: 'rows',
281	        rows: translatedGroup.nextRows,
282	        metaUpdates: Object.keys(translatedGroup.metaUpdates).length > 0 ? translatedGroup.metaUpdates : undefined,
283	        clipOrderOverride: translatedGroup.nextClipOrder,
284	        pinnedShotGroupsOverride: translatedGroup.pinnedShotGroupsOverride,
285	      }, { transactionId });
286	      return;
287	    }
288	
289	    const sourceRow = current.rows.find((row) => row.actions.some((action) => action.id === clipId));
290	    const targetRow = current.rows.find((row) => row.id === targetRowId);
291	    if (!sourceRow || !targetRow) {
292	      return;
293	    }
294	
295	    const sourceTrack = current.tracks.find((track) => track.id === sourceRow.id);
296	    const targetTrack = current.tracks.find((track) => track.id === targetRow.id);
297	    const action = sourceRow.actions.find((candidate) => candidate.id === clipId);
298	    if (!sourceTrack || !targetTrack || !action || sourceTrack.kind !== targetTrack.kind) {
299	      return;
300	    }
301	
302	    const duration = action.end - action.start;
303	    const nextStart = typeof newStartTime === 'number' ? Math.max(0, newStartTime) : action.start;
304	
305	    // Find a free track (target first, then nearest above/below), or create one
306	    const snapResult = trySnapToEdge(
307	      current.rows,
308	      targetRow.id,
309	      nextStart,
310	      duration,
311	      clipId,
312	    );
313	    let effectiveStart = snapResult.snapped ? snapResult.time : nextStart;
314	    let finalTrackId = snapResult.snapped
315	      ? targetRow.id
316	      : findNearestFreeTrack(
317	          current.tracks,
318	          current.rows,
319	          targetRow.id,
320	          sourceTrack.kind,
321	          effectiveStart,
322	          duration,
323	          clipId,
324	        );
325	
326	    if (!finalTrackId) {
327	      const prefix = sourceTrack.kind === 'audio' ? 'A' : 'V';
328	      const nextNumber = getTrackIndex(current.tracks, prefix) + 1;
329	      finalTrackId = `${prefix}${nextNumber}`;
330	      current = {
331	        ...current,
332	        tracks: [...current.tracks, { id: finalTrackId, kind: sourceTrack.kind, label: finalTrackId }],
333	        rows: [...current.rows, { id: finalTrackId, actions: [] }],
334	      };
335	      dataRef.current = current;
336	    }
337	
338	    const nextAction = { ...action, start: effectiveStart, end: effectiveStart + duration };
339	    const nextRows = current.rows.map((row) => {
340	      if (row.id === sourceRow.id && row.id === finalTrackId) {
341	        return {
342	          ...row,
343	          actions: row.actions.map((candidate) => (candidate.id === clipId ? nextAction : candidate)),
344	        };
345	      }
346	
347	      if (row.id === sourceRow.id) {
348	        return { ...row, actions: row.actions.filter((candidate) => candidate.id !== clipId) };
349	      }
350	
351	      if (row.id === finalTrackId) {
352	        return { ...row, actions: [...row.actions, nextAction] };
353	      }
354	
355	      return row;
356	    });
357	
358	    const nextClipOrder = moveClipBetweenTracks(current.clipOrder, clipId, sourceRow.id, finalTrackId);
359	    applyEdit({
360	      type: 'rows',
361	      rows: nextRows,
362	      metaUpdates: {
363	        [clipId]: { track: finalTrackId },
364	      },
365	      clipOrderOverride: nextClipOrder,
366	    }, { transactionId });
367	  }, [applyEdit, dataRef]);
368	
369	  const createTrackAndMoveClip = useCallback((clipId: string, kind: TrackKind, newStartTime?: number, insertAtTop = false) => {
370	    const current = dataRef.current;
371	    if (!current) {
372	      return;
373	    }
374	
375	    const enclosingGroup = findEnclosingPinnedGroup(current.config, clipId);
376	    const sourceClip = current.resolvedConfig.clips.find((clip) => clip.id === clipId);
377	    const sourceTrackId = enclosingGroup
378	      ? resolveGroupTrackId(enclosingGroup.group, current.rows)
379	      : sourceClip?.track;
380	    const sourceTrack = sourceTrackId
381	      ? current.resolvedConfig.tracks.find((track) => track.id === sourceTrackId)
382	      : null;
383	    if ((!sourceClip && !enclosingGroup) || !sourceTrack || sourceTrack.kind !== kind) {
384	      return;
385	    }
386	
387	    const nextResolvedConfigBase = addTrack(current.resolvedConfig, kind, insertAtTop ? 0 : undefined);
388	    const newTrack = nextResolvedConfigBase.tracks.find((track) => {
389	      return !current.resolvedConfig.tracks.some((existingTrack) => existingTrack.id === track.id);
390	    }) ?? nextResolvedConfigBase.tracks[nextResolvedConfigBase.tracks.length - 1];
391	    if (!newTrack) {
392	      return;
393	    }
394	
395	    if (enclosingGroup) {
396	      const previewCurrent: TimelineData = {
397	        ...current,
398	        tracks: nextResolvedConfigBase.tracks.map((track) => ({ ...track })),
399	        rows: nextResolvedConfigBase.tracks.map((track) => (
400	          current.rows.find((row) => row.id === track.id) ?? { id: track.id, actions: [] }
401	        )),
402	        config: {
403	          ...current.config,
404	          tracks: nextResolvedConfigBase.tracks.map((track) => ({ ...track })),
405	        },
406	      };
407	      const resolvedGroup = sourceTrackId === enclosingGroup.group.trackId
408	        ? enclosingGroup.group
409	        : { ...enclosingGroup.group, trackId: sourceTrackId };
410	      const groupStart = getLiveGroupStart(current, resolvedGroup);
411	      if (groupStart == null) {
412	        return;
413	      }
414	      const nextStart = typeof newStartTime === 'number' ? Math.max(0, newStartTime) : groupStart;
415	      const deltaTime = nextStart - groupStart;
416	      const translatedGroup = translateGroupMembers({
417	        current: previewCurrent,
418	        group: resolvedGroup,
419	        targetRowId: newTrack.id,
420	        deltaTime,
421	      });
422	      const nextResolvedConfig = {
423	        ...nextResolvedConfigBase,
424	        clips: nextResolvedConfigBase.clips.map((clip) => (
425	          enclosingGroup.group.clipIds.includes(clip.id)
426	            ? {
427	                ...clip,
428	                at: clip.at + deltaTime,
429	                track: newTrack.id,
430	              }
431	            : clip
432	        )),
433	      };
434	
435	      applyEdit({
436	        type: 'config',
437	        resolvedConfig: nextResolvedConfig,
438	        pinnedShotGroupsOverride: translatedGroup.pinnedShotGroupsOverride,
439	      }, {
440	        selectedClipId: clipId,
441	        selectedTrackId: newTrack.id,
442	      });
443	      return;
444	    }
445	
446	    const nextResolvedConfig = {
447	      ...nextResolvedConfigBase,
448	      clips: nextResolvedConfigBase.clips.map((clip) => {
449	        if (clip.id !== clipId) {
450	          return clip;
451	        }
452	
453	        return {
454	          ...clip,
455	          at: typeof newStartTime === 'number' ? Math.max(0, newStartTime) : clip.at,
456	          track: newTrack.id,
457	        };
458	      }),
459	    };
460	
461	    applyEdit({
462	      type: 'config',
463	      resolvedConfig: nextResolvedConfig,
464	    }, {
465	      selectedClipId: clipId,
466	      selectedTrackId: newTrack.id,
467	    });
468	  }, [applyEdit, dataRef]);
469	
470	  const moveSelectedClipToTrack = useCallback((direction: 'up' | 'down') => {
471	    const current = dataRef.current;
472	    if (!current || !selectedClipId) {
473	      return;
474	    }
475	
476	    const currentRowIndex = current.rows.findIndex((row) => row.actions.some((action) => action.id === selectedClipId));
477	    if (currentRowIndex < 0) {
478	      return;
479	    }
480	
481	    const sourceTrack = current.tracks.find((track) => track.id === current.rows[currentRowIndex]?.id);
482	    if (!sourceTrack) {
483	      return;
484	    }
485	
486	    let targetRowIndex = currentRowIndex;
487	    while (true) {
488	      targetRowIndex += direction === 'up' ? -1 : 1;
489	      if (targetRowIndex < 0 || targetRowIndex >= current.rows.length) {
490	        return;
491	      }
492	
493	      const targetTrack = current.tracks.find((track) => track.id === current.rows[targetRowIndex]?.id);
494	      if (targetTrack?.kind === sourceTrack.kind) {
495	        moveClipToRow(selectedClipId, targetTrack.id);
496	        setSelectedTrackId(targetTrack.id);
497	        return;
498	      }
499	    }
500	  }, [dataRef, moveClipToRow, selectedClipId, setSelectedTrackId]);
501	
502	  const moveSelectedClipsToTrack = useCallback((
503	    direction: 'up' | 'down',
504	    selectedClipIds: ReadonlySet<string>,
505	  ) => {
506	    const current = dataRef.current;
507	    if (!current || selectedClipIds.size === 0) {
508	      return;
509	    }
510	
511	    const trackById = new Map(current.tracks.map((track) => [track.id, track]));
512	    const selection = categorizeSelection([...selectedClipIds], current.config);
513	    const freeClipIdSet = new Set(selection.freeClipIds);
514	    const groupsByRow = new Map<string, { clipIds: string[]; rowIndex: number; kind: TrackKind }>();
515	    const groupedUnits = selection.groups.flatMap((groupEntry) => {
516	      const group = findGroupForTrack(
517	        current.config.pinnedShotGroups ?? [],
518	        groupEntry.groupKey.shotId,
519	        groupEntry.groupKey.trackId,
520	        current.rows,
521	      );
522	      if (!group) {
523	        return [];
524	      }
525	
526	      const resolvedTrackId = resolveGroupTrackId(group, current.rows);
527	      const rowIndex = current.rows.findIndex((row) => row.id === resolvedTrackId);
528	      const track = trackById.get(resolvedTrackId);
529	      if (rowIndex < 0 || !track) {
530	        return [];
531	      }
532	
533	      return [{
534	        kind: 'group' as const,
535	        [REDACTED],
536	        rowIndex,
537	        rowId: resolvedTrackId,
538	        trackKind: track.kind,
539	      }];
540	    });
541	
542	    current.rows.forEach((row, rowIndex) => {
543	      const sourceTrack = trackById.get(row.id);
544	      if (!sourceTrack) {
545	        return;
546	      }
547	
548	      const rowClipIds = row.actions
549	        .map((action) => action.id)
550	        .filter((clipId) => freeClipIdSet.has(clipId));
551	      if (rowClipIds.length === 0) {
552	        return;
553	      }
554	
555	      groupsByRow.set(row.id, {
556	        clipIds: rowClipIds,
557	        rowIndex,
558	        kind: sourceTrack.kind,
559	      });
560	    });
561	
562	    if (groupsByRow.size === 0 && groupedUnits.length === 0) {
563	      return;
564	    }
565	
566	    const freeUnits = [...groupsByRow.entries()].map(([rowId, group]) => ({
567	      kind: 'free' as const,
568	      rowId,
569	      rowIndex: group.rowIndex,
570	      trackKind: group.kind,
571	      clipIds: group.clipIds,
572	    }));
573	    const plannedClipMoves: Array<{ clipId: string; targetRowId: string }> = [];
574	    const plannedGroupMoves: Array<{ groupKey: PinnedGroupKey; targetRowId: string }> = [];
575	
576	    for (const kind of ['visual', 'audio'] as const) {
577	      const kindUnits = [
578	        ...freeUnits.filter((unit) => unit.trackKind === kind),
579	        ...groupedUnits.filter((unit) => unit.trackKind === kind),
580	      ];
581	      if (kindUnits.length === 0) {
582	        continue;
583	      }
584	
585	      const kindClipMoves: Array<{ clipId: string; targetRowId: string }> = [];
586	      const kindGroupMoves: Array<{ groupKey: PinnedGroupKey; targetRowId: string }> = [];
587	      let isBlocked = false;
588	
589	      for (const unit of kindUnits) {
590	        let targetRowIndex = unit.rowIndex;
591	        let targetRowId: string | null = null;
592	
593	        while (true) {
594	          targetRowIndex += direction === 'up' ? -1 : 1;
595	          if (targetRowIndex < 0 || targetRowIndex >= current.rows.length) {
596	            isBlocked = true;
597	            break;
598	          }
599	
600	          const targetTrack = trackById.get(current.rows[targetRowIndex]?.id ?? '');
601	          if (targetTrack?.kind === kind) {
602	            targetRowId = targetTrack.id;
603	            break;
604	          }
605	        }
606	
607	        if (isBlocked || !targetRowId) {
608	          isBlocked = true;
609	          break;
610	        }
611	
612	        if (unit.kind === 'group') {
613	          kindGroupMoves.push({ groupKey: unit.groupKey, targetRowId });
614	          continue;
615	        }
616	
617	        for (const clipId of unit.clipIds) {
618	          kindClipMoves.push({ clipId, targetRowId });
619	        }
620	      }
621	
622	      if (!isBlocked) {
623	        plannedClipMoves.push(...kindClipMoves);
624	        plannedGroupMoves.push(...kindGroupMoves);
625	      }
626	    }
627	
628	    if (plannedClipMoves.length === 0 && plannedGroupMoves.length === 0) {
629	      return;
630	    }
631	
632	    const transactionId = crypto.randomUUID();
633	    if (plannedGroupMoves.length > 0) {
634	      let workingState: TimelineData = current;
635	      const accumulatedMetaUpdates: Record<string, Partial<TimelineData['meta'][string]>> = {};
636	      let appliedGroupMove = false;
637	
638	      for (const move of plannedGroupMoves) {
639	        const existingGroup = findGroupForTrack(
640	          workingState.config.pinnedShotGroups ?? [],
641	          move.groupKey.shotId,
642	          move.groupKey.trackId,
643	          workingState.rows,
644	        );
645	        if (!existingGroup) {
646	          continue;
647	        }
648	
649	        const resolvedTrackId = resolveGroupTrackId(existingGroup, workingState.rows);
650	        const resolvedGroup = resolvedTrackId === existingGroup.trackId
651	          ? existingGroup
652	          : { ...existingGroup, trackId: resolvedTrackId };
653	
654	        const translatedGroup = translateGroupMembers({
655	          current: workingState,
656	          group: resolvedGroup,
657	          targetRowId: move.targetRowId,
658	          deltaTime: 0,
659	        });
660	        Object.assign(accumulatedMetaUpdates, translatedGroup.metaUpdates);
661	        const nextMeta: Record<string, ClipMeta> = {
662	          ...workingState.meta,
663	          ...translatedGroup.metaUpdates,
664	        } as Record<string, ClipMeta>;
665	        workingState = {
666	          ...workingState,
667	          rows: translatedGroup.nextRows,
668	          meta: nextMeta,
669	          clipOrder: translatedGroup.nextClipOrder,
670	          config: {
671	            ...workingState.config,
672	            pinnedShotGroups: translatedGroup.pinnedShotGroupsOverride,
673	          },
674	        };
675	        appliedGroupMove = true;
676	      }
677	
678	      if (appliedGroupMove) {
679	        applyEdit({
680	          type: 'rows',
681	          rows: workingState.rows,
682	          metaUpdates: Object.keys(accumulatedMetaUpdates).length > 0 ? accumulatedMetaUpdates : undefined,
683	          clipOrderOverride: workingState.clipOrder,
684	          pinnedShotGroupsOverride: workingState.config.pinnedShotGroups,
685	        }, { transactionId });
686	      }
687	    }
688	
689	    for (const move of plannedClipMoves) {
690	      moveClipToRow(move.clipId, move.targetRowId, undefined, transactionId);
691	    }
692	
693	    const primaryGroupMove = selectedClipId
694	      ? findEnclosingPinnedGroup(current.config, selectedClipId)
695	      : null;
696	    const primaryMove = primaryGroupMove
697	      ? plannedGroupMoves.find((move) => (
698	          move.groupKey.shotId === primaryGroupMove.groupKey.shotId
699	          && move.groupKey.trackId === primaryGroupMove.groupKey.trackId
700	        ))
701	      : (selectedClipId
702	          ? plannedClipMoves.find((move) => move.clipId === selectedClipId)
703	          : undefined)
704	        ?? plannedGroupMoves[0]
705	        ?? plannedClipMoves[0];
706	    if (primaryMove) {
707	      setSelectedTrackId(primaryMove.targetRowId);
708	    }
709	  }, [applyEdit, dataRef, moveClipToRow, selectedClipId, setSelectedTrackId]);
710	
711	  const handleAddTrack = useCallback((kind: TrackKind) => {
712	    if (!resolvedConfig) {
713	      return;
714	    }
715	
716	    const nextResolvedConfig = addTrack(resolvedConfig, kind);
717	    const nextTrack = nextResolvedConfig.tracks[nextResolvedConfig.tracks.length - 1] ?? null;
718	    applyEdit({ type: 'config', resolvedConfig: nextResolvedConfig }, { selectedTrackId: nextTrack?.id ?? null });
