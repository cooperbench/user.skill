   ```sh
   update_worker_phase_strict() {
       local phase="$1"
       local key="$SUPABASE_SERVICE_ROLE_KEY"
       [ -z "$key" ] && return 1  # missing key is a hard failure, not silent
       local http_code
       http_code=$(curl --silent --show-error --output /dev/null \
           --write-out '%{http_code}' \
           --max-time 10 \
           --retry 0 \
           -X PATCH \
           "${SUPABASE_URL}/rest/v1/workers?id=eq.${WORKER_ID}" \
           -H "Content-Type: application/json" \
           -H "Authorization: Bearer $key" \
           -H "apikey: $key" \
           -H "Prefer: return=minimal" \
           -d "{\"metadata\":{\"startup_phase\":\"$phase\"}}") || return 1
       case "$http_code" in
           2*) return 0 ;;
           *) echo "update_worker_phase_strict: HTTP $http_code" >&2 ; return 1 ;;
       esac
   }
   ```
