    29→def _extract_subtype(finding: dict) -> str | None:
    30→    """Extract the subtype/kind from a finding.

    41→    fid = finding.get("id", "")

    44→        # For IDs like smells::file.py::silent_except, the subtype is parts[2]
    45→        # For IDs like dict_keys::file.py::key::phantom_read, subtype is last part
    46→        candidate = parts[-1]

   150→def _subtype_has_fixer(meta: DetectorMeta, subtype: str | None) -> str | None:
   151→    """Check if a subtype maps to a specific fixer."""

    65→def _apply_subjective_interleave_guardrail(
    66→    items: list[dict],
    67→    *,
    68→    objective_burst: int = 3,
    69→    subjective_burst: int = 1,
    70→) -> list[dict]:
    71→    """Interleave subjective work with objective work to avoid starvation."""
    72→    if objective_burst < 1 or subjective_burst < 1:
    73→        return items
    74→
    75→    subjective_items = deque(
    76→        item for item in items if _is_subjective_queue_item(item)
    77→    )
    78→    objective_items = deque(
    79→        item for item in items if not _is_subjective_queue_item(item)
    80→    )
    81→    if not subjective_items or not objective_items:
    82→        return items
    83→
    84→    interleaved: list[dict] = []
    85→    while objective_items and subjective_items:
    86→        for _ in range(objective_burst):
    87→            if not objective_items:
    88→                break
    89→            interleaved.append(objective_items.popleft())
    90→        for _ in range(subjective_burst):
    91→            if not subjective_items:
    92→                break
    93→            interleaved.append(subjective_items.popleft())
    94→
    95→    interleaved.extend(objective_items)
    96→    interleaved.extend(subjective_items)
    97→    return interleaved

   100→def _apply_plan_order(
   101→    items: list[dict],
   102→    plan: dict,
   103→    *,
   104→    include_skipped: bool = False,
   105→    cluster: str | None = None,
   106→) -> list[dict]:
   107→    """Reorder items according to the living plan.
   108→
   109→    1. Items in ``queue_order`` appear first, in that order.
   110→    2. Remaining items keep their mechanical sort.
   111→    3. Skipped items are appended last (or excluded).
   112→    4. Each item is annotated with plan metadata.
   113→    """
   114→    queue_order: list[str] = plan.get("queue_order", [])
   115→    skipped_map: dict = plan.get("skipped", {})
   116→    skipped_ids: set[str] = set(skipped_map.keys())
   117→    overrides: dict = plan.get("overrides", {})
   118→    clusters: dict = plan.get("clusters", {})
   119→    active_cluster = plan.get("active_cluster")
   120→
   121→    # Build lookup
   122→    by_id: dict[str, dict] = {}
   123→    for item in items:
   124→        by_id[item["id"]] = item
   125→
   126→    # Annotate items with plan metadata
   127→    for item_id, item in by_id.items():
   128→        override = overrides.get(item_id, {})
   129→        if override.get("description"):
   130→            item["plan_description"] = override["description"]
   131→        if override.get("note"):
   132→            item["plan_note"] = override["note"]
   133→        if override.get("cluster"):
   134→            cluster_name = override["cluster"]
   135→            cluster_data = clusters.get(cluster_name, {})
   136→            item["plan_cluster"] = {
   137→                "name": cluster_name,
   138→                "description": cluster_data.get("description"),
   139→                "total_items": len(cluster_data.get("finding_ids", [])),
   140→                "sibling_ids": cluster_data.get("finding_ids", []),
   141→            }
   142→
   143→    # Split into ordered, remaining, skipped
   144→    ordered: list[dict] = []
   145→    ordered_ids: set[str] = set()
   146→    for fid in queue_order:
   147→        if fid in by_id and fid not in skipped_ids:
   148→            ordered.append(by_id[fid])
   149→            ordered_ids.add(fid)
   150→
   151→    skipped_items: list[dict] = []
   152→    remaining: list[dict] = []
   153→    for item in items:
   154→        item_id = item["id"]
   155→        if item_id in ordered_ids:
   156→            continue
   157→        if item_id in skipped_ids:
   158→            skipped_items.append(item)
   159→        else:
   160→            remaining.append(item)
   161→
   162→    # Assign queue positions
   163→    result = ordered + remaining
   164→    if include_skipped:
   165→        result = result + skipped_items
   166→
   167→    for pos, item in enumerate(result):
   168→        item["queue_position"] = pos + 1
   169→        if item["id"] in skipped_ids:
   170→            item["plan_skipped"] = True
   171→            skip_entry = skipped_map.get(item["id"])
   172→            if skip_entry:
   173→                item["plan_skip_kind"] = skip_entry.get("kind", "temporary")
   174→                skip_reason = skip_entry.get("reason")
   175→                if skip_reason:
   176→                    item["plan_skip_reason"] = skip_reason
   177→
   178→    # Filter to cluster if requested
   179→    effective_cluster = cluster or active_cluster
   180→    if effective_cluster:
   181→        cluster_data = clusters.get(effective_cluster, {})
   182→        cluster_member_ids = set(cluster_data.get("finding_ids", []))
   183→        if cluster_member_ids:
   184→            result = [item for item in result if item["id"] in cluster_member_ids]
   185→
   186→    return result

   214→def _collapse_clusters(items: list[dict], plan: dict) -> list[dict]:
   215→    """Replace cluster member items with single cluster meta-items."""
   216→    clusters = plan.get("clusters", {})
   217→    if not clusters:
   218→        return items
   219→
   220→    # Build mapping: finding_id → auto-cluster name
   221→    fid_to_cluster: dict[str, str] = {}
   222→    for name, cluster in clusters.items():
   223→        if not cluster.get("auto"):
   224→            continue
   225→        for fid in cluster.get("finding_ids", []):
   226→            fid_to_cluster[fid] = name
   227→
   228→    if not fid_to_cluster:
   229→        return items
   230→
   231→    # Collect members for each cluster, preserving order
   232→    cluster_members: dict[str, list[dict]] = {}
   233→    non_cluster_items: list[dict] = []
   234→
   235→    for item in items:
   236→        cname = fid_to_cluster.get(item.get("id", ""))
   237→        if cname:
   238→            cluster_members.setdefault(cname, []).append(item)
   239→        else:
   240→            non_cluster_items.append(item)
   241→
   242→    # Build cluster meta-items
   243→    result: list[dict] = list(non_cluster_items)
   244→    for cname, members in cluster_members.items():
   245→        # Don't collapse singletons — show them as individual findings
   246→        if len(members) < 2:
   247→            result.extend(members)
   248→            continue
   249→        cluster_data = clusters.get(cname, {})
   250→        detector = members[0].get("detector", "") if members else ""
   251→        action = cluster_data.get("action") or ""
   252→        # Derive action_type from the actual cluster action, not just the detector
   253→        if "desloppify fix" in action:
   254→            action_type = "auto_fix"
   255→        elif "desloppify move" in action:
   256→            action_type = "reorganize"
   257→        else:
   258→            action_type = _action_type_for_detector(detector)
   259→            # If detector says auto_fix but cluster has no fix command, it's really refactor
   260→            if action_type == "auto_fix" and "desloppify fix" not in action:
   261→                action_type = "refactor"
   262→
   263→        # Use stored description but patch the count if it differs from visible members
   264→        stored_desc = cluster_data.get("description") or ""
   265→        total_in_cluster = len(cluster_data.get("finding_ids", []))
   266→        if stored_desc and total_in_cluster != len(members):
   267→            # Replace the stored count with the visible count
   268→            summary = stored_desc.replace(str(total_in_cluster), str(len(members)))
   269→        else:
   270→            summary = stored_desc or f"{len(members)} findings"
   271→
   272→        primary_command = cluster_data.get("action")
   273→        if not primary_command:
   274→            primary_command = f"desloppify next --cluster {cname} --count 10"
   275→
   276→        meta_item: dict = {
   277→            "id": cname,
   278→            "kind": "cluster",
   279→            "action_type": action_type,
   280→            "summary": summary,
   281→            "members": members,
   282→            "member_count": len(members),
   283→            "primary_command": primary_command,
   284→            "cluster_name": cname,
   285→            "cluster_auto": True,
   286→            "confidence": "high",
   287→            "detector": detector,
   288→            "file": "",
   289→        }
   290→        result.append(meta_item)
   291→
   292→    result.sort(key=item_sort_key)
   293→    return result

   368→        and not resolved_options.scope
   369→        and not resolved_options.cluster
   370→    )
   371→    if should_interleave:

   373→
   374→    counts = tier_counts(all_items)
   375→

   377→        int(resolved_options.tier) if resolved_options.tier is not None else None
   378→    )

   380→    fallback_reason = None

   382→

   402→        elif not filtered:

   404→
   405→    total = len(filtered)
   406→    if resolved_options.count is not None and resolved_options.count > 0:
   407→        filtered = filtered[: resolved_options.count]
   408→
   409→    if resolved_options.explain:

   412→
   413→    available_tiers = [tier for tier, value in counts.items() if value > 0]

   420→        "fallback_reason": fallback_reason,
   421→        "available_tiers": available_tiers,
   422→        "grouped": group_queue_items(filtered, "item"),

   186→def _reconcile_plan_post_scan(runtime: "ScanRuntime") -> None:
   187→    """Reconcile plan queue metadata and stale subjective review dimensions."""
   188→    try:
   189→        plan_path = runtime.state_path.parent / "plan.json" if runtime.state_path else None
   190→        plan = load_plan(plan_path)
   191→        dirty = False
   192→
   193→        if _apply_plan_reconciliation(plan, runtime.state, reconcile_plan_after_scan):
   194→            dirty = True
   195→
   196→        unscored_changed = _sync_unscored_dimensions(plan, runtime.state, sync_unscored_dimensions)
   197→        if unscored_changed:
   198→            dirty = True
   199→            append_log_entry(plan, "sync_unscored", actor="system",
   200→                             detail={"changes": True})
   201→
   202→        stale_changed = _sync_stale_dimensions(plan, runtime.state, sync_stale_dimensions)
   203→        if stale_changed:
   204→            dirty = True
   205→            append_log_entry(plan, "sync_stale", actor="system",
   206→                             detail={"changes": True})
   207→
   208→        auto_changed = _sync_auto_clusters(plan, runtime.state)
   209→        if auto_changed:
   210→            dirty = True
   211→            append_log_entry(plan, "auto_cluster", actor="system",
   212→                             detail={"changes": True})
   213→
   214→        synth_sync = sync_synthesis_needed(plan, runtime.state)
   215→        if synth_sync.changes:
   216→            dirty = True
   217→            if synth_sync.injected:
   218→                print(
   219→                    colorize(
   220→                        "  Plan: synthesis needed — review findings changed since last synthesis.",
   221→                        "cyan",
   222→                    )
   223→                )
   224→                append_log_entry(plan, "sync_synthesis", actor="system",
   225→                                 detail={"injected": True})
   226→
   227→        seeded = _seed_plan_start_scores(plan, runtime.state)
   228→        if seeded:
   229→            dirty = True
   230→            append_log_entry(plan, "seed_start_scores", actor="system",
   231→                             detail={})
   232→        # Only clear scores that existed before this reconcile pass —
   233→        # never clear scores we just seeded in the same scan.
   234→        if not seeded and _clear_plan_start_scores_if_queue_empty(runtime.state, plan):
   235→            dirty = True
   236→            append_log_entry(plan, "clear_start_scores", actor="system",
   237→                             detail={})
   238→
   239→        if dirty:

   362→def _reset_subjective_assessments_for_scan_reset(
   363→    state: state_mod.StateModel,

   387→            payload.pop("components", None)
   388→            payload.pop("component_scores", None)
   389→            continue
   390→        assessments[key] = {
   391→            "score": 0.0,
   392→            "source": "scan_reset_subjective",
   393→            "assessed_at": now,
   394→            "reset_by": "scan_reset_subjective",

   549→def merge_scan_results(
   550→    runtime: ScanRuntime,
   551→    findings: list[dict[str, Any]],
   552→    potentials: dict[str, object],
   553→    codebase_metrics: dict[str, object] | None,
   554→) -> ScanMergeResult:
   555→    """Merge findings into persistent state and return diff + previous score snapshot."""
   556→    scan_path_rel = rel(str(runtime.path))

   559→
   560→    if not path_changed:
   561→        prev = state_mod.score_snapshot(runtime.state)
   562→    else:
   563→        prev = state_mod.ScoreSnapshot(None, None, None, None)
   564→    prev_dim_scores = (
   565→        runtime.state.get("dimension_scores", {}) if not path_changed else {}
   566→    )
   567→
   568→    if runtime.lang and runtime.lang.zone_map is not None:
   569→        runtime.state["zone_distribution"] = runtime.lang.zone_map.counts()
   570→    _persist_scan_coverage(runtime.state, runtime.lang)
   571→
   572→    target_score = target_strict_score_from_config(runtime.config, fallback=95.0)
   573→
   574→    diff = state_mod.merge_scan(
   575→        runtime.state,
   576→        findings,
   577→        options=state_mod.MergeScanOptions(
   578→            lang=runtime.lang.name if runtime.lang else None,
   579→            scan_path=scan_path_rel,
   580→            force_resolve=getattr(runtime.args, "force_resolve", False),
   581→            exclude=get_exclusions(),
   582→            potentials=potentials,
   583→            codebase_metrics=codebase_metrics,
   584→            include_slow=runtime.effective_include_slow,
   585→            ignore=runtime.config.get("ignore", []),

   588→    )

   590→    issues_mod.expire_stale_holistic(

   602→    return ScanMergeResult(

   605→        prev_objective=prev.objective,

## Comprehensive Analysis: Desloppify Planning/Queue System

Based on my thorough exploration of the codebase, here's how the planning and queue system works:

### 1. KEY DATA STRUCTURES

**Plan Schema** (`desloppify/engine/_plan/schema.py`):
- **`queue_order: list[str]`** — Ordered list of finding IDs, determines priority/display order
- **`skipped: dict[str, SkipEntry]`** — Finding IDs removed from active work (temporary/permanent/false_positive)
- **`overrides: dict[str, ItemOverride]`** — Per-finding metadata (custom description, note, cluster assignment)
- **`clusters: dict[str, Cluster]`** — Named groupings of findings with metadata (description, action, auto flag)
- **`active_cluster: str | None`** — Currently focused cluster for filtering
- **`superseded: dict[str, SupersededEntry]`** — Findings that disappeared from state (kept 90 days for history)
- **`execution_log: list[ExecutionLogEntry]`** — Audit trail of all user/system actions
- **`epic_synthesis_meta: dict`** — Synthesis engine metadata

### 2. HOW ITEMS ARE PRIORITIZED & MOVED UP

**Move Operations** (`desloppify/engine/_plan/operations.py::move_items()`):
```python
def move_items(
    plan: PlanModel,
    finding_ids: list[str],
    position: str,  # "top", "bottom", "before", "after", "up", "down"
    target: str | None = None,
    offset: int | None = None,
) -> int:
```

The algorithm:
1. **Remove** the IDs from their current position in `queue_order`
2. **Resolve** the target insertion point:
   - `"top"` → index 0
   - `"bottom"` → end of list
   - `"before"/"after"` + target ID → relative to specific item
   - `"up"/"down"` + offset → move by N positions
3. **Insert** the IDs at the resolved position in original order
4. **Remove from skipped** if present (unskip when moving back to active)

**Key Feature**: Moving items automatically **removes them from skipped**, bringing them back to active work.

### 3. HOW NEW ITEMS FROM SCAN ENTER THE QUEUE (RECONCILIATION)

**Entry Point**: `desloppify/app/commands/scan/scan_workflow.py::_reconcile_plan_post_scan()` (called after every scan merge at line 600)

The reconciliation pipeline runs these in order:

#### Step A: `reconcile_plan_after_scan()` (`desloppify/engine/_plan/reconcile.py`)
- **Finds dead findings** — checks all IDs in `queue_order`, `skipped`, clusters, overrides
- **Moves disappeared IDs to superseded dict** (best-effort, keeps them for history)
- **Removes them from queue_order/clusters** 
- **Resurfaces stale temporary skips** — if `review_after` scan count reached, returns to queue_order
- **Prunes old superseded** — deletes entries >90 days old
- **Logs reconciliation event**

#### Step B: `sync_unscored_dimensions()` (`desloppify/engine/_plan/stale_dimensions.py`)
- **Prepends** unscored (placeholder) subjective dimension IDs to **front** of queue unconditionally
- These are onboarding/initial reviews — highest priority
- Format: `subjective::<slug>`

#### Step C: `sync_stale_dimensions()` (`desloppify/engine/_plan/stale_dimensions.py`)
- **Appends** stale (previously-scored) subjective dimensions to **back** of queue **only** when no objective items remain
- Prevents stale subjective work from starving objective fixes

#### Step D: `auto_cluster_findings()` (`desloppify/engine/_plan/auto_cluster.py`)
- **Regenerates automatic clusters** from open, non-suppressed findings
- Groups by deterministic keys (detector, dimension, file, subtype)
- **Preserves user-modified auto-clusters** — adds new findings but doesn't replace manual edits
- **Deletes stale auto-clusters** — clusters with no living members
- Updates `overrides` to track cluster membership
- Handles special cases: `auto/initial-review` (unscored dimensions) and `auto/stale-review`

#### Step E: `sync_synthesis_needed()`
- **Injects `synthesis::pending`** at front of queue when review findings change
- Never auto-prunes — only explicit completion removes it

**Protection for Manually-Prioritized Items**: 
- ✅ Reconciliation happens **first**, clearing dead findings
- ✅ New subjective items prepend to **front** (unscored) or **append to back** (stale, conditional)
- ✅ Auto-clustering operates on open findings only, **doesn't reorder existing queue**
- ✅ Existing `queue_order` items **stay in their current positions** after reconciliation (reconciliation only removes dead IDs)
- ❌ **NO EXPLICIT PROTECTION** — if user hasn't moved items yet, new findings could still appear in ranking order between manually-prioritized items after `_apply_plan_order()` applies the ordering

### 4. HOW CLUSTERS WORK & ARE ADDED

**Cluster Types**:
- **Manual clusters** (`user_modified: False`) — created by user
- **Auto clusters** (`auto: True`, name starts with `auto/`) — regenerated every scan
  - `auto/*` — groups of open findings by detector/dimension/file
  - `auto/initial-review` — unscored subjective dimensions (min size 1)
  - `auto/stale-review` — stale subjective dimensions (min size 2)
- **Synthesis clusters** (name starts with `epic/`) — multi-stage epics for structured refactoring

**New Cluster Operations** (`desloppify/engine/_plan/operations.py`):
```python

```

**Auto-Clustering Algorithm** (`auto_cluster.py::auto_cluster_findings()`):
1. **Collect** open, non-suppressed findings
2. **Group by deterministic key**:
   - Review findings → `review::{dimension}`
   - Per-file detectors → `file::{detector}::{basename}`
   - Typed detectors → `typed::{detector}::{subtype}`
   - Pure auto-fix → `auto::{detector}`
   - Everything else → `detector::{detector}`
3. **Filter singletons** (min cluster size 2)
4. **Map to existing clusters** by `cluster_key`
5. **Update or create**:
   - If exists + not user-modified → replace membership
   - If exists + user-modified → merge new findings in, keep manual edits
   - If new → create cluster with generated description/action
6. **Update overrides** — each finding's override gets `cluster: cluster_name`
7. **Delete stale clusters** — those with no active group key and not user-modified

### 5. HOW `_apply_plan_order()` REORDERS ITEMS

**Location**: `desloppify/engine/_work_queue/core.py::_apply_plan_order()`

**Algorithm** (lines 100–186):
1. **Annotate** all items with plan metadata:
   - `plan_description` (from override)
   - `plan_note` (from override)
   - `plan_cluster` (from override cluster + cluster data)
2. **Split items into three groups**:
   - `ordered` — items in `queue_order` not in skipped (in that order)
   - `remaining` — items not in `queue_order` and not skipped (mechanical sort preserved)
   - `skipped_items` — items in skipped dict
3. **Concatenate**: `ordered + remaining [+ skipped_items if include_skipped]`
4. **Assign queue positions** starting at 1
5. **Mark skipped items** with `plan_skipped: True`, `plan_skip_kind`, reason
6. **Filter to cluster** if requested (keep only cluster members)

**Key Invariant**:
- Items in `queue_order` appear **first** in that exact order
- Remaining items keep their **mechanical ranking** (tier, dimension health, etc.)
- `queue_order` acts as a "priority override" — anything there jumps ahead of automatic ranking

### 7. PROTECTION MECHANISMS FOR MANUALLY-PRIORITIZED ITEMS

**Strong Protections**:
- ✅ `queue_order` items appear first, in that order
- ✅ Reconciliation removes only dead IDs, doesn't reorder
- ✅ Auto-clustering doesn't touch `queue_order` directly
- ✅ Moving items manually stays via `move_items()` → persisted to disk

**Potential Issues**:
- ⚠️ New findings not yet in `queue_order` will be ranked mechanically (could appear before some manually-prioritized items in the "remaining" group)
- ⚠️ If a manually-prioritized finding disappears (e.g., detector fix), it moves to `superseded` and is removed from queue
- ⚠️ Auto-clusters are regenerated every scan — if a user-modified auto-cluster has all members removed, the cluster is deleted

## How the queue works
- `queue_order: list[str]` is the persisted ordering in `plan.json`
- `_apply_plan_order()` in `_work_queue/core.py` renders: `queue_order items first` + `remaining items (not in queue) in mechanical sort`
- `move_items()` in `operations.py` removes IDs from current position and inserts at specified position (top/bottom/before/after/up/down)
- `move_items()` is only called from user-initiated CLI commands; system operations manipulate `queue_order` directly

## Proposed Solution: `promoted_ids` tracking

Add a `promoted_ids` set to the plan that tracks IDs the user has explicitly positioned via `move_items()`. During reconciliation, system insertions go after the last promoted item instead of at position 0.

    57→    from desloppify.engine._work_queue.helpers import slugify
    58→    from desloppify.engine.planning.scorecard_projection import (
    59→        scorecard_subjective_entries,
    60→    )
    61→
    62→    dim_scores = state.get("dimension_scores", {}) or {}
    63→    if not dim_scores:
    64→        return set()

    66→    stale: set[str] = set()
    67→    for entry in scorecard_subjective_entries(state, dim_scores=dim_scores):
    68→        if not entry.get("stale"):
    69→            continue
    70→        dim_key = entry.get("dimension_key", "")

    85→    assessments = state.get("subjective_assessments")

    87→        return set()

    91→        if not isinstance(payload, dict):
    92→            continue
    93→        if not payload.get("placeholder"):

   101→# Unscored dimension sync (front of queue, unconditional)
   102→# ---------------------------------------------------------------------------
   103→
   104→def sync_unscored_dimensions(
   105→    plan: PlanModel,
   106→    state: StateModel,
   107→) -> UnscoredDimensionSyncResult:
   108→    """Keep the plan queue in sync with unscored (placeholder) subjective dimensions.
   109→
   110→    1. **Prune** — remove ``subjective::*`` IDs from ``queue_order`` that are
   111→       no longer unscored AND not stale (avoids pruning stale IDs — that is
   112→       ``sync_stale_dimensions``' responsibility).
   113→    2. **Inject** — unconditionally prepend currently-unscored IDs to the
   114→       *front* of ``queue_order`` so initial reviews are the first priority.
   115→    """
   116→    ensure_plan_defaults(plan)
   117→    result = UnscoredDimensionSyncResult()
   118→    unscored_ids = _current_unscored_ids(state)
   119→    stale_ids = _current_stale_ids(state)
   120→    order: list[str] = plan["queue_order"]
   121→
   122→    # --- Cleanup: prune subjective IDs that are no longer unscored --------
   123→    # Only prune IDs that are neither unscored nor stale (stale sync owns those).
   124→    to_remove: list[str] = [
   125→        fid for fid in order
   126→        if fid.startswith(SUBJECTIVE_PREFIX)
   127→        and fid not in unscored_ids
   128→        and fid not in stale_ids
   129→    ]
   130→    for fid in to_remove:
   131→        order.remove(fid)
   132→        result.pruned.append(fid)
   133→
   134→    # --- Inject: prepend unscored IDs to front ----------------------------
   135→    existing = set(order)
   136→    for uid in reversed(sorted(unscored_ids)):
   137→        if uid not in existing:
   138→            order.insert(0, uid)
   139→            result.injected.append(uid)
   140→
   141→    return result

   145→# Stale dimension sync (back of queue, conditional)
   146→# ---------------------------------------------------------------------------
   147→
   148→def sync_stale_dimensions(
   149→    plan: PlanModel,
   150→    state: StateModel,
   151→) -> StaleDimensionSyncResult:
   152→    """Keep the plan queue in sync with stale subjective dimensions.
   153→
   154→    1. Remove any ``subjective::*`` IDs from ``queue_order`` that are no
   155→       longer stale and not unscored (avoids pruning IDs owned by
   156→       ``sync_unscored_dimensions``).
   157→    2. If no objective items remain after cleanup, inject all currently-stale
   158→       dimension IDs so the plan surfaces them as actionable work.
   159→    """
   160→    ensure_plan_defaults(plan)
   161→    result = StaleDimensionSyncResult()
   162→    stale_ids = _current_stale_ids(state)
   163→    unscored_ids = _current_unscored_ids(state)
   164→    order: list[str] = plan["queue_order"]
   165→
   166→    # --- Cleanup: prune resolved subjective IDs --------------------------
   167→    # Only prune IDs that are neither stale nor unscored.
   168→    to_remove: list[str] = [
   169→        fid for fid in order
   170→        if fid.startswith(SUBJECTIVE_PREFIX)
   171→        and fid not in stale_ids
   172→        and fid not in unscored_ids
   173→    ]
   174→    for fid in to_remove:
   175→        order.remove(fid)
   176→        result.pruned.append(fid)
   177→
   178→    # --- Inject: populate when no objective items remain -----------------
   179→    has_real_items = any(fid for fid in order if not fid.startswith(SUBJECTIVE_PREFIX))
   180→    if not has_real_items and stale_ids:
   181→        existing = set(order)
   182→        for sid in sorted(stale_ids):
   183→            if sid not in existing:
   184→                order.append(sid)
   185→                result.injected.append(sid)
   186→
   187→    return result
   188→

   190→# ---------------------------------------------------------------------------
   191→# Synthesis snapshot hash + sync
   192→# ---------------------------------------------------------------------------
   193→
   194→def review_finding_snapshot_hash(state: StateModel) -> str:
   195→    """Hash open review finding IDs to detect changes.
   196→
   197→    Returns empty string when there are no open review findings.
   198→    """
   199→    findings = state.get("findings", {})
   200→    review_ids = sorted(
   201→        fid for fid, f in findings.items()
   202→        if f.get("status") == "open"
   203→        and f.get("detector") in ("review", "concerns")
   204→    )
   205→    if not review_ids:
   206→        return ""
   207→    return hashlib.sha256("|".join(review_ids).encode()).hexdigest()[:16]
   208→

   210→@dataclass
   211→class SynthesisSyncResult:
   212→    """What changed during a synthesis sync."""
   213→
   214→    injected: bool = False
   215→    pruned: bool = False
   216→
   217→    @property
   218→    def changes(self) -> int:
   219→        return int(self.injected) + int(self.pruned)
   220→
   221→
   222→def sync_synthesis_needed(
   223→    plan: PlanModel,
   224→    state: StateModel,
   225→) -> SynthesisSyncResult:
   226→    """Inject ``synthesis::pending`` at front of queue when review findings change.
   227→
   228→    Never auto-prunes — only explicit completion (``_apply_completion``) removes it.
   229→    """
   230→    ensure_plan_defaults(plan)
   231→    result = SynthesisSyncResult()
   232→    order: list[str] = plan["queue_order"]
   233→    meta = plan.get("epic_synthesis_meta", {})
   234→    already_present = SYNTHESIS_ID in order
   235→
   236→    current_hash = review_finding_snapshot_hash(state)
   237→    last_hash = meta.get("finding_snapshot_hash", "")
   238→
   239→    # If hash changed and not already present, inject at front
   240→    if current_hash and current_hash != last_hash and not already_present:
   241→        order.insert(0, SYNTHESIS_ID)
   242→        result.injected = True
   243→
   244→    # Never auto-prune: synthesis::pending is only removed by explicit completion
   245→

## Implementation Plan: `promoted_ids` tracking to protect user-positioned items

### Problem Summary

When a scan runs, `sync_unscored_dimensions()` and `sync_synthesis_needed()` in `stale_dimensions.py` insert items at position 0 of `queue_order`, displacing items the user has explicitly moved to the top via `plan move ... top`. The fix introduces a `promoted_ids` set that tracks user-promoted items, causing system insertions to go after the last promoted item rather than at the absolute front.

### Design Validation

After reading all the code, the proposed design is sound with a few refinements:

1. **`move_cluster` delegates to `move_items`** (line 431 in operations.py), so promoted tracking in `move_items` automatically covers cluster moves. No separate handling needed in `move_cluster`.

2. **The `reversed(sorted(...))` loop in `sync_unscored_dimensions`** (line 136) inserts multiple items one-by-one at position 0. With the new helper, each insert goes at `_after_promoted(order, plan)`. Since `_after_promoted` re-scans the list each call, and the newly inserted items are NOT in `promoted_ids`, the helper returns the same index each time. This means all unscored items cluster together right after the promoted block, which is the desired behavior. But the iteration order matters: `reversed(sorted(...))` with `insert(idx, ...)` will produce items in `sorted()` order at that position, which is correct.

3. **The `_supersede_id` function in `reconcile.py`** removes items from `queue_order`, `skipped`, and clusters, but does NOT currently touch any other plan fields. It needs to also remove from `promoted_ids`.

4. **`validate_plan`** does not need changes since `promoted_ids` has no cross-field invariants worth enforcing (it is advisory/optimization only).
