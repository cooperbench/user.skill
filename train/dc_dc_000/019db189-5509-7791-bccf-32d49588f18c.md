   ```bash
   python3 - <<'PY'
   import hashlib, os, sys
   VEND = [REDACTED]
   UP   = "/tmp/wan2gp-upstream"
   SKIP_DIRS = {"__pycache__", ".git"}
   SKIP_SUFFIX = (".pyc",)
   def walk(root):
       out = {}
       for dp, dns, fns in os.walk(root):
           dns[:] = [d for d in dns if d not in SKIP_DIRS]
           for fn in fns:
               if fn.endswith(SKIP_SUFFIX): continue
               p = os.path.join(dp, fn)
               rel = os.path.relpath(p, root)
               with open(p, "rb") as f:
                   out[rel] = hashlib.sha256(f.read()).hexdigest()
       return out
   v, u = walk(VEND), walk(UP)
   keys = sorted(set(v) | set(u))
   for k in keys:
       if k not in u:
           print(f"VENDORED_ONLY\t{k}")
       elif k not in v:
           print(f"UPSTREAM_ONLY\t{k}")
       elif v[k] != u[k]:

   PY
   ```
