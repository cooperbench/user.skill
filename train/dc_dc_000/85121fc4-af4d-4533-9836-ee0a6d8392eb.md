   550→def _platform_status(platform: dict) -> str:
   551→    """Return a plain-text status string for a platform.
   552→
   553→    Returns uncolored text so it can safely be embedded in
   554→    simple_term_menu items (ANSI codes break width calculation).
   555→    """
   556→    token_var = platform["token_var"]
   557→    val = get_env_value(token_var)
   558→    if token_var == "WHATSAPP_ENABLED":
   559→        if val and val.lower() == "true":
   560→            session_file = Path.home() / ".hermes" / "whatsapp" / "session" / "creds.json"
   561→            if session_file.exists():
   562→                return "configured + paired"
   563→            return "enabled, not paired"
   564→        return "not configured"
   565→    if platform.get("key") == "signal":
   566→        account = get_env_value("SIGNAL_ACCOUNT")
   567→        if val and account:
   568→            return "configured"
   569→        if val or account:
   570→            return "partially configured"
   571→        return "not configured"
   572→    if platform.get("key") == "email":
   573→        pwd = get_env_value("EMAIL_PASSWORD")
   574→        imap = get_env_value("EMAIL_IMAP_HOST")
   575→        smtp = get_env_value("EMAIL_SMTP_HOST")
   576→        if all([val, pwd, imap, smtp]):
   577→            return "configured"
   578→        if any([val, pwd, imap, smtp]):
   579→            return "partially configured"
   580→        return "not configured"
   581→    if val:
   582→        return "configured"
   583→    return "not configured"

   586→def _setup_standard_platform(platform: dict):
   587→    """Interactive setup for Telegram, Discord, or Slack."""
   588→    emoji = platform["emoji"]
   589→    label = platform["label"]
   590→    token_var = platform["token_var"]
   591→
   592→    print()
   593→    print(color(f"  ─── {emoji} {label} Setup ───", Colors.CYAN))
   594→
   595→    # Show step-by-step setup instructions if this platform has them
   596→    instructions = platform.get("setup_instructions")
   597→    if instructions:
   598→        print()
   599→        for line in instructions:
   600→            print_info(f"  {line}")
   601→
   602→    existing_token = get_env_value(token_var)
   603→    if existing_token:
   604→        print()
   605→        print_success(f"{label} is already configured.")
   606→        if not prompt_yes_no(f"  Reconfigure {label}?", False):
   607→            return
   608→
   609→    allowed_val_set = None  # Track if user set an allowlist (for home channel offer)
   610→
   611→    for var in platform["vars"]:
   612→        print()
   613→        print_info(f"  {var['help']}")
   614→        existing = get_env_value(var["name"])
   615→        if existing and var["name"] != token_var:
   616→            print_info(f"  Current: {existing}")
   617→
   618→        # Allowlist fields get special handling for the deny-by-default security model
   619→        if var.get("is_allowlist"):
   620→            print_info("  The gateway DENIES all users by default for security.")
   621→            print_info("  Enter user IDs to create an allowlist, or leave empty")
   622→            print_info("  and you'll be asked about open access next.")
   623→            value = prompt(f"  {var['prompt']}", password=False)
   624→            if value:
   625→                cleaned = value.replace(" ", "")
   626→                save_env_value(var["name"], cleaned)
   627→                print_success("  Saved — only these users can interact with the bot.")
   628→                allowed_val_set = cleaned
   629→            else:
   630→                # No allowlist — ask about open access vs DM pairing
   631→                print()
   632→                access_choices = [
   633→                    "Enable open access (anyone can message the bot)",
   634→                    "Use DM pairing (unknown users request access, you approve with 'hermes pairing approve')",
   635→                    "Skip for now (bot will deny all users until configured)",
   636→                ]
   637→                access_idx = prompt_choice("  How should unauthorized users be handled?", access_choices, 1)
   638→                if access_idx == 0:
   639→                    save_env_value("GATEWAY_ALLOW_ALL_USERS", "true")
   640→                    print_warning("  Open access enabled — anyone can use your bot!")
   641→                elif access_idx == 1:
   642→                    print_success("  DM pairing mode — users will receive a code to request access.")
   643→                    print_info("  Approve with: hermes pairing approve {platform} {code}")
   644→                else:
   645→                    print_info("  Skipped — configure later with 'hermes gateway setup'")
   646→            continue
   647→
   648→        value = prompt(f"  {var['prompt']}", [REDACTED]("password", False))
   649→        if value:
   650→            save_env_value(var["name"], value)
   651→            print_success(f"  Saved {var['name']}")
   652→        elif var["name"] == token_var:
   653→            print_warning(f"  Skipped — {label} won't work without this.")
   654→            return
   655→        else:
   656→            print_info("  Skipped (can configure later)")
   657→
   658→    # If an allowlist was set and home channel wasn't, offer to reuse
   659→    # the first user ID (common for Telegram DMs).
   660→    home_var = f"{label.upper()}_HOME_CHANNEL"
   661→    home_val = get_env_value(home_var)
   662→    if allowed_val_set and not home_val and label == "Telegram":
   663→        first_id = allowed_val_set.split(",")[0].strip()
   664→        if first_id and prompt_yes_no(f"  Use your user ID ({first_id}) as the home channel?", True):
   665→            save_env_value(home_var, first_id)
   666→            print_success(f"  Home channel set to {first_id}")
   667→
   668→    print()
   669→    print_success(f"{emoji} {label} configured!")

   679→def _is_service_installed() -> bool:
   680→    """Check if the gateway is installed as a system service."""
   681→    if is_linux():
   682→        return get_systemd_unit_path().exists()
   683→    elif is_macos():
   684→        return get_launchd_plist_path().exists()
   685→    return False
   686→
   687→
   688→def _is_service_running() -> bool:
   689→    """Check if the gateway service is currently running."""
   690→    if is_linux() and get_systemd_unit_path().exists():
   691→        result = subprocess.run(
   692→            ["systemctl", "--user", "is-active", SERVICE_NAME],
   693→            capture_output=True, text=True
   694→        )
   695→        return result.stdout.strip() == "active"
   696→    elif is_macos() and get_launchd_plist_path().exists():
   697→        result = subprocess.run(
   698→            ["launchctl", "list", "ai.hermes.gateway"],
   699→            capture_output=True, text=True
   700→        )
   701→        return result.returncode == 0
   702→    # Check for manual processes
   703→    return len(find_gateway_pids()) > 0

   821→def gateway_setup():
   822→    """Interactive setup for messaging platforms + gateway service."""
   823→
   824→    print()
   825→    print(color("┌─────────────────────────────────────────────────────────┐", Colors.MAGENTA))
   826→    print(color("│             ⚕ Gateway Setup                            │", Colors.MAGENTA))

   878→    # ── Post-setup: offer to install/restart gateway ──
   879→    any_configured = any(
   880→        bool(get_env_value(p["token_var"]))
   881→        for p in _PLATFORMS
   882→        if p["key"] != "whatsapp"
   883→    ) or (get_env_value("WHATSAPP_ENABLED") or "").lower() == "true"
   884→
   885→    if any_configured:
   886→        print()
   887→        print(color("─" * 58, Colors.DIM))
   888→        service_installed = _is_service_installed()
   889→        service_running = _is_service_running()
   890→
   891→        if service_running:
   892→            if prompt_yes_no("  Restart the gateway to pick up changes?", True):
   893→                try:
   894→                    if is_linux():
   895→                        systemd_restart()
   896→                    elif is_macos():
   897→                        launchd_restart()
   898→                    else:
   899→                        kill_gateway_processes()
   900→                        print_info("Start manually: hermes gateway")
   901→                except subprocess.CalledProcessError as e:
   902→                    print_error(f"  Restart failed: {e}")
   903→        elif service_installed:
   904→            if prompt_yes_no("  Start the gateway service?", True):
   905→                try:
   906→                    if is_linux():
   907→                        systemd_start()
   908→                    elif is_macos():
   909→                        launchd_start()
   910→                except subprocess.CalledProcessError as e:
   911→                    print_error(f"  Start failed: {e}")
   912→        else:
   913→            print()
   914→            if is_linux() or is_macos():
   915→                platform_name = "systemd" if is_linux() else "launchd"
   916→                if prompt_yes_no(f"  Install the gateway as a {platform_name} service? (runs in background, starts on boot)", True):
   917→                    try:

   925→                            try:
   926→                                if is_linux():
   927→                                    systemd_start()
   928→                                else:
   929→                                    launchd_start()
   930→                            except subprocess.CalledProcessError as e:
   931→                                print_error(f"  Start failed: {e}")
