### Step 3: Create the shared `useOpenFromDataState` helper (`src/shared/components/ui/overlay/useOpenFromDataState.ts`)
**Scope:** Small
1. **Create** the file with:
   ```tsx
   import * as React from 'react';

   export function useOpenFromDataState(node: HTMLElement | null): boolean {
     const [open, setOpen] = React.useState(false);
     React.useEffect(() => {
       if (!node) { setOpen(false); return; }
       const read = () => {
         const state = node.getAttribute('data-state');
         setOpen(state === 'open' || node.hasAttribute('data-open'));
       };
       read();
       const mo = new MutationObserver(read);
       mo.observe(node, { attributes: true, attributeFilter: ['data-state', 'data-open'] });
       return () => mo.disconnect();
     }, [node]);
     return open;
   }
   ```
   The caller holds `node` in `useState` (per D3), so DOM attachment triggers rerender and this effect re-subscribes with the real node.
