136	                "You have finished investigating. Now fill in this JSON template with your findings "
137	                "and output it as your response. Output ONLY the raw JSON, nothing else.\n\n"
138	                + template
139	            )
140	            summary_result = agent.run_conversation(
141	                user_message=summary_prompt,
142	                conversation_history=messages,
143	            )
144	            raw_output = summary_result.get("final_response", "") or ""
145	            messages = summary_result.get("messages", messages)
146	            if raw_output.strip():
147	                print(f"[hermes-worker] Got JSON from template prompt ({len(raw_output)} chars)", file=sys.stderr)
148	        except Exception as exc:
149	            print(f"[hermes-worker] Template prompt failed: {exc}", file=sys.stderr)
150	
151	    # For template-file phases, check the template file FIRST — we told the
152	    # model to write there, so it's the primary output path.
153	    payload = None
154	    if output_path and output_path.exists():
155	        try:
156	            candidate_payload = json.loads(output_path.read_text(encoding="utf-8"))
157	            if isinstance(candidate_payload, dict):
158	                # Check if the model actually filled in findings (not just the empty template)
159	                has_content = _template_has_content(candidate_payload, step)
160	                if has_content:
161	                    payload = candidate_payload
162	                    print(f"[hermes-worker] Read JSON from template file: {output_path}", file=sys.stderr)
163	                else:
164	                    print(f"[hermes-worker] Template file exists but has no real content", file=sys.stderr)
165	        except (json.JSONDecodeError, OSError):
166	            pass
167	
168	    # Try parsing the final text response
169	    if payload is None:
170	        payload = _parse_json_response(raw_output)
171	
172	    # Fallback: some models (GLM-5) put JSON in reasoning/think tags
173	    # instead of content. Just grab it from there.
174	    if payload is None and messages:
175	        payload = _extract_json_from_reasoning(messages)
176	        if payload is not None:
177	            print(f"[hermes-worker] Extracted JSON from reasoning tags", file=sys.stderr)
178	
179	    # Fallback: check all assistant message content fields (not just final_response)
180	    # The model may have output JSON in an earlier message before making more tool calls
181	    if payload is None and messages:
182	        for msg in reversed(messages):
183	            if msg.get("role") != "assistant":
184	                continue
185	            content = msg.get("content", "")
186	            if isinstance(content, str) and content.strip():
187	                payload = _parse_json_response(content)
188	                if payload is not None:
189	                    print(f"[hermes-worker] Extracted JSON from assistant message content", file=sys.stderr)
190	                    break
191	
192	    # Fallback: for execute phase, reconstruct from tool calls + git diff
193	    if payload is None and step == "execute":
194	        payload = _reconstruct_execute_payload(messages, project_dir, plan_dir)
195	        if payload is not None:
196	            print(f"[hermes-worker] Reconstructed execute payload from tool calls", file=sys.stderr)
197	
198	    # Fallback: the model may have written the JSON to a different file location
199	    if payload is None:
200	        schema_filename = STEP_SCHEMA_FILENAMES.get(step, f"{step}.json")
201	        for candidate in [
202	            plan_dir / f"{step}_output.json",  # template file path
203	            project_dir / schema_filename,
204	            plan_dir / schema_filename,
205	            project_dir / f"{step}.json",
206	        ]:
207	            if candidate.exists() and candidate != output_path:  # skip if already checked
208	                try:
209	                    payload = json.loads(candidate.read_text(encoding="utf-8"))
210	                    print(f"[hermes-worker] Read JSON from file written by model: {candidate}", file=sys.stderr)
211	                    break
212	                except (json.JSONDecodeError, OSError):
213	                    pass
214	
215	    # Last resort for template-file phases: the model investigated and produced
216	    # text findings but didn't write valid JSON anywhere. Ask it to restructure
217	    # its analysis into JSON. This catches MiniMax's pattern of outputting markdown.
218	    if payload is None and output_path and messages:
219	        try:
220	            template = _schema_template(schema)
221	            summary_prompt = (
222	                "You have completed your investigation but your findings were not written as JSON. "
223	                "Take everything you found and fill in this JSON template. "
224	                "Output ONLY the raw JSON, nothing else — no markdown, no explanation.\n\n"
225	                + template
226	            )
227	            print(f"[hermes-worker] Attempting summary prompt to extract JSON from investigation", file=sys.stderr)
228	            summary_result = agent.run_conversation(
229	                user_message=summary_prompt,
230	                conversation_history=messages,
231	            )
232	            summary_output = summary_result.get("final_response", "") or ""
233	            if summary_output.strip():
234	                payload = _parse_json_response(summary_output)
235	                if payload is not None:
236	                    print(f"[hermes-worker] Got JSON from summary prompt ({len(summary_output)} chars)", file=sys.stderr)
237	        except Exception as exc:
238	            print(f"[hermes-worker] Summary prompt failed: {exc}", file=sys.stderr)
239	
240	    if payload is None:
241	        raise CliError(
242	            "worker_parse_error",
243	            f"Hermes worker returned invalid JSON for step '{step}': "
244	            f"could not extract JSON from response ({len(raw_output)} chars)",
245	            extra={"raw_output": raw_output},
246	        )
247	
248	    result["final_response"] = raw_output
249	    result["messages"] = messages
250	    return payload, raw_output

255	    # Strip guide-only fields from critique checks (guidance/prior_findings

272	def run_hermes_step(
273	    step: str,
274	    state: PlanState,
275	    plan_dir: Path,
276	    *,
277	    root: Path,
278	    fresh: bool,
279	    model: str | None = None,
280	    prompt_override: str | None = None,
281	) -> WorkerResult:
282	    """Run a megaplan phase using Hermes Agent via OpenRouter.
283	
284	    Structured output is enforced via the prompt (megaplan prompts already
285	    embed the JSON schema). The final response is parsed and validated.
286	    """
287	    if os.getenv(MOCK_ENV_VAR) == "1":
288	        return mock_worker_output(step, state, plan_dir, prompt_override=prompt_override)
289	
290	    from run_agent import AIAgent
291	    from hermes_state import SessionDB
292	
293	    project_dir = Path(state["config"]["project_dir"])
294	    schema_name = STEP_SCHEMA_FILENAMES[step]
295	    schema = read_json(schemas_root(root) / schema_name)
296	    output_path: Path | None = None
297	
298	    # Session management
299	    session_key = session_key_for(step, "hermes", model=model)
300	    session = state["sessions"].get(session_key, {})
301	    session_id = session.get("id") if not fresh else None
302	
303	    # Reload conversation history for session continuity
304	    conversation_history = None
305	    if session_id:
306	        try:
307	            db = SessionDB()
308	            conversation_history = db.get_messages_as_conversation(session_id)
309	        except Exception:
310	            conversation_history = None
311	
312	    # Generate new session ID if needed
313	    if not session_id:
314	        import uuid
315	        session_id = str(uuid.uuid4())
316	
317	    # Build prompt — megaplan prompts embed the JSON schema, but some models
318	    # ignore formatting instructions buried in long prompts.  Append a clear
319	    # reminder so the final response is valid JSON, not markdown.
320	    prompt = prompt_override if prompt_override is not None else create_hermes_prompt(
321	        step, state, plan_dir, root=root
322	    )
323	    # Add web search guidance for phases that have it
324	    if step in ("plan", "critique", "revise"):
325	        prompt += (
326	            "\n\nWEB SEARCH: You have web_search and web_extract tools. "
327	            "If the task involves a framework API you're not certain about — "
328	            "for example a specific Next.js feature, a particular import path, "
329	            "or a config flag that might have changed between versions — "
330	            "search for the current documentation before committing to an approach. "
331	            "Your training data may be outdated for newer framework features."
332	        )
333	    elif step == "execute":
334	        prompt += (
335	            "\n\nWEB SEARCH: You have web_search available. "
336	            "If you encounter an API you're not sure about while coding, "
337	            "search before writing — a quick lookup is cheaper than a build failure."
338	            "\n\nIMPORTANT: Do NOT rename, modify, or delete EVAL.ts or any test files. "
339	            "They are used for scoring after execution and must remain unchanged."
340	        )
341	
342	    # Critique and review: use custom template writers that pre-populate IDs.
343	    # Other template-file phases: hermes_worker writes a generic template.
344	    if step == "critique":
345	        output_path = plan_dir / "critique_output.json"
346	    elif step == "review":
347	        from megaplan.prompts.review import _write_review_template
348	        output_path = _write_review_template(plan_dir, state)
349	        prompt += (
350	            f"\n\nOUTPUT FILE: {output_path}\n"
351	            "This file is your ONLY output. It contains a JSON template PRE-POPULATED with "
352	            "the task IDs and sense-check IDs you must review.\n"
353	            "Workflow:\n"
354	            "1. Read the file to see all the task IDs and sense-check IDs\n"
355	            "2. Investigate each task — cross-reference executor claims against the git diff\n"
356	            "3. Fill in every reviewer_verdict, evidence_files, verdict, criteria, and summary\n"
357	            "4. Write the completed JSON back to the file\n\n"
358	            "CRITICAL: You MUST fill in ALL task_verdicts and sense_check_verdicts entries. "
359	            "Do NOT leave reviewer_verdict or verdict fields empty. "
360	            "Do NOT put your results in a text response. The file is the only output that matters."
361	        )
362	    elif step in _TEMPLATE_FILE_PHASES:
363	        output_path = plan_dir / f"{step}_output.json"
364	        output_path.write_text(
365	            _build_output_template(step, schema),
366	            encoding="utf-8",
367	        )
368	        prompt += (
369	            f"\n\nOUTPUT FILE: {output_path}\n"
370	            "This file is your ONLY output. It contains a JSON template with the structure to fill in.\n"
371	            "Workflow:\n"
372	            "1. Start by reading the file to see the structure\n"
373	            "2. Do your work\n"
374	            "3. Read the file, add your results, write it back\n\n"
375	            "Do NOT put your results in a text response. The file is the only output that matters."
376	        )
377	    else:
378	        template = _schema_template(schema)
379	        prompt += (
380	            "\n\nIMPORTANT: Your final response MUST be a single valid JSON object. "
381	            "Do NOT use markdown. Do NOT wrap in code fences. Output ONLY raw JSON "
382	            "matching this template:\n\n" + template
383	        )
384	
385	    # Redirect stdout to stderr before creating the agent.  Hermes's
386	    # KawaiiSpinner captures sys.stdout at __init__ time and writes
387	    # directly to it (bypassing _print_fn).  By swapping stdout to
388	    # stderr here, all spinner/progress output flows to stderr while
389	    # megaplan keeps stdout clean for structured JSON results.
390	    real_stdout = sys.stdout
391	    sys.stdout = sys.stderr
392	
393	    # Resolve model provider — support direct API providers via prefix
394	    # e.g. "zhipu:glm-5.1" → base_url=Zhipu API, model="glm-5.1"
395	    # Uses the key pool for key rotation and cooldown on 429s.
396	    from megaplan.key_pool import resolve_model as _resolve_model, acquire_key, report_429
397	    resolved_model, agent_kwargs = _resolve_model(model)
398	
399	    toolsets = _toolsets_for_phase(step)
400	
401	    # Disable reasoning/thinking for models that default to reasoning mode.
402	    # When reasoning is enabled, some models (Qwen3.5, DeepSeek-R1) put all
403	    # output in the reasoning field and return content: null.  Megaplan needs
404	    # structured JSON in the content field, so force reasoning off for any
405	    # model family known to use reasoning by default on OpenRouter.
406	    _model_lower = (resolved_model or "").lower()
407	    _reasoning_families = ("qwen/qwen3", "deepseek/deepseek-r1")
408	    _reasoning_off = (
409	        {"enabled": False}
410	        if any(_model_lower.startswith(prefix) for prefix in _reasoning_families)
411	        else None
412	    )
413	
414	    def _make_agent(agent_model: str, extra_kwargs: dict) -> "AIAgent":
415	        current_agent = AIAgent(
416	            model=agent_model,
417	            quiet_mode=True,
418	            skip_context_files=True,
419	            skip_memory=True,
420	            enabled_toolsets=toolsets,
421	            session_id=session_id,
422	            session_db=SessionDB(),
423	            # Cap output tokens to prevent repetition loops (Qwen generates 330K+
424	            # of repeated text without a limit). 8192 is plenty for any megaplan
425	            # phase response. Execute gets more since it may include verbose output.
426	            max_tokens=16384 if step == "execute" else 8192,
427	            reasoning_config=_reasoning_off,
428	            **extra_kwargs,
429	        )
430	        current_agent._print_fn = lambda *a, **kw: print(*a, **kw, file=sys.stderr)
431	        if not toolsets:
432	            current_agent.set_response_format(schema, name=f"megaplan_{step}")
433	        return current_agent
434	
435	    def _rewrite_output_template(current_output_path: Path | None) -> Path | None:
436	        if current_output_path is None:
437	            return None
438	        if step == "critique":
439	            from megaplan._core import configured_robustness
440	            from megaplan.checks import checks_for_robustness
441	            from megaplan.prompts import _write_critique_template
442	
443	            return _write_critique_template(
444	                plan_dir,
445	                state,
446	                checks_for_robustness(configured_robustness(state)),
447	            )
448	        if step == "review":
449	            from megaplan.prompts.review import _write_review_template
450	            return _write_review_template(plan_dir, state)
451	        current_output_path.write_text(
452	            _build_output_template(step, schema),
453	            encoding="utf-8",
454	        )
455	        return current_output_path
456	
457	    def _failure_reason(exc: Exception) -> str:
458	        if isinstance(exc, CliError):
459	            return exc.message
460	        return str(exc) or exc.__class__.__name__
461	
462	    def _run_attempt(current_agent, current_output_path: Path | None) -> tuple[dict, dict, str]:
463	        current_result = current_agent.run_conversation(
464	            user_message=prompt,
465	            conversation_history=conversation_history,
466	        )
467	        current_payload, current_raw_output = parse_agent_output(
468	            current_agent,
469	            current_result,
470	            output_path=current_output_path,
471	            schema=schema,
472	            step=step,
473	            project_dir=project_dir,
474	            plan_dir=plan_dir,
475	        )
476	        clean_parsed_payload(current_payload, schema, step)
477	        messages = current_result.get("messages", [])
478	
479	        try:
480	            validate_payload(step, current_payload)
481	        except CliError as error:
482	            # For execute, try reconstructed payload if validation fails
483	            if step == "execute":
484	                reconstructed = _reconstruct_execute_payload(messages, project_dir, plan_dir)
485	                if reconstructed is not None:
486	                    try:
487	                        validate_payload(step, reconstructed)
488	                        current_payload = reconstructed
489	                        print(
490	                            "[hermes-worker] Using reconstructed payload (original failed validation)",
491	                            file=sys.stderr,
492	                        )
493	                        error = None
494	                    except CliError:
495	                        pass
496	            if error is not None:
497	                raise CliError(error.code, error.message, extra={"raw_output": current_raw_output}) from error
498	
499	        return current_result, current_payload, current_raw_output
500	
501	    agent = _make_agent(resolved_model, agent_kwargs)
502	    # Don't set response_format when tools are enabled — many models
503	    # (Qwen, GLM-5) hang or produce garbage when both are active.
504	    # The JSON template in the prompt is sufficient; _parse_json_response
505	    # handles code fences and markdown wrapping.
506	
507	    # Run — with fallback to OpenRouter for MiniMax if primary API fails
508	    started = time.monotonic()
509	    try:
510	        try:
511	            result, payload, raw_output = _run_attempt(agent, output_path)
512	        except Exception as exc:
513	            # Report 429 to key pool so it cools down this key
514	            exc_str = str(exc)
515	            if "429" in exc_str:
516	                if model and model.startswith("minimax:"):
517	                    report_429("minimax", agent_kwargs.get("api_key", ""), cooldown_secs=60)
518	                elif model and model.startswith("zhipu:"):
519	                    # Quota exhaustion needs a long cooldown (hours, not seconds)
520	                    cooldown = 3600 if "Limit Exhausted" in exc_str else 120
521	                    report_429("zhipu", agent_kwargs.get("api_key", ""), cooldown_secs=cooldown)
522	                    print(f"[hermes-worker] Z.AI key cooled down for {cooldown}s", file=sys.stderr)
523	            if model and model.startswith("minimax:"):
524	                or_key = acquire_key("openrouter")
525	                if or_key:
526	                    if isinstance(exc, CliError):
527	                        print(
528	                            f"[hermes-worker] MiniMax returned bad content ({_failure_reason(exc)}), falling back to OpenRouter",
529	                            file=sys.stderr,
530	                        )
531	                    else:
532	                        print(f"[hermes-worker] MiniMax failed ({exc}), falling back to OpenRouter", file=sys.stderr)
533	                    from megaplan.key_pool import minimax_openrouter_model
534	                    fallback_model = minimax_openrouter_model(model[len("minimax:"):])
535	                    output_path = _rewrite_output_template(output_path)
536	                    agent = _make_agent(
537	                        fallback_model,
538	                        {
539	                            "base_url": "https://openrouter.ai/api/v1",
540	                            "api_key": or_key,
541	                        },
542	                    )
543	                    try:
544	                        result, payload, raw_output = _run_attempt(agent, output_path)
545	                    except Exception as fallback_exc:
546	                        raise CliError(
547	                            "worker_error",
548	                            (
549	                                f"Hermes worker failed for step '{step}' "
550	                                f"(both MiniMax and OpenRouter): primary={_failure_reason(exc)}; "
551	                                f"fallback={_failure_reason(fallback_exc)}"
552	                            ),
553	                            extra={"session_id": session_id},
554	                        ) from fallback_exc
555	                elif isinstance(exc, CliError):
556	                    raise
557	                else:
558	                    raise CliError(
559	                        "worker_error",
560	                        f"Hermes worker failed for step '{step}': {exc}",
561	                        extra={"session_id": session_id},
562	                    ) from exc
563	            elif isinstance(exc, CliError):
564	                raise
565	            else:
566	                raise CliError(
567	                    "worker_error",
568	                    f"Hermes worker failed for step '{step}': {exc}",
569	                    extra={"session_id": session_id},
570	                ) from exc
571	    finally:
572	        sys.stdout = real_stdout
573	    elapsed_ms = int((time.monotonic() - started) * 1000)
574	
575	    cost_usd = result.get("estimated_cost_usd", 0.0) or 0.0
576	    prompt_tokens = int(result.get("prompt_tokens", 0) or 0)
577	    completion_tokens = int(result.get("completion_tokens", 0) or 0)
578	    total_tokens = int(result.get("total_tokens", 0) or 0)
579	
580	    return WorkerResult(
581	        payload=payload,
582	        raw_output=raw_output,
583	        duration_ms=elapsed_ms,
584	        cost_usd=float(cost_usd),
585	        session_id=session_id,
586	        prompt_tokens=prompt_tokens,
587	        completion_tokens=completion_tokens,
588	        total_tokens=total_tokens,
589	    )

592	def _extract_json_from_reasoning(messages: list) -> dict | None:
593	    """Extract JSON from the last assistant message's reasoning field.
594	
595	    Some models (GLM-5) wrap their entire response in think/reasoning tags,
596	    so the content field is empty but reasoning contains valid JSON.
597	    """
598	    for msg in reversed(messages):
599	        if msg.get("role") != "assistant":
600	            continue
601	        for field in ("reasoning", "reasoning_content"):
602	            text = msg.get(field)
603	            if isinstance(text, str) and text.strip():
604	                result = _parse_json_response(text)
605	                if result is not None:
606	                    return result
607	        # Also check reasoning_details (list of dicts with "content")
608	        details = msg.get("reasoning_details")
609	        if isinstance(details, list):
610	            for item in details:
611	                if isinstance(item, dict):
612	                    text = item.get("content", "")
613	                    if isinstance(text, str) and text.strip():
614	                        result = _parse_json_response(text)
615	                        if result is not None:
616	                            return result
617	    return None

620	def _reconstruct_execute_payload(
621	    messages: list,
622	    project_dir: Path,
623	    plan_dir: Path,
624	) -> dict | None:
625	    """Reconstruct an execute phase response from tool calls and git state.
626	
627	    When the model did the work via tools but couldn't produce the JSON
628	    report (e.g., response trapped in think tags, or timeout), build the
629	    response from what actually happened.
630	    """
631	    import subprocess
632	
633	    # Collect tool calls from messages
634	    tool_calls = []
635	    files_changed = set()
636	    commands_run = []
637	    for msg in messages:
638	        if msg.get("role") != "assistant":
639	            continue
640	        for tc in msg.get("tool_calls") or []:
641	            fn = tc.get("function", {})
642	            if not isinstance(fn, dict):
643	                continue
644	            name = fn.get("name", "")
645	            args_raw = fn.get("arguments", "{}")
646	            try:
647	                args = json.loads(args_raw) if isinstance(args_raw, str) else args_raw
648	            except (json.JSONDecodeError, TypeError):
649	                args = {}
650	            if not isinstance(args, dict):
651	                args = {}
652	
653	            tool_calls.append({"name": name, "args": args})
654	
655	            if name in ("write_file", "patch", "edit_file", "apply_patch"):
656	                path = args.get("path", "")
657	                if isinstance(path, str) and path:
658	                    # Make relative to project dir
659	                    try:
660	                        rel = str(Path(path).relative_to(project_dir))
661	                    except ValueError:
662	                        rel = path
663	                    files_changed.add(rel)
664	            elif name in ("terminal", "shell"):
665	                cmd = args.get("command", "")
666	                if isinstance(cmd, str) and cmd:
667	                    commands_run.append(cmd)
668	
669	    # Also check git diff for files changed
670	    try:
671	        diff_result = subprocess.run(
672	            ["git", "diff", "--name-only", "HEAD"],
673	            cwd=project_dir,
674	            capture_output=True, text=True, timeout=10, check=False,
675	        )
676	        if diff_result.returncode == 0:
677	            for line in diff_result.stdout.splitlines():
678	                if line.strip():
679	                    files_changed.add(line.strip())
680	    except Exception:
681	        pass
682	
683	    # Check for untracked files too
684	    try:
685	        status_result = subprocess.run(
686	            ["git", "status", "--porcelain"],
687	            cwd=project_dir,
688	            capture_output=True, text=True, timeout=10, check=False,
689	        )
690	        if status_result.returncode == 0:
691	            for line in status_result.stdout.splitlines():
692	                if line.startswith("?? ") or line.startswith("A  ") or line.startswith("M  "):
693	                    fname = line[3:].strip()
694	                    if fname and not fname.startswith(".megaplan/"):

815	def _parse_json_response(text: str) -> dict | None:
816	    """Extract a JSON object from a model response.
817	
818	    Tries in order:
819	    1. Direct JSON parse
820	    2. Repair common JSON issues (escaped newlines in structural positions)
821	    3. Extract from ```json ... ``` code block
822	    4. Find first { ... } JSON object in the text
823	
824	    Each step also tries the repaired version.
825	    """
826	    text = text.strip()
827	    if not text:
828	        return None
829	
830	    for candidate in [text, _repair_json(text)]:
831	        # Direct parse
832	        try:
833	            parsed = json.loads(candidate)
834	            if isinstance(parsed, dict):
835	                return parsed
836	        except (json.JSONDecodeError, TypeError):
837	            pass
838	
839	        # Extract from code block
840	        import re
841	        code_block = re.search(r'```(?:json)?\s*\n(.*?)\n```', candidate, re.DOTALL)
842	        if code_block:
843	            block_text = code_block.group(1)
844	            for block_candidate in [block_text, _repair_json(block_text)]:
845	                try:
846	                    parsed = json.loads(block_candidate)
847	                    if isinstance(parsed, dict):
848	                        return parsed
849	                except (json.JSONDecodeError, TypeError):
850	                    pass
851	
852	        # Find first JSON object
853	        decoder = json.JSONDecoder()
854	        for i, ch in enumerate(candidate):
855	            if ch != '{':
856	                continue
857	            try:
858	                parsed, end = decoder.raw_decode(candidate[i:])
859	                if isinstance(parsed, dict):
860	                    return parsed
861	            except json.JSONDecodeError:
862	                continue
863	
864	    return None

867	def _repair_json(text: str) -> str:
868	    """Fix common JSON issues from LLM output.
869	
870	    Models sometimes mix escaped and literal newlines, or produce
871	    backslash-n outside of strings where real whitespace is needed.
872	    """
873	    # Replace literal \n that appear outside of JSON strings with actual newlines.
874	    # This handles the case where the model outputs [\n    "item"] instead of
875	    # [\n    "item"] — the \n is structural whitespace, not string content.
876	    result = []
877	    in_string = False
878	    escape = False
879	    i = 0
880	    while i < len(text):
881	        ch = text[i]
882	        if escape:
883	            result.append(ch)
884	            escape = False
885	            i += 1
886	            continue
887	        if ch == '\\' and in_string:
888	            escape = True
889	            result.append(ch)
890	            i += 1
891	            continue
892	        if ch == '"' and not escape:
893	            in_string = not in_string
894	            result.append(ch)
895	            i += 1
896	            continue
897	        # Outside a string, replace \n with actual newline
898	        if not in_string and ch == '\\' and i + 1 < len(text) and text[i + 1] == 'n':
899	            result.append('\n')
900	            i += 2
901	            continue
902	        result.append(ch)
903	        i += 1
904	    return ''.join(result)
