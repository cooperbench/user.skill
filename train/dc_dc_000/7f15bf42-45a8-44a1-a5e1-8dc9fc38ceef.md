   155→    from .codex_runner import _output_file_has_text, run_triage_stage
   156→
   157→    _log = append_run_log or (lambda _msg: None)
   158→
   159→    batches = group_issues_into_observe_batches(si)
   160→    total = len(batches)
   161→    print(colorize(f"\n  Observe: splitting into {total} parallel batches.", "bold"))
   162→    _log(f"observe-parallel batches={total}")
   163→
   164→    # Build prompts and tasks
   165→    tasks: dict[int, object] = {}
   166→    batch_meta: list[tuple[list[str], Path]] = []
   167→
   168→    for i, (dims, issues_subset) in enumerate(batches):
   169→        prompt = build_observe_batch_prompt(
   170→            batch_index=i + 1,
   171→            total_batches=total,
   172→            dimension_group=dims,
   173→            issues_subset=issues_subset,
   174→            repo_root=repo_root,
   175→        )
   176→        prompt_file = prompts_dir / f"observe_batch_{i}.md"
   177→        safe_write_text(prompt_file, prompt)
   178→
   179→        output_file = output_dir / f"observe_batch_{i}.raw.txt"
   180→        log_file = logs_dir / f"observe_batch_{i}.log"
   181→        batch_meta.append((dims, output_file))
   182→
   183→        if not dry_run:
   184→            tasks[i] = partial(
   185→                run_triage_stage,
   186→                prompt=prompt,
   187→                repo_root=repo_root,
   188→                output_file=output_file,
   189→                log_file=log_file,
   190→                timeout_seconds=timeout_seconds,
   191→                validate_output_fn=_output_file_has_text,
   192→            )
   193→
   194→        dim_list = ", ".join(dims)
   195→        print(colorize(f"    Batch {i + 1}: {len(issues_subset)} issues ({dim_list})", "dim"))
   196→        _log(f"observe-batch batch={i + 1} issues={len(issues_subset)} dims={dim_list}")
   197→
   198→    if dry_run:
   199→        print(colorize("  [dry-run] Would execute parallel observe batches.", "dim"))
   200→        return True, ""
   201→
   202→    # Progress callback — same pattern as review batch execution
   203→    def _progress(event: BatchProgressEvent) -> None:
   204→        idx = event.batch_index
   205→        if event.event == "start":
   206→            print(colorize(f"    Observe batch {idx + 1}/{total} started", "dim"))
   207→            _log(f"observe-batch-start batch={idx + 1}")
   208→        elif event.event == "done":
   209→            elapsed = event.details.get("elapsed_seconds", 0) if event.details else 0
   210→            status = "done" if event.code == 0 else f"failed ({event.code})"
   211→            tone = "dim" if event.code == 0 else "yellow"
   212→            print(colorize(f"    Observe batch {idx + 1}/{total} {status} in {int(elapsed)}s", tone))
   213→            _log(f"observe-batch-done batch={idx + 1} code={event.code} elapsed={int(elapsed)}s")
   214→        elif event.event == "heartbeat":
   215→            details = event.details or {}
   216→            active = details.get("active_batches", [])
   217→            elapsed_map = details.get("elapsed_seconds", {})
   218→            if active:
   219→                parts = [f"#{i + 1}:{int(elapsed_map.get(i, 0))}s" for i in active[:6]]
   220→                print(colorize(f"    Observe heartbeat: {len(active)}/{total} active ({', '.join(parts)})", "dim"))
   221→
   222→    def _error_log(batch_index: int, exc: Exception) -> None:
   223→        _log(f"observe-batch-error batch={batch_index + 1} error={exc}")
   224→
   225→    # Execute in parallel
   226→    failures = execute_batches(
   227→        tasks=tasks,
   228→        options=BatchExecutionOptions(run_parallel=True, heartbeat_seconds=15.0),
   229→        progress_fn=_progress,
   230→        error_log_fn=_error_log,
   231→    )
   232→
   233→    if failures:
   234→        print(colorize(
   235→            f"  Observe: {len(failures)} batch(es) failed: {failures}",
   236→            "red",
   237→        ))
   238→        for idx in failures:
   239→            log_file = logs_dir / f"observe_batch_{idx}.log"
   240→            print(colorize(f"    Check log: {log_file}", "dim"))
   241→        _log(f"observe-parallel-failed failures={failures}")
   242→        return False, ""
   243→
   244→    merged = _merge_observe_outputs(batch_meta)
   245→    print(colorize(f"  Observe: merged {total} batch outputs ({len(merged)} chars).", "green"))
   246→    _log(f"observe-parallel-done merged_chars={len(merged)}")
   247→    return True, merged

   524→def _run_sense_check(
   525→    *,
   526→    plan: dict,
   527→    repo_root: Path,
   528→    prompts_dir: Path,
   529→    output_dir: Path,
   530→    logs_dir: Path,
   531→    timeout_seconds: int,
   532→    dry_run: bool = False,
   533→    append_run_log=None,
   534→) -> tuple[bool, str]:
   535→    """Run sense-check via two parallel codex subprocess batches.
   536→
   537→    Returns (ok, merged_report) where ok is True on success.
   538→    Launches content batches (one per cluster) and one structure batch
   539→    in parallel.
   540→    """

   542→
   543→    from desloppify.app.commands.review._runner_parallel_types import BatchProgressEvent
   544→
   545→    from .codex_runner import _output_file_has_text, run_triage_stage
   546→
   547→    _log = append_run_log or (lambda _msg: None)
   548→
   549→    clusters = manual_clusters_with_issues(plan)
   550→    total_content = len(clusters)
   551→    total = total_content + 1  # +1 for structure
   552→    print(colorize(f"\n  Sense-check: {total_content} content batches + 1 structure batch.", "bold"))
   553→    _log(f"sense-check-parallel content_batches={total_content}")
   554→
   555→    tasks: dict[int, object] = {}
   556→    batch_meta: list[tuple[str, Path]] = []  # (label, output_file)
   557→
   558→    # Content batches — one per cluster
   559→    for i, cluster_name in enumerate(clusters):
   560→        prompt = build_sense_check_content_prompt(
   561→            cluster_name=cluster_name,
   562→            plan=plan,
   563→            repo_root=repo_root,
   564→        )
   565→        prompt_file = prompts_dir / f"sense_check_content_{i}.md"
   566→        safe_write_text(prompt_file, prompt)
   567→
   568→        output_file = output_dir / f"sense_check_content_{i}.raw.txt"
   569→        log_file = logs_dir / f"sense_check_content_{i}.log"
   570→        batch_meta.append((f"content:{cluster_name}", output_file))
   571→
   572→        if not dry_run:
   573→            tasks[i] = partial(
   574→                run_triage_stage,
   575→                prompt=prompt,
   576→                repo_root=repo_root,
   577→                output_file=output_file,
   578→                log_file=log_file,
   579→                timeout_seconds=timeout_seconds,
   580→                validate_output_fn=_output_file_has_text,
   581→            )
   582→        print(colorize(f"    Content batch {i + 1}: {cluster_name}", "dim"))
   583→        _log(f"sense-check-content batch={i + 1} cluster={cluster_name}")
   584→
   585→    # Structure batch — single global
   586→    structure_idx = total_content
   587→    structure_prompt = build_sense_check_structure_prompt(
   588→        plan=plan,
   589→        repo_root=repo_root,
   590→    )
   591→    prompt_file = prompts_dir / "sense_check_structure.md"
   592→    safe_write_text(prompt_file, structure_prompt)
   593→
   594→    structure_output = output_dir / "sense_check_structure.raw.txt"
   595→    structure_log = logs_dir / "sense_check_structure.log"
   596→    batch_meta.append(("structure", structure_output))
   597→
   598→    if not dry_run:
   599→        tasks[structure_idx] = partial(
   600→            run_triage_stage,
   601→            prompt=structure_prompt,
   602→            repo_root=repo_root,
   603→            output_file=structure_output,
   604→            log_file=structure_log,
   605→            timeout_seconds=timeout_seconds,
   606→            validate_output_fn=_output_file_has_text,
   607→        )
   608→    print(colorize("    Structure batch: global dependency check", "dim"))
   609→    _log("sense-check-structure batch=global")
   610→
   611→    if dry_run:
   612→        print(colorize("  [dry-run] Would execute parallel sense-check batches.", "dim"))
   613→        return True, ""
   614→
   615→    def _progress(event: BatchProgressEvent) -> None:
   616→        idx = event.batch_index
   617→        label = batch_meta[idx][0] if idx < len(batch_meta) else f"batch-{idx}"
   618→        if event.event == "start":
   619→            print(colorize(f"    Sense-check {label} started", "dim"))
   620→            _log(f"sense-check-batch-start {label}")
   621→        elif event.event == "done":
   622→            elapsed = event.details.get("elapsed_seconds", 0) if event.details else 0
   623→            status = "done" if event.code == 0 else f"failed ({event.code})"
   624→            tone = "dim" if event.code == 0 else "yellow"
   625→            print(colorize(f"    Sense-check {label} {status} in {int(elapsed)}s", tone))
   626→            _log(f"sense-check-batch-done {label} code={event.code} elapsed={int(elapsed)}s")
   627→
   628→    def _error_log(batch_index: int, exc: Exception) -> None:
   629→        _log(f"sense-check-batch-error batch={batch_index} error={exc}")
   630→
   631→    failures = execute_batches(
   632→        tasks=tasks,
   633→        options=BatchExecutionOptions(run_parallel=True, heartbeat_seconds=15.0),
   634→        progress_fn=_progress,
   635→        error_log_fn=_error_log,
   636→    )
   637→
   638→    if failures:
   639→        print(colorize(
   640→            f"  Sense-check: {len(failures)} batch(es) failed: {failures}",
   641→            "red",
   642→        ))
   643→        _log(f"sense-check-parallel-failed failures={failures}")
   644→        return False, ""
   645→
   646→    # Merge outputs
   647→    parts: list[str] = []
