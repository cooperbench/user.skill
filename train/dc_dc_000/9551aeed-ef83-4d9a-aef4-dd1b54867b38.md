326	def _find_step_by_path(plan, path_tuple):
327	    """Walk a TaskPlan to find the step at ``path_tuple`` (descending NestedStep
328	    children). Returns the step or None if the path does not resolve.
329	    """
330	    if not path_tuple:
331	        return None
332	    steps = plan.steps
333	    for segment in path_tuple[:-1]:
334	        match = next((s for s in steps if s.id == segment), None)
335	        if match is None or not isinstance(match, NestedStep):
336	            return None
337	        steps = match.plan.steps
338	    return next((s for s in steps if s.id == path_tuple[-1]), None)

341	def _completed_items_from_events(events, host_path):
342	    """Return the set of item ids that have a completed/attested event under
343	    ``host_path``. ``host_path`` is the STEP_PATH_SEP-joined string form.
344	    """
345	    path_list = host_path.split(STEP_PATH_SEP) if host_path else []
346	    completed: set[str] = set()

348	        if not isinstance(ev, dict):

523	def _summarize_run_dir(run_dir: Path) -> tuple[str, str, str]:
524	    """Return (status, last_event_kind, last_ts) for a run directory.
525	
526	    Per FLAG-P5-006: only ``aborted`` vs ``in-progress`` are reliably
527	    distinguishable in V1. A naturally completed plan still leaves
528	    ``active_run.json`` in place, so the "complete" bucket is mostly
529	    unobservable.
530	    """
531	    events_path = run_dir / "events.jsonl"
532	    if not events_path.is_file():
533	        return "in-progress", "", ""
534	    events = read_events(events_path)
535	    if not events:
536	        return "in-progress", "", ""
537	    last = events[-1]
538	    last_kind = str(last.get("kind", ""))
539	    last_ts = str(last.get("ts", ""))
540	    status = "aborted" if last_kind == "run_aborted" else "in-progress"
541	    return status, last_kind, last_ts
