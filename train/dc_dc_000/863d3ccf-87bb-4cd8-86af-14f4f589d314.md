   201→    if (!resolution || typeof resolution !== 'string' || !resolution.includes('x')) return null;
   202→    const [w, h] = resolution.split('x').map(Number);
   203→    if (!isNaN(w) && !isNaN(h) && w > 0 && h > 0) {
   204→      return { width: w, height: h };
   205→    }
   206→    return null;
   207→  };
   208→
   209→  const aspectRatioToDimensions = (aspectRatio: string): { width: number; height: number } | null => {
   210→    if (!aspectRatio) return null;
   211→    const directResolution = ASPECT_RATIO_TO_RESOLUTION[aspectRatio];
   212→    if (directResolution) return resolutionToDimensions(directResolution);
   213→    const ratio = parseRatio(aspectRatio);
   214→    if (!isNaN(ratio)) {
   215→      const closestAspectRatio = findClosestAspectRatio(ratio);
   216→      const closestResolution = ASPECT_RATIO_TO_RESOLUTION[closestAspectRatio];
   217→      if (closestResolution) return resolutionToDimensions(closestResolution);
   218→    }
   219→    return null;
   220→  };

   222→  const extractDimensionsFromMedia = useCallback((mediaObj: typeof media): { width: number; height: number } | null => {
   223→    if (!mediaObj) return null;
   224→    const params = mediaObj.params as GenerationParams | undefined;
   225→    const metadata = mediaObj.metadata as Record<string, unknown> | undefined;
   226→
   227→    // Check metadata for width/height (these may exist as extended metadata fields)
   228→    if (metadata?.width && metadata?.height && typeof metadata.width === 'number' && typeof metadata.height === 'number') {
   229→      return { width: metadata.width, height: metadata.height };
   230→    }
   231→
   232→    const resolutionSources = [
   233→      params?.resolution,
   234→      (params as Record<string, unknown>)?.originalParams,
   235→      (params as Record<string, unknown>)?.orchestrator_details,
   236→      metadata?.resolution,
   237→      (metadata?.originalParams as Record<string, unknown> | undefined)?.resolution,
   238→      ((metadata?.originalParams as Record<string, unknown> | undefined)?.orchestrator_details as Record<string, unknown> | undefined)?.resolution,
   239→    ];
   240→    for (const res of resolutionSources) {
   241→      if (typeof res === 'string') {
   242→        const dims = resolutionToDimensions(res);
   243→        if (dims) return dims;
   244→      }
   245→    }
   246→
   247→    const aspectRatioSources = [
   248→      params?.aspect_ratio,
   249→      (params as Record<string, unknown>)?.custom_aspect_ratio,
   250→      (params as Record<string, unknown>)?.originalParams,
   251→      (params as Record<string, unknown>)?.orchestrator_details,
   252→      metadata?.aspect_ratio,
   253→      (metadata?.originalParams as Record<string, unknown> | undefined)?.aspect_ratio,
   254→      ((metadata?.originalParams as Record<string, unknown> | undefined)?.orchestrator_details as Record<string, unknown> | undefined)?.aspect_ratio,
   255→    ];
   256→    for (const ar of aspectRatioSources) {
   257→      if (ar && typeof ar === 'string') {
   258→        const dims = aspectRatioToDimensions(ar);
   259→        if (dims) return dims;
   260→      }
   261→    }
   262→
   263→    return null;
   264→  }, []);
