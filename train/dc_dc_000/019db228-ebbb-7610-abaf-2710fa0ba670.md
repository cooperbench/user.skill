### Step 3: `useHeadMeta` hook covering every mutable meta tag in `index.html` (`src/hooks/useHeadMeta.ts`)
**Scope:** Small
1. **Create** `src/hooks/useHeadMeta.ts`:
   ```ts
   type HeadMeta = {
     title: string;
     description: string;
     ogImage?: string;      // falls back to baseline if omitted
     ogUrl?: string;
     twitterImage?: string; // falls back to baseline if omitted
   };
   const SELECTORS = {
     desc: 'meta[name="description"]',
     ogTitle: 'meta[property="og:title"]',
     ogDesc: 'meta[property="og:description"]',
     ogImg: 'meta[property="og:image"]',
     ogUrl: 'meta[property="og:url"]',
     twTitle: 'meta[name="twitter:title"]',
     twDesc: 'meta[name="twitter:description"]',
     twImg: 'meta[name="twitter:image"]',
   } as const;
   export function useHeadMeta({ title, description, ogImage, ogUrl, twitterImage }: HeadMeta) {
     useEffect(() => {
       const prevTitle = document.title;
       const get = (sel: string) =>
         document.head.querySelector<HTMLMetaElement>(sel)?.content;
       const set = (sel: string, v: string | undefined) => {
         if (v === undefined) return;
         const el = document.head.querySelector<HTMLMetaElement>(sel);
         if (el) el.setAttribute('content', v);
       };
       const prev = Object.fromEntries(
         Object.entries(SELECTORS).map(([k, sel]) => [k, get(sel)])
       ) as Record<keyof typeof SELECTORS, string | undefined>;

       document.title = title;
       set(SELECTORS.desc, description);
       set(SELECTORS.ogTitle, title);
       set(SELECTORS.ogDesc, description);
       set(SELECTORS.ogUrl, ogUrl);
       set(SELECTORS.ogImg, ogImage);
       set(SELECTORS.twTitle, title);
       set(SELECTORS.twDesc, description);
       set(SELECTORS.twImg, twitterImage);
       return () => {
         document.title = prevTitle;
         (Object.keys(SELECTORS) as Array<keyof typeof SELECTORS>).forEach((k) => {
           if (prev[k] !== undefined) set(SELECTORS[k], prev[k]);
         });
       };
     }, [title, description, ogImage, ogUrl, twitterImage]);
   }
   ```
   Covers every mutable tag in `index.html` (title, description, og:title/description/image/url, twitter:title/description/image). Omitted fields fall through to whatever the baseline was at mount — no accidental clobbering.
