   162→def format_queue_block(
   163→    breakdown: QueueBreakdown,
   164→    *,
   165→    frozen_score: float | None = None,
   166→    live_score: float | None = None,
   167→) -> list[tuple[str, str]]:
   168→    """Full queue block: focus banner + queue line + contextual hints.
   169→
   170→    Returns a list of ``(text, style)`` pairs ready for ``colorize()``.
   171→    """
   172→    lines: list[tuple[str, str]] = []
   173→
   174→    # Focus banner (prominent, separate)
   175→    if breakdown.focus_cluster:
   176→        focus_line = (
   177→            f"  Focus: `{breakdown.focus_cluster}` "
   178→            f"\u2014 {breakdown.focus_cluster_count}/{breakdown.focus_cluster_total}"
   179→            f" items remaining"
   180→        )
   181→        lines.append((focus_line, "cyan"))
   182→
   183→    # Score line: show both frozen plan-start and live score when available
   184→    if frozen_score is not None:
   185→        if live_score is not None and abs(live_score - frozen_score) >= 0.05:
   186→            delta = round(live_score - frozen_score, 1)
   187→            delta_str = f"{'+' if delta > 0 else ''}{delta:.1f}"
   188→            lines.append((
   189→                f"  Score: strict {live_score:.1f}/100 (plan start: {frozen_score:.1f}, {delta_str})",
   190→                "cyan",
   191→            ))
   192→        else:
   193→            lines.append((
   194→                f"  Score (frozen at plan start): strict {frozen_score:.1f}/100",
   195→                "cyan",
   196→            ))
   197→
   198→    # Queue headline — always the same
   199→    lines.append((f"  {format_queue_headline(breakdown)}", "bold"))
   200→
   201→    # Contextual hints (dim)
   202→    if breakdown.focus_cluster:
   203→        lines.append((
   204→            f"  Unfocus: `desloppify plan focus --clear`"
   205→            f" \u00b7 Cluster details: `desloppify next --cluster {breakdown.focus_cluster} --count 10`",
   206→            "dim",
   207→        ))
   208→    elif breakdown.plan_ordered > 0 or breakdown.skipped > 0:
   209→        lines.append((
   210→            "  Details: `desloppify plan queue`"
   211→            " \u00b7 Skip: `desloppify plan skip <id>`",
   212→            "dim",
   213→        ))
   214→    else:
   215→        lines.append((
   216→            "  Start planning: `desloppify plan`",
   217→            "dim",
   218→        ))
   219→
   220→    return lines
