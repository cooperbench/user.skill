410	def _collect_megaplan_roots(root: Path, *, tree: bool = False, all_system: bool = False) -> list[Path]:
411	    """Collect .megaplan root directories based on search mode."""
412	    roots: list[Path] = [root]
413	
414	    if all_system:
415	        # Search from home directory downward for all .megaplan directories
416	        home = Path.home()
417	        for megaplan_dir in sorted(home.rglob(".megaplan")):
418	            if megaplan_dir.is_dir() and (megaplan_dir / "plans").is_dir():
419	                candidate = megaplan_dir.parent
420	                if candidate.resolve() != root.resolve():
421	                    roots.append(candidate)
422	    elif tree:
423	        # Walk up to find parent .megaplan directories
424	        current = root.resolve().parent
425	        while True:
426	            if (current / ".megaplan" / "plans").is_dir() and current.resolve() != root.resolve():
427	                roots.append(current)
428	            parent = current.parent
429	            if parent == current:
430	                break
431	            current = parent
432	        # Walk down to find child .megaplan directories
433	        for megaplan_dir in sorted(root.rglob(".megaplan")):
434	            if megaplan_dir.is_dir() and (megaplan_dir / "plans").is_dir():
435	                candidate = megaplan_dir.parent
436	                if candidate.resolve() != root.resolve():
437	                    roots.append(candidate)
438	
439	    return roots

442	def handle_list(root: Path, args: argparse.Namespace) -> StepResponse:
443	    ensure_runtime_layout(root)
444	    filter_status = getattr(args, "filter_status", None)
445	    no_tree = getattr(args, "no_tree", False)
446	    include_done = getattr(args, "include_done", False)
447	    show_summary = getattr(args, "summary", False)
448	    search_all = getattr(args, "all", False)
449	    # Default: tree=True (parent+child), active-only (exclude done/aborted)
450	    # --status overrides the active filter (explicit filter = show exactly that)
451	    search_tree = not no_tree and not search_all
452	    filter_active = not include_done and not filter_status
453	
454	    roots = _collect_megaplan_roots(root, tree=search_tree, all_system=search_all)
455	    total_scanned = 0
456	    allowed_states: set[str] | None = None
457	    if filter_status:
458	        allowed_states = {s.strip() for s in filter_status.split(",")}
459	
460	    items = []
461	    state_counts: dict[str, int] = {}
462	    resolved_root = root.resolve()
463	    for search_root in roots:
464	        resolved_search = search_root.resolve()
465	        is_local = resolved_search == resolved_root
466	        for plan_dir in active_plan_dirs(search_root):
467	            state = read_json(plan_dir / "state.json")
468	            current_state = state["current_state"]
469	            state_counts[current_state] = state_counts.get(current_state, 0) + 1
470	            total_scanned += 1
471	
472	            if filter_active and current_state in TERMINAL_STATES:
473	                continue
474	            if allowed_states and current_state not in allowed_states:
475	                continue
476	
477	            next_steps = infer_next_steps(state)
478	            entry = {
479	                "name": state["name"],
480	                "idea": state["idea"],
481	                "state": current_state,
482	                "iteration": state["iteration"],
483	                "next_step": next_steps[0] if next_steps else None,
484	            }
485	            if not is_local:
486	                try:
487	                    rel = resolved_search.relative_to(resolved_root)
488	                    entry["location"] = f"./{rel}"
489	                    entry["direction"] = "child"
490	                except ValueError:
491	                    try:
492	                        resolved_root.relative_to(resolved_search)
493	                        entry["location"] = os.path.relpath(resolved_search, resolved_root)
494	                        entry["direction"] = "parent"
495	                    except ValueError:
496	                        entry["location"] = str(resolved_search)
497	                        entry["direction"] = "external"
498	            items.append(entry)
499	
500	    summary_parts = [f"Found {len(items)} plans"]
501	    if len(roots) > 1:
502	        summary_parts.append(f"across {len(roots)} directories")
503	    if allowed_states:
504	        summary_parts.append(f"matching {','.join(sorted(allowed_states))}")
505	    if filter_active:
506	        summary_parts.append("(active only)")
507	
508	    result: StepResponse = {
509	        "success": True,
510	        "step": "list",
511	        "summary": f"{'. '.join(summary_parts)}.",
512	        "plans": items,
513	    }
514	    if show_summary:
515	        result["state_summary"] = dict(sorted(state_counts.items()))
516	
517	    # Hints for discovering more plans
518	    hidden_done = total_scanned - len(items) if filter_active else 0
519	    hints: list[str] = []
520	    if hidden_done > 0:
521	        hints.append(f"{hidden_done} completed plans hidden (use --include-done to show)")
522	    if not search_all:
523	        hints.append("Use --all to search all plans system-wide")
524	    if hints:
525	        result["hints"] = hints
526	
527	    return result

816	            except ValueError as exc:

820	            raise CliError(
821	                "invalid_args",

823	            )

841	                "action": "profiles",
842	                "profiles_action": "list",
843	                "project_dir": str(project_dir),
844	                "profiles": profiles,
845	            }

850	                "success": True,
851	                "step": "config",
852	                "action": "profiles",
853	                "profiles_action": "show",
854	                "project_dir": str(project_dir),
855	                "name": args.name,
856	                "profile": resolved,
857	            }
858	        raise CliError("invalid_args", f"Unknown profiles action: {profiles_action}")
859	    if action == "reset":
860	        path = config_dir() / "config.json"
861	        if path.exists():
862	            path.unlink()
863	        return {"success": True, "step": "config", "action": "reset", "summary": "Config file removed. Using defaults."}

1169	def _resolve_project_root(args: argparse.Namespace) -> Path:
1170	    """Pick the authoritative project root for handlers that take ``root``.
1171	
1172	    Precedence:
1173	
1174	    1. ``--project-dir`` (when set on *args*) wins. The CLI flag is a deliberate
1175	       caller override; honoring CWD-based discovery here lets a stray
1176	       ``.megaplan/`` in an ancestor directory hijack the run. That's how
1177	       parallel `megaplan init` invocations from sibling worktrees under
1178	       ``~/Documents/.megaplan-worktrees/<exp>/<profile>/`` collide on a
1179	       ``duplicate_plan`` error — the walk-up hits ``~/Documents/.megaplan/``
1180	       and they all try to write into the same plans dir. See
1181	       ``megaplan/bakeoff/orchestrator.py:_init_profile`` for the spawning side.
1182	    2. Otherwise fall back to ``_find_megaplan_root(Path.cwd())`` — the legacy
1183	       behavior that lets ``megaplan plan`` / ``status`` / etc. find the
1184	       enclosing project without an explicit flag.
1185	    """
1186	    project_dir = getattr(args, "project_dir", None)
1187	    if project_dir:
1188	        resolved = Path(project_dir).expanduser().resolve()
1189	        if not resolved.exists() or not resolved.is_dir():
1190	            raise CliError(
1191	                "invalid_project_dir",
1192	                f"--project-dir does not exist or is not a directory: {project_dir}",
1193	            )
1194	        return resolved
1195	    return _find_megaplan_root(Path.cwd())

1198	def _find_megaplan_root(start: Path) -> Path:
1199	    """Walk up from *start* to find the git-root directory containing ``.megaplan/``.
1200	
1201	    Strategy: find the git root first (like ``git rev-parse --show-toplevel``),
1202	    then check if it has a ``.megaplan/`` directory.  This avoids ambiguity when
1203	    nested subdirectories also have their own ``.megaplan/``.  Falls back to the
1204	    nearest ancestor with ``.megaplan/`` if not in a git repo, and finally to
1205	    *start* if nothing is found.
1206	    """
1207	    resolved = start.resolve()
1208	
1209	    # Try git root first — the canonical project root.
1210	    git_root = _find_git_root(resolved)
1211	    if git_root and (git_root / ".megaplan").is_dir():
1212	        return git_root
1213	
1214	    # Fallback: walk up to find nearest .megaplan
1215	    current = resolved
1216	    while True:
1217	        if (current / ".megaplan").is_dir():
1218	            return current
1219	        parent = current.parent
1220	        if parent == current:
1221	            return start
1222	        current = parent

1226	    """Walk up to find the directory containing ``.git``."""

1229	        if (current / ".git").exists():
1230	            return current
1231	        parent = current.parent
1232	        if parent == current:
1233	            return None
1234	        current = parent
