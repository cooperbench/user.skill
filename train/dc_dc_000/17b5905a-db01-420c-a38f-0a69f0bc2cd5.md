   103→    opts = options or QueueBuildOptions()
   104→    plan, scan_path, status, threshold = _resolve_inputs(opts, state)
   105→
   106→    # 1. Gather
   107→    items = build_issue_items(
   108→        state, scan_path=scan_path, status_filter=status,
   109→        scope=opts.scope, chronic=opts.chronic,
   110→    )
   111→    items += _gather_subjective_items(state, opts, threshold)
   112→    items += _gather_workflow_items(state, plan, status)
   113→
   114→    # 2. Score
   115→    enrich_with_impact(items, state.get("dimension_scores", {}))
   116→
   117→    # 3. Plan-aware ordering (part 1: separate skipped items)
   118→    new_ids, skipped = _plan_presort(items, state, plan)
   119→
   120→    # 4. Lifecycle filter — endgame-only items filtered when objective work remains
   121→    items = _apply_lifecycle_filter(items)
   122→
   123→    # 5. Sort & plan post-processing
   124→    items.sort(key=item_sort_key)
   125→    _plan_postsort(items, skipped, plan, opts)
   126→
   127→    # 6. Finalize
   128→    if not items:
   129→        items += _empty_queue_fallback(plan)
   130→    total = len(items)
   131→    if opts.count is not None and opts.count > 0:
   132→        items = items[:opts.count]
   133→    if opts.explain:
   134→        for item in items:
   135→            item["explain"] = item_explain(item)

   137→    return {
   138→        "items": items,
   139→        "total": total,
   140→        "grouped": group_queue_items(items, "item"),
   141→        "new_ids": new_ids,
   142→    }

   297→def _apply_lifecycle_filter(items: list[WorkQueueItem]) -> list[WorkQueueItem]:

   300→    The queue enforces a phase order: scan → initial review → triage → objective work.
   301→
   302→    1. **Initial reviews pending** → only show initial-review subjective items.
   303→       The user must complete first-time reviews before working objective items.
   304→    2. **Triage in progress** → only show triage stages and workflow items.
   305→       Cluster work items aren't ready until triage completes (enrich adds
   306→       the detail that makes steps actionable).
   307→    3. **Objective work remains** → show objective items, hide endgame-only

   313→        # items stay hidden until initial reviews are complete.
   314→        return [
   315→            i for i in items
   316→            if i.get("kind") == "subjective_dimension" and i.get("initial_review")

   320→        # Cluster work items aren't ready until enrich completes.
   321→        return [

   323→            if i.get("kind") in ("workflow_stage", "workflow_action")
   324→        ]
   325→    if not _has_objective_items(items):
   326→        return items  # endgame: everything visible
   327→    return [i for i in items if not _is_endgame_only(i)]
