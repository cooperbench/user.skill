## Overview

DataClaw today is a CLI that (1) discovers agent sessions (claude/codex/gemini/opencode/openclaw/kimi/custom), (2) redacts secrets + anonymizes identifiers, (3) writes a single flat `conversations.jsonl`, and (4) pushes to one HF dataset repo. All persisted state lives in `~/.dataclaw/config.json`.

Phases 1–3 of `docs/expansion-design.md` layer three things onto this flow without changing any existing public surface:

- **Phase 1** — wider config schema (new TypedDict fields + `chmod 0600`), ISO parsing + universal cooled-session filter keyed on *source-file mtime* (including the OpenCode SQLite DB file), keyword-only `cooled_only` / `since` threading in `parse_project_sessions`, and stamping `_source_file` on every session.
- **Phase 2** — a new `export_to_shards(...)` that writes directly into a caller-supplied staging dir (no `export/runs/` intermediate), per-session-id dedup against HF, full HF exception taxonomy (`RepositoryNotFoundError` / `EntryNotFoundError` / `LocalEntryNotFoundError` / 404 `HfHubHTTPError` → "new", other network → raise), an atomic `.dataclaw/manifest.json`, a manifest-scoped `push_shards_to_huggingface`, a `_build_dataset_card_v2`, and a directory-aware `confirm` with a positional `path` argument (old `--file/-f` still honored as alias).
- **Phase 3** — new config verbs for buckets (`--assign/--unassign`, `--default-bucket`, `--tag-project`, `--bucket-by-tag`) and list management (`--set-redact/--remove-redact` pairs for redact / usernames / excluded, plus `--show-secrets`), a canonical `AUTO_MODE_STEPS` constant, full source-choice coverage across every CLI message, and `bucket`/`tags` columns on `list_projects` output.

   COOL_DOWN_SECONDS = 86400

   def _parse_iso(ts: str | None) -> datetime | None:
       """Parse ISO-8601 with 'Z' or offset; return aware UTC datetime or None."""
       if not ts: return None
       try:
           s = ts.replace("Z", "+00:00") if ts.endswith("Z") else ts
           dt = datetime.fromisoformat(s)
           return dt if dt.tzinfo else dt.replace(tzinfo=timezone.utc)
       except (ValueError, TypeError):
           return None

   def _is_cooled_path(p: Path | str | None, *, now: float | None = None) -> bool:
       """True iff the source file's mtime is ≥ 24h old (or file missing → False)."""
       if p is None: return False
       try:
           mtime = Path(p).stat().st_mtime
       except (OSError, TypeError):
           return False
       return ((now if now is not None else datetime.now(tz=timezone.utc).timestamp()) - mtime) >= COOL_DOWN_SECONDS

   MANIFEST_REL = Path(".dataclaw/manifest.json")

   def _project_candidates(session: dict) -> list[str]:
       """Names a session could match against folder-rules lookups."""
       out = []
       if session.get("_project_dir_name"): out.append(session["_project_dir_name"])
       if session.get("project"): out.append(session["project"])
       # Strip source-prefixed display names like "gemini:foo" or "codex:foo"
       p = session.get("project", "")
       if ":" in p: out.append(p.split(":", 1)[1])
       return out

   def _bucket_for_project(session: dict, config: DataClawConfig) -> str | None:
       rules = config.get("folder_rules") or {}
       buckets = rules.get("buckets") or {}
       tag_map = rules.get("buckets_by_tag") or {}
       default = rules.get("default_bucket")
       tags_by_project = config.get("project_tags") or {}
       candidates = _project_candidates(session)
       # 1. Explicit buckets
       for bucket, projects in buckets.items():
           if any(c in projects for c in candidates):
               return bucket
       # 2. Tag fallback
       for c in candidates:
           for tag in tags_by_project.get(c, []):
               if tag in tag_map: return tag_map[tag]
       # 3. Default
       return default

   def _resolve_shard_path(root: Path, session: dict, config: DataClawConfig) -> Path:
       """Returns run_dir-relative shard path: {bucket/}{source}/{YYYY-MM-DD}.jsonl."""
       bucket = _bucket_for_project(session, config)
       src = session.get("source", "claude")
       end = _parse_iso(session.get("end_time"))
       date = (end.date().isoformat() if end else "unknown")
       rel = Path(src) / f"{date}.jsonl"
       return (Path(bucket) / rel) if bucket else rel

       for s in sessions:
