   def check_read_scope(ctx: TurnContext, requested_scope: str) -> str | None:
       """Return None if allowed, else an error message.

       Resolution: requested_scope='own' resolves to ctx.primary_topic_slug.
       The bot's read_scopes.topics may contain the literal token 'own' (a
       bot-relative wildcard meaning 'whatever this bot's primary topic is'),

       if "all" in topics:
           return None
       if requested_scope == "all":
           return "scope_denied: bot lacks 'all' read scope"
       # requested_scope == "own" or an explicit slug
       target_slug = ctx.primary_topic_slug if requested_scope == "own" else requested_scope
       if "own" in topics and target_slug == ctx.primary_topic_slug:

       if target_slug in topics:

           return None
       if slug in topics:
           return None
       return f"write_scope_denied: topic '{slug}' not in write scope"
