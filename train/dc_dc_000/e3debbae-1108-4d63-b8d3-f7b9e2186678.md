   422→def _fixer_help_lines(langs: list[str]) -> list[str]:
   423→    fixer_help_lines: list[str] = []
   424→    for lang_name in langs:
   425→        try:
   426→            fixer_names = sorted(get_lang(lang_name).fixers.keys())
   427→        except (ImportError, ValueError, TypeError, AttributeError):
   428→            fixer_names = []
   429→        fixer_list = ", ".join(fixer_names) if fixer_names else "none yet"
   430→        fixer_help_lines.append(f"fixers ({lang_name}): {fixer_list}")
   431→    fixer_help_lines.append("special: review — prepare structured review data")
   432→    return fixer_help_lines
