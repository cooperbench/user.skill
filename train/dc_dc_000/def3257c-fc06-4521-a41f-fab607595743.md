   - `def normalize_event(event: dict, *, run_dir: str | None = None) -> dict`:

   ```python
       """Execute the plan in replay mode, emitting events to events.jsonl.

       Code steps get stub outputs from fixture_dir/<step-id>/. 
       Attested steps auto-approve. Nested steps recurse.
       """
       run_id = uuid.uuid4().hex
       slug = "test"
       projects_root = run_dir  # <tmp>/  where <tmp>/projects/test/ = project_root
       plan_hash = compute_plan_hash(project_root / "plan.json")

           _replay_step(step, path, slug, run_id, events_path, project_root, projects_root, run_dir, fixture_dir)

       return read_events(events_path)

       path_str = "/".join(path)
       repeat = getattr(step, "repeat", None)

       if isinstance(step, CodeStep):
           append_event(events_path, make_step_dispatched_event(path_str, step.command))
           _simulate_code_step(step, path, slug, run_id, project_root, run_dir, fixture_dir)
           append_event(events_path, make_step_completed_event(path_str, 0))

               _replay_step(child_step, child_full_path, slug, run_id, events_path, project_root, projects_root, run_dir, fixture_dir)
           append_event(events_path, make_nested_exited_event(path_str, 0))
   ```
