333	    """Walk a TaskPlan to find the step at ``path_tuple`` (descending NestedStep
334	    children). Returns the step or None if the path does not resolve.
335	    """
336	    if not path_tuple:
337	        return None
338	    steps = plan.steps

341	        if match is None or not isinstance(match, NestedStep):
342	            return None
343	        steps = match.plan.steps

128	    """Replay ``events.jsonl`` left-to-right to reconstruct the path-stack cursor.

382	                # Defensive: _finalize_cursor should have popped these already.
383	                cursor.frames.pop()
384	                cursor.frames[-1].child_index += 1
385	                continue
386	            exit_path_str = STEP_PATH_SEP.join(top.path_prefix)
387	            append_fn(make_nested_exited_event(exit_path_str, 0))
388	            cursor.frames.pop()
389	            cursor.frames[-1].child_index += 1
390	            continue
391	        top = cursor.frames[-1]
392	        current_step = top.plan.steps[top.child_index]
393	        current_path = top.path_prefix + (current_step.id,)
394	        path_str = STEP_PATH_SEP.join(current_path)
395	        repeat = getattr(current_step, "repeat", None)
396	        in_repeat_frame = top.repeat_step_id is not None
397	        if repeat is not None and not in_repeat_frame:
398	            if isinstance(repeat, RepeatUntil):
399	                _enter_repeat_until(
400	                    slug=slug,
401	                    cursor=cursor,
402	                    host=current_step,
403	                    repeat=repeat,
404	                    path_str=path_str,
405	                    parent_prefix=top.path_prefix,
406	                    events=events_view,
407	                    append_fn=append_fn,
408	                )
409	                continue
410	            if isinstance(repeat, RepeatForEach):
411	                _enter_repeat_for_each(
412	                    slug=slug,
413	                    cursor=cursor,
414	                    host=current_step,
415	                    repeat=repeat,
416	                    path_str=path_str,
417	                    parent_prefix=top.path_prefix,
418	                    events=events_view,
419	                    append_fn=append_fn,
420	                    project_root=project_root,
421	                    run_id=run_id,
422	                    incoming_command=incoming_command,
423	                )
424	                continue
425	        if isinstance(current_step, NestedStep):
426	            child_hash = _compute_inline_plan_hash(current_step.plan)
427	            append_fn(make_nested_entered_event(path_str, child_hash))
428	            cursor.frames.append(
429	                _Frame(plan=current_step.plan, path_prefix=current_path, child_index=0)
430	            )
431	            continue
432	        return current_step, current_path

435	@dataclass(frozen=True)
436	class PeekResult:
437	    """Read-only view of the next dispatchable step under the current cursor.
438	
439	    Returned by ``peek_current_step`` for ``cmd_next`` / ``cmd_status`` /
440	    ``cmd_ack`` to inspect what the gate would dispatch on next without
441	    actually mutating ``events.jsonl``. ``exhausted=True`` covers both
442	    ``at_root_done`` (plan complete) and ``pinned_failure``
443	    (repeat.until on_exhaust=fail).
444	    """
445	
446	    step: Any
447	    path_tuple: tuple[str, ...]
448	    iteration: int | None
449	    item_id: str | None
450	    exhausted: bool
451	
452	
453	def peek_current_step(
454	    plan: TaskPlan,
455	    events: Sequence[dict[str, Any]],
456	    slug: str,
457	    *,
458	    project_root: Path,
459	    run_id: str,
460	) -> PeekResult:
461	    """Walk the cursor exactly the way the gate would, but with a list-capturing
462	    ``append_fn`` so ``events.jsonl`` is never mutated.
463	
464	    Shares ``_auto_traverse_to_leaf`` with ``gate_command`` so peek and dispatch
465	    cannot drift on iteration / for_each / nested transitions (FLAG-P5-003).
466	    The captured events are kept in ``events_view`` so prior-event scans inside
467	    the auto-traverse helpers (``_count_iteration_failed``, the
468	    ``for_each_expanded`` lookup) see them; after every append we let the helper
469	    proceed and re-evaluate the cursor — which is equivalent to recomputing
470	    ``derive_cursor(plan, events + captured, slug=slug)`` because the helper
471	    performs the same frame mutations that ``derive_cursor`` would on replay.
472	    """
473	    cursor = derive_cursor(plan, events, slug=slug)
474	    if cursor.pinned_failure is not None or cursor.at_root_done:
475	        return PeekResult(step=None, path_tuple=(), iteration=None, item_id=None, exhausted=True)
476	
477	    events_view = list(events)
478	    captured: list[dict[str, Any]] = []
479	
480	    def _peek_append(ev: dict[str, Any]) -> None:
481	        captured.append(ev)
482	        events_view.append(ev)
483	
484	    leaf = _auto_traverse_to_leaf(
485	        slug=slug,
486	        cursor=cursor,
487	        events_view=events_view,
488	        incoming_command="",
489	        project_root=project_root,
490	        run_id=run_id,
491	        append_fn=_peek_append,
492	        raise_on_exhausted=False,
493	    )
494	    if leaf is None:
495	        return PeekResult(step=None, path_tuple=(), iteration=None, item_id=None, exhausted=True)
496	    step, path_tuple = leaf
497	    top = cursor.frames[-1]
498	    iteration = top.iteration if top.repeat_step_id is not None else None
499	    item_id = top.item_id if top.repeat_step_id is not None else None
500	    return PeekResult(
501	        step=step,
502	        path_tuple=path_tuple,
503	        iteration=iteration,
504	        item_id=item_id,
505	        exhausted=False,
506	    )

509	def gate_command(
510	    slug: str,
511	    command: str,
512	    argv: Sequence[str],
513	    *,
514	    root: str | Path | None = None,
515	    reentry: bool = False,
516	) -> GateDecision:
517	    active_run = read_active_run(slug, root=root)
518	    if active_run is None:
519	        if not is_in_task_run(slug):
520	            return GateDecision(active=False)
521	        _reject(slug, "active_run.json is missing", abort=True)
522	
523	    project_root = project_dir(slug, root=root)
524	    plan_path = project_root / "plan.json"
525	    plan_hash = compute_plan_hash(plan_path)
526	    if plan_hash != active_run["plan_hash"]:
527	        _reject(slug, "plan.json hash does not match active_run.json pin", abort=True)
528	
529	    run_id = active_run["run_id"]
530	    events_path = project_root / "runs" / run_id / "events.jsonl"
531	    ok, _last_index, error = verify_chain(events_path)
532	    if not ok:
533	        _reject(slug, error or "events.jsonl chain integrity check failed", abort=True)
534	
535	    plan = load_plan(plan_path)
536	    events = read_events(events_path)
537	    cursor = derive_cursor(plan, events, slug=slug)
538	    if cursor.pinned_failure is not None:
539	        reason, _host_path = cursor.pinned_failure
540	        raise TaskRunGateError(reason=reason, recovery=f"artagents abort --project {slug}")
541	    run_started_actor = _find_run_started_actor(events)
542	
543	    # Auto-traverse: nested_entered/exited for nested plans; iteration_started/
544	    # for_each_expanded/item_started for repeat hosts. We loop until we land on a
545	    # dispatchable leaf (CodeStep or AttestedStep) inside the appropriate frame.
546	    events_view = list(events)
547	
548	    def _gate_append(ev: dict[str, Any]) -> None:
549	        append_event(events_path, ev)
550	        events_view.append(ev)
551	
552	    leaf = _auto_traverse_to_leaf(
553	        slug=slug,
554	        cursor=cursor,
555	        events_view=events_view,
556	        incoming_command=command,
557	        project_root=project_root,
558	        run_id=run_id,
559	        append_fn=_gate_append,
560	        raise_on_exhausted=True,
561	    )
562	    if leaf is None:
563	        # Defensive: raise_on_exhausted=True should always raise inside the helper.
564	        _reject(slug, "plan is exhausted", abort=True)
565	    current_step, current_path = leaf
566	    top = cursor.frames[-1]
567	    path_str = STEP_PATH_SEP.join(current_path)
568	
569	    iteration = top.iteration if top.repeat_step_id is not None else None
570	    item_id = top.item_id if top.repeat_step_id is not None else None
571	
572	    if isinstance(current_step, CodeStep):
573	        return _dispatch_code(
574	            slug=slug,
575	            command=command,
576	            step=current_step,
577	            path_str=path_str,
578	            path_tuple=current_path,
579	            events_path=events_path,
580	            run_id=run_id,
581	            reentry=reentry,
582	            project_root=project_root,
583	            iteration=iteration,
584	            item_id=item_id,
585	        )
586	    if isinstance(current_step, AttestedStep):
587	        return _dispatch_attested(
588	            slug=slug,
589	            command=command,
590	            step=current_step,
591	            path_str=path_str,
592	            path_tuple=current_path,
593	            events_path=events_path,
594	            run_id=run_id,
595	            run_started_actor=run_started_actor,
596	            project_root=project_root,
597	            iteration=iteration,
598	            item_id=item_id,
599	        )
600	    raise TaskRunGateError(
601	        reason=f"unexpected step kind: {type(current_step).__name__}",
602	        recovery=f"artagents next --project {slug}",
603	    )

641	    *,
642	    slug: str,
643	    cursor: CursorPath,
644	    host: Any,
645	    repeat: RepeatUntil,
646	    path_str: str,
647	    parent_prefix: tuple[str, ...],
648	    events: Sequence[dict[str, Any]],
649	    append_fn: Callable[[dict[str, Any]], Any],
650	) -> None:
651	    failed = _count_iteration_failed(events, path_str)
652	    iteration = failed + 1
653	    path_tuple = parent_prefix + (host.id,)
654	    if iteration > repeat.max_iterations:
655	        existing = _has_iteration_exhausted(events, path_str)
656	        if existing is None:
657	            append_fn(
658	                make_iteration_exhausted_event(

665	            raise TaskRunGateError(
666	                reason="repeat.until max_iterations exhausted",
667	                recovery=f"artagents abort --project {slug}",
668	            )
669	        # escalate: park on a synthetic exhaust-override attested step.
670	        override_step = _make_exhaust_override_step(slug, path_str)
671	        override_plan = TaskPlan(
672	            plan_id=f"__exhaust_{host.id}",
673	            version=1,
674	            steps=(override_step,),

676	        cursor.frames.append(
677	            _Frame(
678	                plan=override_plan,
679	                path_prefix=path_tuple,
680	                child_index=0,
681	                repeat_step_id=host.id,
682	            )
683	        )
684	        return
685	    append_fn(make_iteration_started_event(path_tuple, iteration))
686	    cursor.frames.append(_make_iteration_frame(host, parent_prefix, iteration))

689	def _resolve_for_each_items(
690	    *,
691	    slug: str,
692	    repeat: RepeatForEach,
693	    project_root: Path,
694	    run_id: str,
695	) -> tuple[str, ...]:
696	    if repeat.items_source == "static":
697	        items = repeat.items
698	    else:
699	        target_id, produces_name = parse_from_ref(repeat.from_ref or "")
700	        prior_step_dir = step_dir_for_path(slug, run_id, (target_id,), root=project_root.parent)
701	        # Find the prior step's declared produces path. We need the plan for that.
702	        # We re-load plan.json which is allowed at gate_command (not in derive_cursor).
703	        plan = load_plan(project_root / "plan.json")
704	        target_step = next((s for s in plan.steps if s.id == target_id), None)
705	        if target_step is None:
706	            raise TaskRunGateError(
707	                reason=f"for_each.from references unknown sibling step {target_id!r}",
708	                recovery=f"artagents abort --project {slug}",
709	            )
710	        produces_entry = next((p for p in target_step.produces if p.name == produces_name), None)
711	        if produces_entry is None:
712	            raise TaskRunGateError(
713	                reason=f"for_each.from references unknown produces {produces_name!r}",
714	                recovery=f"artagents abort --project {slug}",
715	            )
716	        try:
717	            payload = json.loads((prior_step_dir / produces_entry.path).read_text(encoding="utf-8"))
718	        except (FileNotFoundError, json.JSONDecodeError, OSError) as exc:
719	            raise TaskRunGateError(
720	                reason=f"for_each.from cannot read produces JSON: {exc}",
721	                recovery=f"artagents abort --project {slug}",
722	            ) from exc
723	        if not isinstance(payload, list):
724	            raise TaskRunGateError(reason="for_each items must be unique strings", recovery=f"artagents next --project {slug}")
725	        items = tuple(payload)
726	    if not all(isinstance(x, str) and x for x in items):
727	        raise TaskRunGateError(reason="for_each items must be unique strings", recovery=f"artagents next --project {slug}")
728	    if len(set(items)) != len(items):
729	        raise TaskRunGateError(reason="for_each items must be unique strings", recovery=f"artagents next --project {slug}")
730	    return items

733	def _enter_repeat_for_each(
734	    *,
735	    slug: str,
736	    cursor: CursorPath,
737	    host: Any,
738	    repeat: RepeatForEach,
739	    path_str: str,
740	    parent_prefix: tuple[str, ...],
741	    events: Sequence[dict[str, Any]],
742	    append_fn: Callable[[dict[str, Any]], Any],
743	    project_root: Path,
744	    run_id: str,
745	    incoming_command: str,
746	) -> str | None:
747	    # FLAG-P3-004: scan events for an existing for_each_expanded; if absent, append once.
748	    existing = next(
749	        (
750	            ev
751	            for ev in events
752	            if isinstance(ev, dict)
753	            and ev.get("kind") == "for_each_expanded"
754	            and _path_str_from_event(ev) == path_str
755	        ),
756	        None,
757	    )
758	    if existing is None:
759	        items = _resolve_for_each_items(slug=slug, repeat=repeat, project_root=project_root, run_id=run_id)
760	        path_tuple = parent_prefix + (host.id,)
761	        append_fn(make_for_each_expanded_event(path_tuple, items))
762	        cursor.for_each_progress[path_str] = {"items": items, "completed": set()}
763	    else:
764	        items = tuple(existing.get("item_ids") or ())
765	    progress = cursor.for_each_progress.setdefault(path_str, {"items": items, "completed": set()})
766	    completed = progress["completed"]
767	    # For attested host: the incoming command may target a specific item via --item.
768	    target_item: str | None = None
769	    if isinstance(host, AttestedStep):
770	        _, args = match_attested_command(incoming_command, host.command)
771	        if args.item is not None:
772	            target_item = args.item
773	    if target_item is None:
774	        # Pick first not-yet-completed item.
775	        target_item = next((it for it in items if it not in completed), None)
776	    if target_item is None:
777	        # All items done — finalize will pop. Just return.
778	        return None
779	    if target_item not in items:
780	        _reject(slug, f"for_each --item {target_item!r} not in expanded item set", abort=False)
781	    if target_item in completed:
782	        _reject(slug, f"for_each --item {target_item!r} already completed", abort=False)
783	    path_tuple = parent_prefix + (host.id,)
784	    append_fn(make_item_started_event(path_tuple, target_item))
785	    cursor.frames.append(_make_item_frame(host, parent_prefix, target_item))
786	    return target_item

904	    *,
905	    slug: str,
906	    command: str,

1024	def match_attested_command(incoming: str, expected_prefix: str) -> tuple[bool, AttestedArgs]:
1025	    """Strip ``--agent``/``--actor``/``--evidence``/``--item`` (repeatable evidence)
1026	    tokens from ``incoming`` and compare the canonical remainder to ``expected_prefix``.
1027	    """
1028	    try:
1029	        tokens = shlex.split(incoming)
1030	    except ValueError:
1031	        return False, AttestedArgs(agent=None, actor=None, evidence=())
1032	    agent: str | None = None
1033	    actor: str | None = None
1034	    item: str | None = None
1035	    evidence: list[str] = []
1036	    remaining: list[str] = []
1037	    i = 0
1038	    while i < len(tokens):
1039	        token = tokens[i]
1040	        if token == "--agent" and i + 1 < len(tokens):
1041	            agent = tokens[i + 1]
1042	            i += 2
1043	            continue
1044	        if token == "--actor" and i + 1 < len(tokens):
1045	            actor = tokens[i + 1]
1046	            i += 2
1047	            continue
1048	        if token == "--item" and i + 1 < len(tokens):
1049	            item = tokens[i + 1]
1050	            i += 2
1051	            continue
1052	        if token == "--evidence" and i + 1 < len(tokens):
1053	            evidence.append(tokens[i + 1])
1054	            i += 2
1055	            continue
1056	        remaining.append(token)
1057	        i += 1
1058	    rejoined = " ".join(shlex.quote(token) for token in remaining)
1059	    matched = rejoined == expected_prefix
1060	    return matched, AttestedArgs(agent=agent, actor=actor, evidence=tuple(evidence), item=item)

After a `produces_check_passed` is about to fire in the gate, the artifact file is hashed, atomically moved into `.cas/<hash>`, and the original artifact path is replaced by a relative symlink into `.cas/`. The CAS hash is recorded on the `produces_check_passed` event so audits can correlate steps to interned blobs. Scope is strictly per-project (`<project_dir>/.cas/`); there is no shared CAS, matching SD-008 / SD-029.

## Main Phase

### Step 1: Add the CAS module (`artagents/core/task/cas.py`)
**Scope:** Small
1. **Create** `artagents/core/task/cas.py` with:
   - `cas_path(project_dir: Path, sha256: str) -> Path` returning `project_dir / ".cas" / sha256`.
   - `hash_file(path: Path) -> str` streaming the file in 1 MiB chunks through `hashlib.sha256()` and returning the hex digest.
   - `intern(project_dir: Path, source_path: Path) -> Path` that:
     - rejects `source_path` if it is already a symlink resolving inside `project_dir / ".cas"` — return that existing CAS path unchanged (idempotent re-intern).
     - hashes `source_path`, computes `dest = cas_path(project_dir, digest)`,
     - if `dest.exists()`: `source_path.unlink()` and return `dest` (deduplication path),
     - else `dest.parent.mkdir(parents=True, exist_ok=True)` and `os.replace(source_path, dest)` (same-filesystem atomic move because `.cas` lives inside the project), then return `dest`.
   - `link_into_produces(cas_dest: Path, target_path: Path) -> None` that:
     - removes `target_path` if it currently exists (symlink or file) via `target_path.unlink(missing_ok=True)`,
     - ensures `target_path.parent` exists,
     - computes `os.path.relpath(cas_dest, start=target_path.parent)` so links stay valid if the run dir is moved (per design "single-host file-based"),
     - calls `os.symlink(relative_target, target_path)`.
   - `cas_hash_from_symlink(target_path: Path) -> str | None` — if `target_path.is_symlink()` and resolves into `<…>/.cas/<digest>`, return `<digest>`; else `None`. Used for the "already interned on retry" path.
2. **Export** the helpers in `artagents/core/task/cas.py`'s `__all__`.
