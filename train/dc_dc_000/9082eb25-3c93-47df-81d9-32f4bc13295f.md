79	class UploadHeartbeat:
80	    def __init__(self, *, label: str) -> None:
81	        self.label = label
82	        self.files = 0
83	        self.bytes = 0
84	        self.start = time.monotonic()
85	        self.last_log = self.start
86	        self.every_seconds = float(os.getenv("VIBECOMFY_UPLOAD_PROGRESS_SECONDS", str(DEFAULT_UPLOAD_PROGRESS_SECONDS)))
87	        self.every_files = max(1, int(os.getenv("VIBECOMFY_UPLOAD_PROGRESS_FILES", str(DEFAULT_UPLOAD_PROGRESS_FILES))))
88	
89	    def tick(self, *, files: int = 0, bytes_added: int = 0, force: bool = False) -> None:
90	        self.files += files
91	        self.bytes += bytes_added
92	        now = time.monotonic()
93	        if force or self.files % self.every_files == 0 or now - self.last_log >= self.every_seconds:
94	            elapsed = max(now - self.start, 0.001)
95	            print(
96	                f"{self.label}_progress files={self.files} bytes={self.bytes} "
97	                f"size={_format_bytes(self.bytes)} elapsed_seconds={elapsed:.1f}",
98	                flush=True,
99	            )
100	            self.last_log = now

103	class PodGuard:
104	    def __init__(
105	        self,
106	        *,
107	        name_prefix: str,
108	        max_runtime_seconds_env: str = "VIBECOMFY_RUNPOD_MAX_RUNTIME_SECONDS",
109	        default_max_runtime_seconds: int = 7200,
110	    ) -> None:
111	        self.name_prefix = name_prefix
112	        self.max_runtime_seconds_env = max_runtime_seconds_env
113	        self.default_max_runtime_seconds = default_max_runtime_seconds
114	        self.pod = None
115	        self._watchdog: asyncio.Task[None] | None = None
116	
117	    async def launch(self):
118	        load_dotenv, runpod_config, launch = _runpod_lifecycle()
119	        load_dotenv(runpod_lifecycle_root() / ".env")
120	        config_kwargs = _runpod_config_kwargs()
121	        self.pod = await launch(runpod_config.from_env(**config_kwargs), name=f"{self.name_prefix}-{int(time.time())}")
122	        self._start_watchdog()
123	        return self.pod
124	
125	    async def terminate(self) -> None:
126	        if self._watchdog is not None:
127	            self._watchdog.cancel()
128	            self._watchdog = None
129	        if self.pod is not None:
130	            try:
131	                await self.pod.terminate()
132	            except Exception as exc:
133	                if "not found" in str(exc).lower():
134	                    print(f"pod_already_terminated={self.pod.id}", flush=True)
135	                else:
136	                    raise
137	
138	    def _start_watchdog(self) -> None:
139	        seconds = int(os.getenv(self.max_runtime_seconds_env, str(self.default_max_runtime_seconds)))
140	        self._watchdog = asyncio.create_task(self._terminate_after(seconds))
141	
142	    async def _terminate_after(self, seconds: int) -> None:
143	        try:
144	            await asyncio.sleep(seconds)
145	            if self.pod is not None:
146	                print(f"watchdog_terminating_pod={self.pod.id}", flush=True)
147	                await self.pod.terminate()
148	        except asyncio.CancelledError:
149	            return

177	def upload_dir(sftp, local: Path, remote: str, exclude_set: set[str], *, progress: UploadHeartbeat | None = None) -> None:
178	    try:
179	        sftp.mkdir(remote)
180	    except OSError:
181	        pass
182	    for child in local.iterdir():
183	        if should_skip(child, ROOT, exclude_set):
184	            continue
185	        remote_child = posixpath.join(remote, child.name)
186	        if child.is_dir():
187	            upload_dir(sftp, child, remote_child, exclude_set, progress=progress)
188	        else:
189	            size = child.stat().st_size
190	            if progress is not None:
191	                sent = 0
192	
193	                def _progress(current: int, _total: int) -> None:
194	                    nonlocal sent
195	                    delta = max(0, current - sent)
196	                    sent = current
197	                    progress.tick(bytes_added=delta)
198	
199	                sftp.put(str(child), remote_child, callback=_progress)
200	                progress.tick(files=1, bytes_added=max(0, size - sent))
201	            else:
202	                sftp.put(str(child), remote_child)

205	def install_signal_handlers(loop) -> asyncio.Event:
206	    cancel_event = asyncio.Event()
207	    task = asyncio.current_task(loop=loop)
208	    signal_count = 0
209	
210	    def _handle_signal(sig: signal.Signals) -> None:
211	        nonlocal signal_count
212	        signal_count += 1
213	        print(f"interrupt_requested signal={sig.name} cleanup=true", flush=True)
214	        cancel_event.set()
215	        if task is not None and not task.done():
216	            task.cancel()
217	        if signal_count > 1:
218	            print("interrupt_repeated=true cleanup_still_in_progress=true", flush=True)
219	
220	    for sig in (signal.SIGINT, signal.SIGTERM):
221	        try:
222	            loop.add_signal_handler(sig, _handle_signal, sig)
223	        except (NotImplementedError, RuntimeError):
224	            previous = signal.getsignal(sig)
225	
226	            def _fallback(signum, frame, *, previous=previous) -> None:
227	                loop.call_soon_threadsafe(_handle_signal, signal.Signals(signum))
228	                if callable(previous) and previous not in {signal.SIG_DFL, signal.SIG_IGN}:
229	                    previous(signum, frame)
230	
231	            signal.signal(sig, _fallback)
232	    return cancel_event

235	def _clear_current_task_cancellation() -> None:
236	    task = asyncio.current_task()
237	    if task is None or not hasattr(task, "uncancel"):
238	        return
239	    while task.cancelling():
240	        task.uncancel()

244	    remote_script: str,
245	    *,
246	    name_prefix: str,
247	    exclude: set[str],
248	    upload_mode: Literal["sftp_walk", "tarball"] = "sftp_walk",
249	    timeout: int,
250	) -> int:
251	    guard = PodGuard(name_prefix=name_prefix, default_max_runtime_seconds=max(timeout * 2, 7200))
252	    install_signal_handlers(asyncio.get_running_loop())
253	    result_code: int | None = None
254	    try:
255	        _log_phase("launching", f"name_prefix={name_prefix} upload_mode={upload_mode} timeout={timeout}")
256	        pod = await guard.launch()
257	        _log_pod_identity(pod)
258	        _log_phase("waiting_ssh", f"pod_id={pod.id}")
259	        await pod.wait_ready(timeout=300)
260	        ssh_details = await pod._ensure_ssh_details()
261	        print(f"pod_ssh=root@{ssh_details['ip']} -p {ssh_details['port']}", flush=True)
262	        _log_phase("gpu_check", f"pod_id={pod.id}")
263	        code, stdout, stderr = await pod.exec_ssh("nvidia-smi -L", timeout=60)
264	        print(stdout.strip(), flush=True)
265	        if code != 0:
266	            print(stderr, flush=True)
267	            result_code = code
268	            return result_code
269	
270	        if upload_mode == "tarball":
271	            await _upload_tarball(pod, exclude)
272	        else:
273	            _log_phase("uploading", "mode=sftp_walk")
274	            client = pod.open_ssh_client()
275	            try:
276	                sftp = client.open_sftp()
277	                try:
278	                    progress = UploadHeartbeat(label="sftp_upload")
279	                    upload_dir(sftp, ROOT, REMOTE_ROOT, exclude, progress=progress)
280	                    progress.tick(force=True)
281	                finally:
282	                    sftp.close()
283	            finally:
284	                client.close()
285	        print("upload_complete=true", flush=True)
286	
287	        _log_phase("starting_remote", f"timeout={timeout}")
288	        code, stdout, stderr = await pod.exec_ssh(remote_script, timeout=timeout)
289	        print(stdout, flush=True)
290	        if stderr.strip():
291	            print(stderr, flush=True)
292	        result_code = code
293	        return result_code

295	        _clear_current_task_cancellation()
296	        result_code = 130
297	        _log_phase("cancelled", "exit_code=130")
298	        return result_code
299	    finally:
300	        pod_id = getattr(guard.pod, "id", "")
301	        _log_phase("terminating", f"pod_id={pod_id}" if pod_id else "")
302	        await guard.terminate()
303	        print("terminated_launched_pod=true", flush=True)
304	        if result_code is not None:
305	            _log_phase("done", f"exit_code={result_code}")

561	def _parse_tsv(path: Path) -> list[dict[str, str]]:
562	    try:
563	        with path.open("r", encoding="utf-8", newline="") as handle:
564	            reader = csv.DictReader(handle, delimiter="\t")
565	            if not reader.fieldnames:
566	                return []
567	            rows: list[dict[str, str]] = []
568	            for row in reader:
569	                rows.append({str(key): "" if value is None else value for key, value in row.items() if key is not None})
570	            return rows
571	    except OSError:
572	        return []

592	def _png_info(path: Path) -> dict[str, Any] | None:
593	    if path.suffix.lower() != ".png":
594	        return None
595	    try:
596	        with path.open("rb") as handle:
597	            header = handle.read(24)
598	    except OSError:
599	        return None
600	    if len(header) < 24 or header[:8] != b"\x89PNG\r\n\x1a\n" or header[12:16] != b"IHDR":
601	        return None
602	    width, height = struct.unpack(">II", header[16:24])
603	    return {
604	        "width": width,
605	        "height": height,
606	        "format": "PNG",
607	        "mode": None,
608	    }

# Astrid

A file-based toolkit for agents to make art and creative work alongside a human.

Three kinds of beings live here:

- **Executors** — run one concrete unit of work
- **Orchestrators** — coordinate executors (and other orchestrators) into workflows
- **Elements** — reusable render building blocks (effects, animations, transitions)

`python3 -m astrid` is the executable package gateway. Every summons passes through this one gate.

## First commands

Run from the repository root:

```bash
git status --short
python3 -m astrid --help
python3 -m astrid doctor
python3 -m astrid orchestrators list
python3 -m astrid executors list
python3 -m astrid elements list
python3 -m astrid setup
```

`setup` is dry-run by default; pass `--apply` to mutate.

## Using tools

Find an id:

```bash
python3 -m astrid [executors|orchestrators|elements] list
python3 -m astrid [executors|orchestrators|elements] search <terms>
```

If you don't know which tool to use, run `python3 -m astrid <kind> search <terms>` first — don't guess from id alone.

Inspect to see inputs, outputs, and intent:

```bash
python3 -m astrid [executors|orchestrators|elements] inspect <id> --json
```

Run it:

```bash
python3 -m astrid [executors|orchestrators] run <id> -- <args>
```

Each tool has its own `STAGE.md` next to its `run.py`. That is the source of truth — read it before invoking. The JSON inspect output points at the folder root and `stage_file`; load only the one relevant `STAGE.md`, not all of them.

At the start of any session that will produce runs, run python3 -m astrid thread show @active first. The [thread] prefix on every command output is your continuous indicator; if it shows the wrong thread, run thread new or pass --thread @new to your next command. Selections are append-only; the most recent write is authoritative on read; prior selections are preserved as history but do not affect current keepers.

# Seinfeld Scene Generator — Project Plan

A demo of LTX 2.3 that generates Seinfeld-style scenes from a concept. Built on top of a fine-tuned LoRA, agent-written scripts, and Astrid's timeline composition.

## The spine: a shared vocabulary

One thing connects every stage: a **locked vocabulary** of scenes, characters, costumes, and shot types. Captions use it. Training learns it. Script-to-shot decomposition emits it. Inference prompts consume it. Get this right once, in a single element, and every downstream stage stops being a separate consistency problem.

Concretely: ~3 scenes (Jerry's apt, Monk's diner, one more), ~4 characters with 1–2 outfit variants each, ~4 shot types. Small, formulaic, Seinfeld-shaped.

## Four orchestrators in Astrid

### 1. `dataset_build` — produce the training set
Owns the bucket-fill loop. Reads a **criteria element** (bucket targets + per-bucket inclusion rules + caption template + output schemas), runs until full.

Steps, in order, wrapped in `repeat.until(buckets_full)`:
- `code`: YouTube search → candidate URLs
- `code`: yt-dlp video download (extend existing audio util)
- `code`: scene-segment (reuse `packs/builtin/scenes`)
- `code`: cheap pre-filter (length, transcript keywords)
- `attested` (agent): VLM judge → bucket assignment, schema-checked
- `attested` (agent): VLM caption using locked vocabulary, schema-checked
- **`attested` (human): review ~50 random clip+caption pairs** before training kicks off
- `code`: manifest export

### 2. `lora_train` — train the LoRA
Takes the manifest, runs LTX 2.3 training using **[ai-toolkit by ostris](https://github.com/ostris/ai-toolkit)** as the training framework.

Steps:
- `code`: provision a RunPod GPU instance (see "Cross-cutting infra" below — this should use a shared `runpod` executor at the Astrid level, not the RunPod plumbing currently buried inside vibecomfy)
- `code`: stage manifest + ai-toolkit config to the pod
- `code`: kick off training, stream logs back, pull checkpoints
- `code`: generate eval sample grid — same prompts run against baseline LTX and each candidate checkpoint, side-by-side
- **`attested` (human): review the eval grid and pick the winning checkpoint.** This is the "evals" gate — a human looks at samples and selects the LoRA that best hits the vocabulary/style targets. The produced artifact is a `chosen_checkpoint.json` with the path + notes; the gate verifies it before downstream orchestrators can use it.
- `code`: tear down the pod, register the promoted LoRA

### 3. `script_to_shots` — concept → shot list
Two `attested` (agent) steps:
- Kimi K2.6: concept → script (dialogue + scene headers)
- Kimi K2.6: script → shot JSON, **constrained to the locked vocabulary** (only emits scenes/costumes that exist)

The vocabulary constraint is what keeps script generation aligned with what the LoRA can render. Schema-check the shot JSON.

### 4. `scene_render` — shots → final video

The per-shot generation isn't single-shot — it's an **iteration loop with automated quality judging**, using the existing `video_understand` / `visual_understand` executors as judges.

For each shot, wrapped in `repeat.until(passes_judge OR max_attempts)`:
- `code`: LTX+LoRA generates the clip via vibecomfy (`external.vibecomfy.run`), seed-varied per attempt
- `attested` (agent): `video_understand` judges the clip against the shot spec — does it show the right scene, character, costume, action? Does it look like Seinfeld? Returns a structured pass/fail + reasons, schema-checked.
- If fail and attempts remain: re-roll with a refined prompt (judge's reasons inform the refinement) or a new seed
- If max attempts hit: keep the best-scoring attempt and flag it for the human review gate

After all shots are rendered:
- `code`: optional TTS or on-screen text for dialogue
- `code`: assemble clips into timeline composition in order
- **`attested` (human): review final cut, flag any shots to re-render before final render** — the human is mainly looking at the flagged shots from the per-shot loop, plus overall scene coherence
- `code`: render final video

This is the same iteration pattern that ai-tooling generally needs: generate → automated judge → human gate on what the judge couldn't catch. `visual_understand` can be used the same way on extracted keyframes if you want a faster/cheaper first-pass judge before invoking the full `video_understand`.

## How they compose

A top-level `seinfeld_demo` orchestrator chains the four as `nested` steps:

```
dataset_build → lora_train → script_to_shots → scene_render
```

These aren't strictly serial in practice. `script_to_shots` and `scene_render` can be developed and iterated against the *baseline* LTX (no LoRA) before `lora_train` finishes — they only need the vocabulary, not the LoRA itself. That's the parallelization you want.

## Astrid primitives recap

- **Executor** = one unit of work (one network call, one transformation, one artifact in / one artifact out). The capability.
- **Orchestrator** = composes executors into a workflow. Owns loop logic, conditional branching, retries-across-stages.
- **Element** = reusable schema or render building block. The criteria spec and caption template live here.

At task-run time, every step in a plan is one of:

| Step kind | Who does it | How it advances |
|---|---|---|
| `code` | nobody — deterministic argv subprocess | runs, gate verifies `produces`, moves on |
| `attested` | an agent or a human | actor runs the work, acks with identity + evidence, gate verifies `produces` |
| `nested` | another orchestrator | delegates to a child plan |

Human steps are first-class. An `attested` step with `--actor <name>` is a human step: the human does the work, runs the ack, and the gate validates the produced artifact against a schema before the run advances. Self-acks and unpinned actors are rejected.

Loops/fan-out are `repeat.until` / `repeat.for_each` wrappers on any step.

## Cross-cutting infra: port RunPod up to Astrid

RunPod provisioning currently lives inside vibecomfy. For this project (and beyond), it should be lifted up to an Astrid-level shared executor — something like `external.runpod.{provision,run,teardown}` — so any orchestrator can request a GPU pod, run work on it, and tear it down with the same plumbing.

Why now:
- `lora_train` needs RunPod for ai-toolkit training runs
- `scene_render` needs RunPod for LTX+LoRA inference (when not local)
- Future GPU-bound executors (eval grids, larger captioning runs) will need the same
- Keeping it inside vibecomfy couples GPU lifecycle to one framework; making it a peer executor lets vibecomfy *consume* it like everything else

This is a small refactor that pays back across every other workstream. Worth doing early in Phase 1.

## What lives where

- **Astrid**: criteria/vocabulary element, all four orchestrators, RunPod lifecycle executors (newly lifted from vibecomfy), all executors except the LTX/Kimi inference calls themselves
- **vibecomfy**: LTX inference workflow (with and without LoRA), invoked by Astrid executors; consumes the Astrid-level RunPod executor instead of owning that lifecycle
- **ai-toolkit (ostris)**: LTX 2.3 LoRA training, invoked by `lora_train` on a RunPod pod
- **Existing tools to reuse**: `packs/builtin/scenes`, `packs/builtin/video_understand`, `packs/builtin/visual_understand` (both as automated judges in iteration loops), possibly `pool_build` as a model for `dataset_build`, the existing YouTube audio util (extend to video), existing VLM capability

## Sequencing

**Phase 0 (1 day):** Lock the vocabulary. Write 10 example inference prompts you want to work. Reverse-engineer the caption template from those. Cheapest, highest-leverage hour in the project.

**Phase 1 (parallel, ~1 week):**
- Lift RunPod lifecycle out of vibecomfy into a shared Astrid executor (foundational — unblocks training and inference)
- Build `dataset_build` orchestrator and missing executors
- Get baseline LTX 2.3 inference running through vibecomfy (now consuming the Astrid RunPod executor)
- Prototype `script_to_shots` against baseline LTX

**Phase 2 (~1 week):**
- Run `dataset_build` end-to-end, human-review the dataset

## Part C — what the Astrid pack needs to look like

From reading `astrid/core/executor/runner.py`, `astrid/core/orchestrator/runner.py`, `astrid/orchestrate/dsl.py`, `astrid/core/task/plan.py`, the three existing external packs, and `docs/creating-tools.md`:

A **`pod_handle.json`** artifact written by `provision` and consumed by `exec`/`teardown` is the only honest pattern.

2. **Gate/verify schema is `Check`-based, not a fixed manifest shape.** `astrid/verify/checks.py` registers `file_nonempty`, `json_file`, `json_schema`, etc.; each `produces` declares a `(path, check)` pair. 3. **Guaranteed teardown does not exist as an orchestrator primitive.** Reviewed `plan.py` (`on_exhaust: escalate|fail`), `dsl.py`, `gate.py`. There is no `finally` step, no `on_failure` hook, no `cleanup` lane. Three honest options, in order of preference:

4. **Pack layout — one folder, three (or four) executors.** The vibecomfy pack already declares two executors (`external.vibecomfy.run` + `external.vibecomfy.validate`) in a single `executor.yaml` (a JSON array under `executors`). That is the right precedent. Concretely:

5. **`consumes`/`provides` sketch:**

   - `exec`: consumes `pod_handle` + `remote_script` (file or string input) + `local_root`; provides `exec_result` (`json_file` with `exit_code`, `artifact_dir`).
   - `teardown`: consumes `pod_handle`; provides `teardown_receipt`.
   - `session`: consumes `local_root` + `remote_script`; provides `exec_result` + `artifact_dir`. (Internally provisions + execs + tears down — no pod_handle escapes.)

6. **Detached/long-running.** Astrid executors are subprocesses; the parent (orchestrator runner) waits on them. A 4-hour training run is *fine* under this model — the executor just blocks. The Python-side `asyncio` and the polling loop in `run_pod_detached` are independent of Astrid's process model. There's no reason to invent an "async executor" — `session` just runs synchronously from Astrid's POV.

   - For the composite `session` executor, the watchdog *can* live in-process (it's the same model `run_pod` uses today and works because the executor blocks until done).

214	## Sprint 3 — Step model overhaul, mutable plans, location-agnostic execution (2 weeks)
215	
216	**Goal:** simplify the step model; make plans editable mid-run; make execution location-agnostic; add the claim verb.
217	
218	### Decisions
219	
220	- **Step model collapsed.** One step type:
221	  ```
222	  Step {
223	    id: str
224	    version: int          (bumped when a dispatched step is superseded)
225	    adapter: "local" | "manual" | "remote-artifact"
226	    requires_ack: bool
227	    assignee: "system" | "agent:<id>" | "any-agent" | "human:<name>" | "any-human"
228	    produces: [{name, path, check}]
229	    repeat: until | for_each | none
230	    command: str?         (leaf step: the thing to run; interpreted by adapter)
231	    children: [Step]?     (group step: nesting; replaces old `nested` kind)
232	    cost: {amount, currency, source}?  (recorded at completion)
233	    superseded_by: { to_version: int, scope: "all" | "future-iterations" | "future-items" }?
234	  }
235	  ```
236	  **Supersession scope is required, not optional.** A `repeat.until` step with iteration 2 in flight that's superseded with `scope: "future-iterations"` finishes iteration 2 against the old version, then iteration 3+ uses the new version. `scope: "all"` aborts the in-flight iteration and restarts at iteration 1 of the new version. `scope: "future-items"` is the equivalent for `repeat.for_each`. Without an explicit scope, supersede is rejected.
237	  - Leaf step has `command`, no `children`.
238	  - Group step has `children`, no `command`. Its `produces` aggregates from descendants.
239	  - `code` / `attested` / `nested` step kinds are gone.
240	- **Ack identity at ack time, not plan-author time.** `astrid ack <step> --decision {approve,retry,iterate,abort} --agent <id>|--actor <name> [--evidence <path>]`. Both `--agent` and `--actor` are required (one or the other) — no anonymous acks. Every ack also carries the `writer_epoch` it read; stale acks are rejected by the kernel.
241	- **Authoring keeps reusable composition; runtime stays uniform.** Two template kinds at authoring time:
242	  - **Leaf-template** — what we used to call an executor. One concrete unit of work. Same pack layout on disk.
243	  - **Plan-template** — what we used to call an orchestrator: a reusable composition of leaf-templates and other plan-templates. Materializes into a step subtree at plan-build or plan-mutation time.
244	  Runtime never sees templates — only the resulting step tree. This keeps hype-style work compositional without re-introducing a parallel runtime abstraction.
245	- **Template authoring transition (S3 → S4b).** During Sprint 3 the new template kinds land alongside the legacy `OrchestratorDefinition` YAML — both readable. `astrid pack list / inspect` show both kinds with their type. `astrid pack run` accepts both; legacy YAML is materialized into the new step tree on the way in. Sprint 4b removes legacy YAML acceptance once hype + event_talks + thumbnail_maker are all ported. There is no third intermediate format; users write either the legacy YAML (until 4b) or the new template kinds (from S3 onward).
246	- **Plan mutation invariants** — every mutation, before it's accepted, must pass:
247	  - Schema validation on the proposed effective tree (full re-validate, not a diff).
248	  - Sibling-id uniqueness at every frame (the existing rule in `plan.py:_assert_unique_paths`, applied to the effective tree).
249	  - `produces` reference integrity (no dangling refs to deleted/tombstoned steps).
250	  - Adapter validation (declared adapter is one of `local | manual | remote-artifact`, command shape matches adapter).
251	  - Repeat-source validation (`repeat.for_each.from_ref` resolves to a real prior `produces` in the effective tree).
252	  - Lease epoch CAS (the writer's epoch matches current).
253	  Mutations that fail any check are rejected at the verb, not partially applied.
254	- **Plans are mutable, but mutation rules are explicit.** Verbs: `astrid plan add-step`, `astrid plan edit-step`, `astrid plan remove-step`, `astrid plan supersede-step`. Each emits a `plan_mutated` event carrying the diff, the actor, and the `writer_epoch`. The hash chain extends to plan mutations.
255	  - `plan.json` is the *initial* plan, written once at run start.

257	  - **Append-only once dispatched.** A step is "dispatched" the moment `astrid next` returns its payload (event: `step_dispatched`). After dispatch:
