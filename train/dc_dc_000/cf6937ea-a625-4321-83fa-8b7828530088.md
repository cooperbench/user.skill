   593→def generate_concerns(
   594→    state: StateModel,
   595→) -> list[Concern]:
   596→    """Run all concern generators against current state.
   597→
   598→    Returns deduplicated list sorted by (type, file).
   599→    """
   600→    dismissals = state.get("concern_dismissals", {})
   601→    concerns: list[Concern] = []
   602→    seen_fps: set[str] = set()
   603→

   605→        for concern in gen(state, dismissals):

   608→                concerns.append(concern)
   609→

   611→    return concerns

   614→def cleanup_stale_dismissals(state: StateModel) -> int:

   618→    ``source_issue_ids`` (legacy) are left untouched.

   620→    dismissals = state.get("concern_dismissals", {})

   622→        return 0

   625→        fp
