   189→def _natural_sort_key(item: WorkQueueItem) -> tuple:
   190→    """Compute natural (non-plan-aware) ranking for queue items."""
   191→    kind = item.get("kind", "issue")
   192→
   193→    # Initial-review subjective items: highest priority
   194→    if kind == "subjective_dimension" and item.get("initial_review"):
   195→        return (_RANK_INITIAL_REVIEW, 0, subjective_score_value(item), item.get("id", ""))
   196→
   197→    # Triage stage items: stage order, blocked after unblocked
   198→    if kind == "workflow_stage":
   199→        blocked_penalty = 1 if item.get("is_blocked") else 0
   200→        stage_index = _workflow_stage_index(item)
   201→        return (_RANK_TRIAGE_STAGE, blocked_penalty, stage_index, item.get("id", ""))
   202→
   203→    # Workflow action items (e.g. create-plan)
   204→    if kind == "workflow_action":
   205→        return (_RANK_WORKFLOW, 0, 0, item.get("id", ""))
   206→
   207→    if kind == "cluster":
   208→        # Clusters sort before individual issues, ordered by action type
   209→        action_pri = ACTION_TYPE_PRIORITY.get(
   210→            item.get("action_type", "manual_fix"), 3
   211→        )
   212→        return (
   213→            _RANK_CLUSTER,
   214→            action_pri,
   215→            -int(item.get("member_count", 0)),
   216→            item.get("id", ""),
   217→        )
   218→
   219→    impact = item.get("estimated_impact", 0.0)
   220→
   221→    if kind == "subjective_dimension" or item.get("is_subjective"):
   222→        return (
   223→            _RANK_ISSUE,
   224→            -impact,
   225→            subjective_score_value(item),
   226→            item.get("id", ""),
   227→        )
   228→
   229→    detail = detail_dict(item)
   230→    review_weight = float(item.get("review_weight", 0.0) or 0.0)
   231→    return (
   232→        _RANK_ISSUE,
   233→        -impact,
   234→        CONFIDENCE_ORDER.get(item.get("confidence", "low"), 9),
   235→        -review_weight,
   236→        -int(detail.get("count", 0) or 0),

   241→def item_sort_key(item: WorkQueueItem) -> tuple:
   242→    """Unified sort key: plan position first, then natural ranking.
   243→
   244→    When ``_plan_position`` is stamped (by :func:`stamp_plan_sort_keys`),
   245→    planned items sort first in plan order, then existing items by natural
   246→    ranking, then newly-discovered items by natural ranking.
   247→
   248→    When no plan fields are stamped, falls back to pure natural ranking.
   249→    """
   250→    plan_pos = item.get("_plan_position")

   252→    if plan_pos is not None:
   253→        kind = item.get("kind", "issue")
   254→        # Triage stages: even when explicitly planned, maintain the
   255→        # blocked-before-unblocked invariant for correctness.
   256→        if kind == "workflow_stage":
   257→            blocked = 1 if item.get("is_blocked") else 0
   258→            stage_idx = _workflow_stage_index(item)
   259→            return (_TIER_PLANNED, blocked, stage_idx, item.get("id", ""))
   260→        return (_TIER_PLANNED, 0, plan_pos, item.get("id", ""))
   261→

   263→    group = _TIER_NEW if is_new else _TIER_EXISTING
