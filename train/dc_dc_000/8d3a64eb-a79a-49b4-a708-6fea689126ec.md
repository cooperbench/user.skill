def handle_ticket_new(args: argparse.Namespace) -> int:
    """Create a ticket; prints only ULID + newline to stdout on success."""
    body: str = ""
    if getattr(args, "stdin_body", False):
        body = "-"
    elif args.body:
        body = args.body
    elif args.edit:
        body = "-"
        # --edit: open $EDITOR (deferred — body='' is valid for now)
    else:
        print("Error: one of -b BODY, --edit, or - is required.", file=sys.stderr)
        return 1

    tags = args.tags.split(",") if args.tags else None
    if tags:
        tags = [t.strip() for t in tags if t.strip()]

    try:
        _core_new(args.title, body=body, tags=tags)
    except ValueError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 1
    return 0

> AGENT

Short answer: **no real search exists today.** The plumbing only filters by `--status` and `--tags`.

What's there now:
- `megaplan ticket list --status X --tags a,b [--json]` (`megaplan/cli.py:1827`, dispatched to `handle_ticket_list` → `megaplan.tickets.list_tickets` at `megaplan/tickets/core.py:263`).
- Local mode: walks `.megaplan/tickets/*.md` for the current repo only — no cross-repo discovery, no body/title keyword filter.
- Cloud mode: `DBStore.list_tickets` (`megaplan/store/db.py:2039`) filters by `codebase_id`, `status`, `tags` — again no keyword/`ILIKE` search and no "all codebases" mode.

What you can do as a workaround right now:
- Across the machine, local tickets: `grep -rli "keyword" $(find ~ -type d -name "tickets" -path "*/.megaplan/*")` — they're just markdown with frontmatter.
- For cloud DB tickets: ad-hoc SQL against `tickets` (columns include `codebase_id`, `title`, `body`, `tags`, `slug`).
- Inside a single repo / superproject: `megaplan ticket list --json | jq …`, or grep `.megaplan/tickets/`.

If you want, the cleanest additions would be: a `--query` (title/body ILIKE) flag on `list_tickets` in both store backends, plus a `--all-codebases` mode that drops the `codebase_id` filter on the DB side and walks superproject children on the file side. Want me to sketch that?

79	    - Unset                     → ``source = 'human'``.

83	    actor_id = os.environ.get("MEGAPLAN_ACTOR_ID")
84	    if turn_id:
85	        return ("agent", turn_id, actor_id or None)

162	    title: str,
163	    *,
164	    body: str = "",
165	    tags: Sequence[str] | None = None,
166	    store: Store | None = None,
167	    cwd: Path | None = None,
168	) -> str:
169	    """Create a new ticket and return its ULID.
170	
171	    Parameters
172	    ----------
173	    title:
174	        Ticket title (required).
175	    body:
176	        Markdown body.  ``"-"`` means read from stdin (empty input is rejected).
177	    tags:
178	        Optional tags.
179	    store:
180	        Explicit store.  If *None*, resolved from environment.
181	    cwd:
182	        Working directory for git operations.
183	
184	    Returns
185	    -------
186	    str
187	        The ULID of the newly created ticket (printed to stdout).
188	    """
189	    # Handle stdin body
190	    if body == "-":
191	        body = sys.stdin.read()
192	        if not body.strip():
193	            raise ValueError("stdin body is empty — body is required")
194	
195	    if store is None:
196	        store = _resolve_store()
197	
198	    source, turn_id, actor_id = _derive_source()
199	    ticket_id = str(ULID())
200	    slug = slugify(title)
201	    now = datetime.now(timezone.utc)
202	
203	    # Determine codebase_id (cloud mode only)
204	    codebase_id: str | None
205	    if store is not None and is_cloud_store(store):
206	        codebase_id = _ensure_codebase(store, cwd)
207	    else:
208	        codebase_id = None
209	
210	    # Build record dict for file
211	    record: dict[str, Any] = {
212	        "id": ticket_id,
213	        "title": title,
214	        "status": "open",
215	        "source": source,
216	        "tags": list(tags or []),
217	        "filed_by_actor_id": actor_id,
218	        "filed_in_turn_id": turn_id,
219	        "codebase_id": codebase_id,
220	        "created_at": now,
221	        "last_edited_at": now,
222	        "resolution_note": None,
223	        "addressed_at": None,
224	        "epics": [],
225	        "__body__": body,
226	    }
227	
228	    # Write file (both modes)
229	    if cwd:
230	        repo_root = str(cwd)
231	    else:
232	        repo_root = os.getcwd()
233	    fpath = ticket_file_path(repo_root, ticket_id, slug)
234	    write_ticket_file(fpath, record)
235	
236	    # Cloud: write DB row
237	    if store is not None and is_cloud_store(store) and codebase_id:
238	        from megaplan.store.db import DBStore
239	
240	        db: DBStore = store  # type: ignore[assignment]
241	        db.create_ticket(
242	            codebase_id=codebase_id,
243	            title=title,
244	            body=body,
245	            source=source,
246	            tags=list(tags or []),
247	            filed_by_actor_id=actor_id,
248	            filed_in_turn_id=turn_id,
249	            slug=slug,
250	            ticket_id=ticket_id,
251	        )
252	
253	    # Print only the ULID to stdout (per spec)
254	    print(ticket_id, flush=True)
255	    return ticket_id

421	    ticket_id: str,
422	    *,
423	    title: str | None = None,
424	    body: str | None = None,
425	    status: str | None = None,
426	    add_tag: str | None = None,
427	    remove_tag: str | None = None,
428	    store: Store | None = None,
429	    cwd: Path | None = None,
430	) -> dict[str, Any] | None:
431	    """Edit a ticket's fields.  Frontmatter always updated; DB updated in cloud mode."""
432	    if store is None:
433	        store = _resolve_store()
434	
435	    repo_root = str(cwd) if cwd else os.getcwd()
436	
437	    # Find the file
438	    found_path: Path | None = None
439	    found_fm: dict | None = None
440	    for fpath, fm in iterate_ticket_files(repo_root):
441	        if fm.get("id") == ticket_id:
442	            found_path = fpath
443	            found_fm = fm

445	
446	    if found_fm is None:
447	        return None
448	
449	    # Apply changes
450	    if title is not None:
451	        found_fm["title"] = title
452	        # Update slug in the filename
453	        new_slug = slugify(title)

462	        found_fm["__body__"] = body
463	
464	    if status is not None:
465	        found_fm["status"] = status
466	
467	    if add_tag is not None:
468	        tags = list(found_fm.get("tags") or [])
469	        if add_tag not in tags:
470	            tags.append(add_tag)
471	            found_fm["tags"] = tags
472	
473	    if remove_tag is not None:
474	        tags = list(found_fm.get("tags") or [])
475	        if remove_tag in tags:
476	            tags.remove(remove_tag)
477	            found_fm["tags"] = tags
478	
479	    found_fm["last_edited_at"] = datetime.now(timezone.utc)
480	
481	    # Write file
482	    if found_path:
483	        write_ticket_file(found_path, found_fm)
484	
485	    # Cloud: update DB
486	    if store is not None and is_cloud_store(store):
487	        from megaplan.store.db import DBStore
488	
489	        db: DBStore = store  # type: ignore[assignment]
490	        db_changes: dict[str, Any] = {}
491	        if title is not None:
492	            db_changes["title"] = title
493	            db_changes["slug"] = slugify(title)
494	        if body is not None:
495	            db_changes["body"] = body
496	        if status is not None:
497	            db_changes["status"] = status
498	        if add_tag is not None or remove_tag is not None:
499	            # Build new tags list
500	            tags_now = list(found_fm.get("tags") or [])
501	            db_changes["tags"] = tags_now
502	        if db_changes:
503	            db.update_ticket(ticket_id, **db_changes)
504	
505	    return found_fm

710	    epic_id: str,
711	    *,
712	    store: Store | None = None,
713	    repo_root: str | Path | None = None,
714	) -> list[str]:
715	    """Flip every open ticket linked to *epic_id* with ``resolves_on_complete=true``
716	    to ``'addressed'``.
717	
718	    Parameters
719	    ----------
720	    epic_id:
721	        The epic that just completed.
722	    store:
723	        Explicit store (resolved from env if *None*).
724	    repo_root:
725	        Path to the repo for file walking.  If *None* or the
726	        ``.megaplan/tickets/`` directory does not exist, the file
727	        walk is skipped cleanly.
728	
729	    Returns
730	    -------
731	    list[str]
732	        The ULIDs of tickets that were updated (empty if none).
733	        Idempotent — only ``status='open'`` tickets are affected.
734	    """
735	    if store is None:
736	        store = _resolve_store()
737	
738	    updated: list[str] = []

2	
3	Tickets are short, repo-scoped notes on problems, bugs, or "we should look at this" observations against a codebase. 6
