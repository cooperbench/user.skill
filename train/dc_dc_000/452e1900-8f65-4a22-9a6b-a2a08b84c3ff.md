   220→def cmd_plan_skip(args: argparse.Namespace) -> None:
   221→    """Skip issues — unified command for temporary/permanent/false-positive."""
   222→    runtime = command_runtime(args)
   223→    state = runtime.state
   224→    if not require_completed_scan(state):
   225→        return
   226→
   227→    patterns: list[str] = getattr(args, "patterns", [])
   228→    reason: str | None = getattr(args, "reason", None)
   229→    review_after: int | None = getattr(args, "review_after", None)
   230→    permanent: bool = getattr(args, "permanent", False)
   231→    false_positive: bool = getattr(args, "false_positive", False)
   232→    note: str | None = getattr(args, "note", None)
   233→    attestation: str | None = getattr(args, "attest", None)
   234→
   235→    kind = skip_kind_from_flags(permanent=permanent, false_positive=false_positive)
   236→    if not _validate_skip_requirements(
   237→        kind=kind,
   238→        attestation=attestation,
   239→        note=note,
   240→    ):
   241→        return
   242→
   243→    state_file = runtime.state_path
   244→    plan_file = _plan_file_for_state(state_file)
   245→    plan = load_plan(plan_file)
   246→    issue_ids = resolve_ids_from_patterns(state, patterns, plan=plan)
   247→    if not issue_ids:
   248→        print(colorize("  No matching issues found.", "yellow"))
   249→        return
   250→
   251→    # Bulk skip guard — prevent agents from draining the queue in one shot
   252→    if len(issue_ids) > _BULK_SKIP_THRESHOLD:
   253→        print(colorize(
   254→            f"  Bulk skip: {len(issue_ids)} items will be removed from the active queue.",
   255→            "yellow",
   256→        ), file=sys.stderr)
   257→        if not getattr(args, "confirm", False):
   258→            raise CommandError(
   259→                f"Skipping {len(issue_ids)} items requires --confirm. "
   260→                "Review the items first, or skip individually."
   261→            )
   262→
   263→    # For permanent/false_positive: delegate to state layer for score impact
   264→    state_data = _apply_state_skip_resolution(
   265→        kind=kind,
   266→        state_file=state_file,
   267→        issue_ids=issue_ids,
   268→        note=note,
   269→        attestation=attestation,
   270→    )
   271→
   272→    scan_count = state.get("scan_count", 0)
   273→    count = skip_items(
   274→        plan,
   275→        issue_ids,
   276→        kind=kind,
   277→        reason=reason,
   278→        note=note,
   279→        attestation=attestation,
   280→        review_after=review_after,
   281→        scan_count=scan_count,
   282→    )
   283→
   284→    # Log the skip action
   285→    append_log_entry(
   286→        plan,
   287→        "skip",
   288→        issue_ids=issue_ids,
   289→        actor="user",
   290→        note=note,
   291→        detail={"kind": kind, "reason": reason},
   292→    )
   293→    if state_data is not None:
   294→        _save_plan_state_transactional(
   295→            plan=plan,
   296→            plan_path=plan_file,
   297→            state_data=state_data,
   298→            state_path_value=state_file,
   299→        )
   300→    else:
   301→        save_plan(plan, plan_file)
   302→
   303→    print(colorize(f"  {SKIP_KIND_LABELS[kind]} {count} item(s).", "green"))
   304→    if review_after:
   305→        print(colorize(f"  Will re-surface after {review_after} scan(s).", "dim"))
   306→    print_user_message(
   307→        "Hey — if skipping was the right call, just continue with"
   308→        " what you were doing. If you think a broader re-triage is"
   309→        " needed, use `desloppify plan triage`. Run `desloppify"
   310→        " plan --help` to see all available plan tools. Otherwise"
   311→        " no need to reply, just keep going."
   312→    )

   407→
   408→    print(colorize(f"  Reopened {len(reopened)} issue(s).", "green"))

   416→def _check_cluster_guard(patterns: list[str], plan: dict, state: dict) -> bool:
   417→    """Return True if blocked by cluster guard, False if OK to proceed."""
   418→    clusters = plan.get("clusters", {})
   419→    issues = state.get("issues", {})
   420→    for pattern in patterns:
   421→        if pattern in clusters:
   422→            cluster = clusters[pattern]
   423→            # Filter to alive issues only — stale IDs should not count
   424→            ids = [
   425→                fid for fid in cluster.get("issue_ids", [])
   426→                if fid in issues and issues[fid].get("status") == "open"
   427→            ]
   428→            if len(ids) == 0:
   429→                print(colorize(
   430→                    f"\n  Cluster '{pattern}' is empty — add items before marking it done.\n",
   431→                    "yellow",
   432→                ))
   433→                print(colorize(
   434→                    f"  Use: desloppify plan cluster add {pattern} <issue-id>",
   435→                    "dim",
   436→                ))
   437→                return True  # blocked
   438→            if len(ids) <= _CLUSTER_INDIVIDUAL_THRESHOLD:
   439→                _print_cluster_guard(pattern, ids, state)
   440→                return True  # blocked
   441→    return False  # OK

   480→def _blocked_triage_stages(plan: dict) -> dict[str, list[str]]:
   481→    """Return ``{stage_id: [blocked_by_ids]}`` for triage stages that can't run yet.
   482→
   483→    Uses the dependency graph and confirmed-stage metadata directly —
   484→    no state needed, no queue item construction.
   485→    """
   486→    order_set = set(plan.get("queue_order", []))
   487→    present = order_set & TRIAGE_IDS
   488→    if not present:
   489→        return {}
   490→
   491→    confirmed = set(plan.get("epic_triage_meta", {}).get("triage_stages", {}).keys())
   492→    stage_names = ("observe", "reflect", "organize", "commit")
   493→
   494→    blocked: dict[str, list[str]] = {}
   495→    for sid, name in zip(TRIAGE_STAGE_IDS, stage_names, strict=False):
   496→        if sid not in present or name in confirmed:
   497→            continue
   498→        deps = TRIAGE_STAGE_DEPENDENCIES.get(name, set())
   499→        unmet = sorted(
   500→            f"triage::{dep}" for dep in deps
   501→            if f"triage::{dep}" in present and dep not in confirmed
   502→        )
   503→        if unmet:
   504→            blocked[sid] = unmet
   505→    return blocked

   508→def cmd_plan_resolve(args: argparse.Namespace) -> None:
   509→    """Mark issues as fixed — delegates to cmd_resolve for rich UX."""
   510→    patterns: list[str] = getattr(args, "patterns", [])
   511→    attestation: str | None = getattr(args, "attest", None)
   512→    note: str | None = getattr(args, "note", None)
   513→
   514→    # --confirm: auto-generate attestation from --note
   515→    if getattr(args, "confirm", False):
   516→        if not note:
   517→            print(colorize("  --confirm requires --note to describe what you did.", "red"))
   518→            return
   519→        attestation = f"I have actually {note} and I am not gaming the score."
   520→        args.attest = attestation
   521→
   522→    # Handle synthetic IDs (triage::*, workflow::*, subjective::*) directly
   523→    synthetic_ids, real_patterns = _resolve_synthetic_ids(patterns)
   524→    if synthetic_ids:
   525→        plan = load_plan()
   526→        # Validate triage dependency chain
   527→        blocked_map = _blocked_triage_stages(plan)
   528→        for sid in synthetic_ids:
   529→            if sid in blocked_map:
   530→                deps = ", ".join(b.replace("triage::", "") for b in blocked_map[sid])
   531→                print(colorize(f"  Cannot resolve {sid} — blocked by: {deps}", "red"))
   532→                print(colorize("  Complete those stages first, or use --force-resolve to override.", "dim"))
   533→                if not getattr(args, "force_resolve", False):
   534→                    return
   535→
   536→        # Gate workflow items behind triage completion
   537→        GATED_WORKFLOW_IDS = {WORKFLOW_SCORE_CHECKPOINT_ID, WORKFLOW_CREATE_PLAN_ID}
   538→        gated_ids = [sid for sid in synthetic_ids if sid in GATED_WORKFLOW_IDS]
   539→        if gated_ids:
   540→            force = getattr(args, "force_resolve", False)
   541→            meta = plan.get("epic_triage_meta", {})
   542→            # Triage is complete if last_completed_at is set (stages archive on --complete)
   543→            triage_ever_completed = bool(meta.get("last_completed_at"))
   544→            if triage_ever_completed:
   545→                # --complete archives stages and sets last_completed_at → all 4 done
   546→                missing: set[str] = set()
   547→            else:
   548→                confirmed_stages = set(meta.get("triage_stages", {}).keys())
   549→                required_stages = {"observe", "reflect", "organize", "commit"}
   550→                missing = required_stages - confirmed_stages
   551→
   552→            if missing and not force:
   553→                # Auto-inject triage stages if not already in queue
   554→                if not has_triage_in_queue(plan):
   555→                    inject_triage_stages(plan)
   556→                    meta.setdefault("triage_stages", {})
   557→                    plan["epic_triage_meta"] = meta
   558→                    save_plan(plan)
   559→
   560→                # Print targeted error based on which stage is next
   561→                stage_order = ["observe", "reflect", "organize", "commit"]
   562→                next_stage = next((s for s in stage_order if s in missing), "observe")
   563→
   564→                for wid in gated_ids:
   565→                    print(colorize(f"  Cannot resolve {wid} — triage not complete.", "red"))
   566→                print()
   567→
   568→                if next_stage == "observe":
   569→                    print(colorize("  You must analyze the findings before resolving this.", "yellow"))
   570→                    print(colorize("  Start by examining themes, root causes, and contradictions:", "dim"))
   571→                    print(colorize('    desloppify plan triage --stage observe --report "..."', "dim"))
   572→                    print()
   573→                    print(colorize("  The report must be 100+ chars describing what you found.", "dim"))
   574→                elif next_stage == "reflect":
   575→                    print(colorize("  Observe is done. Now compare against previously completed work:", "yellow"))
   576→                    print(colorize('    desloppify plan triage --stage reflect --report "..."', "dim"))
   577→                    print()
   578→                    print(colorize("  The report must mention recurring dimensions if any exist.", "dim"))
   579→                elif next_stage == "organize":
   580→                    print(colorize("  Reflect is done. Now create clusters and prioritize:", "yellow"))
   581→                    print(colorize('    desloppify plan cluster create <name> --description "..."', "dim"))
   582→                    print(colorize("    desloppify plan cluster add <name> <issue-patterns>", "dim"))
   583→                    print(colorize('    desloppify plan cluster update <name> --steps "step1" "step2"', "dim"))
   584→                    print(colorize('    desloppify plan triage --stage organize --report "..."', "dim"))
   585→                    print()
   586→                    print(colorize("  All manual clusters must have descriptions and action_steps.", "dim"))
   587→                elif next_stage == "commit":
   588→                    print(colorize("  Organize is done. Finalize the execution plan:", "yellow"))
   589→                    print(colorize('    desloppify plan triage --complete --strategy "..."', "dim"))
   590→
   591→                print()
   592→                print(colorize(f"  Remaining stages: {', '.join(sorted(missing))}", "dim"))
   593→                print(colorize(
   594→                    "  To skip triage: --force-resolve --note 'reason for skipping triage'",
   595→                    "dim",
   596→                ))
   597→
   598→                # Log the blocked attempt
   599→                append_log_entry(
   600→                    plan, "workflow_blocked",
   601→                    issue_ids=gated_ids, actor="user",
