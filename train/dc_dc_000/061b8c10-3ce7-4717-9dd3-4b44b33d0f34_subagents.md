497	            payload=payload,
498	            batch_task_ids=batch_task_ids,
499	            issues=deviations,
500	            capture_recursive_snapshot_fn=_capture_git_status_snapshot_recursive,
501	        )
502	        observation_snapshot_fn = capture_git_status_snapshot_fn
503	        if (
504	            attribution_result.records
505	            and attribution_result.recursive_snapshot is not None
506	        ):
507	            cached_snapshot = attribution_result.recursive_snapshot
508	            observation_snapshot_fn = lambda _p, _snap=cached_snapshot: (_snap, None)
509	        deviations.extend(
510	            _observe_git_changes(
511	                project_dir=project_dir,
512	                payload=payload,
513	                before_snapshot=before_snapshot,
514	                before_error=before_error,
515	                batch_number=batch_number,
516	                batches_total=batches_total,
517	                capture_git_status_snapshot_fn=observation_snapshot_fn,
518	            )
519	        )
520	    if is_prose_mode(state):
521	        missing_task_evidence = _check_done_task_evidence(
522	            finalize_data.get("tasks", []),
523	            issues=deviations,
524	            should_classify=lambda task: task.get("id") in batch_task_id_set,
525	            has_evidence=lambda task: bool(task.get("sections_written")),
526	            has_advisory_evidence=lambda task: True,
527	            missing_message="Done tasks missing sections_written: ",
528	            advisory_message="",
529	        )
530	    else:
531	        missing_task_evidence = _check_done_task_evidence(
532	            finalize_data.get("tasks", []),
533	            issues=deviations,
534	            should_classify=lambda task: task.get("id") in batch_task_id_set,
535	            has_evidence=lambda task: bool(task.get("files_changed")),
536	            has_advisory_evidence=lambda task: bool(task.get("commands_run")),
537	            missing_message="Done tasks missing both files_changed and commands_run: ",
538	            advisory_message="Advisory: done tasks rely on commands_run without files_changed (FLAG-006 softening): ",
539	        )
540	    execution_audit = validate_execution_evidence(finalize_data, project_dir, mode=plan_mode, state=state)
541	    if attribution_result.records:
542	        execution_audit["auto_attribution"] = list(attribution_result.records)
543	    if execution_audit["skipped"]:
544	        deviations.append(f"Advisory audit skip: {execution_audit['reason']}")
545	    for finding in execution_audit["findings"]:
546	        deviations.append(f"Advisory audit finding: {finding}")
547	    payload["deviations"] = deviations
548	    atomic_write_json(batch_artifact_path(plan_dir, batch_number), payload)
549	    atomic_write_json(plan_dir / "execution_audit.json", execution_audit)
550	    atomic_write_json(plan_dir / "finalize.json", finalize_data)
551	    atomic_write_text(
552	        plan_dir / "final.md", render_final_md(finalize_data, phase="execute")
553	    )
554	    return BatchResult(
555	        worker=worker,
556	        agent=agent,
557	        mode=mode,
558	        refreshed=refreshed,
559	        payload=payload,
560	        batch_number=batch_number,
561	        batch_task_ids=list(batch_task_ids),
562	        batch_sense_check_ids=list(batch_sense_check_ids),
563	        merged_task_count=merged_count,
564	        total_task_count=total_batch_tasks,
565	        acknowledged_sense_check_count=acknowledged_count,
566	        total_sense_check_count=total_batch_checks,
567	        missing_task_evidence=missing_task_evidence,
568	        execution_audit=execution_audit,
569	        finalize_hash=sha256_file(plan_dir / "finalize.json"),
570	        attribution_records=list(attribution_result.records),

614	def handle_execute_one_batch(
615	    *,
616	    root: Path,
617	    plan_dir: Path,
618	    state: PlanState,
619	    args: argparse.Namespace,
620	    batch_number: int,
621	    auto_approve: bool,
622	    agent: str,
623	    mode: str,
624	    refreshed: bool,
625	    model: str | None = None,
626	) -> StepResponse:
627	    finalize_data = read_json(plan_dir / "finalize.json")
628	    global_config = load_config()
629	    quality_config = global_config.get("quality_checks", {})
630	    project_dir = Path(state["config"]["project_dir"])
631	    global_batches = compute_global_batches(finalize_data)
632	    batches_total = len(global_batches)
633	
634	    if batch_number < 1 or batch_number > batches_total:
635	        raise CliError(
636	            "batch_out_of_range",
637	            f"--batch {batch_number} is out of range. Plan has {batches_total} batch(es) (1-indexed).",
638	        )
639	
640	    tasks = finalize_data.get("tasks", [])
641	    completed_ids = {
642	        task["id"]
643	        for task in tasks
644	        if task.get("status") in {"done", "skipped"} and isinstance(task.get("id"), str)
645	    }
646	    for prior_idx in range(batch_number - 1):
647	        prior_batch = global_batches[prior_idx]
648	        missing = [task_id for task_id in prior_batch if task_id not in completed_ids]
649	        if missing:
650	            raise CliError(
651	                "batch_prerequisites",
652	                f"Batch {batch_number} requires batches 1..{batch_number - 1} to be complete. "
653	                f"Batch {prior_idx + 1} has incomplete tasks: {', '.join(missing)}",
654	            )
655	
656	    batch_task_ids = global_batches[batch_number - 1]
657	    active_task_ids = set(batch_task_ids)
658	    batch_sense_check_ids = _active_sense_check_ids(finalize_data, active_task_ids)
659	    batch_prompt = _execute_batch_prompt(
660	        state, plan_dir, batch_task_ids, completed_ids, root=root
661	    )
662	
663	    try:
664	        result = _run_and_merge_batch(
665	            root=root,
666	            plan_dir=plan_dir,
667	            state=state,
668	            args=args,
669	            agent=agent,
670	            mode=mode,
671	            refreshed=refreshed,
672	            model=model,
673	            prompt_override=batch_prompt,
674	            batch_task_ids=batch_task_ids,
675	            batch_sense_check_ids=batch_sense_check_ids,
676	            finalize_data=finalize_data,
677	            batch_number=batch_number,
678	            batches_total=batches_total,
679	            quality_config=quality_config,
680	            capture_git_status_snapshot_fn=_capture_git_status_snapshot,
681	        )
682	    except CliError as error:
683	        if error.code == "worker_timeout":
684	            return _recover_execute_timeout(
685	                plan_dir=plan_dir,
686	                state=state,
687	                error=error,
688	                agent=agent,
689	                mode=mode,
690	                refreshed=refreshed,
691	                auto_approve=auto_approve,
692	                args=args,
693	                batch_number=batch_number,
694	            )
695	        record_step_failure(
696	            plan_dir, state, step="execute", iteration=state["iteration"], error=error
697	        )
698	        raise
699	
700	    apply_session_update(
701	        state,
702	        "execute",
703	        result.agent,
704	        result.worker.session_id,
705	        mode=result.mode,
706	        refreshed=result.refreshed,
707	    )
708	    trace_written = _append_trace_output(plan_dir, result.worker.trace_output)
709	    blocking_reasons = build_blocking_reasons(
710	        tracked_tasks=result.merged_task_count,
711	        total_tasks=result.total_task_count,
712	        acknowledged_checks=result.acknowledged_sense_check_count,
713	        total_checks=result.total_sense_check_count,
714	        missing_task_evidence=result.missing_task_evidence,
715	    )
716	
717	    all_tasks = finalize_data.get("tasks", [])
718	    is_final_batch = batch_number == batches_total
719	    all_tracked = all(
720	        task.get("status") in {"done", "skipped"}
721	        for task in all_tasks
722	        if isinstance(task.get("id"), str)
723	    )
724	
725	    aggregate_payload: dict[str, Any] | None = None
726	    batch_payloads: list[dict[str, Any]] = []
727	    drift = None
728	    if is_final_batch and all_tracked:
729	        plan_mode = state["config"].get("mode", "code")
730	        batch_payloads = [read_json(path) for path in list_batch_artifacts(plan_dir)]
731	        aggregate_payload = _build_aggregate_execution_payload(
732	            batch_payloads,
733	            completed_batches=len(batch_payloads),
734	            total_batches=batches_total,
735	            mode=plan_mode,
736	            plan_dir=plan_dir,
737	            state=state,
738	        )
739	        # _run_and_merge_batch already wrote execution_audit.json; this handler
740	        # only writes the aggregate execution.json after the batch returns.
741	        atomic_write_json(plan_dir / "execution.json", aggregate_payload)
742	        drift = _compute_execute_scope_drift(project_dir, aggregate_payload)
743	        _append_scope_drift_blocker(blocking_reasons, state, drift)
744	
745	    blocked = bool(blocking_reasons)
746	    if is_final_batch and all_tracked and not blocked:
747	        state["current_state"] = STATE_EXECUTED
748	
749	    user_approved_gate = bool(state["meta"].get("user_approved_gate", False))
750	    approval_mode = _resolve_execute_approval_mode(
751	        auto_approve=auto_approve,
752	        user_approved_gate=user_approved_gate,
753	    )
754	    result_value = (
755	        "blocked"
756	        if blocked
757	        else "success" if (is_final_batch and all_tracked) else "partial"
758	    )
759	    append_history(
760	        state,
761	        make_history_entry(
762	            "execute",
763	            duration_ms=result.worker.duration_ms,
764	            cost_usd=result.worker.cost_usd,
765	            result=result_value,
766	            worker=result.worker,
767	            agent=result.agent,
768	            mode=result.mode,
769	            output_file=f"execution_batch_{batch_number}.json",
770	            artifact_hash=sha256_file(batch_artifact_path(plan_dir, batch_number)),
771	            finalize_hash=result.finalize_hash,
772	            approval_mode=approval_mode,
773	        ),
774	    )
775	    if aggregate_payload is not None and drift is not None:
776	        receipt_worker = WorkerResult(
777	            payload=aggregate_payload,
778	            raw_output="",
779	            duration_ms=result.worker.duration_ms,
780	            cost_usd=result.worker.cost_usd,
781	            session_id=result.worker.session_id,
782	            trace_output=result.worker.trace_output,
783	            prompt_tokens=result.worker.prompt_tokens,
784	            completion_tokens=result.worker.completion_tokens,
785	            total_tokens=result.worker.total_tokens,
786	        )
787	        receipt_metrics = execute_metrics(aggregate_payload, drift)
788	        receipt_metrics["batches"] = batch_payloads
789	        receipt_worker.receipt_metrics = receipt_metrics
790	        try:
791	            artifact_hash = sha256_file(plan_dir / "execution.json")
792	            receipt = build_receipt(
793	                phase="execute",
794	                state=state,
795	                plan_dir=plan_dir,
796	                args=args,
797	                worker=receipt_worker,
798	                agent=result.agent,
799	                mode=result.mode,
800	                output_file="execution.json",
801	                artifact_hash=artifact_hash,
802	                verdict=result_value,
803	                drift=drift,
804	            )
805	            write_receipt(plan_dir, receipt, project_dir=project_dir)
806	        except Exception:
807	            log.warning("Execute receipt emission failed", exc_info=True)
808	    save_state_merge_meta(plan_dir, state)
809	
810	    batches_remaining = batches_total - batch_number
811	    tracking_note = _format_execute_tracking_note(
812	        merged_count=result.merged_task_count,
813	        total_tasks=result.total_task_count,
814	        acknowledged_count=result.acknowledged_sense_check_count,
815	        total_checks=result.total_sense_check_count,
816	    )
817	    artifacts = [
818	        f"execution_batch_{batch_number}.json",
819	        "execution_audit.json",
820	        "finalize.json",
821	        "final.md",
822	    ]
823	    if aggregate_payload is not None and not blocked:
824	        artifacts.insert(0, "execution.json")
825	    if trace_written:
826	        artifacts.append("execution_trace.jsonl")
827	
828	    if blocked:
829	        summary = (
830	            "Blocked: "
831	            + "; ".join(blocking_reasons)
832	            + ". Re-run execute to complete tracking."
833	        )
834	        next_step = "execute"
835	        response_state = STATE_FINALIZED
836	    elif is_final_batch and all_tracked:
837	        summary = result.payload.get("output", "Batch complete.") + tracking_note
838	        next_step = "review"
839	        response_state = STATE_EXECUTED
840	    else:
841	        summary = (
842	            f"Batch {batch_number}/{batches_total} complete.{tracking_note} "
843	            f"{batches_remaining} batch(es) remaining."
844	        )
845	        next_step = "execute"
846	        response_state = STATE_FINALIZED
847	    if drift is not None and drift.severity != "none":
848	        summary = f"[scope_drift={drift.severity}] {summary}"
849	
850	    # Surface worker-reported `status=blocked` tasks prominently. Merged
851	    # task_updates have already been written into finalize_data; count tasks
852	    # in this batch that ended in 'blocked' so supervisors see the real
853	    # reason a batch did not advance instead of just "state=finalized".
854	    batch_blocked_ids = [
855	        task.get("id")
856	        for task in finalize_data.get("tasks", [])
857	        if task.get("id") in set(batch_task_ids)
858	        and task.get("status") == "blocked"
859	    ]
860	    warnings: list[str] = []
861	    if blocked:
862	        warnings.append(summary)
863	    if batch_blocked_ids:
864	        warnings.append(
865	            f"{len(batch_blocked_ids)} task(s) reported status=blocked by the worker "
866	            "— investigate executor_notes before continuing"
867	        )
868	
869	    response: StepResponse = {
870	        "success": not blocked,
871	        "step": "execute",
872	        "summary": summary,
873	        "artifacts": artifacts,
874	        "monitor_hint": build_monitor_hint(plan_dir),
875	        "next_step": next_step,
876	        "state": response_state,
877	        "batch": batch_number,
878	        "batches_total": batches_total,
879	        "batches_remaining": batches_remaining,
880	        "files_changed": result.payload.get("files_changed", []),
881	        "deviations": result.payload.get("deviations", []),
882	        "warnings": warnings,
883	        "auto_approve": auto_approve,
884	        "user_approved_gate": user_approved_gate,
885	        "blocked_task_ids": batch_blocked_ids,
886	    }
887	    if next_step == "execute" and not blocked:
888	        response["guidance"] = f"Run --batch {batch_number + 1}"
889	    _attach_next_step_runtime(response)
890	    return response

927	        for task in tasks

931	        task
932	        for task in tasks
933	        if task.get("status") == "pending" and isinstance(task.get("id"), str)

945	    global_batch_lookup = {
946	        tuple(batch): index + 1 for index, batch in enumerate(global_batches)
947	    }

949	    total_batches = len(batches_to_run) or 1
950	    active_task_ids = set(
951	        all_task_ids if single_batch_mode else [task["id"] for task in pending_tasks]
952	    )
953	    active_sense_check_ids = set(
954	        all_sense_check_ids
955	        if single_batch_mode
956	        else _active_sense_check_ids(finalize_data, active_task_ids)
957	    )
958	
959	    batch_payloads: list[dict[str, Any]] = []
960	    all_attribution_records: list[dict[str, Any]] = []
961	    trace_chunks: list[str] = []
962	    total_duration_ms = 0
963	    total_cost_usd = 0.0
964	    timeout_error: CliError | None = None
965	    latest_session_id: str | None = None
966	    blocking_reasons: list[str] = []
967	    timeout_recovery: StepResponse | None = None
968	
969	    for batch_index, batch_task_ids in enumerate(batches_to_run, start=1):
970	        batch_prompt = (
971	            None
972	            if single_batch_mode
973	            else _execute_batch_prompt(
974	                state,
975	                plan_dir,
976	                batch_task_ids,
977	                completed_task_ids,
978	                root=root,
979	            )
980	        )
981	        batch_number_for_artifact = (
982	            1
983	            if single_batch_mode
984	            else global_batch_lookup.get(tuple(batch_task_ids), batch_index)
985	        )
986	        batch_sense_check_ids = (
987	            all_sense_check_ids
988	            if single_batch_mode
989	            else _active_sense_check_ids(finalize_data, set(batch_task_ids))
990	        )
991	        batches_total_for_observation = 1 if single_batch_mode else len(global_batches)
992	        try:
993	            result = _run_and_merge_batch(
994	                root=root,
995	                plan_dir=plan_dir,
996	                state=state,
997	                args=args,
998	                agent=agent,
999	                mode=mode,
1000	                refreshed=refreshed,
1001	                model=model,
1002	                prompt_override=batch_prompt,
1003	                batch_task_ids=batch_task_ids,
1004	                batch_sense_check_ids=batch_sense_check_ids,
1005	                finalize_data=finalize_data,
1006	                batch_number=batch_number_for_artifact,
1007	                batches_total=batches_total_for_observation,
1008	                quality_config=quality_config,
1009	                capture_git_status_snapshot_fn=_capture_git_status_snapshot,
1010	            )
1011	        except CliError as error:
1012	            if error.code == "worker_timeout":
1013	                timeout_error = error
1014	                latest_session_id = (
1015	                    error.extra.get("session_id")
1016	                    if isinstance(error.extra.get("session_id"), str)
1017	                    else latest_session_id
1018	                )
1019	                timeout_recovery = _recover_execute_timeout(
1020	                    plan_dir=plan_dir,
1021	                    state=state,
1022	                    error=error,
1023	                    agent=agent,
1024	                    mode=mode,
1025	                    refreshed=refreshed,
1026	                    auto_approve=auto_approve,
1027	                    args=args,
1028	                    batch_number=(
1029	                        None if single_batch_mode else batch_number_for_artifact
1030	                    ),
1031	                    persist_state=False,
1032	                )
1033	                finalize_data = read_json(plan_dir / "finalize.json")
1034	                break
1035	            record_step_failure(
1036	                plan_dir,
1037	                state,
1038	                step="execute",
1039	                iteration=state["iteration"],
1040	                error=error,
1041	            )
1042	            raise
1043	
1044	        total_duration_ms += result.worker.duration_ms
1045	        total_cost_usd += result.worker.cost_usd
1046	        latest_session_id = result.worker.session_id
1047	        apply_session_update(
1048	            state,
1049	            "execute",
1050	            result.agent,
1051	            result.worker.session_id,
1052	            mode=result.mode,
1053	            refreshed=result.refreshed,
1054	        )
1055	        batch_payloads.append(result.payload)
1056	        all_attribution_records.extend(result.attribution_records)
1057	        if result.worker.trace_output is not None:
1058	            trace_chunks.append(result.worker.trace_output)
1059	        completed_task_ids.update(
1060	            task_id
1061	            for task_id in batch_task_ids
