   def _push_with_retry(run_dir, repo_id, manifest, logger, *, backoffs=PUSH_BACKOFFS,
                        sleep=time.sleep, jitter=lambda: random.uniform(0, 5)):
       from huggingface_hub.utils import HfHubHTTPError
       last_exc = None
       total_wait = 0
       for attempt, base in enumerate([0, *backoffs], start=1):
           if base:
               wait = base + jitter()
               logger.info("push_retry_wait", extra={"phase":"push","extra_data":{"attempt":attempt,"wait":wait}})
               sleep(wait); total_wait += wait
           try:
               url = push_shards_to_huggingface(run_dir, repo_id, manifest)
               logger.info("push_success", extra={"phase":"push","extra_data":{"attempt":attempt,"url":url}})
               return url, attempt, total_wait
           except HfHubHTTPError as e:
               status = getattr(getattr(e, "response", None), "status_code", None)
               if status in (401, 403, 400):
                   raise PushFailed(e, attempt, total_wait) from e   # non-retriable
               if status == 429:
                   retry_after = int(e.response.headers.get("Retry-After", 0)) if e.response else 0
                   if retry_after: sleep(retry_after); total_wait += retry_after
               last_exc = e
           except (ConnectionError, TimeoutError, OSError) as e:
               last_exc = e
       raise PushFailed(last_exc, len(backoffs)+1, total_wait)
   ```
2. **Use `random.uniform`** (add `import random` at top). The `jitter` argument is plumbed so tests can pass `lambda: 0`.
3. Retriable: `ConnectionError`, `TimeoutError`, `OSError`, `HfHubHTTPError` 5xx and 429 (with `Retry-After` honor). Non-retriable: 401/403/400 — raise immediately as `PushFailed`.

### Step 3: `record_last_auto_run` + `_cleanup_published` + staging-size helpers (`dataclaw/cli.py`)
**Scope:** Small
1. **Add** helpers near `_run_export`:
   ```python
   def record_last_auto_run(result, config, **fields):
       config["last_auto_run"] = {
           "result": result,
           "timestamp": datetime.now(tz=timezone.utc).isoformat(),
           **fields,
       }

   def _cleanup_published(staging_root, *, keep=3):
       pub = staging_root / PUBLISHED_DIRNAME
       if not pub.exists(): return
       dirs = sorted([p for p in pub.iterdir() if p.is_dir()],
                     key=lambda p: p.stat().st_mtime, reverse=True)
       for old in dirs[keep:]:
           shutil.rmtree(old, ignore_errors=True)

   def _staging_size_bytes(staging_root):
       total = 0
       if not staging_root.exists(): return 0
       for entry in staging_root.iterdir():
           if entry.name == PUBLISHED_DIRNAME or not entry.is_dir(): continue
           for p in entry.rglob("*"):
               if p.is_file(): total += p.stat().st_size
       return total

   def _find_failed_run_dir(staging_root):
       if not staging_root.exists(): return None
       cands = [p for p in staging_root.iterdir()
                if p.is_dir() and p.name != PUBLISHED_DIRNAME and (p / MANIFEST_REL).exists()]
       return max(cands, key=lambda p: p.stat().st_mtime, default=None)
   ```

   def _handle_auto(args):
       config = load_config()
       run_id = uuid.uuid4().hex
       logger = dc_logging.setup_logging(run_id)

       if args.retry_only: return _handle_auto_retry_only(args, config, run_id, logger)

       if not (config.get("auto") or {}).get("enabled"):

                                    fetch_existing=True, custom_strings=custom_strings,
                                    include_thinking=not args.no_thinking if hasattr(args,"no_thinking") else True)

       if manifest["total_sessions_new"] == 0:
           record_last_auto_run("noop", config, sessions=0, staging_dir=str(run_dir))

           shutil.rmtree(run_dir, ignore_errors=True)

       # privacy-filter policy branch (strict blocks; permissive auto-acks new findings, re-runs)
       policy = args.policy_override or config["auto"]["policy"]

           pf_new = _scan_pf_new(run_dir, config)
           if pf_new and not args.force:
               record_last_auto_run("blocked", config, sessions=manifest["total_sessions_new"],
                                    staging_dir=str(run_dir), pf_new_count=len(pf_new))
               save_config(config); sys.exit(3)

       if args.dry_run:
           record_last_auto_run("dry-run", config, sessions=manifest["total_sessions_new"],
                                staging_dir=str(run_dir))
           save_config(config); return

       try:
           url, attempts, total_wait = _push_with_retry(run_dir, config["repo"], manifest, logger)
       except PushFailed as e:
           record_last_auto_run("error", config, sessions=manifest["total_sessions_new"],
                                staging_dir=str(run_dir), push_attempts=e.attempts,
                                backoff_seconds_total=e.backoff_seconds_total, error=str(e.cause))
           save_config(config)
           dc_logging.write_run_summary(run_dir, _summary("error", run_id, manifest, run_dir, started_at, attempts=e.attempts))
           sys.exit(4)

       # success
       config["last_export_cutoff"] = {**config.get("last_export_cutoff", {}),
                                        **manifest["max_end_time_by_source"]}
       published = STAGING_ROOT / PUBLISHED_DIRNAME / run_id
       published.parent.mkdir(parents=True, exist_ok=True)
       shutil.move(str(run_dir), str(published))
       record_last_auto_run("pushed", config, sessions=manifest["total_sessions_new"],
                            repo_url=url, staging_dir=str(published),
                            push_attempts=attempts, backoff_seconds_total=total_wait)
       save_config(config)
       dc_logging.write_run_summary(published, _summary("pushed", run_id, manifest, published, started_at, attempts=attempts))
   ```

### Step 8: `dataclaw/scheduler.py` (new module)
**Scope:** Medium
1. **Module skeleton**:
   ```python
   """OS scheduling for dataclaw auto — launchd (macOS), systemd --user (Linux)."""
   import os, plistlib, shutil, subprocess, sys
   from pathlib import Path
   from typing import Iterable

   _ENV_ALLOWLIST = ("PATH", "HOME", "HF_HOME", "HUGGINGFACE_HUB_CACHE",
                     "LANG", "LC_ALL", "TMPDIR")
   LAUNCH_LABEL = "io.dataclaw.auto"
   LAUNCH_PLIST = Path.home() / "Library/LaunchAgents" / f"{LAUNCH_LABEL}.plist"
   SYSTEMD_DIR = Path.home() / ".config/systemd/user"
   LOG_OUT = Path.home() / ".dataclaw/logs/auto.stdout.log"
   LOG_ERR = Path.home() / ".dataclaw/logs/auto.stderr.log"

   def _capture_env(allowlist: Iterable[str] = _ENV_ALLOWLIST) -> dict[str, str]:
       return {k: os.environ[k] for k in allowlist if k in os.environ}

   def _binary_path(config) -> str:
       return (config.get("auto") or {}).get("binary") or shutil.which("dataclaw") or sys.argv[0]

   def _parse_time(hhmm: str) -> tuple[int, int]:
       h, m = hhmm.split(":"); return int(h), int(m)

   def install_macos(config, time_hhmm="03:00"):
       hh, mm = _parse_time(time_hhmm)
       plist = {
           "Label": LAUNCH_LABEL,
           "ProgramArguments": [_binary_path(config), "auto"],
           "WorkingDirectory": str(Path.home()),
           "StartCalendarInterval": {"Hour": hh, "Minute": mm},
           "RunAtLoad": False,
           "EnvironmentVariables": _capture_env(),
           "StandardOutPath": str(LOG_OUT),
           "StandardErrorPath": str(LOG_ERR),
       }
       LAUNCH_PLIST.parent.mkdir(parents=True, exist_ok=True)
       LAUNCH_PLIST.write_bytes(plistlib.dumps(plist))
       os.chmod(LAUNCH_PLIST, 0o600)
       subprocess.run(["launchctl", "bootout", f"gui/{os.getuid()}/{LAUNCH_LABEL}"],
                      check=False, stderr=subprocess.DEVNULL)
       subprocess.run(["launchctl", "bootstrap", f"gui/{os.getuid()}", str(LAUNCH_PLIST)],
                      check=True)
       return LAUNCH_PLIST

   def install_linux(config, time_hhmm="03:00"):
       hh, mm = _parse_time(time_hhmm)
       SYSTEMD_DIR.mkdir(parents=True, exist_ok=True)
       svc = SYSTEMD_DIR / "dataclaw-auto.service"
       tmr = SYSTEMD_DIR / "dataclaw-auto.timer"
       env_lines = "\n".join(f"Environment={k}={v}" for k, v in _capture_env().items())
       svc.write_text(f"""[Unit]
   Description=DataClaw auto run
   [Service]
   Type=oneshot
   ExecStart={_binary_path(config)} auto
   {env_lines}
   StandardOutput=append:{LOG_OUT}
   StandardError=append:{LOG_ERR}
   """)
       tmr.write_text(f"""[Unit]
   Description=DataClaw daily auto
   [Timer]
   OnCalendar=*-*-* {hh:02d}:{mm:02d}:00
   Persistent=true
   Unit=dataclaw-auto.service
   [Install]
   WantedBy=timers.target
   """)
       for p in (svc, tmr): os.chmod(p, 0o600)
       subprocess.run(["systemctl", "--user", "daemon-reload"], check=True)
       subprocess.run(["systemctl", "--user", "enable", "--now", "dataclaw-auto.timer"], check=True)
       return svc, tmr

   def uninstall_macos():
       subprocess.run(["launchctl", "bootout", f"gui/{os.getuid()}/{LAUNCH_LABEL}"],
                      check=False, stderr=subprocess.DEVNULL)
       LAUNCH_PLIST.unlink(missing_ok=True)

   def uninstall_linux():
       subprocess.run(["systemctl", "--user", "disable", "--now", "dataclaw-auto.timer"],
                      check=False)
       (SYSTEMD_DIR / "dataclaw-auto.service").unlink(missing_ok=True)
       (SYSTEMD_DIR / "dataclaw-auto.timer").unlink(missing_ok=True)
       subprocess.run(["systemctl", "--user", "daemon-reload"], check=False)

   def status() -> dict:
       if sys.platform == "darwin":
           r = subprocess.run(["launchctl", "print", f"gui/{os.getuid()}/{LAUNCH_LABEL}"],
                              capture_output=True, text=True, check=False)
           return {"platform": "macos", "installed": LAUNCH_PLIST.exists(),
                   "active": r.returncode == 0, "raw": r.stdout[:1000]}
       r = subprocess.run(["systemctl", "--user", "is-enabled", "dataclaw-auto.timer"],
                          capture_output=True, text=True, check=False)
       return {"platform": "linux",
               "installed": (SYSTEMD_DIR / "dataclaw-auto.timer").exists(),
               "active": r.stdout.strip() == "enabled"}

   def notify(title: str, body: str) -> None:
       try:
           if sys.platform == "darwin":
               subprocess.run(["osascript", "-e",
                               f'display notification {body!r} with title {title!r}'],
                              check=False)
           else:
               subprocess.run(["notify-send", title, body], check=False)
       except OSError:
           pass
   ```

   ```python
   def _handle_rollback(args):
       config = load_config()
       repo_id = args.repo or config.get("repo")
       if not repo_id: fail_json({"error":"No repo configured"}, 2)
       from huggingface_hub import HfApi, CommitOperationAdd, CommitOperationDelete
       api = HfApi()
       if args.list_commits:
           commits = api.list_repo_commits(repo_id, repo_type="dataset")[: args.limit]
           print(json.dumps([{"sha": c.commit_id, "title": c.title,
                               "created_at": c.created_at.isoformat() if c.created_at else None}
                              for c in commits], indent=2))
           return

       previous = api.list_repo_commits(repo_id, repo_type="dataset")[0].commit_id
       target = args.commit
       target_files = api.list_repo_files(repo_id, repo_type="dataset", revision=target)
       current_files = set(api.list_repo_files(repo_id, repo_type="dataset"))
       ops = []
       for path in target_files:
           local = api.hf_hub_download(repo_id=repo_id, filename=path, repo_type="dataset",
                                        revision=target)
           ops.append(CommitOperationAdd(path_in_repo=path, path_or_fileobj=local))
       for path in current_files - set(target_files):
           ops.append(CommitOperationDelete(path_in_repo=path))

       run_id = uuid.uuid4().hex
       logger = dc_logging.setup_logging(run_id)
       logger.info("rollback", extra={"phase":"rollback", "extra_data":{
           "initiator": os.environ.get("USER", "unknown"),
           "target_commit": target, "previous_commit": previous,
           "reverted_files": len(target_files), "dry_run": args.dry_run}})

       if args.dry_run:
           print(json.dumps({"dry_run": True, "target_commit": target,
                              "previous_commit": previous,
                              "operations": [{"type": op.__class__.__name__,
                                              "path": getattr(op, "path_in_repo", None)} for op in ops]}, indent=2))
           return

       commit = api.create_commit(repo_id=repo_id, repo_type="dataset", operations=ops,
                                   commit_message=f"Rollback to {target[:8]}")
       print(json.dumps({"rolled_back": True, "target_commit": target,
                          "previous_commit": previous, "new_commit": commit.oid}, indent=2))
   ```
3. **Dispatch** in `main()`: `if command == "rollback": _handle_rollback(args); return`.
