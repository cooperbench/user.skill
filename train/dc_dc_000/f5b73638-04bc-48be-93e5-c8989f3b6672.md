   ```python
   import base64, os, re

   _DISCORD_BOT_USER_ID_RE = re.compile(r"^\d+$")

   def _decode_discord_user_id(token: str | None) -> str | None:
       if not token:
           return None
       try:
           segment = token.split(".", 1)[0]
           padded = segment + "=" * (-len(segment) % 4)
           decoded = base64.urlsafe_b64decode(padded).decode()
           return decoded if _DISCORD_BOT_USER_ID_RE.match(decoded) else None
       except (ValueError, UnicodeDecodeError):
           return None

   def discord_bot_user_id() -> str | None:
       explicit = os.environ.get("DISCORD_BOT_USER_ID")
       if explicit and _DISCORD_BOT_USER_ID_RE.match(explicit):
           return explicit
       return _decode_discord_user_id(os.environ.get("DISCORD_BOT_TOKEN"))
   ```
