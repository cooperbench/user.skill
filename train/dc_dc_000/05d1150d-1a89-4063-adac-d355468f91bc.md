257	    if "=" not in raw:
258	        usage_error(f"artagents: invalid --asset value {raw!r}; expected KEY=PATH")
259	    key, path_text = raw.split("=", 1)
260	    key = key.strip()
261	    path_text = path_text.strip()
262	    if not key or not path_text:
263	        usage_error(f"artagents: invalid --asset value {raw!r}; expected KEY=PATH")
264	    if key == "main":
265	        usage_error("artagents: asset key 'main' is reserved; pass the primary video via --video")
266	    if asset_cache.is_url(path_text):
267	        return key, path_text
268	    path = Path(path_text).expanduser().resolve()
269	    if not path.exists():
270	        usage_error(f"artagents: asset path not found for {key!r}: {path}")
271	    return key, path

465	def _coerce_frontmatter_value(raw: str) -> object:
466	    """Coerce a YAML-like scalar string into a Python value.
467	
468	    Recognizes: ``true``/``false`` (case-insensitive) -> bool; bare integers and
469	    floats -> numeric; strings wrapped in matching single or double quotes ->
470	    unquoted str; everything else -> raw str (whitespace-trimmed).
471	    """
472	    text = raw.strip()
473	    if not text:
474	        return ""
475	    lowered = text.lower()
476	    if lowered == "true":
477	        return True
478	    if lowered == "false":
479	        return False
480	    if (text.startswith("\"") and text.endswith("\"") and len(text) >= 2) or (
481	        text.startswith("'") and text.endswith("'") and len(text) >= 2
482	    ):
483	        return text[1:-1]
484	    try:
485	        return int(text)
486	    except ValueError:
487	        pass
488	    try:
489	        return float(text)
490	    except ValueError:
491	        pass
492	    return text

495	def parse_brief_frontmatter(text: str) -> tuple[dict[str, object], str]:
496	    """Split an optional YAML-like ``---``-fenced frontmatter block off a brief.
497	
498	    The frontmatter must begin on line 1 with a line containing only ``---``,
499	    end with another line containing only ``---``, and contain
500	    ``key: value`` pairs (one per line). Blank lines and ``#`` comment lines
501	    inside the block are tolerated. When no frontmatter is present, returns
502	    ``({}, text)`` unchanged.
503	    """
504	    if not text.startswith("---"):
505	        return {}, text
506	    lines = text.split("\n")
507	    if lines[0].strip() != "---":
508	        return {}, text
509	    metadata: dict[str, object] = {}
510	    closing_index: int | None = None
511	    for index in range(1, len(lines)):
512	        line = lines[index]
513	        stripped = line.strip()
514	        if stripped == "---":
515	            closing_index = index
516	            break
517	        if not stripped or stripped.startswith("#"):
518	            continue
519	        if ":" not in stripped:
520	            # Malformed line inside frontmatter; treat the file as having no
521	            # frontmatter to avoid silently corrupting a brief that happens to
522	            # start with three dashes (e.g. an em-dash separator).
523	            return {}, text
524	        key, _, value = stripped.partition(":")
525	        key = key.strip()
526	        if not key:
527	            return {}, text
528	        metadata[key] = _coerce_frontmatter_value(value)
529	    if closing_index is None:
530	        return {}, text
531	    body = "\n".join(lines[closing_index + 1 :])
532	    return metadata, body

545	    args.brief_out.mkdir(parents=True, exist_ok=True)

557	    args.brief_copy.write_bytes(body_bytes)

867	def _initial_facts(args: argparse.Namespace) -> set[str]:
868	    """Compute the set of pipeline facts available before any step runs.
869	
870	    Facts are matched against each executor's `pipeline_requirements`; a step
871	    runs when its requirements are a subset of the running facts, where
872	    each step that runs adds its `graph.provides` to the set.
873	    """
874	    facts: set[str] = {"brief", "theme"}
875	    if args.video is not None:
876	        facts.update({"source_video", "video", "source_media"})
877	    if args.audio is not None:
878	        facts.update({"source_audio", "audio", "source_media"})
879	    if getattr(args, "target_duration", None) is not None:
880	        facts.add("target_duration")
881	    # Phase 3 SD-003 precedence: explicit CLI --allow-generative-effects wins,
882	    # else the brief's allow_generative_visuals frontmatter, else False.
883	    if getattr(args, "allow_generative_effects", False) or getattr(
884	        args, "brief_allow_generative_visuals", False
885	    ):
886	        facts.add("generative_visuals_enabled")
887	    return facts

890	def select_steps(args: argparse.Namespace) -> list[Step]:
891	    """Select pipeline steps via manifest-declared requirements.
892	
893	    Walks STEP_ORDER (used as the topological hint) and includes each step
894	    whose executor's `pipeline_requirements` are satisfied by the running
895	    facts set. Each step that runs contributes its `graph.provides` for
896	    downstream steps. Replaces the old mode-fork logic; equivalent for
897	    source-video, audio-only, and pure-generative briefs.
898	    """
899	    from artagents.core.executor.registry import load_default_registry
900	
901	    registry = load_default_registry()
902	    executors_by_step = {
903	        executor.metadata.get("pipeline_step"): executor
904	        for executor in registry.list()
905	        if executor.metadata.get("pipeline_step")
906	    }
907	    facts = _initial_facts(args)
908	    all_steps = {step.name: step for step in build_pool_steps()}
909	    selected: list[Step] = []
910	    for name in STEP_ORDER:
911	        step = all_steps.get(name)
912	        if step is None:
913	            continue
914	        executor = executors_by_step.get(name)
915	        if executor is None:
916	            selected.append(step)
917	            continue
918	        requirements = set(executor.pipeline_requirements)
919	        if not requirements.issubset(facts):
920	            continue
921	        selected.append(step)
922	        facts.update(executor.graph.provides or ())
923	    return selected

926	def _redact_command(cmd: list[str]) -> list[str]:
927	    """Strip env-file values from logged argv (paths can contain secrets)."""
928	    out: list[str] = []
929	    skip_next = False
930	    for token in cmd:
931	        if skip_next:
932	            out.append("<redacted>")
933	            skip_next = False
934	            continue
935	        if token in ("--env-file",):
936	            out.append(token)
937	            skip_next = True
938	            continue
939	        out.append(token)
940	    return out

1013	def should_rerun(step: Step, args: argparse.Namespace, forced: bool) -> bool:
1014	    if forced:
1015	        return True
1016	    if step.always_run:
1017	        return True
1018	    paths = sentinel_paths(step, args)
1019	    existing = [path.exists() for path in paths]
1020	    if step.name == "refine" and all(existing):
1021	        refine_path = paths[0]
1022	        for name in ("hype.timeline.json", "hype.assets.json", "hype.metadata.json"):
1023	            candidate = step_output_root(step, args) / name
1024	            if candidate.exists() and candidate.stat().st_mtime > refine_path.stat().st_mtime:
1025	                return True
1026	    if step.name == "render" and all(existing):
1027	        render_path = paths[0]
1028	        for name in ("hype.timeline.json", "hype.assets.json", "hype.metadata.json", "refine.json"):
1029	            candidate = step_output_root(step, args) / name
1030	            if candidate.exists() and candidate.stat().st_mtime > render_path.stat().st_mtime:
1031	                return True
1032	    if step.name == "editor_review" and all(existing):
1033	        review_path = paths[0]
1034	        candidate = step_output_root(step, args) / "hype.mp4"
1035	        if candidate.exists() and candidate.stat().st_mtime > review_path.stat().st_mtime:
1036	            return True
1037	    if all(existing):
1038	        return False
1039	    if step.name == "cut" and any(existing):
1040	        print("cut: partial prior output detected, rerunning")
1041	    return True

1052	def run_step(step: Step, cmd: list[str], args: argparse.Namespace) -> int:
1053	    logs_dir = log_dir_for_step(step, args)
1054	    logs_dir.mkdir(parents=True, exist_ok=True)
1055	    log_path = logs_dir / f"{step.name}.log"
1056	    with log_path.open("w", encoding="utf-8") as log_handle:
1057	        env = os.environ.copy()
1058	        if getattr(args, "audit", None) is not None:
1059	            env["ARTAGENTS_AUDIT_RUN_DIR"] = str(args.out)
1060	            parent_ids = getattr(args, "audit_parent_ids", [])
1061	            if parent_ids:
1062	                env[PARENT_IDS_ENV] = ",".join(parent_ids)
1063	        if getattr(args, "no_audit", False):
1064	            env["ARTAGENTS_AUDIT_DISABLED"] = "1"
1065	        env.update(thread_subprocess_env())
1066	        process = subprocess.Popen(
1067	            cmd,
1068	            stdout=subprocess.PIPE,
1069	            stderr=subprocess.STDOUT,
1070	            bufsize=1,
1071	            text=True,
1072	            env=env,
1073	        )
1074	        assert process.stdout is not None
1075	        for line in process.stdout:
1076	            log_handle.write(line)
1077	            if args.verbose:
1078	                sys.stdout.write(line)
1079	                sys.stdout.flush()
1080	        returncode = process.wait()
1081	    if returncode != 0:
1082	        print_log_tail(step.name, log_path)
1083	    elif getattr(args, "audit", None) is not None:
1084	        output_ids = _register_step_outputs(step, cmd, args, log_path)
1085	        if output_ids:
1086	            args.audit_parent_ids = output_ids
1087	    return returncode

1319	def pool_main(args: argparse.Namespace) -> int:
1320	    args.out.mkdir(parents=True, exist_ok=True)
1321	    args.audit = None if args.no_audit else AuditContext.for_run(args.out)
1322	    if args.audit is not None:
1323	        _register_run_inputs(args)
1324	    prepare_brief_artifacts(args)
1325	    for label, url in _url_inputs(args):
1326	        _preflight_url_expiry(label, url)
1327	
1328	    # Phase 3 mixed-mode: --dry-run plans the run without invoking executors.
1329	    # Computes step set, builds redacted commands, writes hype.plan.json, exits.
1330	    if getattr(args, "dry_run", False):
1331	        return _write_dry_run_plan(args)
1332	
1333	    _prefetch_url_inputs(args)
