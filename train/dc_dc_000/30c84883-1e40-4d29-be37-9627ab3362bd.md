332	def _find_step_by_path(plan, path_tuple):
333	    """Walk a TaskPlan to find the step at ``path_tuple`` (descending NestedStep
334	    children). Returns the step or None if the path does not resolve.
335	    """
336	    if not path_tuple:
337	        return None
338	    steps = plan.steps
339	    for segment in path_tuple[:-1]:
340	        match = next((s for s in steps if s.id == segment), None)
341	        if match is None or not isinstance(match, NestedStep):
342	            return None
343	        steps = match.plan.steps
344	    return next((s for s in steps if s.id == path_tuple[-1]), None)

348	    """Return the set of item ids that have a completed/attested event under
349	    ``host_path``. ``host_path`` is the STEP_PATH_SEP-joined string form.
350	    """
351	    path_list = host_path.split(STEP_PATH_SEP) if host_path else []
352	    completed: set[str] = set()

354	        if not isinstance(ev, dict):
355	            continue

357	        if kind not in ("item_completed", "item_attested"):

362	        if isinstance(item_id, str):
363	            completed.add(item_id)

529	def _summarize_run_dir(run_dir: Path) -> tuple[str, str, str]:
530	    """Return (status, last_event_kind, last_ts) for a run directory.
531	
532	    Per FLAG-P5-006: only ``aborted`` vs ``in-progress`` are reliably
533	    distinguishable in V1. A naturally completed plan still leaves
534	    ``active_run.json`` in place, so the "complete" bucket is mostly
535	    unobservable.
536	    """
537	    events_path = run_dir / "events.jsonl"
538	    if not events_path.is_file():
539	        return "in-progress", "", ""
540	    events = read_events(events_path)
541	    if not events:
542	        return "in-progress", "", ""
543	    last = events[-1]
544	    last_kind = str(last.get("kind", ""))
545	    last_ts = str(last.get("ts", ""))
546	    status = "aborted" if last_kind == "run_aborted" else "in-progress"
547	    return status, last_kind, last_ts

     - If `source_path` is already a symlink whose resolved target sits inside `project_dir / CAS_DIRNAME`, short-circuit: read the existing CAS hash from the link target's filename and return `(link_target, hash)` without touching anything.
     - Otherwise compute `digest = hash_file(source_path)`, `target = cas_path(project_dir, digest)`.
     - `mkdir(parents=True, exist_ok=True)` on `target.parent`.
     - If `target.exists()`, treat as duplicate content: `source_path.unlink()` (the source is being replaced by a symlink in the next step anyway). Return `(target, digest)`.
     - Else `os.replace(source_path, target)` — atomic on the same filesystem; the projects root is single-host single-volume by design (SD-029).
     - Return `(target, digest)`.
   - `link_into_produces(cas_entry: Path, target_path: Path) -> None`:
     - If `target_path` exists or is a symlink, `target_path.unlink()` first.
     - Compute `rel = os.path.relpath(cas_entry, target_path.parent)` so the link stays portable.
     - `os.symlink(rel, target_path)`.
2. **Keep** the module under ~80 lines; pure stdlib; no logging, no tracing.
3. **Export** `intern`, `link_into_produces`, `cas_path`, `CAS_DIRNAME` via `__all__`.
