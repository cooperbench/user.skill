33	
34	This is using jujustiu, jj + git and we'll see how it goes. Below is my

40	- **Single-step** (recommended for mechanical/focused changes): bump directly to
41	  `X.Y.Z`, implement in one commit. Simpler history.
42	- **Multi-step** (for exploratory/large changes): bump to `X.Y.Z-devN`, implement
43	  across multiple commits, final commit removes `-devN`.

385	/// `ContextZC` impl used inside each actor thread.
386	///
387	/// - Holds borrowed references to the actor's peer senders
388	///   and the shared pool.

397	impl<S: BufRefStore> ContextZC<S> for MultiCtxZC<'_, S> {
398	    fn get_msg(&mut self, size: usize) -> Result<PooledMsg<S>, PoolError> {

403	        // Ignored Err contains the un-sent SignalZC::User(msg);

409	/// Per-thread loop with `tprobe` instrumentation.
410	///
411	/// - One scope per `handle_message` (`inner=1`).
412	/// - `Signal::User` records.
413	/// - `Signal::ClearProbe` resets probe + count.
414	/// - `Signal::Shutdown` (or closed channel) exits.
415	fn actor_loop_probed<S: BufRefStore + 'static>(
416	    mut actor: Box<dyn ActorZC<S> + Send>,
417	    rx: Receiver<SignalZC<S>>,
418	    peers: Vec<Sender<SignalZC<S>>>,
419	    pool: Pool<S>,
420	    probe_name: String,
421	) -> (u64, TProbe) {
422	    let mut probe = TProbe::new(&probe_name);
423	    let mut count: u64 = 0;
424	    loop {
425	        match rx.recv() {
426	            Ok(SignalZC::User(msg)) => {
427	                let id = probe.start(0);
428	                let mut ctx = MultiCtxZC {
429	                    senders: &peers,
430	                    pool: &pool,
431	                };
432	                actor.handle_message(&mut ctx, &msg);
433	                probe.end(id);
434	                drop(msg);
435	                count += 1;
436	            }
437	            Ok(SignalZC::ClearProbe) => {
438	                probe.clear();
439	                count = 0;
440	            }
441	            Ok(SignalZC::Shutdown) | Err(_) => break,
442	        }
443	    }
444	    (count, probe)
445	}
446	
447	/// Per-thread loop with no probe involvement.
448	///
449	/// - Identical to [`actor_loop_probed`] minus `probe.start` /
450	///   `probe.end` / `probe.clear`.
451	/// - `_probe_name` is unused but kept in the signature so the
452	///   function pointer matches [`ActorLoopFn`] without
453	///   special-casing.
454	fn actor_loop<S: BufRefStore + 'static>(
455	    mut actor: Box<dyn ActorZC<S> + Send>,
456	    rx: Receiver<SignalZC<S>>,
457	    peers: Vec<Sender<SignalZC<S>>>,
458	    pool: Pool<S>,
459	    _probe_name: String,
460	) -> u64 {
461	    let mut count: u64 = 0;
462	    loop {
463	        match rx.recv() {
464	            Ok(SignalZC::User(msg)) => {
465	                let mut ctx = MultiCtxZC {
466	                    senders: &peers,
467	                    pool: &pool,
468	                };
469	                actor.handle_message(&mut ctx, &msg);
470	                drop(msg);
471	                count += 1;
472	            }
473	            Ok(SignalZC::ClearProbe) => {
474	                count = 0;
475	            }
476	            Ok(SignalZC::Shutdown) | Err(_) => break,
477	        }
478	    }
479	    count
480	}
