   110→    #[arg(short = 'n', long = "names", num_args = 1..)]
   111→    pub name_filter: Vec<String>,
   112→    /// Run only the named test target (use --target-names to see available names; repeatable)
   113→    #[arg(long = "test", value_name = "TARGET-NAME")]
   114→    pub test_target: Vec<String>,
   115→    /// Include #[ignore]'d tests (equivalent to -- --include-ignored)
   116→    #[arg(long = "all-tests")]
   117→    pub all_tests: bool,
   118→    /// Extra arguments passed after -- to the test binary (e.g., --ignored)
   119→    #[arg(last = true)]
   120→    pub test_args: Vec<String>,
   121→}

   123→impl TestCmd {
   124→    fn effective_profile(&self) -> Option<&str> {
   125→        if let Some(ref p) = self.profile {
   126→            Some(p)
   127→        } else if self.release {
   128→            Some("release")
   129→        } else {
   130→            None
   131→        }
   132→    }
   133→}
   134→
   135→impl Execute for TestCmd {
   136→    fn execute(&self, project_root: &Path, flags: &CargoFlags) -> Result<ExitCode> {
   137→        let cli_profile = self.effective_profile();
   138→
   139→        // Resolve package: --workspace > -p/positional PKG > cwd > all
   140→        let resolved = if self.workspace {
   141→            None
   142→        } else {
   143→            match self.positional.as_deref().or(self.package.as_deref()) {
   144→                Some(pkg) => resolve_package_arg(pkg, project_root)?,
   145→                None => current_package_name(project_root),
   146→            }
   147→        };
   148→
   149→        if self.target_names {
   150→            return list_target_names(resolved.as_deref(), project_root, flags);
   151→        }
   152→
   153→        if self.list {
   154→            return list_tests(resolved.as_deref(), &self.name_filter, project_root, flags);
   155→        }
   156→
   157→        // Build extra_args from --names, --test, --all-tests, and trailing args
   158→        let has_extra = !self.name_filter.is_empty()
   159→            || !self.test_target.is_empty()
   160→            || !self.test_args.is_empty()
   161→            || self.all_tests;
   162→        let flags = if has_extra {
   163→            let mut f = flags.clone();
   164→            for target in &self.test_target {
   165→                f.extra_args.push("--test".to_string());
   166→                f.extra_args.push(target.clone());
   167→            }
   168→            // Name filters, --all-tests, and test_args all go after --
   169→            // The test harness accepts [FILTERS...] as OR-matched substrings
   170→            if !self.name_filter.is_empty() || !self.test_args.is_empty() || self.all_tests {
   171→                f.extra_args.push("--".to_string());
   172→                f.extra_args.extend(self.name_filter.clone());
   173→                if self.all_tests {
   174→                    f.extra_args.push("--include-ignored".to_string());
   175→                }
   176→                f.extra_args.extend(self.test_args.clone());
   177→            }
   178→            f
   179→        } else {
   180→            flags.clone()
   181→        };
   182→
   183→        match resolved {
   184→            None => {
   185→                let workspace = WorkspaceInfo::discover(project_root)?;
   186→
   187→                // When --test targets are specified in workspace mode, only test
   188→                // packages that actually have matching files in tests/
   189→                if !self.test_target.is_empty() {
   190→                    let members = workspace.buildable_members();
   191→                    let matching: Vec<_> = members
   192→                        .iter()
   193→                        .filter(|m| {
   194→                            self.test_target
   195→                                .iter()
   196→                                .any(|t| m.path.join("tests").join(format!("{t}.rs")).exists())
   197→                        })
   198→                        .collect();
   199→
   200→                    if matching.is_empty() {
   201→                        eprintln!(
   202→                            "No packages contain test targets: {}",
   203→                            self.test_target.join(", ")
   204→                        );
   205→                        return Ok(ExitCode::from(1));
   206→                    }
   207→
   208→                    for member in &matching {
   209→                        println!("=== {} ===", member.name);
   210→                        test_package(&member.name, None, cli_profile, project_root, &flags)?;
   211→                    }
   212→                    return Ok(ExitCode::SUCCESS);
   213→                }
   214→
   215→                let results =
   216→                    test_all(&workspace, &self.tspec, cli_profile, self.fail_fast, &flags);
   217→                Ok(print_test_summary(workspace.name(), &results))
   218→            }
   219→            Some(name) => {
   220→                let mut all_lines = Vec::new();
   221→                if self.tspec.is_empty() {
   222→                    all_lines = test_package(&name, None, cli_profile, project_root, &flags)?;
   223→                } else {
   224→                    let package_dir = resolve_package_dir(project_root, Some(&name))?;
   225→                    let pkg_name = get_package_name(&package_dir)?;
   226→                    let spec_paths = find_tspecs(&package_dir, &self.tspec)?;
   227→                    for spec_path in &spec_paths {
   228→                        let spec_str = spec_path.to_string_lossy();
   229→                        let lines = test_package(
   230→                            &pkg_name,
   231→                            Some(&spec_str),
   232→                            cli_profile,
   233→                            project_root,
   234→                            &flags,
   235→                        )?;
   236→                        all_lines.extend(lines);
   237→                    }
   238→                }
   239→                let counts = parse_test_results(&all_lines);
   240→                if counts.total_ran() == 0 {
   241→                    eprintln!("Error: 0 tests ran (filter may not match anything)");
   242→                    return Ok(ExitCode::from(1));
   243→                }
   244→                Ok(ExitCode::SUCCESS)
   245→            }
   246→        }
   247→    }
   248→}

   250→/// Format a cargo test target header into a human-friendly label.
   251→///
   252→/// Cargo emits headers like:
   253→/// - `unittests src/lib.rs (target/debug/deps/tspec-abc123)` → unit tests (no --test name)
   254→/// - `tests/tspec_test.rs (target/debug/deps/tspec_test-abc123)` → integration test
   255→///
   256→/// For integration tests, we show the `--test` usable name prominently.
   257→fn format_target_header(raw: &str) -> String {
   258→    // Strip the "(target/...)" suffix for cleaner output
   259→    let label = if let Some(idx) = raw.find(" (") {
   260→        &raw[..idx]
   261→    } else {
   262→        raw
   263→    };
   264→
   265→    if label.starts_with("unittests ") {
   266→        // "unittests src/lib.rs" → unit tests, not selectable with --test
   267→        format!("unit tests ({})", label.strip_prefix("unittests ").unwrap())
   268→    } else if label.starts_with("tests/") {
   269→        // "tests/tspec_test.rs" → extract basename for --test
   270→        let basename = label
   271→            .strip_prefix("tests/")
   272→            .and_then(|s| s.strip_suffix(".rs"))
   273→            .unwrap_or(label);
   274→        format!("{} (--test {})", label, basename)
   275→    } else {
   276→        label.to_string()
   277→    }
   278→}
   279→
   280→/// List available target names for `--test`.
   281→///
   282→/// Runs `cargo test --no-run` and parses stderr for "Running tests/..." lines,
   283→/// extracting the basename (without .rs) as the target name.
   284→fn list_target_names(
   285→    package: Option<&str>,
   286→    project_root: &Path,
   287→    flags: &CargoFlags,
   288→) -> Result<ExitCode> {
   289→    let mut cmd = std::process::Command::new("cargo");
   290→    cmd.arg("test").arg("--no-run");
   291→    if let Some(pkg) = package {
   292→        cmd.arg("-p").arg(pkg);
   293→    }
   294→    flags.apply_to_command(&mut cmd);
   295→    cmd.current_dir(project_root);
   296→
   297→    let output = cmd.output().context("failed to run cargo test --no-run")?;
   298→
   299→    if !output.status.success() {
   300→        let stderr = String::from_utf8_lossy(&output.stderr);
   301→        eprintln!("{}", stderr);
   302→        return Ok(ExitCode::from(1));
   303→    }
   304→
   305→    let stderr = String::from_utf8_lossy(&output.stderr);
   306→    let mut found = false;
   307→
   308→    println!("Available target names for --test:");
   309→    for line in stderr.lines() {
   310→        let trimmed = line.trim();
   311→        // --no-run emits "Executable tests/foo.rs (target/...)" for integration tests
   312→        if let Some(rest) = trimmed
   313→            .strip_prefix("Executable tests/")
   314→            .or_else(|| trimmed.strip_prefix("Running tests/"))
   315→        {
   316→            let file = if let Some(idx) = rest.find(" (") {
   317→                &rest[..idx]
   318→            } else {
   319→                rest
   320→            };
   321→            if let Some(name) = file.strip_suffix(".rs") {
   322→                println!("  {}    (tests/{})", name, file);
   323→                found = true;
   324→            }
   325→        }
   326→    }
   327→
   328→    if !found {
   329→        println!("  (none — no integration test files in tests/)");
   330→    }
   331→
   332→    Ok(ExitCode::SUCCESS)
   333→}

   335→/// Run `cargo test -- --list` and format the output.
   336→///
   337→/// Groups test functions under their target headers, showing counts per target
   338→/// and a total. Skips targets with zero tests.
   339→fn list_tests(
   340→    package: Option<&str>,
   341→    name_filter: &[String],
   342→    project_root: &Path,
   343→    flags: &CargoFlags,
   344→) -> Result<ExitCode> {
   345→    let mut cmd = std::process::Command::new("cargo");
   346→    cmd.arg("test");
   347→    if let Some(pkg) = package {
   348→        cmd.arg("-p").arg(pkg);
   349→    }
   350→    flags.apply_to_command(&mut cmd);
   351→    cmd.arg("--");
   352→    cmd.args(name_filter);
   353→    cmd.arg("--list");
   354→    cmd.current_dir(project_root);
   355→
   356→    let output = cmd.output().context("failed to run cargo test -- --list")?;
   357→
   358→    if !output.status.success() {
   359→        let stderr = String::from_utf8_lossy(&output.stderr);
   360→        eprintln!("{}", stderr);
   361→        return Ok(ExitCode::from(1));
   362→    }
   363→
   364→    let stderr = String::from_utf8_lossy(&output.stderr);
   365→    let stdout = String::from_utf8_lossy(&output.stdout);
   366→
   367→    // Parse target headers from stderr ("Running ..." and "Doc-tests ..." lines)
   368→    // Produce human-friendly labels that show the --test name when applicable.
   369→    let targets: Vec<String> = stderr
   370→        .lines()
   371→        .filter_map(|line| {
   372→            let trimmed = line.trim();
   373→            if trimmed.starts_with("Running ") {
   374→                let rest = trimmed.strip_prefix("Running ").unwrap();
   375→                Some(format_target_header(rest))
   376→            } else if trimmed.starts_with("Doc-tests ") {
   377→                Some(trimmed.to_string())
   378→            } else {
   379→                None
   380→            }
   381→        })
   382→        .collect();
   383→
   384→    // Parse stdout into groups split by "N tests, M benchmarks" summary lines
   385→    let mut groups: Vec<Vec<&str>> = Vec::new();
   386→    let mut current: Vec<&str> = Vec::new();
   387→    for line in stdout.lines() {
   388→        if line.contains("tests,") && line.contains("benchmarks") {
   389→            groups.push(std::mem::take(&mut current));
   390→        } else if line.ends_with(": test") || line.ends_with(": bench") {
   391→            current.push(line);
   392→        }
   393→    }
   394→
   395→    // Print grouped output
   396→    let mut total = 0;
   397→    for (i, tests) in groups.iter().enumerate() {
   398→        if tests.is_empty() {
   399→            continue;
   400→        }
   401→        let unknown = "unknown target".to_string();
   402→        let header = targets.get(i).unwrap_or(&unknown);
   403→        println!("{} ({} tests)", header, tests.len());
   404→        for test in tests {
   405→            // Strip ": test" / ": bench" suffix for cleaner output
   406→            let name = test
   407→                .strip_suffix(": test")
   408→                .or_else(|| test.strip_suffix(": bench"))
   409→                .unwrap_or(test);
   410→            println!("  {}", name);
   411→        }
   412→        total += tests.len();
   413→        println!();
   414→    }
   415→
   416→    println!("{} tests total", total);
   417→    Ok(ExitCode::SUCCESS)
   418→}
