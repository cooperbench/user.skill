3	//! The VN-100 speaks an ASCII protocol over a serial port. Each command is

8	//! (i.e. everything between `$` and `*`). Replies echo the same form; binary

13	//!   baud                  change the serial baud rate          (register 5)
14	//!   rrg / wrg             generic read/write of any register   ($VNRRG / $VNWRG)

138	/// Verify the trailing `*XX` checksum of a received `$...*XX` line.
139	fn verify_checksum(line: &str) -> Result<(), String> {
140	    let payload = line.strip_prefix('$').ok_or("reply missing leading '$'")?;
141	    let (payload, sum) = payload
142	        .rsplit_once('*')
143	        .ok_or("reply missing '*' checksum delimiter")?;
144	    let given = u8::from_str_radix(sum.trim(), 16)
145	        .map_err(|_| format!("malformed checksum field {sum:?}"))?;
146	    let actual = checksum(payload);
147	    if given == actual {
148	        Ok(())
149	    } else {
150	        Err(format!(
151	            "checksum mismatch: reply says {given:02X}, computed {actual:02X}"
152	        ))
153	    }
154	}

568	            let secs = secs.unwrap_or(5);
569	            if binary {
570	                if type_arg.is_some() {
571	                    return Err(
572	                        "--type only applies to the ASCII bench (binary picks data with --fields)"
573	                            .into(),

575	                }
576	                if hz == 0 || 800 % hz != 0 {
577	                    return Err(format!(
578	                        "--hz {hz} invalid for --bin; the binary rate is 800/divisor, so HZ must \
579	                         divide 800 (e.g. 50, 100, 200, 400)"
580	                    ));
581	                }
582	                let fields = match &fields_arg {
583	                    Some(list) => parse_fields(list)?,
584	                    None => default_fields(),
585	                };
586	                Command::Bench {
587	                    binary: true,
588	                    hz,
589	                    secs,
590	                    fields,
591	                    ascii_type: None,
592	                }
593	            } else {
594	                if fields_arg.is_some() {
595	                    return Err(
596	                        "--fields only applies with --bin (ASCII async uses preset messages, \
597	                         not arbitrary fields)"
598	                            .into(),
599	                    );
600	                }
601	                let ascii_type = match &type_arg {
602	                    Some(t) => Some(parse_ascii_type(t)?),
603	                    None => None,
604	                };
605	                if !VALID_RATES.contains(&hz) {
606	                    return Err(format!(
607	                        "--hz {hz} not valid for the ASCII async output; choose one of \
608	                         {VALID_RATES:?} (or use --bin)"
609	                    ));
610	                }
611	                Command::Bench {
612	                    binary: false,
613	                    hz,
614	                    secs,
615	                    fields: Vec::new(),
616	                    ascii_type,
617	                }
618	            }
619	        }
620	        Some(other) => return Err(format!("unknown command `{other}`")),
621	        None => {
622	            return Err(
623	                "missing command (`get-hz`, `set-hz`, `baud`, `rrg`, `wrg`, `bench`, `reset`, \
624	                 `factory-reset`, or `help`)"

626	            )

633	/// Read from the device until `matches` accepts a line or the deadline passes.
634	///
635	/// Reads raw bytes (not `read_line`) so a stream of garbage with no newline —
636	/// e.g. when the host baud doesn't match the device — can't block us forever:
637	/// we honor an overall `deadline` and cap line length to drop runaway junk.
638	fn read_reply<R, F>(
639	    reader: &mut R,
640	    deadline: Instant,
641	    mut matches: F,
642	) -> std::io::Result<Option<String>>
643	where
644	    R: Read,
645	    F: FnMut(&str) -> bool,
646	{
647	    let mut buf = [0u8; 256];
648	    let mut line: Vec<u8> = Vec::new();
649	    loop {
650	        if Instant::now() >= deadline {
651	            return Ok(None);
652	        }
653	        let n = match reader.read(&mut buf) {
654	            // On a serial port there is no real EOF: a read that yields nothing
655	            // (Ok(0), or a TimedOut/WouldBlock error) just means "no data within
656	            // this read window". Keep waiting — the overall `deadline` is the only
657	            // terminator, so a reply that lags (e.g. while the device reconfigures
658	            // its UART for a baud change) isn't dropped.
659	            Ok(0) => continue,
660	            Ok(n) => n,
661	            Err(ref e)
662	                if matches!(
663	                    e.kind(),
664	                    std::io::ErrorKind::TimedOut | std::io::ErrorKind::WouldBlock
665	                ) =>
666	            {
667	                continue
668	            }
669	            Err(e) => return Err(e),
670	        };
671	        for &b in &buf[..n] {
672	            match b {
673	                b'\n' => {
674	                    let raw = String::from_utf8_lossy(&line);
675	                    // An ASCII reply ($VN...*XX) can arrive with leading bytes on
676	                    // the same line — e.g. binary frames still streaming when the
677	                    // echo lands. The reply starts at the last '$', so slice there
678	                    // before matching/validating.
679	                    let candidate: String = match raw.rfind('$') {
680	                        Some(p) => raw[p..].trim().to_string(),
681	                        None => raw.trim().to_string(),
682	                    };
683	                    line.clear();
684	                    if matches(&candidate) {
685	                        return Ok(Some(candidate));
686	                    }
687	                }
688	                b'\r' => {}
689	                _ => {
690	                    line.push(b);
691	                    if line.len() > 1024 {
692	                        line.clear(); // drop a runaway (likely garbage) line
693	                    }
694	                }
695	            }
696	        }
697	    }
698	}

700	/// Send `cmd`, wait up to `wait` for a reply matching `accept`, and validate it.
701	/// A `$VNERR` reply is surfaced as an error; a checksum mismatch is rejected.
702	fn transact<S: Read + Write>(
703	    port: &mut S,
704	    cmd: &str,
705	    wait: Duration,
706	    accept: impl Fn(&str) -> bool,
707	    missing: &str,
708	) -> Result<String, Box<dyn std::error::Error>> {
709	    print!("TX: {cmd}");
710	    port.write_all(cmd.as_bytes())?;
711	    port.flush()?;
712	
713	    let deadline = Instant::now() + wait;
714	    let reply = read_reply(port, deadline, |l| accept(l) || l.starts_with("$VNERR"))?
715	        .ok_or_else(|| missing.to_string())?;
716	    verify_checksum(&reply)?;
717	    if reply.starts_with("$VNERR") {
718	        return Err(vnerr_message(&reply).into());
719	    }
720	    Ok(reply)
721	}

723	/// Like `transact`, but resends the command up to `attempts` times. A fresh
724	/// open (especially at high baud) can drop the first query or its reply while
725	/// the USB-serial chip locks its divisor and open-time line noise drains, so a
726	/// single shot is unreliable. A device `$VNERR` is returned immediately —
727	/// retrying a rejected command won't change the answer.
728	fn transact_retry<S: Read + Write>(
729	    port: &mut S,
730	    cmd: &str,
731	    attempts: u32,
732	    accept: impl Fn(&str) -> bool,
733	    missing: &str,
734	) -> Result<String, Box<dyn std::error::Error>> {
735	    let mut last: Option<Box<dyn std::error::Error>> = None;
736	    for attempt in 1..=attempts {
737	        match transact(
738	            port,
739	            cmd,
740	            Duration::from_millis(800),
741	            |l| accept(l),
742	            "no reply yet",
743	        ) {
744	            Ok(reply) => return Ok(reply),
745	            // A device-side error is a definitive answer; don't keep retrying it.
746	            Err(e) if e.to_string().starts_with("device error") => return Err(e),
747	            Err(e) => {
748	                last = Some(e);
749	                if attempt < attempts {
750	                    println!("  attempt {attempt}/{attempts}: no response yet, retrying...");
751	                    std::thread::sleep(Duration::from_millis(100));
752	                }
753	            }
754	        }
755	    }
756	    Err(format!(
757	        "{missing} (after {attempts} attempts; last: {})",
758	        last.map(|e| e.to_string()).unwrap_or_default()
759	    )
760	    .into())
761	}

764	/// may or may not arrive before the reset, so a missing reply is NOT an error;
765	/// only a `$VNERR` (e.g. unauthorized) is surfaced.
766	fn send_reboot_command<S: Read + Write>(
767	    port: &mut S,
768	    cmd: &str,
769	    accept: impl Fn(&str) -> bool,
770	) -> Result<(), Box<dyn std::error::Error>> {
771	    print!("TX: {cmd}");
772	    port.write_all(cmd.as_bytes())?;
773	    port.flush()?;
774	
775	    let deadline = Instant::now() + Duration::from_millis(1500);
776	    match read_reply(port, deadline, |l| accept(l) || l.starts_with("$VNERR"))? {
777	        Some(reply) if reply.starts_with("$VNERR") => Err(vnerr_message(&reply).into()),
778	        Some(reply) => {
779	            println!("RX: {reply}");
780	            Ok(())
781	        }
782	        None => {
783	            println!("(no echo — device likely rebooted before replying, which is normal)");
784	            Ok(())
785	        }
786	    }
787	}

789	/// VectorNav 16-bit CRC (CRC-CCITT/XMODEM, the algorithm from their app note).
790	/// A valid binary packet, run from the groups byte through the trailing CRC,
791	/// produces 0. Authoritative: REFERENCE.md "Framing & checksums" (ICD §1.4).
792	fn vn_crc16(data: &[u8]) -> u16 {
793	    let mut crc: u16 = 0;
794	    for &b in data {
795	        crc = crc.rotate_left(8);
796	        crc ^= b as u16;
797	        crc ^= (crc & 0xff) >> 4;
798	        crc ^= crc << 12;
799	        crc ^= (crc & 0x00ff) << 5;
800	    }
801	    crc
802	}

804	// Binary frame: sync 0xFA, groups=0x01 (Common), one 16-bit field mask, the
805	// selected fields' payload, then a 2-byte CRC. Length depends on the field set.
806	const BENCH_SYNC: u8 = 0xFA;
807	const BENCH_GROUPS: u8 = 0x01;
808	const BENCH_HEADER: usize = 1 + 1 + 2; // sync + groups + field mask
809	const BENCH_CRC: usize = 2;

814	/// Read a little-endian f32 from `buf` at `off`.
815	fn rd_f32(buf: &[u8], off: usize) -> f32 {
816	    f32::from_le_bytes(buf[off..off + 4].try_into().unwrap())
817	}

833	                rd_f32(frame, off + 8)
834	            ),
835	            "quat" => format!(

845	                rd_f32(frame, off + 4),

847	            ),
848	            "accel" => format!(

865	                rd_f32(frame, off),

879	/// Measure binary frames of `frame_len` bytes for `secs`. Counts CRC-valid
880	/// frames and captures the first frame's raw bytes for the caller to decode.
881	fn measure_binary<S: Read>(
882	    port: &mut S,
883	    frame_len: usize,
884	    secs: u64,
885	) -> std::io::Result<(u64, u64, f64, Option<Vec<u8>>)> {
886	    let start = Instant::now();
887	    let deadline = start + Duration::from_secs(secs);
888	    let mut buf = [0u8; 1024];
889	    let mut acc: Vec<u8> = Vec::new();
890	    let mut frames = 0u64;
891	    let mut first: Option<Vec<u8>> = None;
892	
893	    while Instant::now() < deadline {
894	        match port.read(&mut buf) {
895	            Ok(0) => continue,
896	            Ok(n) => acc.extend_from_slice(&buf[..n]),
897	            Err(ref e)
898	                if matches!(
899	                    e.kind(),
900	                    std::io::ErrorKind::TimedOut | std::io::ErrorKind::WouldBlock
901	                ) =>
902	            {
903	                continue
904	            }
905	            Err(e) => return Err(e),
906	        }
907	
908	        let mut i = 0;
909	        while i + frame_len <= acc.len() {
910	            if acc[i] != BENCH_SYNC || acc[i + 1] != BENCH_GROUPS {
911	                i += 1;
912	                continue;
913	            }
914	            let frame = &acc[i..i + frame_len];
915	            // CRC over everything after the sync byte (groups..payload..crc) == 0.
916	            if vn_crc16(&frame[1..]) == 0 {
917	                frames += 1;
918	                if first.is_none() {
919	                    first = Some(frame.to_vec());
920	                }
921	                i += frame_len;
922	            } else {
923	                i += 1; // false sync (0xFA can appear in payload); resync
924	            }
925	        }
926	        acc.drain(0..i);
927	        if acc.len() > 8192 {
928	            let keep = acc.len() - frame_len;
929	            acc.drain(0..keep);
930	        }
931	    }
932	
933	    Ok((
934	        frames,
935	        frames * frame_len as u64,
936	        start.elapsed().as_secs_f64(),
937	        first,
938	    ))
939	}

941	/// True if `line` is an async data message (a `$VN...` line that isn't a command
942	/// echo or error) with a valid checksum.
943	fn is_async_line(line: &str) -> bool {
944	    line.starts_with("$VN")
945	        && !line.starts_with("$VNRRG")
946	        && !line.starts_with("$VNWRG")
947	        && !line.starts_with("$VNERR")
948	        && verify_checksum(line).is_ok()
949	}

955	    let deadline = start + Duration::from_secs(secs);
956	    let mut buf = [0u8; 1024];
957	    let mut line: Vec<u8> = Vec::new();
958	    let mut msgs = 0u64;
959	    let mut bytes = 0u64;
960	    let mut sample = None;
961	
962	    while Instant::now() < deadline {
963	        let n = match port.read(&mut buf) {
964	            Ok(0) => continue,
965	            Ok(n) => n,
966	            Err(ref e)
967	                if matches!(
968	                    e.kind(),
969	                    std::io::ErrorKind::TimedOut | std::io::ErrorKind::WouldBlock
970	                ) =>
971	            {
972	                continue
973	            }
974	            Err(e) => return Err(e),
975	        };
976	        for &b in &buf[..n] {
977	            match b {
978	                b'\n' => {

1033	/// Configure a binary output (reg 75) with `fields` at `hz`, measure the frame
1034	/// rate for `secs`, then restore the prior state.
1035	fn bench_binary<S: Read + Write>(
1036	    port: &mut S,
1037	    baud: u32,
1038	    hz: u32,
1039	    secs: u64,
1040	    fields: &[&Field],
1041	) -> Result<(), Box<dyn std::error::Error>> {
1042	    let divisor = 800 / hz; // device IMU base rate is 800 Hz
1043	    let mask: u16 = fields.iter().fold(0, |m, f| m | (1u16 << f.bit));
1044	    let payload: usize = fields.iter().map(|f| f.size).sum();
1045	    let frame_len = BENCH_HEADER + payload + BENCH_CRC;
1046	    let names: Vec<&str> = fields.iter().map(|f| f.name).collect();
1047	
1048	    // Remember the current ASCII async rate so we can put it back.
1049	    let prev = transact_retry(
1050	        port,
1051	        &build_command("VNRRG,07"),
1052	        5,
1053	        |l| parse_reg07(l).is_some(),
1054	        "could not read current async rate",
1055	    )?;
1056	    let prev_hz = parse_reg07(&prev).unwrap();
1057	
1058	    // Configure Binary Output 1 (reg 75) on serial1. A $VNERR here means the
1059	    // chosen fields+rate don't fit the current baud — nothing else has changed.
1060	    let cfg = format!("VNWRG,75,1,{divisor},01,{mask:04X}");
1061	    transact_retry(
1062	        port,
1063	        &build_command(&cfg),
1064	        5,
1065	        |l| l.starts_with("$VNWRG,75"),
1066	        "device did not accept the binary config (a $VNERR means it won't fit at this baud)",
1067	    )?;
1068	    println!(
1069	        "Configured binary output: Common{names:?} @ {hz} Hz (divisor {divisor}, {frame_len} B/frame)."
1070	    );
1071	
1072	    // Silence the ASCII async output so we measure ONLY the binary stream.
1073	    transact_retry(
1074	        port,
1075	        &build_command("VNWRG,07,0"),
1076	        5,
1077	        |l| parse_reg07(l).is_some(),
1078	        "could not disable ASCII async output",
1079	    )?;
1080	
1081	    println!("Measuring for {secs}s...");
1082	    let (frames, bytes, elapsed, first) = measure_binary(port, frame_len, secs)?;
1083	    let sample = first.map(|fr| decode_binary_sample(&fr, fields));
1084	    report_bench("frames", hz, (frames, bytes, elapsed, sample), baud);
1085	
1086	    // Restore: turn the binary output off, put the ASCII rate back.
1087	    let _ = transact_retry(
1088	        port,
1089	        &build_command(&format!("VNWRG,75,0,{divisor},01,{mask:04X}")),
1090	        3,
1091	        |l| l.starts_with("$VNWRG,75"),
1092	        "restore: disable binary output",
1093	    );
1094	    let _ = transact_retry(
1095	        port,
1096	        &build_command(&format!("VNWRG,07,{prev_hz}")),
1097	        3,
1098	        |l| parse_reg07(l).is_some(),
1099	        "restore: ASCII async rate",
1100	    );
1101	    println!("Restored: binary output off, ASCII async back to {prev_hz} Hz.");
1102	
1103	    if frames == 0 {
1104	        return Err(
1105	            "received 0 binary frames — the config may have targeted the wrong serial port \
1106	             (try a VN-100 on serial1), or the device isn't streaming"
1107	                .into(),
1108	        );
1109	    }
1110	    Ok(())
1111	}

1175	    );

1201	/// Print the fields of a `$VN(R|W)RG,<id>,<f1>,<f2>...*XX` reply.
1202	fn print_reg_fields(reply: &str) {
1203	    let body = reply.trim_start_matches('$');
1204	    let body = body.split('*').next().unwrap_or(body);
1205	    let parts: Vec<&str> = body.split(',').collect();
1206	    match parts.as_slice() {
1207	        [_, id, fields @ ..] if !fields.is_empty() => {
1208	            println!("register {id}: {fields:?}");
1209	        }
1210	        [_, id] => println!("register {id}: (no fields)"),
1211	        _ => println!("(unrecognized reply)"),
1212	    }
1213	}

1215	fn main() -> Result<(), Box<dyn std::error::Error>> {
1216	    let (config, command) = match parse_args(std::env::args().skip(1)) {
1217	        Ok(parsed) => parsed,
1218	        Err(e) => {
1219	            eprintln!("error: {e}\n");
1220	            eprint!("{}", help_text());
1221	            std::process::exit(2);
1222	        }
1223	    };
1224	
1225	    if let Command::Help = command {
1226	        print!("{}", help_text());
1227	        return Ok(());
1228	    }
1229	
1230	    println!("Opening {} at {} baud...", config.port, config.baud);
1231	    let mut port = serialport::new(&config.port, config.baud)
1232	        // Short per-read timeout so read_reply re-checks its overall deadline often.
1233	        .timeout(Duration::from_millis(250))
1234	        .open()?;
1235	
1236	    // Let the USB-serial chip lock its baud divisor, then drop any open-time
1237	    // line noise / partial async frame before the first command. Without this,
1238	    // a fresh open at high baud often loses its first query.
1239	    std::thread::sleep(Duration::from_millis(150));
1240	    let _ = port.clear(serialport::ClearBuffer::Input);
1241	
1242	    // Shared "we heard nothing usable" hint — usually a baud mismatch.
1243	    let no_reply = format!(
1244	        "no usable reply from device — is it actually at {} baud? \
1245	         (VN-100 factory default is 115200; use --baud to match, or the `baud` command to change it)",
1246	        config.baud
1247	    );
1248	
1249	    match command {
1250	        Command::Help => unreachable!("handled above"),
1251	
1252	        Command::GetHz => {
1253	            let reply = transact_retry(
1254	                &mut port,
1255	                &build_command("VNRRG,07"),
1256	                5,
1257	                |l| parse_reg07(l).is_some(),
1258	                &no_reply,
1259	            )?;
1260	            println!("RX: {reply}");
1261	            println!("Async output rate: {} Hz", parse_reg07(&reply).unwrap());
1262	        }
1263	
1264	        Command::SetHz { hz, persist } => {
1265	            let reply = transact_retry(
1266	                &mut port,
1267	                &build_command(&format!("VNWRG,07,{hz}")),
1268	                5,
1269	                |l| parse_reg07(l).is_some(),
1270	                &no_reply,
1271	            )?;
1272	            println!("RX: {reply}");
1273	            println!("Async output rate: {} Hz", parse_reg07(&reply).unwrap());
1274	
1275	            if persist {
1276	                let confirm = transact_retry(
1277	                    &mut port,
1278	                    &build_command("VNWNV"),
1279	                    5,
1280	                    |l| l.starts_with("$VNWNV"),
1281	                    &no_reply,
1282	                )?;
1283	                println!("RX: {confirm}");
1284	                println!("Settings written to non-volatile memory.");
1285	            }
1286	        }
1287	
1288	        Command::SetBaud {
1289	            baud: new_baud,
1290	            persist,
1291	        } => {
1292	            // The device replies at the CURRENT baud, then switches to the new one.
1293	            let reply = transact_retry(
1294	                &mut port,
1295	                &build_command(&format!("VNWRG,05,{new_baud}")),
1296	                5,
1297	                |l| l.starts_with("$VNWRG,05,"),
1298	                &no_reply,
1299	            )?;
1300	            println!("RX: {reply}");
1301	            println!("Device acknowledged baud change to {new_baud}.");
1302	
1303	            // Give the device a moment to reconfigure its UART before we talk at
1304	            // the new rate (the vendor SDK waits ~50 ms here), then switch THIS
1305	            // connection in place. We switch in-session rather than close/reopen
1306	            // not because the device would forget the baud — it holds the RAM
1307	            // value across host reconnects — but because each reconnect risks a
1308	            // line transient that, at very high baud (e.g. 921600 on an FT232R),
1309	            // can wedge the link until a power cycle.
1310	            std::thread::sleep(Duration::from_millis(60));
1311	            port.set_baud_rate(new_baud)?;
1312	            // Drop any bytes that were in flight across the switch.
1313	            let _ = port.clear(serialport::ClearBuffer::Input);
1314	
1315	            println!("Verifying at {new_baud} baud...");
1316	            let verify = transact_retry(
1317	                &mut port,
1318	                &build_command("VNRRG,07"),
1319	                5,
1320	                |l| parse_reg07(l).is_some(),
1321	                "device did not respond at the new baud (a power cycle reverts to the flash baud)",
1322	            )?;
1323	            println!("RX: {verify}");
1324	            println!(
1325	                "Verified — device is at {new_baud} baud (async rate {} Hz).",
1326	                parse_reg07(&verify).unwrap()
1327	            );
1328	
1329	            if persist {
1330	                let confirm = transact_retry(
1331	                    &mut port,
1332	                    &build_command("VNWNV"),
1333	                    5,
1334	                    |l| l.starts_with("$VNWNV"),
1335	                    "no $VNWNV confirmation at the new baud",
1336	                )?;
1337	                println!("RX: {confirm}");
1338	                println!("Baud saved to flash; the device will boot at {new_baud} from now on.");
1339	            } else {
1340	                println!(
1341	                    "(Volatile — the device holds this across host reconnects, but a \
1342	                     power cycle or `reset`/`factory-reset` reverts it to the flash baud. \
1343	                     Re-run with `baud {new_baud} --persist` to make it permanent.)"
1344	                );
1345	            }
1346	        }
1347	
1348	        Command::Reset => {
1349	            send_reboot_command(&mut port, &build_command("VNRST"), |l| {
1350	                l.starts_with("$VNRST")
1351	            })?;
1352	            println!("Reset requested — sensor is rebooting and reloading its saved settings.");
1353	        }
1354	
1355	        Command::Rrg { id } => {
1356	            let reply = transact_retry(
1357	                &mut port,
1358	                &build_command(&format!("VNRRG,{id:02}")),
1359	                5,
1360	                |l| l.starts_with("$VNRRG,"),
1361	                &no_reply,
1362	            )?;
1363	            println!("RX: {reply}");
1364	            print_reg_fields(&reply);
1365	        }
1366	
1367	        Command::Wrg { id, params } => {
1368	            let payload = format!("VNWRG,{id:02},{}", params.join(","));
1369	            let reply = transact_retry(
1370	                &mut port,
1371	                &build_command(&payload),
1372	                5,
1373	                |l| l.starts_with("$VNWRG,"),
1374	                &no_reply,
1375	            )?;
1376	            println!("RX: {reply}");
1377	            print_reg_fields(&reply);
1378	        }
1379	
1380	        Command::Bench {
1381	            binary,
1382	            hz,
1383	            secs,
1384	            fields,
1385	            ascii_type,
1386	        } => {
1387	            if binary {
1388	                bench_binary(&mut port, config.baud, hz, secs, &fields)?;
1389	            } else {
1390	                bench_ascii(&mut port, config.baud, hz, secs, ascii_type)?;
1391	            }
1392	        }
1393	
1394	        Command::FactoryReset => {
1395	            println!("Restoring factory defaults — this overwrites flash and cannot be undone.");
1396	            send_reboot_command(&mut port, &build_command("VNRFS"), |l| {
1397	                l.starts_with("$VNRFS")
1398	            })?;
1399	            println!("Factory restore requested — sensor is rebooting.");
1400	            println!(
1401	                "It is now at 115200 baud with the default async output. \
1402	                 Reconnect with the default --baud (115200)."
1403	            );
1404	        }
1405	    }
1406	
1407	    Ok(())
1408	}

207	## Prose form
208	
209	Long-lived prose on this project follows one basic shape: a short intro
210	that explains the *why* or the high-level *what*, then a `-` bullet
211	list for the details. Wrap lines at ≤72 cols (bullet continuations
212	indent two spaces). Avoid wall-of-prose paragraphs — they hide the
213	structure that bullets make scannable.
214	
215	Surfaces that use this shape:
216	
217	- Module / function / struct / field doc comments in `.rs` files —
218	  see [Doc comments](#doc-comments-on-every-file-function-and-method).
219	- Commit message bodies (both app-repo and session-repo). The
220	  ≤50-col title is the commit-specific add-on; see
221	  [Per-commit flow](notes/cycle-protocol.md#per-commit-flow).
222	- Chore descriptions in `notes/chores/chores-NN.md` — see
223	  [Chores section content](#chores-section-content--no-edit-list-git-is-the-record).
224	- Todo and Done entries in `notes/todo.md` when an entry needs more
225	  than one line of detail. Pure one-liners are still fine.
226	
227	Bullet *content* differs by surface:
228	
229	- **Commit bodies** — bullets are file-by-file: one bullet per file
230	  changed, file plus a one-line gist (e.g.
231	  `README.md: new Overview intro`). Source of truth for the
232	  mechanical edit list.
233	- **Chores / todo / done** — bullets are conceptual (design points,
234	  structural notes, the "what landed and why" at a notch above
235	  file-list granularity). Never a copy of the commit's edit list —
236	  see [Chores section content](#chores-section-content--no-edit-list-git-is-the-record).
237	- **Doc comments** — bullets are whatever structure fits (fields,
238	  cases, invariants).
239	
240	### Problem + plan shape
241	
242	`## In Progress` cycle blocks, chores section intros, and `## Todo`
243	entries use a sharper form of the same shape:
244	
245	- **Problem statement** (the why) — one or two sentences;
246	  don't pad with intent, don't restate the plan.
247	- **Plan bullets** (the what/when) — formality differs by
248	  surface:
249	  - In Progress / chores: a committed ladder, one step per
250	    commit — see
251	    [Conventional-commit shape](#conventional-commit-shape-ladder--chores--commit)
252	    for the per-step title + `(current)` / `(done)` form.
253	  - Todo entries: rough informal bullets, no numbering;
254	    formalized only when the entry is picked up into a
255	    cycle.
256	
257	### Semicolons inside bullets
258	
259	A bullet that joins multiple clauses with semicolons (`A; B; C`)
260	is a list hiding inside running prose — break the clauses into
261	sub-bullets so the structure shows. Semicolons in running prose
262	(intro paragraphs, sentence-joins) are fine. Not absolute: very
263	short clauses or tight pairs can stay joined inside a bullet when
264	breaking would be more noise than signal.
265	
266	### Conventional-commit shape (ladder / chores / commit)
267	
268	A ladder step, its chores section, and its commit description
269	share a *title* shape — a
270	[Conventional Commits](https://www.conventionalcommits.org/en/v1.0.0/)
271	title over [Prose form](#prose-form) detail. They differ in the
272	title's prefix / marker (below) and in bullet *content* — commit
273	bodies are file-by-file, ladder / chores conceptual (see "Bullet
274	*content* differs by surface" above). The shared template:
275	
276	```
277	<optional - X.Y.Z[-N]> <title>   # <title> is the commit's `<type>: <desc>`
278	<optional prose intro>
279	  - <optional item>
280	    <optional prose intro>
281	      - <optional sub-item>
282	      ...
283	```
284	
285	The three surfaces apply it as:
286	
287	- **Ladder step** (`notes/todo.md` `## In Progress`): the title is
288	  prefixed with the step's version — `X.Y.Z-N <title>` — and
289	  carries a `(current)` / `(done)` marker. The bare three-element
290	  `X.Y.Z` (no `-N`) is the close-out step. Detail is bulleted,
291	  never `;`-joined inline.
292	- **Chores section** (`notes/chores/chores-NN.md`): no version
293	  prefix — the `##` header *is* the bare title, with a
294	  `Commits: [[ref]]` line first under it (empty until backfilled —
295	  see [Chores commit references](#chores-commit-references)).
296	- **Commit description**: no version prefix — the title is the
297	  ≤50-col first line; the body is the prose (file-by-file for the
298	  app repo, per [Per-commit flow](notes/cycle-protocol.md#per-commit-flow)).
299	
300	The title is **identical** across all three for a given step, so a
301	step's ladder entry, its chores `##` header, and its commit title
302	line up verbatim — pick the commit title first and reuse it.
303	
304	## Notes file conventions
305	
306	Conventions the bot follows when writing notes files
307	(`notes/todo.md`, `notes/todo-backlog.md`, `notes/bugs.md`,
308	`notes/chores/chores-NN.md`, `notes/done.md`). One source
309	of truth lives here; [`notes/README.md`](notes/README.md)
310	points back.
311	
312	### Notes references
313	
314	Reference *citations* are double-bracketed so the brackets render
315	— `[[N]]`, or `[[2]],[[3]]` for several (comma-separated, not
316	`[2,3]` or `[[2]][[3]]`). The `[N]:` definitions in a file's
317	`# References` section and inline `[text](url)` / `[text](#anchor)`
318	links stay single-bracketed.
319	
320	### Reference numbering
321	
322	Every note file (`todo.md`, `todo-backlog.md`, `bugs.md`,
323	`chores-NN.md`, `done.md`) keeps a file-local `# References`
324	section at the bottom. Reference numbers are scoped to that
325	file — `[1]` in `chores-07.md` and `[1]` in `chores-01.md`
326	are independent slots that may point at completely different
327	URLs. New chores files start their numbering at `[1]`.
328	
329	Treat `[N]` like a **footnote**: the number is a local slot, only
330	meaningful within its file's `# References`. So a `[N]` *citation*
331	(bare `[N]`, or doubled `[[N]]`) never reuses another file's number; to cite a target a sibling
332	file references, pick your own next-local slot and define it
333	(the same target may carry a different number in each file). A
334	`[N]` *inside a code span* (`` `[72]` ``) is different — that's
335	a quoted identifier: literal text naming a ref-key (often from
336	another file's namespace), data, not a citation; it needs no
337	definition here. To point at a section of another file from
338	prose, use an inline link with an anchor —
339	`[that section](../chores-07.md#…)` — not a bare number.
340	
341	A `chores-NN.md` `# References` entry is usually a
342	`/notes/<file>.md#anchor` (or `/ARCHITECTURE.md`) path, but may
343	also be a **commit reference** —
344	`[N]: <commit-url-with-12-hex-SHA> "<full-40-hex-SHA>"`, cited by
345	a section's `Commits:` line. See
346	[Chores commit references](#chores-commit-references) for the
347	why and the exact shape.
348	
349	A file's `# References` can be **re-packed** to a contiguous
350	`[1]..[N]` in first-citation-appearance order — walk the file's
351	prose in document order (`todo.md`: `## Todo` then `## Done`;
352	`chores-NN.md`: top to bottom) and number refs as their first
353	`[[N]]` citation appears. This is a file-local rewrite — only
354	that file's `[[N]]` citations and `[N]:` definitions move; every
355	target and sibling file is untouched. A `[[N]]` inside a `` ` ``
356	code span is a literal token, not a citation, and is left alone.
357	Do it opportunistically (when the namespace has drifted enough to
358	annoy), not on a schedule: `todo.md` fragments fastest (entries
359	land and get pruned every cycle) and is the usual candidate;
360	`chores-NN.md` / `done.md` are append-mostly and only need it
361	after an unusual event (e.g. a bulk retrofit that allocated slots
362	out of document order).
363	
364	### Markdown anchor links
365	
366	GitHub anchor algorithm: lowercase, strip non-alphanumeric
367	characters in place, map remaining spaces to hyphens 1-for-1. Do
368	**not** collapse adjacent whitespace — so `a + b` → `a--b` (spaces
369	on both sides of `+`), but `a: b` → `a-b` (only trailing space on
370	`:`). General markdown reference:
371	[markdownguide.org](https://www.markdownguide.org). GitHub
372	publishes no official spec for auto-generated anchors; the
373	de-facto reference implementation is
374	[github-slugger](https://github.com/Flet/github-slugger).
375	
376	### Todo format
377	
378	`todo.md` is organized into `## In Progress`, `## Todo`
379	(strict priority rank, #1 highest; long-tail backlog in
380	[todo-backlog.md](notes/todo-backlog.md)), `## Ideas`,
381	`## Bugs` (pointer to [bugs.md](notes/bugs.md)), and
382	`## Done` sections. Each item is a short description with
383	reference links to more detail.
384	
385	`## Todo` and `## Bugs` entries carry explicit `1.` `2.` …
386	numbers in the source. For `## Todo` the number is its
387	**priority rank** (#1 highest, descending); for `## Bugs`
388	it's just an index. They're for grepping and at-a-glance
389	"let's do #1" — **not stable IDs**: reorder (to
390	reprioritize), insert, or delete freely, then
391	`vc-x1 fix-todo --no-dry-run` renumbers and normalizes
392	continuation-line indent, so any given number is positional.
393	**To refer to a Todo durably, name it by its title — a
394	plain, greppable text mention.** Not its number (positional,
395	renumbered) and not a markdown link: a numbered list item has
396	no anchor to link to.
397	
398	Numbering helps a human orient in a long list but makes links
399	difficult and fragile — especially an external reference
400	pointing in, which can't be auto-fixed when the list
401	renumbers. A robust fix (a number-free anchor, or a
402	number-tolerant dereference that matches the title slug and
403	wildcards the numeric prefix — a GitHub slug like `5-foo` is
404	encoded, not opaque, so the title is recoverable) is a
405	`validate-numbering` design question, out of scope here.
406	
407	`vc-x1 fix-todo` alone only previews; `vc-x1 validate-todo`
408	is the read-only check. The `## Done` section keeps `-`
409	bullets — items aren't referenced by number once completed.
410	
411	Example shape:
412	
413	```
414	# Todo
415	- Add new feature X [details](features.md#feature-x)
416	- Fix bug Y [[1]]
417	
418	# Done
419	- Fixed issue Z [[2]],[[3]]
420	
421	[1]: bugs.md#bug-y
422	[2]: issues.md#issue-z
423	[3]: fixes.md#fix-z
424	```
425	
426	### Retiring Done entries
427	
428	`todo.md`'s `## Done` section is a rolling buffer of recently
429	shipped work, not a permanent log. Move entries into `done.md`
430	at two natural beats:
431	
432	- **Closing a ladder** — when the final `X.Y.Z` (no suffix)
433	  commit ships, decide which prior entries are no longer
434	  needed for nearby context and migrate them.
435	- **Opening a new ladder** — at `X.Y.Z-0`, do the same sweep
436	  before bumping the version.
437	
438	Migration mechanics:
439	
440	- Move the bullet itself from `todo.md > ## Done` to
441	  `done.md` (preserving the original ref number).
442	- Copy any references the moved entries cite into
443	  `done.md`'s `# References` section (those refs are
444	  file-local, so coexisting with `todo.md`'s namespace is
445	  fine).
446	- Prune any references in `todo.md > # References` no longer
447	  cited by anything in `## In Progress` / `## Todo` /
448	  `## Done`. This frees the numbers for future reuse.
449	
450	## Chores conventions
451	
452	### Headings and entries that record a commit
453	
454	A commit's title is reused verbatim across its records — see
455	[Conventional-commit shape](#conventional-commit-shape-ladder--chores--commit)
456	for the rule. Beyond the chores `##` header, that same string is
457	used for the matching `todo.md > ## Done` entry and any `[N]`
458	reference to that section. Titles carry **no `(<version>)`
459	suffix** — see
460	[Commit description](notes/cycle-protocol.md#commit-description)
461	for why.
462	E.g. the chores header `## refactor: extract config loader`
463	and the Done line `- refactor: extract config loader [[3]]`.
464	The `## Done` entry uses the close-out commit's title.
465	
466	This does **not** apply to organizational headings (`## Todo`,
467	`## In Progress`, `# References`) or to design `###` subsections
468	inside a chores section — those are named for whatever fits.
469	Among the commit-recording ones, exact match is the strong
470	default (nothing absolute): a near-miss just makes it harder to
471	line a record up with its commit.
472	
473	A commit-recording header is provisional while the work is in
474	progress; the *last* edit before `vc-x1 push` syncs it — and the
475	`## Done` entry / `[N]` anchor for that commit — to the final
476	commit title. See [Markdown anchor links](#markdown-anchor-links)
477	for the slug algorithm; the pre-commit checklist catches a
478	dangling `#anchor`, and a future `vc-x1 validate-repo` should too
479	(and should verify the recorded title matches the commit).
480	
481	Any pre-existing sections and `## Done` entries that predate
482	this convention keep their free-form text; the convention
483	applies going forward.
484	
485	### Chores section content — no edit list; git is the record
486	
487	A chores section is: a `Commits:` line (first line under the
488	header — see below), then [Prose form](#prose-form) (intro +
489	bullets) for what landed and why, and any `###` design
490	subsections. Bullets here are **conceptual** — design points,
491	structural notes — never a per-file edit list. That lives in the
492	commit message body, which is the source of truth for "what
493	changed mechanically" (immutable, `git show`-able, naturally
494	scoped to the commit). The chores section is the source of truth
495	for the design thinking; the two cross-link, neither restates the
496	other.
497	
498	When the intro starts wanting to explain a mechanism,
499	hypothesis, or wrinkle, don't inflate it — promote that to its
500	own `###` subsection inside the same `chores-NN.md`. If the
501	wrinkle is a live design concern (something that *should*
502	change, not just be recorded), also add a `notes/todo.md` item
503	with a `[N]` ref pointing at that subsection (todo→chores is the
504	normal ref direction).
505	
506	**Why:** a chores edit list and the commit body were specified
507	to be the same content in two places — and detail written twice
508	drifts. Git owns the mechanical record; chores owns the
509	narrative; `Commits:` links them.
510	
511	### Chores commit references
512	
513	The first line under a chores section header is a `Commits:`
514	line citing the git commit(s) that section records:
515	
516	```
517	## refactor: extract config loader
518	
519	Commits: [[3]]
520	
521	<intro paragraph...>
522	```
523	
524	`Commits:` uses the file-local `[N]` reference machinery (see
525	[Reference numbering](#reference-numbering)),
526	**double-bracketed** so the brackets render — `Commits: [[3]]`,
527	or `Commits: [[3]],[[5]]` for several. (`[[3]]` shows as a
528	literal `[`, the `[3]` link, then a literal `]`; the inner
529	`[3]` resolves against its `[3]:` definition — CommonMark /
530	GitHub / VS Code all do this.) The `# References` definition
531	puts the **commit URL** as the destination, with the **full
532	40-hex SHA** in the title slot:
533	
534	```
535	[3]: https://github.com/<owner>/<repo>/commit/<12-hex> "<40-hex>"
536	```
537	
538	- The 12-hex short SHA in the URL keeps it short; GitHub /
539	  GitLab resolve a unique prefix to the canonical commit page
540	  (GitLab's path has a `/-/` before `commit/`).
541	- The full SHA in the title is host-agnostic and unambiguous —
542	  it survives a repo host change, `git show <40-hex>` works in
543	  any clone, and external tooling scraping the notes (a
544	  database, say) gets the canonical identifier.
545	
546	**Timing.** A commit's SHA isn't stable until it lands on a
547	**permanent branch** (`main`, or a long-lived release/patch
548	branch) — a rebase or squash rewrites it on the way. A commit
549	can't record its own SHA, so the fill lands one push later:
550	every section opens with an **empty `Commits:`**, and each push
551	backfills the `Commits:` of the commits the previous push made
552	permanent. On a topic branch a section waits until the branch
553	lands. The commit itself is the record, and
554	`git log --grep "<title>"` finds it. See cycle-protocol
555	[Commits backfill](notes/cycle-protocol.md#commits-backfill).
556	
557	## Cycle Protocol
558	
559	Every change runs as a **cycle**: Preparation (`X.Y.Z-0`) →
560	Work commits (`X.Y.Z-1`, `X.Y.Z-2`, …) → Close-out (bare
561	`X.Y.Z`). Each commit runs through a per-commit flow whose
562	cargo cycle is fmt → clippy → test → install (exact commands
563	in cycle-protocol.md; skip-able for purely-docs commits,
564	mandatory at close-out). The full protocol — numbering, per-commit
565	flow, reviewing changes, close-out, pushing, ochid
566	trailers, sub-cycles — lives in
567	[`notes/cycle-protocol.md`](notes/cycle-protocol.md). Read
568	it before any commit work — and before any push, cycle or
569	not: pushes always need per-push user approval (see
570	[Pushing policy](notes/cycle-protocol.md#policy)).
571	
572	**Hard stop after push/finalize.** Once a push or `.claude`
573	finalize is invoked (`vc-x1 push` does both), emit
574	**nothing** — no tool call, no text, no verification, no
575	summary — until the user speaks. Closing words go *before*
576	the invoke; post-push verification happens next turn at the
577	user's direction. See
578	[After push or finalize](notes/cycle-protocol.md#after-push-or-finalize-stop-and-wait).
579	
580	**`vc-x1 push` behaviors to keep in mind.** Two, independent
581	of project language:
582	
583	- **Preflight** runs the cargo cycle (fmt → clippy → test). On
584	  a cargo project it works as-is, so `vc-x1 push <bookmark>`
585	  runs it before the review gate. On a project with no cargo
586	  cycle — or any time preflight can't run — invoke `vc-x1 push
587	  <bookmark> --from message ...` to skip preflight and the
588	  interactive review gate, doing the diff review in
589	  conversation first.
590	- **ochid trailers** are injected by `vc-x1 push` itself —
591	  don't hand-write them into the commit body or
592	  `--title`/`--body`.
593	
594	**Python/uv.** Where a Python/uv project (no cargo) departs
595	from the general protocol:
596	
597	- No cargo cycle, so the no-cargo case above is the norm —
598	  always push with `vc-x1 push <bookmark> --from message ...`.
599	- Version strings are PEP 440, whose local segment can't carry
600	  a `-`: the ladder/chores/todo notation `X.Y.Z-N.M` becomes
601	  `X.Y.Z+N.M` in `fc-config.toml` `[project].version` (e.g.
602	  `0.3.0-5.3.0` -> `0.3.0+5.3.0`). Hyphen when we talk, `+` in
603	  the version string.
604

606	## Code Conventions
607	
608	### Doc comments on every file, function, and method
609	
610	Every `.rs` file must begin with a `//!` module docstring. Every
611	function and method must have a `///` doc comment. Keep them brief —
612	one sentence of purpose is often enough; the discipline is that the
613	comment exists, not that it be long. Doc comments follow the
614	[Prose form](#prose-form) shape (intro + bullets).
615	
616	This is a deliberate override of the generic "write no comments"
617	default that applies to inline `//` comments. Doc comments on the
618	module / item surface are expected; inline explanatory comments
619	inside function bodies remain discouraged unless they capture a
620	non-obvious WHY.
621	
622	**Clap-derive args:** doc comments on `#[arg(...)]` fields drive
623	`--help` output. Clap reflows by default and collapses bullets
624	into running prose. Add `#[arg(verbatim_doc_comment, ...)]` on
625	any field whose doc comment uses bullets so each `- …` lands on
626	its own line in the rendered help.
627	
628	### `// OK: …` comments on `unwrap*` calls (Rust)
629	
630	Non-test code that calls `.unwrap()`, `.unwrap_or(…)`,
631	`.unwrap_or_default()`, or `.unwrap_or_else(…)` must have a trailing
632	`// OK: …` comment that justifies why the call is acceptable.
633	
634	- `// OK: <specific reason>` — document the real precondition,
635	  invariant, or domain reason. Preferred whenever the reason isn't
636	  self-evident.
637	- `// OK: obvious` — the default is self-evident from context (e.g.
638	  `desc.lines().next().unwrap_or("")` — empty desc → empty title).
639	
640	Bare `// OK` is not used (reads like a truncated comment).
641	Abbreviations (e.g. `SE`) are not used because they require a decoder
642	ring for readers seeing the code out of context.
643	
644	For provably-unreachable `.unwrap()` calls, also prefix with
645	`#[allow(clippy::unwrap_used)]` so the site stays silent if we enable
646	the project-wide `clippy::unwrap_used` lint later.
647	
648	```rust
649	let max = stderr_level.unwrap_or(LevelFilter::Info); // OK: default verbosity when -v/-vv absent
650	let first_line = desc.lines().next().unwrap_or("");  // OK: obvious
651	
652	match matches.len() {
653	    1 => {
654	        #[allow(clippy::unwrap_used)]
655	        // OK: `1 =>` arm guarantees matches.len() == 1
656	        Ok(TitleMatch::One(matches.into_iter().next().unwrap()))
657	    }
658	    // ...
659	}
660	```
661	
662	Tests (`#[cfg(test)]`) are exempt — panicking on setup failure is the
663	correct test behavior.
664	
665	### Ask for clarification on ambiguous input
666	
667	When user input is ambiguous or missing necessary detail, stop and
668	ask a specific question. Do not proceed on a guess and hope the
669	result lands right — a clarifying question costs a few seconds;
670	redoing misaligned work costs much more.
671	
672	### Recognize when stuck
673	
674	If a simple task has eaten 5+ minutes of thinking or back-and-forth
675	without progress, stop. Summarize what's blocking — unclear
676	requirements, unfamiliar API surface, conflicting signals — and ask.
677	Continued flailing produces worse outcomes than a direct "I'm stuck
678	on X."

37	Preparation, `X.Y.Z-3.1` / `X.Y.Z-3.2` Work, `X.Y.Z-3`
38	Close-out. The same three-phase shape applies recursively
39	at every depth. See [Numbering](#numbering) for the
40	suffix rule and [Sub-cycle ladders](#sub-cycle-ladders)
41	for the local-ladder mechanics.
42	
43	## Chores sections
44	
45	A **chores section** is a `##` section in
46	`notes/chores/chores-NN.md` recording landed work. In
47	general, every commit that lands on the permanent branch
48	should have a reference to it on a `Commits:` list in a
49	chores file.
50	
51	The phrase **"Open" the chores section** means append a
52	`##` header to the current `notes/chores/chores-NN.md`
53	with the title it records (e.g. `## refactor: foo bar`),
54	followed by an **empty `Commits:`** line. The narrative
55	body stays empty until close-out (see [Close-out](#close-out));
56	the `Commits:` line is backfilled later, once the commit is
57	permanent (see [Commits backfill](#commits-backfill) below).
58	
59	Fuller chores conventions (content rules, header sync,
60	design subsection pattern, `Commits:` formatting) live in
61	AGENTS.md [Chores conventions](../AGENTS.md#chores-conventions).
62	
63	### Commits backfill
64	
65	A chores section's `Commits:` line cites the commit(s) it
66	records, by SHA — but a SHA isn't stable until the commit
67	lands on a **permanent branch** (`main`, or a long-lived
68	release/patch branch that won't be rewritten); a rebase or
69	squash rewrites it on the way. So:
70	
71	- A chores section is **opened with an empty `Commits:`** line.
72	- **Backfill once the commit is on a permanent branch**,
73	  where its SHA is final. A commit can't record its own SHA
74	  (that would change the hash), so the fill always lands one
75	  push later: **each push backfills the `Commits:` of the
76	  commits the previous push made permanent.** On a topic
77	  branch the sections instead wait until the branch lands —
78	  so no SHA is ever written that a later rebase could
79	  invalidate.
80	
81	Use `[[N]]` refs — several as `[[N]],[[M]]` only when one
82	section records multiple commits (a merge non-ff close-out) —
83	with the commit URL + 40-hex SHA in the file's `# References`
84	(format in AGENTS.md
85	[Chores commit references](../AGENTS.md#chores-commit-references)).
86	A section's `##` title matches its commit title, so a rare
87	deliberate rewrite of a permanent-branch commit re-syncs via
88	`git log --grep "<title>"`.
89	
90	The per-push cadence is a project choice, not dogma — a
91	**per-close-out** model (recording a cycle's SHAs at its
92	close-out) is equally valid. The one invariant: a recorded
93	SHA must be permanent.
94	
95	## Preparation
96	
97	The cycle's first commit (`X.Y.Z-0`):
98	
99	- **Bump the version** in `Cargo.toml` to `X.Y.Z-0`.
100	- **Update `Cargo.lock`** to match — run `cargo build`
101	  (or `cargo check`) so the lockfile's `vc-x1` version
102	  tracks `Cargo.toml` in the same commit.
103	- **Move a `## Todo` item** (if the cycle has one) into
104	  `## In Progress` and the todo item should have:
105	  - A **bold title line** — that will be the chores
106	    section header, minus the `## ` prefix.
107	  - A **succinct problem statement**; add if one is needed
108	  - A **plan ladder**.
109	- **Open the [chores section](#chores-sections)** —
110	  append a `##` header with the title it records — the
111	  cycle's anticipated close-out title.
112	
113	## Work-N
114	
115	The cycle's work commits (`X.Y.Z-1`, `X.Y.Z-2`, …)
116	implement the change. As many as needed:
117	
118	- Each commit runs through the
119	  **[per-commit flow](#per-commit-flow)**.
120	- **Interim pushes** are optional (backup, progress
121	  visibility).
122	- Close-out is the only mandatory push (see
123	  [Pushing](#pushing)).
124	- **Subdivide into a sub-cycle** if a Work commit's
125	  scope grows enough (see
126	  [Sub-cycle ladders](#sub-cycle-ladders)).
127	
128	## Close-out
129	
130	The cycle's last commit (bare `X.Y.Z`) does bookkeeping
131	only — the commit body describes that bookkeeping, not
132	what happens post-squash:
133	
134	- **Move the picked-up item** from `## In Progress` to a
135	  one-line entry (with a chores `[N]` ref) in `## Done`.
136	- **Move the `## In Progress` block into the
137	  [chores section](#chores-sections)** (the title-only
138	  placeholder opened during Preparation):
139	  - Cut the problem statement + plan ladder out of
140	    `## In Progress` (drop the bold title line — the
141	    chores `##` header already carries that string).
142	  - Paste under the chores `##` header.
143	  - Sync the chores header to the **final** commit title
144	    if the cycle's scope shifted; update every anchor
145	    back-reference.
146	  - Add an `### As-built ladder` listing the cycle's
147	    commits; optional `### Outcome` notes.
148	  - Replace `## In Progress` with
149	    `_No cycle currently in progress._`.
150	- **Update `notes/README.md`** if functionality changed
151	  (new flags, new subcommands, changed behavior).
152	
153	Whether to **squash** the cycle into one commit before the
154	publishing push, or push as-is, is decided at push time —
155	see [Pushing](#pushing).
156	
157	## Numbering
158	
159	The version suffix on each commit encodes its phase —
160	the **final identifier `0` marks a Preparation**:
161	
162	- `X.Y.Z-0` — Preparation
163	- `X.Y.Z-1`, `X.Y.Z-2`, … — Work commits
164	- `X.Y.Z` — Close-out (bare version, no suffix)
165	
166	Disambiguation:
167	
168	- `-10` — Work commit #10 (final identifier `10`), not a
169	  Preparation.
170	- `-1.0` — Preparation of the `-1` sub-cycle (final
171	  identifier `0`).
172	
173	**Nesting.** Sub-cycles append another level, recursively:
174	
175	- `X.Y.Z-3.0` — Preparation of the `-3` sub-cycle
176	- `X.Y.Z-3.1`, `X.Y.Z-3.2` — its Work
177	- `X.Y.Z-3` — its Close-out
178	- `X.Y.Z-3.1.0` — Preparation of the `-3.1` sub-sub-cycle
179	
180	Sub-cycles needing no Preparation omit `.0` (`-3.1`,
181	`-3.2`, `-3`); one that grows a Preparation later adds
182	`-3.0` without renumbering siblings.
183	
184	Bump `Cargo.toml` at the start of each phase so `vc-x1 -V`
185	reports the active phase at build time.
186	
187	## Per-commit flow
188	
189	Every commit (Preparation, each Work commit, Close-out) goes
190	through:
191	
192	1. **Mark this commit `(current)`** as the first edit in
193	   `notes/todo.md > ## In Progress`.
194	2. **Do the work** (see [Iterative work](#iterative-work)
195	   for the loop-and-squash technique).
196	3. **Flip this commit `(current)` → `(done)`** in `## In
197	   Progress` — before the cargo cycle and the commit.
198	4. **Cargo cycle** (skip-able for purely-docs commits; full
199	   cycle mandatory at close-out):
200	   1. `cargo fmt`
201	   2. `cargo clippy --all-targets -- -D warnings`
202	   3. `cargo test`
203	   4. `cargo install --path . --locked`
204	   5. (re-test if anything substantive changed)
205	5. **Work review.** Stop *before* writing any description;
206	   tell the user "ready to commit." The user reviews the
207	   changes and we iterate until complete.
208	6. **Write the commit description** — see
209	   [Commit description](#commit-description).
210	7. **Commit Description review.** Show the title + body
211	   and stop. The user reviews the description. Iterate.
212	8. **Commit.** `jj commit -m "title" -m "body" -R .` for
213	   the app repo, `-R .claude` for the bot repo (`-R` last
214	   keeps the verb visible):
215	
216	   ```
217	   jj commit -m \
218	   "<type>: <short description>" \
219	   -m "<intro paragraph>
220	
221	   - file1: gist
222	   - file2: gist
223	
224	   ochid: /.claude/<chid>" \
225	   -R .
226	   ```
227	
228	**Two overrides apply:**
229	
230	- **Deviation or question** — any time the work deviates
231	  from the agreed plan, or a question arises, stop and
232	  surface it; don't push through.
233	- **ESC-ESC** — the user can interrupt at any point to pull
234	  a review or question forward.
235	
236	## Commit description
237	
238	[Conventional Commits](https://www.conventionalcommits.org/):
239	
240	```
241	<type>: <short description>
242	```
243	
244	Titles carry **no trailing `(<version>)` suffix**. The
245	version-of-record lives in `fc-config.toml`
246	(`[project].version`); it's a stable local notion (update
247	cadence TBD), useful for confirming you're running the
248	version you're testing, not a per-commit phase marker. We
249	think per-commit version-in-title is unstable on large
250	projects: many changes are in flight simultaneously, and a
251	version is only stable once merged into the main repo — so
252	titles don't carry one.
253	
254	### Title
255	
256	- ≤50 chars total.
257	- Common types: `feat`, `fix`, `refactor`, `test`,
258	  `docs`, `chore`.
259	- Favor terse phrasings.
260	
261	### Body
262	
263	[Prose form](../AGENTS.md#prose-form) (intro + bullets),
264	wrap ≤72. Bullet content differs per repo:
265	
266	- **App-repo body**: file-by-file. One bullet per file
267	  changed (file plus a one-line gist). Sub-bullets for
268	  files with multiple distinct changes:
269	
270	  ```
271	  - path/to/file1
272	    - first distinct change
273	    - second distinct change, wrapping to next line
274	       with continuation indented 5
275	  ```
276	
277	  The file-by-file list is the source of truth for the
278	  cycle's mechanical change record. Chores carries the
279	  narrative + design, not a copy of it. Promote any
280	  "why" beyond one sentence to a chores `###` subsection.
281	
282	- **Session-repo (`.claude`) body**: bullets describe
283	  in-session activity rather than code changes.
284	
285	### Trailer
286	
287	`ochid:` as the last line of the body — see
288	[Cross-repo linking (ochid trailers)](../AGENTS.md#cross-repo-linking-ochid-trailers)
289	in AGENTS.md for the convention.
290	
291	For breaking changes, use the hyphenated `BREAKING-CHANGE:`
292	trailer key. `BREAKING CHANGE:` (with a space) is the only
293	space-separated key the Conventional Commits spec allows; the
294	hyphenated form is also valid and avoids the space ambiguity.
295	
296	## Reviewing changes
297	
298	Work review looks at the **uncommitted working-copy diff**,
299	on the way to commit. The user opens diffs in their
300	editor (Zed, VSCode); jj commands are for terminal:
301	
302	- `jj diff` — working-copy diff (uncommitted)
303	- `jj diff -r @-` — diff of the previous commit
304	- `jj diff --from <X> --to <Y>` — any two revisions
305	- `jj show -r <X>` — description + diff for one rev
306	
307	Don't `jj edit -r @-` to view a past commit — that marks
308	it mutable and shifts `@`; use `jj diff -r @-` or
309	`jj show -r @-`.
310	
311	See [Sub-cycle ladders](#sub-cycle-ladders) for the
312	close-out squash recipe and recovery; revset primitives
313	are in [`jj-tips.md`](jj-tips.md#revsets).
314	
315	## Pushing
316	
317	### Policy
318	
319	Push is **discretionary** during the cycle (backup,
320	progress visibility) and **mandatory at close-out** —
321	the cycle's result must be published.
322	
323	**Approval is per-push.** Every push — any repo, any kind:
324	cycle push, interim backup, recovery/surgery force-push —
325	happens only after the user has reviewed the changes to be
326	published and explicitly approved that specific push.
327	Approval of a plan that *includes* a push does not authorize
328	the push itself; stop and ask again at the moment of pushing.
329	
330	### Shape at close-out push
331	
332	At close-out the cycle's *work* is done; its *published
333	shape* is the remaining choice, made at push time. Surface
334	the options and get user approval before pushing — once on
335	the target, changing shape is a remote rewrite (force-push,
336	needs approval), so choose deliberately.
337	
338	- **Squash to one commit** — single entry on the target.
339	  Right for straightforward changes where the Work-N is
340	  focused on one or two files.
341	- **Merge non-ff** *(current default)* — `main` gains a
342	  merge commit (`X.Y.Z`); cycle commits stay reachable via
343	  two parents. `jj log -r ..@ -n <N>` shows the trapezoidal
344	  shape. See [Merge non-ff recipe](#merge-non-ff-recipe)
345	  for the full setup sequence.
346	- **Keep separate** — one commit per cycle entry on
347	  `main`. Use when the decomposition itself is
348	  informative. Each chores section keeps its own header /
349	  `Commits:` ref; no consolidation churn.
350	
351	Set up squash/merge before invoking `vc-x1 push`; use
352	`jj git push` directly for non-standard shapes.
353	
354	### Merge non-ff recipe
355	
356	Setting up a [Merge non-ff](#shape-at-close-out-push)
357	close-out is a fixed sequence. `<closeout>` is the cycle's
358	close-out commit, `<prev>` the previous cycle's close-out
359	(the current `main` tip), `<work-tip>` the cycle's last
360	Work commit:
361	
362	1. **Rebase the close-out into a merge** — `jj rebase -r
363	   <closeout> --onto <prev> --onto <work-tip>`.
364	   - `-r <closeout>` keeps the `<closeout>` commit in place.
365	   - `--onto <prev>` becomes its first parent (trunk).
366	   - `--onto <work-tip>` becomes the second parent.
367	
368	   Together these make `<closeout>` a merge of
369	   `<prev>` + `<work-tip>`, forming a trapezoidal commit.
370	2. **Use `jj new <merge>`** to add an empty `@` above the
371	   merge. The rebase left `@` *on* the now-content-bearing
372	   merge, which git/IDE diff views show as uncommitted;
373	   `jj new` restores the clean empty `@` on top.
374	3. **Push** — `jj git push --bookmark main -R .`.
375	
376	**Post-hoc caveat.** If the cycle was already pushed
377	[Keep separate](#shape-at-close-out-push) its commits are
378	immutable: the rebase needs `--ignore-immutable` and the
379	push force-updates `main`.
380	The standard sequence assumes the merge is set up *before*
381	the close-out push.
382	
383	### vc-x1 push wrapper
384	
385	`vc-x1 push <bookmark>` wraps per-push mechanics. See
386	`vc-x1 push --help` for current flags.
387	
388	**Current limitation**: only fully supports the
389	[Keep separate](#shape-at-close-out-push) shape; other
390	shapes need manual jj steps. Improvements tracked /
391	planned:
392	
393	- N:1 code↔bot for Merge non-ff (`## Todo` entry P1).
394	- Symmetric squash (planned, to be captured in `-2`).
395	- Per-repo bookmark names (planned).
396	
397	### .claude cadence
398	
399	**Cadence**: one push = one `.claude` commit, paired
400	with every code commit in that push.
401	
402	The `.claude` working copy accumulates session data
403	across the cycle; its change ID stays stable across
404	snapshots, `jj describe`, and the finalize commit, so
405	code-side `ochid:` trailers resolve.
406	
407	`.claude` is a linear journal — all session work lives
408	on `main`, regardless of the app-side bookmark. **Do not
409	create or maintain `.claude` bookmarks that mirror
410	app-side branches** — risks the bot steering session
411	pushes to the wrong remote ref.
412	
413	Ending a session: if the user runs `/exit` there will be
414	session information created, which we don't worry about.
415	The user can close the terminal instead and `@` will
416	remain empty.
417	
418	### Bot communication at the reviews
419	
420	Use plain prose — no insider jargon ("Gate N signal",
421	"Checkpoint N", etc.):
422	
423	- **At Work review** — summarize what changed and stop.
424	  "Work complete. Please review."
425	- **At Commit Description review** — present `$TITLE`
426	  and `$BODY` explicitly; ask permission to commit/push.
427	  Don't spell out the full `vc-x1 push ... --title ...
428	  --body ...` invocation by default.
429	- **At Post close-out review** — surface the shape
430	  options (squash / merge / keep) and the push target;
431	  wait for the user's choice before any `jj squash` /
432	  `jj rebase` / `jj git push` invocation.
433	
434	### After push or finalize: stop and wait
435	
436	After you **push** (cross the remote boundary) or launch
437	**finalize** on the `.claude` repo (`vc-x1 finalize --repo
438	.claude --squash --push <bookmark> --delay 10 --detach`) — by
439	hand or via the `vc-x1 push` wrapper — you **MUST NEVER**
440	proceed (next step, edit, tool call, text output) until the
441	user explicitly directs you to continue. **Even when the next
442	step seems obvious — wait.** Treat push-or-finalize as a hard
443	stop for the whole turn.
444	
445	`vc-x1 push` performs both as its tail stages — `push-app`
446	then the detached `vc-x1 finalize` on `.claude` — so there is
447	no gap left to speak in once the push is invoked. Put **all**
448	closing words *before* invoking it; do not purposely emit anything
449	after the tool returns.
450	
451	Why (`.claude` is the live journal finalize snapshots →
452	squashes → pushes):
453	
454	- **`--detach`**: lets the tool return immediately so the
455	  harness can keep flushing the *trailing* session-data — the
456	  tool call's own transcript and any words emitted *before*
457	  the invoke — into the snapshot.
458	- **`--delay 10`**: gives the harness time to complete that
459	  flush before the snapshot is taken. It is **not** time for
460	  the bot to do more work.
461	- **silence**: the bot must not summarize or perform any work
462	  after the tool returns.
463	
464	**Known slip**: the bot has emitted a summary after launching
465	finalize. There is no "harmless" closing line — if it's worth
466	saying, say it *before* the invoke.
467	
468	### Recovery
469	
470	- **If push exits before `finalize-claude`** (e.g.
471	  failure between `push-app` and `finalize-claude`), run
472	  finalize by hand:
473	
474	  ```
475	  vc-x1 finalize --repo .claude --squash --push <bookmark> --delay 10 --detach --log /tmp/vc-x1-finalize.log
476	  ```
477	
478	- **Run finalize again if `@` is non-empty** after the
479	  finalize's squash-and-commit (also desirable after
480	  extra activity by the bot's agents).
481	  - Why: finalize snapshots `.claude` at `--delay`, but the
482	    bot keeps writing session data while it runs — the tool's
483	    own invocation plus the bot's closing response land
484	    *after* the snapshot.
485	  - Safe to repeat: bot session data is append-only, so a
486	    re-run never conflicts or overwrites. (This could
487	    change; it is not under the user's control.)
488	  - Fix: run `vc-x1 finalize --repo .claude --squash --push
489	    <bookmark> --delay 10 --detach` another time to capture
490	    that tail; one extra pass is generally enough.
491	  - No guarantees: events outside the bot's control can leave
492	    `@` non-empty — e.g. the bot's back end may decide to
493	    squash/consolidate session data, which can take minutes
494	    and land after the snapshot. The remedy is the same: just
495	    run finalize again. This is why a single pass is never
496	    guaranteed to leave `@` empty.
497	- **Clear push's saved state** after any out-of-band
498	  recovery — `rm .vc-x1/push-state.toml` or `vc-x1 push
499	  <bookmark> --restart` — otherwise push resumes from a
500	  stale stage.
501	- **Late code-repo tweak after `push-app` succeeded**
502	  (e.g. updating AGENTS.md or memory) requires `jj
503	  squash --ignore-immutable` and a re-push; that is a
504	  remote rewrite and needs explicit approval like any
505	  push.
506	
507	## Iterative work
508	
509	When work for a single commit (the **target**) benefits
510	from incremental review, loop:
511	
512	1. `jj new -R .` — fresh empty `@` on top of the target.
513	2. Make the next round of changes.
514	3. User reviews the round (see
515	   [Reviewing changes](#reviewing-changes)).
516	4. `jj squash -R .` folds into the target and creates a
517	   new empty `@`.
518	5. If not done, go to step 2.
519	
520	Same jj mechanics as a
521	[sub-cycle ladder](#sub-cycle-ladders), but at
522	single-commit scope — the cycle's version suffix
523	doesn't change.
524	
525	## Sub-cycle ladders
526	
527	When a Work commit subdivides into a sub-cycle (see
528	[Numbering](#numbering) for suffix nesting), its Work
529	commits typically live as a local jj `@` chain and
530	**collapse into the sub-cycle's Close-out** before the
531	parent cycle continues. Ladder commits are scratch —
532	for review and bisection only.
533	
534	### Per-Work-commit contract within a ladder
535	
536	For each Work commit in the ladder:
537	
538	1. `jj new -R .` — create a fresh empty `@`.
539	2. Do the commit's work.
540	3. Run `cargo test --bins`. **Non-negotiable** — build
541	   and clippy alone miss regressions until a later
542	   commit runs the full suite, raising bisection cost.
543	4. `jj describe -m "..." -m "..." -R .` — working title
544	   only; the sub-cycle Close-out collects everything
545	   into one final commit.
546	
547	### Navigating the ladder
548	
549	Common moves:
550	
551	- `jj log -r '<base>::' -R .` — see the whole ladder
552	  from its base.
553	- `jj edit -r <prefix> -R .` — jump `@` to any ladder
554	  commit by chid prefix; useful for bisection.
555	- `jj edit @-- -R .` — quick-jump back two commits.
556	- `jj diff -r <chid> -R .` — review one commit in
557	  isolation.
558	
559	Modifications to any ladder commit rewrite it in place;
560	descendants auto-rebase.
561	
562	### Close-out: squash the ladder
563	
564	When all ladder Work commits are done and tests pass:
565	
566	```
567	jj squash --from "<base>..@-" --into @ -u -R .
568	```
569	
570	`<base>` is the parent of the first ladder commit; `-u`
571	keeps `@`'s description and discards the sources'.
572	After squash, history is linear: `<base> → @`;
573	intermediate commits are auto-abandoned.
574	
575	Then `vc-x1 push <bookmark>` as for any other commit.
576	
577	For N = 1 the squash is a no-op (`<base>..@-` is empty
578	when `@-` is `<base>`); push the single commit directly.
579	
580	### Recovery
581	
582	If a ladder commit goes wrong, back out without losing
583	prior commits:
584	
585	- **Discard the current commit.** `jj abandon @ -R .`
586	  drops it; you get a fresh empty `@` on the same
587	  parent.
588	- **Edit an earlier commit.** `jj edit -r <chid> -R .`,
589	  make corrections, then `jj edit -r <last-ladder-chid>`
590	  to return. Descendants auto-rebase.
591	- **Discard the entire ladder.** `jj op log -R .` shows
592	  the op history; `jj op restore <op-id> -R .` reverts
593	  to that point. Full undo — removes *all* ladder work
594	  after the chosen op. Use only to start over.

100	fc's 100 Hz control loop in `src/fc.py` calls `os.fsync` on the CSV

104	internal garbage-collection / erase cycle can block the loop for
105	hundreds of ms; the loop loses its cadence and the next `print` is
106	delayed — the periodic terminal hesitation seen on Alpine (SanDisk

112	  the loop is flat at the 10 ms target (sync max 0.03 ms, 0 spikes) —
113	  pinning the cause to `os.fsync`.
114	- No latency guarantee exists at any layer:
115	  - the SD card is an opaque FTL device whose GC / wear-leveling can
116	    stall writes for hundreds of ms to seconds; the SD spec gives a
117	    busy signal plus command timeouts, but a timeout is a failure
