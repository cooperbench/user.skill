    11→    let header = format!("[{}]", section);
    12→    content.lines().any(|line| line.trim() == header)
    13→}

    15→/// Extract package name from Cargo.toml
    16→pub fn get_package_name(crate_dir: &Path) -> Result<String> {
    17→    let cargo_toml = crate_dir.join("Cargo.toml");
    18→    let content = std::fs::read_to_string(&cargo_toml)
    19→        .with_context(|| format!("failed to read {}", cargo_toml.display()))?;
    20→
    21→    // Simple parsing - look for name = "..." in [package] section
    22→    let mut in_package = false;
    23→    for line in content.lines() {
    24→        let line = line.trim();
    25→        if line == "[package]" {
    26→            in_package = true;
    27→            continue;
    28→        }
    29→        if line.starts_with('[') {
    30→            in_package = false;
    31→            continue;
    32→        }
    33→        if in_package
    34→            && line.starts_with("name")
    35→            && let Some(eq_pos) = line.find('=')
    36→        {
    37→            let value = line[eq_pos + 1..].trim();
    38→            let value = value.trim_matches('"').trim_matches('\'');
    39→            return Ok(value.to_string());
    40→        }
    41→    }
    42→    bail!("could not find package name in {}", cargo_toml.display());
    43→}

    45→/// Find the project root by looking for Cargo.toml with [workspace] or [package]
    46→/// For workspaces, returns the directory containing the workspace Cargo.toml
    47→/// For POPs (Plain Old Packages), returns the directory containing the package Cargo.toml
    48→pub fn find_project_root() -> Result<PathBuf> {
    49→    let mut dir = std::env::current_dir()?;
    50→    let mut package_root: Option<PathBuf> = None;
    51→
    52→    loop {
    53→        let cargo_toml = dir.join("Cargo.toml");
    54→        if cargo_toml.exists() {
    55→            let content = std::fs::read_to_string(&cargo_toml)?;
    56→            // Workspace takes precedence
    57→            if has_toml_section_exact(&content, "workspace") {
    58→                return Ok(dir);
    59→            }
    60→            // Remember the first (deepest) package we find as potential POP root
    61→            if package_root.is_none() && has_toml_section_exact(&content, "package") {
    62→                package_root = Some(dir.clone());
    63→            }
    64→        }
    65→        if !dir.pop() {
    66→            // No workspace found, use the POP root if we found one
    67→            if let Some(root) = package_root {
    68→                return Ok(root);
    69→            }
    70→            bail!("could not find project root (no Cargo.toml with [workspace] or [package])");
    71→        }
    72→    }
    73→}

    75→/// Check if a project root is a POP (Plain Old Package) vs a workspace
    76→pub fn is_pop(project_root: &Path) -> bool {
    77→    let cargo_toml = project_root.join("Cargo.toml");
    78→    if let Ok(content) = std::fs::read_to_string(&cargo_toml) {
    79→        // POP has [package] but no [workspace]
    80→        has_toml_section_exact(&content, "package")
    81→            && !has_toml_section_exact(&content, "workspace")
    82→    } else {
    83→        false
    84→    }
    85→}

    87→/// Resolve package directory from optional name, defaulting to current directory
    88→/// If package is None, uses current directory (must contain Cargo.toml)
    89→/// If package is Some, looks up the package by name
    90→pub fn resolve_package_dir(workspace: &Path, package: Option<&str>) -> Result<PathBuf> {
    91→    match package {
    92→        Some(name) => find_package_dir(workspace, name),
    93→        None => {
    94→            let cwd = std::env::current_dir()?;
    95→            if cwd.join("Cargo.toml").exists() {
    96→                Ok(cwd)
    97→            } else {
    98→                bail!(
    99→                    "not in a package directory (no Cargo.toml found). Use -p to specify a package."
   100→                )
   101→            }
   102→        }
   103→    }
   104→}

   106→/// Find a package's directory - tries as path first, then searches standard locations
   107→/// For POPs, checks if name matches the root package
   108→pub fn find_package_dir(project_root: &Path, name: &str) -> Result<PathBuf> {
   109→    // Try as path first (relative or absolute)
   110→    let as_path = PathBuf::from(name);
   111→    if as_path.join("Cargo.toml").exists() {
   112→        return Ok(as_path.canonicalize().unwrap_or(as_path));
   113→    }
   114→
   115→    // For POPs, check if name matches the root package
   116→    if is_pop(project_root) {
   117→        if let Ok(pkg_name) = get_package_name(project_root)
   118→            && pkg_name == name
   119→        {
   120→            return Ok(project_root.to_path_buf());
   121→        }
   122→        bail!(
   123→            "package '{}' not found (this is a single-package project with package '{}')",
   124→            name,
   125→            get_package_name(project_root).unwrap_or_else(|_| "unknown".to_string())
   126→        );
   127→    }
   128→
   129→    // Workspace: search libs/, apps/, tools/
   130→    for prefix in ["libs", "apps", "tools"] {
   131→        let path = project_root.join(prefix).join(name);
   132→        if path.join("Cargo.toml").exists() {
   133→            return Ok(path);
   134→        }
   135→    }
   136→
   137→    // Special case: nested test crates like libs/rlibc-x2/tests
   138→    for prefix in ["libs", "apps"] {
   139→        for entry in std::fs::read_dir(project_root.join(prefix))
   140→            .into_iter()
   141→            .flatten()
   142→            .flatten()
   143→        {
   144→            let nested = entry.path().join("tests");
   145→            if nested.join("Cargo.toml").exists()
   146→                && let Ok(pkg_name) = get_package_name(&nested)
   147→                && pkg_name == name
   148→            {
   149→                return Ok(nested);
   150→            }
   151→        }
   152→    }
   153→
   154→    bail!("package '{}' not found", name);
   155→}

   157→/// Find the tspec for a package - tries as path first, then relative to pkg_dir
   158→/// Returns None if no tspec exists (plain cargo build will be used)
   159→pub fn find_tspec(pkg_dir: &Path, explicit: Option<&str>) -> Result<Option<PathBuf>> {
   160→    match explicit {
   161→        Some(name) => {
   162→            // Try as path first (relative to cwd or absolute)
   163→            let as_path = PathBuf::from(name);
   164→            if as_path.exists() {
   165→                return Ok(Some(as_path.canonicalize().unwrap_or(as_path)));
   166→            }
   167→
   168→            // Fallback: relative to package directory
   169→            let in_pkg = pkg_dir.join(name);
   170→            if in_pkg.exists() {
   171→                return Ok(Some(in_pkg));
   172→            }
   173→
   174→            // Try with TSPEC_SUFFIX if name has no extension
   175→            if !name.contains('.') {
   176→                let with_suffix = pkg_dir.join(format!("{}{}", name, TSPEC_SUFFIX));
   177→                if with_suffix.exists() {
   178→                    return Ok(Some(with_suffix));
   179→                }
   180→            }
   181→
   182→            bail!("tspec not found: {}", name);
   183→        }
   184→        None => {
   185→            let default = pkg_dir.join(format!("tspec{}", TSPEC_SUFFIX));
   186→            if default.exists() {
   187→                Ok(Some(default))
   188→            } else {
   189→                Ok(None) // No tspec = plain cargo build
   190→            }
   191→        }
   192→    }
   193→}

   195→/// Find multiple tspecs by glob patterns
   196→/// If no patterns given, defaults to "tspec*{TSPEC_SUFFIX}"
   197→/// Returns sorted list of paths, errors if none found
   198→pub fn find_tspecs(pkg_dir: &Path, patterns: &[String]) -> Result<Vec<PathBuf>> {
   199→    let default_pattern = format!("tspec*{}", TSPEC_SUFFIX);
   200→    let patterns: Vec<&str> = if patterns.is_empty() {
   201→        vec![&default_pattern]
   202→    } else {
   203→        patterns.iter().map(|s| s.as_str()).collect()
   204→    };
   205→
   206→    let mut results = Vec::new();
   207→
   208→    for pattern in &patterns {
   209→        // Try as literal path first (relative to cwd or absolute)
   210→        let as_path = PathBuf::from(pattern);
   211→        if as_path.exists() && as_path.is_file() {
   212→            results.push(as_path.canonicalize().unwrap_or(as_path));
   213→            continue;
   214→        }
   215→
   216→        // Try as literal path relative to pkg_dir
   217→        let in_pkg = pkg_dir.join(pattern);
   218→        if in_pkg.exists() && in_pkg.is_file() {
   219→            results.push(in_pkg);
   220→            continue;
   221→        }
   222→
   223→        // Try as glob pattern in pkg_dir
   224→        let glob_pattern =
   225→            Pattern::new(pattern).with_context(|| format!("invalid glob pattern: {}", pattern))?;
   226→
   227→        let entries: Vec<_> = std::fs::read_dir(pkg_dir)
   228→            .with_context(|| format!("cannot read directory: {}", pkg_dir.display()))?
   229→            .filter_map(|e| e.ok())
   230→            .filter(|e| {
   231→                let name = e.file_name().to_string_lossy().to_string();
   232→                e.path().is_file() && glob_pattern.matches(&name)
   233→            })
   234→            .map(|e| e.path())
   235→            .collect();
   236→
   237→        results.extend(entries);
   238→    }
   239→
   240→    // Remove duplicates and sort
   241→    results.sort();
   242→    results.dedup();
   243→
   244→    if results.is_empty() {
   245→        let pattern_list = patterns.join(", ");
   246→        bail!(
   247→            "no tspec files found matching '{}' in {}",
   248→            pattern_list,
   249→            pkg_dir.display()
   250→        );
   251→    }
   252→
   253→    Ok(results)
   254→}

   256→/// Get binary path for a build with a spec
   257→pub fn get_binary_path(
   258→    workspace: &Path,
   259→    crate_name: &str,
   260→    spec: &Spec,
   261→    release: bool,
   262→    expanded_target_dir: Option<&str>,
   263→) -> PathBuf {
   264→    let is_release = release || spec.cargo.profile.as_ref() == Some(&Profile::Release);
   265→
   266→    let target = spec.cargo.target_triple.clone().or_else(|| {
   267→        spec.cargo
   268→            .target_json
   269→            .as_ref()
   270→            .and_then(|p| p.file_stem().map(|s| s.to_string_lossy().to_string()))
   271→    });
   272→
   273→    let profile_dir = if is_release { "release" } else { "debug" };
   274→
   275→    let base = match expanded_target_dir {
   276→        Some(td) => workspace.join("target").join(td),
   277→        None => workspace.join("target"),
   278→    };
   279→
   280→    match target {
   281→        Some(t) => base.join(t).join(profile_dir).join(crate_name),
   282→        None => base.join(profile_dir).join(crate_name),
   283→    }
   284→}

   290→}

     5→**Status:** Active development. Works with both workspaces and single-package projects (POPs).
     6→
     7→## Installation
     8→
     9→```bash
    10→cargo install --path .
    11→```
    12→
    13→Or run directly:
    14→```bash
    15→cargo run -- build
    16→```
    17→
    18→## Usage
    19→
    20→```bash
    21→tspec build                                # Build current package (or all in workspace) with tspec.ts.toml if present
    22→tspec build -p myapp                       # Build specific package with tspec.ts.toml if present
    23→tspec build -p myapp -t tspec-opt          # Build with alternative spec tspec-opt.ts.toml
    24→tspec build -p myapp -t tspec-opt.ts.toml  # Build with alternative spec tspec-opt.ts.toml
    25→tspec build -p myapp -r -s                 # Build release, strip symbol with tspec.ts.toml if present
    26→tspec run -p myapp                         # Build and run with tspec.ts.toml if present
    27→tspec test -p myapp                        # Build and test with tspec.ts.toml if present
    28→tspec build -w                             # Build all packages (even from inside a package dir)
    29→tspec compare -p myapp -r                  # Compare all tspec*.ts.toml by binary size
    30→```
    31→
    32→The `-p` flag specifies a package (defaults to current directory if in a package, otherwise all packages).
    33→Use `-w, --workspace` to force all-packages mode even when inside a package directory.
    34→The `-t` flag selects a tspec file; if omitted and `tspec.ts.toml` exists, it's used automatically.
    35→
    36→## tspec Files
    37→
    38→Translation specs are TOML files (conventionally `*.ts.toml`) that configure builds:
    39→
    40→```toml
    41→# tspec.ts.toml - all supported fields shown
    42→
    43→# Top-level high-level options (expand to lower-level cargo/rustc flags)
    44→panic = "abort"                            # "unwind" (default), "abort", "immediate-abort" (nightly)
    45→strip = "symbols"                          # "none" (default), "debuginfo", "symbols"
    46→
    47→[cargo]
    48→profile = "release"                        # "debug" (default), "release"
    49→target_triple = "x86_64-unknown-linux-musl"
    50→target_json = "path/to/custom-target.json" # custom target spec (mutually exclusive with target_triple)
    51→unstable = ["panic-immediate-abort"]       # -Z flags (nightly only)
    52→target_dir = "{name}"                      # per-spec target dir; supports {name} and {hash} placeholders
    53→
    54→[rustc]
    55→opt_level = "z"                            # "0", "1", "2", "3", "s", "z"
    56→panic = "abort"                            # "abort", "unwind", "immediate-abort" (prefer top-level panic)
    57→lto = true
    58→codegen_units = 1
    59→build_std = ["core", "alloc"]              # crates to rebuild with -Z build-std (nightly)
    60→flags = ["-C", "relocation-model=static"]  # raw rustc flags passed through
    61→
    62→[linker]
    63→args = ["-static", "-nostartfiles"]
    64→
    65→[linker.version_script]                    # symbol visibility control (enables --gc-sections)
    66→global = ["_start"]                        # symbols to keep
    67→local = "*"                                # default: "*" (hide everything else)
    68→```
    69→
    70→The top-level `panic` sets both cargo `-Z` and rustc `-C` flags automatically. If both
    71→top-level `panic` and `rustc.panic` are set, both are applied (prefer one or the other).
    72→
    73→### Key Concepts
    74→
    75→- **One tspec per package** - A tspec applies to all crates (targets) within a Cargo package. For per-crate control, use separate packages in a workspace.
    76→- **tspec.ts.toml is optional** - Packages without one get plain `cargo build/test`
    77→- **Generated build.rs** - tspec generates temporary build.rs for scoped linker flags
    78→- **Spec comparison** - Compare binary sizes across different specs
    79→
    80→### How Linker Flags Work
    81→
    82→tspec generates a temporary `build.rs` to apply linker flags with binary-scoped directives:
    83→
    84→```rust
    85→// Generated by tspec - deleted after build
    86→fn main() {
    87→    println!("cargo:rustc-link-arg-bin=myapp=-static");
    88→    println!("cargo:rustc-link-arg-bin=myapp=-nostdlib");
    89→}
    90→```
    91→
    92→This is necessary because RUSTFLAGS affects ALL compilations (including dependency
    93→build scripts), causing crashes with flags like `-nostartfiles`. The generated
    94→build.rs scopes flags to just the binary.
    95→
    96→## tspec Management (ts subcommand)
    97→
    98→Manage spec files without manual TOML editing. All commands that modify files preserve comments and formatting.
    99→
   100→| Command | Description |
   101→|---------|-------------|
   102→| `ts list` | List tspec files found in the workspace or a specific package |
   103→| `ts show` | Display the TOML contents of a tspec file |
   104→| `ts hash` | Print the content hash (used in backup filenames) of a tspec |
   105→| `ts new` | Create a new tspec with defaults, or copy an existing one with `-f` (byte-for-byte) |
   106→| `ts set` | Set a scalar value or replace an entire array |
   107→| `ts unset` | Remove a field (scalar or array) from a tspec entirely |
   108→| `ts add` | Add items to an array field (append or insert at position) |
   109→| `ts remove` | Remove items from an array field (by value or by index) |
   110→| `ts backup` | Create a versioned backup copy (`name-NNN-hash.ts.toml`) |
   111→| `ts restore` | Restore a tspec by copying a versioned backup back to its original name |
   112→
   113→```bash
   114→# List / show / hash
   115→tspec ts list                            # List all tspec files
   116→tspec ts list -p myapp                   # List tspec files for a package
   117→tspec ts show -p myapp                   # Show all tspec contents
   118→tspec ts show -p myapp -t tspec-opt      # Show specific tspec-opt.ts.toml
   119→tspec ts hash -p myapp                   # Show content hash of tspec.ts.toml
   120→
   121→# Create new tspec files
   122→tspec ts new -p myapp                    # Create tspec.ts.toml with defaults
   123→tspec ts new experiment -p myapp         # Create experiment.ts.toml
   124→tspec ts new opt2 -p myapp -f tspec-opt  # Copy from existing spec (byte-for-byte)
   125→
   126→# Set scalar values
   127→tspec ts set strip symbols -p myapp              # Set a scalar field
   128→tspec ts set rustc.lto true -t tspec-opt         # Set in a specific tspec
   129→tspec ts set cargo.profile release               # No quoting needed
   130→
   131→# Set (replace) entire arrays — each element is a separate arg
   132→tspec ts set linker.args -static -nostdlib        # Replace array
   133→tspec ts set rustc.build_std core alloc           # Replace array
   134→
   135→# Add items to arrays (append by default)
   136→tspec ts add linker.args -Wl,--gc-sections        # Append one item
   137→tspec ts add linker.args -nostdlib -pie            # Append multiple items
   138→tspec ts add -i 0 linker.args -nostartfiles        # Insert at position 0
   139→
   140→# Remove items from arrays
   141→tspec ts remove linker.args -nostdlib              # Remove by value
   142→tspec ts remove linker.args -static -pie           # Remove multiple by value
   143→tspec ts remove -i 2 linker.args                   # Remove by index
   144→
   145→# Remove a field entirely
   146→tspec ts unset rustc.lto -p myapp                  # Remove scalar field
   147→tspec ts unset linker.args -t tspec-opt            # Remove array field
   148→
   149→# Backup and restore (byte-for-byte copies)
   150→tspec ts backup -p myapp                 # Backup tspec.ts.toml → tspec-001-abcd1234.ts.toml
   151→tspec ts backup -p myapp -t tspec-opt    # Backup tspec-opt.ts.toml
   152→tspec ts restore -t t1-001-abcd1234      # Restore t1.ts.toml from backup
   153→```
   154→
   155→Backups are valid spec files and can be used directly with `-t`.
   156→
   157→`ts set`, `ts unset`, `ts add`, and `ts remove` use `toml_edit` for surgical editing — comments and formatting are preserved.
   158→`ts backup`, `ts restore`, and `ts new -f` use byte-for-byte file copy — comments are preserved exactly.
   159→
   160→### Supported keys
   161→
   162→Keys use two forms: `FIELD` for top-level fields (`panic`, `strip`) or `TABLE.FIELD` for nested fields (`rustc.lto`, `linker.args`).
   163→
   164→| Key | Type | Values |
   165→|-----|------|--------|
   166→| `panic` | scalar | `unwind`, `abort`, `immediate-abort` |
   167→| `strip` | scalar | `none`, `debuginfo`, `symbols` |
   168→| `cargo.profile` | scalar | `debug`, `release` |
   169→| `cargo.target_triple` | scalar | any string |
   170→| `cargo.target_json` | scalar | any path |
   171→| `cargo.target_dir` | scalar | any string (supports `{name}`, `{hash}`) |
   172→| `cargo.unstable` | array | e.g. `panic-immediate-abort` |
   173→| `rustc.opt_level` | scalar | `0`, `1`, `2`, `3`, `s`, `z` |
   174→| `rustc.panic` | scalar | `abort`, `unwind`, `immediate-abort` |
   175→| `rustc.lto` | scalar | `true`, `false` |
   176→| `rustc.codegen_units` | scalar | integer |
   177→| `rustc.build_std` | array | e.g. `core alloc` |
   178→| `rustc.flags` | array | e.g. `-Cforce-frame-pointers=yes` |
   179→| `linker.args` | array | e.g. `-static -nostdlib` |
   180→
   181→### Array operations
   182→
   183→Four orthogonal commands operate on arrays at two levels:
   184→
   185→| Level | Command | Purpose |
   186→|-------|---------|---------|
   187→| Field-level | `ts set` | Replace entire array with new values |
   188→| Field-level | `ts unset` | Remove the array field entirely |
   189→| Item-level | `ts add` | Append items (or insert at position with `-i`) |
   190→| Item-level | `ts remove` | Remove items by value (or by index with `-i`) |
   191→
   192→```bash
   193→tspec ts set linker.args -static -nostdlib   # replace all with these
   194→tspec ts unset linker.args                   # clear the field
   195→tspec ts add linker.args -Wl,--gc-sections   # append (skips duplicates)
   196→tspec ts add -i 0 linker.args -nostartfiles  # insert at position
   197→tspec ts remove linker.args -static          # remove by value
   198→tspec ts remove -i 2 linker.args             # remove by index
   199→```
   200→
   201→Values starting with `-` work without quoting thanks to `allow_hyphen_values`.
   202→Use `--` as an escape hatch if a value collides with a flag like `-p`, `-t`, or `-i`.
   203→
