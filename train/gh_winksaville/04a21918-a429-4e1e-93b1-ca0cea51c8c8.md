fn requires_nightly(spec: &Spec, mode: CargoMode) -> bool {
    let panic_needs = match mode {
        CargoMode::Test => spec.panic.map(|p| p.rustc_panic_value().is_some()).unwrap_or(false),
        CargoMode::Build => spec.panic.map(|p| p.requires_nightly()).unwrap_or(false),
    };
    panic_needs || !spec.cargo.build_std.is_empty() || !spec.cargo.unstable.is_empty()
}
