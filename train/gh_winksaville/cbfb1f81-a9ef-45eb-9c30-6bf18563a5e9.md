163	/// Recursively copy non-hidden entries from `src` to `dst`. Any entry whose
164	/// file name starts with `.` is skipped. Symlinks are skipped with a debug
165	/// log — templates don't need them, and following them risks escaping the
166	/// template tree.
167	pub(crate) fn copy_template_recursive(
168	    src: &Path,
169	    dst: &Path,
170	) -> Result<(), Box<dyn std::error::Error>> {
171	    for entry in std::fs::read_dir(src)? {
172	        let entry = entry?;
173	        let name = entry.file_name();
174	        let name_str = name.to_string_lossy();
175	        if name_str.starts_with('.') {
176	            continue;
177	        }
178	        let src_path = entry.path();
179	        let dst_path = dst.join(&name);
180	        let ft = entry.file_type()?;
181	        if ft.is_symlink() {
182	            debug!("skip symlink {}", src_path.display());
183	            continue;
184	        }
185	        if ft.is_dir() {
186	            mkdir_p(&dst_path)?;
187	            copy_template_recursive(&src_path, &dst_path)?;
188	        } else if ft.is_file() {
189	            debug!("copy {} -> {}", src_path.display(), dst_path.display());
190	            std::fs::copy(&src_path, &dst_path)?;
191	        }
192	    }
193	    Ok(())
194	}

196	/// Replace the first line of `<dir>/README.md` with `# <name>`. If
197	/// `README.md` is absent, this is a no-op. Trailing content after the
198	/// first newline is preserved verbatim; a file with no newline becomes
199	/// just `# <name>`.
200	pub(crate) fn rewrite_readme_first_line(
201	    dir: &Path,
202	    name: &str,
203	) -> Result<(), Box<dyn std::error::Error>> {
204	    let readme = dir.join("README.md");
205	    if !readme.exists() {
206	        return Ok(());
207	    }
208	    let content = std::fs::read_to_string(&readme)?;
209	    let rest = match content.find('\n') {
210	        Some(pos) => &content[pos..],
211	        None => "",
212	    };
213	    let new_content = format!("# {name}{rest}");
214	    std::fs::write(&readme, new_content)?;
215	    debug!("rewrote first line of {}", readme.display());
216	    Ok(())
217	}

281	pub fn init(args: &InitArgs) -> Result<(), Box<dyn std::error::Error>> {
282	    debug!("init: enter");
283	    let parent_dir = match &args.dir {
284	        Some(d) => d.clone(),
285	        None => std::env::current_dir()?,
286	    };
287	    let project_dir = parent_dir.join(&args.name);
288	    let session_dir = project_dir.join(".claude");
289	
290	    // Preflight checks
291	    info!("Preflight checks...");
292	
293	    // Check tools
294	    run("gh", &["auth", "status"], Path::new("."))
295	        .map_err(|_| "gh is not installed or not authenticated (run: gh auth login)")?;
296	    run("jj", &["--version"], Path::new(".")).map_err(|_| "jj is not installed")?;
297	
298	    let owner = match &args.owner {
299	        Some(o) => o.clone(),
300	        None => gh_whoami()?,
301	    };
302	
303	    if project_dir.exists() {
304	        return Err(format!("'{}' already exists", project_dir.display()).into());
305	    }
306	
307	    let code_repo = format!("{owner}/{}", args.name);
308	    let session_repo = format!("{owner}/{}.claude", args.name);
309	
310	    if gh_repo_exists(&owner, &args.name)? {
311	        return Err(format!("GitHub repo '{code_repo}' already exists").into());
312	    }
313	    if gh_repo_exists(&owner, &format!("{}.claude", args.name))? {
314	        return Err(format!("GitHub repo '{session_repo}' already exists").into());
315	    }
316	
317	    let templates = match &args.use_template {
318	        Some(s) => {
319	            let (code_t, bot_t) = parse_use_template(s)?;
320	            validate_templates(&code_t, &bot_t)?;
321	            Some((code_t, bot_t))
322	        }
323	        None => None,
324	    };
325	
326	    let visibility = if args.private {
327	        "--private"
328	    } else {
329	        "--public"
330	    };
331	
332	    if args.dry_run {
333	        info!("Dry run — would execute:");
334	        info!("  1. Create directories: {}", project_dir.display());
335	        info!("  2. git init + jj git init --colocate on both repos");
336	        info!("  3. Write .vc-config.toml and .gitignore to both");
337	        match &templates {
338	            Some((c, b)) => {
339	                info!(
340	                    "  4. Copy templates (non-hidden) into both repos + rewrite README.md first line"
341	                );
342	                info!("       code: {}", c.display());
343	                info!("       bot:  {}", b.display());
344	            }
345	            None => info!("  4. (skipped — no --use-template)"),
346	        }
347	        info!("  5. jj commit both with placeholder ochids");
348	        info!("  6. Get both chids, jj describe both with correct ochids");
349	        info!("  7. Remove jj from both (git clean -xdf)");
350	        info!("  8. gh repo create {session_repo} {visibility}, push");
351	        info!("  9. gh repo create {code_repo} {visibility}, push");
352	        info!("  10. jj git init --colocate on both repos");
353	        info!("  11. Create Claude Code symlink");
354	        return Ok(());
355	    }
356	
357	    // Step 1: Create directories
358	    info!("Step 1: Creating project directories...");
359	    mkdir_p(&project_dir)?;
360	    mkdir_p(&session_dir)?;
361	
362	    // Step 2: git init + jj init on both repos
363	    info!("Step 2: Initializing repos...");
364	    run("git", &["init"], &project_dir)?;
365	    run("jj", &["git", "init", "--colocate"], &project_dir)?;
366	    run("git", &["init"], &session_dir)?;
367	    run("jj", &["git", "init", "--colocate"], &session_dir)?;
368	
369	    // Step 3: Write config files
370	    info!("Step 3: Writing config files...");
371	    write_file(&project_dir.join(".vc-config.toml"), VC_CONFIG_CODE)?;
372	    write_file(&project_dir.join(".gitignore"), GITIGNORE_CODE)?;
373	    write_file(&session_dir.join(".vc-config.toml"), VC_CONFIG_SESSION)?;
374	    write_file(&session_dir.join(".gitignore"), GITIGNORE_SESSION)?;
375	
376	    // Step 4: Copy templates (if --use-template)
377	    if let Some((code_t, bot_t)) = &templates {
378	        info!("Step 4: Copying templates...");
379	        info!("  code: {} -> {}", code_t.display(), project_dir.display());
380	        copy_template_recursive(code_t, &project_dir)?;
381	        rewrite_readme_first_line(&project_dir, &args.name)?;
382	        info!("  bot:  {} -> {}", bot_t.display(), session_dir.display());
383	        copy_template_recursive(bot_t, &session_dir)?;
384	        let session_name = format!("{}.claude", args.name);
385	        rewrite_readme_first_line(&session_dir, &session_name)?;
386	    }
387	
388	    // Step 5: jj commit both with placeholder ochids
389	    info!("Step 5: Committing both repos with placeholder ochids...");
390	    run(
391	        "jj",
392	        &["commit", "-m", "Initial commit\n\nochid: /none"],
393	        &project_dir,
394	    )?;
395	    run(
396	        "jj",
397	        &["commit", "-m", "Initial commit\n\nochid: /none"],
398	        &session_dir,
399	    )?;
400	
401	    // Step 6: Get both chids, then describe both with correct ochids
402	    info!("Step 6: Setting ochid cross-references...");
403	    let code_chid = jj_chid("@-", &project_dir)?;
404	    let session_chid = jj_chid("@-", &session_dir)?;
405	
406	    let code_desc = format!("Initial commit\n\nochid: /.claude/{session_chid}");
407	    let session_desc = format!("Initial commit\n\nochid: /{code_chid}");
408	    run("jj", &["describe", "@-", "-m", &code_desc], &project_dir)?;
409	    run("jj", &["describe", "@-", "-m", &session_desc], &session_dir)?;
410	
411	    {
412	        let hash = run("git", &["rev-parse", "HEAD"], &project_dir)?;
413	        debug!("code repo: chid={code_chid} hash={hash}");
414	        let hash = run("git", &["rev-parse", "HEAD"], &session_dir)?;
415	        debug!(".claude:   chid={session_chid} hash={hash}");
416	    }
417	
418	    // Step 7: Set bookmarks (creates git branches), then remove jj
419	    // Bookmarks must be set before removing .jj/ so git has a 'main' branch to push
420	    info!("Step 7: Setting bookmarks and removing jj...");
421	    run("jj", &["bookmark", "set", "main", "-r", "@-"], &project_dir)?;
422	    run("jj", &["bookmark", "set", "main", "-r", "@-"], &session_dir)?;
423	    // Clean .claude first, then code repo with --exclude to preserve .claude/
424	    run("git", &["clean", "-xdf"], &session_dir)?;
425	    run(
426	        "git",
427	        &["clean", "-xdf", "--exclude", ".claude"],
428	        &project_dir,
429	    )?;
430	    // After removing .jj/, git HEAD is detached — reattach to main
431	    run("git", &["checkout", "main"], &session_dir)?;
432	    run("git", &["checkout", "main"], &project_dir)?;
433	
434	    // Step 8: Create .claude GitHub repo and push
435	    let session_url = format!("git@github.com:{session_repo}.git");
436	    info!("Step 8: Creating GitHub repo {session_repo}...");
437	    run(
438	        "gh",
439	        &["repo", "create", &session_repo, visibility],
440	        &project_dir,
441	    )?;
442	    run(
443	        "git",
444	        &["remote", "add", "origin", &session_url],
445	        &session_dir,
446	    )?;
447	    run_retry(
448	        "git",
449	        &["push", "-u", "origin", "main"],
450	        &session_dir,
451	        args.push_retries,
452	        args.push_retry_delay,
453	    )?;
454	
455	    // Step 9: Create code GitHub repo and push
456	    let code_url = format!("git@github.com:{code_repo}.git");
457	    info!("Step 9: Creating GitHub repo {code_repo}...");
458	    run(
459	        "gh",
460	        &["repo", "create", &code_repo, visibility],
461	        &project_dir,
462	    )?;
463	    run("git", &["remote", "add", "origin", &code_url], &project_dir)?;
464	    run_retry(
465	        "git",
466	        &["push", "-u", "origin", "main"],
467	        &project_dir,
468	        args.push_retries,
469	        args.push_retry_delay,
470	    )?;
471	
472	    // Step 10: Re-initialize jj on both repos
473	    info!("Step 10: Re-initializing jj on both repos...");
474	    run(
475	        "jj",
476	        &["--quiet", "git", "init", "--colocate"],
477	        &project_dir,
478	    )?;
479	    run("jj", &["bookmark", "set", "main", "-r", "@-"], &project_dir)?;
480	    run(
481	        "jj",
482	        &["bookmark", "track", "main", "--remote=origin"],
483	        &project_dir,
484	    )?;
485	    let code_chid_final = jj_chid("@-", &project_dir)?;
486	
487	    run(
488	        "jj",
489	        &["--quiet", "git", "init", "--colocate"],
490	        &session_dir,
491	    )?;
492	    run("jj", &["bookmark", "set", "main", "-r", "@-"], &session_dir)?;
493	    run(
494	        "jj",
495	        &["bookmark", "track", "main", "--remote=origin"],
496	        &session_dir,
497	    )?;
498	    let session_chid_final = jj_chid("@-", &session_dir)?;
499	
500	    // Step 11: Create Claude Code symlink
501	    info!("Step 11: Creating Claude Code symlink...");
502	    let symlink_dir = {
503	        let home =
504	            std::env::var("HOME").map_err(|_| "HOME environment variable not set".to_string())?;
505	        PathBuf::from(home).join(".claude").join("projects")
506	    };
507	
508	    let sl = symlink::SymLink::new(&project_dir, Path::new(".claude"), &symlink_dir)?;
509	    sl.create(false)?;
510	
511	    info!("");
512	    info!("Done! Project created at {}", project_dir.display());
513	    info!("  Code repo:    {code_repo}  (chid={code_chid_final})");
514	    info!("  Session repo: {session_repo}  (chid={session_chid_final})");
515	    info!(
516	        "  Symlink:      {} -> {}",
517	        sl.symlink_path.display(),
518	        sl.abs_target.display()
519	    );
520	
521	    debug!("init: exit");
522	    Ok(())
523	}

327	/// Collect commit IDs for `..x..` (both directions) or any dot notation.
328	///
329	/// Returns `(commit_ids, anchor_index)` — the IDs in display order
330	/// (descendants closest to anchor, anchor, ancestors closest to anchor)
331	/// and the index of the anchor commit within the list.
332	///
333	/// - `desc_count`: `Some(n)` = n closest descendants, `None` = all, `Some(0)` = none
334	/// - `anc_count`: `Some(n)` = n closest ancestors, `None` = all, `Some(0)` = none
335	pub fn collect_ids(
336	    workspace: &Workspace,
337	    repo: &Arc<ReadonlyRepo>,
338	    rev: &str,
339	    desc_count: Option<usize>,
340	    anc_count: Option<usize>,
341	) -> Result<(Vec<CommitId>, usize), Box<dyn std::error::Error>> {
342	    let anchor_ids = resolve_revset(workspace, repo, rev)?;
343	    if anchor_ids.is_empty() {
344	        return Err(format!("no commit found for revision '{rev}'").into());
345	    }
346	    let anchor_id = &anchor_ids[0];
347	    let root_commit_id = repo.store().root_commit_id().clone();
348	
349	    // Descendants (closest to anchor)
350	    let mut desc_ids: Vec<CommitId> = Vec::new();
351	    if desc_count != Some(0) {
352	        let descendant_ids = resolve_revset(workspace, repo, &format!("{rev}::"))?;
353	        desc_ids = descendant_ids
354	            .into_iter()
355	            .filter(|id| *id != root_commit_id && *id != *anchor_id)
356	            .collect();
357	        if let Some(n) = desc_count {
358	            let start = desc_ids.len().saturating_sub(n);
359	            desc_ids = desc_ids[start..].to_vec();
360	        }
361	    }
362	
363	    // Ancestors (closest to anchor)
364	    let mut anc_ids: Vec<CommitId> = Vec::new();
365	    if anc_count != Some(0) {
366	        let ancestor_ids = resolve_revset(workspace, repo, &format!("::{rev}"))?;
367	        let limit = anc_count.unwrap_or(usize::MAX); // OK: no --ancestors limit → unbounded
368	        let mut count = 0;
369	        for commit_id in ancestor_ids {
370	            if commit_id == root_commit_id || commit_id == *anchor_id {
371	                continue;
372	            }
373	            anc_ids.push(commit_id);
374	            count += 1;
375	            if count >= limit {
376	                break;
377	            }
378	        }
379	    }
380	
381	    let anchor_index = desc_ids.len();
382	    let mut result = desc_ids;
383	    result.push(anchor_id.clone());
384	    result.extend(anc_ids);
385	    Ok((result, anchor_index))
386	}

408	/// Expand comma-separated repo paths, default to `["."]`, and run a closure for each.
409	///
410	/// Header style controls output between repos.
411	/// Continues past errors, printing them to stderr, and returns an error if any failed.
412	pub fn for_each_repo<F>(
413	    raw_repos: &[PathBuf],
414	    header: &Header,
415	    mut body: F,
416	) -> Result<(), Box<dyn std::error::Error>>
417	where
418	    F: FnMut(&Workspace, &Arc<ReadonlyRepo>) -> Result<(), Box<dyn std::error::Error>>,
419	{
420	    let repos: Vec<PathBuf> = if raw_repos.is_empty() {
421	        vec![PathBuf::from(".")]
422	    } else {
423	        raw_repos
424	            .iter()
425	            .flat_map(|p| {
426	                p.to_string_lossy()
427	                    .split(',')
428	                    .map(|s| PathBuf::from(s.trim()))
429	                    .collect::<Vec<_>>()
430	            })
431	            .collect()
432	    };
433	    let multi = repos.len() > 1;
434	    let mut first = true;
435	    let mut errors: Vec<String> = Vec::new();
436	
437	    for repo_path in &repos {
438	        if multi {
439	            match header {
440	                Header::Label(deco) => {
441	                    if !first {
442	                        info!("");
443	                    }
444	                    info!(
445	                        "{}",
446	                        bold(&format!("{deco} {} {deco}", repo_path.display()))
447	                    );
448	                }
449	                Header::None => {}
450	            }
451	        }
452	        first = false;
453	
454	        match load_repo(repo_path) {
455	            Ok((workspace, repo)) => {
456	                if let Err(e) = body(&workspace, &repo) {
457	                    error!("({}): {e}", repo_path.display());
458	                    errors.push(format!("{}: {e}", repo_path.display()));
459	                }
460	            }
461	            Err(e) => {
462	                error!("({}): {e}", repo_path.display());
463	                errors.push(format!("{}: {e}", repo_path.display()));
464	            }
465	        }
466	    }
467	
468	    if errors.is_empty() {
469	        Ok(())
470	    } else {
471	        Err(format!("{} repo(s) had errors", errors.len()).into())
472	    }
473	}

489	/// Parse and evaluate a revset string, returning commit IDs.
490	pub fn resolve_revset(
491	    workspace: &Workspace,
492	    repo: &Arc<ReadonlyRepo>,
493	    revset_str: &str,
494	) -> Result<Vec<CommitId>, Box<dyn std::error::Error>> {
495	    let aliases_map = RevsetAliasesMap::new();
496	    let fileset_aliases_map = FilesetAliasesMap::new();
497	    let extensions = RevsetExtensions::default();
498	    let workspace_root = workspace.workspace_root().to_path_buf();
499	    let path_converter = RepoPathUiConverter::Fs {
500	        cwd: workspace_root.clone(),
501	        base: workspace_root,
502	    };
503	    let workspace_name = workspace.workspace_name();
504	    let workspace_ctx = RevsetWorkspaceContext {
505	        path_converter: &path_converter,
506	        workspace_name,
507	    };
508	    let context = RevsetParseContext {
509	        aliases_map: &aliases_map,
510	        local_variables: HashMap::new(),
511	        user_email: "",
512	        date_pattern_context: chrono::Utc::now().fixed_offset().into(),
513	        default_ignored_remote: None,
514	        fileset_aliases_map: &fileset_aliases_map,
515	        use_glob_by_default: false,
516	        extensions: &extensions,
517	        workspace: Some(workspace_ctx),
518	    };
519	
520	    let mut diagnostics = RevsetDiagnostics::new();
521	    let expression = jj_lib::revset::parse(&mut diagnostics, revset_str, &context)?;
522

524	    let symbol_resolver = SymbolResolver::new(repo.as_ref(), no_extensions);

527

529	    for result in revset.iter() {
530	        commit_ids.push(result?);

532	    Ok(commit_ids)

697	## Add `sync` subcommand (0.35.0)
698	
699	Single-command fetch + sync for both repos (`.` and `.claude`).
700	Dry-run by default; `--no-dry-run` applies. Rebase-on-divergence is
701	automatic when no conflicts arise; any failure reverts both repos to
702	their pre-sync state via `jj op restore` so the caller sees an
703	atomic outcome.
704	
705	### Flag shape
706	
707	```
708	vc-x1 sync [--no-dry-run] [--bookmark <NAME>] [--remote <NAME>]
709	```
710	
711	- `--no-dry-run` — default is dry-run (classify + report without
712	  mutating). User confirmed this default: "dry-run will be the default
713	  until we feel how it goes."
714	- `--bookmark` — default `main` (the vc-x1 project convention).
715	- `--remote` — default `origin`.
716	
717	Repos are hardcoded: `[".", ".claude"]`. The user's rule is "always
718	both, at least for now" — an `-R` override can be added later if the
719	need surfaces.
720	
721	### No pre-flight: trust `jj op restore` + explicit `@` rebase
722	
723	The first draft of `sync` required `@` to be empty (`@ & empty()`
724	revset) in every repo before running, on the theory that the revert
725	path (`jj op restore`) would clobber uncommitted work. That ruled out
726	the most common real-world case — `.claude`'s `@` is always non-empty
727	after `/exit` writes its tail.
728	
729	Two empirical findings from fixture testing changed the design:
730	
731	1. **`jj op restore` preserves working-copy content across conflicted
732	   rebases.** Setup: local commit modifying `shared.txt`, `@` above
733	   with `trailing.jsonl`, remote with a conflicting `shared.txt`.
734	   Fetch + rebase → 2 conflicted commits. `jj op restore <pre-sync>`
735	   → `main` and `main@origin` back to their pre-sync ids, `@` back to
736	   the *same* commit id it had before the sync (not a new revert
737	   commit), `conflicts()` empty, both files (`trailing.jsonl` and
738	   `shared.txt`) on disk with their original content. jj's own output
739	   confirms it: "Existing conflicts were resolved or abandoned from 2
740	   commits."
741	
742	2. **`jj git fetch`'s auto-fast-forward leaves `@` behind.** When a
743	   tracked local bookmark is a strict ancestor of the freshly fetched
744	   remote, jj advances the bookmark — but `@`'s parent pointer stays
745	   on the *old* bookmark commit. `@` is now on a dangling branch,
746	   which means any subsequent `jj new`, commit, or finalize would
747	   build on stale history. Easy to reproduce, easy to miss.
748	
749	So the design became:
750	- **No pre-flight check** — the revert path preserves files.
751	- **After the bookmark action, always call `ensure_at_on_main(repo,
752	  bookmark)`.** If `main::@` is empty (i.e., `@` isn't reachable from
753	  the bookmark), run `jj rebase -b @ -d <bookmark>` to pull `@` and
754	  any commits between the old parent and `@` forward. Then check
755	  `conflicts()` — a conflicted rebase trips the outer revert the same
756	  way a conflicted `act_on_state` rebase does.
757	- **Revert preserves trailing writes.** An integration test
758	  (`sync_conflict_preserves_trailing_at_on_revert`) pins this
759	  behavior so a future jj version that changed it would fail CI.
760	
761	### States and actions
762	
763	| State | Detection | `--no-dry-run` action |
764	|------|----------|-----------------------|
765	| `UpToDate` | `bookmarks(exact:<b>) == {<b>@<remote>}` | bookmark no-op (but `ensure_at_on_main` still fires) |
766	| `Behind` | single local head is strict ancestor of remote | `jj bookmark set <b> -r <b>@<remote>` |
767	| `Ahead` | single local head is strict descendant of remote | none (push is separate) |
768	| `Diverged` | conflicted bookmark (multiple heads) *or* neither is ancestor | `jj rebase -b <local-only-head> -d <b>@<remote>`, then check `conflicts()` |
769	| `NoRemote` | `<b>@<remote>` does not resolve | none — skip |
770	
771	After whichever action (or no action) above, **every repo** runs
772	`ensure_at_on_main`: if `<b>::@` is empty, `jj rebase -b @ -d <b>` +
773	conflict check. This catches the common `jj git fetch` auto-ff case
774	— main moved forward, but `@` was left parented on the old main —
775	and orphans in the `Diverged` path where the local-only-head rebase
776	covered local commits but a separate `@` subtree still needed to be
777	brought along.
778	
779	Note: the `Behind` branch is largely redundant with jj's own
780	fetch-auto-fast-forward behavior (tracked bookmarks whose local is a
781	strict ancestor auto-advance on fetch), so in practice the classify
782	result post-fetch is usually `UpToDate`. `Behind` is kept for untracked
783	bookmarks and edge configs where auto-advance is disabled.
784	
785	### `bookmarks(exact:<b>)` for conflicted detection
786	
787	When `jj git fetch` produces a divergence, the local bookmark becomes
788	conflicted — it points at *both* its old local tip and the
789	freshly-fetched remote tip. The bare `main` revset then errors with
790	`Name \`main\` is conflicted`. `bookmarks(exact:main)` returns all
791	heads instead, so `classify` can see the multi-head case and classify
792	it as `Diverged` directly. The rebase picks the non-remote head from
793	that set as the rebase source.
794	
795	### Revert on failure
796	
797	Before any mutation, `sync` snapshots each repo's current operation id
798	(`jj op log -n 1 -T id.short(12)`). On any error, every repo is
799	reverted in a best-effort loop (`jj op restore <id>`); revert failures
800	are logged as `warn!` but don't override the original error. The caller
801	sees the first root-cause error as the process exit value.
802	
803	### Integration tests (`src/sync.rs::integration_tests`)
804	
805	Five integration tests drive real dual-repo fixtures (built via
806	`crate::test_fixture::test_fixture`, seeded optionally from the

811

813	repo, but `sync` itself shells out to `jj` so the tests can't avoid

815	
816	- **`sync_up_to_date`** — fresh fixture, `--no-dry-run`; assert both
817	  repos' `main` commit ids unchanged.
818	- **`sync_tolerates_trailing_at_up_to_date`** — write a file into

821	  file content is preserved. Pins the "no pre-flight" rule.
822	- **`sync_rebases_trailing_at_when_main_moves`** — remote advances
823	  via a second clone; local `@` has trailing jsonl content; sync
824	  (`--no-dry-run`) runs fetch + auto-ff + `ensure_at_on_main`. Asserts
825	  local `main` matches the pushed remote head, `@` is now reachable
826	  from main (not orphaned on the pre-fetch commit), and trailing

187	fn format_branches(
188	    commit: &Commit,
189	    workspace: &Workspace,
190	    repo: &Arc<ReadonlyRepo>,
191	) -> Result<String, Box<dyn std::error::Error>> {
192	    let commit_hex = commit.id().hex();
193	    let view = repo.view();
194	    let mut names: Vec<String> = Vec::new();
195	
196	    for (name, local_remote) in view.bookmarks() {
197	        for bookmark_id in local_remote.local_target.added_ids() {
198	            let revset_str = format!("{} & ::{}", commit_hex, bookmark_id.hex());
199	            let results = common::resolve_revset(workspace, repo, &revset_str)?;
200	            if !results.is_empty() {
201	                names.push(name.as_str().to_string());
202	                break;
203	            }
204	        }
205	
206	        for (remote_name, remote_ref) in &local_remote.remote_refs {
207	            if remote_name.as_str() == "git" {

209	            }
210	            for bookmark_id in remote_ref.target.added_ids() {

212	                let results = common::resolve_revset(workspace, repo, &revset_str)?;

214	                    names.push(format!("{}@{}", name.as_str(), remote_name.as_str()));
215	                    break;
216	                }
217	            }
218	        }
219	    }

253	fn print_diff(
254	    commit: &Commit,
255	    repo: &Arc<ReadonlyRepo>,
256	    file_limit: FileLimit,
257	) -> Result<(), Box<dyn std::error::Error>> {
258	    let parent_tree = commit.parent_tree(repo.as_ref()).block_on()?;
259	    let commit_tree = commit.tree();
260	
261	    let diff_iter = TreeDiffIterator::new(&parent_tree, &commit_tree, &EverythingMatcher);
262	
263	    let mut lines: Vec<String> = Vec::new();
264	    for entry in diff_iter {
265	        let diff = entry.values?;
266	        let before = diff.before.as_resolved().cloned().flatten();
267	        let after = diff.after.as_resolved().cloned().flatten();
268	
269	        if matches!(&before, Some(TreeValue::Tree(_))) || matches!(&after, Some(TreeValue::Tree(_)))
270	        {
271	            continue;
272	        }
273	
274	        let path_str = entry.path.as_internal_file_string().to_string();
275	
276	        let line = match (before, after) {
277	            (None, Some(value)) => {
278	                format!("    Added {} {path_str}", file_type_str(&value))
279	            }
280	            (Some(_), None) => {
281	                format!(
282	                    "    Removed {} {path_str}",
283	                    file_type_str_from_merged(&diff.before),
284	                )
285	            }
286	            (Some(_), Some(value)) => {
287	                format!("    Modified {} {path_str}", file_type_str(&value))
288	            }
289	            (None, None) => continue,
290	        };
291	        lines.push(line);
292	    }
293	
294	    let total = lines.len();
295	    let cap = match file_limit {
296	        FileLimit::All | FileLimit::None => 0,
297	        FileLimit::Cap(n) => n,
298	    };
299	    let show_all = cap == 0 || total <= cap;
300	
301	    if show_all {
302	        info!("Changed files: {total}");
303	        for line in &lines {
304	            info!("{line}");
305	        }
306	    } else {
307	        info!("Changed files: {total}, showing first {cap}");
308	        for line in &lines[..cap] {
309	            info!("{line}");
310	        }
311	    }
312	
313	    Ok(())
314	}
