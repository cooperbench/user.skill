19	function sceneOp(f: number, sc: Scene, fade = 14): number {
20	  if (f < sc.s || f >= sc.e) return 0;
21	  return Math.min(clamp01((f - sc.s) / fade), clamp01((sc.e - f) / fade));
22	}
23	function prog(f: number, sc: Scene): number {
24	  return clamp01((f - sc.s) / (sc.e - sc.s));
25	}

28	  let s = seed;
