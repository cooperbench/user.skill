149	        }
150	      }
151	      // Check if we're on the error alert sound page for toggle functionality
152	      else if (_current_page == e_config_page_error_alert_sound) {
153	        auto* settings = workers.worker<LocalStorage>(k_worker_local_storage);
154	        bool current_setting = settings->get_error_alert_sound();

156

159	        // Save to SD card

161	          SDInterface::i().write_safezen_file_from_settings(*settings, false);

178	          }

180	          M5_LOGD("Dim brightness decreased to: %u%%", new_value);

190	          settings->set_audio_volume(new_value, false);

195	          M5_LOGD("Audio volume decreased to: %u%%", new_value);

275	          }

310	          uint8_t step = button2->get_data().longPress ? 20 : 5;
311	          uint16_t temp = current + step; // use wider type to avoid overflow before clamp
312	          uint8_t new_value = (temp > 100) ? 100 : (uint8_t)temp;

316	            if (SDInterface::i().ready()) {

318	            }
319	            force_next_render();
320	            M5_LOGD("Audio volume increased to: %u%%", new_value);
321	          }
322	          break;
323	        }
324	        default:
325	          break;
326	      }
327	    }
328	    if (button3->is_fresh() && button3->get_data().shortPress) {

330	      switch (_current_page) {

332	        case e_config_page_click_sound:
333	        case e_config_page_error_alert_sound:
334	        case e_config_page_dim_brightness:
335	        case e_config_page_audio_volume:
336	        case e_config_page_set_home_gps:

339	          M5.Lcd.clear(LCD_COLOR_BACKGROUND);  // Clear screen to avoid visual artifacts
340	          open_menu(false);
341	          force_next_render();
342	          return nullptr;  // Stay on config screen
343	        default:
344	          // For other pages, go back to main menu
345	          return &MenuWindow_i;
346	      }
347	    }
348	  }
349	
350	  return nullptr;
351	}
