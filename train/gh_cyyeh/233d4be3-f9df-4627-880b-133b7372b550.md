  lines = buffer.split('\n');

    16→async function streamSSE(
    17→  response: Response,
    18→  callbacks: AgentCallbacks,
    19→  signal?: AbortSignal,
    20→): Promise<void> {
    21→  const reader = response.body?.getReader();
    22→  if (!reader) {
    23→    callbacks.onError('No response stream');
    24→    return;
    25→  }
    26→
    27→  let doneReceived = false;
    28→  const wrappedCallbacks: AgentCallbacks = {
    29→    ...callbacks,
    30→    onDone: (sessionId) => {
    31→      doneReceived = true;
    32→      callbacks.onDone(sessionId);
    33→    },
    34→    onError: (error) => {
    35→      doneReceived = true;
    36→      callbacks.onError(error);
    37→    },
    38→  };
    39→
    40→  try {
    41→    const decoder = new TextDecoder();
    42→    let buffer = '';
    43→    let eventType = '';
    44→
    45→    while (true) {
    46→      const { done, value } = await reader.read();
    47→      if (done) break;
    48→
    49→      buffer += decoder.decode(value, { stream: true });
    50→      const lines = buffer.split('\n');
    51→      buffer = lines.pop() ?? '';
    52→      for (const line of lines) {
    53→        if (line.startsWith('event: ')) {
    54→          eventType = line.slice(7).trim();
    55→        } else if (line.startsWith('data: ')) {
    56→          const data = line.slice(6);
    57→          try {
    58→            const parsed = JSON.parse(data);
    59→            handleSSEEvent(eventType, parsed, wrappedCallbacks);
    60→          } catch {
    61→            // Skip malformed JSON
    62→          }
    63→          eventType = '';
    64→        }
    65→      }
    66→    }
    67→
    68→    // Process any remaining data left in buffer after stream ends
    69→    if (buffer.trim()) {
    70→      const remainingLines = buffer.split('\n');
    71→      for (const line of remainingLines) {
    72→        if (line.startsWith('event: ')) {
    73→          eventType = line.slice(7).trim();
    74→        } else if (line.startsWith('data: ')) {

    80→            // Skip malformed JSON
    81→          }

    83→        }
    84→      }

    88→    if (!doneReceived) {

    90→    }
    91→  } catch (e: unknown) {
    92→    if (signal?.aborted) return;
    93→    const msg = e instanceof Error ? e.message : 'Connection failed';

    95→  }

   148→def _build_message_with_history(
   149→    message: str, conversation_history: list[dict] | None = None
   150→) -> str:
   151→    """Prepend conversation history context to the user message when editing."""
   152→    if not conversation_history:
   153→        return message
   154→
   155→    history_text = "Previous conversation (for context, I am now editing a message):\n"
   156→    for entry in conversation_history:
   157→        role = entry.get("role", "user").capitalize()
   158→        content = entry.get("content", "")
   159→        history_text += f"\n{role}: {content}\n"
   160→    history_text += f"\n---\n\nMy updated message:\n{message}"
   161→    return history_text

   164→def _extract_chart_spec(text: str) -> dict | None:
   165→    """Extract chart_spec from subagent output text.
   166→
   167→    The chart-builder subagent returns markdown containing a JSON code block.
   168→    Try pure JSON first, then extract from ```json ... ``` fenced blocks.
   169→    """
   170→
   171→    # Try pure JSON first
   172→    try:
   173→        parsed = json.loads(text)
   174→        if "chart_spec" in parsed:
   175→            return parsed["chart_spec"]
   176→    except (json.JSONDecodeError, TypeError):
   177→        pass
   178→
   179→    # Extract from markdown fenced code blocks (```json ... ``` or ``` ... ```)
   180→    pattern = r"```(?:json)?\s*\n?(.*?)\n?\s*```"
   181→    for match in re.finditer(pattern, text, re.DOTALL):
   182→        block = match.group(1).strip()
   183→        try:
   184→            parsed = json.loads(block)
   185→            if "chart_spec" in parsed:
   186→                return parsed["chart_spec"]
   187→            # The block itself might BE the chart_spec
   188→            if "data" in parsed and isinstance(parsed["data"], list):
   189→                return parsed
   190→        except (json.JSONDecodeError, TypeError):
   191→            continue
   192→
   193→    return None

   196→def _extract_tool_result_text(content: object) -> str:
   197→    """Extract text from ToolResultBlock.content."""
   198→    if content is None:
   199→        return ""
   200→    if isinstance(content, str):
   201→        return content
   202→    if isinstance(content, list):
   203→        parts = []
   204→        for item in content:
   205→            if isinstance(item, dict) and item.get("type") == "text":
   206→                parts.append(item.get("text", ""))
   207→        return "\n".join(parts)
   208→    return str(content)

   211→async def _stream_chat_container(
   212→    message: str,
   213→    session_id: str | None,
   214→    db: Database,
   215→    conversation_history: list[dict] | None,
   216→    container_manager,
   217→    backend_session_id: str | None = None,
   218→    langfuse_session_id: str | None = None,
   219→) -> AsyncIterator[str]:
   220→    """Stream chat via containerized sidecar instead of local subprocess."""
   221→    import httpx
   222→    import asyncio
   223→
   224→    query_message = _build_message_with_history(message, conversation_history)
   225→    system_prompt = build_system_prompt(db)
   226→
   227→    session_token=[REDACTED].create_token()
   228→
   229→    # Pass Langfuse credentials to the container so the sidecar's
   230→    # TypeScript Langfuse SDK can create traces directly.
   231→    env: dict[str, str] = {
   232→        "ANTHROPIC_API_KEY": session_token,
   233→        "ANTHROPIC_BASE_URL": f"{PROXY_BASE_URL}/anthropic",
   234→    }
   235→    if LANGFUSE_ENABLED:
   236→        env["LANGFUSE_PUBLIC_KEY"] = LANGFUSE_PUBLIC_KEY
   237→        env["LANGFUSE_SECRET_KEY"] = LANGFUSE_SECRET_KEY
   238→        env["LANGFUSE_BASE_URL"] = LANGFUSE_BASE_URL
   239→
   240→    if "127.0.0.1" in PROXY_BASE_URL or "localhost" in PROXY_BASE_URL:
   241→        logger.warning(
   242→            "PROXY_BASE_URL=%s uses localhost which is unreachable from containers. "
   243→            "Set PROXY_BASE_URL to the host's Docker-accessible address "
   244→            "(e.g., http://host.docker.internal:10000).",
   245→            PROXY_BASE_URL,
   246→        )
   247→
   248→    # Use the backend session ID (X-Session-ID header) for both:
   249→    # 1. MCP SSE URL — so the container queries the correct DuckDB instance
   250→    # 2. Container lifecycle key — so the same container is reused across
   251→    #    requests from the same browser tab (the Claude agent session_id
   252→    #    changes after the first response, which would orphan the container)
   253→    stable_session = backend_session_id or session_id or "default"
   254→
   255→    try:
   256→        # Send SSE keepalive immediately so the HTTP response starts and
   257→        # intermediate proxies (Vite, nginx) don't drop the idle connection
   258→        # before we've finished the blocking Docker container creation.
   259→        yield ": keepalive\n\n"
   260→
   261→        # container_manager.create() is synchronous (blocking Docker API call).
   262→        # Run it in a thread executor so the event loop stays responsive and
   263→        # can continue flushing keepalives to the client during startup.
   264→        # gVisor (runsc) containers can take 10-30 seconds to spin up.
   265→        loop = asyncio.get_event_loop()
   266→        create_future = loop.run_in_executor(None, container_manager.create, stable_session, env)
   267→
   268→        max_create_wait = 60.0
   269→        elapsed = 0.0
   270→        while not create_future.done():
   271→            await asyncio.sleep(2.0)
   272→            elapsed += 2.0
   273→            if elapsed >= max_create_wait:
   274→                create_future.cancel()
   275→                raise RuntimeError(f"Container creation timed out after {max_create_wait:.0f}s")
   276→            yield ": keepalive\n\n"
   277→        info = await create_future
   278→
   279→        # Wait for container to be ready
   280→        for attempt in range(10):
   281→            try:
   282→                async with httpx.AsyncClient(timeout=httpx.Timeout(5.0)) as check_client:
   283→                    resp = await check_client.get(f"{info.url}/health")
   284→                    if resp.status_code == 200:
   285→                        break
   286→            except Exception:
   287→                pass
   288→            yield ": keepalive\n\n"
   289→            await asyncio.sleep(1)
   290→        else:
   291→            raise RuntimeError("Sidecar container failed health check after 10 attempts")
   292→
   293→        payload: dict = {
   294→            "message": query_message,
   295→            "session_id": session_id,
   296→            "system_prompt": system_prompt,
   297→            "model": ANTHROPIC_MODEL,
   298→            "mcp_server_url": f"{PROXY_BASE_URL}/mcp/sse?session_id={stable_session}",
   299→            "env": {
   300→                "ANTHROPIC_API_KEY": session_token,
   301→                "ANTHROPIC_BASE_URL": f"{PROXY_BASE_URL}/anthropic",
   302→            },
   303→            "agents": {
   304→                name: {
   305→                    "description": agent_def.description,
   306→                    "prompt": agent_def.prompt,
   307→                    "tools": agent_def.tools,
   308→                    "model": agent_def.model,
   309→                }
   310→                for name, agent_def in build_subagent_definitions(db).items()
   311→            },
   312→        }
   313→        if langfuse_session_id:
   314→            payload["langfuse_session_id"] = langfuse_session_id
   315→        # Pass original message & history separately for Langfuse trace metadata
   316→        if conversation_history:
   317→            payload["original_message"] = message
   318→            payload["conversation_history"] = conversation_history
   319→
   320→        has_tool_calls = False
   321→        has_thinking = False
   322→        done_sent = False
   323→        tool_names: dict[str, str] = {}
   324→        tool_sqls: dict[str, str] = {}
   325→        actual_session_id = session_id
   326→
   327→        async with httpx.AsyncClient(timeout=httpx.Timeout(300.0)) as client:
   328→            async with client.stream("POST", f"{info.url}/query", json=payload) as response:
   329→                async for line in response.aiter_lines():
   330→                    if not line.startswith("data: "):
   331→                        continue
   332→                    raw = line[6:]
   333→                    try:
   334→                        msg = json.loads(raw)
   335→                    except json.JSONDecodeError:
   336→                        continue
   337→
   338→                    msg_type = msg.get("type")
   339→
   340→                    # --- Token-level streaming events from SDK ---
   341→                    if msg_type == "stream_event":
   342→                        event = msg.get("event", {})
   343→                        event_type = event.get("type", "")
   344→
   345→                        if event_type == "content_block_delta":
   346→                            delta = event.get("delta", {})
   347→                            delta_type = delta.get("type", "")
   348→                            if delta_type == "thinking_delta":
   349→                                text = delta.get("thinking", "")
   350→                                if text:
   351→                                    yield f"event: thinking\ndata: {json.dumps({'text': text})}\n\n"
   352→                            elif delta_type == "text_delta":
   353→                                text = delta.get("text", "")
   354→                                if text:
   355→                                    event_name = "answer" if has_tool_calls else "thinking"
   356→                                    yield f"event: {event_name}\ndata: {json.dumps({'text': text})}\n\n"
   357→
   358→                        elif event_type == "content_block_start":
   359→                            block = event.get("content_block", {})
   360→                            block_type = block.get("type")
   361→                            if block_type == "thinking":
   362→                                has_thinking = True
   363→                            elif block_type == "text":
   364→                                if has_thinking:
   365→                                    yield f"event: thinking_done\ndata: {json.dumps({})}\n\n"
   366→                            elif block_type == "tool_use":
   367→                                has_thinking = False
   368→                                has_tool_calls = True
   369→
   370→                    # --- Complete assistant message (contains tool_use blocks) ---
   371→                    elif msg_type == "assistant":
   372→                        message_obj = msg.get("message", {})
   373→                        for block in message_obj.get("content", []):
   374→                            block_type = block.get("type")
   375→                            if block_type == "tool_use":
   376→                                has_tool_calls = True
   377→                                tool_id = block.get("id", "")
   378→                                tool_name = block.get("name", "")
   379→                                tool_input = block.get("input", {})
   380→                                tool_names[tool_id] = tool_name
   381→                                is_execute_sql = "execute_sql" in tool_name
   382→                                sql = tool_input.get("sql", "") if is_execute_sql else ""
   383→                                if sql:
   384→                                    tool_sqls[tool_id] = sql
   385→                                tool_call_data: dict = {"id": tool_id, "name": tool_name}
   386→                                if sql:
   387→                                    tool_call_data["sql"] = sql
   388→                                else:
   389→                                    tool_call_data["input"] = tool_input
   390→                                yield f"event: tool_call\ndata: {json.dumps(tool_call_data, default=str)}\n\n"
   391→                                if tool_name == "Task":
   392→                                    subagent_name = tool_input.get("subagent_type", "unknown")
   393→                                    subagent_prompt = tool_input.get("prompt", "")
   394→                                    tool_names[tool_id] = subagent_name
   395→                                    yield f"event: subagent_start\ndata: {json.dumps({'id': tool_id, 'name': subagent_name, 'prompt': subagent_prompt})}\n\n"
   396→
   397→                    # --- Tool results from user messages ---
   398→                    elif msg_type == "user":
   399→                        message_obj = msg.get("message", {})
   400→                        for block in message_obj.get("content", []):
   401→                            if block.get("type") != "tool_result":
   402→                                continue
   403→                            tool_id = block.get("tool_use_id", "")
   404→                            name = tool_names.get(tool_id, "")
   405→                            content_parts = block.get("content", [])
   406→                            text = ""
   407→                            if isinstance(content_parts, list):
   408→                                for part in content_parts:
   409→                                    if isinstance(part, dict) and part.get("type") == "text":
   410→                                        text = part.get("text", "")
   411→                            elif isinstance(content_parts, str):
   412→                                text = content_parts
   413→
   414→                            # Try to parse structured MCP result
   415→                            result_data: dict = {"id": tool_id, "name": name}
   416→                            # Include the SQL from the original tool_call
   417→                            original_sql = tool_sqls.get(tool_id, "")
   418→                            if original_sql:
   419→                                result_data["sql"] = original_sql
   420→                            try:
   421→                                parsed = json.loads(text)
   422→                                if parsed.get("status") == "success":
   423→                                    if "chart_spec" in parsed:
   424→                                        result_data["chart_spec"] = parsed["chart_spec"]
   425→                                    else:
   426→                                        result_data["columns"] = parsed.get("columns", [])
   427→                                        result_data["rows"] = parsed.get("rows", [])[:100]
   428→                                        result_data["rowCount"] = parsed.get("rowCount", 0)
   429→                                elif parsed.get("status") == "error":
   430→                                    result_data["error"] = parsed.get("error", "")
   431→                                else:
   432→                                    result_data["output"] = text
   433→                            except (json.JSONDecodeError, AttributeError):
   434→                                result_data["output"] = text
   435→                            if block.get("is_error"):
   436→                                try:
   437→                                    parsed_err = json.loads(text)
   438→                                    result_data["error"] = parsed_err.get("error", text)
   439→                                except (json.JSONDecodeError, AttributeError):
   440→                                    result_data["error"] = text
   441→                            # Detect subagent result (Task tool)
   442→                            if name in ("sql-analyst", "chart-builder"):
   443→                                end_data: dict = {"id": tool_id, "name": name}
   444→                                chart_spec = _extract_chart_spec(text)
   445→                                if chart_spec:
   446→                                    end_data["chart_spec"] = chart_spec
   447→                                else:
   448→                                    end_data["result"] = text
   449→                                yield f"event: subagent_end\ndata: {json.dumps(end_data, default=str)}\n\n"
   450→                                continue
   451→                            yield f"event: tool_result\ndata: {json.dumps(result_data, default=str)}\n\n"
   452→
   453→                    # --- Final result ---
   454→                    elif msg_type == "result":
   455→                        actual_session_id = msg.get("session_id") or actual_session_id
   456→                        if msg.get("is_error"):
   457→                            errors = msg.get("errors", [])
   458→                            error_text = msg.get("result") or "; ".join(errors) or "Unknown error"
   459→                            yield f"event: error\ndata: {json.dumps({'message': error_text})}\n\n"
   460→                        yield f"event: done\ndata: {json.dumps({'session_id': actual_session_id})}\n\n"
   461→                        done_sent = True
   462→
   463→                    # --- Sidecar error (e.g. SDK/CLI crash inside container) ---
   464→                    elif msg_type == "error":
   465→                        error_text = msg.get("message") or "Sidecar error"
   466→                        logger.error("Sidecar reported error: %s", error_text)
   467→                        yield f"event: error\ndata: {json.dumps({'message': error_text})}\n\n"
   468→
   469→                    # --- Extract session_id early from system init ---
   470→                    elif msg_type == "system":
   471→                        sys_session = msg.get("session_id")
   472→                        if sys_session:
   473→                            actual_session_id = sys_session
   474→
   475→        # Guard: always send done even if sidecar ended without result message
   476→        if not done_sent:
   477→            logger.warning("Sidecar stream ended without result message; sending done event")
   478→            yield f"event: done\ndata: {json.dumps({'session_id': actual_session_id})}\n\n"
   479→
   480→    except Exception as e:
   481→        logger.error("Container agent error: %s", str(e))
   482→        yield f"event: error\ndata: {json.dumps({'message': str(e)})}\n\n"
   483→    finally:
   484→        proxy_token_store.revoke_token(session_token)
   485→        # Container intentionally kept alive for session resume (--resume flag).
   486→        # Containers are cleaned up by the background cleanup loop after
   487→        # CONTAINER_MAX_LIFETIME_SECONDS, or on application shutdown.
   488→

   606→                        event_name = "thinking" if not has_tool_calls else "answer"
   607→                        yield f"event: {event_name}\ndata: {json.dumps({'text': text})}\n\n"
   608→
   609→                elif event_type == "content_block_start":
   610→                    block = event.get("content_block", {})

Now let me propose approaches. The core architectural challenge is: **how to pause the agent mid-stream, get user input, and feed it back**.

## Approach 1: MCP Tool with Async Wait (Recommended)

**How it works:**
- Add `ask_user_question` as an MCP tool on the backend
- When called, the tool handler stores the question in a shared state and `await`s an `asyncio.Event`
- The SSE generator detects the `ToolUseBlock`, emits a `user_question` SSE event to the frontend
- Frontend renders inline buttons; user clicks one
- Frontend POSTs to a new `/api/chat/respond` endpoint
- Backend sets the `asyncio.Event`, tool handler wakes up and returns the answer
- Agent continues with the user's answer

**Trade-offs:**
- (+) Clean integration with existing MCP/SDK architecture
- (+) Agent subprocess stays alive and resumes naturally
- (+) Works for both subprocess and container paths
- (-) Requires careful async coordination (shared state between SSE generator and POST endpoint)
- (-) Needs SSE keepalives while waiting for user response

## Approach 2: Stream Interruption + Resume

**How it works:**
- When the agent calls `ask_user_question`, the backend detects it, terminates the stream, and sends the question as a special `done`-like event
- Frontend renders the question. When user answers, it sends a new chat request with the answer injected as context
- A fresh agent session resumes with the answer

**Trade-offs:**
- (+) Simpler — no async coordination, no shared state
- (-) Loses agent session continuity (new subprocess for each question)
- (-) More complex conversation reconstruction
- (-) Slower (spawning new subprocess per question)

## Approach 3: Proxy Tool Result Injection

**How it works:**
- Don't define ask_user_question as an MCP tool. Instead, intercept the agent's text output when it contains a structured question pattern
- Parse the question from the agent's text, present to user, then inject the answer as a new user message

**Trade-offs:**
- (+) No tool coordination needed
- (-) Fragile — relies on parsing agent text
- (-) Agent doesn't know to pause and wait
- (-) Not reliable across model changes

**My recommendation: Approach 1 (MCP Tool with Async Wait).** It integrates cleanly with the existing architecture, maintains session continuity, and gives the agent a proper tool to call. The async coordination is manageable with `asyncio.Event`.
