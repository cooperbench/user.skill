   158→
   159→```python
   160→import logging
   161→from dataclasses import dataclass, field
   162→from datetime import datetime, timedelta
   163→
   164→from app.database import Database
   165→
   166→logger = logging.getLogger(__name__)
   167→
   168→
   169→@dataclass
   170→class SessionEntry:
   171→    db: Database
   172→    last_seen_at: datetime = field(default_factory=datetime.utcnow)
   173→
   174→
   175→class SessionManager:
   176→    def __init__(self) -> None:
   177→        self._sessions: dict[str, SessionEntry] = {}
   178→
   179→    def get_or_create(self, session_id: str) -> Database:
   180→        if session_id not in self._sessions:
   181→            logger.info("Creating new session: %s", session_id)
   182→            self._sessions[session_id] = SessionEntry(db=Database())
   183→        else:
   184→            self._sessions[session_id].last_seen_at = datetime.utcnow()
   185→        return self._sessions[session_id].db
   186→
   187→    def touch(self, session_id: str) -> bool:
   188→        if session_id not in self._sessions:
   189→            return False
   190→        self._sessions[session_id].last_seen_at = datetime.utcnow()
   191→        return True
   192→
   193→    def destroy(self, session_id: str) -> None:
   194→        if session_id not in self._sessions:
   195→            return
   196→        try:
   197→            self._sessions[session_id].db.conn.close()
   198→        except Exception:
   199→            pass
   200→        del self._sessions[session_id]
   201→        logger.info("Destroyed session: %s", session_id)
   202→
   203→    def cleanup_stale(self, ttl_seconds: int = 300) -> int:
   204→        cutoff = datetime.utcnow() - timedelta(seconds=ttl_seconds)
   205→        stale = [
   206→            sid for sid, entry in self._sessions.items()
   207→            if entry.last_seen_at < cutoff
   208→        ]
   209→        for sid in stale:
   210→            self.destroy(sid)
   211→        return len(stale)
   212→
   213→
   214→session_manager = SessionManager()
   215→```
