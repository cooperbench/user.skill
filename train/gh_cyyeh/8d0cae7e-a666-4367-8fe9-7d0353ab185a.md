### 6. Error Handling

**In `agent.py`:**
- Top-level `try/except Exception` catches any error and yields:
  ```
  event: error
  data: {"message": "<error>"}
  ```
- Container creation timeout (60s): raises `RuntimeError`, caught by the outer except
- Sidecar health check failure (10 attempts): raises `RuntimeError`, caught by the outer except
- `msg_type == "error"` from sidecar: logs `logger.error()` and emits `error` + `done` events
- `msg_type == "result"` with `is_error`: emits `error` event before `done`
- Guard: if the sidecar stream ends without a `result` message, always sends `done`

- `AbortError` is silently swallowed (expected on client disconnect)

   128→def _build_message_with_history(
   129→    message: str, conversation_history: list[dict] | None = None
   130→) -> str:
   131→    """Prepend conversation history context to the user message when editing."""
   132→    if not conversation_history:
   133→        return message
   134→
   135→    history_text = "Previous conversation (for context, I am now editing a message):\n"
   136→    for entry in conversation_history:
   137→        role = entry.get("role", "user").capitalize()
   138→        content = entry.get("content", "")
   139→        history_text += f"\n{role}: {content}\n"
   140→    history_text += f"\n---\n\nMy updated message:\n{message}"
   141→    return history_text

   145→def _build_chart_spec_from_stream_messages(
   146→    msg: dict, tool_names: dict[str, str], subagent_chart_specs: dict
   147→) -> None:
   148→    """Inspect an assistant message and capture render_chart input as chart_spec.
   149→
   150→    When the chart-builder subagent calls render_chart, the tool_use block
   151→    appears in an assistant message whose parent_tool_use_id is the Task
   152→    tool_use_id for the chart-builder.  We extract input directly so we never
   153→    need to parse free-text JSON.
   154→    """
   155→    parent_id = msg.get("parent_tool_use_id")
   156→    if not parent_id or tool_names.get(parent_id) != "chart-builder":
   157→        return
   158→    for block in msg.get("message", {}).get("content", []):
   159→        if block.get("type") == "tool_use" and "render_chart" in block.get("name", ""):
   160→            subagent_chart_specs[parent_id] = block.get("input", {})
   161→            return

   164→async def stream_chat(
   165→    message: str,
   166→    session_id: str | None = None,
   167→    db: Database | None = None,
   168→    conversation_history: list[dict] | None = None,
   169→    langfuse_session_id: str | None = None,
   170→    backend_session_id: str | None = None,
   171→) -> AsyncIterator[str]:
   172→    """Stream agent chat responses as SSE events via containerized sidecar."""
   173→    from app.container_manager import container_manager
   174→    if container_manager is None:
   175→        raise RuntimeError("Docker is not available. Container mode requires Docker.")
   176→    if db is None:
   177→        raise ValueError("db must be provided")
   178→
   179→    import httpx
   180→    import asyncio
   181→
   182→    query_message = _build_message_with_history(message, conversation_history)
   183→    system_prompt = build_system_prompt(db)
   184→
   185→    # Pass LiteLLM master key so the sidecar can authenticate with the
   186→    # LiteLLM proxy.  Real provider API keys live only in LiteLLM config.
   187→    env: dict[str, str] = {
   188→        "ANTHROPIC_API_KEY=[REDACTED],
   189→        "ANTHROPIC_BASE_URL": LITELLM_URL,
   190→    }
   191→    if LANGFUSE_ENABLED:
   192→        env["LANGFUSE_PUBLIC_KEY"] = LANGFUSE_PUBLIC_KEY
   193→        env["LANGFUSE_SECRET_KEY"] = LANGFUSE_SECRET_KEY
   194→        env["LANGFUSE_BASE_URL"] = LANGFUSE_BASE_URL
   195→
   196→    if "127.0.0.1" in PROXY_BASE_URL or "localhost" in PROXY_BASE_URL:
   197→        logger.warning(
   198→            "PROXY_BASE_URL=%s uses localhost which is unreachable from containers. "
   199→            "Set PROXY_BASE_URL to the host's Docker-accessible address "
   200→            "(e.g., http://host.docker.internal:10000).",
   201→            PROXY_BASE_URL,
   202→        )
   203→
   204→    # Use the backend session ID (X-Session-ID header) for both:
   205→    # 1. MCP SSE URL — so the container queries the correct DuckDB instance
   206→    # 2. Container lifecycle key — so the same container is reused across
   207→    #    requests from the same browser tab (the Claude agent session_id
   208→    #    changes after the first response, which would orphan the container)
   209→    stable_session = backend_session_id or session_id or "default"
   210→
   211→    try:
   212→        # Send SSE keepalive immediately so the HTTP response starts and
   213→        # intermediate proxies (Vite, nginx) don't drop the idle connection
   214→        # before we've finished the blocking Docker container creation.
   215→        yield ": keepalive\n\n"
   216→
   217→        # container_manager.create() is synchronous (blocking Docker API call).
   218→        # Run it in a thread executor so the event loop stays responsive and
   219→        # can continue flushing keepalives to the client during startup.
   220→        # gVisor (runsc) containers can take 10-30 seconds to spin up.
   221→        loop = asyncio.get_event_loop()
   222→        create_future = loop.run_in_executor(None, container_manager.create, stable_session, env)
   223→
   224→        max_create_wait = 60.0
   225→        elapsed = 0.0
   226→        while not create_future.done():
   227→            await asyncio.sleep(2.0)
   228→            elapsed += 2.0
   229→            if elapsed >= max_create_wait:
   230→                create_future.cancel()
   231→                raise RuntimeError(f"Container creation timed out after {max_create_wait:.0f}s")
   232→            yield ": keepalive\n\n"
   233→        info = await create_future
   234→
   235→        # Wait for container to be ready
   236→        for attempt in range(10):
   237→            try:
   238→                async with httpx.AsyncClient(timeout=httpx.Timeout(5.0)) as check_client:
   239→                    resp = await check_client.get(f"{info.url}/health")
   240→                    if resp.status_code == 200:
   241→                        break
   242→            except Exception:
   243→                pass
   244→            yield ": keepalive\n\n"
   245→            await asyncio.sleep(1)
   246→        else:
   247→            raise RuntimeError("Sidecar container failed health check after 10 attempts")
   248→
   249→        payload: dict = {
   250→            "message": query_message,
   251→            "session_id": session_id,
   252→            "system_prompt": system_prompt,
   253→            "model": ANTHROPIC_MODEL,
   254→            "mcp_server_url": f"{PROXY_BASE_URL}/mcp/sse?session_id={stable_session}",
   255→            "env": {
   256→                "ANTHROPIC_API_KEY=[REDACTED],
   257→                "ANTHROPIC_BASE_URL": LITELLM_URL,
   258→            },
   259→            "agents": {
   260→                name: {
   261→                    "description": agent_def.description,
   262→                    "prompt": agent_def.prompt,
   263→                    "tools": agent_def.tools,
   264→                    "model": agent_def.model,
   265→                }
   266→                for name, agent_def in build_subagent_definitions(db).items()
   267→            },
   268→        }
   269→        if langfuse_session_id:
   270→            payload["langfuse_session_id"] = langfuse_session_id
   271→        # Pass original message & history separately for Langfuse trace metadata
   272→        if conversation_history:
   273→            payload["original_message"] = message
   274→            payload["conversation_history"] = conversation_history
   275→
   276→        has_tool_calls = False
   277→        has_thinking = False
   278→        done_sent = False
   279→        waiting_for_user = False
   280→        tool_names: dict[str, str] = {}
   281→        tool_sqls: dict[str, str] = {}
   282→        # Track subagent text output from intermediate assistant messages.
   283→        # The TypeScript SDK's Task tool_result contains only metadata (agentId,
   284→        # usage), not the subagent's actual output.  The real output arrives in
   285→        # assistant messages whose parent_tool_use_id matches the Task tool ID.
   286→        subagent_texts: dict[str, str] = {}
   287→        subagent_chart_specs: dict[str, dict] = {}
   288→        actual_session_id = session_id
   289→
   290→        async with httpx.AsyncClient(timeout=httpx.Timeout(300.0)) as client:
   291→            async with client.stream("POST", f"{info.url}/query", json=payload) as response:
   292→                line_iter = response.aiter_lines().__aiter__()
   293→                while True:
   294→                    try:
   295→                        if waiting_for_user:
   296→                            try:
   297→                                line = await asyncio.wait_for(line_iter.__anext__(), timeout=5.0)
   298→                            except asyncio.TimeoutError:
   299→                                yield ": keepalive\n\n"
   300→                                continue
   301→                        else:
   302→                            line = await line_iter.__anext__()
   303→                    except StopAsyncIteration:
   304→                        break
   305→
   306→                    if not line.startswith("data: "):
   307→                        continue
   308→                    raw = line[6:]
   309→                    waiting_for_user = False
   310→                    try:
   311→                        msg = json.loads(raw)
   312→                    except json.JSONDecodeError:
   313→                        continue
   314→
   315→                    msg_type = msg.get("type")
   316→                    # --- Token-level streaming events from SDK ---
   317→                    if msg_type == "stream_event":
   318→                        event = msg.get("event", {})
   319→                        event_type = event.get("type", "")
   320→                        stream_parent = msg.get("parent_tool_use_id")
   321→                        is_subagent_event = bool(stream_parent and stream_parent in tool_names)
   322→
   323→                        if event_type == "content_block_delta":
   324→                            delta = event.get("delta", {})
   325→                            delta_type = delta.get("type", "")
   326→                            if delta_type == "thinking_delta":
   327→                                text = delta.get("thinking", "")
   328→                                if text and not is_subagent_event:
   329→                                    yield f"event: thinking\ndata: {json.dumps({'text': text})}\n\n"
   330→                            elif delta_type == "text_delta":
   331→                                text = delta.get("text", "")
   332→                                if text:
   333→                                    # Accumulate subagent text for chart_spec
   334→                                    # extraction (the SDK may not yield a
   335→                                    # complete assistant message for the
   336→                                    # subagent's final turn).
   337→                                    if is_subagent_event:
   338→                                        subagent_texts[stream_parent] = subagent_texts.get(stream_parent, "") + text
   339→                                    else:
   340→                                        event_name = "answer" if has_tool_calls else "thinking"
   341→                                        yield f"event: {event_name}\ndata: {json.dumps({'text': text})}\n\n"
   342→
   343→                        elif event_type == "content_block_start":
   344→                            block = event.get("content_block", {})
   345→                            block_type = block.get("type")
   346→                            if block_type == "thinking":
   347→                                if not is_subagent_event:
   348→                                    has_thinking = True
   349→                            elif block_type == "text":
   350→                                if has_thinking and not is_subagent_event:
   351→                                    yield f"event: thinking_done\ndata: {json.dumps({})}\n\n"
   352→                            elif block_type == "tool_use":
   353→                                if not is_subagent_event:
   354→                                    has_thinking = False
   355→                                    has_tool_calls = True
   356→
   357→                    # --- Complete assistant message (contains tool_use blocks) ---
   358→                    elif msg_type == "assistant":
   359→                        message_obj = msg.get("message", {})
   360→                        parent_tool_use_id = msg.get("parent_tool_use_id")
   361→
   362→                        # Capture text from subagent assistant messages.
   363→                        # These carry the real subagent output (e.g. chart JSON)
   364→                        # that the Task tool_result metadata lacks.
   365→                        if parent_tool_use_id and parent_tool_use_id in tool_names:
   366→                            text_parts = []
   367→                            for block in message_obj.get("content", []):
   368→                                if block.get("type") == "text" and block.get("text"):
   369→                                    text_parts.append(block["text"])
   370→                            if text_parts:
   371→                                subagent_texts[parent_tool_use_id] = "\n".join(text_parts)
   372→                        _build_chart_spec_from_stream_messages(msg, tool_names, subagent_chart_specs)
   373→
   374→                        for block in message_obj.get("content", []):
   375→                            block_type = block.get("type")
   376→                            if block_type == "tool_use":
   377→                                has_tool_calls = True
   378→                                tool_id = block.get("id", "")
   379→                                tool_name = block.get("name", "")
   380→                                tool_input = block.get("input", {})
   381→                                tool_names[tool_id] = tool_name
   382→                                is_execute_sql = "execute_sql" in tool_name
   383→                                sql = tool_input.get("sql", "") if is_execute_sql else ""
   384→                                if sql:
   385→                                    tool_sqls[tool_id] = sql
   386→                                tool_call_data: dict = {"id": tool_id, "name": tool_name}
   387→                                if sql:
   388→                                    tool_call_data["sql"] = sql
   389→                                else:
   390→                                    tool_call_data["input"] = tool_input
   391→                                yield f"event: tool_call\ndata: {json.dumps(tool_call_data, default=str)}\n\n"
   392→                                if tool_name == "Task":
   393→                                    subagent_name = tool_input.get("subagent_type", "unknown")
   394→                                    subagent_prompt = tool_input.get("prompt", "")
   395→                                    tool_names[tool_id] = subagent_name
   396→                                    yield f"event: subagent_start\ndata: {json.dumps({'id': tool_id, 'name': subagent_name, 'prompt': subagent_prompt})}\n\n"
   397→
   398→                                # Detect ask_user_question tool
   399→                                if "ask_user_question" in tool_name:
   400→                                    from app.pending_questions import pending_question_store
   401→                                    import asyncio as _asyncio
   402→                                    for _ in range(50):
   403→                                        pending = pending_question_store.get_pending(stable_session)
   404→                                        if pending:
   405→                                            yield f"event: user_question\ndata: {json.dumps({'question_id': pending['question_id'], **pending['data']})}\n\n"
   406→                                            waiting_for_user = True
   407→                                            break
   408→                                        await _asyncio.sleep(0.1)
   409→
   410→                    # --- Tool results from user messages ---
   411→                    elif msg_type == "user":
   412→                        message_obj = msg.get("message", {})
   413→                        # The SDK attaches the subagent's actual output in
   414→                        # tool_use_result.content (a list of content blocks).
   415→                        # Extract it so we can use it for chart_spec extraction.
   416→                        tool_use_result = msg.get("tool_use_result")
   417→                        tool_use_result_text = ""
   418→                        if isinstance(tool_use_result, dict):
   419→                            tur_content = tool_use_result.get("content")
   420→                            if isinstance(tur_content, list):
   421→                                parts = []
   422→                                for item in tur_content:
   423→                                    if isinstance(item, dict) and item.get("type") == "text":
   424→                                        parts.append(item.get("text", ""))
   425→                                tool_use_result_text = "\n".join(parts)
   426→                        for block in message_obj.get("content", []):
   427→                            if block.get("type") != "tool_result":
   428→                                continue
   429→                            tool_id = block.get("tool_use_id", "")
   430→                            name = tool_names.get(tool_id, "")
   431→                            content_parts = block.get("content", [])
   432→                            text = ""
   433→                            if isinstance(content_parts, list):
   434→                                for part in content_parts:
   435→                                    if isinstance(part, dict) and part.get("type") == "text":
   436→                                        text = part.get("text", "")
   437→                            elif isinstance(content_parts, str):
   438→                                text = content_parts
   439→
   440→                            # Try to parse structured MCP result
   441→                            result_data: dict = {"id": tool_id, "name": name}
   442→                            # Include the SQL from the original tool_call
   443→                            original_sql = tool_sqls.get(tool_id, "")
   444→                            if original_sql:
   445→                                result_data["sql"] = original_sql
   446→                            try:
   447→                                parsed = json.loads(text)
   448→                                if parsed.get("status") == "success":
   449→                                    if "chart_spec" in parsed:
   450→                                        result_data["chart_spec"] = parsed["chart_spec"]
   451→                                    else:
   452→                                        result_data["columns"] = parsed.get("columns", [])
   453→                                        result_data["rows"] = parsed.get("rows", [])[:100]
   454→                                        result_data["rowCount"] = parsed.get("rowCount", 0)
   455→                                elif parsed.get("status") == "error":
   456→                                    result_data["error"] = parsed.get("error", "")
   457→                                else:
   458→                                    result_data["output"] = text
   459→                            except (json.JSONDecodeError, AttributeError):
   460→                                result_data["output"] = text
   461→                            if block.get("is_error"):
   462→                                try:
   463→                                    parsed_err = json.loads(text)
   464→                                    result_data["error"] = parsed_err.get("error", text)
   465→                                except (json.JSONDecodeError, AttributeError):
   466→                                    result_data["error"] = text
   467→                            # Detect subagent result (Task tool)
   468→                            if name in ("sql-analyst", "chart-builder"):
   469→                                # Only chart-builder produces chart_spec JSON.
   470→                                # sql-analyst returns plain text results.
   471→                                if name == "chart-builder":
   472→                                    chart_spec = subagent_chart_specs.get(tool_id)
   473→                                    end_data: dict = {"id": tool_id, "name": name}
   474→                                    if chart_spec:
   475→                                        end_data["chart_spec"] = chart_spec
   476→                                    else:
   477→                                        end_data["result"] = tool_use_result_text or subagent_texts.get(tool_id, text)
   478→                                        logger.warning(
   479→                                            "[container] render_chart tool_use not found for chart-builder %s",
   480→                                            tool_id,
   481→                                        )
   482→                                    yield f"event: subagent_end\ndata: {json.dumps(end_data, default=str)}\n\n"
   483→                                    continue
   484→                                else:
   485→                                    end_data: dict = {"id": tool_id, "name": name}
   486→                                    end_data["result"] = tool_use_result_text or subagent_texts.get(tool_id, text)
   487→                                    yield f"event: subagent_end\ndata: {json.dumps(end_data, default=str)}\n\n"
   488→                                    continue
   489→                            yield f"event: tool_result\ndata: {json.dumps(result_data, default=str)}\n\n"
   490→
   491→                    # --- Final result ---
   492→                    elif msg_type == "result":
   493→                        actual_session_id = msg.get("session_id") or actual_session_id
   494→                        if msg.get("is_error"):
   495→                            errors = msg.get("errors", [])
   496→                            error_text = msg.get("result") or "; ".join(errors) or "Unknown error"
   497→                            yield f"event: error\ndata: {json.dumps({'message': error_text})}\n\n"
   498→                        yield f"event: done\ndata: {json.dumps({'session_id': actual_session_id})}\n\n"
   499→                        done_sent = True
   500→
   501→                    # --- Sidecar error (e.g. SDK/CLI crash inside container) ---
   502→                    elif msg_type == "error":
   503→                        error_text = msg.get("message") or "Sidecar error"
   504→                        logger.error("Sidecar reported error: %s", error_text)
   505→                        yield f"event: error\ndata: {json.dumps({'message': error_text})}\n\n"
   506→                        yield f"event: done\ndata: {json.dumps({'session_id': actual_session_id})}\n\n"
   507→                        done_sent = True
   508→
   509→                    # --- Extract session_id early from system init ---
   510→                    elif msg_type == "system":
   511→                        sys_session = msg.get("session_id")
   512→                        if sys_session:
   513→                            actual_session_id = sys_session
   514→
   515→        # Guard: always send done even if sidecar ended without result message
   516→        if not done_sent:
   517→            logger.warning("Sidecar stream ended without result message; sending done event")
   518→            yield f"event: done\ndata: {json.dumps({'session_id': actual_session_id})}\n\n"
   519→
