package resilience

import (
	"errors"
	"fmt"
	"time"
)

// ErrInvalidPolicy is returned by NewPolicy when configuration validation fails.
var ErrInvalidPolicy = errors.New("resilience: invalid policy")

// Policy bundles the timeout and circuit-breaker configuration that gets
// applied to a single external dependency (e.g. the AI worker gRPC client).
type Policy struct {
	Name               string
	Timeout            time.Duration
	BreakerThreshold   uint32
	BreakerCooldown    time.Duration
	BreakerHalfOpenMax uint32
}

// Option mutates a Policy during construction.
type Option func(*Policy)

// WithTimeout sets the per-call deadline.
func WithTimeout(d time.Duration) Option {
	return func(p *Policy) { p.Timeout = d }
}

// WithBreakerThreshold sets the consecutive-failure count that flips the
// breaker from Closed to Open.
func WithBreakerThreshold(n uint32) Option {
	return func(p *Policy) { p.BreakerThreshold = n }
}

// WithBreakerCooldown sets how long the breaker stays Open before the
// next probe transitions it to Half-Open.
func WithBreakerCooldown(d time.Duration) Option {
	return func(p *Policy) { p.BreakerCooldown = d }
}

// WithBreakerHalfOpenMax caps in-flight probes during Half-Open.
func WithBreakerHalfOpenMax(n uint32) Option {
	return func(p *Policy) { p.BreakerHalfOpenMax = n }
}

// NewPolicy returns a validated Policy. Defaults: 5s timeout,
// breaker opens after 5 consecutive failures, 30s cooldown, 1 half-open probe.
func NewPolicy(name string, opts ...Option) (*Policy, error) {
	p := &Policy{
		Name:               name,
		Timeout:            5 * time.Second,
		BreakerThreshold:   5,
		BreakerCooldown:    30 * time.Second,
		BreakerHalfOpenMax: 1,
	}
	for _, opt := range opts {
		opt(p)
	}
	if err := p.validate(); err != nil {
		return nil, err
	}
	return p, nil
}

func (p *Policy) validate() error {
	if p.Name == "" {
		return fmt.Errorf("%w: name must not be empty", ErrInvalidPolicy)
	}
	if p.Timeout <= 0 {
		return fmt.Errorf("%w: timeout must be > 0", ErrInvalidPolicy)
	}
	if p.BreakerThreshold == 0 {
		return fmt.Errorf("%w: breaker threshold must be > 0", ErrInvalidPolicy)
	}
	if p.BreakerCooldown <= 0 {
		return fmt.Errorf("%w: breaker cooldown must be > 0", ErrInvalidPolicy)
	}
	if p.BreakerHalfOpenMax == 0 {
		return fmt.Errorf("%w: breaker half-open max must be > 0", ErrInvalidPolicy)
	}
	return nil
}

package resilience

import (
	"context"
	"time"

	"google.golang.org/grpc"
	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/status"
)

// IsGRPCCallerError reports whether err is a gRPC status that represents
// a caller-side fault (bad input, missing resource, missing auth) rather
// than a server-side failure. Pass this to NewBreaker via WithIsSuccessful
// so caller errors propagate to the caller AND don't trip the breaker.
//
// Returns true for nil so it satisfies the IsSuccessful contract (success
// counts as success).
//
// Codes treated as caller errors (success-equivalent for breaker accounting):
//
//	InvalidArgument, NotFound, AlreadyExists, PermissionDenied,
//	Unauthenticated, FailedPrecondition, OutOfRange.
//
// Codes counted as breaker failures (caller still sees them):
//
//	Unavailable, DeadlineExceeded, Internal, ResourceExhausted, Unknown,
//	Aborted, Canceled, DataLoss.
func IsGRPCCallerError(err error) bool {
	if err == nil {
		return true
	}
	switch status.Code(err) {
	case codes.InvalidArgument,
		codes.NotFound,
		codes.AlreadyExists,
		codes.PermissionDenied,
		codes.Unauthenticated,
		codes.FailedPrecondition,
		codes.OutOfRange:
		return true
	}
	return false
}

// UnaryClientInterceptor returns a gRPC unary client interceptor that:
//
//   - Applies policy.Timeout to each call (only if no shorter deadline is set).
//   - Routes the call through the breaker.
//   - Returns the invoker's error verbatim. Whether the error counts toward
//     the breaker's failure tally is decided by the breaker's IsSuccessful
//     predicate (set via NewBreaker(p, WithIsSuccessful(...))).
func UnaryClientInterceptor(p *Policy, br *Breaker) grpc.UnaryClientInterceptor {
	return func(
		ctx context.Context,
		method string,
		req, reply any,
		cc *grpc.ClientConn,
		invoker grpc.UnaryInvoker,
		opts ...grpc.CallOption,
	) error {
		callCtx, cancel := withTimeoutIfShorter(ctx, p.Timeout)
		defer cancel()

		_, err := br.Execute(callCtx, func(c context.Context) (any, error) {
			return nil, invoker(c, method, req, reply, cc, opts...)
		})
		return err
	}
}

func withTimeoutIfShorter(ctx context.Context, d time.Duration) (context.Context, context.CancelFunc) {
	if dl, ok := ctx.Deadline(); ok && time.Until(dl) <= d {
		return context.WithCancel(ctx)
	}
	return context.WithTimeout(ctx, d)
}

import (

// taskBudgets maps a task type to its per-execution deadline.
// Keys come from the TypeXxx constants in tasks.go (or per-file constants),
// plus the on-demand task types defined in internal/queue. Anything not in
// the map falls back to defaultBudget.
var taskBudgets = map[string]time.Duration{
	queue.TypeDocumentProcess: 5 * time.Minute,
	queue.TypeAirbyteSync:     5 * time.Minute,
	TypeEmailSummary:          2 * time.Minute,
	TypeSendEmail:             30 * time.Second,
	TypeVoiceUsageAggregation: 30 * time.Second,
	TypeUsageAggregation:      30 * time.Second,
	TypeWebhookDelivery:       30 * time.Second,
	TypeRecrawlSources:        2 * time.Minute,
	TypeCleanupSessions:       2 * time.Minute,
}

// defaultBudget is applied when a task type has no entry in taskBudgets.
const defaultBudget = 1 * time.Minute

// DeadlineMiddleware wraps every Asynq handler with a per-task-type
// context.WithTimeout. Apply via mux.Use(DeadlineMiddleware) so it covers
// handlers regardless of whether they are method receivers or HandlerFunc
// factories. Tasks not present in taskBudgets fall back to defaultBudget.
func DeadlineMiddleware(next asynq.Handler) asynq.Handler {
	return asynq.HandlerFunc(func(ctx context.Context, t *asynq.Task) error {
		budget, ok := taskBudgets[t.Type()]
		if !ok {
			budget = defaultBudget
		}
		ctx, cancel := context.WithTimeout(ctx, budget)
		defer cancel()
		return next.ProcessTask(ctx, t)
	})
}

## Question 1 — **Data race on `currentState` in `WithObservability` (breaker.go).**

Look at the live code in `internal/resilience/breaker.go`:

```go
var currentState gobreaker.State                          // shared variable

gauge, err := o.meter.Int64ObservableGauge("resilience.breaker.state", ...)
if err == nil {
    _, _ = o.meter.RegisterCallback(func(_ context.Context, obs metric.Observer) error {
        obs.ObserveInt64(gauge, int64(currentState), ...)  // READ
        return nil
    }, gauge)
}
settings.OnStateChange = func(name string, from, to gobreaker.State) {
    currentState = to                                       // WRITE
    ...
}
```

- The **read** runs on the OTel SDK's periodic collection goroutine.
- The **write** runs on whichever goroutine made the `gobreaker.Execute` call that tripped the state transition. `gobreaker` calls `OnStateChange` while holding its own internal mutex, but **that mutex doesn't synchronise with the OTel collector** — they're separate goroutine domains.

This is an unguarded shared `int` (gobreaker.State is an `int`). It's a data race by the Go memory model. `go test -race ./internal/resilience/...` would flag it if a test exercised both goroutines, but `breaker_test.go` likely doesn't drive the OTel collector + a state transition concurrently.

In practice, the gauge will read a stale value occasionally — not catastrophic, but it's a real race and CI will eventually flag it once any test stresses both paths.

**My recommended fix:** make `currentState` an `atomic.Int32` (or `atomic.Value`):

```go
var currentState atomic.Int32      // 0=Closed, 1=HalfOpen, 2=Open

// in callback:
obs.ObserveInt64(gauge, int64(currentState.Load()), ...)

// in OnStateChange:
currentState.Store(int32(to))
```

Skill confirms: **`atomic.Int32` (typed atomic, Go 1.19+) is the right primitive**. The skill's decision table is explicit:

| Scenario | Use | Why |
|---|---|---|
| Simple counters, flags | `sync/atomic` | Lock-free, lower overhead |

And the Sync Primitives table reinforces it: *"Prefer typed atomics (Go 1.19+): `atomic.Int64`, `atomic.Bool`"*.

Why atomic beats mutex here:

1. **No compound invariant.** We write one int, we read one int. Nothing else needs to move in lockstep with the state change. Mutexes are for "these N things must update together"; atomic is for "this one value must be visible coherently."
2. **The metric collector runs on every OTel scrape (typically 5–15s).** A mutex would force the callback to contend with `OnStateChange` writers; atomic loads are lock-free reads — important because the OTel SDK collects many metrics in one tick and any blocking primitive can slow the whole scrape.
3. **`gobreaker.State` is `type State int`**, and only three values exist (`Closed=0`, `HalfOpen=1`, `Open=2`). `atomic.Int32` fits cleanly with a one-line cast.

## Question 2 — **Hardcoded `Retry-After: 30` will drift from the breaker cooldown.**

In `pkg/apierror/apierror.go` the `ErrCircuitOpen` branch hard-codes both the header and the user-facing message:

```go
if errors.Is(err, resilience.ErrCircuitOpen) {
    c.Header("Retry-After", "30")
    c.JSON(http.StatusServiceUnavailable, &AppError{
        Code:    http.StatusServiceUnavailable,
        Message: "Service Unavailable",
        Detail:  "circuit breaker open; please retry after 30 seconds",
    })
}
```

Meanwhile `policy.go` exposes `BreakerCooldown` as a tunable, and `.env.example` documents `RAVEN_AI_WORKER_BREAKER_COOLDOWN=30s` as configurable. The moment an operator sets `RAVEN_AI_WORKER_BREAKER_COOLDOWN=10s` (a faster-recovery edge deployment) or `120s` (a flakier upstream), the `Retry-After` value sent to clients **lies**:

- Setting cooldown to **10s** but advertising `Retry-After: 30` makes clients wait 3× too long. Bad for latency-sensitive callers.
- Setting cooldown to **120s** but advertising `Retry-After: 30` makes clients hammer the API 4× before the breaker actually closes. Bad for the failing upstream.

There's also a smaller wart: the user-facing `Detail` says `"circuit breaker open; please retry after 30 seconds"` — leaking the implementation detail "circuit breaker" to API consumers. They don't care that we use a Sony gobreaker; they want to know "the upstream is sick, try again in N seconds."

**My recommended fix:**

1. **Plumb the cooldown into the apierror branch.** The breaker (or the policy that built it) is the source of truth. Two options:
   - **(a)** Add a method `Policy.RetryAfter() time.Duration` and have the gRPC interceptor attach the policy's value as a context value or wrap `ErrCircuitOpen` in a typed error that carries the duration. Then `apierror` reads it off the error.
   - **(b)** Make `ErrCircuitOpen` a struct (`type CircuitOpenError struct { RetryAfter time.Duration }`) instead of a sentinel `var ErrCircuitOpen = errors.New(...)`. `errors.Is` still works if you give it an `Is(error) bool` method. The wrapping site (the interceptor or `Breaker.Execute`) sets the duration from the policy.

   **I'd pick (b)** — keeps the error self-describing without smuggling state through ctx values. The sentinel pattern is too lossy for an error that carries a duration. `errors.As(err, &cOpen)` in apierror gives you the value.

2. **Reword the user-facing `Detail`** to drop "circuit breaker":
   - From: `"circuit breaker open; please retry after 30 seconds"`
   - To: `"upstream service temporarily unavailable; retry after N seconds"`
   (Or just `"upstream temporarily unavailable"` — the `Retry-After` header is the machine-readable contract; the Detail is for humans browsing the response.)

**Side question this opens:** the breaker cooldown is per-policy, but right now only one breaker exists (the AI worker). If/when a second breaker lands (LiveKit, SeaweedFS, Razorpay — per main's deferred HTTP audit), each will have its own cooldown. The struct-shaped error scales; the sentinel doesn't.

Do you agree this needs fixing, and is option (b) (typed `*CircuitOpenError` carrying `RetryAfter`) the right approach — or do you have a reason to keep the sentinel + hardcoded value (e.g., "we'll never tune cooldown in prod, it's fine")?
