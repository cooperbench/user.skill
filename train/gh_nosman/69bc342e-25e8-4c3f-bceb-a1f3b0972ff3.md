604	// Visible character width, skipping ANSI escape sequences.
605	fn visible_width(s: &str) -> usize {
606	    let mut w = 0usize;
607	    let mut chars = s.chars().peekable();
608	    while let Some(c) = chars.next() {
609	        if c == '\x1b' {
610	            // consume up to and including the final byte of the CSI sequence
611	            for nc in chars.by_ref() {
612	                if nc.is_ascii_alphabetic() { break; }
613	            }
614	        } else {
615	            w += 1;
616	        }
617	    }
618	    w
619	}

621	// Apply a selection background to a pre-colored ANSI string.
622	// Parses SGR sequences properly so that any explicit background color emitted
623	// by termimad (e.g. `\x1b[48;5;238m` for inline code) is replaced with the
624	// selection background rather than overriding it.
625	fn with_bg(s: &str, bg: &str) -> String {
626	    let bg_seq  = format!("\x1b[{bg}m");
627	    let mut out = bg_seq.clone();
628	    let bytes   = s.as_bytes();
629	    let mut i   = 0;
630	
631	    while i < bytes.len() {
632	        if bytes[i] == b'\x1b' && bytes.get(i + 1) == Some(&b'[') {
633	            // Parse CSI sequence: ESC [ <params> <cmd>
634	            let esc_start   = i;
635	            i += 2;
636	            let params_start = i;
637	            while i < bytes.len() && !bytes[i].is_ascii_alphabetic() { i += 1; }
638	            let cmd    = *bytes.get(i).unwrap_or(&b'm');
639	            let params = &s[params_start..i];
640	            i += 1;
641	
642	            if cmd == b'm' {
643	                out.push_str(&rewrite_sgr(params, &bg_seq));
644	            } else {
645	                out.push_str(&s[esc_start..i]);
646	            }
647	        } else {
648	            let ch_len = s[i..].chars().next().map_or(1, |c| c.len_utf8());
649	            out.push_str(&s[i..i + ch_len]);
650	            i += ch_len;
651	        }
652	    }
653	
654	    out
655	}

657	// Rewrite a single SGR parameter string so that any background colour is
658	// replaced with the selection background. Foreground / attribute codes are
659	// kept; a pure reset gets the background re-appended.
660	fn rewrite_sgr(params: &str, bg_seq: &str) -> String {
661	    if params.is_empty() || params == "0" {
662	        return format!("\x1b[0m{bg_seq}");
663	    }
664	
665	    let mut non_bg: Vec<&str> = Vec::new();
666	    let mut found_bg = false;
667	    let mut segs = params.split(';').peekable();
668	
669	    while let Some(seg) = segs.next() {
670	        match seg {
671	            // Standard background colours 40-47 and bright 100-107
672	            "40"|"41"|"42"|"43"|"44"|"45"|"46"|"47"
673	            |"100"|"101"|"102"|"103"|"104"|"105"|"106"|"107" => {
674	                found_bg = true;
675	            }
676	            // 256-colour or true-colour background: 48;5;N or 48;2;R;G;B
677	            "48" => {
678	                found_bg = true;
679	                match segs.next().as_deref() {
680	                    Some("5") => { segs.next(); }          // skip N
681	                    Some("2") => { segs.next(); segs.next(); segs.next(); } // skip R;G;B
682	                    _ => {}
683	                }
684	            }
685	            // \x1b[49m — reset background to default (termimad uses this, not \x1b[0m)
686	            "49" => { found_bg = true; }
687	            // Reset within a combined sequence — keep it
688	            "0" => non_bg.push("0"),
689	            other => non_bg.push(other),
690	        }
691	    }
692	
693	    if !found_bg {
694	        return format!("\x1b[{params}m");
695	    }
696	
697	    let mut result = String::new();
698	    if !non_bg.is_empty() {
699	        result.push_str(&format!("\x1b[{}m", non_bg.join(";")));
700	    }
701	    result.push_str(bg_seq);
702	    result
703	}

705	fn wrap(text: &str, width: usize) -> Vec<String> {
706	    if width < 4 { return text.lines().map(str::to_string).collect(); }
707	    let mut out = Vec::new();
708	    for raw_line in text.lines() {
709	        if raw_line.is_empty() { out.push(String::new()); continue; }
710	        let mut remaining = raw_line;
711	        loop {
712	            if remaining.chars().count() <= width {
713	                out.push(remaining.to_string());
714	                break;
715	            }
716	            let break_pos = remaining.chars().take(width)
717	                .collect::<String>()
718	                .rfind(' ')
719	                .unwrap_or(width);
720	            let byte_pos = remaining.char_indices().nth(break_pos)
721	                .map(|(i, _)| i)
722	                .unwrap_or(remaining.len());
723	            out.push(remaining[..byte_pos].to_string());
724	            remaining = remaining[byte_pos..].trim_start_matches(' ');
725	        }
726	    }
727	    out
728	}

730	fn rel_time(iso: &str) -> String {
731	    let Ok(dt) = DateTime::parse_from_rfc3339(iso) else { return iso.to_string(); };
732	    let secs = (Utc::now() - dt.with_timezone(&Utc)).num_seconds().max(0);
733	    if secs < 604_800 {
734	        match secs {
735	            s if s < 60     => "just now".into(),
736	            s if s < 3_600  => format!("{} min{} ago", s/60,     if s/60==1     {""} else {"s"}),
737	            s if s < 86_400 => format!("{} hr{} ago",  s/3_600,  if s/3_600==1  {""} else {"s"}),
738	            s               => format!("{} day{} ago", s/86_400, if s/86_400==1 {""} else {"s"}),
739	        }
740	    } else {
741	        dt.with_timezone(&Local).format("%m/%d/%y").to_string()
742	    }
743	}

747	fn git_current_branch(cwd: &str) -> Option<String> {
748	    let out = std::process::Command::new("git")
749	        .args(["rev-parse", "--abbrev-ref", "HEAD"])
750	        .current_dir(cwd)
751	        .output()
752	        .ok()?;
753	    if !out.status.success() { return None; }
754	    let s = String::from_utf8_lossy(&out.stdout).trim().to_string();
755	    if s.is_empty() || s == "HEAD" { None } else { Some(s) }
756	}

758	fn create_resume_worktree(cwd: &str, branch: &str) -> Option<String> {
759	    let root_out = std::process::Command::new("git")
760	        .args(["rev-parse", "--show-toplevel"])
761	        .current_dir(cwd)
762	        .output()
763	        .ok()?;
764	    if !root_out.status.success() { return None; }
765	    let repo_root = String::from_utf8_lossy(&root_out.stdout).trim().to_string();
766	
767	    let repo_path = std::path::Path::new(&repo_root);
768	    let parent = repo_path.parent()?;
769	    let repo_name = repo_path.file_name()?.to_str()?;
770	    let wt_path = parent.join(format!("{repo_name}-{branch}"));
771	
772	    // Branch already exists; add a worktree pointing at it.
773	    let _ = std::process::Command::new("git")
774	        .args(["worktree", "add", wt_path.to_str()?, branch])
775	        .current_dir(&repo_root)
776	        .output();
777	
778	    if wt_path.exists() { Some(wt_path.to_string_lossy().to_string()) } else { None }
779	}

781	fn resume_via_entire(dir: &str, branch: &str) {
782	    use std::os::unix::process::CommandExt;
783	
784	    let launch_dir = if !branch.is_empty() {
785	        let current = git_current_branch(dir);
786	        if current.as_deref() != Some(branch) {
787	            create_resume_worktree(dir, branch).unwrap_or_else(|| dir.to_string())
788	        } else {
789	            dir.to_string()
790	        }
791	    } else {
792	        dir.to_string()
793	    };
794	
795	    let mut entire = std::process::Command::new("entire");
796	    entire.arg("resume").current_dir(&launch_dir);
797	    if !branch.is_empty() { entire.arg(branch); }
798	
799	    let Ok(out) = entire.output() else { eprintln!("entire resume failed"); return; };
800	
801	    let cmd_str = String::from_utf8_lossy(&out.stdout).trim().to_string();
802	    if cmd_str.is_empty() { eprintln!("entire resume returned no command"); return; }
803	
804	    let mut parts = cmd_str.split_whitespace();
805	    let Some(prog) = parts.next() else { return; };
806	    let args: Vec<&str> = parts.collect();
807	    let err = std::process::Command::new(prog).args(&args).current_dir(&launch_dir).exec();
808	    eprintln!("exec failed: {err}");
809	}

811	/// Find a Claude session by ID. Returns `(cwd, git_branch)` from the JSONL if found,
812	/// so `claude --resume` can be run from the correct directory and branch is known.
813	fn find_claude_session_cwd(session_id: &str) -> Option<(String, String)> {
814	    use std::io::BufRead;
815	    let home = std::env::var("HOME").ok()?;
816	    let projects = std::path::PathBuf::from(&home).join(".claude/projects");
817	    for dir in std::fs::read_dir(&projects).ok()?.flatten() {
818	        let jsonl = dir.path().join(format!("{session_id}.jsonl"));
819	        if !jsonl.exists() { continue; }
820	        let Ok(file) = std::fs::File::open(&jsonl) else { return Some((String::new(), String::new())); };
821	        let mut cwd = String::new();
822	        let mut branch = String::new();
823	        for line in std::io::BufReader::new(file).lines().flatten() {
824	            let Ok(v) = serde_json::from_str::<serde_json::Value>(&line) else { continue };
825	            if matches!(v["type"].as_str(), Some("user") | Some("system")) {
826	                if cwd.is_empty() {
827	                    if let Some(c) = v["cwd"].as_str() { cwd = c.to_string(); }
828	                }
829	                if branch.is_empty() {
830	                    if let Some(b) = v["gitBranch"].as_str() { branch = b.to_string(); }
831	                }
832	                if !cwd.is_empty() && !branch.is_empty() { break; }
833	            }
834	        }
835	        return Some((cwd, branch));
836	    }
837	    None
838	}

840	fn do_resume(_agent: &str, session_id: &str, session_branch: &str, session_cwd: &str) {
841	    let current = git_current_branch(session_cwd);
842	    let same_branch = !session_branch.is_empty()
843	        && current.as_deref() == Some(session_branch);
844	
845	    let dir = if same_branch || session_branch.is_empty() {
846	        session_cwd.to_string()
847	    } else {
848	        create_resume_worktree(session_cwd, session_branch)
849	            .unwrap_or_else(|| session_cwd.to_string())
850	    };
851	
852	    resume_session(session_id, &dir);
853	}

855	/// Resume a session by ID, falling back to `entire resume` if the Claude JSONL is absent.
856	/// `fallback_cwd` is used for `entire resume` if the Claude session isn't found.
857	pub fn resume_session(session_id: &str, fallback_cwd: &str) {
858	    use std::os::unix::process::CommandExt;
859	    if let Some((session_cwd, session_branch)) = find_claude_session_cwd(session_id) {
860	        let launch_dir = if session_cwd.is_empty() || !std::path::Path::new(&session_cwd).exists() {
861	            fallback_cwd.to_string()
862	        } else {
863	            session_cwd
864	        };
865	        let err = std::process::Command::new("claude")
866	            .arg("--resume")
867	            .arg(session_id)
868	            .current_dir(&launch_dir)
869	            .exec();
870	        eprintln!("exec failed: {err}");
871	        // exec failed — fall back to entire resume, checking out the session branch first
872	        resume_via_entire(&launch_dir, &session_branch);
873	    } else {
874	        resume_via_entire(fallback_cwd, "");
875	    }
876	}

880	fn build_flat(
881	    cards: &[Card],
882	    term_w: usize,
883	    collapsed: &std::collections::HashSet<usize>,
884	) -> (Vec<(usize, String)>, Vec<Selectable>, Vec<usize>) {
885	    let w = term_w.saturating_sub(2);
886	    let mut flat: Vec<(usize, String)> = Vec::new();
887	    let mut selectables: Vec<Selectable> = Vec::new();
888	    let mut starts: Vec<usize> = Vec::new();
889	
890	    // Pull the agent name from the Header card so render_card can use it.
891	    let agent: &str = cards.iter().find_map(|c| {
892	        if let Card::Header { agent, .. } = c { Some(agent.as_str()) } else { None }
893	    }).unwrap_or("");
894	
895	    for (card_idx, card) in cards.iter().enumerate() {
896	        if let Card::ToolRound { parts } = card {
897	            if collapsed.contains(&card_idx) {
898	                let si = selectables.len();
899	                starts.push(flat.len());
900	                selectables.push(Selectable::Card(card_idx));
901	                for l in render_tool_summary(parts) { flat.push((si, l)); }
902	            } else {
903	                // Header row (collapses the round when space is pressed)
904	                let si = selectables.len();
905	                starts.push(flat.len());
906	                selectables.push(Selectable::ToolHeader(card_idx));
907	                for l in render_tool_header(parts) { flat.push((si, l)); }
908	                // One selectable per individual tool call

Looking at your screenshot, I can see exactly what's happening — your terminal is light but gossamer is applying the **dark** theme. The selected user card has the dark-mode selection background (`48;5;236`, near-black) which clashes badly with your white terminal background, and the assistant text is the dark-mode light-gray which is too faint on white.

The root cause: macOS terminals (iTerm2, Terminal.app) don't set `COLORFGBG`, so our detection falls back to dark. We need to actively query the terminal for its real background via **OSC 11** (`ESC ] 11 ; ? BEL`), then compute luminance.

Let me implement that. The approach: open `/dev/tty`, put it in raw mode, write the query, poll with a short timeout, parse `rgb:RRRR/GGGG/BBBB` from the response, compute luminance. This needs minimal libc bindings.

262	                        Some(b) if !b.trim().is_empty() => Cmd::NewWorktree(b.trim().to_string()),
263	                        _ => Cmd::Redraw,
264	                    },
265	                    KeyCode::Char('s') => {
266	                        let repo_dir = repos[*repo_idx].directory.clone();
267	                        match new_session_wizard(stdout, &repo_dir, w, h) {
268	                            Some(cfg) => Cmd::NewSession(cfg),
269	                            None => Cmd::Redraw,
270	                        }
271	                    }
272	                    KeyCode::Char('t') => Cmd::Tidy,
273	                    _ => Cmd::None,
274	                },
275	            },
276	            Event::Resize(_, _) => Cmd::Redraw,
277	            _ => Cmd::None,
278	        };
279	
280	        match cmd {
281	            Cmd::None => {}
282	            Cmd::Break => break,
283	            Cmd::Back => {
284	                stack.pop();
285	                if stack.is_empty() { break; }
286	            }
287	            Cmd::PushSessions(idx) => {
288	                let sessions = load_sessions(&repos[idx].directory);
289	                let worktrees = fetch_worktrees(&repos[idx].directory);
290	                stack.push(Screen::Sessions { repo_idx: idx, sel: 0, sessions, worktrees });
291	            }
292	            Cmd::Cd(path) => return Ok(Some(TuiOutcome::CdTo(path))),
293	            Cmd::ResumeSession(id, cwd) => return Ok(Some(TuiOutcome::ResumeSession { id, cwd })),
294	            Cmd::ShowSession(id) => {
295	                execute!(stdout, LeaveAlternateScreen, cursor::Show).ok();
296	                terminal::disable_raw_mode().ok();
297	                let _ = super::show::run(&id);
298	                terminal::enable_raw_mode().ok();
299	                execute!(stdout, EnterAlternateScreen, cursor::Hide).ok();
300	                execute!(stdout, terminal::Clear(ClearType::All)).ok();
301	            }
302	            Cmd::Search(query) => {
303	                execute!(stdout, LeaveAlternateScreen, cursor::Show).ok();
304	                terminal::disable_raw_mode().ok();
305	                let _ = super::search::run(&query, 10, false);
306	                terminal::enable_raw_mode().ok();
307	                execute!(stdout, EnterAlternateScreen, cursor::Hide).ok();
308	                execute!(stdout, terminal::Clear(ClearType::All)).ok();
309	            }
310	            Cmd::NewWorktree(branch) => {
311	                if let Some(Screen::Sessions { repo_idx, worktrees, .. }) = stack.last_mut() {
312	                    let repo_dir = &repos[*repo_idx].directory;
313	                    let notification = create_worktree(repo_dir, &branch);
314	                    // Refresh the worktree list whether or not it succeeded.
315	                    *worktrees = fetch_worktrees(repo_dir);
316	                    // Show the result in the status bar until the next keypress.
317	                    let (tw, th) = terminal::size().unwrap_or((120, 40));
318	                    let msg: String = notification.chars().take(tw as usize).collect();
319	                    let padded = format!("{:<width$}", msg, width = tw as usize);
320	                    execute!(stdout, cursor::MoveTo(0, th - 1)).ok();
321	                    write!(stdout, "\x1b[7m{padded}\x1b[0m").ok();
322	                    stdout.flush().ok();
323	                }
324	            }
325	            Cmd::NewSession(cfg) => {
326	                // Create worktree if requested, then launch the agent.
327	                let launch_dir = if let Some(ref branch) = cfg.branch {
328	                    let _ = create_worktree(&cfg.repo_dir, branch);
329	                    if let Some(Screen::Sessions { repo_idx, worktrees, .. }) = stack.last_mut() {
330	                        *worktrees = fetch_worktrees(&repos[*repo_idx].directory);
331	                    }
332	                    let repo_path = std::path::Path::new(&cfg.repo_dir);
333	                    let parent = repo_path.parent().unwrap_or(repo_path);
334	                    let repo_name = repo_path.file_name()
335	                        .map(|s| s.to_string_lossy().to_string())
336	                        .unwrap_or_else(|| "repo".to_string());
337	                    parent.join(format!("{repo_name}-{}", branch)).to_string_lossy().to_string()
338	                } else {
339	                    cfg.repo_dir.clone()
340	                };
341	
342	                // Copy prompt to clipboard as a convenience.
343	                let prompt = cfg.prompt.trim().to_string();
344	                if !prompt.is_empty() {
345	                    if let Ok(mut child) = std::process::Command::new("pbcopy")
346	                        .stdin(std::process::Stdio::piped())
347	                        .spawn()
348	                    {
349	                        if let Some(stdin) = child.stdin.as_mut() {
350	                            let _ = stdin.write_all(prompt.as_bytes());
351	                        }
352	                        let _ = child.wait();
353	                    }
354	                }
355	
356	                let session_name = cfg.session_name.trim().to_string();
357	                return Ok(Some(TuiOutcome::LaunchAgent {
358	                    cli: cfg.agent_cli,
359	                    cwd: launch_dir,
360	                    session_name,
361	                    prompt,
362	                }));
363	            }
364	            Cmd::Tidy => {
365	                let repo_idx_opt = if let Some(Screen::Sessions { repo_idx, .. }) = stack.last() {
366	                    Some(*repo_idx)
367	                } else {
368	                    None
369	                };
370	                if let Some(repo_idx) = repo_idx_opt {
371	                    let repo = &repos[repo_idx];
372	                    let (tw, th) = terminal::size().unwrap_or((120, 40));
373	                    let changed = super::tidy::tui_tidy(
374	                        stdout,
375	                        std::slice::from_ref(repo),
376	                        7,
377	                        false,
378	                        tw as usize,
379	                        th as usize,
380	                    );
381	                    if changed {
382	                        if let Some(Screen::Sessions { worktrees, .. }) = stack.last_mut() {
383	                            *worktrees = fetch_worktrees(&repos[repo_idx].directory);
384	                        }
385	                    }
386	                    execute!(stdout, terminal::Clear(ClearType::All)).ok();
387	                }
388	            }
389	            Cmd::Redraw => { execute!(stdout, terminal::Clear(ClearType::All)).ok(); }
390	        }
391	    }
392	
393	    Ok(None)
394	}

398	fn draw_repos(
399	    stdout: &mut impl Write,
400	    repos: &[Repository],
401	    sel: usize,
402	    current_repo_dir: Option<&str>,
403	    w: usize,
404	    h: usize,

406	) -> io::Result<()> {
407	    let t = crate::theme::get();
408	    let sel_bg = t.sel_bg;
409	
410	    execute!(stdout, cursor::MoveTo(0, 0))?;
411	
412	    let content_h = h.saturating_sub(1);
413	    let mut row = 0usize;
414	
415	    let name_w = repos.iter().map(|r| r.name.chars().count()).max().unwrap_or(0);
416	    let dir_w  = repos.iter().map(|r| r.directory.chars().count()).max().unwrap_or(0);
417

422	        let is_cur = current_repo_dir == Some(repo.directory.as_str());
423	        let dot_col = if is_cur { t.fresh } else { t.text_dim };
424

426	        let dir_padded  = format!("{:<dir_w$}",  repo.directory);
427	        let line = format!(

429	            repo.remote, pm = t.text_primary, dm = t.text_dim,
430	        );
431	
432	        print_row(stdout, &line, is_sel, sel_bg, w, row as u16)?;
433	        row += 1;
434	    }
435	
436	    // Clear remaining rows
437	    while row < content_h {
438	        execute!(stdout, cursor::MoveTo(0, row as u16), terminal::Clear(ClearType::UntilNewLine))?;
439	        row += 1;
440	    }
441	
442	    // Status bar
443	    let cd_hint = if has_cd { "   c: cd" } else { "" };
444	    let bar = format!(
445	        "  {} repos   ↑↓/jk navigate   space: sessions   s: new session   /: search{}   q: quit  ",
446	        repos.len(), cd_hint
447	    );
448	    draw_status(stdout, &bar, w, h)?;
449	
450	    stdout.flush()
451	}

453	fn draw_sessions(
454	    stdout: &mut impl Write,
455	    repos: &[Repository],
456	    repo_idx: usize,
457	    sessions: &[RepoSession],
458	    worktrees: &[RepoWorktree],
459	    sel: usize,
460	    w: usize,
461	    h: usize,
462	) -> io::Result<()> {
463	    let t = crate::theme::get();
464	    let sel_bg = t.sel_bg;
465	
466	    execute!(stdout, cursor::MoveTo(0, 0))?;
467	    write!(stdout, "\x1b[{hd}m{}\x1b[0m", repos[repo_idx].name, hd = t.header)?;
468	    execute!(stdout, terminal::Clear(ClearType::UntilNewLine))?;
469	
470	    let content_h = h.saturating_sub(2); // header + status bar
471	    let mut row = 1usize;
472	
473	    // ── Worktrees ────────────────────────────────────────────────────────────
474	    if !worktrees.is_empty() {
475	        for wt in worktrees.iter().take(content_h.saturating_sub(row + 1)) {
476	            if row >= content_h { break; }
477	
478	            let (branch_col, branch_label) = if wt.branch == "(detached)" {
479	                (t.error, format!("detached:{}", &wt.head))
480	            } else if wt.is_main {
481	                (t.accent, wt.branch.clone())
482	            } else {
483	                (t.link, wt.branch.clone())
484	            };
485	
486	            let path_short = short_path_s(&wt.path);
487	            let line = format!(
488	                "\x1b[{dm}m  @ \x1b[{branch_col}m{branch_label}\x1b[0m  \x1b[{dm}m{path_short}\x1b[0m",
489	                dm = t.text_dim,
490	            );
491	            execute!(stdout, cursor::MoveTo(0, row as u16))?;
492	            write!(stdout, "{line}")?;
493	            execute!(stdout, terminal::Clear(ClearType::UntilNewLine))?;
494	            row += 1;
495	        }
496	
497	        // Blank separator between worktrees and sessions
498	        if row < content_h {
499	            execute!(stdout, cursor::MoveTo(0, row as u16), terminal::Clear(ClearType::UntilNewLine))?;
500	            row += 1;
501	        }
502	    }
503	
504	    // ── Sessions ─────────────────────────────────────────────────────────────
505	    let sessions_start = row;
506	    let sessions_h = content_h.saturating_sub(sessions_start);
507	    let scroll = if sel >= sessions_h { sel + 1 - sessions_h } else { 0 };
508	
509	    if sessions.is_empty() {
510	        if row < content_h {
511	            execute!(stdout, cursor::MoveTo(0, row as u16))?;
512	            write!(stdout, "\x1b[{dm}m  no sessions found\x1b[0m", dm = t.text_dim)?;
513	            execute!(stdout, terminal::Clear(ClearType::UntilNewLine))?;
514	            row += 1;
515	        }
516	    } else {
517	        // Pre-compute column widths for table alignment
518	        let name_w   = sessions.iter().map(|s| s.session_name.trim().chars().count()).max().unwrap_or(0).min(40);
519	        let branch_w = sessions.iter().map(|s| s.branch.chars().count()).max().unwrap_or(0);
520	        let agent_w  = sessions.iter().map(|s| s.agent.chars().count()).max().unwrap_or(0);
521	
522	        for (i, s) in sessions.iter().enumerate().skip(scroll) {
523	            if row >= content_h { break; }
524	
525	            let id_short: String = s.session_id.chars().take(8).collect();
526	            let ts = relative_time(s.updated_at);
527	            let name: String = s.session_name.trim().chars().take(name_w).collect();
528	            let age = (Utc::now() - s.updated_at).num_seconds().max(0);
529	            let dot_col = match age {
530	                a if a < 900   => t.fresh,
531	                a if a < 3_600 => t.moderate,
532	                _              => t.text_dim,
533	            };
534	
535	            let (name_col, meta_col, dot_char) = if s.backed_up {
536	                (t.backed_name, t.backed_meta, "*")
537	            } else {
538	                (t.unbacked_name, t.unbacked_meta, "·")
539	            };
540	            let branch_col = if s.backed_up { t.link } else { t.stale };
541	
542	            let name_padded = format!("{:<name_w$}", name);
543	            let mut line = format!("\x1b[{dot_col}m{dot_char}\x1b[0m \x1b[{name_col}m{name_padded}\x1b[0m");
544	
545	            if branch_w > 0 {
546	                let b: String = s.branch.chars().take(branch_w).collect();
547	                let pad = " ".repeat(branch_w - b.chars().count());
548	                line.push_str(&format!("  \x1b[{branch_col}m{b}{pad}\x1b[0m"));
549	            }
550	
551	            if agent_w > 0 {
552	                let col = if s.backed_up { agent_color(&s.agent) } else { t.stale_agent };
553	                let a: String = s.agent.chars().take(agent_w).collect();
554	                let pad = " ".repeat(agent_w - a.chars().count());
555	                line.push_str(&format!("  \x1b[38;5;{col}m{a}{pad}\x1b[0m"));
556	            }
557	
558	            line.push_str(&format!("  \x1b[{meta_col}m{id_short}  {ts}\x1b[0m"));
559	
560	            print_row(stdout, &line, i == sel, sel_bg, w, row as u16)?;
561	            row += 1;
562	        }
563	    }
564	
565	    while row < content_h {
566	        execute!(stdout, cursor::MoveTo(0, row as u16), terminal::Clear(ClearType::UntilNewLine))?;
567	        row += 1;
568	    }
569	
570	    let bar = format!(
571	        "  {} sessions   ↑↓/jk navigate   space: view   r: resume   s: new session   n: new worktree   t: tidy   /: search   ←/h: back   q: quit  ",
572	        sessions.len()
573	    );
574	    draw_status(stdout, &bar, w, h)?;
575	
576	    stdout.flush()
577	}

600	// ── Worktree loading ──────────────────────────────────────────────────────────
601	
602	fn fetch_worktrees(repo_dir: &str) -> Vec<RepoWorktree> {
603	    let out = std::process::Command::new("git")
604	        .args(["worktree", "list", "--porcelain"])
605	        .current_dir(repo_dir)
606	        .output();
607	
608	    let Ok(out) = out else { return vec![]; };
609	    if !out.status.success() { return vec![]; }
610	    let text = String::from_utf8_lossy(&out.stdout);
611	
612	    let mut result = Vec::new();
613	    let mut first = true;
614	
615	    // Each worktree block is separated by a blank line.
616	    for block in text.split("\n\n") {
617	        let mut path = String::new();
618	        let mut head = String::new();
619	        let mut branch = String::new();
620	        let mut detached = false;
621	
622	        for line in block.lines() {
623	            if let Some(v) = line.strip_prefix("worktree ") {
624	                path = v.to_string();
625	            } else if let Some(v) = line.strip_prefix("HEAD ") {
626	                head = v.chars().take(8).collect();
627	            } else if let Some(v) = line.strip_prefix("branch refs/heads/") {
628	                branch = v.to_string();
629	            } else if line == "detached" {
630	                detached = true;
631	            }
632	        }
633	
634	        if path.is_empty() { continue; }
635	        if detached { branch = "(detached)".to_string(); }
636	        if branch.is_empty() { continue; } // bare worktree — skip
637	
638	        result.push(RepoWorktree { path, branch, head, is_main: first });
639	        first = false;
640	    }
641	
642	    result
643	}

645	// ── Session loading ───────────────────────────────────────────────────────────
646	
647	fn load_sessions(cwd_prefix: &str) -> Vec<RepoSession> {
648	    let mut sessions: Vec<RepoSession> = Vec::new();
649	
650	    // From DB
651	    if let Ok(conn) = db::connect() {
652	        if let Ok(mut stmt) = conn.prepare(
653	            "SELECT session_id, session_name, updated_at, agent_name FROM sessions WHERE cwd LIKE ?1 ORDER BY updated_at DESC"
654	        ) {
655	            let prefix_pattern = format!("{cwd_prefix}%");
656	            let _ = stmt.query_map([&prefix_pattern], |row| {
657	                let ts_str: String = row.get(2)?;
658	                let updated_at = DateTime::parse_from_rfc3339(&ts_str)
659	                    .map(|dt| dt.with_timezone(&Utc))
660	                    .unwrap_or_else(|_| Utc::now());
661	                Ok(RepoSession {
662	                    session_id: row.get(0)?,
663	                    session_name: row.get(1)?,
664	                    updated_at,
665	                    branch: String::new(),
666	                    agent: row.get(3)?,
667	                    backed_up: true,
668	                })
669	            }).map(|rows| {
670	                for r in rows.flatten() { sessions.push(r); }
671	            });
672	        }
673	    }
674	
675	    // From ~/.claude/projects/ — refresh existing DB sessions and add new ones
676	    if let Ok(home) = std::env::var("HOME") {
677	        let projects = PathBuf::from(&home).join(".claude/projects");
678	        if let Ok(dirs) = std::fs::read_dir(&projects) {
679	            for dir_entry in dirs.flatten() {
680	                let dir = dir_entry.path();
681	                if !dir.is_dir() { continue }
682	                if let Ok(files) = std::fs::read_dir(&dir) {
683	                    for f in files.flatten() {
684	                        let path = f.path();
685	                        if path.extension().and_then(|e| e.to_str()) != Some("jsonl") { continue }
686	                        let session_id = match path.file_stem().and_then(|s| s.to_str()) {
687	                            Some(s) => s.to_string(),
688	                            None => continue,
689	                        };
690	
691	                        let file_mtime = f.metadata().ok()
692	                            .and_then(|m| m.modified().ok())
693	                            .map(DateTime::<Utc>::from)
694	                            .unwrap_or_else(Utc::now);
695	
696	                        let Ok(file) = std::fs::File::open(&path) else { continue };
697	                        let reader = std::io::BufReader::new(file);
698	                        let mut session_name = String::new();
699	                        let mut cwd_found = String::new();
700	                        let mut last_prompt = String::new();
701	                        let mut git_branch = String::new();
702	
703	                        for line in reader.lines().flatten() {
704	                            let Ok(v) = serde_json::from_str::<serde_json::Value>(&line) else { continue };
705	                            match v["type"].as_str() {
706	                                Some("custom-title") => {
707	                                    if let Some(t) = v["customTitle"].as_str() { session_name = t.to_string(); }
708	                                }
709	                                Some("user") | Some("system") => {
710	                                    if cwd_found.is_empty() {
711	                                        if let Some(c) = v["cwd"].as_str() { cwd_found = c.to_string(); }
712	                                    }
713	                                }
714	                                _ => {}
715	                            }
716	                            if v["type"].as_str() == Some("user") {
717	                                if let Some(t) = user_text(&v["message"]["content"]) {
718	                                    last_prompt = t;
719	                                }
720	                            }
721	                            // Track the most recent branch seen in this session
722	                            if let Some(b) = v["gitBranch"].as_str() {
723	                                if !b.is_empty() { git_branch = b.to_string(); }
724	                            }
725	                        }
726	
727	                        let display_name = if !session_name.is_empty() { session_name } else { last_prompt };
728	
729	                        if let Some(existing) = sessions.iter_mut().find(|s| s.session_id == session_id) {
730	                            // Refresh the name, timestamp, and branch from live JSONL
731	                            if !display_name.is_empty() { existing.session_name = display_name; }
732	                            if file_mtime > existing.updated_at { existing.updated_at = file_mtime; }
733	                            if !git_branch.is_empty() { existing.branch = git_branch; }
734	                            continue;
735	                        }
736	
737	                        if !cwd_found.starts_with(cwd_prefix) { continue }
738	
739	                        sessions.push(RepoSession {
740	                            session_id,
741	                            session_name: display_name,
742	                            updated_at: file_mtime,
743	                            branch: git_branch,
744	                            agent: String::new(),
745	                            backed_up: false,
746	                        });
747	                    }
748	                }
749	            }
750	        }
751	    }
752	
753	    sessions.sort_by(|a, b| b.updated_at.cmp(&a.updated_at));
754	    sessions
755	}

759	/// Generic single-line input shown in the status bar. Returns None on Esc.
760	fn collect_text_input(stdout: &mut impl Write, prefix: &str, w: usize, h: usize) -> Option<String> {
761	    let mut value = String::new();
762	    execute!(stdout, cursor::Show).ok();
763	    draw_input_bar(stdout, prefix, &value, w, h).ok();
764	
765	    loop {
766	        match event::read().ok()? {
767	            Event::Key(k) => match k.code {
768	                KeyCode::Esc => {
769	                    execute!(stdout, cursor::Hide).ok();
770	                    return None;
771	                }
772	                KeyCode::Enter => {
773	                    execute!(stdout, cursor::Hide).ok();
774	                    return Some(value);
775	                }
776	                KeyCode::Backspace => {
777	                    value.pop();
778	                    draw_input_bar(stdout, prefix, &value, w, h).ok();
779	                }
780	                KeyCode::Char(c) => {
781	                    value.push(c);
782	                    draw_input_bar(stdout, prefix, &value, w, h).ok();
783	                }
784	                _ => {}
785	            },
786	            _ => {}
787	        }
788	    }
789	}

791	fn draw_input_bar(stdout: &mut impl Write, prefix: &str, value: &str, w: usize, h: usize) -> io::Result<()> {
792	    let prefix_len = prefix.chars().count();
793	    let text: String = value.chars().take(w.saturating_sub(prefix_len + 2)).collect();
794	    let content = format!("{prefix}{text}");
795	    let padded = format!("{:<width$}", content, width = w);

189	pub(crate) fn parse_session(
190	    meta_bytes: &[u8],
191	    jsonl_bytes: &[u8],
192	    user: &str,
193	) -> Result<(String, String, String, String, String, String)> {
194	    #[derive(Deserialize)]
195	    struct SessionMetadata {
196	        session_id: String,
197	        agent: Option<String>,
198	        created_at: Option<String>,
199	        summary: Option<Summary>,
200	    }
201	    #[derive(Deserialize)]
202	    struct Summary {
203	        intent: Option<String>,
204	    }
205	
206	    let meta: SessionMetadata = serde_json::from_slice(meta_bytes)
207	        .context("failed to parse metadata.json")?;
208	
209	    let created_at: DateTime<Utc> = meta
210	        .created_at
211	        .as_deref()
212	        .and_then(|s| DateTime::parse_from_rfc3339(s).ok())
213	        .map(|dt| dt.with_timezone(&Utc))
214	        .unwrap_or_else(Utc::now);
215	
216	    let session_name = meta
217	        .summary
218	        .and_then(|s| s.intent)
219	        .filter(|s| !s.is_empty())
220	        .unwrap_or_else(|| format!("session:{}", &meta.session_id[..8]));
221	
222	    let agent_name = meta.agent.unwrap_or_else(|| "unknown".to_string());
223	
224	    let mut latest: Option<DateTime<Utc>> = None;
225	    let mut cwd = String::new();
226	    let mut last_prompt: Option<String> = None;
227	    let mut custom_title: Option<String> = None;
228	
229	    for line in jsonl_bytes.split(|&b| b == b'\n') {
230	        if line.is_empty() { continue; }
231	        let v: Value = match serde_json::from_slice(line) {
232	            Ok(v) => v,
233	            Err(_) => continue,
234	        };
235	
236	        if let Some(ts) = v.get("timestamp").and_then(Value::as_str) {
237	            if let Ok(dt) = DateTime::parse_from_rfc3339(ts) {
238	                let dt: DateTime<Utc> = dt.with_timezone(&Utc);
239	                if latest.map_or(true, |l| dt > l) { latest = Some(dt); }
240	            }
241	        }
242	
243	        if cwd.is_empty() {
244	            if let Some(c) = v.get("cwd").and_then(Value::as_str) { cwd = c.to_string(); }
245	        }
246	
247	        match v.get("type").and_then(Value::as_str) {
248	            Some("custom-title") => {
249	                if let Some(t) = v.get("customTitle").and_then(Value::as_str) {
250	                    if !t.trim().is_empty() { custom_title = Some(t.trim().to_string()); }
251	                }
252	            }
253	            Some("user") => {
254	                if let Some(Value::String(text)) = v.get("message").and_then(|m| m.get("content")) {
255	                    let text = text.trim().to_string();
256	                    if !text.is_empty() { last_prompt = Some(text); }
257	                }
258	            }
259	            _ => {}
260	        }
261	    }
262	
263	    let updated_at = latest.unwrap_or(created_at);
264	    // Priority: explicit rename > last user prompt > metadata summary
265	    let session_name = custom_title.or(last_prompt).unwrap_or(session_name);
266	
267	    Ok((
268	        meta.session_id,
269	        agent_name,
270	        created_at.to_rfc3339(),
271	        updated_at.to_rfc3339(),
272	        cwd,
273	        session_name,
274	    ))
275	}
