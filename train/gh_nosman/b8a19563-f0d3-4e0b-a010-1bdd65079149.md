404	fn draw(
405	    stdout: &mut impl Write,
406	    groups: &[Group],
407	    query:  &str,
408	    ms:     u128,
409	    sel:    usize,
410	    scroll: usize,
411	    w:      usize,
412	    h:      usize,

503	                    };

506	                    screen_row += 1;
507	                }
508	                abs_row += 1;
509	            }
510	        }
511	
512	        // Blank separator row between groups
513	        if abs_row >= scroll && screen_row <= content_h {
514	            queue!(buf, cursor::MoveTo(0, screen_row as u16), terminal::Clear(ClearType::UntilNewLine))?;
515	            screen_row += 1;

517	        abs_row += 1;
518	    }

521	    while screen_row <= content_h {
522	        queue!(buf, cursor::MoveTo(0, screen_row as u16), terminal::Clear(ClearType::UntilNewLine))?;

524	    }

529	    queue!(buf, cursor::MoveTo(0, (h - 1) as u16))?;
530	    write!(buf, "\x1b[7m{padded}\x1b[0m")?;
531	
532	    stdout.write_all(&buf)?;
533	    stdout.flush()
534	}

640	fn rel_time_hit(iso: &str) -> String {
641	    if iso.is_empty() { return String::new(); }
642	    let Ok(dt) = DateTime::parse_from_rfc3339(iso) else { return String::new(); };
643	    let secs = (Utc::now() - dt.with_timezone(&Utc)).num_seconds().max(0);
644	    if secs < 604_800 {
645	        match secs {
646	            s if s < 60     => "just now".into(),
647	            s if s < 3_600  => format!("{} min ago", s / 60),
648	            s if s < 86_400 => format!("{} hr ago",  s / 3_600),
649	            s               => format!("{} day{} ago", s / 86_400, if s / 86_400 == 1 { "" } else { "s" }),
650	        }
651	    } else {
652	        dt.with_timezone(&chrono::Local).format("%m/%d/%y").to_string()
653	    }
654	}

236	            timestamp,

## Session Name Question

**No, SessionStart and Stop hooks do NOT receive the session name/customTitle** (the value passed via `-n`/`--name` flag). The documentation explicitly states that neither hook receives this field.

However, you *can* set a session title via **UserPromptSubmit** hooks using the output:

```json
{
  "hookSpecificOutput": {
    "hookEventName": "UserPromptSubmit",
    "sessionTitle": "My session title"
  }
}
```

This is relevant to gossamer's current implementation in `session_start.rs` — if you're trying to capture the user-provided session name, you'd need to either:
1. Use the `UserPromptSubmit` hook instead
2. Have the user pass the name through a different mechanism (like `~/.gossamer/pending_session_name`, which gossamer already does for `gossamer new-session`)

64	        .output()?;
65	
66	    if !check.status.success() {
67	        return Ok(0);
68	    }
69	
70	    let current_head = String::from_utf8(check.stdout)?.trim().to_string();
71	
72	    // Load stored watermark.
73	    let stored_head: Option<String> = conn.query_row(
74	        "SELECT last_indexed_commit FROM repositories WHERE directory = ?1",
75	        [repo_dir],
76	        |row| row.get(0),
77	    ).ok().flatten();
78	
79	    if stored_head.as_deref() == Some(current_head.as_str()) {
80	        return Ok(0); // nothing new
81	    }
82	
83	    // Collect metadata paths: added or modified since watermark (or all if first run).
84	    let meta_paths: Vec<String> = if let Some(ref last) = stored_head {
85	        // git diff gives a deduplicated file list between two tree states, so each
86	        // session appears at most once even if it was checkpointed many times.
87	        let out = Command::new("git")
88	            .args(["diff", "--name-only", "--diff-filter=AM", last, BRANCH])
89	            .current_dir(repo_dir)
90	            .output()?;
91	        String::from_utf8(out.stdout)?
92	            .lines()
93	            .filter(|l| is_meta_path(l))
94	            .map(str::to_string)
95	            .collect()
96	    } else {
97	        let out = Command::new("git")
98	            .args(["ls-tree", "-r", "--name-only", BRANCH])
99	            .current_dir(repo_dir)
100	            .output()?;
101	        String::from_utf8(out.stdout)?
102	            .lines()
103	            .filter(|l| is_meta_path(l))
104	            .map(str::to_string)
105	            .collect()
106	    };
107	
108	    let user = std::env::var("USER")
109	        .or_else(|_| std::env::var("USERNAME"))
110	        .unwrap_or_else(|_| "unknown".to_string());
111	
112	    let mut count = 0;
113	    for meta_path in &meta_paths {
114	        let jsonl_path = format!(
115	            "{}full.jsonl",
116	            &meta_path[..meta_path.len() - "metadata.json".len()]
117	        );
118	        let meta_bytes = match git_show(repo_dir, meta_path) {
119	            Ok(b) => b,
120	            Err(e) => { eprintln!("  skipping {}: {}", meta_path, e); continue; }
121	        };
122	        let jsonl_bytes = match git_show(repo_dir, &jsonl_path) {
123	            Ok(b) => b,
124	            Err(_) => continue,
125	        };
126	        match parse_session(&meta_bytes, &jsonl_bytes, &user) {
127	            Ok((session_id, agent_name, created_at, updated_at, cwd, session_name)) => {
128	                upsert_session(conn, &session_id, &agent_name, &user,
129	                               &created_at, &updated_at, &cwd, &session_name)?;
130	                count += 1;
131	            }
132	            Err(e) => eprintln!("  skipping {}: {}", meta_path, e),
133	        }
134	    }
135	
136	    // Advance watermark.
137	    conn.execute(
138	        "UPDATE repositories SET last_indexed_commit = ?1 WHERE directory = ?2",
139	        rusqlite::params![current_head, repo_dir],
140	    )?;
141	
142	    Ok(count)
143	}
