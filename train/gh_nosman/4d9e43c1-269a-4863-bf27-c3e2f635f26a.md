385	    let mut row = 0usize;
386	
387	    let name_w = repos.iter().map(|r| r.name.chars().count()).max().unwrap_or(0);
388	    let dir_w  = repos.iter().map(|r| r.directory.chars().count()).max().unwrap_or(0);
389	
390	    for (i, repo) in repos.iter().enumerate() {
391	        if row >= content_h { break }
392	
393	        let is_sel = i == sel;
394	        let is_cur = current_repo_dir == Some(repo.directory.as_str());
395	        let dot_col = if is_cur { "38;5;82" } else { "38;5;240" };
396	
397	        let name_padded = format!("{:<name_w$}", repo.name);
398	        let dir_padded  = format!("{:<dir_w$}",  repo.directory);
399	        let line = format!(
400	            "\x1b[{dot_col}m*\x1b[0m \x1b[38;5;255m{name_padded}\x1b[0m  \x1b[38;5;240m{dir_padded}  {}\x1b[0m",
401	            repo.remote
402	        );
403	
404	        print_row(stdout, &line, is_sel, SEL_BG, w, row as u16)?;
405	        row += 1;
406	    }
407	
408	    // Clear remaining rows
409	    while row < content_h {
410	        execute!(stdout, cursor::MoveTo(0, row as u16), terminal::Clear(ClearType::UntilNewLine))?;
411	        row += 1;
412	    }
413	
414	    // Status bar
415	    let cd_hint = if has_cd { "   c: cd" } else { "" };
416	    let bar = format!(
417	        "  {} repos   ↑↓/jk navigate   space: sessions   s: new session   /: search{}   q: quit  ",
418	        repos.len(), cd_hint
419	    );
420	    draw_status(stdout, &bar, w, h)?;
421	
422	    stdout.flush()
423	}

425	fn draw_sessions(
426	    stdout: &mut impl Write,
427	    repos: &[Repository],
428	    repo_idx: usize,
429	    sessions: &[RepoSession],
430	    worktrees: &[RepoWorktree],
431	    sel: usize,
432	    w: usize,
433	    h: usize,
434	) -> io::Result<()> {
435	    const SEL_BG: &str = "48;5;236";
436	
437	    execute!(stdout, cursor::MoveTo(0, 0))?;
438	    write!(stdout, "\x1b[1;38;5;229m{}\x1b[0m", repos[repo_idx].name)?;
439	    execute!(stdout, terminal::Clear(ClearType::UntilNewLine))?;
440	
441	    let content_h = h.saturating_sub(2); // header + status bar
442	    let mut row = 1usize;
443	
444	    // ── Worktrees ────────────────────────────────────────────────────────────
445	    if !worktrees.is_empty() {
446	        for wt in worktrees.iter().take(content_h.saturating_sub(row + 1)) {
447	            if row >= content_h { break; }
448	
449	            let (branch_col, branch_label) = if wt.branch == "(detached)" {
450	                ("38;5;196", format!("detached:{}", &wt.head))
451	            } else if wt.is_main {
452	                ("38;5;229", wt.branch.clone())   // gold — main worktree
453	            } else {
454	                ("38;5;75",  wt.branch.clone())   // blue — linked worktree
455	            };
456	
457	            let path_short = short_path_s(&wt.path);
458	            let line = format!(
459	                "\x1b[38;5;240m  @ \x1b[{branch_col}m{branch_label}\x1b[0m  \x1b[38;5;240m{path_short}\x1b[0m"
460	            );
461	            execute!(stdout, cursor::MoveTo(0, row as u16))?;
462	            write!(stdout, "{line}")?;
463	            execute!(stdout, terminal::Clear(ClearType::UntilNewLine))?;
464	            row += 1;
465	        }
466	
467	        // Blank separator between worktrees and sessions
468	        if row < content_h {
469	            execute!(stdout, cursor::MoveTo(0, row as u16), terminal::Clear(ClearType::UntilNewLine))?;
470	            row += 1;
471	        }
472	    }
473	
474	    // ── Sessions ─────────────────────────────────────────────────────────────
475	    let sessions_start = row;
476	    let sessions_h = content_h.saturating_sub(sessions_start);
477	    let scroll = if sel >= sessions_h { sel + 1 - sessions_h } else { 0 };
478	
479	    if sessions.is_empty() {
480	        if row < content_h {
481	            execute!(stdout, cursor::MoveTo(0, row as u16))?;
482	            write!(stdout, "\x1b[38;5;240m  no sessions found\x1b[0m")?;
483	            execute!(stdout, terminal::Clear(ClearType::UntilNewLine))?;
484	            row += 1;
485	        }
486	    } else {
487	        // Pre-compute column widths for table alignment
488	        let name_w   = sessions.iter().map(|s| s.session_name.trim().chars().count()).max().unwrap_or(0).min(40);
489	        let branch_w = sessions.iter().map(|s| s.branch.chars().count()).max().unwrap_or(0);
490	        let agent_w  = sessions.iter().map(|s| s.agent.chars().count()).max().unwrap_or(0);
491	
492	        for (i, s) in sessions.iter().enumerate().skip(scroll) {
493	            if row >= content_h { break; }
494	
495	            let id_short: String = s.session_id.chars().take(8).collect();
496	            let ts = relative_time(s.updated_at);
497	            let name: String = s.session_name.trim().chars().take(name_w).collect();
498	            let age = (Utc::now() - s.updated_at).num_seconds().max(0);
499	            let dot_col = match age {
500	                a if a < 900   => "38;5;82",
501	                a if a < 3_600 => "38;5;214",
502	                _              => "38;5;240",
503	            };
504	
505	            let (name_col, meta_col, dot_char) = if s.backed_up {
506	                ("1;38;5;255", "38;5;240", "*")
507	            } else {
508	                ("38;5;242", "38;5;238", "·")
509	            };
510	            let branch_col = if s.backed_up { "38;5;75" } else { "38;5;239" };
511	
512	            let name_padded = format!("{:<name_w$}", name);
513	            let mut line = format!("\x1b[{dot_col}m{dot_char}\x1b[0m \x1b[{name_col}m{name_padded}\x1b[0m");
514	
515	            if branch_w > 0 {
516	                let b: String = s.branch.chars().take(branch_w).collect();
517	                let pad = " ".repeat(branch_w - b.chars().count());
518	                line.push_str(&format!("  \x1b[{branch_col}m{b}{pad}\x1b[0m"));
519	            }
520	
521	            if agent_w > 0 {
522	                let col = if s.backed_up { agent_color(&s.agent) } else { 239 };
523	                let a: String = s.agent.chars().take(agent_w).collect();
524	                let pad = " ".repeat(agent_w - a.chars().count());
525	                line.push_str(&format!("  \x1b[38;5;{col}m{a}{pad}\x1b[0m"));
526	            }
527	
528	            line.push_str(&format!("  \x1b[{meta_col}m{id_short}  {ts}\x1b[0m"));
529	
530	            print_row(stdout, &line, i == sel, SEL_BG, w, row as u16)?;
531	            row += 1;
532	        }
533	    }
534	
535	    while row < content_h {
536	        execute!(stdout, cursor::MoveTo(0, row as u16), terminal::Clear(ClearType::UntilNewLine))?;
537	        row += 1;
538	    }
539	
540	    let bar = format!(
541	        "  {} sessions   ↑↓/jk navigate   space: view   r: resume   s: new session   n: new worktree   /: search   ←/h: back   q: quit  ",
542	        sessions.len()
543	    );
544	    draw_status(stdout, &bar, w, h)?;
545	
546	    stdout.flush()
547	}

570	// ── Worktree loading ──────────────────────────────────────────────────────────
571	
572	fn fetch_worktrees(repo_dir: &str) -> Vec<RepoWorktree> {
573	    let out = std::process::Command::new("git")
574	        .args(["worktree", "list", "--porcelain"])
575	        .current_dir(repo_dir)
576	        .output();
577	
578	    let Ok(out) = out else { return vec![]; };
579	    if !out.status.success() { return vec![]; }
580	    let text = String::from_utf8_lossy(&out.stdout);
581	
582	    let mut result = Vec::new();
583	    let mut first = true;
584	
585	    // Each worktree block is separated by a blank line.
586	    for block in text.split("\n\n") {
587	        let mut path = String::new();
588	        let mut head = String::new();
589	        let mut branch = String::new();
590	        let mut detached = false;
591	
592	        for line in block.lines() {
593	            if let Some(v) = line.strip_prefix("worktree ") {
594	                path = v.to_string();
595	            } else if let Some(v) = line.strip_prefix("HEAD ") {
596	                head = v.chars().take(8).collect();
597	            } else if let Some(v) = line.strip_prefix("branch refs/heads/") {
598	                branch = v.to_string();
599	            } else if line == "detached" {
600	                detached = true;
601	            }
602	        }
603	
604	        if path.is_empty() { continue; }
605	        if detached { branch = "(detached)".to_string(); }
606	        if branch.is_empty() { continue; } // bare worktree — skip
607	
608	        result.push(RepoWorktree { path, branch, head, is_main: first });
609	        first = false;
610	    }
611	
612	    result
613	}

615	// ── Session loading ───────────────────────────────────────────────────────────
616	
617	fn load_sessions(cwd_prefix: &str) -> Vec<RepoSession> {
618	    let mut sessions: Vec<RepoSession> = Vec::new();
619	
620	    // From DB
621	    if let Ok(conn) = db::connect() {
622	        if let Ok(mut stmt) = conn.prepare(
623	            "SELECT session_id, session_name, updated_at, agent_name FROM sessions WHERE cwd LIKE ?1 ORDER BY updated_at DESC"
624	        ) {
625	            let prefix_pattern = format!("{cwd_prefix}%");
626	            let _ = stmt.query_map([&prefix_pattern], |row| {
627	                let ts_str: String = row.get(2)?;
628	                let updated_at = DateTime::parse_from_rfc3339(&ts_str)
629	                    .map(|dt| dt.with_timezone(&Utc))
630	                    .unwrap_or_else(|_| Utc::now());
631	                Ok(RepoSession {
632	                    session_id: row.get(0)?,
633	                    session_name: row.get(1)?,
634	                    updated_at,
635	                    branch: String::new(),
636	                    agent: row.get(3)?,
637	                    backed_up: true,
638	                })
639	            }).map(|rows| {
640	                for r in rows.flatten() { sessions.push(r); }
641	            });
642	        }
643	    }
644	
645	    // From ~/.claude/projects/ — refresh existing DB sessions and add new ones
646	    if let Ok(home) = std::env::var("HOME") {
647	        let projects = PathBuf::from(&home).join(".claude/projects");
648	        if let Ok(dirs) = std::fs::read_dir(&projects) {
649	            for dir_entry in dirs.flatten() {
650	                let dir = dir_entry.path();
651	                if !dir.is_dir() { continue }
652	                if let Ok(files) = std::fs::read_dir(&dir) {
653	                    for f in files.flatten() {
654	                        let path = f.path();
655	                        if path.extension().and_then(|e| e.to_str()) != Some("jsonl") { continue }
656	                        let session_id = match path.file_stem().and_then(|s| s.to_str()) {
657	                            Some(s) => s.to_string(),
658	                            None => continue,
659	                        };
660	
661	                        let file_mtime = f.metadata().ok()
662	                            .and_then(|m| m.modified().ok())
663	                            .map(DateTime::<Utc>::from)
664	                            .unwrap_or_else(Utc::now);
665	
666	                        let Ok(file) = std::fs::File::open(&path) else { continue };
667	                        let reader = std::io::BufReader::new(file);
668	                        let mut session_name = String::new();
669	                        let mut cwd_found = String::new();
670	                        let mut last_prompt = String::new();
671	                        let mut git_branch = String::new();
672	
673	                        for line in reader.lines().flatten() {
674	                            let Ok(v) = serde_json::from_str::<serde_json::Value>(&line) else { continue };
675	                            match v["type"].as_str() {
676	                                Some("custom-title") => {
677	                                    if let Some(t) = v["customTitle"].as_str() { session_name = t.to_string(); }
678	                                }
679	                                Some("user") | Some("system") => {
680	                                    if cwd_found.is_empty() {
681	                                        if let Some(c) = v["cwd"].as_str() { cwd_found = c.to_string(); }
682	                                    }
683	                                }
684	                                _ => {}
685	                            }
686	                            if v["type"].as_str() == Some("user") {
687	                                if let Some(t) = user_text(&v["message"]["content"]) {
688	                                    last_prompt = t;
689	                                }
690	                            }
691	                            // Track the most recent branch seen in this session
692	                            if let Some(b) = v["gitBranch"].as_str() {
693	                                if !b.is_empty() { git_branch = b.to_string(); }
694	                            }
695	                        }
696	
697	                        let display_name = if !session_name.is_empty() { session_name } else { last_prompt };
698	
699	                        if let Some(existing) = sessions.iter_mut().find(|s| s.session_id == session_id) {
700	                            // Refresh the name, timestamp, and branch from live JSONL
701	                            if !display_name.is_empty() { existing.session_name = display_name; }
702	                            if file_mtime > existing.updated_at { existing.updated_at = file_mtime; }
703	                            if !git_branch.is_empty() { existing.branch = git_branch; }
704	                            continue;
705	                        }
706	
707	                        if !cwd_found.starts_with(cwd_prefix) { continue }
708	
709	                        sessions.push(RepoSession {
710	                            session_id,
711	                            session_name: display_name,
712	                            updated_at: file_mtime,
713	                            branch: git_branch,
714	                            agent: String::new(),
715	                            backed_up: false,
716	                        });
717	                    }
718	                }
719	            }
720	        }
721	    }
722	
723	    sessions.sort_by(|a, b| b.updated_at.cmp(&a.updated_at));
724	    sessions
725	}

730	fn collect_text_input(stdout: &mut impl Write, prefix: &str, w: usize, h: usize) -> Option<String> {
731	    let mut value = String::new();
732	    execute!(stdout, cursor::Show).ok();
733	    draw_input_bar(stdout, prefix, &value, w, h).ok();

736	        match event::read().ok()? {
737	            Event::Key(k) => match k.code {
738	                KeyCode::Esc => {
739	                    execute!(stdout, cursor::Hide).ok();
740	                    return None;
741	                }
742	                KeyCode::Enter => {
743	                    execute!(stdout, cursor::Hide).ok();
744	                    return Some(value);
745	                }
746	                KeyCode::Backspace => {
747	                    value.pop();
748	                    draw_input_bar(stdout, prefix, &value, w, h).ok();
749	                }
750	                KeyCode::Char(c) => {
751	                    value.push(c);
752	                    draw_input_bar(stdout, prefix, &value, w, h).ok();
753	                }
754	                _ => {}
755	            },
756	            _ => {}
757	        }
758	    }
759	}

770	    stdout.flush()

778	
779	/// A 4-step panel drawn over the bottom of the screen. Each completed step
780	/// stays visible with its confirmed value while the user fills the next one.
781	fn new_session_wizard(stdout: &mut impl Write, repo_dir: &str, w: usize, h: usize) -> Option<NewSessionConfig> {
782	    const LABEL_W: usize = 24;
783	    // Panel rows: title, blank, agent, branch, name, prompt, blank  (7 rows + 1 status bar)
784	    const PANEL_H: usize = 7;
785	
786	    let panel_top = h.saturating_sub(PANEL_H + 1);
787	
788	    let mut step = 0usize;           // 0=agent 1=branch 2=name 3=prompt
789	    let mut agent_sel = 0usize;
790	    let mut confirmed_agent = 0usize;
791	    let mut inputs = [String::new(), String::new(), String::new()]; // branch, name, prompt
792	
793	    execute!(stdout, cursor::Show).ok();
794	
795	    loop {
796	        // ── Draw panel ────────────────────────────────────────────────────────
797	        // Title row
798	        execute!(stdout, cursor::MoveTo(0, panel_top as u16)).ok();
799	        write!(stdout, "\x1b[1;38;5;229m  New Session\x1b[0m").ok();
800	        execute!(stdout, terminal::Clear(ClearType::UntilNewLine)).ok();
801	
802	        // Blank separator
803	        execute!(stdout, cursor::MoveTo(0, (panel_top + 1) as u16)).ok();
804	        execute!(stdout, terminal::Clear(ClearType::UntilNewLine)).ok();
805	
806	        let step_rows  = [panel_top + 2, panel_top + 3, panel_top + 4, panel_top + 5];
807	        let step_labels = ["  Agent", "  Worktree branch", "  Session name", "  Prompt"];
808	
809	        for (i, (&row, &label)) in step_rows.iter().zip(step_labels.iter()).enumerate() {
810	            execute!(stdout, cursor::MoveTo(0, row as u16)).ok();
811	            let lpart = format!("{label:<LABEL_W$}");
812	
813	            if i < step {
814	                // Completed — dim label, value in colour
815	                let val = if i == 0 {
816	                    let (name, _, col) = AGENTS[confirmed_agent];
817	                    format!("\x1b[38;5;{col}m{name}\x1b[0m")
818	                } else {
819	                    let v = inputs[i - 1].trim();
820	                    if v.is_empty() { "\x1b[38;5;238m(skip)\x1b[0m".to_string() }
821	                    else            { format!("\x1b[38;5;255m{v}\x1b[0m") }
822	                };
823	                write!(stdout, "\x1b[38;5;240m{lpart}\x1b[0m{val}").ok();
824	            } else if i == step {
825	                // Active step
826	                write!(stdout, "\x1b[38;5;255m{lpart}\x1b[0m").ok();
827	                if i == 0 {
828	                    // Inline agent picker
829	                    for (ai, (name, _, col)) in AGENTS.iter().enumerate() {
830	                        if ai > 0 { write!(stdout, "\x1b[38;5;238m │ \x1b[0m").ok(); }
831	                        if ai == agent_sel {
832	                            write!(stdout, "\x1b[48;5;236;38;5;{col}m {name} \x1b[0m").ok();
833	                        } else {
834	                            write!(stdout, "\x1b[38;5;240m {name}\x1b[0m").ok();
835	                        }
836	                    }
837	                } else {
838	                    // Text input — print typed chars, terminal cursor sits here
839	                    write!(stdout, "\x1b[38;5;255m{}\x1b[0m", inputs[i - 1]).ok();
840	                }
841	            } else {
842	                // Future step — very dim label only
843	                write!(stdout, "\x1b[38;5;238m{lpart}\x1b[0m").ok();
844	            }
845	            execute!(stdout, terminal::Clear(ClearType::UntilNewLine)).ok();
846	        }
847	
848	        // Trailing blank row
849	        let blank2 = panel_top + 6;
850	        if blank2 < h.saturating_sub(1) {
851	            execute!(stdout, cursor::MoveTo(0, blank2 as u16)).ok();
852	            execute!(stdout, terminal::Clear(ClearType::UntilNewLine)).ok();
853	        }
854	
855	        // Status bar hint
856	        let bar = if step == 0 {
857	            "  ↑↓/jk: select agent   enter: confirm   esc: cancel  "
858	        } else {
859	            "  type to enter   enter: next   esc: cancel  "
860	        };
861	        draw_status(stdout, bar, w, h).ok();
862	
863	        // Position terminal cursor for text input steps
864	        if step > 0 {
865	            let row = step_rows[step];
866	            let col = (LABEL_W + inputs[step - 1].len()).min(w.saturating_sub(1));
867	            execute!(stdout, cursor::MoveTo(col as u16, row as u16)).ok();
868	        }
869	        stdout.flush().ok();
870	
871	        // ── Handle input ──────────────────────────────────────────────────────
872	        match event::read().ok()? {
873	            Event::Key(k) => match k.code {
874	                KeyCode::Esc => {
875	                    execute!(stdout, cursor::Hide).ok();
876	                    return None;
877	                }
878	                KeyCode::Up   | KeyCode::Char('k') if step == 0 => {
879	                    if agent_sel > 0 { agent_sel -= 1; }
880	                }
881	                KeyCode::Down | KeyCode::Char('j') if step == 0 => {
882	                    if agent_sel + 1 < AGENTS.len() { agent_sel += 1; }
883	                }
884	                KeyCode::Backspace if step > 0 => {
885	                    inputs[step - 1].pop();
886	                }
887	                KeyCode::Char(c) if step > 0 && !k.modifiers.contains(crossterm::event::KeyModifiers::CONTROL) => {
888	                    inputs[step - 1].push(c);
889	                }
890	                KeyCode::Enter => match step {
891	                    0 => { confirmed_agent = agent_sel; step = 1; }
892	                    1 | 2 => { step += 1; }
893	                    _ => {
894	                        execute!(stdout, cursor::Hide).ok();
895	                        let (agent_name, agent_cli, _) = AGENTS[confirmed_agent];
896	                        let branch_raw = inputs[0].trim().to_string();

66	// ── Entry point ───────────────────────────────────────────────────────────────
67	
68	pub fn run(query: &str, top_k: usize, json: bool) -> Result<()> {
69	    let assets = crate::config::resolve_warp_assets()
70	        .ok_or_else(|| anyhow::anyhow!(
71	            "Witchcraft assets not configured.\nRun: gossamer config <path-to-witchcraft-assets>"
72	        ))?;
73	
74	    let db_path = dirs::home_dir()
75	        .ok_or_else(|| anyhow::anyhow!("cannot determine home directory"))?
76	        .join(".gossamer/search.db");
77	
78	    let db = witchcraft::DB::new_reader(db_path)
79	        .map_err(|e| anyhow::anyhow!("failed to open search DB: {e}"))?;
80	    let device = witchcraft::make_device();
81	    let embedder = witchcraft::Embedder::new(&device, &assets)
82	        .context("failed to load embedder")?;
83	    let mut cache = witchcraft::EmbeddingsCache::new(1);
84	
85	    const FLOOR:         f32 = 0.20;
86	    const LOG_THRESHOLD: f32 = 0.40;
87	    const SES_THRESHOLD: f32 = 0.25;
88	    const REP_THRESHOLD: f32 = 0.25;
89	
90	    let t0 = std::time::Instant::now();
91	    let raw = witchcraft::search(&db, &embedder, &mut cache, query, FLOOR, top_k * 4, true, None)?;
92	    let ms = t0.elapsed().as_millis();
93	
94	    let mut hits: Vec<SearchHit> = raw
95	        .iter()
96	        .filter_map(|(score, meta, bodies, sub_idx, _date)| {
97	            let hit = parse_hit(meta, bodies, *sub_idx as u32);
98	            let threshold = match hit.kind {
99	                HitKind::Log     => LOG_THRESHOLD,
100	                HitKind::Session => SES_THRESHOLD,
101	                HitKind::Repo    => REP_THRESHOLD,
102	            };
103	            if *score >= threshold { Some(hit) } else { None }
104	        })
105	        .take(top_k)
106	        .collect();
107	
108	    for hit in search_sessions_by_name(query) {
109	        let already = hits.iter().any(|h| {
110	            matches!(h.kind, HitKind::Session) && h.session_id == hit.session_id
111	        });
112	        if !already { hits.insert(0, hit); }
113	    }
114	    for hit in search_repos_by_name(query) {
115	        let already = hits.iter().any(|h| {
116	            matches!(h.kind, HitKind::Repo) && h.repo_dir == hit.repo_dir
117	        });
118	        if !already { hits.insert(0, hit); }
119	    }
120	    hits.truncate(top_k);
121	    enrich_hits(&mut hits);
122	
123	    let groups = build_groups(hits);
124	
125	    if json || !io::stdout().is_terminal() {
126	        let arr: Vec<serde_json::Value> = groups.iter().map(|g| {
127	            let hits_json: Vec<serde_json::Value> = g.rows.iter().map(|r| serde_json::json!({
128	                "timestamp": r.hit_ts,
129	                "excerpt": r.lines.join(" "),
130	            })).collect();
131	            serde_json::json!({
132	                "kind": g.kind.label(),
133	                "session_id": g.session_id,
134	                "session_name": g.title,
135	                "dir": g.dir,
136	                "branch": g.branch,
137	                "agent": g.agent,
138	                "updated_at": g.updated_at,
139	                "backed_up": g.backed_up,
140	                "hits": hits_json,
141	            })
142	        }).collect();
143	        println!("{}", serde_json::to_string_pretty(&serde_json::json!({
144	            "query": query,
145	            "results": arr,
146	        }))?);
147	        return Ok(());
148	    }
149	
150	    let orig_hook = std::panic::take_hook();
151	    std::panic::set_hook(Box::new(move |info| {
152	        let mut out = io::stdout();
153	        let _ = execute!(out, LeaveAlternateScreen, cursor::Show);
154	        let _ = terminal::disable_raw_mode();
155	        orig_hook(info);
156	    }));
157	
158	    let mut stdout = io::stdout();
159	    terminal::enable_raw_mode()?;
160	    execute!(stdout, EnterAlternateScreen, cursor::Hide)?;
161	
162	    let result = tui_loop(&mut stdout, &groups, query, ms);
163	
164	    execute!(stdout, LeaveAlternateScreen, cursor::Show)?;
165	    terminal::disable_raw_mode()?;
166	
167	    result
168	}

170	// ── Hit parsing ───────────────────────────────────────────────────────────────
171	
172	fn parse_hit(metadata_json: &str, bodies: &[String], sub_idx: u32) -> SearchHit {
173	    let sub_idx = sub_idx as usize;
174	    let meta: serde_json::Value = serde_json::from_str(metadata_json).unwrap_or_default();
175	    let source = meta["source"].as_str().unwrap_or("claude");
176	
177	    match source {
178	        "session" => {
179	            let name = meta["session_name"].as_str().unwrap_or("").to_string();
180	            let cwd  = meta["cwd"].as_str().unwrap_or("").to_string();
181	            SearchHit {
182	                kind: HitKind::Session,
183	                title: name,
184	                dir: short_path(&cwd),
185	                excerpt_lines: vec![],
186	                session_id: meta["session_id"].as_str().map(str::to_string),
187	                repo_dir: None,
188	                start_ts: None, hit_ts: None, branch: String::new(),
189	                agent: String::new(), backed_up: false, updated_at: String::new(),
190	            }
191	        }
192	        "repo" => {
193	            let name = meta["repo_name"].as_str().unwrap_or("").to_string();
194	            let dir  = meta["repo_dir"].as_str().unwrap_or("").to_string();
195	            SearchHit {
196	                kind: HitKind::Repo,
197	                title: name,
198	                dir: short_path(&dir),
199	                excerpt_lines: vec![],
200	                session_id: None,
201	                repo_dir: Some(dir),
202	                start_ts: None, hit_ts: None, branch: String::new(),
203	                agent: String::new(), backed_up: false, updated_at: String::new(),
204	            }
205	        }
206	        _ => {
207	            // "claude" — session log turn
208	            let name   = meta["session_name"].as_str().unwrap_or("").to_string();
209	            let project = meta["project"].as_str().unwrap_or("").to_string();
210	            let branch  = meta["branch"].as_str().unwrap_or("").to_string();
211	
212	            // Collect context: previous turn, matched turn, next turn.
213	            // bodies[0] is the header; bodies[N] for N>0 is a conversation turn.
214	            let mut excerpt_lines: Vec<String> = Vec::new();
215	            if sub_idx > 1 {
216	                if let Some(prev) = bodies.get(sub_idx - 1) {
217	                    let t: String = prev.trim().chars().take(220).collect();
218	                    if !t.is_empty() { excerpt_lines.push(t); }
219	                }
220	            }
221	            if let Some(matched) = bodies.get(sub_idx) {
222	                let t: String = matched.trim().chars().take(400).collect();
223	                if !t.is_empty() { excerpt_lines.push(t); }
224	            }
225	            if sub_idx + 1 < bodies.len() {
226	                if let Some(next) = bodies.get(sub_idx + 1) {
227	                    let t: String = next.trim().chars().take(220).collect();
228	                    if !t.is_empty() { excerpt_lines.push(t); }
229	                }
230	            }
231	
232	            let hit_ts = if sub_idx > 0 {
233	                meta["turns"][sub_idx - 1]["timestamp"].as_str()
234	                    .filter(|s| !s.is_empty())
235	                    .map(str::to_string)
236	            } else {
237	                None
238	            };
239	
240	            SearchHit {
241	                kind: HitKind::Log,
242	                title: name,
243	                dir: short_path(&project),
244	                excerpt_lines,
245	                session_id: meta["session_id"].as_str().map(str::to_string),
246	                repo_dir: None,
247	                start_ts: hit_ts.clone(),
248	                hit_ts,
249	                branch,
250	                agent: String::new(), backed_up: false, updated_at: String::new(),
251	            }
252	        }
253	    }
254	}

258	fn build_groups(hits: Vec<SearchHit>) -> Vec<Group> {
259	    let mut groups: Vec<Group> = Vec::new();
260	    let mut log_group_idx: std::collections::HashMap<String, usize> = std::collections::HashMap::new();
261	
262	    for hit in hits {
263	        match hit.kind {
264	            HitKind::Log => {
265	                let sid = hit.session_id.clone().unwrap_or_default();
266	                if let Some(&gi) = log_group_idx.get(&sid) {
267	                    groups[gi].rows.push(GroupRow {
268	                        hit_ts: hit.hit_ts,
269	                        start_ts: hit.start_ts,
270	                        lines: hit.excerpt_lines,
271	                    });
272	                } else {
273	                    let gi = groups.len();
274	                    if !sid.is_empty() { log_group_idx.insert(sid, gi); }
275	                    groups.push(Group {
276	                        title: hit.title,
277	                        dir: hit.dir,
278	                        branch: hit.branch,
279	                        agent: hit.agent,
280	                        backed_up: hit.backed_up,
281	                        updated_at: hit.updated_at,
282	                        session_id: hit.session_id,
283	                        repo_dir: None,
284	                        kind: HitKind::Log,
285	                        rows: vec![GroupRow {
286	                            hit_ts: hit.hit_ts,
287	                            start_ts: hit.start_ts,
288	                            lines: hit.excerpt_lines,
289	                        }],
290	                    });
291	                }
292	            }
293	            HitKind::Session => {
294	                groups.push(Group {
295	                    title: hit.title,
296	                    dir: hit.dir,
297	                    branch: String::new(),
298	                    agent: hit.agent,
299	                    backed_up: hit.backed_up,
300	                    updated_at: hit.updated_at,
301	                    session_id: hit.session_id,
302	                    repo_dir: None,
303	                    kind: HitKind::Session,
304	                    rows: vec![],
305	                });
306	            }
307	            HitKind::Repo => {
308	                groups.push(Group {
309	                    title: hit.title,
310	                    dir: hit.dir,
311	                    branch: String::new(),
312	                    agent: String::new(),
313	                    backed_up: false,
314	                    updated_at: String::new(),
315	                    session_id: None,
316	                    repo_dir: hit.repo_dir,
317	                    kind: HitKind::Repo,
318	                    rows: vec![],
319	                });
320	            }
321	        }
322	    }
323	
324	    groups
325	}

327	// ── TUI ───────────────────────────────────────────────────────────────────────
328	
329	fn tui_loop(stdout: &mut impl Write, groups: &[Group], query: &str, ms: u128) -> Result<()> {
330	    let mut sel    = 0usize;
331	    let mut scroll = 0usize;
332	
333	    loop {
334	        let (w, h) = terminal::size().unwrap_or((120, 40));
335	        let w = w as usize;
336	        let h = h as usize;
337	        let content_h = h.saturating_sub(2);
338	
339	        let sel_start = group_start_of(groups, sel);
340	        let sel_end   = sel_start + groups.get(sel).map_or(1, rows_for_group);
341	        if sel_start < scroll                { scroll = sel_start; }
342	        else if sel_end > scroll + content_h { scroll = sel_end.saturating_sub(content_h); }
343	
344	        draw(stdout, groups, query, ms, sel, scroll, w, h)?;
345	
346	        match event::read()? {
347	            Event::Key(k) => match k.code {
348	                KeyCode::Char('q') | KeyCode::Esc => break,
349	                KeyCode::Char('c') if k.modifiers.contains(crossterm::event::KeyModifiers::CONTROL) => {
350	                    execute!(stdout, LeaveAlternateScreen, cursor::Show).ok();
351	                    terminal::disable_raw_mode().ok();
352	                    std::process::exit(0);
353	                }
354	                KeyCode::Up   | KeyCode::Char('k') => { if sel > 0 { sel -= 1; } }
355	                KeyCode::Down | KeyCode::Char('j') => { if sel + 1 < groups.len() { sel += 1; } }
356	                KeyCode::Char('g') => { sel = 0; scroll = 0; }
357	                KeyCode::Char('G') => { sel = groups.len().saturating_sub(1); }
358	                KeyCode::Char(' ') | KeyCode::Right | KeyCode::Enter => {
359	                    if let Some(group) = groups.get(sel) {
360	                        execute!(stdout, LeaveAlternateScreen, cursor::Show).ok();
361	                        terminal::disable_raw_mode().ok();
362	
363	                        match group.kind {
364	                            HitKind::Log | HitKind::Session => {
365	                                if let Some(id) = &group.session_id {
366	                                    let ts = group.rows.first().and_then(|r| r.start_ts.as_deref());
367	                                    let _ = super::show::run_at(id, ts);
368	                                }
369	                            }
370	                            HitKind::Repo => {
371	                                if let Some(dir) = &group.repo_dir {
372	                                    let _ = super::status::run_for_dir(dir);
373	                                }
374	                            }
375	                        }
376	
377	                        terminal::enable_raw_mode().ok();
378	                        execute!(stdout, EnterAlternateScreen, cursor::Hide).ok();
379	                        execute!(stdout, terminal::Clear(ClearType::All)).ok();
380	                    }
381	                }
382	                _ => {}
383	            },
384	            Event::Resize(_, _) => {
385	                execute!(stdout, terminal::Clear(ClearType::All)).ok();
386	            }
387	            _ => {}
388	        }
389	    }
390	
391	    Ok(())
392	}

394	fn rows_for_group(g: &Group) -> usize {
395	    1 + g.rows.iter().map(|r| r.lines.len().max(1)).sum::<usize>() + 1 // +1 for blank separator
396	}
397	
398	fn group_start_of(groups: &[Group], idx: usize) -> usize {
399	    groups[..idx.min(groups.len())].iter().map(rows_for_group).sum()
400	}

402	// ── Rendering ─────────────────────────────────────────────────────────────────
403	
404	fn draw(
405	    stdout: &mut impl Write,
406	    groups: &[Group],
407	    query:  &str,
408	    ms:     u128,
409	    sel:    usize,
410	    scroll: usize,
411	    w:      usize,
412	    h:      usize,
413	) -> io::Result<()> {
414	    use crossterm::queue;
415	    const SEL_BG: &str  = "48;5;236";
416	    const TS_W:   usize = 12; // fixed width of the timestamp column in excerpt rows
417	
418	    let content_h = h.saturating_sub(2);
419	    let mut buf: Vec<u8> = Vec::with_capacity((w + 60) * (h + 2));
420	
421	    // Title bar
422	    queue!(buf, cursor::MoveTo(0, 0), terminal::Clear(ClearType::UntilNewLine))?;
423	    let hdr = format!("  search: \"{}\"  {} result(s)  {}ms", query, groups.len(), ms);
424	    write!(buf, "\x1b[1m{}\x1b[0m", hdr.chars().take(w).collect::<String>())?;
425	
426	    // Pre-compute column widths across all groups for tabular alignment.
427	    let name_w   = groups.iter().map(|g| g.title.trim().chars().count()).max().unwrap_or(0).min(45);
428	    let agent_w  = groups.iter().map(|g| g.agent.chars().count()).max().unwrap_or(0);
429	    let branch_w = groups.iter().map(|g| g.branch.chars().count()).max().unwrap_or(0).min(30);
430	
431	    let mut screen_row = 1usize; // next terminal row to write (row 0 is title bar)
432	    let mut abs_row    = 0usize; // absolute content row (before scroll is applied)
433	
434	    for (gi, group) in groups.iter().enumerate() {
435	        if screen_row > content_h { break; }
436	        let selected = gi == sel;
437	
438	        // ── Session header ────────────────────────────────────────────────
439	        if abs_row >= scroll && screen_row <= content_h {
440	            let age     = age_secs_hit(&group.updated_at);
441	            let dot_col = match age {
442	                a if a < 900   => "38;5;82",
443	                a if a < 3_600 => "38;5;214",
444	                _              => "38;5;240",
445	            };
446	            let (name_col, meta_col, dot_char) = if group.backed_up {
447	                ("1;38;5;229", "38;5;240", "*")
448	            } else {
449	                ("38;5;242", "38;5;238", "·")
450	            };
451	
452	            let name: String = group.title.trim().chars().take(name_w).collect();
453	            let name_padded  = format!("{name:<name_w$}");
454	            let mut line = format!(
455	                "\x1b[{dot_col}m{dot_char}\x1b[0m \x1b[{name_col}m{name_padded}\x1b[0m  \x1b[38;5;240m{}\x1b[0m",
456	                group.dir,
457	            );
458	
459	            if branch_w > 0 {
460	                let branch_col = if group.backed_up { "38;5;75" } else { "38;5;239" };
461	                let b: String = group.branch.chars().take(branch_w).collect();
462	                let pad = " ".repeat(branch_w - b.chars().count());
463	                line.push_str(&format!("  \x1b[{branch_col}m{b}{pad}\x1b[0m"));
464	            }
465	
466	            if agent_w > 0 {
467	                let col = if group.backed_up { agent_color(&group.agent) } else { 239 };
468	                let a: String = group.agent.chars().take(agent_w).collect();
469	                let pad = " ".repeat(agent_w - a.chars().count());
470	                line.push_str(&format!("  \x1b[38;5;{col}m{a}{pad}\x1b[0m"));
471	            }
472	
473	            if let Some(sid) = &group.session_id {
474	                let id_short: String = sid.chars().take(8).collect();
475	                let ts = rel_time_hit(&group.updated_at);
476	                line.push_str(&format!("  \x1b[{meta_col}m{id_short}  {ts}\x1b[0m"));
477	            } else if matches!(group.kind, HitKind::Repo) {
478	                line.push_str(&format!("  \x1b[{meta_col}mrepo\x1b[0m"));
479	            }
480	
481	            render_row(&mut buf, &line, selected, SEL_BG, screen_row, w)?;
482	            screen_row += 1;
483	        }
484	        abs_row += 1;
485	
486	        // ── Excerpt rows ──────────────────────────────────────────────────
487	        for row in &group.rows {
488	            let ts_str = row.hit_ts.as_deref().map(rel_time_hit).unwrap_or_default();
489	            let n_lines = row.lines.len().max(1);
490	            let excerpt_indent = " ".repeat(2 + TS_W + 2);
491	
492	            for li in 0..n_lines {
493	                if abs_row >= scroll && screen_row <= content_h {
494	                    let text = row.lines.get(li).map(String::as_str).unwrap_or("");
495	                    let avail = w.saturating_sub(2 + TS_W + 2);
496	                    let text_t: String = text.chars().take(avail).collect();
497	
498	                    let exc_line = if li == 0 {
499	                        let ts_padded = format!("{ts_str:>TS_W$}");
500	                        format!("  \x1b[38;5;241m{ts_padded}\x1b[0m  \x1b[38;5;245m{text_t}\x1b[0m")
501	                    } else {
502	                        format!("{excerpt_indent}\x1b[38;5;245m{text_t}\x1b[0m")
503	                    };
504	
505	                    render_row(&mut buf, &exc_line, selected, SEL_BG, screen_row, w)?;
506	                    screen_row += 1;
507	                }
508	                abs_row += 1;
509	            }
510	        }
511	
512	        // Blank separator row between groups
513	        if abs_row >= scroll && screen_row <= content_h {
514	            queue!(buf, cursor::MoveTo(0, screen_row as u16), terminal::Clear(ClearType::UntilNewLine))?;
515	            screen_row += 1;
516	        }
517	        abs_row += 1;
518	    }
519	
520	    // Clear any leftover rows below the results.
521	    while screen_row <= content_h {
522	        queue!(buf, cursor::MoveTo(0, screen_row as u16), terminal::Clear(ClearType::UntilNewLine))?;
523	        screen_row += 1;
524	    }
525	
526	    // Status bar
527	    let bar = "  ↑↓/jk: navigate   space/→: open   q/esc: quit  ";
528	    let padded = format!("{:<width$}", bar.chars().take(w).collect::<String>(), width = w);
529	    queue!(buf, cursor::MoveTo(0, (h - 1) as u16))?;
530	    write!(buf, "\x1b[7m{padded}\x1b[0m")?;
531	
532	    stdout.write_all(&buf)?;
533	    stdout.flush()
534	}

536	fn render_row(buf: &mut Vec<u8>, line: &str, selected: bool, bg: &str, row: usize, w: usize) -> io::Result<()> {
537	    use crossterm::queue;
538	    queue!(buf, cursor::MoveTo(0, row as u16), terminal::Clear(ClearType::UntilNewLine))?;
539	    if selected {
540	        let colored = with_bg(line, bg);
541	        let vis = visible_len(line);
542	        let pad = w.saturating_sub(vis);
543	        write!(buf, "\x1b[{bg}m{colored}{}\x1b[0m", " ".repeat(pad))?;
544	    } else {
545	        write!(buf, "{line}")?;
546	    }
547	    Ok(())
548	}

550	// ── DB enrichment ─────────────────────────────────────────────────────────────
551	
552	fn enrich_hits(hits: &mut Vec<SearchHit>) {
553	    let Ok(conn) = crate::db::connect() else { return };
554	    for hit in hits.iter_mut() {
555	        let Some(sid) = &hit.session_id else { continue };
556	        if let Ok((agent, updated_at)) = conn.query_row(
557	            "SELECT agent_name, updated_at FROM sessions WHERE session_id = ?1",
558	            [sid.as_str()],
559	            |row| Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?)),
560	        ) {
561	            hit.agent      = agent;

Two root causes:

**Hang (60+ seconds):** `gossamer clean` calls `Command::new("entire").arg("clean").arg(session_id).status()` and blocks waiting for it to finish. If you have `entire` installed, the real binary runs — possibly making network calls or doing git work before exiting. The `clean` test was stuck here.

**Garbled output:** `entire` can print ANSI-colored progress output to its stdout, which gossamer inherits (since it just calls `.status()` without redirecting the subprocess's I/O). That output reaches the terminal directly, corrupting what cargo test is showing.

**Fix:** `TestEnv` now creates a temp directory containing a one-line `entire` stub (`#!/bin/sh\nexit 0`) and prepends it to `PATH` for every gossamer subprocess. The real `entire` is never reached. Same fix covers any other gossamer command that shells out to `entire` (`init` wizard, `attach`).
