8	def get_angular_radius(z: float, impact_kpc: float) -> u.Quantity:
9	    """Calculate angular radius for a given physical impact parameter at redshift z."""
10	    if z <= 0:
11	        return 0.0 * u.arcmin
12	    
13	    d_a = COSMO.angular_diameter_distance(z)
14	    with u.set_enabled_equivalencies(u.dimensionless_angles()):
15	        theta_rad = (impact_kpc * u.kpc / d_a).to(u.rad)
16	    return theta_rad.to(u.arcmin)
17	
18	def calculate_impact_parameter(
19	    ra_gal: float, 
20	    dec_gal: float, 
21	    z_gal: float, 
22	    ra_sight: float, 
23	    dec_sight: float
24	) -> float:
25	    """Calculate physical impact parameter in kpc."""
26	    if z_gal <= 0:
27	        return np.nan
28	        
29	    c_sight = SkyCoord(ra_sight, dec_sight, unit='deg', frame='icrs')
30	    c_gal = SkyCoord(ra_gal, dec_gal, unit='deg', frame='icrs')
31	    sep = c_sight.separation(c_gal)
32	    
33	    d_a = COSMO.angular_diameter_distance(z_gal)
34	    impact_kpc = (sep.radian * d_a).to(u.kpc).value
35	    return impact_kpc

97	def query_ps1_gi_mags(coord: SkyCoord, match_radius: u.Quantity = PS1_MATCH_RADIUS):
98	    """Nearest Pan-STARRS1 (II/349/ps1) source to ``coord`` within ``match_radius``.
99	
100	    Returns ``(g_mag, i_mag, sep_arcsec)``. Kron magnitudes are preferred over
101	    mean-PSF mags: Taylor+2011 (the g-i/M_i estimator these feed) was calibrated
102	    on Kron-like GAMA total magnitudes, and Kron better captures extended-source
103	    flux. PS1 AB mags are used directly; the small PS1->SDSS color terms are
104	    neglected at the order-of-magnitude precision of these halo-mass estimates.
105	
106	    The DESI VII/292 photo-z catalog carries no photometry, so this PS1 match is
107	    the only route to a measured stellar mass for DESI-only sightlines. Faint,
108	    high-z galaxies fall below the shallow PS1 3pi depth and simply will not
109	    match; callers then fall back to an assumed mass. Either of ``g_mag`` /
110	    ``i_mag`` may be ``None`` if not finite, and the result is
111	    ``(None, None, None)`` on no match or query failure.
112	    """
113	    try:
114	        v = Vizier(columns=["RAJ2000", "DEJ2000", "gmag", "imag", "gKmag", "iKmag"], row_limit=50)
115	        res = v.query_region(coord, radius=match_radius, catalog=PS1_CATALOG)
116	    except Exception as e:
117	        print(f"PS1 query failed: {e}")
118	        return None, None, None
119	
120	    if not res or len(res[0]) == 0:
121	        return None, None, None
122	
123	    t = res[0]
124	    matches = SkyCoord(t["RAJ2000"], t["DEJ2000"], unit="deg")
125	    seps = coord.separation(matches).arcsec
126	    j = int(np.argmin(seps))
127	
128	    def _pick(kron_col: str, psf_col: str):
129	        for col in (kron_col, psf_col):
130	            if col in t.colnames:
131	                val = t[col][j]
132	                if val is not None and np.isfinite(val):
133	                    return float(val)
134	        return None
135	
136	    return _pick("gKmag", "gmag"), _pick("iKmag", "imag"), float(seps[j])
137

175	    z_col = _find_column(df, ("flux_z", "fluxz", "zflux", "z_flux", "flx_z"))

180	    r_flux = pd.to_numeric(df[r_col], errors="coerce")
181	    z_flux = pd.to_numeric(df[z_col], errors="coerce")
182	    redshift = pd.to_numeric(df["z"], errors="coerce")
183	    valid = (g_flux > 0.0) & (r_flux > 0.0) & (z_flux > 0.0) & (redshift > 0.0)
184	    if not valid.any():
185	        return df
186	
187	    g_mag = 22.5 - 2.5 * np.log10(g_flux[valid])
188	    r_mag = 22.5 - 2.5 * np.log10(r_flux[valid])
189	    d_l_pc = COSMO.luminosity_distance(redshift[valid].to_numpy()).to(u.pc).value
190	    distance_modulus = 5.0 * np.log10(d_l_pc / 10.0)
191	    absolute_r = r_mag - distance_modulus
192	    log_mass = -0.68 + 0.70 * (g_mag - r_mag) - 0.4 * absolute_r
193	
194	    df.loc[valid, "M_star"] = log_mass
