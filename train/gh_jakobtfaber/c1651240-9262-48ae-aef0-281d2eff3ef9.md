742	        g_f (see gain_spectrum) is the scintillation/Delta-nu_d probe. A purely
743	        temporal (shape) misfit -- e.g. a model that cannot reach the burst peak --
744	        is NOT absorbed by g_f and remains visible.
745	        """
746	        if self.data is None or self.noise_std is None:
747	            raise RuntimeError("need observed data + noise_std for likelihood")
748	        if self.valid is None or not np.any(self.valid):
749	            return -np.inf
750	
751	        K = self(replace(p, c0=1.0, gamma=0.0), model, freq_subset=self.valid)
752	        d = self.data[self.valid]
753	        sig = np.clip(self.noise_std[self.valid], 1e-9, None)
754	        var = sig**2
755	        S_dd = np.einsum("ij,ij->i", d, d)
756	        S_dk = np.einsum("ij,ij->i", d, K)
757	        S_kk = np.einsum("ij,ij->i", K, K)
758	        ok = S_kk > 1e-30
759	        S_kk_safe = np.where(ok, S_kk, 1.0)
760	        # gain=0 baseline (S_dd/var) where the model has no support in a channel
761	        chi2min = np.where(ok, (S_dd - S_dk**2 / S_kk_safe) / var, S_dd / var)
762	        occam = np.where(ok, -0.5 * np.log(S_kk_safe), 0.0)
763	        const = 0.5 * np.log(2.0 * np.pi * var)
764	        ll = float(np.sum(-0.5 * chi2min + occam + const))
765	        return ll if np.isfinite(ll) else -np.inf
766	
767	    def gain_spectrum(self, p: FRBParams, model: str = "M3") -> NDArray[np.floating]:
768	        """Profiled per-channel gain g_f = S_dk/S_kk at parameters *p* (all channels).
769	
770	        g_f is the matched-filter amplitude of the unit kernel K_f in each channel:
771	        the burst spectrum modulated by diffractive scintillation. Autocorrelating
772	        g_f over frequency yields the scintillation bandwidth Delta-nu_d (the second
773	        scattering probe). Channels with no model support return 0.
774	        """
775	        K = self(replace(p, c0=1.0, gamma=0.0), model)
776	        d = self.data
777	        S_dk = np.einsum("ij,ij->i", d, K)
778	        S_kk = np.einsum("ij,ij->i", K, K)
779	        return np.where(S_kk > 1e-30, S_dk / np.where(S_kk > 1e-30, S_kk, 1.0), 0.0)

\subsection{Isotropic-equivalent energies}
\label{sec:burst-energies}

For each sightline with a spectroscopic host redshift we estimate the
isotropic-equivalent burst energy directly from the joint CHIME--DSA fit,
without extrapolating either band's spectrum beyond where it is constrained.
The joint fit returns a per-band spectral amplitude and index,
$F_X(\nu) = c_{0,X}\,(\nu/\nu_{\mathrm{ref},X})^{\gamma_X}$ (in
$\mathrm{Jy\,ms}$), for $X\in\{\mathrm{CHIME},\mathrm{DSA}\}$. We integrate
each power law across its own observing band---CHIME over
$0.400$--$0.800\,\mathrm{GHz}$ and DSA over $1.311$--$1.499\,\mathrm{GHz}$---and
sum the two contributions,
%
\begin{equation}
E_{\mathrm{iso}} = 4\pi D_L^2(z)\left[
  \int_{\nu_1^{C}}^{\nu_2^{C}} F_{\mathrm{CHIME}}(\nu)\,d\nu

\right],
\label{eq:eiso}
\end{equation}
%
