        echo "${name}: no pid file"
        return 0
    fi

    local pid
    pid=$(cat "$pidfile")

    if ! kill -0 "$pid" 2>/dev/null; then
        echo "${name}: not running (stale pid ${pid})"
        rm -f "$pidfile"
        return 0
    fi

    echo "Stopping ${name} (pid ${pid})..."
    kill "$pid"

    local i=0
    while kill -0 "$pid" 2>/dev/null; do
        sleep 1
        i=$((i + 1))
        if [ "$i" -ge "$timeout" ]; then
            echo "  ${name} did not stop in ${timeout}s — sending SIGKILL"
            kill -9 "$pid" 2>/dev/null || true
            sleep 1
            break
        fi
    done

    rm -f "$pidfile"
    echo "  ${name} stopped"
}

## Root Cause

The new dockyard commit *"Add bundled sysbox service"* introduced `sc_sysbox.service`, which starts bundled sysbox binaries. Two issues cascaded:

1. **Flag renamed in sysbox 0.6.7**: `sysbox-fs` removed `--data-root` and replaced it with `--mountpoint`. Dockyard was still passing `--data-root`, causing `sysbox-fs` to crash immediately with `"flag provided but not defined: -data-root"`. The stale `/run/sysbox/sysfs.sock` from the previous system sysbox was left on disk (nothing listening), causing the "no such file or directory" error.

2. **Socket conflict with system sysbox**: The server had sysbox installed via apt (`sysbox-ce` 0.6.6). When the installer ran `dockyard create`, the bundled `sc_sysbox.service` and the system `sysbox-fs.service` both try to own `/run/sysbox/sysfs.sock`.

- Changed `sysbox-fs --data-root` → `sysbox-fs --mountpoint` in both the startup function and the systemd service template
