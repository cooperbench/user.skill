   593→# ═══ cmd_destroy ═════════════════════════════════════════════════════════════
   594→
   595→cmd_destroy() {
   596→  local auto_confirm="${1:-false}"
   597→
   598→  require_root
   599→
   600→  # Load installed env for CREATED_USER/CREATED_GROUP
   601→  if [ -f "$SANDCASTLE_HOME/etc/sandcastle.env" ]; then
   602→    # shellcheck source=/dev/null
   603→    source "$SANDCASTLE_HOME/etc/sandcastle.env"
   604→  fi
   605→
   606→  warn "This will remove Sandcastle from $SANDCASTLE_HOME (containers, config, services)"
   607→  if [ "$auto_confirm" != "true" ]; then
   608→    read -rp "Are you sure? (yes to confirm): " CONFIRM
   609→    [ "$CONFIRM" = "yes" ] || die "Aborted"
   610→  fi
   611→
   612→  info "Tearing down Sandcastle..."
   613→  cd /
   614→
   615→  if [ -x "$DOCKER" ]; then
   616→    for container in $($DOCKER ps -a --filter "name=^sc-ts-" --format '{{.Names}}' 2>/dev/null); do
   617→      info "Removing Tailscale sidecar: $container"
   618→      $DOCKER rm -f "$container" 2>/dev/null || true
   619→    done
   620→    for container in $($DOCKER ps -a --filter "runtime=sysbox-runc" --format '{{.Names}}' 2>/dev/null); do
   621→      info "Removing container: $container"
   622→      $DOCKER rm -f "$container" 2>/dev/null || true
   623→    done
   624→    for net in $($DOCKER network ls --filter "name=^sc-ts-net-" --format '{{.Name}}' 2>/dev/null); do
   625→      info "Removing network: $net"
   626→      $DOCKER network rm "$net" 2>/dev/null || true
   627→    done
   628→    if [ -f "$SANDCASTLE_HOME/docker-compose.yml" ]; then
   629→      $DOCKER compose -f "$SANDCASTLE_HOME/docker-compose.yml" --env-file "$SANDCASTLE_HOME/.env" down --rmi all --volumes --remove-orphans 2>/dev/null || true
   630→    fi
   631→    $DOCKER network rm sandcastle-web 2>/dev/null || true
   632→  fi
   633→
   634→  # Destroy Dockyard
   635→  DOCKYARD_ENV_FILE="$SANDCASTLE_HOME/etc/dockyard.env"
   636→  if [ -f "$DOCKYARD_ENV_FILE" ] || systemctl cat "${DOCKYARD_DOCKER_PREFIX}docker.service" &>/dev/null; then
   637→    info "Destroying Dockyard..."
   638→    if [ -f "$DOCKYARD_ENV_FILE" ] && wget -q "https://raw.githubusercontent.com/thieso2/dockyard/refs/heads/main/dockyard.sh" -O /tmp/dockyard.sh 2>/dev/null; then
   639→      echo "y" | DOCKYARD_ENV="$DOCKYARD_ENV_FILE" bash /tmp/dockyard.sh destroy >/dev/null 2>&1 || true
   640→      rm -f /tmp/dockyard.sh
   641→    else
   642→      systemctl stop "${DOCKYARD_DOCKER_PREFIX}docker" 2>/dev/null || true
   643→      systemctl disable "${DOCKYARD_DOCKER_PREFIX}docker" 2>/dev/null || true
   644→      rm -f "/etc/systemd/system/${DOCKYARD_DOCKER_PREFIX}docker.service"
   645→      systemctl daemon-reload 2>/dev/null || true
   646→      ip link delete "${DOCKYARD_DOCKER_PREFIX}docker0" 2>/dev/null || true
   647→    fi
   648→    ok "Dockyard destroyed"
   649→  fi
   650→
   651→  # Remove user/group if we created them
   652→  if [ "${CREATED_USER:-}" = "true" ] && id "$SANDCASTLE_USER" &>/dev/null; then
   653→    userdel "$SANDCASTLE_USER" 2>/dev/null || true
   654→    ok "Removed user '${SANDCASTLE_USER}'"
   655→  fi
   656→  if [ "${CREATED_GROUP:-}" = "true" ] && getent group "$SANDCASTLE_GROUP" &>/dev/null; then
   657→    groupdel "$SANDCASTLE_GROUP" 2>/dev/null || true
   658→    ok "Removed group '${SANDCASTLE_GROUP}'"
   659→  fi
   660→
   661→  # sandcastle-nat.service no longer used (dockyard handles NAT for user-defined networks)
   662→
   663→  # Remove sudoers file
   664→  if [ -f /etc/sudoers.d/sandcastle ]; then
   665→    rm -f /etc/sudoers.d/sandcastle
   666→    ok "Removed sudoers file"
   667→  fi
   668→
   669→  # Remove PATH export from .bashrc (try both old and new format for compatibility)
   670→  if [ -f "$SANDCASTLE_HOME/.bashrc" ]; then
   671→    local old_path_export="export PATH=${SANDCASTLE_HOME}/docker-runtime/bin/:\$PATH"
   672→    local new_path_export="export PATH=${SANDCASTLE_HOME}/docker-runtime/bin:\$PATH"
   673→    for path_export in "$old_path_export" "$new_path_export"; do
   674→      if grep -qF "$path_export" "$SANDCASTLE_HOME/.bashrc" 2>/dev/null; then
   675→        grep -vF "$path_export" "$SANDCASTLE_HOME/.bashrc" > "$SANDCASTLE_HOME/.bashrc.tmp"
   676→        mv "$SANDCASTLE_HOME/.bashrc.tmp" "$SANDCASTLE_HOME/.bashrc"
   677→        ok "Removed PATH export from .bashrc"
   678→      fi
   679→    done
   680→  fi
   681→
   682→  # Remove login banner
   683→  if [ -f "/etc/profile.d/sandcastle-banner.sh" ]; then
   684→    rm -f /etc/profile.d/sandcastle-banner.sh
   685→    ok "Removed login banner"
   686→  fi
   687→
   688→  # Remove Sandcastle files — keep user data (data/users, data/sandboxes, data/postgres)
   689→  rm -f "$SANDCASTLE_HOME/.env"
   690→  rm -f "$SANDCASTLE_HOME/docker-compose.yml"
   691→  rm -rf "$SANDCASTLE_HOME/etc"
   692→  rm -rf "$SANDCASTLE_HOME/data/traefik"
   693→  rmdir "$SANDCASTLE_HOME/data" 2>/dev/null || true
   694→  rmdir "$SANDCASTLE_HOME" 2>/dev/null || true
   695→  rm -rf "/run/${DOCKYARD_DOCKER_PREFIX}docker"
   696→
   697→  if [ -d "$SANDCASTLE_HOME/data/users" ] || [ -d "$SANDCASTLE_HOME/data/sandboxes" ] || [ -d "$SANDCASTLE_HOME/data/postgres" ]; then
   698→    warn "User data preserved in $SANDCASTLE_HOME/data/ — remove manually if no longer needed"
   699→  fi
   700→
   701→  ok "Sandcastle destroyed"
   702→}

   704→# ═══ cmd_install ═════════════════════════════════════════════════════════════
   705→
   706→cmd_install() {
   707→  require_root
   708→
   709→  if ! grep -qi ubuntu /etc/os-release 2>/dev/null; then
   710→    warn "This script is tested on Ubuntu 24.04. Other distros may work but are unsupported."
   711→  fi
   712→  if [ "$ARCH" != "amd64" ] && [ "$ARCH" != "arm64" ]; then
   713→    die "Sandcastle requires amd64 or arm64 architecture (got: $ARCH)"
   714→  fi
   715→
   716→  echo ""
   717→  echo -e "${BLUE}═══ Sandcastle Installer ═══${NC}"
   718→  echo ""
   719→  info "Available images (${ARCH}):"
   720→  show_image_info "sandcastle"
   721→  show_image_info "sandcastle-sandbox"
   722→  echo ""
   723→
   724→  # ── Install Dockyard ──────────────────────────────────────────────────────
   725→
   726→  if [ -S "$DOCKER_SOCK" ]; then
   727→    ok "Dockyard already installed"
   728→  else
   729→    info "Installing Dockyard (Docker + Sysbox)..."
   730→    wget -q "https://raw.githubusercontent.com/thieso2/dockyard/refs/heads/main/dockyard.sh" -O /tmp/dockyard.sh
   731→
   732→    mkdir -p "$SANDCASTLE_HOME/etc"
   733→    cat > "$SANDCASTLE_HOME/etc/dockyard.env" <<DYEOF
   734→DOCKYARD_ROOT=${DOCKYARD_ROOT}
   735→DOCKYARD_DOCKER_PREFIX=${DOCKYARD_DOCKER_PREFIX}
   736→DOCKYARD_BRIDGE_CIDR=${DOCKYARD_BRIDGE_CIDR}
   737→DOCKYARD_FIXED_CIDR=${DOCKYARD_FIXED_CIDR}
   738→DOCKYARD_POOL_BASE=${DOCKYARD_POOL_BASE}
   739→DOCKYARD_POOL_SIZE=${DOCKYARD_POOL_SIZE}
   740→DYEOF
   741→    wrote "$SANDCASTLE_HOME/etc/dockyard.env"
   742→
   743→    DOCKYARD_ENV="$SANDCASTLE_HOME/etc/dockyard.env" bash /tmp/dockyard.sh create
   744→    rm -f /tmp/dockyard.sh
   745→
   746→    for i in $(seq 1 30); do
   747→      [ -S "$DOCKER_SOCK" ] && break
   748→      sleep 1
   749→    done
   750→    [ -S "$DOCKER_SOCK" ] || die "Dockyard socket not found at $DOCKER_SOCK after 30s"
   751→    ok "Dockyard installed"
   752→  fi
   753→
   754→  # ── Configure UFW ─────────────────────────────────────────────────────────
   755→
   756→  if command -v ufw &>/dev/null; then
   757→    info "Configuring firewall..."
   758→    ufw --force reset >/dev/null 2>&1
   759→    ufw default deny incoming >/dev/null 2>&1
   760→    ufw default allow outgoing >/dev/null 2>&1
   761→    ufw allow 22/tcp >/dev/null 2>&1
   762→    ufw allow "${SANDCASTLE_HTTP_PORT}/tcp" >/dev/null 2>&1
   763→    ufw allow "${SANDCASTLE_HTTPS_PORT}/tcp" >/dev/null 2>&1
   764→    ufw allow 2201:2299/tcp >/dev/null 2>&1
   765→    ufw --force enable >/dev/null 2>&1
   766→    ok "Firewall configured (22, ${SANDCASTLE_HTTP_PORT}, ${SANDCASTLE_HTTPS_PORT}, 2201-2299)"
   767→  else
   768→    warn "UFW not found — skipping firewall setup"
   769→  fi
   770→
   771→  # ── Create system user/group ──────────────────────────────────────────────
   772→
   773→  CREATED_GROUP=false
   774→  CREATED_USER=false
   775→
   776→  if getent group "$SANDCASTLE_GROUP" &>/dev/null; then
   777→    ok "Group '${SANDCASTLE_GROUP}' already exists"
   778→  else
   779→    groupadd --system --gid "$SANDCASTLE_GID" "$SANDCASTLE_GROUP"
   780→    CREATED_GROUP=true
   781→    ok "Created group '${SANDCASTLE_GROUP}' (GID ${SANDCASTLE_GID})"
   782→  fi
   783→
   784→  if id "$SANDCASTLE_USER" &>/dev/null; then
   785→    ok "User '${SANDCASTLE_USER}' already exists"
   786→  else
   787→    useradd --system --uid "$SANDCASTLE_UID" --gid "$SANDCASTLE_GID" \
   788→      --home-dir "$SANDCASTLE_HOME" --shell /bin/bash "$SANDCASTLE_USER"
   789→    CREATED_USER=true
   790→    ok "Created user '${SANDCASTLE_USER}' (UID ${SANDCASTLE_UID})"
   791→  fi
   792→
   793→  # ── Create directories ────────────────────────────────────────────────────
   794→
   795→  ensure_dirs
   796→
   797→  # ── SSH & sudo setup ──────────────────────────────────────────────────────
   798→
   799→  setup_ssh_keys
   800→  setup_passwordless_sudo
   801→  setup_bashrc_path
   802→  setup_login_banner
   803→
   804→  # ── Detect fresh install vs upgrade ─────────────────────────────────────
   805→
   806→  FRESH_INSTALL=false
   807→  [ ! -f "$SANDCASTLE_HOME/.env" ] && FRESH_INSTALL=true
   808→
   809→  # ── Fresh install: validate & generate .env ─────────────────────────────
   810→
   811→  if [ "$FRESH_INSTALL" = true ]; then
   812→    [ -z "${SANDCASTLE_HOST:-}" ] && die "SANDCASTLE_HOST is required (set in sandcastle.env)"
   813→    [ -z "${SANDCASTLE_ADMIN_EMAIL:-}" ] && die "SANDCASTLE_ADMIN_EMAIL is required (set in sandcastle.env)"
   814→    [ -z "${SANDCASTLE_ADMIN_PASSWORD:-}" ] && die "SANDCASTLE_ADMIN_PASSWORD is required (set in sandcastle.env or use SANDCASTLE_ADMIN_PASSWORD_FILE)"
   815→    [ ${#SANDCASTLE_ADMIN_PASSWORD} -lt 6 ] && die "SANDCASTLE_ADMIN_PASSWORD must be at least 6 characters"
   816→
   817→    # SANDCASTLE_SUBNET no longer needed - networks allocated from dockyard pool
   818→
   819→    SECRET_KEY_BASE=$(openssl rand -hex 64)
   820→    DOCKER_GID=$(stat -c '%g' "$DOCKER_SOCK" 2>/dev/null || echo "988")
   821→
   822→    # Preserve DB password across reinstalls (user data)
   823→    POSTGRES_SECRETS_FILE="$SANDCASTLE_HOME/data/postgres/.secrets"
   824→    if [ -f "$POSTGRES_SECRETS_FILE" ]; then
   825→      # shellcheck source=/dev/null
   826→      source "$POSTGRES_SECRETS_FILE"
   827→      info "Reusing existing database password"
   828→    else
   829→      DB_PASSWORD=$(openssl rand -hex 32)
   830→      mkdir -p "$SANDCASTLE_HOME/data/postgres"
   831→      echo "DB_PASSWORD=$DB_PASSWORD" > "$POSTGRES_SECRETS_FILE"
   832→      chmod 600 "$POSTGRES_SECRETS_FILE"
   833→      wrote "$POSTGRES_SECRETS_FILE"

   839→SANDCASTLE_HOST="${SANDCASTLE_HOST}"
   840→SANDCASTLE_TLS_MODE="${SANDCASTLE_TLS_MODE}"
   841→SANDCASTLE_USER="${SANDCASTLE_USER}"
   842→SANDCASTLE_GROUP="${SANDCASTLE_GROUP}"
   843→SANDCASTLE_UID="${SANDCASTLE_UID}"
   844→SANDCASTLE_GID="${SANDCASTLE_GID}"
   845→SECRET_KEY_BASE="${SECRET_KEY_BASE}"
   846→DB_PASSWORD="${DB_PASSWORD}"
   847→SANDCASTLE_ADMIN_USER="${SANDCASTLE_ADMIN_USER}"
   848→SANDCASTLE_ADMIN_EMAIL="${SANDCASTLE_ADMIN_EMAIL}"
   849→SANDCASTLE_ADMIN_PASSWORD="${SANDCASTLE_ADMIN_PASSWORD}"
   850→SANDCASTLE_ADMIN_SSH_KEY="${SANDCASTLE_ADMIN_SSH_KEY:-}"
   851→DOCKER_GID="${DOCKER_GID}"
   852→DOCKER_SOCK="${DOCKER_SOCK}"
   853→ACME_EMAIL="${ACME_EMAIL:-}"
   854→GITHUB_CLIENT_ID="${GITHUB_CLIENT_ID:-}"
   855→GITHUB_CLIENT_SECRET="${GITHUB_CLIENT_SECRET:-}"
   856→GOOGLE_CLIENT_ID="${GOOGLE_CLIENT_ID:-}"
   857→GOOGLE_CLIENT_SECRET="${GOOGLE_CLIENT_SECRET:-}"
   858→EOF
   859→    chmod 600 "$SANDCASTLE_HOME/.env"
   860→    wrote "$SANDCASTLE_HOME/.env"
   861→  else
   862→    info "Existing install — loading $SANDCASTLE_HOME/.env"
   863→  fi
   864→
   865→  # shellcheck source=/dev/null
   866→  source "$SANDCASTLE_HOME/.env"

   872→  fi

   874→    DOCKER_SOCK="${DOCKYARD_ROOT}/docker.sock"
   875→    echo "DOCKER_SOCK=$DOCKER_SOCK" >> "$SANDCASTLE_HOME/.env"
   876→  fi
   877→  # Backfill OAuth vars from sandcastle.env into .env
   878→  grep -q '^GITHUB_CLIENT_ID=' "$SANDCASTLE_HOME/.env" 2>/dev/null || \
   879→    echo "GITHUB_CLIENT_ID=${GITHUB_CLIENT_ID:-}" >> "$SANDCASTLE_HOME/.env"

   881→    echo "GITHUB_CLIENT_SECRET=${GITHUB_CLIENT_SECRET:-}" >> "$SANDCASTLE_HOME/.env"
   882→  grep -q '^GOOGLE_CLIENT_ID=' "$SANDCASTLE_HOME/.env" 2>/dev/null || \
   883→    echo "GOOGLE_CLIENT_ID=${GOOGLE_CLIENT_ID:-}" >> "$SANDCASTLE_HOME/.env"
   884→  grep -q '^GOOGLE_CLIENT_SECRET=' "$SANDCASTLE_HOME/.env" 2>/dev/null || \
   885→    echo "GOOGLE_CLIENT_SECRET=${GOOGLE_CLIENT_SECRET:-}" >> "$SANDCASTLE_HOME/.env"
   886→
   887→  # ── Write installed sandcastle.env ────────────────────────────────────────

   894→SANDCASTLE_USER="${SANDCASTLE_USER}"
   895→SANDCASTLE_GROUP="${SANDCASTLE_GROUP}"
