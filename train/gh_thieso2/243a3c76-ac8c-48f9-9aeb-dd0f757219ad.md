   418→cmd_destroy() {
   419→  local auto_confirm="${1:-false}"
   420→
   421→  require_root
   422→
   423→  # Load installed env for CREATED_USER/CREATED_GROUP
   424→  if [ -f "$SANDCASTLE_HOME/etc/sandcastle.env" ]; then
   425→    # shellcheck source=/dev/null
   426→    source "$SANDCASTLE_HOME/etc/sandcastle.env"
   427→  fi
   428→
   429→  warn "This will remove Sandcastle from $SANDCASTLE_HOME (containers, config, services)"
   430→  if [ "$auto_confirm" != "true" ]; then
   431→    read -rp "Are you sure? (yes to confirm): " CONFIRM
   432→    [ "$CONFIRM" = "yes" ] || die "Aborted"
   433→  fi
   434→
   435→  info "Tearing down Sandcastle..."
   436→  cd /
   437→
   438→  if [ -x "$DOCKER" ]; then
   439→    for container in $($DOCKER ps -a --filter "name=^sc-ts-" --format '{{.Names}}' 2>/dev/null); do
   440→      info "Removing Tailscale sidecar: $container"
   441→      $DOCKER rm -f "$container" 2>/dev/null || true
   442→    done
   443→    for container in $($DOCKER ps -a --filter "runtime=sysbox-runc" --format '{{.Names}}' 2>/dev/null); do
   444→      info "Removing container: $container"
   445→      $DOCKER rm -f "$container" 2>/dev/null || true
   446→    done
   447→    for net in $($DOCKER network ls --filter "name=^sc-ts-net-" --format '{{.Name}}' 2>/dev/null); do
   448→      info "Removing network: $net"
   449→      $DOCKER network rm "$net" 2>/dev/null || true
   450→    done
   451→    if [ -f "$SANDCASTLE_HOME/docker-compose.yml" ]; then
   452→      $DOCKER compose -f "$SANDCASTLE_HOME/docker-compose.yml" --env-file "$SANDCASTLE_HOME/.env" down --rmi all --volumes --remove-orphans 2>/dev/null || true
   453→    fi
   454→    $DOCKER network rm sandcastle-web 2>/dev/null || true
   455→  fi
   456→
   457→  # Destroy Dockyard
   458→  DOCKYARD_ENV_FILE="$SANDCASTLE_HOME/etc/dockyard.env"
   459→  if [ -f "$DOCKYARD_ENV_FILE" ] || systemctl cat "${DOCKYARD_DOCKER_PREFIX}docker.service" &>/dev/null; then
   460→    info "Destroying Dockyard..."
   461→    if [ -f "$DOCKYARD_ENV_FILE" ] && wget -q "https://raw.githubusercontent.com/thieso2/dockyard/refs/heads/main/dockyard.sh" -O /tmp/dockyard.sh 2>/dev/null; then
   462→      echo "y" | DOCKYARD_ENV="$DOCKYARD_ENV_FILE" bash /tmp/dockyard.sh destroy >/dev/null 2>&1 || true
   463→      rm -f /tmp/dockyard.sh
   464→    else
   465→      systemctl stop "${DOCKYARD_DOCKER_PREFIX}docker" 2>/dev/null || true
   466→      systemctl disable "${DOCKYARD_DOCKER_PREFIX}docker" 2>/dev/null || true
   467→      rm -f "/etc/systemd/system/${DOCKYARD_DOCKER_PREFIX}docker.service"
   468→      systemctl daemon-reload 2>/dev/null || true
   469→      ip link delete "${DOCKYARD_DOCKER_PREFIX}docker0" 2>/dev/null || true
   470→    fi
   471→    ok "Dockyard destroyed"
   472→  fi
   473→
   474→  # Remove user/group if we created them
   475→  if [ "${CREATED_USER:-}" = "true" ] && id "$SANDCASTLE_USER" &>/dev/null; then
   476→    userdel "$SANDCASTLE_USER" 2>/dev/null || true
   477→    ok "Removed user '${SANDCASTLE_USER}'"
   478→  fi
   479→  if [ "${CREATED_GROUP:-}" = "true" ] && getent group "$SANDCASTLE_GROUP" &>/dev/null; then
   480→    groupdel "$SANDCASTLE_GROUP" 2>/dev/null || true
   481→    ok "Removed group '${SANDCASTLE_GROUP}'"
   482→  fi
   483→
   484→  # Remove NAT service
   485→  if [ -f /etc/systemd/system/sandcastle-nat.service ]; then
   486→    systemctl disable --now sandcastle-nat 2>/dev/null || true
   487→    rm -f /etc/systemd/system/sandcastle-nat.service
   488→    systemctl daemon-reload 2>/dev/null || true
   489→  fi
   490→
   491→  # Remove Sandcastle files — keep user data (data/users, data/sandboxes)
   492→  rm -f "$SANDCASTLE_HOME/.env"
   493→  rm -f "$SANDCASTLE_HOME/docker-compose.yml"
   494→  rm -rf "$SANDCASTLE_HOME/etc"
   495→  rm -rf "$SANDCASTLE_HOME/data/traefik"
   496→  rmdir "$SANDCASTLE_HOME/data" 2>/dev/null || true
   497→  rmdir "$SANDCASTLE_HOME" 2>/dev/null || true
   498→  rm -rf "/run/${DOCKYARD_DOCKER_PREFIX}docker"
   499→
   500→  if [ -d "$SANDCASTLE_HOME/data/users" ] || [ -d "$SANDCASTLE_HOME/data/sandboxes" ]; then
   501→    warn "User data preserved in $SANDCASTLE_HOME/data/ — remove manually if no longer needed"
   502→  fi
   503→
   504→  ok "Sandcastle destroyed"
   505→}

Here's the root cause:

**The Rails `web` container mounts `${DATA_MOUNT}:/data`** (host `/sandcastle/data` → container `/data`), and `SANDCASTLE_DATA_DIR=/data`. The services then use `/data/users/...`, `/data/sandboxes/...` etc. for **two purposes**:

2. **Docker bind mount paths** when creating sandbox/WeTTY/Tailscale containers — **broken**, because Docker interprets these as HOST paths, creating `/data/users/...` on the host instead of using `/sandcastle/data/users/...`

The fix is to mount the data volume at the same path as on the host, so paths are consistent between the Rails container and Docker daemon.

In `write_compose()`, the change is:
- Volume: `${DATA_MOUNT}:/data` → `${DATA_MOUNT}:${DATA_MOUNT}`
- Env: `SANDCASTLE_DATA_DIR: /data` → `SANDCASTLE_DATA_DIR: ${DATA_MOUNT}`

Fix looks correct. Here's the summary:

**Root cause:** `write_compose()` in `installer.sh` mounted the data volume as `${DATA_MOUNT}:/data` (line 283), mapping host `/sandcastle/data` to container `/data`. It set `SANDCASTLE_DATA_DIR: /data` inside the container. When the Rails app created sandbox containers via the Docker socket, it used paths like `/data/users/{name}/home` as bind mount sources. Docker interpreted these as **host** paths and created `/data/` on the host — instead of using the actual host data at `/sandcastle/data/`.

**Fix:** Changed the volume mount from `${DATA_MOUNT}:/data` to `${DATA_MOUNT}:${DATA_MOUNT}` and the env from `SANDCASTLE_DATA_DIR: /data` to `SANDCASTLE_DATA_DIR: ${DATA_MOUNT}`. Now the container-internal path matches the host path (`/sandcastle/data`), so Docker bind mounts resolve correctly.

After deploying this fix, you'll want to:
1. Move any data Docker created at `/data` back to `/sandcastle/data` (or just `rm -rf /data` if it's a fresh install)
2. Re-run `sudo ./installer.sh install` to regenerate the compose file and restart
