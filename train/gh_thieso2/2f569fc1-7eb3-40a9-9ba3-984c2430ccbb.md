   473→    fi
   474→    $DOCKER network rm sandcastle-web 2>/dev/null || true
   475→  fi
   476→
   477→  # Destroy Dockyard
   478→  DOCKYARD_ENV_FILE="$SANDCASTLE_HOME/etc/dockyard.env"
   479→  if [ -f "$DOCKYARD_ENV_FILE" ] || systemctl cat "${DOCKYARD_DOCKER_PREFIX}docker.service" &>/dev/null; then
   480→    info "Destroying Dockyard..."
   481→    if [ -f "$DOCKYARD_ENV_FILE" ] && wget -q "https://raw.githubusercontent.com/thieso2/dockyard/refs/heads/main/dockyard.sh" -O /tmp/dockyard.sh 2>/dev/null; then
   482→      echo "y" | DOCKYARD_ENV="$DOCKYARD_ENV_FILE" bash /tmp/dockyard.sh destroy >/dev/null 2>&1 || true
   483→      rm -f /tmp/dockyard.sh
   484→    else
   485→      systemctl stop "${DOCKYARD_DOCKER_PREFIX}docker" 2>/dev/null || true
   486→      systemctl disable "${DOCKYARD_DOCKER_PREFIX}docker" 2>/dev/null || true
   487→      rm -f "/etc/systemd/system/${DOCKYARD_DOCKER_PREFIX}docker.service"
   488→      systemctl daemon-reload 2>/dev/null || true
   489→      ip link delete "${DOCKYARD_DOCKER_PREFIX}docker0" 2>/dev/null || true
   490→    fi
   491→    ok "Dockyard destroyed"
   492→  fi
   493→
   494→  # Remove user/group if we created them
   495→  if [ "${CREATED_USER:-}" = "true" ] && id "$SANDCASTLE_USER" &>/dev/null; then
   496→    userdel "$SANDCASTLE_USER" 2>/dev/null || true
   497→    ok "Removed user '${SANDCASTLE_USER}'"
   498→  fi
   499→  if [ "${CREATED_GROUP:-}" = "true" ] && getent group "$SANDCASTLE_GROUP" &>/dev/null; then
   500→    groupdel "$SANDCASTLE_GROUP" 2>/dev/null || true
   501→    ok "Removed group '${SANDCASTLE_GROUP}'"
   502→  fi
   503→
   504→  # Remove NAT service
   505→  if [ -f /etc/systemd/system/sandcastle-nat.service ]; then
   506→    systemctl disable --now sandcastle-nat 2>/dev/null || true
   507→    rm -f /etc/systemd/system/sandcastle-nat.service
   508→    systemctl daemon-reload 2>/dev/null || true
   509→  fi
   510→
   511→  # Remove Sandcastle files — keep user data (data/users, data/sandboxes)
   512→  rm -f "$SANDCASTLE_HOME/.env"
   513→  rm -f "$SANDCASTLE_HOME/docker-compose.yml"
   514→  rm -rf "$SANDCASTLE_HOME/etc"
   515→  rm -rf "$SANDCASTLE_HOME/data/traefik"
   516→  rmdir "$SANDCASTLE_HOME/data" 2>/dev/null || true
   517→  rmdir "$SANDCASTLE_HOME" 2>/dev/null || true
   518→  rm -rf "/run/${DOCKYARD_DOCKER_PREFIX}docker"

   520→  if [ -d "$SANDCASTLE_HOME/data/users" ] || [ -d "$SANDCASTLE_HOME/data/sandboxes" ]; then
   521→    warn "User data preserved in $SANDCASTLE_HOME/data/ — remove manually if no longer needed"
   522→  fi
   523→
   524→  ok "Sandcastle destroyed"
   525→}
