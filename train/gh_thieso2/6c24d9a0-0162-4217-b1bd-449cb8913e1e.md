   ```bash
     -c "CREATE DATABASE sandcastle_production_queue OWNER sandcastle;"
   ```
   Then restart: `docker compose restart web`

2. **Or nuke and restart** (if it's a fresh install anyway):
   ```bash
   docker compose down -v
   docker compose up -d
   ```

```
   145→                    f'https://ghcr.io/v2/thieso2/{repo}/manifests/{p[\"digest\"]}',
   146→                    headers={**h, 'Accept': 'application/vnd.oci.image.manifest.v1+json'})
   147→                m = json.load(urllib.request.urlopen(req))
   148→                break
   149→    cfg = json.load(urllib.request.urlopen(
   150→        urllib.request.Request(f'https://ghcr.io/v2/thieso2/{repo}/blobs/{m[\"config\"][\"digest\"]}', headers=h)))
   151→    ts = re.sub(r'(\.\d{6})\d+', r'\1', cfg['created']).replace('Z', '+00:00')
   152→    dt = datetime.fromisoformat(ts)
   153→    secs = int((datetime.now(timezone.utc) - dt).total_seconds())
   154→    if secs < 0: secs = 0
   155→    if secs >= 86400: ago = f'{secs // 86400}d ago'
   156→    elif secs >= 3600: ago = f'{secs // 3600}h ago'
   157→    else: ago = f'{max(1, secs // 60)}m ago'
   158→    print(f'  ghcr.io/thieso2/{repo}:latest    built {dt.strftime(\"%Y-%m-%d %H:%M UTC\")} ({ago})')
   159→except Exception:
   160→    print(f'  ghcr.io/thieso2/{repo}:latest    (unable to fetch build info)')
   161→" 2>/dev/null || echo "  ghcr.io/thieso2/${repo}:latest"
   162→}

   164→find_free_subnet() {
   165→  local used
   166→  used=$(
   167→    { ip route 2>/dev/null; ip addr 2>/dev/null; netstat -rn 2>/dev/null; } \
   168→      | grep -oE '(10|172|192)\.[0-9]+\.[0-9]+\.[0-9]+' \
   169→      | sort -un
   170→    $DOCKER network ls -q 2>/dev/null | while read -r nid; do
   171→      $DOCKER network inspect --format '{{range .IPAM.Config}}{{.Subnet}}{{end}}' "$nid" 2>/dev/null
   172→    done | grep -oE '(10|172|192)\.[0-9]+\.[0-9]+\.[0-9]+' \
   173→      | sort -un
   174→  )
   175→  for b in $(shuf -i 16-31); do
   176→    for c in $(shuf -i 1-254 | head -5); do
   177→      if ! echo "$used" | grep -q "^172\.${b}\.${c}\."; then
   178→        echo "172.${b}.${c}.0/24"
   179→        return
   180→      fi
   181→    done
   182→  done
   183→  for b in $(shuf -i 1-254 | head -20); do
   184→    for c in $(shuf -i 1-254 | head -5); do
   185→      if ! echo "$used" | grep -q "^10\.${b}\.${c}\."; then
   186→        echo "10.${b}.${c}.0/24"
   187→        return
   188→      fi
   189→    done
   190→  done
   191→  echo "172.30.99.0/24"
   192→}

   271→cmd_destroy() {
   272→  local auto_confirm="${1:-false}"
   273→
   274→  require_root
   275→
   276→  # Load installed env for CREATED_USER/CREATED_GROUP
   277→  if [ -f "$SANDCASTLE_HOME/etc/sandcastle.env" ]; then
   278→    # shellcheck source=/dev/null
   279→    source "$SANDCASTLE_HOME/etc/sandcastle.env"
   280→  fi
   281→
   282→  warn "This will remove Sandcastle from $SANDCASTLE_HOME (containers, config, services)"
   283→  if [ "$auto_confirm" != "true" ]; then
   284→    read -rp "Are you sure? (yes to confirm): " CONFIRM
   285→    [ "$CONFIRM" = "yes" ] || die "Aborted"
   286→  fi
   287→
   288→  info "Tearing down Sandcastle..."
   289→  cd /
   290→
   291→  if [ -x "$DOCKER" ]; then
   292→    for container in $($DOCKER ps -a --filter "name=^sc-ts-" --format '{{.Names}}' 2>/dev/null); do
   293→      info "Removing Tailscale sidecar: $container"
   294→      $DOCKER rm -f "$container" 2>/dev/null || true
   295→    done
   296→    for container in $($DOCKER ps -a --filter "runtime=sysbox-runc" --format '{{.Names}}' 2>/dev/null); do
   297→      info "Removing container: $container"
   298→      $DOCKER rm -f "$container" 2>/dev/null || true
   299→    done
   300→    for net in $($DOCKER network ls --filter "name=^sc-ts-net-" --format '{{.Name}}' 2>/dev/null); do
   301→      info "Removing network: $net"
   302→      $DOCKER network rm "$net" 2>/dev/null || true
   303→    done
   304→    if [ -f "$SANDCASTLE_HOME/docker-compose.yml" ]; then
   305→      $DOCKER compose -f "$SANDCASTLE_HOME/docker-compose.yml" --env-file "$SANDCASTLE_HOME/.env" down --rmi all --volumes --remove-orphans 2>/dev/null || true
   306→    fi
   307→    $DOCKER network rm sandcastle-web 2>/dev/null || true
   308→  fi
   309→
   310→  # Destroy Dockyard
   311→  DOCKYARD_ENV_FILE="$SANDCASTLE_HOME/etc/dockyard.env"
   312→  if [ -f "$DOCKYARD_ENV_FILE" ] || systemctl cat "${DOCKYARD_DOCKER_PREFIX}docker.service" &>/dev/null; then
   313→    info "Destroying Dockyard..."
   314→    if [ -f "$DOCKYARD_ENV_FILE" ] && wget -q "https://raw.githubusercontent.com/thieso2/dockyard/refs/heads/main/dockyard.sh" -O /tmp/dockyard.sh 2>/dev/null; then
   315→      echo "y" | DOCKYARD_ENV="$DOCKYARD_ENV_FILE" bash /tmp/dockyard.sh destroy >/dev/null 2>&1 || true
   316→      rm -f /tmp/dockyard.sh
   317→    else
   318→      systemctl stop "${DOCKYARD_DOCKER_PREFIX}docker" 2>/dev/null || true
   319→      systemctl disable "${DOCKYARD_DOCKER_PREFIX}docker" 2>/dev/null || true
   320→      rm -f "/etc/systemd/system/${DOCKYARD_DOCKER_PREFIX}docker.service"
   321→      systemctl daemon-reload 2>/dev/null || true
   322→      ip link delete "${DOCKYARD_DOCKER_PREFIX}docker0" 2>/dev/null || true
   323→    fi
   324→    ok "Dockyard destroyed"
   325→  fi
   326→
   327→  # Remove user/group if we created them
   328→  if [ "${CREATED_USER:-}" = "true" ] && id "$SANDCASTLE_USER" &>/dev/null; then
   329→    userdel "$SANDCASTLE_USER" 2>/dev/null || true
   330→    ok "Removed user '${SANDCASTLE_USER}'"
   331→  fi
   332→  if [ "${CREATED_GROUP:-}" = "true" ] && getent group "$SANDCASTLE_GROUP" &>/dev/null; then
   333→    groupdel "$SANDCASTLE_GROUP" 2>/dev/null || true
   334→    ok "Removed group '${SANDCASTLE_GROUP}'"
   335→  fi
   336→
   337→  # Remove NAT service
   338→  if [ -f /etc/systemd/system/sandcastle-nat.service ]; then
   339→    systemctl disable --now sandcastle-nat 2>/dev/null || true
   340→    rm -f /etc/systemd/system/sandcastle-nat.service
   341→    systemctl daemon-reload 2>/dev/null || true
   342→  fi
   343→
   344→  # Remove Sandcastle files — keep user data (data/users, data/sandboxes)
   345→  rm -f "$SANDCASTLE_HOME/.env"
   346→  rm -f "$SANDCASTLE_HOME/docker-compose.yml"
   347→  rm -rf "$SANDCASTLE_HOME/etc"
   348→  rm -rf "$SANDCASTLE_HOME/data/traefik"
   349→  rmdir "$SANDCASTLE_HOME/data" 2>/dev/null || true
   350→  rmdir "$SANDCASTLE_HOME" 2>/dev/null || true
   351→  rm -rf "/run/${DOCKYARD_DOCKER_PREFIX}docker"
   352→
   353→  if [ -d "$SANDCASTLE_HOME/data/users" ] || [ -d "$SANDCASTLE_HOME/data/sandboxes" ]; then
   354→    warn "User data preserved in $SANDCASTLE_HOME/data/ — remove manually if no longer needed"
   355→  fi
   356→
   357→  ok "Sandcastle destroyed"
   358→}

   387→
   388→    mkdir -p "$SANDCASTLE_HOME/etc"
   389→    cat > "$SANDCASTLE_HOME/etc/dockyard.env" <<DYEOF

   397→
   398→    DOCKYARD_ENV="$SANDCASTLE_HOME/etc/dockyard.env" bash /tmp/dockyard.sh create
   399→    rm -f /tmp/dockyard.sh
   400→
   401→    for i in $(seq 1 30); do
   402→      [ -S "$DOCKER_SOCK" ] && break
   403→      sleep 1
   404→    done
   405→    [ -S "$DOCKER_SOCK" ] || die "Dockyard socket not found at $DOCKER_SOCK after 30s"

   407→  fi

   410→

   412→    info "Configuring firewall..."
   413→    ufw --force reset >/dev/null 2>&1
   414→    ufw default deny incoming >/dev/null 2>&1
   415→    ufw default allow outgoing >/dev/null 2>&1
   416→    ufw allow 22/tcp >/dev/null 2>&1
   417→    ufw allow "${SANDCASTLE_HTTP_PORT}/tcp" >/dev/null 2>&1
   418→    ufw allow "${SANDCASTLE_HTTPS_PORT}/tcp" >/dev/null 2>&1
   419→    ufw allow 2201:2299/tcp >/dev/null 2>&1
   420→    ufw --force enable >/dev/null 2>&1
   421→    ok "Firewall configured (22, ${SANDCASTLE_HTTP_PORT}, ${SANDCASTLE_HTTPS_PORT}, 2201-2299)"
   422→  else
   423→    warn "UFW not found — skipping firewall setup"
   424→  fi
   425→
   426→  # ── Create system user/group ──────────────────────────────────────────────
   427→
   428→  CREATED_GROUP=false
   429→  CREATED_USER=false
   430→
   431→  if getent group "$SANDCASTLE_GROUP" &>/dev/null; then
   432→    ok "Group '${SANDCASTLE_GROUP}' already exists"
   433→  else
   434→    groupadd --system --gid "$SANDCASTLE_GID" "$SANDCASTLE_GROUP"
   435→    CREATED_GROUP=true
   436→    ok "Created group '${SANDCASTLE_GROUP}' (GID ${SANDCASTLE_GID})"
   437→  fi
   438→
   439→  if id "$SANDCASTLE_USER" &>/dev/null; then
   440→    ok "User '${SANDCASTLE_USER}' already exists"
   441→  else
   442→    useradd --system --uid "$SANDCASTLE_UID" --gid "$SANDCASTLE_GID" \
   443→      --home-dir "$SANDCASTLE_HOME" --shell /bin/bash "$SANDCASTLE_USER"

   445→    ok "Created user '${SANDCASTLE_USER}' (UID ${SANDCASTLE_UID})"
   446→  fi
   447→
   448→  # ── Create directories ────────────────────────────────────────────────────
   449→
   450→  mkdir -p "$SANDCASTLE_HOME"/etc
   451→  mkdir -p "$SANDCASTLE_HOME"/data/{users,sandboxes}
   452→  mkdir -p "$SANDCASTLE_HOME"/data/traefik/{dynamic,certs}
   453→  chown "${SANDCASTLE_USER}:${SANDCASTLE_GROUP}" "$SANDCASTLE_HOME"
   454→  chown -R "${SANDCASTLE_UID}:${SANDCASTLE_GID}" "$SANDCASTLE_HOME"/data/users "$SANDCASTLE_HOME"/data/sandboxes "$SANDCASTLE_HOME"/data/traefik/dynamic
   455→  usermod -d "$SANDCASTLE_HOME" "$SANDCASTLE_USER" 2>/dev/null || true
   456→
   457→  # ── Detect fresh install vs upgrade ─────────────────────────────────────
```
