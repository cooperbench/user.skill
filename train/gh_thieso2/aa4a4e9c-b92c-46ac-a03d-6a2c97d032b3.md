   405→SANDCASTLE_UID=${uid}
   406→SANDCASTLE_GID=${gid}
   407→
   408→# ─── Network & TLS ──────────────────────────────────────────────────────────
   409→SANDCASTLE_HOST=${host}
   410→SANDCASTLE_TLS_MODE=${tls_mode}
   411→#ACME_EMAIL=admin@example.com
   412→SANDCASTLE_HTTP_PORT=${http_port}
   413→SANDCASTLE_HTTPS_PORT=${https_port}
   414→#SANDCASTLE_SUBNET=

   418→SANDCASTLE_ADMIN_EMAIL=${admin_email}

   426→#GOOGLE_CLIENT_ID=
   427→#GOOGLE_CLIENT_SECRET=

   435→DOCKYARD_POOL_SIZE=${dy_pool_size}

   438→  ok "Generated $out — edit it, then run: sudo installer.sh install"

   443→cmd_destroy() {
   444→  local auto_confirm="${1:-false}"
   445→
   446→  require_root
   447→
   448→  # Load installed env for CREATED_USER/CREATED_GROUP
   449→  if [ -f "$SANDCASTLE_HOME/etc/sandcastle.env" ]; then
   450→    # shellcheck source=/dev/null
   451→    source "$SANDCASTLE_HOME/etc/sandcastle.env"
   452→  fi
   453→
   454→  warn "This will remove Sandcastle from $SANDCASTLE_HOME (containers, config, services)"
   455→  if [ "$auto_confirm" != "true" ]; then
   456→    read -rp "Are you sure? (yes to confirm): " CONFIRM
   457→    [ "$CONFIRM" = "yes" ] || die "Aborted"
   458→  fi
   459→
   460→  info "Tearing down Sandcastle..."
   461→  cd /
   462→
   463→  if [ -x "$DOCKER" ]; then
   464→    for container in $($DOCKER ps -a --filter "name=^sc-ts-" --format '{{.Names}}' 2>/dev/null); do
   465→      info "Removing Tailscale sidecar: $container"
   466→      $DOCKER rm -f "$container" 2>/dev/null || true
   467→    done
   468→    for container in $($DOCKER ps -a --filter "runtime=sysbox-runc" --format '{{.Names}}' 2>/dev/null); do
   469→      info "Removing container: $container"
   470→      $DOCKER rm -f "$container" 2>/dev/null || true
   471→    done
   472→    for net in $($DOCKER network ls --filter "name=^sc-ts-net-" --format '{{.Name}}' 2>/dev/null); do
   473→      info "Removing network: $net"
   474→      $DOCKER network rm "$net" 2>/dev/null || true
   475→    done
   476→    if [ -f "$SANDCASTLE_HOME/docker-compose.yml" ]; then
   477→      $DOCKER compose -f "$SANDCASTLE_HOME/docker-compose.yml" --env-file "$SANDCASTLE_HOME/.env" down --rmi all --volumes --remove-orphans 2>/dev/null || true
   478→    fi
   479→    $DOCKER network rm sandcastle-web 2>/dev/null || true
   480→  fi
   481→
   482→  # Destroy Dockyard
   483→  DOCKYARD_ENV_FILE="$SANDCASTLE_HOME/etc/dockyard.env"
   484→  if [ -f "$DOCKYARD_ENV_FILE" ] || systemctl cat "${DOCKYARD_DOCKER_PREFIX}docker.service" &>/dev/null; then
   485→    info "Destroying Dockyard..."
   486→    if [ -f "$DOCKYARD_ENV_FILE" ] && wget -q "https://raw.githubusercontent.com/thieso2/dockyard/refs/heads/main/dockyard.sh" -O /tmp/dockyard.sh 2>/dev/null; then
   487→      echo "y" | DOCKYARD_ENV="$DOCKYARD_ENV_FILE" bash /tmp/dockyard.sh destroy >/dev/null 2>&1 || true
   488→      rm -f /tmp/dockyard.sh
   489→    else
   490→      systemctl stop "${DOCKYARD_DOCKER_PREFIX}docker" 2>/dev/null || true
   491→      systemctl disable "${DOCKYARD_DOCKER_PREFIX}docker" 2>/dev/null || true
   492→      rm -f "/etc/systemd/system/${DOCKYARD_DOCKER_PREFIX}docker.service"
   493→      systemctl daemon-reload 2>/dev/null || true
   494→      ip link delete "${DOCKYARD_DOCKER_PREFIX}docker0" 2>/dev/null || true
   495→    fi
   496→    ok "Dockyard destroyed"
   497→  fi
   498→
   499→  # Remove user/group if we created them
   500→  if [ "${CREATED_USER:-}" = "true" ] && id "$SANDCASTLE_USER" &>/dev/null; then
   501→    userdel "$SANDCASTLE_USER" 2>/dev/null || true
   502→    ok "Removed user '${SANDCASTLE_USER}'"
   503→  fi
   504→  if [ "${CREATED_GROUP:-}" = "true" ] && getent group "$SANDCASTLE_GROUP" &>/dev/null; then
   505→    groupdel "$SANDCASTLE_GROUP" 2>/dev/null || true
   506→    ok "Removed group '${SANDCASTLE_GROUP}'"
   507→  fi
   508→
   509→  # Remove NAT service
   510→  if [ -f /etc/systemd/system/sandcastle-nat.service ]; then
   511→    systemctl disable --now sandcastle-nat 2>/dev/null || true
   512→    rm -f /etc/systemd/system/sandcastle-nat.service
   513→    systemctl daemon-reload 2>/dev/null || true
   514→  fi
   515→
   516→  # Remove Sandcastle files — keep user data (data/users, data/sandboxes)
   517→  rm -f "$SANDCASTLE_HOME/.env"
   518→  rm -f "$SANDCASTLE_HOME/docker-compose.yml"
   519→  rm -rf "$SANDCASTLE_HOME/etc"
   520→  rm -rf "$SANDCASTLE_HOME/data/traefik"
   521→  rmdir "$SANDCASTLE_HOME/data" 2>/dev/null || true
   522→  rmdir "$SANDCASTLE_HOME" 2>/dev/null || true
   523→  rm -rf "/run/${DOCKYARD_DOCKER_PREFIX}docker"
   524→
   525→  if [ -d "$SANDCASTLE_HOME/data/users" ] || [ -d "$SANDCASTLE_HOME/data/sandboxes" ]; then
   526→    warn "User data preserved in $SANDCASTLE_HOME/data/ — remove manually if no longer needed"
   527→  fi
   528→
   529→  ok "Sandcastle destroyed"
   530→}

   532→# ═══ cmd_install ═════════════════════════════════════════════════════════════

   534→cmd_install() {
   535→  require_root
   536→
   537→  if ! grep -qi ubuntu /etc/os-release 2>/dev/null; then
   538→    warn "This script is tested on Ubuntu 24.04. Other distros may work but are unsupported."
   539→  fi
   540→  if [ "$ARCH" != "amd64" ] && [ "$ARCH" != "arm64" ]; then
   541→    die "Sandcastle requires amd64 or arm64 architecture (got: $ARCH)"
   542→  fi
   543→
   544→  echo ""
   545→  echo -e "${BLUE}═══ Sandcastle Installer ═══${NC}"
   546→  echo ""
   547→  info "Available images (${ARCH}):"
   548→  show_image_info "sandcastle"
   549→  show_image_info "sandcastle-sandbox"
   550→  echo ""
   551→
   552→  # ── Install Dockyard ──────────────────────────────────────────────────────
   553→
   554→  if [ -S "$DOCKER_SOCK" ]; then
   555→    ok "Dockyard already installed"
   556→  else
   557→    info "Installing Dockyard (Docker + Sysbox)..."
   558→    wget -q "https://raw.githubusercontent.com/thieso2/dockyard/refs/heads/main/dockyard.sh" -O /tmp/dockyard.sh
   559→
   560→    mkdir -p "$SANDCASTLE_HOME/etc"
   561→    cat > "$SANDCASTLE_HOME/etc/dockyard.env" <<DYEOF
   562→DOCKYARD_ROOT=${DOCKYARD_ROOT}
   563→DOCKYARD_DOCKER_PREFIX=${DOCKYARD_DOCKER_PREFIX}
   564→DOCKYARD_BRIDGE_CIDR=${DOCKYARD_BRIDGE_CIDR}
   565→DOCKYARD_FIXED_CIDR=${DOCKYARD_FIXED_CIDR}
   566→DOCKYARD_POOL_BASE=${DOCKYARD_POOL_BASE}
   567→DOCKYARD_POOL_SIZE=${DOCKYARD_POOL_SIZE}
   568→DYEOF
   569→    wrote "$SANDCASTLE_HOME/etc/dockyard.env"
   570→
   571→    DOCKYARD_ENV="$SANDCASTLE_HOME/etc/dockyard.env" bash /tmp/dockyard.sh create
   572→    rm -f /tmp/dockyard.sh
   573→
   574→    for i in $(seq 1 30); do
   575→      [ -S "$DOCKER_SOCK" ] && break
   576→      sleep 1
   577→    done
   578→    [ -S "$DOCKER_SOCK" ] || die "Dockyard socket not found at $DOCKER_SOCK after 30s"
   579→    ok "Dockyard installed"
   580→  fi
   581→
   582→  # ── Configure UFW ─────────────────────────────────────────────────────────
   583→
   584→  if command -v ufw &>/dev/null; then
   585→    info "Configuring firewall..."
   586→    ufw --force reset >/dev/null 2>&1
   587→    ufw default deny incoming >/dev/null 2>&1
   588→    ufw default allow outgoing >/dev/null 2>&1
   589→    ufw allow 22/tcp >/dev/null 2>&1
   590→    ufw allow "${SANDCASTLE_HTTP_PORT}/tcp" >/dev/null 2>&1
   591→    ufw allow "${SANDCASTLE_HTTPS_PORT}/tcp" >/dev/null 2>&1
   592→    ufw allow 2201:2299/tcp >/dev/null 2>&1
   593→    ufw --force enable >/dev/null 2>&1
   594→    ok "Firewall configured (22, ${SANDCASTLE_HTTP_PORT}, ${SANDCASTLE_HTTPS_PORT}, 2201-2299)"
   595→  else
   596→    warn "UFW not found — skipping firewall setup"
   597→  fi
   598→
   599→  # ── Create system user/group ──────────────────────────────────────────────
   600→
   601→  CREATED_GROUP=false
   602→  CREATED_USER=false
   603→
   604→  if getent group "$SANDCASTLE_GROUP" &>/dev/null; then
   605→    ok "Group '${SANDCASTLE_GROUP}' already exists"
   606→  else
   607→    groupadd --system --gid "$SANDCASTLE_GID" "$SANDCASTLE_GROUP"
   608→    CREATED_GROUP=true
   609→    ok "Created group '${SANDCASTLE_GROUP}' (GID ${SANDCASTLE_GID})"
   610→  fi
   611→
   612→  if id "$SANDCASTLE_USER" &>/dev/null; then
   613→    ok "User '${SANDCASTLE_USER}' already exists"
   614→  else
   615→    useradd --system --uid "$SANDCASTLE_UID" --gid "$SANDCASTLE_GID" \
   616→      --home-dir "$SANDCASTLE_HOME" --shell /bin/bash "$SANDCASTLE_USER"
   617→    CREATED_USER=true
   618→    ok "Created user '${SANDCASTLE_USER}' (UID ${SANDCASTLE_UID})"
   619→  fi
   620→
   621→  # ── Create directories ────────────────────────────────────────────────────
   622→
   623→  ensure_dirs
   624→
   625→  # ── Detect fresh install vs upgrade ─────────────────────────────────────
   626→
   627→  FRESH_INSTALL=false
   628→  [ ! -f "$SANDCASTLE_HOME/.env" ] && FRESH_INSTALL=true
   629→
   630→  # ── Fresh install: validate & generate .env ─────────────────────────────
   631→
   632→  if [ "$FRESH_INSTALL" = true ]; then
   633→    [ -z "${SANDCASTLE_HOST:-}" ] && die "SANDCASTLE_HOST is required (set in sandcastle.env)"
   634→    [ -z "${SANDCASTLE_ADMIN_EMAIL:-}" ] && die "SANDCASTLE_ADMIN_EMAIL is required (set in sandcastle.env)"
   635→    [ -z "${SANDCASTLE_ADMIN_PASSWORD:-}" ] && die "SANDCASTLE_ADMIN_PASSWORD is required (set in sandcastle.env or use SANDCASTLE_ADMIN_PASSWORD_FILE)"
   636→    [ ${#SANDCASTLE_ADMIN_PASSWORD} -lt 6 ] && die "SANDCASTLE_ADMIN_PASSWORD must be at least 6 characters"
   637→
   638→    if [ -z "${SANDCASTLE_SUBNET:-}" ]; then
   639→      SANDCASTLE_SUBNET=$(find_free_subnet)
   640→      info "Auto-detected subnet: $SANDCASTLE_SUBNET"
   641→    fi
   642→
   643→    SECRET_KEY_BASE=$(openssl rand -hex 64)
   644→    DB_PASSWORD=$(openssl rand -hex 32)
   645→    DOCKER_GID=$(stat -c '%g' "$DOCKER_SOCK" 2>/dev/null || echo "988")
   646→
   647→    cat > "$SANDCASTLE_HOME/.env" <<EOF
   648→# Sandcastle runtime — generated $(date -Iseconds)
   649→SANDCASTLE_HOME="${SANDCASTLE_HOME}"
   650→SANDCASTLE_HOST="${SANDCASTLE_HOST}"
   651→SANDCASTLE_TLS_MODE="${SANDCASTLE_TLS_MODE}"
   652→SANDCASTLE_USER="${SANDCASTLE_USER}"
   653→SANDCASTLE_GROUP="${SANDCASTLE_GROUP}"
   654→SANDCASTLE_UID="${SANDCASTLE_UID}"
   655→SANDCASTLE_GID="${SANDCASTLE_GID}"
   656→SECRET_KEY_BASE="${SECRET_KEY_BASE}"
   657→DB_PASSWORD="${DB_PASSWORD}"
   658→SANDCASTLE_ADMIN_USER="${SANDCASTLE_ADMIN_USER}"
   659→SANDCASTLE_ADMIN_EMAIL="${SANDCASTLE_ADMIN_EMAIL}"
   660→SANDCASTLE_ADMIN_PASSWORD="${SANDCASTLE_ADMIN_PASSWORD}"
   661→SANDCASTLE_SUBNET="${SANDCASTLE_SUBNET}"
   662→SANDCASTLE_ADMIN_SSH_KEY="${SANDCASTLE_ADMIN_SSH_KEY:-}"
   663→DOCKER_GID="${DOCKER_GID}"
   664→DOCKER_SOCK="${DOCKER_SOCK}"
   665→ACME_EMAIL="${ACME_EMAIL:-}"
   666→GITHUB_CLIENT_ID="${GITHUB_CLIENT_ID:-}"
   667→GITHUB_CLIENT_SECRET="${GITHUB_CLIENT_SECRET:-}"
   668→GOOGLE_CLIENT_ID="${GOOGLE_CLIENT_ID:-}"
   669→GOOGLE_CLIENT_SECRET="${GOOGLE_CLIENT_SECRET:-}"
   670→EOF
   671→    chmod 600 "$SANDCASTLE_HOME/.env"
   672→    wrote "$SANDCASTLE_HOME/.env"
   673→  else

   686→    DOCKER_SOCK="${DOCKYARD_ROOT}/docker.sock"
