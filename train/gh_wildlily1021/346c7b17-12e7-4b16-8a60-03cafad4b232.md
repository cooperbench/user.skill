172	  2. bytes 长度 >= functionCodeOffset + 4
173	  3. bytes[functionCodeOffset] === hex(scoeIdentifier)
174	  4. bytes[functionCodeOffset + 2] === 0xAA && bytes[functionCodeOffset + 3] === 0xAA
175	  5. 如果 scoeFramesLoaded === false:

495	
496	### 5.1 SEND_FRAME → TaskDefinition
497	
498	使用 `schedule: { kind: 'immediate' }`（顺序执行所有步骤）。
499	
500	```typescript
501	function buildSendFrameTask(
502	  command: ParsedCommand,
503	  ctx: CommandContext,
504	): TaskDefinition {
505	  const config = command.commandConfig;
506	  const steps: TaskStepDefinition[] = [];
507	
508	  // 每个帧映射 → 一个 send step
509	  for (const mapping of config.frameMappings ?? []) {
510	    const userFieldValues: Record<string, string | number | boolean> = {};
511	    for (const fm of mapping.fieldMappings) {
512	      if (fm.source === 'param' && fm.paramId) {
513	        userFieldValues[fm.fieldId] = command.resolvedParams[fm.paramId] ?? '';
514	      } else if (fm.fixedValue !== undefined) {
515	        userFieldValues[fm.fieldId] = fm.fixedValue;
516	      }
517	    }
518	
519	    steps.push({
520	      id: `send-${mapping.instanceId}`,
521	      kind: 'send',
522	      config: {
523	        frameId: mapping.frameId,
524	        targetId: mapping.targetId,
525	        userFieldValues,
526	      },
527	    });
528	
529	    // 发送间隔（帧间）→ intervalAfterMs
530	    if (config.sendInterval && config.sendInterval > 0) {
531	      steps.push({
532	        id: `delay-after-${mapping.instanceId}`,
533	        kind: 'delay',
534	        config: { durationMs: config.sendInterval },
535	      });
536	    }
537	  }
538	
539	  // 完成条件 → wait-condition step（支持多条件数组）
540	  const waitConditions = buildWaitConditions(command, config.completionConditions);
541	  if (waitConditions.length > 0) {
542	    steps.push({
543	      id: 'wait-completion',
544	      kind: 'wait-condition',
545	      config: {
546	        conditions: waitConditions,
547	        timeoutMs: config.completionTimeout ?? 5000,
548	        onTimeout: 'fail',
549	      },
550	    });
551	  }
552	
553	  return {
554	    id: `scoe-send-${command.commandId}`,
555	    name: `SCOE SEND_FRAME: ${config.label}`,
556	    schedule: { kind: 'immediate' },
557	    steps,
558	    errorPolicy: { onFailure: 'stop' },
559	  };
560	}
561	```

624	### 5.3 完成条件 → ConditionTerm[]
625	
626	task-real design 将 `WaitConditionStepConfig.condition` 改为 `WaitConditionConfig.conditions` 数组，支持多条件 AND/OR 组合。所有完成条件可合并为单个 `conditions` 数组。
627	
628	```typescript
629	function buildWaitConditions(
630	  command: ParsedCommand,
631	  condConfigs?: readonly CompletionConditionConfig[],
632	): ConditionTerm[] {
633	  if (!condConfigs || condConfigs.length === 0) return [];
634	
635	  return condConfigs.map((condConfig) => {
636	    if (condConfig.useParam && condConfig.targetParamId) {
637	      // 参数化模式：从 resolvedParams 取值 → 查 param.options 索引 → 同索引取 condition.options
638	      const paramValue = command.resolvedParams[condConfig.targetParamId];
639	      const param = command.commandConfig.params?.find(p => p.id === condConfig.targetParamId);
640	      const optionIndex = param?.options.findIndex(o => o.value === paramValue) ?? -1;
641	      const condOption = condConfig.options?.[optionIndex];
642	
643	      return {
644	        frameId: condConfig.sourceFrameId,
645	        fieldId: condConfig.sourceFieldId,
646	        operator: condOption?.operator ?? 'eq' as ComparisonOperator,
647	        threshold: condOption?.matchValue ?? '',
648	      };
649	    } else {
650	      // 固定值模式
651	      return {
652	        frameId: condConfig.sourceFrameId,
653	        fieldId: condConfig.sourceFieldId,
654	        operator: condConfig.operator ?? 'eq' as ComparisonOperator,
655	        threshold: condConfig.targetFixedValue ?? '',
656	      };
657	    }
658	  });
659	}
660	```
661	
662	来源：`scoeFrame.ts:416-491` checkCompletionConditions 逻辑。task-real design §1.1 ConditionTerm 支持 `logicOperator`（默认 `'and'`），与旧系统"所有条件都满足"语义一致，无需显式设置。

671	  for param in params:
672	    bytes = data[param.offset ... param.offset + param.length]
673	    hex = bytesToUpperCaseHex(bytes)

690	## 7. 确认帧发送
691	
692	### 7.1 策略
693	
694	- 成功：adapter 在 `taskService.onSettled(instanceId)` resolve 后，通过 `connectionService.write()` 发送 `successFrameId` 帧（UDP）
695	- 失败/超时：只更新领域状态（errorCount++、lastErrorReason），不发任何回应帧
696	
697	### 7.2 发送流程
698	
699	**选定路径**：通过 `sendService.execute()` 发送确认帧。sendService 已有 frame→bytes 序列化 + connection write 能力，command-ingress 不需要自行序列化帧。
700	
701	```typescript
702	async function sendAckFrame(
703	  command: ParsedCommand,
704	  ctx: CommandContext,
705	): Promise<void> {
706	  const successFrameId = command.commandConfig.successFrameId
707	    ?? ctx.stateReader.globalConfig().successFrameId;
708	  if (!successFrameId) return;
709	
710	  const frame = ctx.frameReader.getFrame(successFrameId);
711	  if (!frame) return;
712	
713	  // 通过 sendService 发送：序列化帧 + 写入 connection（复用 send service 的 frameToBuffer）
714	  await ctx.sendService.execute({
715	    frameId: successFrameId,
716	    targetId: command.commandConfig.frameMappings?.[0]?.targetId  // UDP targetId
717	              ?? ctx.stateReader.globalConfig().udpTargetId,
718	    userFieldValues: {},
719	  });
720	}
721	```
722	
723	**前提**：CommandContext 需新增 `sendService: SendService` 字段。sendService.execute() 的 `SendRequest` 接口已在 send feature 中定义，接受 `{ frameId, targetId, fieldValues }`。
724	
725	**GAP 关闭**：G3 已解决，确认帧发送路径锁定为 sendService。
726	
727	### 7.3 onSettled 注册方式
728	
729	```typescript
730	// 在 handler 中
731	ctx.taskService.startTask(instance.instanceId);
732	
733	// fire-and-forget：监听 settled 后发确认帧
734	ctx.taskService.onSettled(instance.instanceId).then(() => {
735	  const inst = ctx.taskService.getInstance(instance.instanceId);
736	  if (inst?.lifecycle === 'completed') {
737	    sendAckFrame(command, ctx);
738	    ctx.stateWriter.incrementSuccessCount();
739	  } else {
