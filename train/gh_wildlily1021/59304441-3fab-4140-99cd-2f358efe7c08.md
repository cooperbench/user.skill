189	// ===== Stats panel helpers =====
190	function getStatsDisplay(key: string): string | number {
191	  const stats = monitor.statistics.value;
192	  const status = monitor.runtimeStatus.value;
193	  switch (key) {
194	    case 'runtimeSeconds': return stats.runtimeSeconds || '—';
195	    case 'satelliteIdRuntimeSeconds': return stats.satelliteIdRuntimeSeconds || '—';
196	    case 'commandReceiveCount': return stats.commandReceiveCount;
197	    case 'commandSuccessCount': return stats.commandSuccessCount;
198	    case 'commandErrorCount': return stats.commandErrorCount;
199	    case 'lastCommandCode': return status.lastCommandCode || '—';
200	    case 'lastErrorReason': return stats.lastErrorReason || '—';
201	    case 'loadedSatelliteId': return status.loadedSatelliteId || '—';
202	    default: return '—';
203	  }
204	}
205	
206	function isStatusKey(key: string): boolean {
207	  return key === 'healthStatus' || key === 'linkTestResult';
208	}
209	
210	function getStatusMap(key: string): Record<string, { label: string; color: string }> {
211	  return key === 'healthStatus' ? healthStatusMap : linkTestStatusMap;
212	}
213	
214	function getStatusValue(key: string): string {
215	  const status = monitor.runtimeStatus.value;
216	  return key === 'healthStatus' ? status.healthStatus : status.linkTestResult;
