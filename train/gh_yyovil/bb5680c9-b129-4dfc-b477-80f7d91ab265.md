335	  try {

347	      ([id, p]) => p.sessionPrefix ?? id,

350	      (session) =>

354	          allSessionPrefixes,

361	    );

367	    for (const [projectId, project] of Object.entries(config.projects)) {
368	      if (availableSlots <= 0) break;
