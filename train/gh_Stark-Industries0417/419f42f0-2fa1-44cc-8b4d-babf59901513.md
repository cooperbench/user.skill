   312→// mergeFilesTouched merges multiple file lists into existing touched files, deduplicating.
   313→func mergeFilesTouched(existing []string, fileLists ...[]string) []string {
   314→	seen := make(map[string]bool)
   315→	for _, f := range existing {
   316→		seen[f] = true
   317→	}
   318→
   319→	for _, list := range fileLists {
   320→		for _, f := range list {
   321→			seen[f] = true
   322→		}
   323→	}
   324→
   325→	result := make([]string, 0, len(seen))
   326→	for f := range seen {
   327→		result = append(result, f)
   328→	}
   329→
   330→	// Sort for deterministic output
   331→	sort.Strings(result)
   332→	return result
   333→}

   335→// accumulateTokenUsage adds new token usage to existing accumulated usage.
   336→// If existing is nil, returns a copy of incoming. If incoming is nil, returns existing unchanged.
   337→func accumulateTokenUsage(existing, incoming *agent.TokenUsage) *agent.TokenUsage {
   338→	if incoming == nil {
   339→		return existing
   340→	}
   341→	if existing == nil {
   342→		// Return a copy to avoid sharing the pointer
   343→		return &agent.TokenUsage{
   344→			InputTokens:         incoming.InputTokens,
   345→			CacheCreationTokens: incoming.CacheCreationTokens,
   346→			CacheReadTokens:     incoming.CacheReadTokens,
   347→			OutputTokens:        incoming.OutputTokens,
   348→			APICallCount:        incoming.APICallCount,
   349→			SubagentTokens:      incoming.SubagentTokens,
   350→		}
   351→	}
   352→
   353→	// Accumulate values
   354→	existing.InputTokens += incoming.InputTokens
   355→	existing.CacheCreationTokens += incoming.CacheCreationTokens
   356→	existing.CacheReadTokens += incoming.CacheReadTokens
   357→	existing.OutputTokens += incoming.OutputTokens
   358→	existing.APICallCount += incoming.APICallCount
   359→
   360→	// Accumulate subagent tokens if present
   361→	if incoming.SubagentTokens != nil {
   362→		existing.SubagentTokens = accumulateTokenUsage(existing.SubagentTokens, incoming.SubagentTokens)
   363→	}
   364→
   365→	return existing
   366→}

   368→// deleteShadowBranch deletes a shadow branch by name using an existing repo handle.
   369→// Returns nil if the branch doesn't exist (idempotent).
   370→func deleteShadowBranch(repo *git.Repository, branchName string) error {
   371→	refName := plumbing.NewBranchReferenceName(branchName)
   372→
   373→	// Check if reference exists
   374→	ref, err := repo.Reference(refName, true)
   375→	if err != nil {
   376→		// Branch doesn't exist - nothing to delete (idempotent)
   377→		return nil //nolint:nilerr // Not an error condition - branch already gone
   378→	}
   379→
   380→	// Delete the reference
   381→	if err := repo.Storer.RemoveReference(ref.Name()); err != nil {
   382→		return fmt.Errorf("failed to delete branch %s: %w", branchName, err)
   383→	}
   384→
   385→	return nil
