   293→		// No new content - check if any session has a LastCheckpointID to reuse
   294→		// This handles the case where user splits Claude's work across multiple commits
   295→		// Reuse if LastCheckpointID exists AND staged files overlap with session files.
   296→		// After condensation FilesTouched is nil; we fall back to condensed metadata
   297→		// so split-commit detection still works.
   298→
   299→		// Get current HEAD to filter sessions
   300→		head, err := repo.Head()
   301→		if err != nil {
   302→			return nil //nolint:nilerr // Hook must be silent on failure
   303→		}
   304→		currentHeadHash := head.Hash().String()
   305→
   306→		// Filter to sessions where BaseCommit matches current HEAD
   307→		// This prevents reusing checkpoint IDs from old sessions
   308→		// Note: BaseCommit is kept current both when new content is condensed (in the
   309→		// condensation process) and when no new content is found (via PostCommit when
   310→		// reusing checkpoint IDs). If none match, we don't add a trailer rather than
   311→		// falling back to old sessions which could have stale checkpoint IDs.
   312→		var currentSessions []*SessionState
   313→		for _, session := range sessions {
   314→			if session.BaseCommit == currentHeadHash {
   315→				currentSessions = append(currentSessions, session)
   316→			}
   317→		}
   318→
   319→		if len(currentSessions) == 0 {
   320→			// No sessions match current HEAD - don't try to reuse checkpoint IDs
   321→			// from old sessions as they may be stale
   322→			logging.Debug(logCtx, "prepare-commit-msg: no sessions match current HEAD",
   323→				slog.String("strategy", "manual-commit"),
   324→				slog.String("source", source),
   325→				slog.String("current_head", truncateHash(currentHeadHash)),
   326→				slog.Int("total_sessions", len(sessions)),
   327→			)
   328→			return nil
   329→		}
   330→
   331→		stagedFiles := getStagedFiles(repo)
   332→		for _, session := range currentSessions {
   333→			if session.LastCheckpointID.IsEmpty() {
   334→				continue
   335→			}
   336→			filesToCheck := session.FilesTouched
   337→			// After condensation FilesTouched is reset. Fall back to the
   338→			// condensed metadata's files_touched so split-commit detection
   339→			// still works (staged files checked against previously-condensed files).
   340→			if len(filesToCheck) == 0 && session.StepCount == 0 {
   341→				if condensedFiles := s.getCondensedFilesTouched(repo, session.LastCheckpointID); len(condensedFiles) > 0 {
   342→					filesToCheck = condensedFiles
   343→				}
   344→			}
   345→			if len(filesToCheck) == 0 || hasOverlappingFiles(stagedFiles, filesToCheck) {
   346→				checkpointID = session.LastCheckpointID
   347→				reusedSession = session
   348→				break
   349→			}
   350→		}
   351→		if checkpointID.IsEmpty() {
   352→			// No new content and no previous checkpoint to reference (or staged files are unrelated)
   353→			logging.Debug(logCtx, "prepare-commit-msg: no content to link",
   354→				slog.String("strategy", "manual-commit"),
   355→				slog.String("source", source),
   356→				slog.Int("sessions_found", len(sessions)),
   357→				slog.Int("sessions_with_content", len(sessionsWithContent)),
   358→			)
   359→			return nil
   360→		}
   361→	}
   362→
   363→	// Read current commit message
   364→	content, err := os.ReadFile(commitMsgFile) //nolint:gosec // commitMsgFile is provided by git hook
   365→	if err != nil {
   366→		return nil //nolint:nilerr // Hook must be silent on failure
   367→	}
   368→
   369→	message := string(content)
   370→
   371→	// Get or generate checkpoint ID (ParseCheckpoint validates format, so found==true means valid)
   372→	if existingCpID, found := trailers.ParseCheckpoint(message); found {
   373→		// Trailer already exists (e.g., amend) - keep it
   374→		logging.Debug(logCtx, "prepare-commit-msg: trailer already exists",
   375→			slog.String("strategy", "manual-commit"),
   376→			slog.String("source", source),
   377→			slog.String("existing_checkpoint_id", existingCpID.String()),
   378→		)
   379→		return nil
   380→	}
   381→
   382→	if hasNewContent {
   383→		// New content: check PendingCheckpointID first (set during previous condensation),
   384→		// otherwise generate a new one. This ensures idempotent IDs across hook invocations.
   385→		for _, state := range sessionsWithContent {
   386→			if state.PendingCheckpointID != "" {
   387→				if cpID, err := id.NewCheckpointID(state.PendingCheckpointID); err == nil {
   388→					checkpointID = cpID
   389→					break
   390→				}
   391→			}
   392→		}
   393→		if checkpointID.IsEmpty() {
   394→			cpID, err := id.Generate()
   395→			if err != nil {
   396→				return fmt.Errorf("failed to generate checkpoint ID: %w", err)
   397→			}
   398→			checkpointID = cpID
   399→		}
   400→	}
   401→	// Otherwise checkpointID is already set to LastCheckpointID from above
   402→
   403→	// Determine agent type and last prompt from session
   404→	agentType := DefaultAgentType // default for backward compatibility
   405→	var lastPrompt string
   406→	if hasNewContent && len(sessionsWithContent) > 0 {
   407→		session := sessionsWithContent[0]
   408→		if session.AgentType != "" {
   409→			agentType = session.AgentType
   410→		}
   411→		lastPrompt = s.getLastPrompt(repo, session)
   412→	} else if reusedSession != nil {
   413→		// Reusing checkpoint from existing session - get agent type and prompt from that session
   414→		if reusedSession.AgentType != "" {
   415→			agentType = reusedSession.AgentType
   416→		}
   417→		lastPrompt = s.getLastPrompt(repo, reusedSession)
   418→	}
   419→
   420→	// Prepare prompt for display: collapse newlines/whitespace, then truncate (rune-safe)
   421→	displayPrompt := stringutil.TruncateRunes(stringutil.CollapseWhitespace(lastPrompt), 80, "...")
   422→
   423→	// Check if we're restoring an existing checkpoint ID (already condensed)
   424→	// vs linking a genuinely new checkpoint. Restoring doesn't need user confirmation
   425→	// since the data is already committed — this handles git commit --amend -m "..."
   426→	// and non-interactive environments (e.g., Claude doing commits).
   427→	isRestoringExisting := false
   428→	if !hasNewContent && reusedSession != nil {
   429→		// Reusing LastCheckpointID from a previous condensation
   430→		isRestoringExisting = true
   431→	} else if hasNewContent {
   432→		for _, state := range sessionsWithContent {
   433→			if state.PendingCheckpointID != "" {
   434→				isRestoringExisting = true
   435→				break
   436→			}
   437→		}
   438→	}
   439→
   440→	// Add trailer differently based on commit source
   441→	switch {
   442→	case source == "message" && !isRestoringExisting:
   443→		// Using -m or -F with genuinely new content: ask user interactively
   444→		// whether to add trailer (comments won't be stripped by git in this mode)
   445→
   446→		// Build context string for interactive prompt
   447→		var promptContext string
   448→		if displayPrompt != "" {
   449→			promptContext = "You have an active " + string(agentType) + " session.\nLast Prompt: " + displayPrompt
   450→		}
   451→
   452→		if !askConfirmTTY("Link this commit to "+string(agentType)+" session context?", promptContext, true) {
   453→			// User declined - don't add trailer
   454→			logging.Debug(logCtx, "prepare-commit-msg: user declined trailer",
   455→				slog.String("strategy", "manual-commit"),
   456→				slog.String("source", source),
   457→			)
   458→			return nil
   459→		}
   460→		message = addCheckpointTrailer(message, checkpointID)
   461→	case source == "message":
   462→		// Restoring existing checkpoint ID (amend, split commit, or non-interactive)
   463→		// No confirmation needed — data is already condensed
   464→		message = addCheckpointTrailer(message, checkpointID)
   465→	default:
   466→		// Normal editor flow: add trailer with explanatory comment (will be stripped by git)
   467→		message = addCheckpointTrailerWithComment(message, checkpointID, string(agentType), displayPrompt)
   468→	}
   469→
   470→	logging.Info(logCtx, "prepare-commit-msg: trailer added",
   471→		slog.String("strategy", "manual-commit"),
   472→		slog.String("source", source),
   473→		slog.String("checkpoint_id", checkpointID.String()),
   474→		slog.Bool("has_new_content", hasNewContent),
   475→	)
   476→
   477→	// Write updated message back
   478→	if err := os.WriteFile(commitMsgFile, []byte(message), 0o600); err != nil {
   479→		return nil //nolint:nilerr // Hook must be silent on failure
   480→	}
   481→
   482→	return nil
   483→}

   485→// handleAmendCommitMsg handles the prepare-commit-msg hook for amend operations
   486→// (source="commit"). It preserves existing trailers or restores from PendingCheckpointID.
   487→func (s *ManualCommitStrategy) handleAmendCommitMsg(logCtx context.Context, commitMsgFile string) error {
   488→	// Read current commit message
   489→	content, err := os.ReadFile(commitMsgFile) //nolint:gosec // commitMsgFile is provided by git hook
   490→	if err != nil {
   491→		return nil //nolint:nilerr // Hook must be silent on failure
   492→	}
   493→
   494→	message := string(content)
   495→
   496→	// If message already has a trailer, keep it unchanged
   497→	if existingCpID, found := trailers.ParseCheckpoint(message); found {
   498→		logging.Debug(logCtx, "prepare-commit-msg: amend preserves existing trailer",
   499→			slog.String("strategy", "manual-commit"),
   500→			slog.String("checkpoint_id", existingCpID.String()),
   501→		)
   502→		return nil
   503→	}
   504→
   505→	// No trailer in message — check if any session has PendingCheckpointID to restore
   506→	worktreePath, err := GetWorktreePath()
   507→	if err != nil {
   508→		return nil //nolint:nilerr // Hook must be silent on failure
   509→	}
   510→
   511→	sessions, err := s.findSessionsForWorktree(worktreePath)
   512→	if err != nil || len(sessions) == 0 {
   513→		return nil //nolint:nilerr // No sessions - nothing to restore
   514→	}
   515→
   516→	// For amend, HEAD^ is the commit being amended, and HEAD is where we are now.
   517→	// We need to match sessions whose BaseCommit equals HEAD (the commit being amended
   518→	// was created from this base). This prevents stale sessions from injecting
   519→	// unrelated checkpoint IDs.
   520→	repo, repoErr := OpenRepository()
   521→	if repoErr != nil {
   522→		return nil //nolint:nilerr // Hook must be silent on failure
   523→	}
   524→	head, headErr := repo.Head()
   525→	if headErr != nil {
   526→		return nil //nolint:nilerr // Hook must be silent on failure
   527→	}
   528→	currentHead := head.Hash().String()
   529→
   530→	// Find first matching session with PendingCheckpointID or LastCheckpointID to restore.
   531→	// PendingCheckpointID is set during ACTIVE_COMMITTED (deferred condensation).
   532→	// LastCheckpointID is set after condensation completes.
   533→	for _, state := range sessions {
   534→		if state.BaseCommit != currentHead {
   535→			continue
   536→		}
   537→		var cpID id.CheckpointID
   538→		source := ""
   539→
   540→		if state.PendingCheckpointID != "" {
   541→			if parsed, parseErr := id.NewCheckpointID(state.PendingCheckpointID); parseErr == nil {
   542→				cpID = parsed
   543→				source = "PendingCheckpointID"
   544→			}
   545→		}
   546→		if cpID.IsEmpty() && !state.LastCheckpointID.IsEmpty() {
   547→			cpID = state.LastCheckpointID
   548→			source = "LastCheckpointID"
   549→		}
   550→		if cpID.IsEmpty() {
   551→			continue
   552→		}
   553→
   554→		// Restore the trailer
   555→		message = addCheckpointTrailer(message, cpID)
   556→		if writeErr := os.WriteFile(commitMsgFile, []byte(message), 0o600); writeErr != nil {
   557→			return nil //nolint:nilerr // Hook must be silent on failure
   558→		}
   559→
   560→		logging.Info(logCtx, "prepare-commit-msg: restored trailer on amend",
   561→			slog.String("strategy", "manual-commit"),
   562→			slog.String("checkpoint_id", cpID.String()),
   563→			slog.String("session_id", state.SessionID),
   564→			slog.String("source", source),
   565→		)
   566→		return nil
   567→	}
   568→
   569→	// No checkpoint ID found - leave message unchanged
   570→	logging.Debug(logCtx, "prepare-commit-msg: amend with no checkpoint to restore",
   571→		slog.String("strategy", "manual-commit"),
   572→	)
   573→	return nil
   574→}

   576→// PostCommit is called by the git post-commit hook after a commit is created.
   577→// Uses the session state machine to determine what action to take per session:
   578→//   - ACTIVE → ACTIVE_COMMITTED: defer condensation (agent still working)
   579→//   - IDLE → condense immediately
   580→//   - ACTIVE_COMMITTED → migrate shadow branch (additional commit during same turn)
   581→//   - ENDED → condense if files touched, discard if empty
   582→//
   583→// Shadow branches are only deleted when ALL sessions sharing the branch are non-active.
   584→// During rebase/cherry-pick/revert operations, phase transitions are skipped entirely.
   585→//
   586→//nolint:unparam // error return required by interface but hooks must return nil
   587→func (s *ManualCommitStrategy) PostCommit() error {
   588→	logCtx := logging.WithComponent(context.Background(), "checkpoint")
   589→
   590→	repo, err := OpenRepository()
   591→	if err != nil {
   592→		return nil //nolint:nilerr // Hook must be silent on failure
   593→	}
   594→
   595→	// Get HEAD commit to check for trailer
   596→	head, err := repo.Head()
   597→	if err != nil {
   598→		return nil //nolint:nilerr // Hook must be silent on failure
   599→	}
   600→
   601→	commit, err := repo.CommitObject(head.Hash())
   602→	if err != nil {
   603→		return nil //nolint:nilerr // Hook must be silent on failure
   604→	}
   605→
   606→	// Check if commit has checkpoint trailer (ParseCheckpoint validates format)
   607→	checkpointID, found := trailers.ParseCheckpoint(commit.Message)
   608→	if !found {
   609→		// No trailer — user removed it or it was never added (mid-turn commit).
   610→		// Still update BaseCommit for active sessions so future commits can match.
   611→		s.postCommitUpdateBaseCommitOnly(logCtx, head)
   612→		return nil
   613→	}
   614→
   615→	worktreePath, err := GetWorktreePath()
   616→	if err != nil {
   617→		return nil //nolint:nilerr // Hook must be silent on failure
   618→	}
   619→
   620→	// Find all active sessions for this worktree
   621→	sessions, err := s.findSessionsForWorktree(worktreePath)
   622→	if err != nil || len(sessions) == 0 {
   623→		logging.Warn(logCtx, "post-commit: no active sessions despite trailer",
   624→			slog.String("strategy", "manual-commit"),
   625→			slog.String("checkpoint_id", checkpointID.String()),
   626→		)
   627→		return nil //nolint:nilerr // Intentional: hooks must be silent on failure
   628→	}
   629→
   630→	// Build transition context
   631→	isRebase := isGitSequenceOperation()
   632→	transitionCtx := session.TransitionContext{
   633→		IsRebaseInProgress: isRebase,
   634→	}
   635→
   636→	if isRebase {
   637→		logging.Debug(logCtx, "post-commit: rebase/sequence in progress, skipping phase transitions",
   638→			slog.String("strategy", "manual-commit"),
   639→		)
   640→	}
   641→
   642→	// Track shadow branch names and whether they can be deleted
   643→	shadowBranchesToDelete := make(map[string]struct{})
   644→	// Track sessions that are still active AFTER transitions
   645→	activeSessionsOnBranch := make(map[string]bool)
   646→
   647→	newHead := head.Hash().String()
   648→
   649→	// Two-pass processing: condensation first, migration second.
   650→	// This prevents a migration from renaming a shadow branch before another
   651→	// session sharing that branch has had a chance to condense from it.
   652→	type pendingMigration struct {
   653→		state *SessionState
   654→	}
   655→	var pendingMigrations []pendingMigration
   656→
   657→	// Pass 1: Run transitions and dispatch condensation/discard actions.
   658→	// Defer migration actions to pass 2.
   659→	for _, state := range sessions {
   660→		shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   661→
   662→		// Check for new content (needed for TransitionContext and condensation).
   663→		// Fail-open: if content check errors, assume new content exists so we
   664→		// don't silently skip data that should have been condensed.
   665→		hasNew, contentErr := s.sessionHasNewContent(repo, state)
   666→		if contentErr != nil {
   667→			hasNew = true
   668→			logging.Debug(logCtx, "post-commit: error checking session content, assuming new content",
   669→				slog.String("session_id", state.SessionID),
   670→				slog.String("error", contentErr.Error()),
   671→			)
   672→		}
   673→		transitionCtx.HasFilesTouched = len(state.FilesTouched) > 0
   674→
   675→		// Run the state machine transition
   676→		remaining := TransitionAndLog(state, session.EventGitCommit, transitionCtx)
   677→
   678→		// Dispatch strategy-specific actions.
   679→		// Each branch handles its own BaseCommit update so there is no
   680→		// fallthrough conditional at the end. On condensation failure,
   681→		// BaseCommit is intentionally NOT updated to preserve access to
   682→		// the shadow branch (which is named after the old BaseCommit).
   683→		for _, action := range remaining {
   684→			switch action {
   685→			case session.ActionCondense:
   686→				if hasNew {
   687→					s.condenseAndUpdateState(logCtx, repo, checkpointID, state, head, shadowBranchName, shadowBranchesToDelete)
   688→					// condenseAndUpdateState updates BaseCommit on success.
   689→					// On failure, BaseCommit is preserved so the shadow branch remains accessible.
   690→				} else {
   691→					// No new content to condense — just update BaseCommit
   692→					s.updateBaseCommitIfChanged(logCtx, state, newHead)
   693→				}
   694→			case session.ActionCondenseIfFilesTouched:
   695→				// The state machine already gates this action on HasFilesTouched,
   696→				// but hasNew is an additional content-level check (transcript has
   697→				// new content beyond what was previously condensed).
   698→				if len(state.FilesTouched) > 0 && hasNew {
   699→					s.condenseAndUpdateState(logCtx, repo, checkpointID, state, head, shadowBranchName, shadowBranchesToDelete)
   700→					// On failure, BaseCommit is preserved (same as ActionCondense).
   701→				} else {
   702→					s.updateBaseCommitIfChanged(logCtx, state, newHead)
   703→				}
   704→			case session.ActionDiscardIfNoFiles:
   705→				if len(state.FilesTouched) == 0 {
   706→					logging.Debug(logCtx, "post-commit: skipping empty ended session (no files to condense)",
   707→						slog.String("session_id", state.SessionID),
   708→					)
   709→				}
   710→				s.updateBaseCommitIfChanged(logCtx, state, newHead)
   711→			case session.ActionMigrateShadowBranch:
   712→				// Deferred to pass 2 so condensation reads the old shadow branch first.
   713→				// Migration updates BaseCommit as part of the rename.
   714→				// Store checkpointID so HandleTurnEnd can reuse it for deferred condensation.
   715→				state.PendingCheckpointID = checkpointID.String()
   716→				pendingMigrations = append(pendingMigrations, pendingMigration{state: state})
   717→			case session.ActionClearEndedAt, session.ActionUpdateLastInteraction:
   718→				// Handled by session.ApplyCommonActions above
   719→			case session.ActionWarnStaleSession:
   720→				// Not produced by EventGitCommit; listed for switch exhaustiveness
   721→			}
   722→		}
   723→
   724→		// Save the updated state
   725→		if err := s.saveSessionState(state); err != nil {
   726→			fmt.Fprintf(os.Stderr, "[entire] Warning: failed to update session state: %v\n", err)
   727→		}
   728→
   729→		// Track whether any session on this shadow branch is still active
   730→		if state.Phase.IsActive() {
   731→			activeSessionsOnBranch[shadowBranchName] = true
   732→		}
   733→	}
   734→
   735→	// Pass 2: Run deferred migrations now that all condensations are complete.
   736→	for _, pm := range pendingMigrations {
   737→		if _, migErr := s.migrateShadowBranchIfNeeded(repo, pm.state); migErr != nil {
   738→			logging.Warn(logCtx, "post-commit: shadow branch migration failed",
   739→				slog.String("session_id", pm.state.SessionID),
   740→				slog.String("error", migErr.Error()),
   741→			)
   742→		}
   743→		// Save the migrated state
   744→		if err := s.saveSessionState(pm.state); err != nil {
   745→			fmt.Fprintf(os.Stderr, "[entire] Warning: failed to update session state after migration: %v\n", err)
   746→		}
   747→	}
   748→
   749→	// Clean up shadow branches — only delete when ALL sessions on the branch are non-active
   750→	for shadowBranchName := range shadowBranchesToDelete {
   751→		if activeSessionsOnBranch[shadowBranchName] {
   752→			logging.Debug(logCtx, "post-commit: preserving shadow branch (active session exists)",
   753→				slog.String("shadow_branch", shadowBranchName),
   754→			)
   755→			continue
   756→		}
   757→		if err := deleteShadowBranch(repo, shadowBranchName); err != nil {
   758→			fmt.Fprintf(os.Stderr, "[entire] Warning: failed to clean up %s: %v\n", shadowBranchName, err)
   759→		} else {
   760→			fmt.Fprintf(os.Stderr, "[entire] Cleaned up shadow branch: %s\n", shadowBranchName)
   761→			logging.Info(logCtx, "shadow branch deleted",
   762→				slog.String("strategy", "manual-commit"),
   763→				slog.String("shadow_branch", shadowBranchName),
   764→			)
   765→		}
   766→	}
   767→
   768→	return nil
   769→}

   771→// condenseAndUpdateState runs condensation for a session and updates state afterward.
   772→// Returns true if condensation succeeded.
   773→func (s *ManualCommitStrategy) condenseAndUpdateState(
   774→	logCtx context.Context,
   775→	repo *git.Repository,
   776→	checkpointID id.CheckpointID,
   777→	state *SessionState,
   778→	head *plumbing.Reference,
   779→	shadowBranchName string,
   780→	shadowBranchesToDelete map[string]struct{},
   781→) {
   782→	result, err := s.CondenseSession(repo, checkpointID, state)
   783→	if err != nil {
   784→		fmt.Fprintf(os.Stderr, "[entire] Warning: condensation failed for session %s: %v\n",
   785→			state.SessionID, err)
   786→		logging.Warn(logCtx, "post-commit: condensation failed",
   787→			slog.String("session_id", state.SessionID),
   788→			slog.String("error", err.Error()),
   789→		)
   790→		return
   791→	}
   792→
   793→	// Track this shadow branch for cleanup
   794→	shadowBranchesToDelete[shadowBranchName] = struct{}{}
   795→
   796→	// Update session state for the new base commit
   797→	newHead := head.Hash().String()
   798→	state.BaseCommit = newHead
   799→	state.AttributionBaseCommit = newHead
   800→	state.StepCount = 0
   801→	state.CheckpointTranscriptStart = result.TotalTranscriptLines
   802→
   803→	// Clear attribution tracking — condensation already used these values
   804→	state.PromptAttributions = nil
   805→	state.PendingPromptAttribution = nil
   806→	state.FilesTouched = nil
   807→
   808→	// Save checkpoint ID so subsequent commits can reuse it
   809→	state.LastCheckpointID = checkpointID
   810→	// Clear PendingCheckpointID after condensation — it was used for deferred
   811→	// condensation (ACTIVE_COMMITTED flow) and should not persist. The amend
   812→	// handler uses LastCheckpointID instead.
   813→	state.PendingCheckpointID = ""
   814→
   815→	shortID := state.SessionID
   816→	if len(shortID) > 8 {
   817→		shortID = shortID[:8]
   818→	}
   819→	fmt.Fprintf(os.Stderr, "[entire] Condensed session %s: %s (%d checkpoints)\n",
   820→		shortID, result.CheckpointID, result.CheckpointsCount)

   822→	logging.Info(logCtx, "session condensed",
   823→		slog.String("strategy", "manual-commit"),
   824→		slog.String("checkpoint_id", result.CheckpointID.String()),
   825→		slog.Int("checkpoints_condensed", result.CheckpointsCount),
   826→		slog.Int("transcript_lines", result.TotalTranscriptLines),
   827→	)

   841→// postCommitUpdateBaseCommitOnly updates BaseCommit for all sessions on the current
   842→// worktree when a commit has no Entire-Checkpoint trailer. This prevents BaseCommit
   843→// from going stale, which would cause future PrepareCommitMsg calls to skip the
   844→// session (BaseCommit != currentHeadHash filter).
   845→//
   846→// Unlike the full PostCommit flow, this does NOT fire EventGitCommit or trigger
   847→// condensation — it only keeps BaseCommit in sync with HEAD.
   848→func (s *ManualCommitStrategy) postCommitUpdateBaseCommitOnly(logCtx context.Context, head *plumbing.Reference) {
   849→	worktreePath, err := GetWorktreePath()
   850→	if err != nil {
   851→		return // Silent failure — hooks must be resilient
   852→	}
   853→
   854→	sessions, err := s.findSessionsForWorktree(worktreePath)
   855→	if err != nil || len(sessions) == 0 {
   856→		return
   857→	}
   858→
   859→	newHead := head.Hash().String()
   860→	for _, state := range sessions {
   861→		// Only update active sessions. Idle/ended sessions are kept around for
   862→		// LastCheckpointID reuse and should not be advanced to HEAD.
   863→		if !state.Phase.IsActive() {
   864→			continue
   865→		}
   866→		if state.BaseCommit != newHead {
   867→			logging.Debug(logCtx, "post-commit (no trailer): updating BaseCommit",
   868→				slog.String("session_id", state.SessionID),
   869→				slog.String("old_base", truncateHash(state.BaseCommit)),
   870→				slog.String("new_head", truncateHash(newHead)),
   871→			)
   872→			state.BaseCommit = newHead
   873→			if err := s.saveSessionState(state); err != nil {
   874→				fmt.Fprintf(os.Stderr, "[entire] Warning: failed to update session state: %v\n", err)
   875→			}
   876→		}
   877→	}
   878→}

   908→// sessionHasNewContent checks if a session has new transcript content
   909→// beyond what was already condensed.
   910→func (s *ManualCommitStrategy) sessionHasNewContent(repo *git.Repository, state *SessionState) (bool, error) {
   911→	// Get shadow branch
   912→	shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   913→	refName := plumbing.NewBranchReferenceName(shadowBranchName)
   914→	ref, err := repo.Reference(refName, true)
   915→	if err != nil {
   916→		// No shadow branch means no Stop has happened since the last condensation.
   917→		// However, the agent may have done work (including commits) without a Stop.
   918→		// Check the live transcript to detect this scenario.
   919→		return s.sessionHasNewContentFromLiveTranscript(repo, state)
   920→	}
   921→
   922→	commit, err := repo.CommitObject(ref.Hash())
   923→	if err != nil {
   924→		return false, fmt.Errorf("failed to get commit object: %w", err)
   925→	}
   926→
   927→	tree, err := commit.Tree()
   928→	if err != nil {
   929→		return false, fmt.Errorf("failed to get commit tree: %w", err)
   930→	}
   931→
   932→	// Look for transcript file
   933→	metadataDir := paths.EntireMetadataDir + "/" + state.SessionID
   934→	var transcriptLines int
   935→
   936→	if file, fileErr := tree.File(metadataDir + "/" + paths.TranscriptFileName); fileErr == nil {
   937→		if content, contentErr := file.Contents(); contentErr == nil {
   938→			transcriptLines = countTranscriptItems(state.AgentType, content)
   939→		}
   940→	} else if file, fileErr := tree.File(metadataDir + "/" + paths.TranscriptFileNameLegacy); fileErr == nil {
   941→		if content, contentErr := file.Contents(); contentErr == nil {
   942→			transcriptLines = countTranscriptItems(state.AgentType, content)
   943→		}
   944→	}
   945→
   946→	// Has new content if there are more lines than already condensed
   947→	return transcriptLines > state.CheckpointTranscriptStart, nil
   948→}

   950→// sessionHasNewContentFromLiveTranscript checks if a session has new content
   951→// by examining the live transcript file. This is used when no shadow branch exists
   952→// (i.e., no Stop has happened yet) but the agent may have done work.
   953→//
   954→// Returns true if:
   955→//  1. The transcript has grown since the last condensation, AND
   956→//  2. The new transcript portion contains file modifications, AND
   957→//  3. At least one modified file overlaps with the currently staged files
   958→//
   959→// The overlap check ensures we don't add checkpoint trailers to commits that are
   960→// unrelated to the agent's recent changes.
   961→//
   962→// This handles the scenario where the agent commits mid-session before Stop.
   963→func (s *ManualCommitStrategy) sessionHasNewContentFromLiveTranscript(repo *git.Repository, state *SessionState) (bool, error) {
   964→	logCtx := logging.WithComponent(context.Background(), "checkpoint")
   965→
   966→	// Need both transcript path and agent type to analyze
   967→	if state.TranscriptPath == "" || state.AgentType == "" {
   968→		logging.Debug(logCtx, "live transcript check: missing transcript path or agent type",
   969→			slog.String("session_id", state.SessionID),
   970→			slog.String("transcript_path", state.TranscriptPath),
   971→			slog.String("agent_type", string(state.AgentType)),
   972→		)
   973→		return false, nil
   974→	}
   975→
   976→	// Get the agent for transcript analysis
   977→	ag, err := agent.GetByAgentType(state.AgentType)
   978→	if err != nil {
   979→		return false, nil //nolint:nilerr // Unknown agent type, fail gracefully
   980→	}
   981→
   982→	// Cast to TranscriptAnalyzer
   983→	analyzer, ok := ag.(agent.TranscriptAnalyzer)
   984→	if !ok {
   985→		return false, nil // Agent doesn't support transcript analysis
   986→	}
   987→
   988→	// Get current transcript position
   989→	currentPos, err := analyzer.GetTranscriptPosition(state.TranscriptPath)
   990→	if err != nil {
   991→		return false, nil //nolint:nilerr // Error reading transcript, fail gracefully
   992→	}
   993→
   994→	// Check if transcript has grown since last condensation
   995→	if currentPos <= state.CheckpointTranscriptStart {
   996→		logging.Debug(logCtx, "live transcript check: no new content",
   997→			slog.String("session_id", state.SessionID),
   998→			slog.Int("current_pos", currentPos),
   999→			slog.Int("start_offset", state.CheckpointTranscriptStart),
  1000→		)
  1001→		return false, nil // No new content
  1002→	}
  1003→
  1004→	// Transcript has grown - check if there are file modifications in the new portion
  1005→	modifiedFiles, _, err := analyzer.ExtractModifiedFilesFromOffset(state.TranscriptPath, state.CheckpointTranscriptStart)
  1006→	if err != nil {
  1007→		return false, nil //nolint:nilerr // Error parsing transcript, fail gracefully
  1008→	}
  1009→
  1010→	// No file modifications means no new content to checkpoint
  1011→	if len(modifiedFiles) == 0 {
  1012→		logging.Debug(logCtx, "live transcript check: transcript grew but no file modifications",
  1013→			slog.String("session_id", state.SessionID),
  1014→		)
  1015→		return false, nil
  1016→	}
  1017→
  1018→	// Normalize modified files from absolute to repo-relative paths.
  1019→	// Transcript tool_use entries contain absolute paths (e.g., /Users/alex/project/src/main.go)
  1020→	// but getStagedFiles returns repo-relative paths (e.g., src/main.go).
  1021→	// Use state.WorktreePath (already resolved) to avoid an extra git subprocess.
  1022→	basePath := state.WorktreePath
  1023→	if basePath == "" {
  1024→		if wp, wpErr := GetWorktreePath(); wpErr == nil {
  1025→			basePath = wp
  1026→		}
  1027→	}
  1028→	if basePath != "" {
  1029→		normalized := make([]string, 0, len(modifiedFiles))
  1030→		for _, f := range modifiedFiles {
  1031→			if rel := paths.ToRelativePath(f, basePath); rel != "" {
  1032→				normalized = append(normalized, rel)
  1033→			} else {
  1034→				// Already relative or outside repo — keep as-is
  1035→				normalized = append(normalized, f)
  1036→			}
  1037→		}
  1038→		modifiedFiles = normalized
  1039→	}
  1040→
  1041→	logging.Debug(logCtx, "live transcript check: found file modifications",
  1042→		slog.String("session_id", state.SessionID),
  1043→		slog.Int("modified_files", len(modifiedFiles)),
  1044→	)
  1045→
  1046→	// Check if any modified files overlap with currently staged files
  1047→	// This ensures we only add checkpoint trailers to commits that include
  1048→	// files the agent actually modified
  1049→	stagedFiles := getStagedFiles(repo)
  1050→
  1051→	logging.Debug(logCtx, "live transcript check: comparing staged vs modified",
  1052→		slog.String("session_id", state.SessionID),
  1053→		slog.Int("staged_files", len(stagedFiles)),
  1054→		slog.Int("modified_files", len(modifiedFiles)),
  1055→	)
  1056→
  1057→	if !hasOverlappingFiles(stagedFiles, modifiedFiles) {
  1058→		logging.Debug(logCtx, "live transcript check: no overlap between staged and modified files",
  1059→			slog.String("session_id", state.SessionID),
  1060→		)
  1061→		return false, nil // No overlap - staged files are unrelated to agent's work
  1062→	}
  1063→
  1064→	return true, nil
  1065→}

  1067→// addTrailerForAgentCommit handles the fast path when an agent is committing
  1068→// (ACTIVE session + no TTY). Generates a checkpoint ID and adds the trailer
  1069→// directly, bypassing content detection and interactive prompts.
  1070→func (s *ManualCommitStrategy) addTrailerForAgentCommit(logCtx context.Context, commitMsgFile string, state *SessionState, source string) error {
  1071→	// Use PendingCheckpointID if set, otherwise generate a new one
  1072→	var cpID id.CheckpointID
  1073→	if state.PendingCheckpointID != "" {
  1074→		var err error
  1075→		cpID, err = id.NewCheckpointID(state.PendingCheckpointID)
  1076→		if err != nil {
  1077→			cpID = "" // fall through to generate
  1078→		}
  1079→	}
  1080→	if cpID.IsEmpty() {
  1081→		var err error
  1082→		cpID, err = id.Generate()
  1083→		if err != nil {
  1084→			return nil //nolint:nilerr // Hook must be silent on failure
  1085→		}
  1086→	}
  1087→

  1089→	if err != nil {
  1090→		return nil //nolint:nilerr // Hook must be silent on failure
  1091→	}
  1092→

  1094→

  1097→		return nil

  1099→

  1104→		slog.String("source", source),
  1105→		slog.String("checkpoint_id", cpID.String()),
  1106→		slog.String("session_id", state.SessionID),
  1107→	)

  1110→		return nil //nolint:nilerr // Hook must be silent on failure
  1111→	}
  1112→	return nil
  1113→}

  1115→// addCheckpointTrailer adds the Entire-Checkpoint trailer to a commit message.
  1116→// Handles proper trailer formatting (blank line before trailers if needed).
  1117→func addCheckpointTrailer(message string, checkpointID id.CheckpointID) string {
  1118→	trailer := trailers.CheckpointTrailerKey + ": " + checkpointID.String()
  1119→
  1120→	// If message already ends with trailers (lines starting with key:), just append
  1121→	// Otherwise, add a blank line first
  1122→	lines := strings.Split(strings.TrimRight(message, "\n"), "\n")
  1123→
  1124→	// Check if last non-empty, non-comment line looks like a trailer
  1125→	// Git comment lines start with # and may contain ": " (e.g., "# Changes to be committed:")
  1126→	hasTrailers := false
  1127→	for i := len(lines) - 1; i >= 0; i-- {
  1128→		line := strings.TrimSpace(lines[i])
  1129→		if line == "" {
  1130→			break
  1131→		}
  1132→		// Skip git comment lines
  1133→		if strings.HasPrefix(line, "#") {
  1134→			continue
  1135→		}
  1136→		if strings.Contains(line, ": ") {
  1137→			hasTrailers = true
  1138→			break
  1139→		}
  1140→		// Non-comment, non-trailer line found - no existing trailers
  1141→		break
  1142→	}
  1143→
  1144→	if hasTrailers {
  1145→		// Append trailer directly
  1146→		return strings.TrimRight(message, "\n") + "\n" + trailer + "\n"
  1147→	}
  1148→
  1149→	// Add blank line before trailer
  1150→	return strings.TrimRight(message, "\n") + "\n\n" + trailer + "\n"
  1151→}

  1153→// addCheckpointTrailerWithComment adds the Entire-Checkpoint trailer with an explanatory comment.
  1154→// The trailer is placed above the git comment block but below the user's message area,
  1155→// with a comment explaining that the user can remove it if they don't want to link the commit
  1156→// to the agent session. If prompt is non-empty, it's shown as context.
  1157→func addCheckpointTrailerWithComment(message string, checkpointID id.CheckpointID, agentName, prompt string) string {
  1158→	trailer := trailers.CheckpointTrailerKey + ": " + checkpointID.String()
  1159→	commentLines := []string{
  1160→		"# Remove the Entire-Checkpoint trailer above if you don't want to link this commit to " + agentName + " session context.",
  1161→	}
  1162→	if prompt != "" {
  1163→		commentLines = append(commentLines, "# Last Prompt: "+prompt)
  1164→	}
  1165→	commentLines = append(commentLines, "# The trailer will be added to your next commit based on this branch.")
  1166→	comment := strings.Join(commentLines, "\n")
  1167→
  1168→	lines := strings.Split(message, "\n")
  1169→
  1170→	// Find where the git comment block starts (first # line)
  1171→	commentStart := -1
  1172→	for i, line := range lines {
  1173→		if strings.HasPrefix(line, "#") {
  1174→			commentStart = i
  1175→			break
  1176→		}
  1177→	}
  1178→
  1179→	if commentStart == -1 {
  1180→		// No git comments, append trailer at the end
  1181→		return strings.TrimRight(message, "\n") + "\n\n" + trailer + "\n" + comment + "\n"
  1182→	}
  1183→
  1184→	// Split into user content and git comments
  1185→	userContent := strings.Join(lines[:commentStart], "\n")
  1186→	gitComments := strings.Join(lines[commentStart:], "\n")
  1187→
  1188→	// Build result: user content, blank line, trailer, comment, blank line, git comments
  1189→	userContent = strings.TrimRight(userContent, "\n")
  1190→	if userContent == "" {
  1191→		// No user content yet - leave space for them to type, then trailer
  1192→		// Two newlines: first for user's message line, second for blank separator
  1193→		return "\n\n" + trailer + "\n" + comment + "\n\n" + gitComments
  1194→	}
  1195→	return userContent + "\n\n" + trailer + "\n" + comment + "\n\n" + gitComments
  1196→}

  1198→// InitializeSession creates session state for a new session or updates an existing one.
  1199→// This implements the optional SessionInitializer interface.
  1200→// Called during UserPromptSubmit to allow git hooks to detect active sessions.
  1201→//
  1202→// If the session already exists and HEAD has moved (e.g., user committed), updates
  1203→// BaseCommit to the new HEAD so future checkpoints go to the correct shadow branch.
  1204→//
  1205→// If there's an existing shadow branch with commits from a different session ID,
  1206→// returns a SessionIDConflictError to prevent orphaning existing session work.
  1207→//
  1208→// agentType is the human-readable name of the agent (e.g., "Claude Code").
  1209→// transcriptPath is the path to the live transcript file (for mid-session commit detection).
  1210→// userPrompt is the user's prompt text (stored truncated as FirstPrompt for display).
  1211→func (s *ManualCommitStrategy) InitializeSession(sessionID string, agentType agent.AgentType, transcriptPath string, userPrompt string) error {
  1212→	repo, err := OpenRepository()
  1213→	if err != nil {
  1214→		return fmt.Errorf("failed to open git repository: %w", err)
  1215→	}
  1216→
  1217→	// Check if session already exists
  1218→	state, err := s.loadSessionState(sessionID)
  1219→	if err != nil {
  1220→		return fmt.Errorf("failed to check session state: %w", err)
  1221→	}
  1222→
  1223→	if state != nil && state.BaseCommit != "" {
  1224→		// Session is fully initialized — apply phase transition for TurnStart
  1225→		TransitionAndLog(state, session.EventTurnStart, session.TransitionContext{})
  1226→
  1227→		// Backfill AgentType if empty or set to the generic default "Agent"
  1228→		if !isSpecificAgentType(state.AgentType) && agentType != "" {
  1229→			state.AgentType = agentType
  1230→		}
  1231→
  1232→		// Backfill FirstPrompt if empty (for sessions created before the first_prompt field was added)
  1233→		if state.FirstPrompt == "" && userPrompt != "" {
  1234→			state.FirstPrompt = truncatePromptForStorage(userPrompt)
  1235→		}
  1236→
  1237→		// Update transcript path if provided (may change on session resume)
  1238→		if transcriptPath != "" && state.TranscriptPath != transcriptPath {
  1239→			state.TranscriptPath = transcriptPath
  1240→		}
  1241→
  1242→		// Clear checkpoint IDs on every new prompt
  1243→		// These are set during PostCommit when a checkpoint is created, and should be
  1244→		// cleared when the user enters a new prompt (starting fresh work)
  1245→		if state.LastCheckpointID != "" {
  1246→			state.LastCheckpointID = ""
  1247→		}
  1248→		if state.PendingCheckpointID != "" {
  1249→			state.PendingCheckpointID = ""
  1250→		}
  1251→
  1252→		// Calculate attribution at prompt start (BEFORE agent makes any changes)
  1253→		// This captures user edits since the last checkpoint (or base commit for first prompt).
  1254→		// IMPORTANT: Always calculate attribution, even for the first checkpoint, to capture
  1255→		// user edits made before the first prompt. The inner CalculatePromptAttribution handles
  1256→		// nil lastCheckpointTree by falling back to baseTree.
  1257→		promptAttr := s.calculatePromptAttributionAtStart(repo, state)
  1258→		state.PendingPromptAttribution = &promptAttr
  1259→
  1260→		// Check if HEAD has moved (user pulled/rebased or committed)
  1261→		// migrateShadowBranchIfNeeded handles renaming the shadow branch and updating state.BaseCommit
  1262→		if _, err := s.migrateShadowBranchIfNeeded(repo, state); err != nil {
  1263→			return fmt.Errorf("failed to check/migrate shadow branch: %w", err)
  1264→		}
  1265→
  1266→		if err := s.saveSessionState(state); err != nil {
  1267→			return fmt.Errorf("failed to update session state: %w", err)
  1268→		}
  1269→		return nil
  1270→	}
  1271→	// If state exists but BaseCommit is empty, it's a partial state from concurrent warning
  1272→	// Continue below to properly initialize it
  1273→
  1274→	// Initialize new session
  1275→	state, err = s.initializeSession(repo, sessionID, agentType, transcriptPath, userPrompt)
  1276→	if err != nil {
  1277→		return fmt.Errorf("failed to initialize session: %w", err)
  1278→	}
  1279→
  1280→	// Apply phase transition: new session starts as ACTIVE
  1281→	TransitionAndLog(state, session.EventTurnStart, session.TransitionContext{})
  1282→
  1283→	// Calculate attribution for pre-prompt edits
  1284→	// This captures any user edits made before the first prompt
  1285→	promptAttr := s.calculatePromptAttributionAtStart(repo, state)
  1286→	state.PendingPromptAttribution = &promptAttr
  1287→	if err = s.saveSessionState(state); err != nil {
  1288→		return fmt.Errorf("failed to save attribution: %w", err)
  1289→	}
  1290→
  1291→	fmt.Fprintf(os.Stderr, "Initialized shadow session: %s\n", sessionID)
  1292→	return nil
  1293→}

  1295→// calculatePromptAttributionAtStart calculates attribution at prompt start (before agent runs).
  1296→// This captures user changes since the last checkpoint - no filtering needed since
  1297→// the agent hasn't made any changes yet.
  1298→//
  1299→// IMPORTANT: This reads from the worktree (not staging area) to match what WriteTemporary
  1300→// captures in checkpoints. If we read staged content but checkpoints capture worktree content,
  1301→// unstaged changes would be in the checkpoint but not counted in PromptAttribution, causing
  1302→// them to be incorrectly attributed to the agent later.
  1303→func (s *ManualCommitStrategy) calculatePromptAttributionAtStart(
  1304→	repo *git.Repository,
  1305→	state *SessionState,
  1306→) PromptAttribution {
  1307→	logCtx := logging.WithComponent(context.Background(), "attribution")
  1308→	nextCheckpointNum := state.StepCount + 1
  1309→	result := PromptAttribution{CheckpointNumber: nextCheckpointNum}
  1310→

**`.github/workflows/sync-fork-checkpoints.yml`** — A sample workflow showing how to use it.

**How it works:**

1. Triggers when a fork PR is merged
2. Scans the merged commits for `Entire-Checkpoint: <id>` trailers
3. Falls back to checking the original PR commits (handles squash merges that strip trailers)
4. Fetches `entire/checkpoints/v1` from the fork's clone URL
5. **Selectively imports** only the checkpoint directories referenced by those IDs (sharded path `<id[:2]>/<id[2:]>/`), not the entire branch
6. Commits and pushes to upstream's `entire/checkpoints/v1`
7. Retry logic handles concurrent PR merges racing on the push

**Key design decisions:**
- **Selective import** — only copies checkpoint directories for IDs found in the PR, not the fork's entire branch (avoids importing unrelated session data)
- **Idempotent** — exits cleanly if checkpoints already exist on upstream
- **Graceful degradation** — exits cleanly if fork has no checkpoints branch, or if no trailers are found
- **No contributor permissions needed** — runs on the upstream repo's CI, fetching from the fork's public URL

**Limitation:** For private forks, you'd pass a PAT via the `token` input that has read access to the fork.

> DEVELOPER

how would we get that PAT?

> AGENT

Two options, from simplest to most proper:

**Option 1: Fine-grained PAT (quick)**

1. Go to GitHub > Settings > Developer settings > Personal access tokens > Fine-grained tokens
2. Generate new token:
   - **Resource owner**: your org (entireio)
   - **Repository access**: "Only select repositories" — pick the repos that need it
   - **Permissions**: Contents → Read and write (nothing else)
3. Add it as a repo/org secret: Settings > Secrets > Actions > `CHECKPOINTS_PAT`

Downside: tied to a personal account. If that person leaves the org, the token dies.

**Option 2: GitHub App (proper for orgs)**

1. Org Settings > Developer settings > GitHub Apps > New GitHub App
2. Give it only **Contents: Read and write** permission
3. Install it on the repos that need it
4. Store the App ID and private key as org secrets (`APP_ID`, `APP_PRIVATE_KEY`)
5. In the workflow, generate a short-lived token:

```yaml
steps:
  - uses: actions/create-github-app-token@v1
    id: app-token
    with:
      app-id: ${{ secrets.APP_ID }}
      private-key: ${{ secrets.APP_PRIVATE_KEY }}

  - uses: entireio/cli/.github/actions/sync-fork-checkpoints@main
    with:
      token: ${{ steps.app-token.outputs.token }}
```

The GitHub App approach is better for orgs — tokens are short-lived (1 hour), not tied to any person, and have an audit trail. Which way do you want to go?
