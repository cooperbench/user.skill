   448→// extractUserPromptsFromLines extracts user prompts from JSONL transcript lines.
   449→// IDE-injected context tags (like <ide_opened_file>) are stripped from the results.
   450→func extractUserPromptsFromLines(lines []string) []string {
   451→	var prompts []string
   452→	for _, line := range lines {
   453→		line = strings.TrimSpace(line)
   454→		if line == "" {
   455→			continue
   456→		}
   457→
   458→		var entry map[string]interface{}
   459→		if err := json.Unmarshal([]byte(line), &entry); err != nil {
   460→			continue
   461→		}
   462→
   463→		// Check for user message (supports both "human" and "user" types)
   464→		msgType, ok := entry["type"].(string)
   465→		if !ok || (msgType != "human" && msgType != "user") {
   466→			continue
   467→		}
   468→
   469→		// Extract message content
   470→		message, ok := entry["message"].(map[string]interface{})
   471→		if !ok {
   472→			continue
   473→		}
   474→
   475→		// Handle string content
   476→		if content, ok := message["content"].(string); ok && content != "" {
   477→			cleaned := textutil.StripIDEContextTags(content)
   478→			if cleaned != "" {
   479→				prompts = append(prompts, cleaned)
   480→			}
   481→			continue
   482→		}
   483→
   484→		// Handle array content (e.g., multiple text blocks from VSCode)
   485→		if arr, ok := message["content"].([]interface{}); ok {
   486→			var texts []string
   487→			for _, item := range arr {
   488→				if m, ok := item.(map[string]interface{}); ok {
   489→					if m["type"] == "text" {
   490→						if text, ok := m["text"].(string); ok {
   491→							texts = append(texts, text)
   492→						}
   493→					}
   494→				}
   495→			}
   496→			if len(texts) > 0 {
   497→				cleaned := textutil.StripIDEContextTags(strings.Join(texts, "\n\n"))
   498→				if cleaned != "" {
   499→					prompts = append(prompts, cleaned)
   500→				}
   501→			}
   502→		}
   503→	}
   504→	return prompts
   505→}
