# Fix: Wingman auto-apply never triggers on session close + improve logging

## Context

When a user closes a Claude session with a pending `REVIEW.md`, the auto-apply never fires. The root cause is a phase check bug in the auto-apply subprocess: `runWingmanApply` calls `isSessionIdle()` which only returns `true` for `PhaseIdle`. But when triggered from the SessionEnd hook, the session has already been marked `PhaseEnded`, so `isSessionIdle` returns `false` and the subprocess aborts with "session became active during spawn."

Additionally, stale session state files (like orphaned `test.json` with `phase: active_committed`) cause `hasAnyLiveSession()` to return `true` perpetually, blocking auto-apply from even being attempted.

## Changes

### 1. Fix phase check in `runWingmanApply` (`wingman_review.go:497-503`)

Replace the `isSessionIdle` check with a direct phase check that only aborts when the session is truly busy (ACTIVE or ACTIVE_COMMITTED). IDLE and ENDED both mean "safe to auto-apply."

```go
// Re-check session hasn't become active (user may have typed during spawn delay).
// IDLE and ENDED are safe — only ACTIVE/ACTIVE_COMMITTED should block.
if state != nil && state.SessionID != "" {
    sessDir := findSessionStateDir(repoRoot)
    if sessDir != "" {
        phase := readSessionPhase(sessDir, state.SessionID)
        if session.Phase(phase).IsActive() {
            wingmanLog("session is active (phase=%s), aborting (next stop hook will retry)", phase)
            return nil
        }
        wingmanLog("session phase=%s, safe to proceed", phase)
    }
}
```

Uses existing `session.Phase.IsActive()` helper which returns true for ACTIVE/ACTIVE_COMMITTED only.

### 2. Add staleness check to `hasAnyLiveSession` (`wingman_review.go:386-421`)

Skip session state files that haven't been updated in over 2 hours. This prevents stale/orphaned sessions from permanently blocking auto-apply.

```go
const staleSessionThreshold = 2 * time.Hour

// In the for loop, after reading the entry:
info, statErr := entry.Info()
if statErr != nil || time.Since(info.ModTime()) > staleSessionThreshold {
    continue // Skip stale/unreadable sessions
}
```

### 3. Improve logging throughout the auto-apply flow

**In `triggerWingmanAutoApplyIfPending` (~line 842):** Add more detailed logging for the skip reasons — currently `hasAnyLiveSession` returns true silently when there are stale sessions.

**In `triggerWingmanAutoApplyOnSessionEnd` (~line 874):** Add logging when conditions aren't met (wingman not enabled, no review pending, etc.) so silent returns are diagnosable.

**In `spawnDetachedWingmanReview` and `spawnDetachedWingmanApply`:** Log the PID of the spawned process (or failure) to stderr and the structured log.

**In `runWingmanApply`:** Log each decision point (review exists, state loaded, phase check, etc.) with `wingmanLog`.

   835→// triggerWingmanAutoApplyIfPending checks for a pending REVIEW.md and spawns
   836→// the auto-apply subprocess if conditions are met. Called from the stop hook
   837→// on every turn end (both with-changes and no-changes paths).
   838→//
   839→// When a live session exists, this is a no-op: the prompt-submit injection
   840→// will deliver the review visibly in the user's terminal instead. Background
   841→// auto-apply is only used when no sessions are alive (all ended).
   842→func triggerWingmanAutoApplyIfPending(repoRoot string) {
   843→	if !settings.IsWingmanEnabled() || os.Getenv("ENTIRE_WINGMAN_APPLY") != "" {
   844→		return
   845→	}
   846→	logCtx := logging.WithComponent(context.Background(), "wingman")
   847→	reviewPath := filepath.Join(repoRoot, wingmanReviewFile)
   848→	if _, statErr := os.Stat(reviewPath); statErr != nil {
   849→		return
   850→	}
   851→	wingmanState := loadWingmanStateDirect(repoRoot)
   852→	if wingmanState != nil && wingmanState.ApplyAttemptedAt != nil {
   853→		logging.Debug(logCtx, "wingman auto-apply already attempted, skipping")
   854→		return
   855→	}
   856→	// Don't spawn background auto-apply if a live session exists.
   857→	// The prompt-submit hook will inject REVIEW.md as additionalContext,
   858→	// which is visible to the user in their terminal.
   859→	if hasAnyLiveSession(repoRoot) {
   860→		logging.Debug(logCtx, "wingman auto-apply deferred: live session will handle via injection")
   861→		fmt.Fprintf(os.Stderr, "[wingman] Review pending — will be injected on next prompt\n")
   862→		return
   863→	}
   864→	fmt.Fprintf(os.Stderr, "[wingman] Pending review found, spawning auto-apply (no live sessions)\n")
   865→	logging.Info(logCtx, "wingman auto-apply spawning (no live sessions)",
   866→		slog.String("review_path", reviewPath),
   867→	)
   868→	spawnDetachedWingmanApply(repoRoot)
   869→}

   893→		return
   894→	}

   896→	logging.Info(logCtx, "wingman auto-apply on session-end (no live sessions remain)",
   897→		slog.String("review_path", reviewPath),

   980→		offset = 0

   983→	if _, err := f.ReadAt(buf, offset); err != nil {
   984→		return false
   985→	}

   988→	lines := strings.Split(string(buf), "\n")

  1013→		}

   196→			NewPhase: PhaseIdle,
   197→			Actions:  []Action{ActionUpdateLastInteraction},

   292→		return TransitionResult{NewPhase: PhaseEnded}

   298→// ApplyCommonActions applies the common (non-strategy-specific) actions from a
   299→// TransitionResult to the given State. It updates Phase, LastInteractionTime,
   300→// and EndedAt as indicated by the transition.
   301→//
   302→// Returns the subset of actions that require strategy-specific handling
   303→// (e.g., ActionCondense, ActionMigrateShadowBranch, ActionWarnStaleSession).
   304→// The caller is responsible for dispatching those.
   305→func ApplyCommonActions(state *State, result TransitionResult) []Action {
   306→	state.Phase = result.NewPhase
   307→
   308→	var remaining []Action
   309→	for _, action := range result.Actions {
   310→		switch action {
   311→		case ActionUpdateLastInteraction:
   312→			now := time.Now()
   313→			state.LastInteractionTime = &now
   314→		case ActionClearEndedAt:
   315→			state.EndedAt = nil
   316→		case ActionCondense, ActionCondenseIfFilesTouched, ActionDiscardIfNoFiles,
   317→			ActionMigrateShadowBranch, ActionWarnStaleSession:
   318→			// Strategy-specific actions — pass through to caller.
   319→			remaining = append(remaining, action)
   320→		}
   321→	}
   322→	return remaining
   323→}

   325→// MermaidDiagram generates a Mermaid state diagram from the transition table.
   326→// The diagram is derived by calling Transition() for representative context
   327→// variants per phase/event, so it stays in sync with the implementation.
   328→func MermaidDiagram() string {
   329→	var b strings.Builder
   330→	b.WriteString("stateDiagram-v2\n")
   331→
   332→	// State declarations with descriptions.
   333→	b.WriteString("    state \"IDLE\" as idle\n")
   334→	b.WriteString("    state \"ACTIVE\" as active\n")
   335→	b.WriteString("    state \"ACTIVE_COMMITTED\" as active_committed\n")
   336→	b.WriteString("    state \"ENDED\" as ended\n")
   337→	b.WriteString("\n")
   338→
   339→	// Context variants for GitCommit: rebase, files/no-files.
   340→	type contextVariant struct {
   341→		label string
   342→		ctx   TransitionContext
   343→	}
   344→
   345→	// For each phase/event, generate edges.
   346→	for _, phase := range allPhases {

   349→			switch {

   352→				// other phases produce identical edges for both values.
   353→				variants = []contextVariant{
   354→					{"[files]", TransitionContext{HasFilesTouched: true}},
   355→					{"[no files]", TransitionContext{HasFilesTouched: false}},
   356→					{"[rebase]", TransitionContext{IsRebaseInProgress: true}},
   357→				}
   358→			case event == EventGitCommit:
   359→				variants = []contextVariant{
   360→					{"", TransitionContext{}},
   361→					{"[rebase]", TransitionContext{IsRebaseInProgress: true}},
   362→				}
   363→			default:
   364→				variants = []contextVariant{
   365→					{"", TransitionContext{}},
   366→				}
   367→			}
   368→
   369→			for _, v := range variants {
   370→				result := Transition(phase, event, v.ctx)
   371→
   372→				// Build edge label.
   373→				label := event.String()
   374→				if v.label != "" {
   375→					label += " " + v.label
   376→				}
   377→				if len(result.Actions) > 0 {
   378→					actionNames := make([]string, 0, len(result.Actions))
   379→					for _, a := range result.Actions {
   380→						actionNames = append(actionNames, a.String())
   381→					}
   382→					label += " / " + strings.Join(actionNames, ", ")
   383→				}
   384→
   385→				from := string(phase)
   386→				to := string(result.NewPhase)
   387→
   388→				fmt.Fprintf(&b, "    %s --> %s : %s\n", from, to, label)
   389→			}
   390→		}
   391→	}
   392→
   393→	return b.String()
   394→}

// triggerWingmanFromCommit builds a wingman payload from the HEAD commit and
// triggers a review. Used by the git post-commit hook for manual-commit strategy
// where files are committed by the user (not by SaveChanges).
func triggerWingmanFromCommit() {
	// Prevent infinite recursion: skip if inside wingman auto-apply
	if os.Getenv("ENTIRE_WINGMAN_APPLY") != "" {
		return
	}
	if !settings.IsWingmanEnabled() {
		return
	}

	repoRoot, err := paths.RepoRoot()
	if err != nil {
		return
	}

	head := resolveHEAD(repoRoot)
	if head == "" {
		return
	}

	// Get changed files from the commit
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	//nolint:gosec // G204: head is from git rev-parse, not user input
	cmd := exec.CommandContext(ctx, "git", "diff-tree", "--no-commit-id", "--name-status", "-r", head)
	cmd.Dir = repoRoot
	out, err := cmd.Output()
	if err != nil || len(out) == 0 {
		return
	}

	var modified, newFiles, deleted []string
	for _, line := range strings.Split(strings.TrimSpace(string(out)), "\n") {
		if len(line) < 3 {
			continue
		}
		status := line[0]
		file := strings.TrimSpace(line[1:])
		switch status {
		case 'M':
			modified = append(modified, file)
		case 'A':
			newFiles = append(newFiles, file)
		case 'D':
			deleted = append(deleted, file)
		}
	}

	if len(modified)+len(newFiles)+len(deleted) == 0 {
		return
	}

	// Get commit message
	//nolint:gosec // G204: head is from git rev-parse, not user input
	msgCmd := exec.CommandContext(ctx, "git", "log", "-1", "--format=%B", head)
	msgCmd.Dir = repoRoot
	msgOut, _ := msgCmd.Output() //nolint:errcheck // best-effort commit message
	commitMessage := strings.TrimSpace(string(msgOut))

	sessionID := strategy.FindMostRecentSession()

	triggerWingmanReview(WingmanPayload{
		SessionID:     sessionID,
		RepoRoot:      repoRoot,
		ModifiedFiles: modified,
		NewFiles:      newFiles,
		DeletedFiles:  deleted,
		CommitMessage: commitMessage,
	})
}
