# Plan: Simplify `entire enable` Onboarding Flow

## Context

The current `entire enable` flow is verbose and implementation-focused (mentions strategies, individual hook types, directory creation). The goal is to make it clean and user-friendly:

- **First run**: Shows agent, installs everything, asks telemetry, says "Ready."
- **Second run**: Detects already-enabled state and shows a brief summary instead of re-running setup.

## Target Output

**First run:**
```
entire enable
Agent: Claude Code (use --agent to change)

✓ Hooks installed
✓ Project configured (.entire/settings.json)

  Help improve Entire CLI?
  Share anonymous usage data. No code or personal info collected.
  [Yes] [No]

Ready.
```

**Second run (already fully enabled):**
```
entire enable
Already enabled. Everything looks good.

  Agent: Claude Code
  Config: .entire/settings.json
```

#### 1. Add `isFullyEnabled()` helper function

   678→// shellCompletionTarget returns the rc file path and completion lines for the

   680→func shellCompletionTarget() (shellName, rcFile, completionLine string, err error) {
   681→	home, err := os.UserHomeDir()

   691→			"source <(entire completion zsh)",
   692→			nil
   693→	case strings.Contains(shell, "bash"):
   694→		bashRC := filepath.Join(home, ".bashrc")
   695→		if _, err := os.Stat(filepath.Join(home, ".bash_profile")); err == nil {
   696→			bashRC = filepath.Join(home, ".bash_profile")
   697→		}
   698→		return "Bash",
   699→			bashRC,
   700→			"source <(entire completion bash)",
   701→			nil
   702→	default:
   703→		return "", "", "", errUnsupportedShell
   704→	}

   825→// runUninstall completely removes Entire from the repository.
   826→func runUninstall(w, errW io.Writer, force bool) error {
   827→	// Check if we're in a git repository
   828→	if _, err := paths.RepoRoot(); err != nil {
   829→		fmt.Fprintln(errW, "Not a git repository. Nothing to uninstall.")
   830→		return NewSilentError(errors.New("not a git repository"))
   831→	}
   832→
   833→	// Gather counts for display
   834→	sessionStateCount := countSessionStates()
   835→	shadowBranchCount := countShadowBranches()
   836→	gitHooksInstalled := strategy.IsGitHookInstalled()
   837→	claudeHooksInstalled := checkClaudeCodeHooksInstalled()
   838→	geminiHooksInstalled := checkGeminiCLIHooksInstalled()
   839→	entireDirExists := checkEntireDirExists()
   840→
   841→	// Check if there's anything to uninstall
   842→	if !entireDirExists && !gitHooksInstalled && sessionStateCount == 0 &&
   843→		shadowBranchCount == 0 && !claudeHooksInstalled && !geminiHooksInstalled {
   844→		fmt.Fprintln(w, "Entire is not installed in this repository.")
   845→		return nil
   846→	}
   847→
   848→	// Show confirmation prompt unless --force
   849→	if !force {
   850→		fmt.Fprintln(w, "\nThis will completely remove Entire from this repository:")
   851→		if entireDirExists {
   852→			fmt.Fprintln(w, "  - .entire/ directory")
   853→		}
   854→		if gitHooksInstalled {
   855→			fmt.Fprintln(w, "  - Git hooks (prepare-commit-msg, commit-msg, post-commit, pre-push)")
   856→		}
   857→		if sessionStateCount > 0 {
   858→			fmt.Fprintf(w, "  - Session state files (%d)\n", sessionStateCount)
   859→		}
   860→		if shadowBranchCount > 0 {
   861→			fmt.Fprintf(w, "  - Shadow branches (%d)\n", shadowBranchCount)
   862→		}
   863→		switch {
   864→		case claudeHooksInstalled && geminiHooksInstalled:
   865→			fmt.Fprintln(w, "  - Agent hooks (Claude Code, Gemini CLI)")
   866→		case claudeHooksInstalled:
   867→			fmt.Fprintln(w, "  - Agent hooks (Claude Code)")
   868→		case geminiHooksInstalled:
   869→			fmt.Fprintln(w, "  - Agent hooks (Gemini CLI)")
   870→		}
   871→		fmt.Fprintln(w)
   872→
   873→		var confirmed bool
   874→		form := NewAccessibleForm(
   875→			huh.NewGroup(
   876→				huh.NewConfirm().
   877→					Title("Are you sure you want to uninstall Entire?").
   878→					Affirmative("Yes, uninstall").
   879→					Negative("Cancel").
   880→					Value(&confirmed),
   881→			),
   882→		)
   883→
   884→		if err := form.Run(); err != nil {
   885→			return fmt.Errorf("confirmation cancelled: %w", err)
   886→		}
   887→
   888→		if !confirmed {
   889→			fmt.Fprintln(w, "Uninstall cancelled.")
   890→			return nil
   891→		}
   892→	}
   893→
   894→	fmt.Fprintln(w, "\nUninstalling Entire CLI...")
   895→
   896→	// 1. Remove agent hooks (lowest risk)
   897→	if err := removeAgentHooks(w); err != nil {
   898→		fmt.Fprintf(errW, "Warning: failed to remove agent hooks: %v\n", err)
   899→	}
   900→
   901→	// 2. Remove git hooks
   902→	removed, err := strategy.RemoveGitHook()
   903→	if err != nil {
   904→		fmt.Fprintf(errW, "Warning: failed to remove git hooks: %v\n", err)
   905→	} else if removed > 0 {
   906→		fmt.Fprintf(w, "  Removed git hooks (%d)\n", removed)
   907→	}
   908→

   919→		fmt.Fprintf(errW, "Warning: failed to remove .entire directory: %v\n", err)

   930→	}
   931→
   932→	fmt.Fprintln(w, "\nEntire CLI uninstalled successfully.")
   933→	return nil
   934→}
