   634→// addCheckpointTrailer adds the Entire-Checkpoint trailer to a commit message.
   635→// Handles proper trailer formatting (blank line before trailers if needed).
   636→func addCheckpointTrailer(message string, checkpointID id.CheckpointID) string {
   637→	trailer := trailers.CheckpointTrailerKey + ": " + checkpointID.String()
   638→
   639→	// If message already ends with trailers (lines starting with key:), just append
   640→	// Otherwise, add a blank line first
   641→	lines := strings.Split(strings.TrimRight(message, "\n"), "\n")
   642→
   643→	// Check if last non-empty, non-comment line looks like a trailer
   644→	// Git comment lines start with # and may contain ": " (e.g., "# Changes to be committed:")
   645→	hasTrailers := false
   646→	for i := len(lines) - 1; i >= 0; i-- {
   647→		line := strings.TrimSpace(lines[i])
   648→		if line == "" {
   649→			break
   650→		}
   651→		// Skip git comment lines
   652→		if strings.HasPrefix(line, "#") {
   653→			continue
   654→		}
   655→		if strings.Contains(line, ": ") {
   656→			hasTrailers = true
   657→			break
   658→		}
   659→		// Non-comment, non-trailer line found - no existing trailers
   660→		break
   661→	}
   662→
   663→	if hasTrailers {
   664→		// Append trailer directly
   665→		return strings.TrimRight(message, "\n") + "\n" + trailer + "\n"
   666→	}
   667→
   668→	// Add blank line before trailer
   669→	return strings.TrimRight(message, "\n") + "\n\n" + trailer + "\n"
   670→}

   672→// addCheckpointTrailerWithComment adds the Entire-Checkpoint trailer with an explanatory comment.
   673→// The trailer is placed above the git comment block but below the user's message area,
   674→// with a comment explaining that the user can remove it if they don't want to link the commit
   675→// to the agent session. If prompt is non-empty, it's shown as context.
   676→func addCheckpointTrailerWithComment(message string, checkpointID id.CheckpointID, agentName, prompt string) string {
   677→	trailer := trailers.CheckpointTrailerKey + ": " + checkpointID.String()
   678→	commentLines := []string{
   679→		"# Remove the Entire-Checkpoint trailer above if you don't want to link this commit to " + agentName + " session context.",
   680→	}
   681→	if prompt != "" {
   682→		commentLines = append(commentLines, "# Last Prompt: "+prompt)
   683→	}
   684→	commentLines = append(commentLines, "# The trailer will be added to your next commit based on this branch.")
   685→	comment := strings.Join(commentLines, "\n")
   686→
   687→	lines := strings.Split(message, "\n")
   688→
   689→	// Find where the git comment block starts (first # line)
   690→	commentStart := -1
   691→	for i, line := range lines {
   692→		if strings.HasPrefix(line, "#") {
   693→			commentStart = i
   694→			break
   695→		}
   696→	}
   697→
   698→	if commentStart == -1 {
   699→		// No git comments, append trailer at the end
   700→		return strings.TrimRight(message, "\n") + "\n\n" + trailer + "\n" + comment + "\n"
   701→	}
   702→
   703→	// Split into user content and git comments
   704→	userContent := strings.Join(lines[:commentStart], "\n")
   705→	gitComments := strings.Join(lines[commentStart:], "\n")
   706→
   707→	// Build result: user content, blank line, trailer, comment, blank line, git comments
   708→	userContent = strings.TrimRight(userContent, "\n")
   709→	if userContent == "" {
   710→		// No user content yet - leave space for them to type, then trailer
   711→		// Two newlines: first for user's message line, second for blank separator
   712→		return "\n\n" + trailer + "\n" + comment + "\n\n" + gitComments
   713→	}
   714→	return userContent + "\n\n" + trailer + "\n" + comment + "\n\n" + gitComments
