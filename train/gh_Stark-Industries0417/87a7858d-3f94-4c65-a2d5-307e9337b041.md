   870→// GitCommitWithShadowHooks stages and commits files, simulating the prepare-commit-msg and post-commit hooks.
   871→// This is used for testing manual-commit strategy which needs:
   872→// - prepare-commit-msg hook: adds the Entire-Checkpoint trailer

   874→func (env *TestEnv) GitCommitWithShadowHooks(message string, files ...string) {
   875→	env.T.Helper()

   877→	// Stage files using go-git
   878→	for _, file := range files {
   879→		env.GitAdd(file)

   885→		env.T.Fatalf("failed to write commit message file: %v", err)

   900→	}

   915→			Name:  "Test User",

   925→	// This triggers condensation if the commit has an Entire-Checkpoint trailer

   934→// GitCommitWithTrailerRemoved stages and commits files, simulating what happens when
   935→// a user removes the Entire-Checkpoint trailer during commit message editing.
   936→// This tests the opt-out behavior where removing the trailer skips condensation.
   937→func (env *TestEnv) GitCommitWithTrailerRemoved(message string, files ...string) {
   938→	env.T.Helper()
   939→
   940→	// Stage files using go-git
   941→	for _, file := range files {
   942→		env.GitAdd(file)
   943→	}
   944→
   945→	// Create a temp file for the commit message (prepare-commit-msg hook modifies this)
   946→	msgFile := filepath.Join(env.RepoDir, ".git", "COMMIT_EDITMSG")
   947→	if err := os.WriteFile(msgFile, []byte(message), 0o644); err != nil {
   948→		env.T.Fatalf("failed to write commit message file: %v", err)
   949→	}
   950→
   951→	// Run prepare-commit-msg hook using the shared binary
   952→	prepCmd := exec.Command(getTestBinary(), "hooks", "git", "prepare-commit-msg", msgFile)
   953→	prepCmd.Dir = env.RepoDir
   954→	if output, err := prepCmd.CombinedOutput(); err != nil {
   955→		env.T.Logf("prepare-commit-msg output: %s", output)
   956→	}
   957→
   958→	// Read the modified message (with trailer added by hook)
   959→	modifiedMsg, err := os.ReadFile(msgFile)
   960→	if err != nil {
   961→		env.T.Fatalf("failed to read modified commit message: %v", err)
   962→	}
   963→
   964→	// REMOVE the Entire-Checkpoint trailer (simulating user editing the message)
   965→	lines := strings.Split(string(modifiedMsg), "\n")
   966→	var cleanedLines []string
   967→	for _, line := range lines {
   968→		// Skip the trailer and the comments about it
   969→		if strings.HasPrefix(line, "Entire-Checkpoint:") {
   970→			continue
   971→		}
   972→		if strings.Contains(line, "Remove the Entire-Checkpoint trailer") {
   973→			continue
   974→		}
   975→		if strings.Contains(line, "trailer will be added to your next commit") {
   976→			continue
   977→		}
   978→		cleanedLines = append(cleanedLines, line)
   979→	}
   980→	cleanedMsg := strings.TrimRight(strings.Join(cleanedLines, "\n"), "\n") + "\n"
   981→
   982→	// Create the commit using go-git with the cleaned message (no trailer)
   983→	repo, err := git.PlainOpen(env.RepoDir)
   984→	if err != nil {
