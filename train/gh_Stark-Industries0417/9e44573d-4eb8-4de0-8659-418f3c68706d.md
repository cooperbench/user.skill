   508→// writeTranscript writes the transcript file from in-memory content or file path.
   509→// If the transcript exceeds MaxChunkSize, it's split into multiple chunk files.
   510→func (s *GitStore) writeTranscript(opts WriteCommittedOptions, basePath string, entries map[string]object.TreeEntry) error {
   511→	transcript := opts.Transcript
   512→	if len(transcript) == 0 && opts.TranscriptPath != "" {
   513→		var readErr error
   514→		transcript, readErr = os.ReadFile(opts.TranscriptPath)
   515→		if readErr != nil {
   516→			// Non-fatal: transcript may not exist yet
   517→			transcript = nil
   518→		}
   519→	}
   520→	if len(transcript) == 0 {
   521→		return nil
   522→	}
   523→
   524→	// Redact secrets before chunking so content hash reflects redacted content
   525→	transcript, err := redact.JSONLBytes(transcript)
   526→	if err != nil {
   527→		return fmt.Errorf("failed to redact transcript secrets: %w", err)
   528→	}
   529→
   530→	// Chunk the transcript if it's too large
   531→	chunks, err := agent.ChunkTranscript(transcript, opts.Agent)
   532→	if err != nil {
   533→		return fmt.Errorf("failed to chunk transcript: %w", err)
   534→	}
   535→
   536→	// Write chunk files
   537→	for i, chunk := range chunks {
   538→		chunkPath := basePath + agent.ChunkFileName(paths.TranscriptFileName, i)
   539→		blobHash, err := CreateBlobFromContent(s.repo, chunk)
   540→		if err != nil {
   541→			return err
   542→		}
   543→		entries[chunkPath] = object.TreeEntry{
   544→			Name: chunkPath,
   545→			Mode: filemode.Regular,
   546→			Hash: blobHash,
   547→		}
   548→	}
   549→
   550→	// Content hash for deduplication (hash of full transcript)
   551→	contentHash := fmt.Sprintf("sha256:%x", sha256.Sum256(transcript))
   552→	hashBlob, err := CreateBlobFromContent(s.repo, []byte(contentHash))
   553→	if err != nil {
   554→		return err
   555→	}
   556→	entries[basePath+paths.ContentHashFileName] = object.TreeEntry{
   557→		Name: basePath + paths.ContentHashFileName,
   558→		Mode: filemode.Regular,
   559→		Hash: hashBlob,
   560→	}
   561→	return nil
   562→}

   564→// mergeFilesTouched combines two file lists, removing duplicates.
   565→func mergeFilesTouched(existing, additional []string) []string {
   566→	seen := make(map[string]bool)
   567→	var result []string
   568→
   569→	for _, f := range existing {
   570→		if !seen[f] {
   571→			seen[f] = true
   572→			result = append(result, f)
   573→		}
   574→	}
   575→	for _, f := range additional {
   576→		if !seen[f] {
   577→			seen[f] = true
   578→			result = append(result, f)
   579→		}
   580→	}
   581→
   582→	sort.Strings(result)
   583→	return result
   584→}

   594→func (s *GitStore) buildCommitMessage(opts WriteCommittedOptions, taskMetadataPath string) string {
   595→	var commitMsg strings.Builder

   597→	// Subject line is always the checkpoint ID for consistent formatting

   604→	commitMsg.WriteString(fmt.Sprintf("%s: %s\n", trailers.SessionTrailerKey, opts.SessionID))

   638→// Returns only the CheckpointSummary (paths + aggregated stats), not actual content.
   639→// Use ReadSessionContent to read actual transcript/prompts/context.
   640→// Returns nil, nil if the checkpoint doesn't exist.
   641→//
   642→// The storage format uses numbered subdirectories for each session (0-based):
   643→//
   644→//	<checkpoint-id>/
   645→//	├── metadata.json      # CheckpointSummary with sessions map
   646→//	├── 0/                 # First session
   647→//	│   ├── metadata.json  # Session-specific metadata
   648→//	│   └── full.jsonl     # Transcript
   649→//	├── 1/                 # Second session

   652→	_ = ctx // Reserved for future use
   653→
   654→	tree, err := s.getSessionsBranchTree()

   656→		return nil, nil //nolint:nilnil,nilerr // No sessions branch means no checkpoint exists

   661→	if err != nil {
   662→		return nil, nil //nolint:nilnil,nilerr // Checkpoint directory not found

   664→

   666→	metadataFile, err := checkpointTree.File(paths.MetadataFileName)

   668→		return nil, nil //nolint:nilnil,nilerr // metadata.json not found
   669→	}

   671→	content, err := metadataFile.Contents()
   672→	if err != nil {
   673→		return nil, fmt.Errorf("failed to read metadata.json: %w", err)
   674→	}
   675→

   677→	if err := json.Unmarshal([]byte(content), &summary); err != nil {
   678→		return nil, fmt.Errorf("failed to parse metadata.json: %w", err)

   684→// ReadSessionContent reads the actual content for a specific session within a checkpoint.
   685→// sessionIndex is 0-based (0 for first session, 1 for second, etc.).
   686→// Returns the session's metadata, transcript, prompts, and context.
   687→// Returns an error if the checkpoint or session doesn't exist.
   688→func (s *GitStore) ReadSessionContent(ctx context.Context, checkpointID id.CheckpointID, sessionIndex int) (*SessionContent, error) {
   689→	_ = ctx // Reserved for future use
   690→
   691→	tree, err := s.getSessionsBranchTree()
   692→	if err != nil {
   693→		return nil, ErrCheckpointNotFound
   694→	}
   695→
   696→	checkpointPath := checkpointID.Path()
   697→	checkpointTree, err := tree.Tree(checkpointPath)
   698→	if err != nil {
   699→		return nil, ErrCheckpointNotFound
   700→	}
   701→
   702→	// Get the session subdirectory
   703→	sessionDir := strconv.Itoa(sessionIndex)
   704→	sessionTree, err := checkpointTree.Tree(sessionDir)
   705→	if err != nil {
   706→		return nil, fmt.Errorf("session %d not found: %w", sessionIndex, err)
   707→	}
   708→
   709→	result := &SessionContent{}
   710→
   711→	// Read session-specific metadata
   712→	var agentType agent.AgentType
   713→	if metadataFile, fileErr := sessionTree.File(paths.MetadataFileName); fileErr == nil {
   714→		if content, contentErr := metadataFile.Contents(); contentErr == nil {
   715→			if jsonErr := json.Unmarshal([]byte(content), &result.Metadata); jsonErr == nil {
   716→				agentType = result.Metadata.Agent
   717→			}
   718→		}
   719→	}
   720→
   721→	// Read transcript
   722→	if transcript, transcriptErr := readTranscriptFromTree(sessionTree, agentType); transcriptErr == nil && transcript != nil {
   723→		result.Transcript = transcript
   724→	}
   725→
   726→	// Read prompts
   727→	if file, fileErr := sessionTree.File(paths.PromptFileName); fileErr == nil {
   728→		if content, contentErr := file.Contents(); contentErr == nil {
   729→			result.Prompts = content
   730→		}
   731→	}
   732→
   733→	// Read context
   734→	if file, fileErr := sessionTree.File(paths.ContextFileName); fileErr == nil {
   735→		if content, contentErr := file.Contents(); contentErr == nil {
   736→			result.Context = content
   737→		}
   738→	}
   739→
   740→	return result, nil
   741→}

   788→// ListCommitted lists all committed checkpoints from the entire/checkpoints/v1 branch.
   789→// Scans sharded paths: <id[:2]>/<id[2:]>/ directories containing metadata.json.
   790→//
   791→
   792→func (s *GitStore) ListCommitted(ctx context.Context) ([]CommittedInfo, error) {
   793→	_ = ctx // Reserved for future use
   794→
   795→	tree, err := s.getSessionsBranchTree()
   796→	if err != nil {
   797→		return []CommittedInfo{}, nil //nolint:nilerr // No sessions branch means empty list
   798→	}
   799→
   800→	var checkpoints []CommittedInfo
   801→
   802→	// Scan sharded structure: <2-char-prefix>/<remaining-id>/metadata.json
   803→	for _, bucketEntry := range tree.Entries {
   804→		if bucketEntry.Mode != filemode.Dir {
   805→			continue
   806→		}
   807→		// Bucket should be 2 hex chars
   808→		if len(bucketEntry.Name) != 2 {
   809→			continue
   810→		}
   811→
   812→		bucketTree, treeErr := s.repo.TreeObject(bucketEntry.Hash)
   813→		if treeErr != nil {
   814→			continue
   815→		}
   816→
   817→		// Each entry in the bucket is the remaining part of the checkpoint ID
   818→		for _, checkpointEntry := range bucketTree.Entries {
   819→			if checkpointEntry.Mode != filemode.Dir {
   820→				continue
   821→			}
   822→
   823→			checkpointTree, cpTreeErr := s.repo.TreeObject(checkpointEntry.Hash)
   824→			if cpTreeErr != nil {
   825→				continue
   826→			}
   827→
   828→			// Reconstruct checkpoint ID: <bucket><remaining>
   829→			checkpointIDStr := bucketEntry.Name + checkpointEntry.Name
   830→			checkpointID, cpIDErr := id.NewCheckpointID(checkpointIDStr)
   831→			if cpIDErr != nil {
   832→				// Skip invalid checkpoint IDs (shouldn't happen with our own data)
   833→				continue
   834→			}
   835→
   836→			info := CommittedInfo{
   837→				CheckpointID: checkpointID,
   838→			}
   839→
   840→			// Get details from root metadata file (CheckpointSummary format)
   841→			if metadataFile, fileErr := checkpointTree.File(paths.MetadataFileName); fileErr == nil {
   842→				if content, contentErr := metadataFile.Contents(); contentErr == nil {
   843→					var summary CheckpointSummary
   844→					if err := json.Unmarshal([]byte(content), &summary); err == nil {
   845→						info.CheckpointsCount = summary.CheckpointsCount
   846→						info.FilesTouched = summary.FilesTouched
   847→						info.SessionCount = len(summary.Sessions)
   848→
   849→						// Read session metadata from latest session to get Agent, SessionID, CreatedAt
   850→						if len(summary.Sessions) > 0 {
   851→							latestIndex := len(summary.Sessions) - 1
   852→							latestDir := strconv.Itoa(latestIndex)
   853→							if sessionTree, treeErr := checkpointTree.Tree(latestDir); treeErr == nil {
   854→								if sessionMetadataFile, smErr := sessionTree.File(paths.MetadataFileName); smErr == nil {
   855→									if sessionContent, scErr := sessionMetadataFile.Contents(); scErr == nil {
   856→										var sessionMetadata CommittedMetadata
   857→										if json.Unmarshal([]byte(sessionContent), &sessionMetadata) == nil {
   858→											info.Agent = sessionMetadata.Agent
   859→											info.SessionID = sessionMetadata.SessionID
   860→											info.CreatedAt = sessionMetadata.CreatedAt
   861→										}
   862→									}
   863→								}
   864→							}
   865→						}
   866→					}
   867→				}
   868→			}
   869→
   870→			checkpoints = append(checkpoints, info)
   871→		}
   872→	}
   873→
   874→	// Sort by time (most recent first)
   875→	sort.Slice(checkpoints, func(i, j int) bool {
   876→		return checkpoints[i].CreatedAt.After(checkpoints[j].CreatedAt)
   877→	})
   878→
   879→	return checkpoints, nil
   880→}

   927→// UpdateSummary updates the summary field in the latest session's metadata.
   928→// Returns ErrCheckpointNotFound if the checkpoint doesn't exist.
   929→func (s *GitStore) UpdateSummary(ctx context.Context, checkpointID id.CheckpointID, summary *Summary) error {
   930→	_ = ctx // Reserved for future use
   931→
   932→	// Ensure sessions branch exists
   933→	if err := s.ensureSessionsBranch(); err != nil {
   934→		return fmt.Errorf("failed to ensure sessions branch: %w", err)
   935→	}
   936→
   937→	// Get current branch tip and flatten tree
   938→	ref, entries, err := s.getSessionsBranchEntries()
   939→	if err != nil {
   940→		return err
   941→	}
   942→
   943→	// Read root CheckpointSummary to find the latest session
   944→	basePath := checkpointID.Path() + "/"
   945→	rootMetadataPath := basePath + paths.MetadataFileName
   946→	entry, exists := entries[rootMetadataPath]
   947→	if !exists {
   948→		return ErrCheckpointNotFound
   949→	}
   950→
   951→	checkpointSummary, err := s.readSummaryFromBlob(entry.Hash)
   952→	if err != nil {
   953→		return fmt.Errorf("failed to read checkpoint summary: %w", err)
   954→	}
   955→
   956→	// Find the latest session's metadata path (0-based indexing)
   957→	latestIndex := len(checkpointSummary.Sessions) - 1
   958→	sessionMetadataPath := fmt.Sprintf("%s%d/%s", basePath, latestIndex, paths.MetadataFileName)
   959→	sessionEntry, exists := entries[sessionMetadataPath]
   960→	if !exists {
   961→		return fmt.Errorf("session metadata not found at %s", sessionMetadataPath)
   962→	}
   963→
   964→	// Read and update session metadata
   965→	existingMetadata, err := s.readMetadataFromBlob(sessionEntry.Hash)
   966→	if err != nil {
   967→		return fmt.Errorf("failed to read session metadata: %w", err)
   968→	}
   969→
   970→	// Update the summary
   971→	existingMetadata.Summary = summary
   972→
   973→	// Write updated session metadata
   974→	metadataJSON, err := jsonutil.MarshalIndentWithNewline(existingMetadata, "", "  ")
   975→	if err != nil {
   976→		return fmt.Errorf("failed to marshal metadata: %w", err)
   977→	}
   978→	metadataHash, err := CreateBlobFromContent(s.repo, metadataJSON)
   979→	if err != nil {
   980→		return fmt.Errorf("failed to create metadata blob: %w", err)
   981→	}
   982→	entries[sessionMetadataPath] = object.TreeEntry{
   983→		Name: sessionMetadataPath,
   984→		Mode: filemode.Regular,
   985→		Hash: metadataHash,
   986→	}
   987→
   988→	// Build and commit
   989→	newTreeHash, err := BuildTreeFromEntries(s.repo, entries)
   990→	if err != nil {
   991→		return err
   992→	}
   993→
   994→	authorName, authorEmail := getGitAuthorFromRepo(s.repo)
   995→	commitMsg := fmt.Sprintf("Update summary for checkpoint %s (session: %s)", checkpointID, existingMetadata.SessionID)
   996→	newCommitHash, err := s.createCommit(newTreeHash, ref.Hash(), commitMsg, authorName, authorEmail)
   997→	if err != nil {
   998→		return err
   999→	}
  1000→
  1001→	refName := plumbing.NewBranchReferenceName(paths.MetadataBranchName)
  1002→	newRef := plumbing.NewHashReference(refName, newCommitHash)
  1003→	if err := s.repo.Storer.SetReference(newRef); err != nil {
  1004→		return fmt.Errorf("failed to set branch reference: %w", err)
  1005→	}
  1006→
  1007→	return nil
  1008→}

  1041→	ref, err := s.repo.Reference(refName, true)
  1042→	if err != nil {

  1044→		remoteRefName := plumbing.NewRemoteReferenceName("origin", paths.MetadataBranchName)
  1045→		ref, err = s.repo.Reference(remoteRefName, true)
  1046→		if err != nil {
  1047→			return nil, fmt.Errorf("sessions branch not found: %w", err)
  1048→		}
  1049→	}
  1050→
  1051→	commit, err := s.repo.CommitObject(ref.Hash())
  1052→	if err != nil {
  1053→		return nil, fmt.Errorf("failed to get commit object: %w", err)

  1143→// createRedactedBlobFromFile reads a file, applies secrets redaction, and creates a git blob.
  1144→// JSONL files get JSONL-aware redaction; all other files get plain string redaction.
  1145→func createRedactedBlobFromFile(repo *git.Repository, filePath, treePath string) (plumbing.Hash, filemode.FileMode, error) {
  1146→	info, err := os.Stat(filePath)

  1148→		return plumbing.ZeroHash, 0, fmt.Errorf("failed to stat file: %w", err)
  1149→	}
  1150→
  1151→	mode := filemode.Regular
  1152→	if info.Mode()&0o111 != 0 {
  1153→		mode = filemode.Executable
  1154→	}

  1156→	content, err := os.ReadFile(filePath) //nolint:gosec // filePath comes from walking the metadata directory

  1162→	// running string replacement on them would corrupt the data.
  1163→	isBin, binErr := binary.IsBinary(bytes.NewReader(content))

  1168→		}

  1172→	if strings.HasSuffix(treePath, ".jsonl") {

  1175→			return plumbing.ZeroHash, 0, fmt.Errorf("failed to redact secrets: %w", err)
  1176→		}

  1182→	if err != nil {
  1183→		return plumbing.ZeroHash, 0, fmt.Errorf("failed to create blob: %w", err)
  1184→	}
  1185→	return hash, mode, nil

  1208→// readTranscriptFromTree reads a transcript from a git tree, handling both chunked and non-chunked formats.
  1209→// It checks for chunk files first (.001, .002, etc.), then falls back to the base file.
  1210→// The agentType is used for reassembling chunks in the correct format.
  1211→func readTranscriptFromTree(tree *object.Tree, agentType agent.AgentType) ([]byte, error) {

  1213→	var chunkFiles []string
  1214→	var hasBaseFile bool
  1215→

   428→// calculateTokenUsage calculates token usage from raw transcript data.
   429→// startOffset is the line number (Claude Code) or message index (Gemini CLI)

   431→// of the transcript since the last checkpoint.
   432→func calculateTokenUsage(agentType agent.AgentType, data []byte, startOffset int) *agent.TokenUsage {
   433→	if len(data) == 0 {
   434→		return &agent.TokenUsage{}
   435→	}
   436→
   437→	// Try Gemini format first if agentType is Gemini, or as fallback if Unknown
   438→	if agentType == agent.AgentTypeGemini || agentType == agent.AgentTypeUnknown {
   439→		// Attempt to parse as Gemini JSON
   440→		transcript, err := geminicli.ParseTranscript(data)
   441→		if err == nil && transcript != nil && len(transcript.Messages) > 0 {
   442→			return geminicli.CalculateTokenUsage(data, startOffset)
   443→		}
   444→		// If agentType is explicitly Gemini but parsing failed, return empty usage
   445→		if agentType == agent.AgentTypeGemini {
   446→			return &agent.TokenUsage{}
   447→		}
   448→		// Otherwise fall through to JSONL parsing for Unknown type
   449→	}

   452→	lines, err := claudecode.ParseTranscript(data)
   453→	if err != nil || len(lines) == 0 {
   454→		return &agent.TokenUsage{}
   455→	}

   457→	if startOffset > 0 && startOffset < len(lines) {
   458→		lines = lines[startOffset:]

   463→// extractUserPromptsFromLines extracts user prompts from JSONL transcript lines.
   464→// IDE-injected context tags (like <ide_opened_file>) are stripped from the results.
   465→func extractUserPromptsFromLines(lines []string) []string {
   466→	var prompts []string
   467→	for _, line := range lines {
   468→		line = strings.TrimSpace(line)
   469→		if line == "" {
   470→			continue
   471→		}
   472→
   473→		var entry map[string]interface{}
   474→		if err := json.Unmarshal([]byte(line), &entry); err != nil {
   475→			continue
   476→		}
   477→
   478→		// Check for user message (supports both "human" and "user" types)
   479→		msgType, ok := entry["type"].(string)
   480→		if !ok || (msgType != "human" && msgType != "user") {
   481→			continue
   482→		}
   483→
   484→		// Extract message content
   485→		message, ok := entry["message"].(map[string]interface{})
   486→		if !ok {
   487→			continue
   488→		}
   489→
   490→		// Handle string content
   491→		if content, ok := message["content"].(string); ok && content != "" {
   492→			cleaned := textutil.StripIDEContextTags(content)
   493→			if cleaned != "" {
   494→				prompts = append(prompts, cleaned)
   495→			}
   496→			continue
   497→		}
   498→
   499→		// Handle array content (e.g., multiple text blocks from VSCode)
   500→		if arr, ok := message["content"].([]interface{}); ok {
   501→			var texts []string
   502→			for _, item := range arr {
   503→				if m, ok := item.(map[string]interface{}); ok {
   504→					if m["type"] == "text" {
   505→						if text, ok := m["text"].(string); ok {
   506→							texts = append(texts, text)
   507→						}
   508→					}
   509→				}
   510→			}
   511→			if len(texts) > 0 {
   512→				cleaned := textutil.StripIDEContextTags(strings.Join(texts, "\n\n"))
   513→				if cleaned != "" {
   514→					prompts = append(prompts, cleaned)
   515→				}
   516→			}
   517→		}
   518→	}
   519→	return prompts
   520→}

   546→// CondenseSessionByID force-condenses a session by its ID and cleans up.

   629→// cleanupShadowBranchIfUnused deletes a shadow branch if no other active sessions reference it.
   630→func (s *ManualCommitStrategy) cleanupShadowBranchIfUnused(_ *git.Repository, shadowBranchName, excludeSessionID string) error {
   631→	// List all session states to check if any other session uses this shadow branch
   632→	allStates, err := s.listAllSessionStates()

   634→		return fmt.Errorf("failed to list session states: %w", err)
   635→	}
   636→
   637→	for _, state := range allStates {
   638→		if state.SessionID == excludeSessionID {
   639→			continue
   640→		}
   641→		otherShadow := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   642→		if otherShadow == shadowBranchName && state.StepCount > 0 {
   643→			// Another session still needs this shadow branch
   644→			return nil
   645→		}
   646→	}
   647→
   648→	// No other sessions need it, delete the shadow branch via CLI
   649→	// (go-git v5's RemoveReference doesn't persist with packed refs/worktrees)

   651→		// Branch already gone is not an error
   652→		if errors.Is(err, ErrBranchNotFound) {
   653→			return nil
   654→		}
   655→		return fmt.Errorf("failed to remove shadow branch: %w", err)
   656→	}
   657→	return nil
   658→}

    73→// Scans sharded paths: <id[:2]>/<id[2:]>/ directories containing metadata.json.
    74→// Used by both manual-commit and auto-commit strategies.
    75→func ListCheckpoints() ([]CheckpointInfo, error) {
    76→	repo, err := OpenRepository()
    77→	if err != nil {
    78→		return nil, fmt.Errorf("failed to open git repository: %w", err)
    79→	}
    80→
    81→	refName := plumbing.NewBranchReferenceName(paths.MetadataBranchName)
    82→	ref, err := repo.Reference(refName, true)
    83→	if err != nil {
    84→		//nolint:nilerr // No sessions branch yet is expected, return empty list
    85→		return []CheckpointInfo{}, nil
    86→	}
    87→
    88→	commit, err := repo.CommitObject(ref.Hash())
    89→	if err != nil {
    90→		return nil, fmt.Errorf("failed to get commit object: %w", err)
    91→	}
    92→
    93→	tree, err := commit.Tree()
    94→	if err != nil {
    95→		return nil, fmt.Errorf("failed to get commit tree: %w", err)
    96→	}
    97→
    98→	var checkpoints []CheckpointInfo
    99→
   100→	// Scan sharded structure: <2-char-prefix>/<remaining-id>/metadata.json
   101→	// The tree has 2-character directories (hex buckets)
   102→	for _, bucketEntry := range tree.Entries {
   103→		if bucketEntry.Mode != filemode.Dir {
   104→			continue
   105→		}
   106→		// Bucket should be 2 hex chars
   107→		if len(bucketEntry.Name) != 2 {
   108→			continue
   109→		}
   110→
   111→		bucketTree, treeErr := repo.TreeObject(bucketEntry.Hash)
   112→		if treeErr != nil {
   113→			continue
   114→		}
   115→
   116→		// Each entry in the bucket is the remaining part of the checkpoint ID
   117→		for _, checkpointEntry := range bucketTree.Entries {
   118→			if checkpointEntry.Mode != filemode.Dir {
   119→				continue
   120→			}
   121→
   122→			checkpointTree, cpTreeErr := repo.TreeObject(checkpointEntry.Hash)
   123→			if cpTreeErr != nil {
   124→				continue
   125→			}
   126→
   127→			// Reconstruct checkpoint ID: <bucket><remaining>
   128→			checkpointIDStr := bucketEntry.Name + checkpointEntry.Name
   129→			checkpointID, cpErr := id.NewCheckpointID(checkpointIDStr)
   130→			if cpErr != nil {
   131→				// Skip invalid checkpoint IDs
   132→				continue
   133→			}
   134→
   135→			info := CheckpointInfo{
   136→				CheckpointID: checkpointID,
   137→			}
   138→
   139→			// Get details from metadata file (CheckpointSummary format)
   140→			if metadataFile, fileErr := checkpointTree.File(paths.MetadataFileName); fileErr == nil {
   141→				if content, contentErr := metadataFile.Contents(); contentErr == nil {
   142→					var summary checkpoint.CheckpointSummary
   143→					if json.Unmarshal([]byte(content), &summary) == nil && len(summary.Sessions) > 0 {
   144→						info.CheckpointsCount = summary.CheckpointsCount
   145→						info.FilesTouched = summary.FilesTouched
   146→						info.SessionCount = len(summary.Sessions)
   147→
   148→						// Read session-level metadata for Agent, SessionID, CreatedAt, SessionIDs
   149→						for i, sessionPaths := range summary.Sessions {
   150→							if sessionPaths.Metadata != "" {
   151→								// SessionFilePaths now contains absolute paths with leading "/"
   152→								// Strip the leading "/" for tree.File() which expects paths without leading slash
   153→								sessionMetadataPath := strings.TrimPrefix(sessionPaths.Metadata, "/")
   154→								if sessionFile, sErr := tree.File(sessionMetadataPath); sErr == nil {
   155→									if sessionContent, scErr := sessionFile.Contents(); scErr == nil {
   156→										var sessionMetadata checkpoint.CommittedMetadata
   157→										if json.Unmarshal([]byte(sessionContent), &sessionMetadata) == nil {
   158→											info.SessionIDs = append(info.SessionIDs, sessionMetadata.SessionID)
   159→											// Use first session's metadata for Agent, SessionID, CreatedAt
   160→											if i == 0 {
   161→												info.Agent = sessionMetadata.Agent
   162→												info.SessionID = sessionMetadata.SessionID
   163→												info.CreatedAt = sessionMetadata.CreatedAt
   164→												info.IsTask = sessionMetadata.IsTask
   165→												info.ToolUseID = sessionMetadata.ToolUseID
   166→											}
   167→										}
   168→									}
   169→								}
   170→							}
   171→						}
   172→					}
   173→				}
   174→			}
   175→
   176→			checkpoints = append(checkpoints, info)
   177→		}
   178→	}
   179→
   180→	// Sort by time (most recent first)
   181→	sort.Slice(checkpoints, func(i, j int) bool {
   182→		return checkpoints[i].CreatedAt.After(checkpoints[j].CreatedAt)
   183→	})
   184→
   185→	return checkpoints, nil
   186→}

   244→// ensureMetadataBranch creates the orphan entire/checkpoints/v1 branch if it doesn't exist.
   245→// This branch has no parent and starts with an empty tree.
   246→func EnsureMetadataBranch(repo *git.Repository) error {
   247→	refName := plumbing.NewBranchReferenceName(paths.MetadataBranchName)
   248→
   249→	// Check if branch already exists
   250→	_, err := repo.Reference(refName, true)
   251→	if err == nil {
   252→		// Branch already exists
   253→		return nil
   254→	}
   255→
   256→	// Create empty tree (no files)
   257→	emptyTree := &object.Tree{Entries: []object.TreeEntry{}}
   258→	obj := repo.Storer.NewEncodedObject()
   259→	if err := emptyTree.Encode(obj); err != nil {
   260→		return fmt.Errorf("failed to encode empty tree: %w", err)
   261→	}
   262→	emptyTreeHash, err := repo.Storer.SetEncodedObject(obj)
   263→	if err != nil {
   264→		return fmt.Errorf("failed to store empty tree: %w", err)
   265→	}
   266→
   267→	// Create orphan commit (no parent)
   268→	now := time.Now()
   269→	sig := object.Signature{
   270→		Name:  "Entire",
   271→		Email: "entire@local",
   272→		When:  now,
   273→	}
   274→
   275→	commit := &object.Commit{
   276→		TreeHash:  emptyTreeHash,
   277→		Author:    sig,
   278→		Committer: sig,
   279→		Message:   "Initialize metadata branch\n\nThis branch stores session metadata for the auto-commit strategy.\n",
   280→	}
   281→	// Note: No ParentHashes - this is an orphan commit
   282→
   283→	commitObj := repo.Storer.NewEncodedObject()
   284→	if err := commit.Encode(commitObj); err != nil {
   285→		return fmt.Errorf("failed to encode orphan commit: %w", err)
   286→	}
   287→	commitHash, err := repo.Storer.SetEncodedObject(commitObj)
   288→	if err != nil {
   289→		return fmt.Errorf("failed to store orphan commit: %w", err)
   290→	}
   291→
   292→	// Create branch reference
   293→	ref := plumbing.NewHashReference(refName, commitHash)
   294→	if err := repo.Storer.SetReference(ref); err != nil {
   295→		return fmt.Errorf("failed to create metadata branch: %w", err)
   296→	}
   297→
   298→	fmt.Fprintf(os.Stderr, "✓ Created orphan branch '%s' for session metadata\n", paths.MetadataBranchName)
   299→	return nil
   300→}

   390→// ExtractFirstPrompt extracts and truncates the first meaningful prompt from prompt content.
   391→// Prompts are separated by "\n\n---\n\n". Skips empty prompts and separator-only content.
   392→// Returns empty string if no valid prompt is found.
   393→func ExtractFirstPrompt(content string) string {
   394→	if content == "" {
   395→		return ""
   396→	}
   397→
   398→	// Prompts are separated by "\n\n---\n\n"
   399→	// Find the first non-empty prompt
   400→	prompts := strings.Split(content, "\n\n---\n\n")
   401→	var firstPrompt string
   402→	for _, p := range prompts {
   403→		cleaned := strings.TrimSpace(p)
   404→		// Skip empty prompts or prompts that are just dashes/separators
   405→		if cleaned == "" || isOnlySeparators(cleaned) {
   406→			continue
   407→		}
   408→		firstPrompt = cleaned
   409→		break
   410→	}
   411→
   412→	if firstPrompt == "" {
   413→		return ""
   414→	}
   415→
   416→	return TruncateDescription(firstPrompt, MaxDescriptionLength)
   417→}

   446→// ReadAllSessionPromptsFromTree reads the first prompt for all sessions in a multi-session checkpoint.
   447→// Returns a slice of prompts parallel to sessionIDs (oldest to newest).
   448→// For single-session checkpoints, returns a slice with just the root prompt.
   449→func ReadAllSessionPromptsFromTree(tree *object.Tree, checkpointPath string, sessionCount int, sessionIDs []string) []string {
   450→	if sessionCount <= 1 || len(sessionIDs) <= 1 {
   451→		// Single session - just return the root prompt
   452→		prompt := ReadSessionPromptFromTree(tree, checkpointPath)
   453→		if prompt != "" {
   454→			return []string{prompt}
   455→		}
   456→		return nil
   457→	}
   458→
   459→	// Multi-session: read prompts from archived folders (0/, 1/, etc.) and root
   460→	prompts := make([]string, len(sessionIDs))
   461→
   462→	// Read archived session prompts (folders 0, 1, ... N-2)
   463→	for i := range sessionCount - 1 {
   464→		archivedPath := fmt.Sprintf("%s/%d", checkpointPath, i)
   465→		prompts[i] = ReadSessionPromptFromTree(tree, archivedPath)
   466→	}
   467→
   468→	// Read the most recent session prompt (at root level)
   469→	prompts[len(prompts)-1] = ReadSessionPromptFromTree(tree, checkpointPath)
   470→
   471→	return prompts
   472→}

   519→// OpenRepository opens the git repository with linked worktree support enabled.
   520→// It uses git.PlainOpenWithOptions with EnableDotGitCommonDir set to true,
   521→// which is required for proper operation in git worktrees created via 'git worktree add'.
   522→//
   523→// Without EnableDotGitCommonDir, go-git operations in worktrees can silently fail:
   524→// - Commits appear to succeed but are not persisted
   525→// - Refs are written to incorrect locations
   526→// - The worktree's HEAD/index don't get updated properly
   527→//
   528→// This happens because worktrees use .git as a file (pointing to the main repo)
   529→// rather than a directory, and go-git needs to route paths correctly between
   530→// shared (.git/) and per-worktree (.git/worktrees/<name>/) locations.
   531→//
   532→// The function first uses 'git rev-parse --show-toplevel' to find the repository
   533→// root, which works correctly even when called from a subdirectory within the repo.
   534→func OpenRepository() (*git.Repository, error) {
   535→	// First, find the repository root using git rev-parse --show-toplevel
   536→	// This works correctly from any subdirectory within the repository
   537→	repoRoot, err := GetWorktreePath()
   538→	if err != nil {
   539→		// Fallback to current directory if git command fails
   540→		// (e.g., if git is not installed or we're not in a repo)
   541→		repoRoot = "."
   542→	}
   543→
   544→	repo, err := git.PlainOpenWithOptions(repoRoot, &git.PlainOpenOptions{
   545→		EnableDotGitCommonDir: true,
   546→	})
   547→	if err != nil {
   548→		return nil, fmt.Errorf("failed to open repository: %w", err)
   549→	}
   550→	return repo, nil
   551→}

   553→// IsInsideWorktree returns true if the current directory is inside a git worktree
   554→// (as opposed to the main repository). Worktrees have .git as a file pointing
   555→// to the main repo, while the main repo has .git as a directory.
   556→// This function works correctly from any subdirectory within the repository.
   557→func IsInsideWorktree() bool {
   558→	// First find the repository root
   559→	repoRoot, err := GetWorktreePath()
   560→	if err != nil {
   561→		return false
   562→	}
   563→
   564→	gitPath := filepath.Join(repoRoot, gitDir)
   565→	gitInfo, err := os.Stat(gitPath)
   566→	if err != nil {
   567→		return false
   568→	}
   569→	return !gitInfo.IsDir()
   570→}

   572→// GetMainRepoRoot returns the root directory of the main repository.
   573→// In the main repo, this is the worktree path (repo root).
   574→// In a worktree, this parses the .git file to find the main repo.
   575→// This function works correctly from any subdirectory within the repository.
   576→//
   577→// Per gitrepository-layout(5), a worktree's .git file is a "gitfile" containing
   578→// "gitdir: <path>" pointing to $GIT_DIR/worktrees/<id> in the main repository.
   579→// See: https://git-scm.com/docs/gitrepository-layout
   580→func GetMainRepoRoot() (string, error) {
   581→	// First find the worktree/repo root
   582→	repoRoot, err := GetWorktreePath()
   583→	if err != nil {
   584→		return "", fmt.Errorf("failed to get worktree path: %w", err)
   585→	}
   586→
   587→	if !IsInsideWorktree() {
   588→		return repoRoot, nil
   589→	}
   590→
   591→	// Worktree .git file contains: "gitdir: /path/to/main/.git/worktrees/<id>"
   592→	gitFilePath := filepath.Join(repoRoot, gitDir)
   593→	content, err := os.ReadFile(gitFilePath) //nolint:gosec // G304: gitFilePath is constructed from repo root, not user input
   594→	if err != nil {
   595→		return "", fmt.Errorf("failed to read .git file: %w", err)
   596→	}
   597→
   598→	gitdir := strings.TrimSpace(string(content))
   599→	gitdir = strings.TrimPrefix(gitdir, "gitdir: ")
   600→
   601→	// Extract main repo root: everything before "/.git/"
   602→	idx := strings.LastIndex(gitdir, "/.git/")
   603→	if idx < 0 {
   604→		return "", fmt.Errorf("unexpected gitdir format: %s", gitdir)
   605→	}
   606→	return gitdir[:idx], nil
   607→}

   609→// GetGitCommonDir returns the path to the shared git directory.
   610→// In a regular checkout, this is .git/
   611→// In a worktree, this is the main repo's .git/ (not .git/worktrees/<name>/)
   612→// Uses git rev-parse --git-common-dir for reliable handling of worktrees.
   613→func GetGitCommonDir() (string, error) {
   614→	ctx := context.Background()
   615→	cmd := exec.CommandContext(ctx, "git", "rev-parse", "--git-common-dir")
   616→	cmd.Dir = "."
   617→	output, err := cmd.Output()
   618→	if err != nil {
   619→		return "", fmt.Errorf("failed to get git common dir: %w", err)
   620→	}
   621→
   622→	commonDir := strings.TrimSpace(string(output))
   623→
   624→	// git rev-parse --git-common-dir returns relative paths from the working directory,
   625→	// so we need to make it absolute if it isn't already
   626→	if !filepath.IsAbs(commonDir) {
   627→		commonDir = filepath.Join(".", commonDir)
   628→	}
   629→
   630→	return filepath.Clean(commonDir), nil

   699→// checkCanRewind checks if working directory is clean enough for rewind.

   701→func checkCanRewind() (bool, string, error) {
   702→	repo, err := OpenRepository()
   703→	if err != nil {
   704→		return false, "", fmt.Errorf("failed to open git repository: %w", err)
   705→	}
   706→
   707→	worktree, err := repo.Worktree()
   708→	if err != nil {
   709→		return false, "", fmt.Errorf("failed to get worktree: %w", err)
   710→	}
   711→
   712→	status, err := worktree.Status()
   713→	if err != nil {
   714→		return false, "", fmt.Errorf("failed to get status: %w", err)
   715→	}
   716→
   717→	if status.IsClean() {
   718→		return true, "", nil
   719→	}
   720→

   722→	for file, st := range status {
   723→		// Skip .entire directory
   724→		if paths.IsInfrastructurePath(file) {
   725→			continue
   726→		}
   727→

   731→		}

   738→		case st.Staging == git.Modified || st.Worktree == git.Modified:
   739→			modified = append(modified, file)
   740→		}
   741→	}

   744→		return true, "", nil

   748→	msg.WriteString("You have uncommitted changes:\n")

   750→		msg.WriteString(fmt.Sprintf("  modified:   %s\n", f))
   751→	}
   752→	for _, f := range added {
   753→		msg.WriteString(fmt.Sprintf("  added:      %s\n", f))
   754→	}
   755→	for _, f := range deleted {
   756→		msg.WriteString(fmt.Sprintf("  deleted:    %s\n", f))
   757→	}
   758→	msg.WriteString("\nPlease commit or stash your changes before rewinding.")
   759→
   760→	return false, msg.String(), nil

   764→// Unlike checkCanRewind, this always returns canRewind=true but includes a warning message
   765→// with +/- line stats for uncommitted changes. Used by manual-commit strategy.
   766→func checkCanRewindWithWarning() (bool, string, error) {

   768→	if err != nil {
   769→		// Can't open repo - still allow rewind but without stats
   770→		return true, "", nil //nolint:nilerr // Rewind allowed even if repo can't be opened
   771→	}
   772→
   773→	worktree, err := repo.Worktree()
   774→	if err != nil {
   775→		return true, "", nil //nolint:nilerr // Rewind allowed even if worktree can't be accessed
   776→	}
   777→

   780→		return true, "", nil //nolint:nilerr // Rewind allowed even if status can't be retrieved
   781→	}

   791→	}
   792→

   794→	if err != nil {
   795→		return true, "", nil //nolint:nilerr // Rewind allowed even if commit lookup fails
   796→	}
   797→
   798→	headTree, err := headCommit.Tree()
   799→	if err != nil {
   800→		return true, "", nil //nolint:nilerr // Rewind allowed even if tree lookup fails
   801→	}
   802→
   803→	type fileChange struct {
   804→		status   string // "modified", "added", "deleted"
   805→		added    int
   806→		removed  int
   807→		filename string
   808→	}
   809→
   810→	var changes []fileChange
   811→	// Use repo root, not cwd - git status returns paths relative to repo root
   812→	repoRoot, err := GetWorktreePath()
   813→	if err != nil {
   814→		return true, "", nil //nolint:nilerr // Rewind allowed even if repo root lookup fails
   815→	}
   816→
   817→	for file, st := range status {
   818→		// Skip .entire directory
   819→		if paths.IsInfrastructurePath(file) {
   820→			continue
   821→		}
   822→
   823→		// Skip untracked files
   824→		if st.Worktree == git.Untracked {
   825→			continue
   826→		}
   827→
   828→		var change fileChange
   829→		change.filename = file
   830→
   831→		switch {
   832→		case st.Staging == git.Added || st.Worktree == git.Added:
   833→			change.status = "added"
   834→			// New file - count all lines as added
   835→			absPath := filepath.Join(repoRoot, file)
   836→			if content, err := os.ReadFile(absPath); err == nil { //nolint:gosec // absPath is repo root + relative path from git status
   837→				change.added = countLines(content)
   838→			}
   839→		case st.Staging == git.Deleted || st.Worktree == git.Deleted:
   840→			change.status = "deleted"
   841→			// Deleted file - count lines from HEAD as removed
   842→			if entry, err := headTree.File(file); err == nil {
   843→				if content, err := entry.Contents(); err == nil {
   844→					change.removed = countLines([]byte(content))
   845→				}
   846→			}
   847→		case st.Staging == git.Modified || st.Worktree == git.Modified:
   848→			change.status = "modified"
   849→			// Modified file - compute diff stats
   850→			var headContent, workContent []byte
   851→			if entry, err := headTree.File(file); err == nil {
   852→				if content, err := entry.Contents(); err == nil {
   853→					headContent = []byte(content)
   854→				}
   855→			}
   856→			absPath := filepath.Join(repoRoot, file)
   857→			if content, err := os.ReadFile(absPath); err == nil { //nolint:gosec // absPath is repo root + relative path from git status
   858→				workContent = content
   859→			}
   860→			if headContent != nil && workContent != nil {
   861→				change.added, change.removed = computeDiffStats(headContent, workContent)
   862→			}
   863→		default:
   864→			continue
   865→		}
   866→
   867→		changes = append(changes, change)
   868→	}
   869→
   870→	if len(changes) == 0 {
   871→		return true, "", nil
   872→	}
   873→
   874→	// Sort changes by filename for consistent output
   875→	sort.Slice(changes, func(i, j int) bool {
   876→		return changes[i].filename < changes[j].filename
   877→	})
   878→
   879→	var msg strings.Builder
   880→	msg.WriteString("The following uncommitted changes will be reverted:\n")
   881→
   882→	totalAdded, totalRemoved := 0, 0
   883→	for _, c := range changes {
   884→		totalAdded += c.added
   885→		totalRemoved += c.removed
   886→
   887→		var stats string
   888→		switch {
   889→		case c.added > 0 && c.removed > 0:
   890→			stats = fmt.Sprintf("+%d/-%d", c.added, c.removed)
   891→		case c.added > 0:
   892→			stats = fmt.Sprintf("+%d", c.added)
   893→		case c.removed > 0:
   894→			stats = fmt.Sprintf("-%d", c.removed)
   895→		}
   896→
   897→		msg.WriteString(fmt.Sprintf("  %-10s %s", c.status+":", c.filename))
   898→		if stats != "" {
   899→			msg.WriteString(fmt.Sprintf(" (%s)", stats))
   900→		}
   901→		msg.WriteString("\n")
   902→	}
   903→
   904→	if totalAdded > 0 || totalRemoved > 0 {
   905→		msg.WriteString(fmt.Sprintf("\nTotal: +%d/-%d lines\n", totalAdded, totalRemoved))
   906→	}
   907→
   908→	return true, msg.String(), nil
   909→}

   911→// countLines counts the number of lines in content.
   912→func countLines(content []byte) int {
   913→	if len(content) == 0 {
   914→		return 0
   915→	}
   916→	count := 1
   917→	for _, b := range content {
   918→		if b == '\n' {
   919→			count++
   920→		}
   921→	}
   922→	// Don't count trailing newline as extra line
   923→	if len(content) > 0 && content[len(content)-1] == '\n' {
   924→		count--
   925→	}
   926→	return count
   927→}

   929→// computeDiffStats computes added and removed line counts between old and new content.
   930→// Uses a simple line-based diff algorithm.
   931→func computeDiffStats(oldContent, newContent []byte) (added, removed int) {
   932→	oldLines := splitLines(oldContent)
   933→	newLines := splitLines(newContent)
   934→
   935→	// Build a set of old lines with counts
   936→	oldSet := make(map[string]int)
   937→	for _, line := range oldLines {
   938→		oldSet[line]++
   939→	}
   940→
   941→	// Check which new lines are truly new
   942→	for _, line := range newLines {
   943→		if oldSet[line] > 0 {
   944→			oldSet[line]--
   945→		} else {
   946→			added++
   947→		}
   948→	}
   949→
   950→	// Remaining old lines are removed
   951→	for _, count := range oldSet {
   952→		removed += count
   953→	}
   954→
   955→	return added, removed
   956→}

   958→// splitLines splits content into lines, preserving empty lines.
   959→func splitLines(content []byte) []string {
   960→	if len(content) == 0 {
   961→		return nil
   962→	}
   963→	s := string(content)
   964→	// Remove trailing newline to avoid empty last element
   965→	s = strings.TrimSuffix(s, "\n")
   966→	return strings.Split(s, "\n")
   967→}

   985→// StageFiles stages modified, new, and deleted files to the git worktree.
   986→//
   987→// This function handles three categories of file changes:
   988→//  1. Modified files: existing files that have been changed
   989→//  2. New files: files that were created during the session
   990→//  3. Deleted files: files that were removed during the session
   991→//
   992→// Error Handling Strategy:
   993→//   - Individual file staging errors are logged to stderr but don't fail the operation
   994→//   - This ensures that partial staging succeeds even if some files have issues
   995→//   - If a modified file no longer exists, it's treated as a deletion
   996→//
   997→// The stageCtx parameter is used for user-facing messages to indicate whether
   998→// this is staging for a session checkpoint or a task checkpoint.
   999→func StageFiles(worktree *git.Worktree, modified, newFiles, deleted []string, stageCtx StageFilesContext) {
  1000→	// Get repo root for resolving file paths
  1001→	// This is critical because fileExists() uses os.Stat() which resolves relative to CWD,
  1002→	// but worktree.Add/Remove resolve relative to repo root. If CWD != repo root,
  1003→	// fileExists() could return false for existing files, causing worktree.Remove()
  1004→	// to incorrectly delete them.
  1005→	repoRoot := worktree.Filesystem.Root()
  1006→
  1007→	// Stage modified files
  1008→	for _, file := range modified {
  1009→		// Resolve path relative to repo root for existence check
  1010→		absPath := filepath.Join(repoRoot, file)
  1011→		if fileExists(absPath) {
  1012→			if _, err := worktree.Add(file); err != nil {
  1013→				fmt.Fprintf(os.Stderr, "  Failed to stage %s: %v\n", file, err)
  1014→			} else {
  1015→				fmt.Fprintf(os.Stderr, "  Staged: %s\n", file)
  1016→			}
  1017→		} else {
  1018→			// File was deleted - stage the deletion
  1019→			if _, err := worktree.Remove(file); err != nil {
  1020→				fmt.Fprintf(os.Stderr, "  File not found or deleted: %s\n", file)
  1021→			}
  1022→		}
  1023→	}
  1024→
  1025→	// Stage new files
  1026→	if len(newFiles) > 0 {
  1027→		fmt.Fprintf(os.Stderr, "Staging %d new files created during %s:\n", len(newFiles), stageCtx)
  1028→		for _, file := range newFiles {
  1029→			if _, err := worktree.Add(file); err != nil {
  1030→				fmt.Fprintf(os.Stderr, "  Failed to stage %s: %v\n", file, err)
  1031→			} else {
  1032→				fmt.Fprintf(os.Stderr, "  Staged new file: %s\n", file)
  1033→			}
  1034→		}
  1035→	}
  1036→
  1037→	// Stage deleted files
  1038→	for _, file := range deleted {
  1039→		if _, err := worktree.Remove(file); err != nil {
  1040→			fmt.Fprintf(os.Stderr, "  Failed to stage deleted file %s: %v\n", file, err)
  1041→		} else {
  1042→			fmt.Fprintf(os.Stderr, "  Staged deleted file: %s\n", file)
  1043→		}
  1044→	}
  1045→}

  1136→// ErrBranchNotFound is returned by DeleteBranchCLI when the branch does not exist.
  1137→var ErrBranchNotFound = errors.New("branch not found")

  1139→// DeleteBranchCLI deletes a git branch using the git CLI.
  1140→// Uses `git branch -D` instead of go-git's RemoveReference because go-git v5
  1141→// doesn't properly persist deletions when refs are packed (.git/packed-refs)
  1142→// or in a worktree context. This is the same class of go-git v5 bug that
  1143→// affects checkout and reset --hard (see HardResetWithProtection).
  1144→//
  1145→// Returns ErrBranchNotFound if the branch does not exist, allowing callers
  1146→// to use errors.Is for idempotent deletion patterns.
  1147→func DeleteBranchCLI(branchName string) error {
  1148→	ctx := context.Background()
  1149→
  1150→	// Pre-check: verify the branch exists so callers get a structured error
  1151→	// instead of parsing git's output string (which varies across locales).
  1152→	// git show-ref exits 1 for "not found" and 128+ for fatal errors (corrupt
  1153→	// repo, permissions, not a git directory). Only map exit code 1 to
  1154→	// ErrBranchNotFound; propagate other failures as-is.
  1155→	check := exec.CommandContext(ctx, "git", "show-ref", "--verify", "--quiet", "refs/heads/"+branchName) //nolint:gosec // branchName comes from internal shadow branch naming
  1156→	if err := check.Run(); err != nil {
  1157→		var exitErr *exec.ExitError
  1158→		if errors.As(err, &exitErr) && exitErr.ExitCode() == 1 {
  1159→			return fmt.Errorf("%w: %s", ErrBranchNotFound, branchName)
  1160→		}
  1161→		return fmt.Errorf("failed to check branch %s: %w", branchName, err)
  1162→	}
  1163→
  1164→	cmd := exec.CommandContext(ctx, "git", "branch", "-D", "--", branchName)
  1165→	if output, err := cmd.CombinedOutput(); err != nil {
  1166→		return fmt.Errorf("failed to delete branch %s: %s: %w", branchName, strings.TrimSpace(string(output)), err)
  1167→	}
  1168→	return nil
  1169→}

  1182→// HardResetWithProtection performs a git reset --hard to the specified commit.
  1183→// Uses the git CLI instead of go-git because go-git's HardReset incorrectly

   216→					sessionPrompts = []string{sessionPrompt}
   217→				}
   218→			}
   219→		}
   220→
   221→		points = append(points, RewindPoint{
   222→			ID:             c.Hash.String(),
   223→			Message:        message,
   224→			Date:           c.Author.When,
   225→			IsLogsOnly:     true,
   226→			CheckpointID:   cpInfo.CheckpointID,
   227→			Agent:          cpInfo.Agent,
   228→			SessionID:      cpInfo.SessionID,
   229→			SessionPrompt:  sessionPrompt,
   230→			SessionCount:   cpInfo.SessionCount,
   231→			SessionIDs:     cpInfo.SessionIDs,
   232→			SessionPrompts: sessionPrompts,
   233→		})
   234→
   235→		return nil
   236→	})
   237→
   238→	if err != nil && !errors.Is(err, errStop) {
   239→		return nil, fmt.Errorf("error iterating commits: %w", err)
   240→	}
   241→
   242→	if len(points) > limit {
   243→		points = points[:limit]
   244→	}
   245→
   246→	return points, nil
   247→}

   252→func (s *ManualCommitStrategy) Rewind(point RewindPoint) error {
   253→	repo, err := OpenRepository()
   254→	if err != nil {
   255→		return fmt.Errorf("failed to open git repository: %w", err)
   256→	}
   257→
   258→	// Get the checkpoint commit
   259→	commitHash := plumbing.NewHash(point.ID)
   260→	commit, err := repo.CommitObject(commitHash)
   261→	if err != nil {
   262→		return fmt.Errorf("failed to get commit: %w", err)
   263→	}
   264→
   265→	tree, err := commit.Tree()
   266→	if err != nil {
   267→		return fmt.Errorf("failed to get tree: %w", err)
   268→	}
   269→
   270→	// Reset the shadow branch to the rewound checkpoint
   271→	// This ensures the next checkpoint will only include prompts from this point forward
   272→	if err := s.resetShadowBranchToCheckpoint(repo, commit); err != nil {
   273→		// Log warning but don't fail - file restoration is the primary operation
   274→		fmt.Fprintf(os.Stderr, "[entire] Warning: failed to reset shadow branch: %v\n", err)
   275→	}
   276→
   277→	// Load session state to get untracked files that existed at session start
   278→	sessionID, hasSessionTrailer := trailers.ParseSession(commit.Message)
   279→	var preservedUntrackedFiles map[string]bool
   280→	if hasSessionTrailer {
   281→		state, stateErr := s.loadSessionState(sessionID)
   282→		if stateErr == nil && state != nil && len(state.UntrackedFilesAtStart) > 0 {
   283→			preservedUntrackedFiles = make(map[string]bool)
   284→			for _, f := range state.UntrackedFilesAtStart {
   285→				preservedUntrackedFiles[f] = true
   286→			}
   287→		}
   288→	}
   289→
   290→	// Build set of files in the checkpoint tree (excluding metadata)
   291→	checkpointFiles := make(map[string]bool)
   292→	err = tree.Files().ForEach(func(f *object.File) error {
   293→		if !strings.HasPrefix(f.Name, entireDir) {
   294→			checkpointFiles[f.Name] = true
   295→		}
   296→		return nil
   297→	})
   298→	if err != nil {
   299→		return fmt.Errorf("failed to list checkpoint files: %w", err)
   300→	}
   301→
   302→	// Get HEAD tree to identify tracked files
   303→	head, err := repo.Head()
   304→	if err != nil {
   305→		return fmt.Errorf("failed to get HEAD: %w", err)
   306→	}
   307→	headCommit, err := repo.CommitObject(head.Hash())
   308→	if err != nil {
   309→		return fmt.Errorf("failed to get HEAD commit: %w", err)
   310→	}
   311→	headTree, err := headCommit.Tree()
   312→	if err != nil {
   313→		return fmt.Errorf("failed to get HEAD tree: %w", err)
   314→	}
   315→
   316→	// Build set of files tracked in HEAD
   317→	trackedFiles := make(map[string]bool)
   318→	//nolint:errcheck // Error is not critical for rewind
   319→	_ = headTree.Files().ForEach(func(f *object.File) error {
   320→		trackedFiles[f.Name] = true
   321→		return nil
   322→	})
   323→
   324→	// Get repository root to walk from there
   325→	repoRoot, err := GetWorktreePath()
   326→	if err != nil {
   327→		repoRoot = "." // Fallback to current directory
   328→	}
   329→
   330→	// Find and delete untracked files that aren't in the checkpoint
   331→	// These are likely files created by the agent in later checkpoints
   332→	err = filepath.Walk(repoRoot, func(path string, info os.FileInfo, walkErr error) error {
   333→		if walkErr != nil {
   334→			return nil //nolint:nilerr // Skip filesystem errors during walk
   335→		}
   336→
   337→		// Get path relative to repo root
   338→		relPath, relErr := filepath.Rel(repoRoot, path)
   339→		if relErr != nil {
   340→			return nil //nolint:nilerr // Skip paths we can't make relative
   341→		}
   342→
   343→		// Skip directories and protected paths
   344→		if info.IsDir() {
   345→			if isProtectedPath(relPath) {
   346→				return filepath.SkipDir
   347→			}
   348→			return nil
   349→		}
   350→
   351→		// Skip files in protected directories
   352→		if isProtectedPath(relPath) {
   353→			return nil
   354→		}
   355→
   356→		// If file is in checkpoint, it will be restored
   357→		if checkpointFiles[relPath] {
   358→			return nil
   359→		}
   360→
   361→		// If file is tracked in HEAD, don't delete (user's committed work)
   362→		if trackedFiles[relPath] {
   363→			return nil
   364→		}
   365→
   366→		// If file existed at session start, preserve it (untracked user files)
   367→		if preservedUntrackedFiles[relPath] {
   368→			return nil
   369→		}
   370→
   371→		// File is untracked and not in checkpoint - delete it (use absolute path)
   372→		if removeErr := os.Remove(path); removeErr == nil {
   373→			fmt.Fprintf(os.Stderr, "  Deleted: %s\n", relPath)
   374→		}
   375→
   376→		return nil
   377→	})
   378→	if err != nil {
   379→		// Non-fatal - continue with restoration
   380→		fmt.Fprintf(os.Stderr, "Warning: error walking directory: %v\n", err)
   381→	}
   382→
   383→	// Restore files from checkpoint
   384→	err = tree.Files().ForEach(func(f *object.File) error {
   385→		// Skip metadata directories - these are for checkpoint storage, not working dir
   386→		if strings.HasPrefix(f.Name, entireDir) {
   387→			return nil
   388→		}
   389→
   390→		contents, err := f.Contents()
   391→		if err != nil {
   392→			return fmt.Errorf("failed to read file %s: %w", f.Name, err)
   393→		}
   394→
   395→		// Ensure directory exists
   396→		dir := filepath.Dir(f.Name)
   397→		if dir != "." {
   398→			//nolint:gosec // G301: Need 0o755 for user directories during rewind
   399→			if err := os.MkdirAll(dir, 0o755); err != nil {
   400→				return fmt.Errorf("failed to create directory %s: %w", dir, err)
   401→			}
   402→		}
   403→
   404→		// Write file with appropriate permissions
   405→		var perm os.FileMode = 0o644
   406→		if f.Mode == filemode.Executable {
   407→			perm = 0o755
   408→		}
   409→		if err := os.WriteFile(f.Name, []byte(contents), perm); err != nil {
   410→			return fmt.Errorf("failed to write file %s: %w", f.Name, err)
   411→		}
   412→
   413→		fmt.Fprintf(os.Stderr, "  Restored: %s\n", f.Name)
   414→		return nil
   415→	})
   416→	if err != nil {
   417→		return fmt.Errorf("failed to iterate tree files: %w", err)
   418→	}
   419→
   420→	fmt.Println()
   421→	if len(point.ID) >= 7 {
   422→		fmt.Printf("Restored files from shadow commit %s\n", point.ID[:7])
   423→	} else {
   424→		fmt.Printf("Restored files from shadow commit %s\n", point.ID)
   425→	}
   426→	fmt.Println()
   427→
   428→	return nil
   429→}

   431→// resetShadowBranchToCheckpoint resets the shadow branch HEAD to the given checkpoint.
   432→// This ensures that when the user commits after rewinding, the next checkpoint will only
   433→// include prompts from the rewound point, not prompts from later checkpoints.
   434→func (s *ManualCommitStrategy) resetShadowBranchToCheckpoint(repo *git.Repository, commit *object.Commit) error {
   435→	// Extract session ID from the checkpoint commit's Entire-Session trailer
   436→	sessionID, found := trailers.ParseSession(commit.Message)
   437→	if !found {
   438→		return errors.New("checkpoint has no Entire-Session trailer")
   439→	}
   440→
   441→	// Load session state to get the shadow branch name
   442→	state, err := s.loadSessionState(sessionID)
   443→	if err != nil {
   444→		return fmt.Errorf("failed to load session state: %w", err)
   445→	}
   446→	if state == nil {
   447→		return fmt.Errorf("session %s not found", sessionID)
   448→	}
   449→
   450→	// Reset the shadow branch to the checkpoint commit
   451→	shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   452→	refName := plumbing.NewBranchReferenceName(shadowBranchName)
   453→
   454→	// Update the reference to point to the checkpoint commit
   455→	ref := plumbing.NewHashReference(refName, commit.Hash)
   456→	if err := repo.Storer.SetReference(ref); err != nil {
   457→		return fmt.Errorf("failed to update shadow branch: %w", err)
   458→	}
   459→
   460→	fmt.Fprintf(os.Stderr, "[entire] Reset shadow branch %s to checkpoint %s\n", shadowBranchName, commit.Hash.String()[:7])
   461→	return nil
   462→}

   614→// RestoreLogsOnly restores session logs from a logs-only rewind point.
   615→// This fetches the transcript from entire/checkpoints/v1 and writes it to the agent's session directory.
   616→// Does not modify the working directory.
   617→// When multiple sessions were condensed to the same checkpoint, ALL sessions are restored.
   618→// If force is false, prompts for confirmation when local logs have newer timestamps.
   619→// Returns info about each restored session so callers can print correct per-session resume commands.
   620→func (s *ManualCommitStrategy) RestoreLogsOnly(point RewindPoint, force bool) ([]RestoredSession, error) {
   621→	if !point.IsLogsOnly {
   622→		return nil, errors.New("not a logs-only rewind point")
   623→	}
   624→
   625→	if point.CheckpointID.IsEmpty() {
   626→		return nil, errors.New("missing checkpoint ID")
   627→	}
   628→
   629→	// Get checkpoint store
   630→	store, err := s.getCheckpointStore()
   631→	if err != nil {
   632→		return nil, fmt.Errorf("failed to get checkpoint store: %w", err)
   633→	}
   634→
   635→	// Read checkpoint summary to get session count
   636→	summary, err := store.ReadCommitted(context.Background(), point.CheckpointID)
   637→	if err != nil {
   638→		return nil, fmt.Errorf("failed to read checkpoint: %w", err)
   639→	}
   640→	if summary == nil {
   641→		return nil, fmt.Errorf("checkpoint not found: %s", point.CheckpointID)
   642→	}
   643→
   644→	// Get repo root for agent session directory lookup
   645→	repoRoot, err := paths.RepoRoot()
   646→	if err != nil {
   647→		return nil, fmt.Errorf("failed to get repository root: %w", err)
   648→	}
   649→
   650→	// Check for newer local logs if not forcing
   651→	if !force {
   652→		sessions := s.classifySessionsForRestore(context.Background(), repoRoot, store, point.CheckpointID, summary)
   653→		hasConflicts := false
   654→		for _, sess := range sessions {
   655→			if sess.Status == StatusLocalNewer {
   656→				hasConflicts = true
   657→				break
   658→			}
   659→		}
   660→		if hasConflicts {
   661→			shouldOverwrite, promptErr := PromptOverwriteNewerLogs(sessions)
   662→			if promptErr != nil {
   663→				return nil, promptErr
   664→			}
   665→			if !shouldOverwrite {
   666→				fmt.Fprintf(os.Stderr, "Resume cancelled. Local session logs preserved.\n")
   667→				return nil, nil
   668→			}
   669→		}
   670→	}
   671→
   672→	// Count sessions to restore
   673→	totalSessions := len(summary.Sessions)
   674→	if totalSessions > 1 {
   675→		fmt.Fprintf(os.Stderr, "Restoring %d sessions from checkpoint:\n", totalSessions)
   676→	}
   677→
   678→	// Restore all sessions (oldest to newest, using 0-based indexing)
   679→	var restored []RestoredSession
   680→	for i := range totalSessions {
   681→		content, readErr := store.ReadSessionContent(context.Background(), point.CheckpointID, i)
   682→		if readErr != nil {
   683→			fmt.Fprintf(os.Stderr, "  Warning: failed to read session %d: %v\n", i, readErr)
   684→			continue
   685→		}
   686→		if content == nil || len(content.Transcript) == 0 {
   687→			continue
   688→		}
   689→
   690→		sessionID := content.Metadata.SessionID
   691→		if sessionID == "" {
   692→			fmt.Fprintf(os.Stderr, "  Warning: session %d has no session ID, skipping\n", i)
   693→			continue
   694→		}
   695→
   696→		// Resolve per-session agent from metadata — skip if agent is unknown
   697→		if content.Metadata.Agent == "" {
   698→			fmt.Fprintf(os.Stderr, "  Warning: session %d (%s) has no agent metadata, skipping (cannot determine target directory)\n", i, sessionID)
   699→			continue
   700→		}
   701→		sessionAgent, agErr := ResolveAgentForRewind(content.Metadata.Agent)
   702→		if agErr != nil {
   703→			fmt.Fprintf(os.Stderr, "  Warning: session %d (%s) has unknown agent %q, skipping\n", i, sessionID, content.Metadata.Agent)
   704→			continue
   705→		}
   706→
   707→		sessionAgentDir, dirErr := sessionAgent.GetSessionDir(repoRoot)
   708→		if dirErr != nil {
   709→			fmt.Fprintf(os.Stderr, "  Warning: failed to get session dir for session %d: %v\n", i, dirErr)
   710→			continue
   711→		}
   712→
   713→		sessionFile := ResolveSessionFilePath(sessionID, sessionAgent, sessionAgentDir)
   714→
   715→		// Get first prompt for display
   716→		promptPreview := ExtractFirstPrompt(content.Prompts)
   717→
   718→		if totalSessions > 1 {
   719→			isLatest := i == totalSessions-1
   720→			if promptPreview != "" {
   721→				if isLatest {
   722→					fmt.Fprintf(os.Stderr, "  Session %d (latest): %s\n", i+1, promptPreview)
   723→				} else {
   724→					fmt.Fprintf(os.Stderr, "  Session %d: %s\n", i+1, promptPreview)
   725→				}
   726→			}
   727→			fmt.Fprintf(os.Stderr, "    Writing to: %s\n", sessionFile)
   728→		} else {
   729→			fmt.Fprintf(os.Stderr, "Writing transcript to: %s\n", sessionFile)
   730→		}
   731→
   732→		// Ensure parent directory exists (session file may be in a different dir than sessionAgentDir)
   733→		if mkdirErr := os.MkdirAll(filepath.Dir(sessionFile), 0o750); mkdirErr != nil {
   734→			fmt.Fprintf(os.Stderr, "    Warning: failed to create directory: %v\n", mkdirErr)
   735→			continue
   736→		}
   737→
   738→		if writeErr := os.WriteFile(sessionFile, content.Transcript, 0o600); writeErr != nil {
   739→			if totalSessions > 1 {
   740→				fmt.Fprintf(os.Stderr, "    Warning: failed to write transcript: %v\n", writeErr)
   741→				continue
   742→			}
   743→			return nil, fmt.Errorf("failed to write transcript: %w", writeErr)
   744→		}
   745→
   746→		restored = append(restored, RestoredSession{
   747→			SessionID: sessionID,
   748→			Agent:     sessionAgent.Type(),
   749→			Prompt:    promptPreview,
   750→		})
   751→	}
   752→
   753→	return restored, nil
   754→}

   767→	if err != nil {

   769→	}
   770→	return ag, nil

   773→// ResolveSessionFilePath determines the correct file path for an agent's session transcript.
   774→// Checks session state for transcript_path first (needed for agents like Gemini that store
   775→// transcripts at paths that GetSessionDir can't reconstruct, e.g. SHA-256 hashed directories).
   776→// Falls back to the agent's ExtractAgentSessionID + ResolveSessionFile with fallbackSessionDir.

   883→// ClassifyTimestamps determines the restore status based on local and checkpoint timestamps.
   884→func ClassifyTimestamps(localTime, checkpointTime time.Time) SessionRestoreStatus {
   885→	// Local file doesn't exist (no timestamp found)
   886→	if localTime.IsZero() {
   887→		return StatusNew
   888→	}
   889→
   890→	// Can't determine checkpoint time - treat as new/safe
   891→	if checkpointTime.IsZero() {
   892→		return StatusNew
   893→	}
   894→
   895→	// Compare timestamps
   896→	if localTime.After(checkpointTime) {
   897→		return StatusLocalNewer
   898→	}
   899→	if checkpointTime.After(localTime) {
   900→		return StatusCheckpointNewer
   901→	}
   902→	return StatusUnchanged
   903→}
