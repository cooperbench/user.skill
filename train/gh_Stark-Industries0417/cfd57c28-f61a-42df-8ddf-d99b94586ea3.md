   672→// Shell completion stanza management.
   673→// The stanza is a versioned, delimited block written to ~/.zshrc or ~/.bashrc:

   699→// findStanzaVersion scans content for the opening stanza marker and returns
   700→// the version number. Returns 0 if no stanza is found.
   701→func findStanzaVersion(content string) int {

   704→		var version int
   705→		if _, err := fmt.Sscanf(trimmed, shellCompletionOpenFormat, &version); err == nil {
   706→			return version
   707→		}

   709→	return 0

   713→func checkStanzaStatus(content string, currentVersion int) stanzaStatus {

   732→// countStanzaOpeners counts how many opening stanza markers exist in content.
   733→func countStanzaOpeners(content string) int {
   734→	count := 0
   735→	for line := range strings.SplitSeq(content, "\n") {
   736→		trimmed := strings.TrimSpace(line)
   737→		var version int
   738→		if _, err := fmt.Sscanf(trimmed, shellCompletionOpenFormat, &version); err == nil {
   739→			count++
   740→		}
   741→	}

   745→// replaceStanza finds the open+close markers in content and replaces the block
   746→// with a new stanza. Returns errStanzaCorrupted if open found without close,
   747→// errMultipleStanzas if count > 1. Returns unchanged content with nil error if
   748→// no stanza found.
   749→func replaceStanza(content string, version int, completionLine string) (string, error) {
   750→	openerCount := countStanzaOpeners(content)
   751→	if openerCount == 0 {
   752→		return content, nil
   753→	}
   754→	if openerCount > 1 {
   755→		return content, errMultipleStanzas
   756→	}
   757→
   758→	lines := strings.Split(content, "\n")
   759→	openIdx := -1
   760→	closeIdx := -1
   761→	for i, line := range lines {
   762→		trimmed := strings.TrimSpace(line)
   763→		var v int
   764→		if _, err := fmt.Sscanf(trimmed, shellCompletionOpenFormat, &v); err == nil {
   765→			openIdx = i
   766→		}
   767→		if openIdx >= 0 && trimmed == shellCompletionClose {
   768→			closeIdx = i
   769→			break
   770→		}
   771→	}
   772→
   773→	if openIdx >= 0 && closeIdx < 0 {
   774→		return content, errStanzaCorrupted
   775→	}
   776→
   777→	newStanza := buildStanza(version, completionLine)
   778→	var result []string
   779→	result = append(result, lines[:openIdx]...)
   780→	result = append(result, newStanza)
   781→	result = append(result, lines[closeIdx+1:]...)
   782→	return strings.Join(result, "\n"), nil
   783→}

   785→// removeStanza finds and removes the stanza block from content.
   786→// Returns errStanzaCorrupted if open found without close,
   787→// errMultipleStanzas if count > 1. Returns unchanged content with nil error if
   788→// no stanza found.
   789→func removeStanza(content string) (string, error) {
   790→	openerCount := countStanzaOpeners(content)
   791→	if openerCount == 0 {
   792→		return content, nil
   793→	}
   794→	if openerCount > 1 {
   795→		return content, errMultipleStanzas
   796→	}
   797→
   798→	lines := strings.Split(content, "\n")
   799→	openIdx := -1
   800→	closeIdx := -1
   801→	for i, line := range lines {
   802→		trimmed := strings.TrimSpace(line)
   803→		var v int
   804→		if _, err := fmt.Sscanf(trimmed, shellCompletionOpenFormat, &v); err == nil {
   805→			openIdx = i
   806→		}
   807→		if openIdx >= 0 && trimmed == shellCompletionClose {
   808→			closeIdx = i
   809→			break
   810→		}
   811→	}
   812→
   813→	if openIdx >= 0 && closeIdx < 0 {
   814→		return content, errStanzaCorrupted
   815→	}
   816→
   817→	var result []string
   818→	result = append(result, lines[:openIdx]...)
   819→	// Skip blank line immediately after stanza if present
   820→	tail := lines[closeIdx+1:]
   821→	if len(tail) > 0 && strings.TrimSpace(tail[0]) == "" {
   822→		tail = tail[1:]
   823→	}
   824→	result = append(result, tail...)
   825→	return strings.Join(result, "\n"), nil
   826→}
