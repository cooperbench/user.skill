   333→		if err != nil {
   334→			// Checkpoint exists in commit but no metadata found - skip this commit
   335→			return nil //nolint:nilerr // Intentional: skip commits without metadata
   336→		}
   337→
   338→		message := strings.Split(c.Message, "\n")[0]
   339→
   340→		// Determine if this is a full rewind or logs-only
   341→		// Full rewind is allowed if commit is only on this branch (not reachable from main)
   342→		isLogsOnly := false
   343→		if mainBranchHash != plumbing.ZeroHash {
   344→			if IsAncestorOf(repo, c.Hash, mainBranchHash) {
   345→				isLogsOnly = true
   346→			}
   347→		}
   348→
   349→		// Build metadata path - for task checkpoints, include the task path
   350→		metadataDir := checkpointPath
   351→		if metadata.IsTask && metadata.ToolUseID != "" {
   352→			metadataDir = checkpointPath + "/tasks/" + metadata.ToolUseID
   353→		}
   354→
   355→		// Read session prompt from metadata tree
   356→		sessionPrompt := ReadSessionPromptFromTree(metadataTree, checkpointPath)
   357→
   358→		points = append(points, RewindPoint{
   359→			ID:               c.Hash.String(),
   360→			Message:          message,
   361→			MetadataDir:      metadataDir,
   362→			Date:             c.Author.When,
   363→			IsLogsOnly:       isLogsOnly,
   364→			CheckpointID:     cpID,
   365→			IsTaskCheckpoint: metadata.IsTask,
   366→			ToolUseID:        metadata.ToolUseID,
   367→			Agent:            metadata.Agent,
   368→			SessionID:        metadata.SessionID,
   369→			SessionPrompt:    sessionPrompt,
   370→		})
   371→
   372→		return nil
   373→	})
   374→
   375→	if err != nil && !errors.Is(err, errStop) {
   376→		return nil, fmt.Errorf("failed to iterate commits: %w", err)
   377→	}
   378→
   379→	return points, nil
   380→}

   382→// findTaskMetadataPathForCommit looks up the task metadata path for a task checkpoint commit
   383→// by searching the entire/sessions branch commit history for the checkpoint directory.
   384→// Returns ("", nil) if metadata is not found - this is expected for commits without metadata.
   385→func (s *AutoCommitStrategy) findTaskMetadataPathForCommit(repo *git.Repository, commitSHA, toolUseID string) (string, error) {
   386→	// Get the entire/sessions branch
   387→	refName := plumbing.NewBranchReferenceName(paths.MetadataBranchName)

   389→	if err != nil {
   390→		if isNotFoundError(err) {
   391→			return "", nil // No metadata branch yet
   392→		}
   393→		return "", fmt.Errorf("failed to get metadata branch: %w", err)

   396→	// Search commit history for a commit referencing this code commit SHA and tool use ID
   397→	shortSHA := commitSHA
   398→	if len(shortSHA) > 7 {
   399→		shortSHA = shortSHA[:7]
   400→	}
   401→
   402→	iter, err := repo.Log(&git.LogOptions{From: ref.Hash()})

   404→		return "", fmt.Errorf("failed to get commit log: %w", err)
   405→	}

   408→	err = iter.ForEach(func(commit *object.Commit) error {

   411→			strings.Contains(commit.Message, toolUseID) {

   414→				foundTaskPath = taskPath
   415→				return errStop // Found it
   416→			}
   417→		}
   418→		return nil
   419→	})
   420→	if err != nil && !errors.Is(err, errStop) {
   421→		return "", fmt.Errorf("failed to iterate commits: %w", err)
   422→	}
   423→
   424→	return foundTaskPath, nil

   487→// SaveTaskCheckpoint creates a checkpoint commit for a completed task.
   488→// For auto-commit strategy:
   489→// 1. Commit code changes to active branch (no trailers - clean history)
   490→// 2. Commit task metadata to entire/sessions branch with checkpoint format
   491→func (s *AutoCommitStrategy) SaveTaskCheckpoint(ctx TaskCheckpointContext) error {
   492→	repo, err := OpenRepository()
   493→	if err != nil {
   494→		return fmt.Errorf("failed to open git repository: %w", err)
   495→	}
   496→
   497→	// Ensure entire/sessions branch exists
   498→	if err := EnsureMetadataBranch(repo); err != nil {
   499→		return fmt.Errorf("failed to ensure metadata branch: %w", err)
   500→	}
   501→
   502→	// Generate checkpoint ID for this task checkpoint
   503→	cpID, err := id.Generate()
   504→	if err != nil {
   505→		return fmt.Errorf("failed to generate checkpoint ID: %w", err)
   506→	}
   507→
   508→	// Step 1: Commit code changes to active branch with checkpoint ID trailer
   509→	// We do code first to avoid orphaned metadata if this step fails.
   510→	_, err = s.commitTaskCodeToActive(repo, ctx, cpID)
   511→	if err != nil {
   512→		return fmt.Errorf("failed to commit task code to active branch: %w", err)
   513→	}
   514→
   515→	// Step 2: Commit task metadata to entire/sessions branch at sharded path
   516→	_, err = s.commitTaskMetadataToMetadataBranch(repo, ctx, cpID)
   517→	if err != nil {
   518→		return fmt.Errorf("failed to commit task metadata to entire/sessions branch: %w", err)
   519→	}
   520→
   521→	// Log task checkpoint creation
   522→	logCtx := logging.WithComponent(context.Background(), "checkpoint")
   523→	attrs := []any{
   524→		slog.String("strategy", "auto-commit"),
   525→		slog.String("checkpoint_type", "task"),
   526→		slog.String("checkpoint_id", cpID.String()),
   527→		slog.String("checkpoint_uuid", ctx.CheckpointUUID),
   528→		slog.String("tool_use_id", ctx.ToolUseID),
   529→		slog.String("subagent_type", ctx.SubagentType),
   530→		slog.Int("modified_files", len(ctx.ModifiedFiles)),
   531→		slog.Int("new_files", len(ctx.NewFiles)),
   532→		slog.Int("deleted_files", len(ctx.DeletedFiles)),
   533→	}
   534→	if ctx.IsIncremental {
   535→		attrs = append(attrs,
   536→			slog.Bool("is_incremental", true),
   537→			slog.String("incremental_type", ctx.IncrementalType),
   538→			slog.Int("incremental_sequence", ctx.IncrementalSequence),
   539→		)
   540→	}
   541→	logging.Info(logCtx, "task checkpoint saved", attrs...)
   542→
   543→	return nil
   544→}

   546→// commitTaskCodeToActive commits task code changes to the active branch.
   547→// Adds an Entire-Checkpoint trailer for metadata lookup that survives amend/rebase.
   548→// Skips commit creation if there are no file changes.
   549→func (s *AutoCommitStrategy) commitTaskCodeToActive(repo *git.Repository, ctx TaskCheckpointContext, checkpointID id.CheckpointID) (plumbing.Hash, error) {
   550→	hasFileChanges := len(ctx.ModifiedFiles) > 0 || len(ctx.NewFiles) > 0 || len(ctx.DeletedFiles) > 0
   551→
   552→	// If no file changes, skip code commit
   553→	if !hasFileChanges {
   554→		fmt.Fprintf(os.Stderr, "No code changes to commit for task checkpoint\n")
   555→		// Return current HEAD hash so metadata can still be stored
   556→		head, err := repo.Head()
   557→		if err != nil {
   558→			return plumbing.ZeroHash, fmt.Errorf("failed to get HEAD: %w", err)
   559→		}
   560→		return head.Hash(), nil
   561→	}
   562→
   563→	worktree, err := repo.Worktree()
   564→	if err != nil {
   565→		return plumbing.ZeroHash, fmt.Errorf("failed to get worktree: %w", err)
   566→	}
   567→
   568→	// Stage code changes
   569→	StageFiles(worktree, ctx.ModifiedFiles, ctx.NewFiles, ctx.DeletedFiles, StageForTask)
   570→
   571→	// Build commit message with checkpoint trailer
   572→	shortToolUseID := ctx.ToolUseID
   573→	if len(shortToolUseID) > 12 {
   574→		shortToolUseID = shortToolUseID[:12]
   575→	}
   576→
   577→	var subject string
   578→	if ctx.IsIncremental {
   579→		subject = FormatIncrementalSubject(
   580→			ctx.IncrementalType,
   581→			ctx.SubagentType,
   582→			ctx.TaskDescription,
   583→			ctx.TodoContent,
   584→			ctx.IncrementalSequence,
   585→			shortToolUseID,
   586→		)
   587→	} else {
   588→		subject = FormatSubagentEndMessage(ctx.SubagentType, ctx.TaskDescription, shortToolUseID)
   589→	}
   590→
   591→	// Add checkpoint ID trailer to commit message
   592→	commitMsg := subject + "\n\n" + trailers.CheckpointTrailerKey + ": " + checkpointID.String()
   593→
   594→	author := &object.Signature{
   595→		Name:  ctx.AuthorName,
   596→		Email: ctx.AuthorEmail,
   597→		When:  time.Now(),
   598→	}
   599→
   600→	commitHash, err := commitOrHead(repo, worktree, commitMsg, author)
   601→	if err != nil {
   602→		return plumbing.ZeroHash, err
   603→	}
   604→
   605→	if ctx.IsIncremental {
   606→		fmt.Fprintf(os.Stderr, "Committed incremental checkpoint #%d to active branch (%s)\n", ctx.IncrementalSequence, commitHash.String()[:7])
   607→	} else {
   608→		fmt.Fprintf(os.Stderr, "Committed task checkpoint to active branch (%s)\n", commitHash.String()[:7])
   609→	}
   610→	return commitHash, nil
   611→}

   785→					// Strip the leading "/" for tree.File() which expects paths without leading slash

   803→		if err != nil {
   804→			return nil, fmt.Errorf("failed to read transcript: %w", err)
   805→		}
   806→		return []byte(content), nil
   807→	}
   808→
   809→	return nil, fmt.Errorf("invalid metadata path format: %s", metadataDir)
   810→}

    64→
    65→	response = strings.TrimSpace(strings.ToLower(response))
    66→	switch response {
    67→	case "y", "yes":
    68→		return true
    69→	case "n", "no":
    70→		return false
    71→	default:
    72→		// Empty or invalid input - use default
    73→		return defaultYes
    74→	}
    75→}

    77→// CommitMsg is called by the git commit-msg hook after the user edits the message.
    78→// If the message contains only our trailer (no actual user content), strip it
    79→// so git will abort the commit due to empty message.
    80→//
    81→//nolint:unparam // error return required by interface but hooks must return nil
    82→func (s *ManualCommitStrategy) CommitMsg(commitMsgFile string) error {
    83→	content, err := os.ReadFile(commitMsgFile) //nolint:gosec // Path comes from git hook
    84→	if err != nil {
    85→		return nil //nolint:nilerr // Hook must be silent on failure
    86→	}
    87→
    88→	message := string(content)
    89→
    90→	// Check if our trailer is present (ParseCheckpoint validates format, so found==true means valid)
    91→	if _, found := trailers.ParseCheckpoint(message); !found {
    92→		// No trailer, nothing to do
    93→		return nil
    94→	}
    95→
    96→	// Check if there's any user content (non-comment, non-trailer lines)
    97→	if !hasUserContent(message) {
    98→		// No user content - strip the trailer so git aborts
    99→		message = stripCheckpointTrailer(message)
   100→		if err := os.WriteFile(commitMsgFile, []byte(message), 0o600); err != nil {
   101→			return nil //nolint:nilerr // Hook must be silent on failure
   102→		}
   103→	}
   104→
   105→	return nil
   106→}

   108→// hasUserContent checks if the message has any content besides comments and our trailer.
   109→func hasUserContent(message string) bool {
   110→	trailerPrefix := trailers.CheckpointTrailerKey + ":"
   111→	for _, line := range strings.Split(message, "\n") {
   112→		trimmed := strings.TrimSpace(line)
   113→		// Skip empty lines
   114→		if trimmed == "" {
   115→			continue
   116→		}
   117→		// Skip comment lines
   118→		if strings.HasPrefix(trimmed, "#") {
   119→			continue
   120→		}
   121→		// Skip our trailer line
   122→		if strings.HasPrefix(trimmed, trailerPrefix) {
   123→			continue
   124→		}
   125→		// Found user content
   126→		return true
   127→	}
   128→	return false
   129→}

   131→// stripCheckpointTrailer removes the Entire-Checkpoint trailer line from the message.
   132→func stripCheckpointTrailer(message string) string {
   133→	trailerPrefix := trailers.CheckpointTrailerKey + ":"
   134→	var result []string
   135→	for _, line := range strings.Split(message, "\n") {
   136→		if !strings.HasPrefix(strings.TrimSpace(line), trailerPrefix) {
   137→			result = append(result, line)
   138→		}
   139→	}
   140→	return strings.Join(result, "\n")
   141→}

   143→// isGitSequenceOperation checks if git is currently in the middle of a rebase,
   144→// cherry-pick, or revert operation. During these operations, commits are being
   145→// replayed and should not be linked to agent sessions.
   146→//
   147→// Detects:
   148→//   - rebase: .git/rebase-merge/ or .git/rebase-apply/ directories
   149→//   - cherry-pick: .git/CHERRY_PICK_HEAD file
   150→//   - revert: .git/REVERT_HEAD file
   151→func isGitSequenceOperation() bool {
   152→	// Get git directory (handles worktrees and relative paths correctly)
   153→	gitDir, err := GetGitDir()
   154→	if err != nil {
   155→		return false // Can't determine, assume not in sequence operation
   156→	}
   157→
   158→	// Check for rebase state directories
   159→	if _, err := os.Stat(filepath.Join(gitDir, "rebase-merge")); err == nil {
   160→		return true
   161→	}
   162→	if _, err := os.Stat(filepath.Join(gitDir, "rebase-apply")); err == nil {
   163→		return true
   164→	}
   165→
   166→	// Check for cherry-pick and revert state files
   167→	if _, err := os.Stat(filepath.Join(gitDir, "CHERRY_PICK_HEAD")); err == nil {
   168→		return true
   169→	}
   170→	if _, err := os.Stat(filepath.Join(gitDir, "REVERT_HEAD")); err == nil {
   171→		return true
   172→	}
   173→
   174→	return false
   175→}

   177→// PrepareCommitMsg is called by the git prepare-commit-msg hook.
   178→// Adds an Entire-Checkpoint trailer to the commit message with a stable checkpoint ID.
   179→// Only adds a trailer if there's actually new session content to condense.
   180→// The actual condensation happens in PostCommit - if the user removes the trailer,
   181→// the commit will not be linked to the session (useful for "manual" commits).
   182→// For amended commits, preserves the existing checkpoint ID.
   183→//
   184→// The source parameter indicates how the commit was initiated:
   185→//   - "" or "template": normal editor flow - adds trailer with explanatory comment
   186→//   - "message": using -m or -F flag - prompts user interactively via /dev/tty
   187→//   - "merge", "squash", "commit": skip trailer entirely (auto-generated or amend commits)
   188→//
   189→
   190→func (s *ManualCommitStrategy) PrepareCommitMsg(commitMsgFile string, source string) error {
   191→	logCtx := logging.WithComponent(context.Background(), "checkpoint")
   192→
   193→	// Skip during rebase, cherry-pick, or revert operations
   194→	// These are replaying existing commits and should not be linked to agent sessions
   195→	if isGitSequenceOperation() {
   196→		logging.Debug(logCtx, "prepare-commit-msg: skipped during git sequence operation",
   197→			slog.String("strategy", "manual-commit"),
   198→			slog.String("source", source),
   199→		)
   200→		return nil
   201→	}
   202→
   203→	// Skip for merge, squash, and commit (amend) sources
   204→	// These are auto-generated or reusing existing messages - not from Claude sessions
   205→	switch source {
   206→	case "merge", "squash", "commit":
   207→		logging.Debug(logCtx, "prepare-commit-msg: skipped for source",
   208→			slog.String("strategy", "manual-commit"),
   209→			slog.String("source", source),
   210→		)
   211→		return nil
   212→	}
   213→	repo, err := OpenRepository()
   214→	if err != nil {
   215→		return nil //nolint:nilerr // Hook must be silent on failure
   216→	}
   217→
   218→	worktreePath, err := GetWorktreePath()
   219→	if err != nil {
   220→		return nil //nolint:nilerr // Hook must be silent on failure
   221→	}
   222→
   223→	// Find all active sessions for this worktree
   224→	// We match by worktree (not BaseCommit) because the user may have made
   225→	// intermediate commits without entering new prompts, causing HEAD to diverge
   226→	sessions, err := s.findSessionsForWorktree(worktreePath)
   227→	if err != nil || len(sessions) == 0 {
   228→		// No active sessions or error listing - silently skip (hooks must be resilient)
   229→		logging.Debug(logCtx, "prepare-commit-msg: no active sessions",
   230→			slog.String("strategy", "manual-commit"),
   231→			slog.String("source", source),
   232→		)
   233→		return nil //nolint:nilerr // Intentional: hooks must be silent on failure
   234→	}
   235→
   236→	// Check if any session has new content to condense
   237→	sessionsWithContent := s.filterSessionsWithNewContent(repo, sessions)
   238→
   239→	// Determine which checkpoint ID to use
   240→	var checkpointID id.CheckpointID
   241→	var hasNewContent bool
   242→	var reusedSession *SessionState
   243→
   244→	if len(sessionsWithContent) > 0 {
   245→		// New content exists - will generate new checkpoint ID below
   246→		hasNewContent = true
   247→	} else {
   248→		// No new content - check if any session has a LastCheckpointID to reuse
   249→		// This handles the case where user splits Claude's work across multiple commits
   250→		// Reuse if: LastCheckpointID exists AND (FilesTouched is empty OR files overlap)
   251→		// - FilesTouched empty: commits made before session stop, reuse checkpoint
   252→		// - FilesTouched populated: only reuse if files overlap (prevents unrelated commits from reusing)
   253→
   254→		// Get current HEAD to filter sessions
   255→		head, err := repo.Head()
   256→		if err != nil {
   257→			return nil //nolint:nilerr // Hook must be silent on failure
   258→		}
   259→		currentHeadHash := head.Hash().String()
   260→
   261→		// Filter to sessions where BaseCommit matches current HEAD
   262→		// This prevents reusing checkpoint IDs from old sessions
   263→		// Note: BaseCommit is kept current both when new content is condensed (in the
   264→		// condensation process) and when no new content is found (via PostCommit when
   265→		// reusing checkpoint IDs). If none match, we don't add a trailer rather than
   266→		// falling back to old sessions which could have stale checkpoint IDs.
   267→		var currentSessions []*SessionState
   268→		for _, session := range sessions {
   269→			if session.BaseCommit == currentHeadHash {
   270→				currentSessions = append(currentSessions, session)
   271→			}
   272→		}
   273→
   274→		if len(currentSessions) == 0 {
   275→			// No sessions match current HEAD - don't try to reuse checkpoint IDs
   276→			// from old sessions as they may be stale
   277→			logging.Debug(logCtx, "prepare-commit-msg: no sessions match current HEAD",
   278→				slog.String("strategy", "manual-commit"),
   279→				slog.String("source", source),
   280→				slog.String("current_head", currentHeadHash[:7]),
   281→				slog.Int("total_sessions", len(sessions)),
   282→			)
   283→			return nil
   284→		}
   285→
   286→		stagedFiles := getStagedFiles(repo)
   287→		for _, session := range currentSessions {
   288→			if !session.LastCheckpointID.IsEmpty() &&
   289→				(len(session.FilesTouched) == 0 || hasOverlappingFiles(stagedFiles, session.FilesTouched)) {
   290→				checkpointID = session.LastCheckpointID
   291→				reusedSession = session
   292→				break
   293→			}
   294→		}
   295→		if checkpointID.IsEmpty() {
   296→			// No new content and no previous checkpoint to reference (or staged files are unrelated)
   297→			logging.Debug(logCtx, "prepare-commit-msg: no content to link",
   298→				slog.String("strategy", "manual-commit"),
   299→				slog.String("source", source),
   300→				slog.Int("sessions_found", len(sessions)),
   301→				slog.Int("sessions_with_content", len(sessionsWithContent)),
   302→			)
   303→			return nil
   304→		}
   305→	}
   306→
   307→	// Read current commit message
   308→	content, err := os.ReadFile(commitMsgFile) //nolint:gosec // commitMsgFile is provided by git hook
   309→	if err != nil {
   310→		return nil //nolint:nilerr // Hook must be silent on failure
   311→	}
   312→
   313→	message := string(content)
   314→
   315→	// Get or generate checkpoint ID (ParseCheckpoint validates format, so found==true means valid)
   316→	if existingCpID, found := trailers.ParseCheckpoint(message); found {
   317→		// Trailer already exists (e.g., amend) - keep it
   318→		logging.Debug(logCtx, "prepare-commit-msg: trailer already exists",
   319→			slog.String("strategy", "manual-commit"),
   320→			slog.String("source", source),
   321→			slog.String("existing_checkpoint_id", existingCpID.String()),
   322→		)
   323→		return nil
   324→	}
   325→
   326→	if hasNewContent {
   327→		// New content: generate new checkpoint ID
   328→		cpID, err := id.Generate()
   329→		if err != nil {
   330→			return fmt.Errorf("failed to generate checkpoint ID: %w", err)
   331→		}
   332→		checkpointID = cpID
   333→	}
   334→	// Otherwise checkpointID is already set to LastCheckpointID from above
   335→
   336→	// Determine agent type and last prompt from session
   337→	agentType := DefaultAgentType // default for backward compatibility
   338→	var lastPrompt string
   339→	if hasNewContent && len(sessionsWithContent) > 0 {
   340→		session := sessionsWithContent[0]
   341→		if session.AgentType != "" {
   342→			agentType = session.AgentType
   343→		}
   344→		lastPrompt = s.getLastPrompt(repo, session)
   345→	} else if reusedSession != nil {
   346→		// Reusing checkpoint from existing session - get agent type and prompt from that session
   347→		if reusedSession.AgentType != "" {
   348→			agentType = reusedSession.AgentType
   349→		}
   350→		lastPrompt = s.getLastPrompt(repo, reusedSession)
   351→	}
   352→
   353→	// Prepare prompt for display: collapse newlines/whitespace, then truncate (rune-safe)
   354→	displayPrompt := stringutil.TruncateRunes(stringutil.CollapseWhitespace(lastPrompt), 80, "...")
   355→
   356→	// Add trailer differently based on commit source
   357→	if source == "message" {
   358→		// Using -m or -F: ask user interactively whether to add trailer
   359→		// (comments won't be stripped by git in this mode)
   360→
   361→		// Build context string for interactive prompt
   362→		var promptContext string
   363→		if displayPrompt != "" {
   364→			promptContext = "You have an active " + string(agentType) + " session.\nLast Prompt: " + displayPrompt
   365→		}
   366→
   367→		if !askConfirmTTY("Link this commit to "+string(agentType)+" session context?", promptContext, true) {
   368→			// User declined - don't add trailer
   369→			logging.Debug(logCtx, "prepare-commit-msg: user declined trailer",
   370→				slog.String("strategy", "manual-commit"),
   371→				slog.String("source", source),
   372→			)
   373→			return nil
   374→		}
   375→		message = addCheckpointTrailer(message, checkpointID)
   376→	} else {
   377→		// Normal editor flow: add trailer with explanatory comment (will be stripped by git)
   378→		message = addCheckpointTrailerWithComment(message, checkpointID, string(agentType), displayPrompt)
   379→	}
   380→
   381→	logging.Info(logCtx, "prepare-commit-msg: trailer added",
   382→		slog.String("strategy", "manual-commit"),
   383→		slog.String("source", source),
   384→		slog.String("checkpoint_id", checkpointID.String()),
   385→		slog.Bool("has_new_content", hasNewContent),
   386→	)
   387→
   388→	// Write updated message back
   389→	if err := os.WriteFile(commitMsgFile, []byte(message), 0o600); err != nil {
   390→		return nil //nolint:nilerr // Hook must be silent on failure
   391→	}
   392→
   393→	return nil
   394→}

   396→// PostCommit is called by the git post-commit hook after a commit is created.
   397→// Checks if the commit has an Entire-Checkpoint trailer and if so, condenses
   398→// session data from shadow branches to entire/sessions.
   399→// If the user removed the trailer during commit message editing, this is treated
   400→// as a "manual" commit and no condensation happens.
   401→//
   402→//nolint:unparam // error return required by interface but hooks must return nil
   403→func (s *ManualCommitStrategy) PostCommit() error {
   404→	logCtx := logging.WithComponent(context.Background(), "checkpoint")
   405→
   406→	repo, err := OpenRepository()
   407→	if err != nil {
   408→		return nil //nolint:nilerr // Hook must be silent on failure
   409→	}
   410→
   411→	// Get HEAD commit to check for trailer
   412→	head, err := repo.Head()
   413→	if err != nil {
   414→		return nil //nolint:nilerr // Hook must be silent on failure
   415→	}
   416→
   417→	commit, err := repo.CommitObject(head.Hash())
   418→	if err != nil {
   419→		return nil //nolint:nilerr // Hook must be silent on failure
   420→	}
   421→
   422→	// Check if commit has checkpoint trailer (ParseCheckpoint validates format)
   423→	checkpointID, found := trailers.ParseCheckpoint(commit.Message)
   424→	if !found {
   425→		// No trailer - user removed it, treat as manual commit
   426→		logging.Debug(logCtx, "post-commit: no checkpoint trailer",
   427→			slog.String("strategy", "manual-commit"),
   428→		)
   429→		return nil
   430→	}
   431→
   432→	worktreePath, err := GetWorktreePath()
   433→	if err != nil {
   434→		return nil //nolint:nilerr // Hook must be silent on failure
   435→	}
   436→
   437→	// Find all active sessions for this worktree
   438→	sessions, err := s.findSessionsForWorktree(worktreePath)
   439→	if err != nil || len(sessions) == 0 {
   440→		logging.Warn(logCtx, "post-commit: no active sessions despite trailer",
   441→			slog.String("strategy", "manual-commit"),
   442→			slog.String("checkpoint_id", checkpointID.String()),
   443→		)
   444→		return nil //nolint:nilerr // Intentional: hooks must be silent on failure
   445→	}
   446→
   447→	// Filter to sessions with new content
   448→	sessionsWithContent := s.filterSessionsWithNewContent(repo, sessions)
   449→	if len(sessionsWithContent) == 0 {
   450→		logging.Debug(logCtx, "post-commit: no new content to condense",
   451→			slog.String("strategy", "manual-commit"),
   452→			slog.String("checkpoint_id", checkpointID.String()),
   453→			slog.Int("sessions_found", len(sessions)),
   454→		)
   455→		// Still update BaseCommit for all sessions in this worktree
   456→		// This prevents stale BaseCommit when commits happen without condensation
   457→		// (e.g., when reusing a previous checkpoint ID for split commits)
   458→		newHead := head.Hash().String()
   459→		for _, state := range sessions {
   460→			if state.BaseCommit != newHead {
   461→				state.BaseCommit = newHead
   462→				if err := s.saveSessionState(state); err != nil {
   463→					logging.Warn(logCtx, "post-commit: failed to update session BaseCommit",
   464→						slog.String("session_id", state.SessionID),
   465→						slog.String("error", err.Error()),
   466→					)
   467→				}
   468→			}
   469→		}
   470→		return nil
   471→	}
   472→
   473→	// Track shadow branch names to clean up after successful condensation
   474→	shadowBranchesToDelete := make(map[string]struct{})
   475→
   476→	// Condense sessions that have new content
   477→	for _, state := range sessionsWithContent {
   478→		// Compute shadow branch name BEFORE condensation modifies state.BaseCommit
   479→		shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   480→
   481→		result, err := s.CondenseSession(repo, checkpointID, state)
   482→		if err != nil {
   483→			fmt.Fprintf(os.Stderr, "[entire] Warning: condensation failed for session %s: %v\n",
   484→				state.SessionID, err)
   485→			continue
   486→		}
   487→
   488→		// Track this shadow branch for cleanup
   489→		shadowBranchesToDelete[shadowBranchName] = struct{}{}
   490→
   491→		// Update session state for the new base commit
   492→		// After condensation, the session continues from the NEW commit (HEAD), so we:
   493→		// 1. Update BaseCommit to new HEAD - session now tracks from new commit
   494→		// 2. Reset StepCount to 0 - no checkpoints exist on new shadow branch yet
   495→		// 3. Update CheckpointTranscriptStart - track transcript position for detecting new content
   496→		// 4. Clear PromptAttributions - they were already used in condensation, reset for next cycle
   497→		//
   498→		// This is critical: if we don't update BaseCommit, listAllSessionStates will try
   499→		// to find shadow branch for old commit (which gets deleted), and since StepCount > 0,
   500→		// it will clean up (delete) the session state file. By updating to new HEAD with
   501→		// StepCount = 0, the session is preserved even without a shadow branch.
   502→		state.BaseCommit = head.Hash().String()
   503→		state.StepCount = 0
   504→		state.CheckpointTranscriptStart = result.TotalTranscriptLines
   505→
   506→		// Clear attribution tracking - condensation already used these values
   507→		// If we don't clear them, they'll be double-counted in the next condensation
   508→		state.PromptAttributions = nil
   509→		state.PendingPromptAttribution = nil
   510→
   511→		// Save checkpoint ID so subsequent commits without new content can reuse it
   512→		state.LastCheckpointID = checkpointID
   513→
   514→		if err := s.saveSessionState(state); err != nil {
   515→			fmt.Fprintf(os.Stderr, "[entire] Warning: failed to update session state: %v\n", err)
   516→		}
   517→
   518→		shortID := state.SessionID
   519→		if len(shortID) > 8 {
   520→			shortID = shortID[:8]
   521→		}
   522→		fmt.Fprintf(os.Stderr, "[entire] Condensed session %s: %s (%d checkpoints)\n",
   523→			shortID, result.CheckpointID, result.CheckpointsCount)
   524→
   525→		// Log condensation
   526→		logCtx := logging.WithComponent(context.Background(), "checkpoint")
   527→		logging.Info(logCtx, "session condensed",
   528→			slog.String("strategy", "manual-commit"),
   529→			slog.String("checkpoint_id", result.CheckpointID.String()),
   530→			slog.Int("checkpoints_condensed", result.CheckpointsCount),
   531→			slog.Int("transcript_lines", result.TotalTranscriptLines),
   532→		)
   533→	}
   534→
   535→	// Clean up shadow branches after successful condensation
   536→	// Data is now preserved on entire/sessions, so shadow branches are no longer needed
   537→	for shadowBranchName := range shadowBranchesToDelete {
   538→		if err := deleteShadowBranch(repo, shadowBranchName); err != nil {
   539→			fmt.Fprintf(os.Stderr, "[entire] Warning: failed to clean up %s: %v\n", shadowBranchName, err)
   540→		} else {
   541→			fmt.Fprintf(os.Stderr, "[entire] Cleaned up shadow branch: %s\n", shadowBranchName)
   542→
   543→			// Log branch cleanup
   544→			logCtx := logging.WithComponent(context.Background(), "checkpoint")
   545→			logging.Info(logCtx, "shadow branch deleted",
   546→				slog.String("strategy", "manual-commit"),
   547→				slog.String("shadow_branch", shadowBranchName),
   548→			)
   549→		}
   550→	}
   551→
   552→	return nil
   553→}

   555→// filterSessionsWithNewContent returns sessions that have new transcript content
   556→// beyond what was already condensed.
   557→func (s *ManualCommitStrategy) filterSessionsWithNewContent(repo *git.Repository, sessions []*SessionState) []*SessionState {
   558→	var result []*SessionState
   559→
   560→	for _, state := range sessions {
   561→		hasNew, err := s.sessionHasNewContent(repo, state)
   562→		if err != nil {
   563→			// On error, include the session (fail open for hooks)
   564→			result = append(result, state)
   565→			continue
   566→		}
   567→		if hasNew {
   568→			result = append(result, state)
   569→		}
   570→	}
   571→
   572→	return result
   573→}

   575→// sessionHasNewContent checks if a session has new transcript content
   576→// beyond what was already condensed.
   577→func (s *ManualCommitStrategy) sessionHasNewContent(repo *git.Repository, state *SessionState) (bool, error) {
   578→	// Get shadow branch
   579→	shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   580→	refName := plumbing.NewBranchReferenceName(shadowBranchName)
   581→	ref, err := repo.Reference(refName, true)
   582→	if err != nil {
   583→		// No shadow branch means no Stop has happened since the last condensation.
   584→		// However, the agent may have done work (including commits) without a Stop.
   585→		// Check the live transcript to detect this scenario.
   586→		return s.sessionHasNewContentFromLiveTranscript(repo, state)
   587→	}
   588→
   589→	commit, err := repo.CommitObject(ref.Hash())
   590→	if err != nil {
   591→		return false, fmt.Errorf("failed to get commit object: %w", err)
   592→	}
   593→
   594→	tree, err := commit.Tree()
   595→	if err != nil {
   596→		return false, fmt.Errorf("failed to get commit tree: %w", err)
   597→	}

   396→// PostCommit is called by the git post-commit hook after a commit is created.
   397→// Checks if the commit has an Entire-Checkpoint trailer and if so, condenses
   398→// session data from shadow branches to entire/sessions.
   399→// If the user removed the trailer during commit message editing, this is treated
   400→// as a "manual" commit and no condensation happens.
   401→//
   402→//nolint:unparam // error return required by interface but hooks must return nil
   403→func (s *ManualCommitStrategy) PostCommit() error {
   404→	logCtx := logging.WithComponent(context.Background(), "checkpoint")
   405→
   406→	repo, err := OpenRepository()
   407→	if err != nil {
   408→		return nil //nolint:nilerr // Hook must be silent on failure
   409→	}
   410→
   411→	// Get HEAD commit to check for trailer
   412→	head, err := repo.Head()
   413→	if err != nil {
   414→		return nil //nolint:nilerr // Hook must be silent on failure
   415→	}
   416→
   417→	commit, err := repo.CommitObject(head.Hash())
   418→	if err != nil {
   419→		return nil //nolint:nilerr // Hook must be silent on failure
   420→	}
   421→
   422→	// Check if commit has checkpoint trailer (ParseCheckpoint validates format)
   423→	checkpointID, found := trailers.ParseCheckpoint(commit.Message)
   424→	if !found {
   425→		// No trailer - user removed it, treat as manual commit
   426→		logging.Debug(logCtx, "post-commit: no checkpoint trailer",
   427→			slog.String("strategy", "manual-commit"),
   428→		)
   429→		return nil
   430→	}
   431→
   432→	worktreePath, err := GetWorktreePath()
   433→	if err != nil {
   434→		return nil //nolint:nilerr // Hook must be silent on failure
   435→	}
   436→
   437→	// Find all active sessions for this worktree
   438→	sessions, err := s.findSessionsForWorktree(worktreePath)
   439→	if err != nil || len(sessions) == 0 {
   440→		logging.Warn(logCtx, "post-commit: no active sessions despite trailer",
   441→			slog.String("strategy", "manual-commit"),
   442→			slog.String("checkpoint_id", checkpointID.String()),
   443→		)
   444→		return nil //nolint:nilerr // Intentional: hooks must be silent on failure
   445→	}
   446→
   447→	// Filter to sessions with new content
   448→	sessionsWithContent := s.filterSessionsWithNewContent(repo, sessions)
   449→	if len(sessionsWithContent) == 0 {
   450→		logging.Debug(logCtx, "post-commit: no new content to condense",
   451→			slog.String("strategy", "manual-commit"),
   452→			slog.String("checkpoint_id", checkpointID.String()),
   453→			slog.Int("sessions_found", len(sessions)),
   454→		)
   455→		// Still update BaseCommit for all sessions in this worktree
   456→		// This prevents stale BaseCommit when commits happen without condensation
   457→		// (e.g., when reusing a previous checkpoint ID for split commits)
   458→		newHead := head.Hash().String()
   459→		for _, state := range sessions {
   460→			if state.BaseCommit != newHead {
   461→				state.BaseCommit = newHead
   462→				if err := s.saveSessionState(state); err != nil {
   463→					logging.Warn(logCtx, "post-commit: failed to update session BaseCommit",
   464→						slog.String("session_id", state.SessionID),
   465→						slog.String("error", err.Error()),
   466→					)
   467→				}
   468→			}
   469→		}
   470→		return nil
   471→	}
   472→
   473→	// Track shadow branch names to clean up after successful condensation
   474→	shadowBranchesToDelete := make(map[string]struct{})
   475→
   476→	// Condense sessions that have new content
   477→	for _, state := range sessionsWithContent {
   478→		// Compute shadow branch name BEFORE condensation modifies state.BaseCommit
   479→		shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   480→
   481→		result, err := s.CondenseSession(repo, checkpointID, state)
   482→		if err != nil {
   483→			fmt.Fprintf(os.Stderr, "[entire] Warning: condensation failed for session %s: %v\n",
   484→				state.SessionID, err)
   485→			continue
   486→		}
   487→
   488→		// Track this shadow branch for cleanup
   489→		shadowBranchesToDelete[shadowBranchName] = struct{}{}
   490→
   491→		// Update session state for the new base commit
   492→		// After condensation, the session continues from the NEW commit (HEAD), so we:
   493→		// 1. Update BaseCommit to new HEAD - session now tracks from new commit
   494→		// 2. Reset StepCount to 0 - no checkpoints exist on new shadow branch yet
   495→		// 3. Update CheckpointTranscriptStart - track transcript position for detecting new content
   496→		// 4. Clear PromptAttributions - they were already used in condensation, reset for next cycle
   497→		//
   498→		// This is critical: if we don't update BaseCommit, listAllSessionStates will try
   499→		// to find shadow branch for old commit (which gets deleted), and since StepCount > 0,
   500→		// it will clean up (delete) the session state file. By updating to new HEAD with
   501→		// StepCount = 0, the session is preserved even without a shadow branch.
   502→		state.BaseCommit = head.Hash().String()
   503→		state.StepCount = 0
   504→		state.CheckpointTranscriptStart = result.TotalTranscriptLines
   505→
   506→		// Clear attribution tracking - condensation already used these values
   507→		// If we don't clear them, they'll be double-counted in the next condensation
   508→		state.PromptAttributions = nil
   509→		state.PendingPromptAttribution = nil
   510→
   511→		// Save checkpoint ID so subsequent commits without new content can reuse it
   512→		state.LastCheckpointID = checkpointID
   513→
   514→		if err := s.saveSessionState(state); err != nil {
   515→			fmt.Fprintf(os.Stderr, "[entire] Warning: failed to update session state: %v\n", err)
   516→		}
   517→
   518→		shortID := state.SessionID
   519→		if len(shortID) > 8 {
   520→			shortID = shortID[:8]
   521→		}
   522→		fmt.Fprintf(os.Stderr, "[entire] Condensed session %s: %s (%d checkpoints)\n",
   523→			shortID, result.CheckpointID, result.CheckpointsCount)
   524→
   525→		// Log condensation
   526→		logCtx := logging.WithComponent(context.Background(), "checkpoint")
   527→		logging.Info(logCtx, "session condensed",
   528→			slog.String("strategy", "manual-commit"),
   529→			slog.String("checkpoint_id", result.CheckpointID.String()),
   530→			slog.Int("checkpoints_condensed", result.CheckpointsCount),
   531→			slog.Int("transcript_lines", result.TotalTranscriptLines),
   532→		)
   533→	}
   534→
   535→	// Clean up shadow branches after successful condensation
   536→	// Data is now preserved on entire/sessions, so shadow branches are no longer needed
   537→	for shadowBranchName := range shadowBranchesToDelete {
   538→		if err := deleteShadowBranch(repo, shadowBranchName); err != nil {
   539→			fmt.Fprintf(os.Stderr, "[entire] Warning: failed to clean up %s: %v\n", shadowBranchName, err)
   540→		} else {
   541→			fmt.Fprintf(os.Stderr, "[entire] Cleaned up shadow branch: %s\n", shadowBranchName)
   542→
   543→			// Log branch cleanup
   544→			logCtx := logging.WithComponent(context.Background(), "checkpoint")
   545→			logging.Info(logCtx, "shadow branch deleted",
   546→				slog.String("strategy", "manual-commit"),
   547→				slog.String("shadow_branch", shadowBranchName),
   548→			)
   549→		}
   550→	}
   551→
   552→	return nil
   553→}

   566→		}

   569→		}
   570→	}

   575→// sessionHasNewContent checks if a session has new transcript content
   576→// beyond what was already condensed.
   577→func (s *ManualCommitStrategy) sessionHasNewContent(repo *git.Repository, state *SessionState) (bool, error) {
   578→	// Get shadow branch
   579→	shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   580→	refName := plumbing.NewBranchReferenceName(shadowBranchName)
   581→	ref, err := repo.Reference(refName, true)
   582→	if err != nil {
   583→		// No shadow branch means no Stop has happened since the last condensation.
   584→		// However, the agent may have done work (including commits) without a Stop.
   585→		// Check the live transcript to detect this scenario.
   586→		return s.sessionHasNewContentFromLiveTranscript(repo, state)
   587→	}
   588→
   589→	commit, err := repo.CommitObject(ref.Hash())
   590→	if err != nil {
   591→		return false, fmt.Errorf("failed to get commit object: %w", err)
   592→	}
   593→
   594→	tree, err := commit.Tree()
   595→	if err != nil {
   596→		return false, fmt.Errorf("failed to get commit tree: %w", err)
   597→	}
   598→
   599→	// Look for transcript file
   600→	metadataDir := paths.EntireMetadataDir + "/" + state.SessionID
   601→	var transcriptLines int
   602→
   603→	if file, fileErr := tree.File(metadataDir + "/" + paths.TranscriptFileName); fileErr == nil {
   604→		if content, contentErr := file.Contents(); contentErr == nil {
   605→			transcriptLines = countTranscriptLines(content)
   606→		}
   607→	} else if file, fileErr := tree.File(metadataDir + "/" + paths.TranscriptFileNameLegacy); fileErr == nil {
   608→		if content, contentErr := file.Contents(); contentErr == nil {
   609→			transcriptLines = countTranscriptLines(content)
   610→		}
   611→	}
   612→
   613→	// Has new content if there are more lines than already condensed
   614→	return transcriptLines > state.CheckpointTranscriptStart, nil
   615→}

   617→// countTranscriptLines counts lines in a transcript, matching the counting method used
   618→// in extractSessionData for consistency. This trims trailing empty lines (from final \n
   619→// in JSONL) but includes empty lines in the middle of the file.
   620→func countTranscriptLines(content string) int {
   621→	lines := strings.Split(content, "\n")
   622→	// Trim trailing empty lines (from final \n in JSONL)
   623→	for len(lines) > 0 && strings.TrimSpace(lines[len(lines)-1]) == "" {
   624→		lines = lines[:len(lines)-1]
   625→	}
   626→	return len(lines)
   627→}

   629→// sessionHasNewContentFromLiveTranscript checks if a session has new content
   630→// by examining the live transcript file. This is used when no shadow branch exists
   631→// (i.e., no Stop has happened yet) but the agent may have done work.
   632→//
   633→// Returns true if:
   634→//  1. The transcript has grown since the last condensation, AND
   635→//  2. The new transcript portion contains file modifications, AND
   636→//  3. At least one modified file overlaps with the currently staged files
   637→//
   638→// The overlap check ensures we don't add checkpoint trailers to commits that are
   639→// unrelated to the agent's recent changes.
   640→//
   641→// This handles the scenario where the agent commits mid-session before Stop.
   642→func (s *ManualCommitStrategy) sessionHasNewContentFromLiveTranscript(repo *git.Repository, state *SessionState) (bool, error) {
   643→	// Need both transcript path and agent type to analyze
   644→	if state.TranscriptPath == "" || state.AgentType == "" {
   645→		return false, nil
   646→	}
   647→
   648→	// Get the agent for transcript analysis
   649→	ag, err := agent.GetByAgentType(state.AgentType)
   650→	if err != nil {
   651→		return false, nil //nolint:nilerr // Unknown agent type, fail gracefully
   652→	}
   653→
   654→	// Cast to TranscriptAnalyzer
   655→	analyzer, ok := ag.(agent.TranscriptAnalyzer)
   656→	if !ok {
   657→		return false, nil // Agent doesn't support transcript analysis
   658→	}
   659→
   660→	// Get current transcript position
   661→	currentPos, err := analyzer.GetTranscriptPosition(state.TranscriptPath)
   662→	if err != nil {
   663→		return false, nil //nolint:nilerr // Error reading transcript, fail gracefully
   664→	}
   665→
   666→	// Check if transcript has grown since last condensation
   667→	if currentPos <= state.CheckpointTranscriptStart {
   668→		return false, nil // No new content
   669→	}
   670→
   671→	// Transcript has grown - check if there are file modifications in the new portion
   672→	modifiedFiles, _, err := analyzer.ExtractModifiedFilesFromOffset(state.TranscriptPath, state.CheckpointTranscriptStart)
   673→	if err != nil {
   674→		return false, nil //nolint:nilerr // Error parsing transcript, fail gracefully
   675→	}
   676→
   677→	// No file modifications means no new content to checkpoint
   678→	if len(modifiedFiles) == 0 {
   679→		return false, nil
   680→	}
   681→
   682→	// Check if any modified files overlap with currently staged files
   683→	// This ensures we only add checkpoint trailers to commits that include
   684→	// files the agent actually modified
   685→	stagedFiles := getStagedFiles(repo) // TODO: does it work? Do we have staged files if this is called at post-commit hook ?
   686→	if !hasOverlappingFiles(stagedFiles, modifiedFiles) {
   687→		return false, nil // No overlap - staged files are unrelated to agent's work
   688→	}
   689→
   690→	return true, nil
   691→}

   693→// addCheckpointTrailer adds the Entire-Checkpoint trailer to a commit message.
   694→// Handles proper trailer formatting (blank line before trailers if needed).
   695→func addCheckpointTrailer(message string, checkpointID id.CheckpointID) string {
   696→	trailer := trailers.CheckpointTrailerKey + ": " + checkpointID.String()
   697→
   698→	// If message already ends with trailers (lines starting with key:), just append
   699→	// Otherwise, add a blank line first
   700→	lines := strings.Split(strings.TrimRight(message, "\n"), "\n")
   701→
   702→	// Check if last non-empty, non-comment line looks like a trailer
   703→	// Git comment lines start with # and may contain ": " (e.g., "# Changes to be committed:")
   704→	hasTrailers := false
   705→	for i := len(lines) - 1; i >= 0; i-- {
   706→		line := strings.TrimSpace(lines[i])
   707→		if line == "" {
   708→			break
   709→		}
   710→		// Skip git comment lines
   711→		if strings.HasPrefix(line, "#") {
   712→			continue
   713→		}
   714→		if strings.Contains(line, ": ") {
   715→			hasTrailers = true
   716→			break
   717→		}
   718→		// Non-comment, non-trailer line found - no existing trailers
   719→		break
   720→	}
   721→
   722→	if hasTrailers {
   723→		// Append trailer directly
   724→		return strings.TrimRight(message, "\n") + "\n" + trailer + "\n"
   725→	}
   726→
   727→	// Add blank line before trailer
   728→	return strings.TrimRight(message, "\n") + "\n\n" + trailer + "\n"
   729→}

   731→// addCheckpointTrailerWithComment adds the Entire-Checkpoint trailer with an explanatory comment.
   732→// The trailer is placed above the git comment block but below the user's message area,
   733→// with a comment explaining that the user can remove it if they don't want to link the commit
   734→// to the agent session. If prompt is non-empty, it's shown as context.
   735→func addCheckpointTrailerWithComment(message string, checkpointID id.CheckpointID, agentName, prompt string) string {
   736→	trailer := trailers.CheckpointTrailerKey + ": " + checkpointID.String()
   737→	commentLines := []string{
   738→		"# Remove the Entire-Checkpoint trailer above if you don't want to link this commit to " + agentName + " session context.",
   739→	}
   740→	if prompt != "" {
   741→		commentLines = append(commentLines, "# Last Prompt: "+prompt)
   742→	}
   743→	commentLines = append(commentLines, "# The trailer will be added to your next commit based on this branch.")
   744→	comment := strings.Join(commentLines, "\n")
   745→
   746→	lines := strings.Split(message, "\n")
   747→
   748→	// Find where the git comment block starts (first # line)
   749→	commentStart := -1
   750→	for i, line := range lines {
   751→		if strings.HasPrefix(line, "#") {
   752→			commentStart = i
   753→			break
   754→		}
   755→	}
   756→
   757→	if commentStart == -1 {
   758→		// No git comments, append trailer at the end
   759→		return strings.TrimRight(message, "\n") + "\n\n" + trailer + "\n" + comment + "\n"
   760→	}
   761→
   762→	// Split into user content and git comments
   763→	userContent := strings.Join(lines[:commentStart], "\n")
   764→	gitComments := strings.Join(lines[commentStart:], "\n")
   765→
   766→	// Build result: user content, blank line, trailer, comment, blank line, git comments
   767→	userContent = strings.TrimRight(userContent, "\n")
   768→	if userContent == "" {
   769→		// No user content yet - leave space for them to type, then trailer
   770→		// Two newlines: first for user's message line, second for blank separator
   771→		return "\n\n" + trailer + "\n" + comment + "\n\n" + gitComments
   772→	}
   773→	return userContent + "\n\n" + trailer + "\n" + comment + "\n\n" + gitComments
   774→}

   883→// calculatePromptAttributionAtStart calculates attribution at prompt start (before agent runs).
   884→// This captures user changes since the last checkpoint - no filtering needed since
   885→// the agent hasn't made any changes yet.

   890→// them to be incorrectly attributed to the agent later.

   897→	result := PromptAttribution{CheckpointNumber: nextCheckpointNum}

   900→	// For the first checkpoint, no shadow branch exists yet - this is fine,
   901→	// CalculatePromptAttribution will use baseTree as the reference instead.

   903→	shadowBranchName := checkpoint.ShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   904→	refName := plumbing.NewBranchReferenceName(shadowBranchName)

   906→	if err != nil {
