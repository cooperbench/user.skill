   687→	// Get user's home directory
   688→	home, err := os.UserHomeDir()

   701→		rcFile = filepath.Join(home, ".zshrc")
   702→		completionLine = "source <(entire completion zsh)"
   703→	case strings.Contains(shell, "bash"):
   704→		rcFile = filepath.Join(home, ".bashrc")
   705→		completionLine = "source <(entire completion bash)"
   706→	default:
   707→		return nil // Unsupported shell, skip silently
   708→	}
   709→
   710→	// Check if completion is already configured
   711→	if isCompletionConfigured(rcFile) {

   716→	var selected string

   722→					huh.NewOption("Yes", "yes"),
   723→					huh.NewOption("No", "no"),
   724→				).
   725→				Value(&selected),
   726→		),
   727→	)
   728→
   729→	if err := form.Run(); err != nil {
   730→		//nolint:nilerr // User cancelled - not a fatal error, just skip
   731→		return nil
   732→	}

   735→		return nil
   736→	}
   737→
   738→	// Append completion to rc file
   739→	if err := appendShellCompletion(rcFile, completionLine); err != nil {
   740→		return fmt.Errorf("failed to update %s: %w", rcFile, err)
   741→	}
   742→
   743→	fmt.Fprintf(w, "✓ Shell completion added to %s\n", rcFile)
   744→	fmt.Fprintln(w, "  Run `source "+rcFile+"` or restart your shell to activate")
   745→
   746→	return nil
   747→}
