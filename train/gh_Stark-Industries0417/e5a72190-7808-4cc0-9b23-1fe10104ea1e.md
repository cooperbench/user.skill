# Fix: Gemini session path during restore

## Context

When restoring Gemini session transcripts from checkpoints, both the **directory** and **filename** are wrong:
- **Directory**: `GetSessionDir()` uses `SanitizePathForGemini()` (replaces non-alphanumeric → `-`) but Gemini uses a SHA256 `projectHash` from the transcript JSON
- **Filename**: Fallback creates `<sessionID>.json` but Gemini uses `session-<YYYY-MM-DDTHH-MM>-<shortid>.json`

Correct path: `~/.gemini/tmp/<projectHash>/chats/session-2026-02-10T23-57-9f5659bb.json`

## Changes

### 1. Add `geminiTranscriptMeta` extractor and filename helper

**`strategy/manual_commit_rewind.go`** — Extract `projectHash` and `startTime` from the Gemini transcript JSON:
```go
type geminiTranscriptMeta struct {
    ProjectHash string `json:"projectHash"`
    StartTime   string `json:"startTime"`
}

func extractGeminiMeta(transcript []byte) geminiTranscriptMeta {
    var meta geminiTranscriptMeta
    json.Unmarshal(transcript, &meta)
    return meta
}

func geminiSessionFileName(agentSessionID string, startTime string) string {
    shortID := agentSessionID
    if len(shortID) > 8 {
        shortID = shortID[:8]
    }
    // startTime from transcript is like "2026-02-10T23:57:..."
    // Gemini filename uses "2026-02-10T23-57" (colons replaced with dashes, truncated to minute)
    t, err := time.Parse(time.RFC3339, startTime)
    if err != nil {
        // Try parsing just the date+time portion
        t, err = time.Parse("2006-01-02T15:04", startTime[:16])
        if err != nil {
            return agentSessionID + ".json"
        }
    }
    return fmt.Sprintf("session-%s-%s.json", t.Format("2006-01-02T15-04"), shortID)
}
```

   223→			Message:        message,
   224→			Date:           c.Author.When,
   225→			IsLogsOnly:     true,
   226→			CheckpointID:   cpInfo.CheckpointID,
   227→			Agent:          cpInfo.Agent,
   228→			SessionID:      cpInfo.SessionID,
   229→			SessionPrompt:  sessionPrompt,
   230→			SessionCount:   cpInfo.SessionCount,
   231→			SessionIDs:     cpInfo.SessionIDs,
   232→			SessionPrompts: sessionPrompts,
   233→		})
   234→
   235→		return nil
   236→	})
   237→
   238→	if err != nil && !errors.Is(err, errStop) {
   239→		return nil, fmt.Errorf("error iterating commits: %w", err)
   240→	}
   241→
   242→	if len(points) > limit {
   243→		points = points[:limit]
   244→	}
   245→
   246→	return points, nil

   252→func (s *ManualCommitStrategy) Rewind(point RewindPoint) error {
   253→	repo, err := OpenRepository()
   254→	if err != nil {
   255→		return fmt.Errorf("failed to open git repository: %w", err)
   256→	}
   257→
   258→	// Get the checkpoint commit
   259→	commitHash := plumbing.NewHash(point.ID)
   260→	commit, err := repo.CommitObject(commitHash)
   261→	if err != nil {
   262→		return fmt.Errorf("failed to get commit: %w", err)
   263→	}
   264→
   265→	tree, err := commit.Tree()
   266→	if err != nil {
   267→		return fmt.Errorf("failed to get tree: %w", err)
   268→	}
   269→
   270→	// Reset the shadow branch to the rewound checkpoint
   271→	// This ensures the next checkpoint will only include prompts from this point forward
   272→	if err := s.resetShadowBranchToCheckpoint(repo, commit); err != nil {
   273→		// Log warning but don't fail - file restoration is the primary operation
   274→		fmt.Fprintf(os.Stderr, "[entire] Warning: failed to reset shadow branch: %v\n", err)
   275→	}
   276→
   277→	// Load session state to get untracked files that existed at session start
   278→	sessionID, hasSessionTrailer := trailers.ParseSession(commit.Message)
   279→	var preservedUntrackedFiles map[string]bool
   280→	if hasSessionTrailer {
   281→		state, stateErr := s.loadSessionState(sessionID)
   282→		if stateErr == nil && state != nil && len(state.UntrackedFilesAtStart) > 0 {
   283→			preservedUntrackedFiles = make(map[string]bool)
   284→			for _, f := range state.UntrackedFilesAtStart {
   285→				preservedUntrackedFiles[f] = true
   286→			}
   287→		}
   288→	}
   289→
   290→	// Build set of files in the checkpoint tree (excluding metadata)
   291→	checkpointFiles := make(map[string]bool)
   292→	err = tree.Files().ForEach(func(f *object.File) error {
   293→		if !strings.HasPrefix(f.Name, entireDir) {
   294→			checkpointFiles[f.Name] = true
   295→		}
   296→		return nil
   297→	})
   298→	if err != nil {
   299→		return fmt.Errorf("failed to list checkpoint files: %w", err)
   300→	}
   301→
   302→	// Get HEAD tree to identify tracked files
   303→	head, err := repo.Head()
   304→	if err != nil {
   305→		return fmt.Errorf("failed to get HEAD: %w", err)
   306→	}
   307→	headCommit, err := repo.CommitObject(head.Hash())
   308→	if err != nil {
   309→		return fmt.Errorf("failed to get HEAD commit: %w", err)
   310→	}
   311→	headTree, err := headCommit.Tree()
   312→	if err != nil {
   313→		return fmt.Errorf("failed to get HEAD tree: %w", err)
   314→	}
   315→
   316→	// Build set of files tracked in HEAD
   317→	trackedFiles := make(map[string]bool)
   318→	//nolint:errcheck // Error is not critical for rewind
   319→	_ = headTree.Files().ForEach(func(f *object.File) error {
   320→		trackedFiles[f.Name] = true
   321→		return nil
   322→	})
   323→
   324→	// Get repository root to walk from there
   325→	repoRoot, err := GetWorktreePath()
   326→	if err != nil {
   327→		repoRoot = "." // Fallback to current directory
   328→	}
   329→
   330→	// Find and delete untracked files that aren't in the checkpoint
   331→	// These are likely files created by the agent in later checkpoints
   332→	err = filepath.Walk(repoRoot, func(path string, info os.FileInfo, walkErr error) error {
   333→		if walkErr != nil {
   334→			return nil //nolint:nilerr // Skip filesystem errors during walk
   335→		}
   336→
   337→		// Get path relative to repo root
   338→		relPath, relErr := filepath.Rel(repoRoot, path)
   339→		if relErr != nil {
   340→			return nil //nolint:nilerr // Skip paths we can't make relative
   341→		}
   342→
   343→		// Skip directories and protected paths
   344→		if info.IsDir() {
   345→			if isProtectedPath(relPath) {
   346→				return filepath.SkipDir
   347→			}
   348→			return nil
   349→		}
   350→
   351→		// Skip files in protected directories
   352→		if isProtectedPath(relPath) {
   353→			return nil
   354→		}
   355→
   356→		// If file is in checkpoint, it will be restored
   357→		if checkpointFiles[relPath] {
   358→			return nil
   359→		}
   360→
   361→		// If file is tracked in HEAD, don't delete (user's committed work)
   362→		if trackedFiles[relPath] {
   363→			return nil
   364→		}
   365→
   366→		// If file existed at session start, preserve it (untracked user files)
   367→		if preservedUntrackedFiles[relPath] {
   368→			return nil
   369→		}
   370→
   371→		// File is untracked and not in checkpoint - delete it (use absolute path)
   372→		if removeErr := os.Remove(path); removeErr == nil {
   373→			fmt.Fprintf(os.Stderr, "  Deleted: %s\n", relPath)
   374→		}
   375→
   376→		return nil
   377→	})
   378→	if err != nil {
   379→		// Non-fatal - continue with restoration
   380→		fmt.Fprintf(os.Stderr, "Warning: error walking directory: %v\n", err)
   381→	}
   382→
   383→	// Restore files from checkpoint
   384→	err = tree.Files().ForEach(func(f *object.File) error {
   385→		// Skip metadata directories - these are for checkpoint storage, not working dir
   386→		if strings.HasPrefix(f.Name, entireDir) {
   387→			return nil
   388→		}
   389→
   390→		contents, err := f.Contents()
   391→		if err != nil {
   392→			return fmt.Errorf("failed to read file %s: %w", f.Name, err)
   393→		}
   394→
   395→		// Ensure directory exists
   396→		dir := filepath.Dir(f.Name)
   397→		if dir != "." {
   398→			//nolint:gosec // G301: Need 0o755 for user directories during rewind
   399→			if err := os.MkdirAll(dir, 0o755); err != nil {
   400→				return fmt.Errorf("failed to create directory %s: %w", dir, err)
   401→			}
   402→		}
   403→
   404→		// Write file with appropriate permissions
   405→		var perm os.FileMode = 0o644
   406→		if f.Mode == filemode.Executable {
   407→			perm = 0o755
   408→		}
   409→		if err := os.WriteFile(f.Name, []byte(contents), perm); err != nil {
   410→			return fmt.Errorf("failed to write file %s: %w", f.Name, err)
   411→		}
   412→
   413→		fmt.Fprintf(os.Stderr, "  Restored: %s\n", f.Name)
   414→		return nil
   415→	})
   416→	if err != nil {
   417→		return fmt.Errorf("failed to iterate tree files: %w", err)
   418→	}
   419→
   420→	fmt.Println()
   421→	if len(point.ID) >= 7 {
   422→		fmt.Printf("Restored files from shadow commit %s\n", point.ID[:7])
   423→	} else {
   424→		fmt.Printf("Restored files from shadow commit %s\n", point.ID)
   425→	}
   426→	fmt.Println()
   427→
   428→	return nil
   429→}

   431→// resetShadowBranchToCheckpoint resets the shadow branch HEAD to the given checkpoint.
   432→// This ensures that when the user commits after rewinding, the next checkpoint will only
   433→// include prompts from the rewound point, not prompts from later checkpoints.
   434→func (s *ManualCommitStrategy) resetShadowBranchToCheckpoint(repo *git.Repository, commit *object.Commit) error {
   435→	// Extract session ID from the checkpoint commit's Entire-Session trailer
   436→	sessionID, found := trailers.ParseSession(commit.Message)
   437→	if !found {
   438→		return errors.New("checkpoint has no Entire-Session trailer")
   439→	}
   440→
   441→	// Load session state to get the shadow branch name
   442→	state, err := s.loadSessionState(sessionID)
   443→	if err != nil {
   444→		return fmt.Errorf("failed to load session state: %w", err)
   445→	}
   446→	if state == nil {
   447→		return fmt.Errorf("session %s not found", sessionID)
   448→	}
   449→
   450→	// Reset the shadow branch to the checkpoint commit
   451→	shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   452→	refName := plumbing.NewBranchReferenceName(shadowBranchName)
   453→
   454→	// Update the reference to point to the checkpoint commit
   455→	ref := plumbing.NewHashReference(refName, commit.Hash)
   456→	if err := repo.Storer.SetReference(ref); err != nil {
   457→		return fmt.Errorf("failed to update shadow branch: %w", err)
   458→	}
   459→
   460→	fmt.Fprintf(os.Stderr, "[entire] Reset shadow branch %s to checkpoint %s\n", shadowBranchName, commit.Hash.String()[:7])
   461→	return nil
   462→}

   614→// RestoreLogsOnly restores session logs from a logs-only rewind point.
   615→// This fetches the transcript from entire/checkpoints/v1 and writes it to the agent's session directory.
   616→// Does not modify the working directory.
   617→// When multiple sessions were condensed to the same checkpoint, ALL sessions are restored.
   618→// If force is false, prompts for confirmation when local logs have newer timestamps.
   619→// Returns info about each restored session so callers can print correct per-session resume commands.
   620→func (s *ManualCommitStrategy) RestoreLogsOnly(point RewindPoint, force bool) ([]RestoredSession, error) {
   621→	if !point.IsLogsOnly {
   622→		return nil, errors.New("not a logs-only rewind point")
   623→	}
   624→
   625→	if point.CheckpointID.IsEmpty() {
   626→		return nil, errors.New("missing checkpoint ID")
   627→	}
   628→
   629→	// Get checkpoint store
   630→	store, err := s.getCheckpointStore()
   631→	if err != nil {
   632→		return nil, fmt.Errorf("failed to get checkpoint store: %w", err)
   633→	}
   634→
   635→	// Read checkpoint summary to get session count
   636→	summary, err := store.ReadCommitted(context.Background(), point.CheckpointID)
   637→	if err != nil {
   638→		return nil, fmt.Errorf("failed to read checkpoint: %w", err)
   639→	}
   640→	if summary == nil {
   641→		return nil, fmt.Errorf("checkpoint not found: %s", point.CheckpointID)
   642→	}
   643→
   644→	// Get repo root for agent session directory lookup
   645→	repoRoot, err := paths.RepoRoot()
   646→	if err != nil {
   647→		return nil, fmt.Errorf("failed to get repository root: %w", err)
   648→	}
   649→
   650→	// Check for newer local logs if not forcing
   651→	if !force {
   652→		sessions := s.classifySessionsForRestore(context.Background(), repoRoot, store, point.CheckpointID, summary)
   653→		hasConflicts := false
   654→		for _, sess := range sessions {
   655→			if sess.Status == StatusLocalNewer {
   656→				hasConflicts = true
   657→				break
   658→			}
   659→		}
   660→		if hasConflicts {
   661→			shouldOverwrite, promptErr := PromptOverwriteNewerLogs(sessions)
   662→			if promptErr != nil {
   663→				return nil, promptErr
   664→			}
   665→			if !shouldOverwrite {
   666→				fmt.Fprintf(os.Stderr, "Resume cancelled. Local session logs preserved.\n")
   667→				return nil, nil
   668→			}
   669→		}
   670→	}
   671→
   672→	// Count sessions to restore
   673→	totalSessions := len(summary.Sessions)
   674→	if totalSessions > 1 {
   675→		fmt.Fprintf(os.Stderr, "Restoring %d sessions from checkpoint:\n", totalSessions)
   676→	}
   677→
   678→	// Restore all sessions (oldest to newest, using 0-based indexing)
   679→	var restored []RestoredSession
   680→	for i := range totalSessions {
   681→		content, readErr := store.ReadSessionContent(context.Background(), point.CheckpointID, i)
   682→		if readErr != nil {
   683→			fmt.Fprintf(os.Stderr, "  Warning: failed to read session %d: %v\n", i, readErr)
   684→			continue
   685→		}
   686→		if content == nil || len(content.Transcript) == 0 {
   687→			continue
   688→		}
   689→
   690→		sessionID := content.Metadata.SessionID
   691→		if sessionID == "" {
   692→			fmt.Fprintf(os.Stderr, "  Warning: session %d has no session ID, skipping\n", i)
   693→			continue
   694→		}
   695→
   696→		// Resolve per-session agent from metadata — skip if agent is unknown
   697→		if content.Metadata.Agent == "" {
   698→			fmt.Fprintf(os.Stderr, "  Warning: session %d (%s) has no agent metadata, skipping (cannot determine target directory)\n", i, sessionID)
   699→			continue
   700→		}
   701→		sessionAgent, agErr := ResolveAgentForRewind(content.Metadata.Agent)
   702→		if agErr != nil {
   703→			fmt.Fprintf(os.Stderr, "  Warning: session %d (%s) has unknown agent %q, skipping\n", i, sessionID, content.Metadata.Agent)
   704→			continue
   705→		}
   706→
   707→		sessionAgentDir, dirErr := sessionAgent.GetSessionDir(repoRoot)
   708→		if dirErr != nil {
   709→			fmt.Fprintf(os.Stderr, "  Warning: failed to get session dir for session %d: %v\n", i, dirErr)
   710→			continue
   711→		}
   712→
   713→		sessionFile := ResolveSessionFilePath(sessionID, sessionAgent, sessionAgentDir)
   714→
   715→		// Get first prompt for display
   716→		promptPreview := ExtractFirstPrompt(content.Prompts)
   717→
   718→		if totalSessions > 1 {
   719→			isLatest := i == totalSessions-1
   720→			if promptPreview != "" {
   721→				if isLatest {
   722→					fmt.Fprintf(os.Stderr, "  Session %d (latest): %s\n", i+1, promptPreview)
   723→				} else {
   724→					fmt.Fprintf(os.Stderr, "  Session %d: %s\n", i+1, promptPreview)
   725→				}
   726→			}
   727→			fmt.Fprintf(os.Stderr, "    Writing to: %s\n", sessionFile)
   728→		} else {
   729→			fmt.Fprintf(os.Stderr, "Writing transcript to: %s\n", sessionFile)
   730→		}
   731→
   732→		// Ensure parent directory exists (session file may be in a different dir than sessionAgentDir)
   733→		if mkdirErr := os.MkdirAll(filepath.Dir(sessionFile), 0o750); mkdirErr != nil {
   734→			fmt.Fprintf(os.Stderr, "    Warning: failed to create directory: %v\n", mkdirErr)
   735→			continue
   736→		}
   737→
   738→		if writeErr := os.WriteFile(sessionFile, content.Transcript, 0o600); writeErr != nil {
   739→			if totalSessions > 1 {
   740→				fmt.Fprintf(os.Stderr, "    Warning: failed to write transcript: %v\n", writeErr)
   741→				continue
   742→			}
   743→			return nil, fmt.Errorf("failed to write transcript: %w", writeErr)
   744→		}
   745→
   746→		restored = append(restored, RestoredSession{
   747→			SessionID: sessionID,
   748→			Agent:     sessionAgent.Type(),
   749→			Prompt:    promptPreview,
   750→		})
   751→	}
   752→
   753→	return restored, nil
   754→}

   767→	if err != nil {
   768→		return nil, fmt.Errorf("resolving agent %q: %w", agentType, err)
   769→	}
   770→	return ag, nil

   773→// ResolveSessionFilePath determines the correct file path for an agent's session transcript.
   774→// Checks session state for transcript_path first (needed for agents like Gemini that store
   775→// transcripts at paths that GetSessionDir can't reconstruct, e.g. SHA-256 hashed directories).
   776→// Falls back to the agent's ExtractAgentSessionID + ResolveSessionFile with fallbackSessionDir.
   777→func ResolveSessionFilePath(sessionID string, ag agent.Agent, fallbackSessionDir string) string {
   778→	state, err := LoadSessionState(sessionID)
   779→	if err == nil && state != nil && state.TranscriptPath != "" {
   780→		return state.TranscriptPath
   781→	}

   883→// ClassifyTimestamps determines the restore status based on local and checkpoint timestamps.
   884→func ClassifyTimestamps(localTime, checkpointTime time.Time) SessionRestoreStatus {
   885→	// Local file doesn't exist (no timestamp found)
   886→	if localTime.IsZero() {
   887→		return StatusNew
   888→	}
   889→
   890→	// Can't determine checkpoint time - treat as new/safe
   891→	if checkpointTime.IsZero() {
   892→		return StatusNew
   893→	}

   897→		return StatusLocalNewer
   898→	}
   899→	if checkpointTime.After(localTime) {
   900→		return StatusCheckpointNewer
   901→	}
   902→	return StatusUnchanged
   903→}

   394→// ChunkTranscript splits a Gemini JSON transcript by distributing messages across chunks.
   395→// Gemini uses JSON format with a {"messages": [...]} structure, so chunking splits
   396→// the messages array while preserving the JSON structure in each chunk.
   397→func (g *GeminiCLIAgent) ChunkTranscript(content []byte, maxSize int) ([][]byte, error) {
   398→	var transcript GeminiTranscript
   399→	if err := json.Unmarshal(content, &transcript); err != nil {
   400→		// Fall back to JSONL chunking if not valid Gemini JSON
   401→		chunks, chunkErr := agent.ChunkJSONL(content, maxSize)
   402→		if chunkErr != nil {
   403→			return nil, fmt.Errorf("failed to chunk as JSONL: %w", chunkErr)
   404→		}
   405→		return chunks, nil
   406→	}
   407→
   408→	if len(transcript.Messages) == 0 {
   409→		return [][]byte{content}, nil
   410→	}
   411→
   412→	var chunks [][]byte
   413→	var currentMessages []GeminiMessage
   414→	currentSize := len(`{"messages":[]}`) // Base JSON structure size
   415→
   416→	for i, msg := range transcript.Messages {
   417→		// Marshal message to get its size
   418→		msgBytes, err := json.Marshal(msg)
   419→		if err != nil {
   420→			logging.Warn(context.Background(), "failed to marshal Gemini message during chunking",
   421→				slog.Int("message_index", i),
   422→				slog.String("error", err.Error()),
   423→			)
   424→			continue
   425→		}
   426→		msgSize := len(msgBytes) + 1 // +1 for comma separator
   427→
   428→		if currentSize+msgSize > maxSize && len(currentMessages) > 0 {
   429→			// Save current chunk
   430→			chunkData, err := json.Marshal(GeminiTranscript{Messages: currentMessages})

   432→				return nil, fmt.Errorf("failed to marshal chunk: %w", err)
   433→			}

   435→
   436→			// Start new chunk
   437→			currentMessages = nil
   438→			currentSize = len(`{"messages":[]}`)
   439→		}
   440→
   441→		currentMessages = append(currentMessages, msg)
   442→		currentSize += msgSize
   443→	}
   444→
   445→	// Add the last chunk
   446→	if len(currentMessages) > 0 {
   447→		chunkData, err := json.Marshal(GeminiTranscript{Messages: currentMessages})
   448→		if err != nil {
   449→			return nil, fmt.Errorf("failed to marshal final chunk: %w", err)

   439→// InitialAttribution captures line-level attribution metrics at commit time.
   440→// This is a point-in-time snapshot comparing the checkpoint tree (agent work)
   441→// against the committed tree (may include human edits).
   442→//
   443→// Attribution Metrics:
   444→//   - TotalCommitted measures "net additions" (lines added that remain in the commit)
   445→//   - AgentPercentage represents "of the new code added, what percentage came from the agent"
   446→//   - Deletion work is tracked separately in HumanRemoved (not included in percentage)

   448→// Deletion-Only Commits:
   449→// For commits with only deletions (no additions), TotalCommitted will be 0 and
   450→// AgentPercentage will be 0. This is by design - the percentage metric is only
   451→// meaningful for commits that add code. Deletion contributions are captured in
   452→// the HumanRemoved field but don't affect the attribution percentage.
   453→type InitialAttribution struct {
   454→	CalculatedAt    time.Time `json:"calculated_at"`
   455→	AgentLines      int       `json:"agent_lines"`      // Lines added by agent (base → shadow diff)
   456→	HumanAdded      int       `json:"human_added"`      // Lines added by human (excluding modifications)
   457→	HumanModified   int       `json:"human_modified"`   // Lines modified by human (estimate: min(added, removed))
   458→	HumanRemoved    int       `json:"human_removed"`    // Lines removed by human (excluding modifications)
   459→	TotalCommitted  int       `json:"total_committed"`  // Net additions in commit (agent + human new lines, not total file size)
   460→	AgentPercentage float64   `json:"agent_percentage"` // agent_lines / total_committed * 100 (0 for deletion-only commits)

    99→	if err != nil {
   100→		return err
   101→	}
   102→

   104→	newRef := plumbing.NewHashReference(refName, newCommitHash)
   105→	if err := s.repo.Storer.SetReference(newRef); err != nil {
   106→		return fmt.Errorf("failed to set branch reference: %w", err)
   107→	}
   108→
   109→	return nil
   110→}

   226→// writeStandardCheckpointEntries writes session files to numbered subdirectories and
   227→// maintains a CheckpointSummary at the root level with aggregated statistics.
   228→//
   229→// Structure:
   230→//
   231→//	basePath/
   232→//	├── metadata.json         # CheckpointSummary (aggregated stats)
   233→//	├── 1/                    # First session
   234→//	│   ├── metadata.json     # CommittedMetadata (session-specific, includes initial_attribution)
   235→//	│   ├── full.jsonl
   236→//	│   ├── prompt.txt
   237→//	│   ├── context.md
   238→//	│   └── content_hash.txt
   239→//	├── 2/                    # Second session
   240→//	└── ...
   241→func (s *GitStore) writeStandardCheckpointEntries(opts WriteCommittedOptions, basePath string, entries map[string]object.TreeEntry) error {
   242→	// Read existing summary to get current session count
   243→	var existingSummary *CheckpointSummary
   244→	metadataPath := basePath + paths.MetadataFileName
   245→	if entry, exists := entries[metadataPath]; exists {
   246→		existing, err := s.readSummaryFromBlob(entry.Hash)
   247→		if err == nil {
   248→			existingSummary = existing
   249→		}
   250→	}
   251→
   252→	// Determine session index: reuse existing slot if session ID matches, otherwise append
   253→	sessionIndex := s.findSessionIndex(basePath, existingSummary, entries, opts.SessionID)
   254→
   255→	// Write session files to numbered subdirectory
   256→	sessionPath := fmt.Sprintf("%s%d/", basePath, sessionIndex)
   257→	sessionFilePaths, err := s.writeSessionToSubdirectory(opts, sessionPath, entries)
   258→	if err != nil {
   259→		return err
   260→	}
   261→
   262→	// Copy additional metadata files from directory if specified (to session subdirectory)
   263→	if opts.MetadataDir != "" {
   264→		if err := s.copyMetadataDir(opts.MetadataDir, sessionPath, entries); err != nil {
   265→			return fmt.Errorf("failed to copy metadata directory: %w", err)
   266→		}
   267→	}
   268→
   269→	// Build the sessions array
   270→	var sessions []SessionFilePaths
   271→	if existingSummary != nil {
   272→		sessions = make([]SessionFilePaths, max(len(existingSummary.Sessions), sessionIndex+1))
   273→		copy(sessions, existingSummary.Sessions)
   274→	} else {
   275→		sessions = make([]SessionFilePaths, 1)
   276→	}
   277→	sessions[sessionIndex] = sessionFilePaths
   278→
   279→	// Update root metadata.json with CheckpointSummary
   280→	return s.writeCheckpointSummary(opts, basePath, entries, sessions)
   281→}

   283→// writeSessionToSubdirectory writes a single session's files to a numbered subdirectory.
   284→// Returns the absolute file paths from the git tree root for the sessions map.
   285→func (s *GitStore) writeSessionToSubdirectory(opts WriteCommittedOptions, sessionPath string, entries map[string]object.TreeEntry) (SessionFilePaths, error) {
   286→	filePaths := SessionFilePaths{}
   287→
   288→	// Clear any existing entries at this path so stale files from a previous
   289→	// write (e.g. prompt.txt, context.md) don't persist on overwrite.
   290→	for key := range entries {
   291→		if strings.HasPrefix(key, sessionPath) {
   292→			delete(entries, key)
   293→		}
   294→	}
   295→
   296→	// Write transcript
   297→	if err := s.writeTranscript(opts, sessionPath, entries); err != nil {
   298→		return filePaths, err
   299→	}
   300→	filePaths.Transcript = "/" + sessionPath + paths.TranscriptFileName
   301→	filePaths.ContentHash = "/" + sessionPath + paths.ContentHashFileName
   302→
   303→	// Write prompts
   304→	if len(opts.Prompts) > 0 {
   305→		promptContent := redact.String(strings.Join(opts.Prompts, "\n\n---\n\n"))
   306→		blobHash, err := CreateBlobFromContent(s.repo, []byte(promptContent))
   307→		if err != nil {
   308→			return filePaths, err
   309→		}
   310→		entries[sessionPath+paths.PromptFileName] = object.TreeEntry{
   311→			Name: sessionPath + paths.PromptFileName,
   312→			Mode: filemode.Regular,
   313→			Hash: blobHash,
   314→		}
   315→		filePaths.Prompt = "/" + sessionPath + paths.PromptFileName
   316→	}
   317→
   318→	// Write context
   319→	if len(opts.Context) > 0 {
   320→		blobHash, err := CreateBlobFromContent(s.repo, redact.Bytes(opts.Context))
   321→		if err != nil {
   322→			return filePaths, err
   323→		}
   324→		entries[sessionPath+paths.ContextFileName] = object.TreeEntry{
   325→			Name: sessionPath + paths.ContextFileName,
   326→			Mode: filemode.Regular,
   327→			Hash: blobHash,
   328→		}
   329→		filePaths.Context = "/" + sessionPath + paths.ContextFileName
   330→	}
   331→
   332→	// Write session-level metadata.json (CommittedMetadata with all fields including initial_attribution)
   333→	sessionMetadata := CommittedMetadata{
   334→		CheckpointID:                opts.CheckpointID,
   335→		SessionID:                   opts.SessionID,
   336→		Strategy:                    opts.Strategy,
   337→		CreatedAt:                   time.Now().UTC(),
   338→		Branch:                      opts.Branch,
   339→		CheckpointsCount:            opts.CheckpointsCount,
   340→		FilesTouched:                opts.FilesTouched,
   341→		Agent:                       opts.Agent,
   342→		IsTask:                      opts.IsTask,
   343→		ToolUseID:                   opts.ToolUseID,
   344→		TranscriptIdentifierAtStart: opts.TranscriptIdentifierAtStart,
   345→		CheckpointTranscriptStart:   opts.CheckpointTranscriptStart,
   346→		TranscriptLinesAtStart:      opts.CheckpointTranscriptStart, // Deprecated: kept for backward compat
   347→		TokenUsage:                  opts.TokenUsage,
   348→		InitialAttribution:          opts.InitialAttribution,
   349→		Summary:                     opts.Summary,
   350→		CLIVersion:                  buildinfo.Version,
   351→	}
   352→
   353→	metadataJSON, err := jsonutil.MarshalIndentWithNewline(sessionMetadata, "", "  ")
   354→	if err != nil {
   355→		return filePaths, fmt.Errorf("failed to marshal session metadata: %w", err)
   356→	}
   357→	metadataHash, err := CreateBlobFromContent(s.repo, metadataJSON)
   358→	if err != nil {
   359→		return filePaths, err
   360→	}
   361→	entries[sessionPath+paths.MetadataFileName] = object.TreeEntry{
   362→		Name: sessionPath + paths.MetadataFileName,
   363→		Mode: filemode.Regular,
   364→		Hash: metadataHash,
   365→	}
   366→	filePaths.Metadata = "/" + sessionPath + paths.MetadataFileName
   367→
   368→	return filePaths, nil
   369→}

   407→// findSessionIndex returns the index of an existing session with the given ID,
   408→// or the next available index if not found. This prevents duplicate session entries.
   409→func (s *GitStore) findSessionIndex(basePath string, existingSummary *CheckpointSummary, entries map[string]object.TreeEntry, sessionID string) int {
   410→	if existingSummary == nil {
   411→		return 0
   412→	}
   413→	for i := range len(existingSummary.Sessions) {
   414→		path := fmt.Sprintf("%s%d/%s", basePath, i, paths.MetadataFileName)
   415→		if entry, exists := entries[path]; exists {
   416→			meta, err := s.readMetadataFromBlob(entry.Hash)
   417→			if err != nil {
   418→				logging.Warn(context.Background(), "failed to read session metadata during dedup check",
   419→					slog.Int("session_index", i),
   420→					slog.String("session_id", sessionID),
   421→					slog.String("error", err.Error()),
   422→				)
   423→				continue
   424→			}
   425→			if meta.SessionID == sessionID {
   426→				return i
   427→			}
   428→		}
   429→	}
   430→	return len(existingSummary.Sessions)
   431→}

   433→// reaggregateFromEntries reads all session metadata from the entries map and
   434→// reaggregates CheckpointsCount, FilesTouched, and TokenUsage.
   435→func (s *GitStore) reaggregateFromEntries(basePath string, sessionCount int, entries map[string]object.TreeEntry) (int, []string, *agent.TokenUsage, error) {
   436→	var totalCount int
   437→	var allFiles []string
   438→	var totalTokens *agent.TokenUsage
   439→
   440→	for i := range sessionCount {
   441→		path := fmt.Sprintf("%s%d/%s", basePath, i, paths.MetadataFileName)
   442→		entry, exists := entries[path]
   443→		if !exists {
   444→			return 0, nil, nil, fmt.Errorf("session %d metadata not found at %s", i, path)
   445→		}
   446→		meta, err := s.readMetadataFromBlob(entry.Hash)
   447→		if err != nil {
   448→			return 0, nil, nil, fmt.Errorf("failed to read session %d metadata: %w", i, err)
   449→		}
   450→		totalCount += meta.CheckpointsCount
   451→		allFiles = mergeFilesTouched(allFiles, meta.FilesTouched)
   452→		totalTokens = aggregateTokenUsage(totalTokens, meta.TokenUsage)
   453→	}
   454→
   455→	return totalCount, allFiles, totalTokens, nil
   456→}

   458→// readJSONFromBlob reads JSON from a blob hash and decodes it to the given type.
   459→func readJSONFromBlob[T any](repo *git.Repository, hash plumbing.Hash) (*T, error) {
   460→	blob, err := repo.BlobObject(hash)
   461→	if err != nil {
   462→		return nil, fmt.Errorf("failed to get blob: %w", err)
   463→	}
   464→
   465→	reader, err := blob.Reader()
   466→	if err != nil {
   467→		return nil, fmt.Errorf("failed to get blob reader: %w", err)
   468→	}
   469→	defer reader.Close()
   470→
   471→	var result T
   472→	if err := json.NewDecoder(reader).Decode(&result); err != nil {
   473→		return nil, fmt.Errorf("failed to decode: %w", err)
   474→	}
   475→
   476→	return &result, nil
   477→}

   484→// aggregateTokenUsage sums two TokenUsage structs.
   485→// Returns nil if both inputs are nil.
   486→func aggregateTokenUsage(a, b *agent.TokenUsage) *agent.TokenUsage {
   487→	if a == nil && b == nil {
   488→		return nil
   489→	}
   490→	result := &agent.TokenUsage{}
   491→	if a != nil {
   492→		result.InputTokens = a.InputTokens
   493→		result.CacheCreationTokens = a.CacheCreationTokens
   494→		result.CacheReadTokens = a.CacheReadTokens
   495→		result.OutputTokens = a.OutputTokens
   496→		result.APICallCount = a.APICallCount
   497→	}
   498→	if b != nil {
   499→		result.InputTokens += b.InputTokens
   500→		result.CacheCreationTokens += b.CacheCreationTokens
   501→		result.CacheReadTokens += b.CacheReadTokens
   502→		result.OutputTokens += b.OutputTokens
   503→		result.APICallCount += b.APICallCount
   504→	}
   505→	return result

   508→// writeTranscript writes the transcript file from in-memory content or file path.
   509→// If the transcript exceeds MaxChunkSize, it's split into multiple chunk files.
   510→func (s *GitStore) writeTranscript(opts WriteCommittedOptions, basePath string, entries map[string]object.TreeEntry) error {
   511→	transcript := opts.Transcript
   512→	if len(transcript) == 0 && opts.TranscriptPath != "" {
   513→		var readErr error
   514→		transcript, readErr = os.ReadFile(opts.TranscriptPath)
   515→		if readErr != nil {
   516→			// Non-fatal: transcript may not exist yet
   517→			transcript = nil
   518→		}
   519→	}
   520→	if len(transcript) == 0 {
   521→		return nil
   522→	}
   523→
   524→	// Redact secrets before chunking so content hash reflects redacted content
   525→	transcript, err := redact.JSONLBytes(transcript)
   526→	if err != nil {
   527→		return fmt.Errorf("failed to redact transcript secrets: %w", err)
   528→	}
   529→
   530→	// Chunk the transcript if it's too large
   531→	chunks, err := agent.ChunkTranscript(transcript, opts.Agent)
   532→	if err != nil {
   533→		return fmt.Errorf("failed to chunk transcript: %w", err)
   534→	}
   535→
   536→	// Write chunk files
   537→	for i, chunk := range chunks {
   538→		chunkPath := basePath + agent.ChunkFileName(paths.TranscriptFileName, i)
   539→		blobHash, err := CreateBlobFromContent(s.repo, chunk)
   540→		if err != nil {
   541→			return err
   542→		}
   543→		entries[chunkPath] = object.TreeEntry{
   544→			Name: chunkPath,
   545→			Mode: filemode.Regular,
   546→			Hash: blobHash,
   547→		}
   548→	}
   549→
   550→	// Content hash for deduplication (hash of full transcript)
   551→	contentHash := fmt.Sprintf("sha256:%x", sha256.Sum256(transcript))
   552→	hashBlob, err := CreateBlobFromContent(s.repo, []byte(contentHash))
   553→	if err != nil {
   554→		return err
   555→	}
   556→	entries[basePath+paths.ContentHashFileName] = object.TreeEntry{
   557→		Name: basePath + paths.ContentHashFileName,
   558→		Mode: filemode.Regular,
   559→		Hash: hashBlob,
   560→	}
   561→	return nil
   562→}

   564→// mergeFilesTouched combines two file lists, removing duplicates.
   565→func mergeFilesTouched(existing, additional []string) []string {
   566→	seen := make(map[string]bool)
   567→	var result []string
   568→
   569→	for _, f := range existing {
   570→		if !seen[f] {
   571→			seen[f] = true
   572→			result = append(result, f)
   573→		}
   574→	}
   575→	for _, f := range additional {
   576→		if !seen[f] {
   577→			seen[f] = true
   578→			result = append(result, f)
   579→		}
   580→	}
   581→
   582→	sort.Strings(result)
   583→	return result
   584→}

   591→// buildCommitMessage constructs the commit message with proper trailers.
   592→// The commit subject is always "Checkpoint: <id>" for consistency.
   593→// If CommitSubject is provided (e.g., for task checkpoints), it's included in the body.
   594→func (s *GitStore) buildCommitMessage(opts WriteCommittedOptions, taskMetadataPath string) string {
   595→	var commitMsg strings.Builder
   596→
   597→	// Subject line is always the checkpoint ID for consistent formatting
   598→	commitMsg.WriteString(fmt.Sprintf("Checkpoint: %s\n\n", opts.CheckpointID))
   599→
   600→	// Include custom description in body if provided (e.g., task checkpoint details)
   601→	if opts.CommitSubject != "" {
   602→		commitMsg.WriteString(opts.CommitSubject + "\n\n")
   603→	}
   604→	commitMsg.WriteString(fmt.Sprintf("%s: %s\n", trailers.SessionTrailerKey, opts.SessionID))
   605→	commitMsg.WriteString(fmt.Sprintf("%s: %s\n", trailers.StrategyTrailerKey, opts.Strategy))
   606→	if opts.Agent != "" {
   607→		commitMsg.WriteString(fmt.Sprintf("%s: %s\n", trailers.AgentTrailerKey, opts.Agent))
   608→	}
   609→	if opts.EphemeralBranch != "" {
   610→		commitMsg.WriteString(fmt.Sprintf("%s: %s\n", trailers.EphemeralBranchTrailerKey, opts.EphemeralBranch))
   611→	}
   612→	if taskMetadataPath != "" {
   613→		commitMsg.WriteString(fmt.Sprintf("%s: %s\n", trailers.MetadataTaskTrailerKey, taskMetadataPath))
   614→	}
   615→
   616→	return commitMsg.String()
   617→}

   637→// ReadCommitted reads a committed checkpoint's summary by ID from the entire/checkpoints/v1 branch.
   638→// Returns only the CheckpointSummary (paths + aggregated stats), not actual content.
   639→// Use ReadSessionContent to read actual transcript/prompts/context.
   640→// Returns nil, nil if the checkpoint doesn't exist.
   641→//
   642→// The storage format uses numbered subdirectories for each session (0-based):
   643→//
   644→//	<checkpoint-id>/
   645→//	├── metadata.json      # CheckpointSummary with sessions map
   646→//	├── 0/                 # First session
   647→//	│   ├── metadata.json  # Session-specific metadata
   648→//	│   └── full.jsonl     # Transcript
   649→//	├── 1/                 # Second session
   650→//	└── ...
   651→func (s *GitStore) ReadCommitted(ctx context.Context, checkpointID id.CheckpointID) (*CheckpointSummary, error) {
   652→	_ = ctx // Reserved for future use
   653→
   654→	tree, err := s.getSessionsBranchTree()
   655→	if err != nil {
   656→		return nil, nil //nolint:nilnil,nilerr // No sessions branch means no checkpoint exists
   657→	}
   658→
   659→	checkpointPath := checkpointID.Path()
   660→	checkpointTree, err := tree.Tree(checkpointPath)
   661→	if err != nil {
   662→		return nil, nil //nolint:nilnil,nilerr // Checkpoint directory not found
   663→	}
   664→
   665→	// Read root metadata.json as CheckpointSummary
   666→	metadataFile, err := checkpointTree.File(paths.MetadataFileName)
   667→	if err != nil {
   668→		return nil, nil //nolint:nilnil,nilerr // metadata.json not found
   669→	}
   670→
   671→	content, err := metadataFile.Contents()
   672→	if err != nil {
   673→		return nil, fmt.Errorf("failed to read metadata.json: %w", err)
   674→	}
   675→
   676→	var summary CheckpointSummary
   677→	if err := json.Unmarshal([]byte(content), &summary); err != nil {
   678→		return nil, fmt.Errorf("failed to parse metadata.json: %w", err)
   679→	}
   680→
   681→	return &summary, nil
   682→}

   684→// ReadSessionContent reads the actual content for a specific session within a checkpoint.
   685→// sessionIndex is 0-based (0 for first session, 1 for second, etc.).
   686→// Returns the session's metadata, transcript, prompts, and context.
   687→// Returns an error if the checkpoint or session doesn't exist.
   688→func (s *GitStore) ReadSessionContent(ctx context.Context, checkpointID id.CheckpointID, sessionIndex int) (*SessionContent, error) {
   689→	_ = ctx // Reserved for future use
   690→
   691→	tree, err := s.getSessionsBranchTree()
   692→	if err != nil {
   693→		return nil, ErrCheckpointNotFound
   694→	}
   695→
   696→	checkpointPath := checkpointID.Path()
   697→	checkpointTree, err := tree.Tree(checkpointPath)
   698→	if err != nil {
   699→		return nil, ErrCheckpointNotFound
   700→	}
   701→
   702→	// Get the session subdirectory
   703→	sessionDir := strconv.Itoa(sessionIndex)
   704→	sessionTree, err := checkpointTree.Tree(sessionDir)
   705→	if err != nil {
   706→		return nil, fmt.Errorf("session %d not found: %w", sessionIndex, err)
   707→	}
   708→
   709→	result := &SessionContent{}
   710→
   711→	// Read session-specific metadata
   712→	var agentType agent.AgentType
   713→	if metadataFile, fileErr := sessionTree.File(paths.MetadataFileName); fileErr == nil {
   714→		if content, contentErr := metadataFile.Contents(); contentErr == nil {
   715→			if jsonErr := json.Unmarshal([]byte(content), &result.Metadata); jsonErr == nil {
   716→				agentType = result.Metadata.Agent
   717→			}
   718→		}
   719→	}
   720→
   721→	// Read transcript
   722→	if transcript, transcriptErr := readTranscriptFromTree(sessionTree, agentType); transcriptErr == nil && transcript != nil {
   723→		result.Transcript = transcript
   724→	}
   725→
   726→	// Read prompts
   727→	if file, fileErr := sessionTree.File(paths.PromptFileName); fileErr == nil {
   728→		if content, contentErr := file.Contents(); contentErr == nil {
   729→			result.Prompts = content
   730→		}
   731→	}
   732→
   733→	// Read context
   734→	if file, fileErr := sessionTree.File(paths.ContextFileName); fileErr == nil {
   735→		if content, contentErr := file.Contents(); contentErr == nil {
   736→			result.Context = content
   737→		}
   738→	}
   739→
   740→	return result, nil
   741→}

   743→// ReadLatestSessionContent is a convenience method that reads the latest session's content.
   744→// This is equivalent to ReadSessionContent(ctx, checkpointID, len(summary.Sessions)-1).
   745→func (s *GitStore) ReadLatestSessionContent(ctx context.Context, checkpointID id.CheckpointID) (*SessionContent, error) {
   746→	summary, err := s.ReadCommitted(ctx, checkpointID)
   747→	if err != nil {
   748→		return nil, err
   749→	}
   750→	if summary == nil {
   751→		return nil, ErrCheckpointNotFound
   752→	}
   753→	if len(summary.Sessions) == 0 {
   754→		return nil, fmt.Errorf("checkpoint has no sessions: %s", checkpointID)

   762→// This is useful when you have the session ID but don't know its index within the checkpoint.

   769→	}

   788→// ListCommitted lists all committed checkpoints from the entire/checkpoints/v1 branch.
   789→// Scans sharded paths: <id[:2]>/<id[2:]>/ directories containing metadata.json.
   790→//
   791→
   792→func (s *GitStore) ListCommitted(ctx context.Context) ([]CommittedInfo, error) {
   793→	_ = ctx // Reserved for future use
   794→
   795→	tree, err := s.getSessionsBranchTree()
   796→	if err != nil {
   797→		return []CommittedInfo{}, nil //nolint:nilerr // No sessions branch means empty list
   798→	}
   799→
   800→	var checkpoints []CommittedInfo
   801→
   802→	// Scan sharded structure: <2-char-prefix>/<remaining-id>/metadata.json
   803→	for _, bucketEntry := range tree.Entries {
   804→		if bucketEntry.Mode != filemode.Dir {
   805→			continue
   806→		}
   807→		// Bucket should be 2 hex chars
   808→		if len(bucketEntry.Name) != 2 {
   809→			continue
   810→		}
   811→
   812→		bucketTree, treeErr := s.repo.TreeObject(bucketEntry.Hash)
   813→		if treeErr != nil {
   814→			continue
   815→		}
   816→
   817→		// Each entry in the bucket is the remaining part of the checkpoint ID
   818→		for _, checkpointEntry := range bucketTree.Entries {
   819→			if checkpointEntry.Mode != filemode.Dir {
   820→				continue
   821→			}
   822→
   823→			checkpointTree, cpTreeErr := s.repo.TreeObject(checkpointEntry.Hash)
   824→			if cpTreeErr != nil {
   825→				continue
   826→			}
   827→
   828→			// Reconstruct checkpoint ID: <bucket><remaining>
   829→			checkpointIDStr := bucketEntry.Name + checkpointEntry.Name
   830→			checkpointID, cpIDErr := id.NewCheckpointID(checkpointIDStr)
   831→			if cpIDErr != nil {
   832→				// Skip invalid checkpoint IDs (shouldn't happen with our own data)
   833→				continue
   834→			}
   835→
   836→			info := CommittedInfo{
   837→				CheckpointID: checkpointID,
   838→			}
   839→
   840→			// Get details from root metadata file (CheckpointSummary format)
   841→			if metadataFile, fileErr := checkpointTree.File(paths.MetadataFileName); fileErr == nil {
   842→				if content, contentErr := metadataFile.Contents(); contentErr == nil {
   843→					var summary CheckpointSummary
   844→					if err := json.Unmarshal([]byte(content), &summary); err == nil {
   845→						info.CheckpointsCount = summary.CheckpointsCount
   846→						info.FilesTouched = summary.FilesTouched
   847→						info.SessionCount = len(summary.Sessions)
   848→
   849→						// Read session metadata from latest session to get Agent, SessionID, CreatedAt
   850→						if len(summary.Sessions) > 0 {
   851→							latestIndex := len(summary.Sessions) - 1
   852→							latestDir := strconv.Itoa(latestIndex)
   853→							if sessionTree, treeErr := checkpointTree.Tree(latestDir); treeErr == nil {
   854→								if sessionMetadataFile, smErr := sessionTree.File(paths.MetadataFileName); smErr == nil {
   855→									if sessionContent, scErr := sessionMetadataFile.Contents(); scErr == nil {
   856→										var sessionMetadata CommittedMetadata
   857→										if json.Unmarshal([]byte(sessionContent), &sessionMetadata) == nil {
   858→											info.Agent = sessionMetadata.Agent
   859→											info.SessionID = sessionMetadata.SessionID
   860→											info.CreatedAt = sessionMetadata.CreatedAt
   861→										}
   862→									}
   863→								}
   864→							}
   865→						}
   866→					}
   867→				}
   868→			}
   869→
   870→			checkpoints = append(checkpoints, info)
   871→		}
   872→	}
   873→
   874→	// Sort by time (most recent first)
   875→	sort.Slice(checkpoints, func(i, j int) bool {
   876→		return checkpoints[i].CreatedAt.After(checkpoints[j].CreatedAt)
   877→	})
   878→
   879→	return checkpoints, nil
   880→}

   927→// UpdateSummary updates the summary field in the latest session's metadata.
   928→// Returns ErrCheckpointNotFound if the checkpoint doesn't exist.
   929→func (s *GitStore) UpdateSummary(ctx context.Context, checkpointID id.CheckpointID, summary *Summary) error {
   930→	_ = ctx // Reserved for future use
   931→
   932→	// Ensure sessions branch exists
   933→	if err := s.ensureSessionsBranch(); err != nil {
   934→		return fmt.Errorf("failed to ensure sessions branch: %w", err)
   935→	}
   936→
   937→	// Get current branch tip and flatten tree
   938→	ref, entries, err := s.getSessionsBranchEntries()
   939→	if err != nil {
   940→		return err
   941→	}
   942→
   943→	// Read root CheckpointSummary to find the latest session
   944→	basePath := checkpointID.Path() + "/"
   945→	rootMetadataPath := basePath + paths.MetadataFileName
   946→	entry, exists := entries[rootMetadataPath]
   947→	if !exists {
   948→		return ErrCheckpointNotFound
   949→	}
   950→
   951→	checkpointSummary, err := s.readSummaryFromBlob(entry.Hash)
   952→	if err != nil {
   953→		return fmt.Errorf("failed to read checkpoint summary: %w", err)
   954→	}
   955→
   956→	// Find the latest session's metadata path (0-based indexing)
   957→	latestIndex := len(checkpointSummary.Sessions) - 1
   958→	sessionMetadataPath := fmt.Sprintf("%s%d/%s", basePath, latestIndex, paths.MetadataFileName)
   959→	sessionEntry, exists := entries[sessionMetadataPath]
   960→	if !exists {
   961→		return fmt.Errorf("session metadata not found at %s", sessionMetadataPath)
   962→	}
   963→
   964→	// Read and update session metadata
   965→	existingMetadata, err := s.readMetadataFromBlob(sessionEntry.Hash)
   966→	if err != nil {
   967→		return fmt.Errorf("failed to read session metadata: %w", err)
   968→	}
   969→
   970→	// Update the summary
   971→	existingMetadata.Summary = summary
   972→
   973→	// Write updated session metadata
   974→	metadataJSON, err := jsonutil.MarshalIndentWithNewline(existingMetadata, "", "  ")
   975→	if err != nil {
   976→		return fmt.Errorf("failed to marshal metadata: %w", err)
   977→	}
   978→	metadataHash, err := CreateBlobFromContent(s.repo, metadataJSON)
   979→	if err != nil {
   980→		return fmt.Errorf("failed to create metadata blob: %w", err)
   981→	}
   982→	entries[sessionMetadataPath] = object.TreeEntry{
   983→		Name: sessionMetadataPath,
   984→		Mode: filemode.Regular,
   985→		Hash: metadataHash,
   986→	}
   987→
   988→	// Build and commit
   989→	newTreeHash, err := BuildTreeFromEntries(s.repo, entries)
   990→	if err != nil {
   991→		return err
   992→	}
   993→
   994→	authorName, authorEmail := getGitAuthorFromRepo(s.repo)
   995→	commitMsg := fmt.Sprintf("Update summary for checkpoint %s (session: %s)", checkpointID, existingMetadata.SessionID)
   996→	newCommitHash, err := s.createCommit(newTreeHash, ref.Hash(), commitMsg, authorName, authorEmail)
   997→	if err != nil {
   998→		return err
   999→	}
  1000→
  1001→	refName := plumbing.NewBranchReferenceName(paths.MetadataBranchName)
  1002→	newRef := plumbing.NewHashReference(refName, newCommitHash)
  1003→	if err := s.repo.Storer.SetReference(newRef); err != nil {
  1004→		return fmt.Errorf("failed to set branch reference: %w", err)
  1005→	}
  1006→
  1007→	return nil
  1008→}

  1010→// ensureSessionsBranch ensures the entire/checkpoints/v1 branch exists.
  1011→func (s *GitStore) ensureSessionsBranch() error {
  1012→	refName := plumbing.NewBranchReferenceName(paths.MetadataBranchName)
  1013→	_, err := s.repo.Reference(refName, true)
  1014→	if err == nil {
  1015→		return nil // Branch exists
  1016→	}
  1017→
  1018→	// Create orphan branch with empty tree
  1019→	emptyTreeHash, err := BuildTreeFromEntries(s.repo, make(map[string]object.TreeEntry))
  1020→	if err != nil {
  1021→		return err
  1022→	}
  1023→
  1024→	authorName, authorEmail := getGitAuthorFromRepo(s.repo)
  1025→	commitHash, err := s.createCommit(emptyTreeHash, plumbing.ZeroHash, "Initialize sessions branch", authorName, authorEmail)
  1026→	if err != nil {
  1027→		return err
  1028→	}
  1029→
  1030→	newRef := plumbing.NewHashReference(refName, commitHash)
  1031→	if err := s.repo.Storer.SetReference(newRef); err != nil {
  1032→		return fmt.Errorf("failed to set branch reference: %w", err)
  1033→	}
  1034→	return nil
  1035→}

  1037→// getSessionsBranchTree returns the tree object for the entire/checkpoints/v1 branch.
  1038→// Falls back to origin/entire/checkpoints/v1 if the local branch doesn't exist.
  1039→func (s *GitStore) getSessionsBranchTree() (*object.Tree, error) {
  1040→	refName := plumbing.NewBranchReferenceName(paths.MetadataBranchName)
  1041→	ref, err := s.repo.Reference(refName, true)
  1042→	if err != nil {
  1043→		// Local branch doesn't exist, try remote-tracking branch
  1044→		remoteRefName := plumbing.NewRemoteReferenceName("origin", paths.MetadataBranchName)
  1045→		ref, err = s.repo.Reference(remoteRefName, true)
  1046→		if err != nil {
  1047→			return nil, fmt.Errorf("sessions branch not found: %w", err)
  1048→		}
  1049→	}
  1050→
  1051→	commit, err := s.repo.CommitObject(ref.Hash())
  1052→	if err != nil {
  1053→		return nil, fmt.Errorf("failed to get commit object: %w", err)
  1054→	}
  1055→
  1056→	tree, err := commit.Tree()
  1057→	if err != nil {
  1058→		return nil, fmt.Errorf("failed to get commit tree: %w", err)
  1059→	}
  1060→
  1061→	return tree, nil
  1062→}

  1064→// CreateBlobFromContent creates a blob object from in-memory content.
  1065→// Exported for use by strategy package (session_test.go)
  1066→func CreateBlobFromContent(repo *git.Repository, content []byte) (plumbing.Hash, error) {
  1067→	obj := repo.Storer.NewEncodedObject()
  1068→	obj.SetType(plumbing.BlobObject)
  1069→	obj.SetSize(int64(len(content)))
  1070→
  1071→	writer, err := obj.Writer()
  1072→	if err != nil {
  1073→		return plumbing.ZeroHash, fmt.Errorf("failed to get object writer: %w", err)
  1074→	}
  1075→
  1076→	_, err = writer.Write(content)
  1077→	if err != nil {
  1078→		_ = writer.Close()
  1079→		return plumbing.ZeroHash, fmt.Errorf("failed to write blob content: %w", err)
  1080→	}
  1081→	if err := writer.Close(); err != nil {
  1082→		return plumbing.ZeroHash, fmt.Errorf("failed to close blob writer: %w", err)
  1083→	}
  1084→
  1085→	hash, err := repo.Storer.SetEncodedObject(obj)
  1086→	if err != nil {
  1087→		return plumbing.ZeroHash, fmt.Errorf("failed to store blob object: %w", err)
  1088→	}
  1089→	return hash, nil
  1090→}

  1092→// copyMetadataDir copies all files from a directory to the checkpoint path.
  1093→// Used to include additional metadata files like task checkpoints, subagent transcripts, etc.
  1094→func (s *GitStore) copyMetadataDir(metadataDir, basePath string, entries map[string]object.TreeEntry) error {
  1095→	err := filepath.Walk(metadataDir, func(path string, info os.FileInfo, err error) error {
  1096→		if err != nil {
  1097→			return err
  1098→		}
  1099→		if info.IsDir() {
  1100→			return nil
  1101→		}
  1102→
  1103→		// Skip symlinks to prevent reading files outside the metadata directory.
  1104→		// A symlink could point to sensitive files (e.g., /etc/passwd) which would
  1105→		// then be captured in the checkpoint and stored in git history.
  1106→		if info.Mode()&os.ModeSymlink != 0 {
  1107→			return nil
  1108→		}
  1109→
  1110→		// Get relative path within metadata dir
  1111→		relPath, err := filepath.Rel(metadataDir, path)
  1112→		if err != nil {
  1113→			return fmt.Errorf("failed to get relative path for %s: %w", path, err)
  1114→		}
  1115→
  1116→		// Prevent path traversal via symlinks pointing outside the metadata dir
  1117→		if strings.HasPrefix(relPath, "..") {
  1118→			return fmt.Errorf("path traversal detected: %s", relPath)
  1119→		}
  1120→
  1121→		// Create blob from file with secrets redaction
  1122→		blobHash, mode, err := createRedactedBlobFromFile(s.repo, path, relPath)
  1123→		if err != nil {
  1124→			return fmt.Errorf("failed to create blob for %s: %w", path, err)
  1125→		}
  1126→
  1127→		// Store at checkpoint path
  1128→		fullPath := basePath + relPath
  1129→		entries[fullPath] = object.TreeEntry{
  1130→			Name: fullPath,
  1131→			Mode: mode,
  1132→			Hash: blobHash,
  1133→		}
  1134→
  1135→		return nil
  1136→	})
  1137→	if err != nil {
  1138→		return fmt.Errorf("failed to walk metadata directory: %w", err)
  1139→	}
  1140→	return nil
  1141→}

  1143→// createRedactedBlobFromFile reads a file, applies secrets redaction, and creates a git blob.
  1144→// JSONL files get JSONL-aware redaction; all other files get plain string redaction.
  1145→func createRedactedBlobFromFile(repo *git.Repository, filePath, treePath string) (plumbing.Hash, filemode.FileMode, error) {
  1146→	info, err := os.Stat(filePath)
  1147→	if err != nil {
  1148→		return plumbing.ZeroHash, 0, fmt.Errorf("failed to stat file: %w", err)
  1149→	}
  1150→
  1151→	mode := filemode.Regular
  1152→	if info.Mode()&0o111 != 0 {
  1153→		mode = filemode.Executable
  1154→	}
  1155→
  1156→	content, err := os.ReadFile(filePath) //nolint:gosec // filePath comes from walking the metadata directory
  1157→	if err != nil {
  1158→		return plumbing.ZeroHash, 0, fmt.Errorf("failed to read file: %w", err)
  1159→	}
  1160→
  1161→	// Skip redaction for binary files — they can't contain text secrets and
  1162→	// running string replacement on them would corrupt the data.
  1163→	isBin, binErr := binary.IsBinary(bytes.NewReader(content))
  1164→	if binErr != nil || isBin {
  1165→		hash, err := CreateBlobFromContent(repo, content)
  1166→		if err != nil {
  1167→			return plumbing.ZeroHash, 0, fmt.Errorf("failed to create blob: %w", err)
  1168→		}
  1169→		return hash, mode, nil
  1170→	}
  1171→
  1172→	if strings.HasSuffix(treePath, ".jsonl") {
  1173→		content, err = redact.JSONLBytes(content)
  1174→		if err != nil {
