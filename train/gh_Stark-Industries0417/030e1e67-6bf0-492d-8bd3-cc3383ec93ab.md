   197→// ComputeFileChanges returns new files (created during session) and deleted files
   198→// (tracked files that were deleted) using a single git status call.
   199→// This is more efficient than calling ComputeNewFiles and ComputeDeletedFiles separately.
   200→//
   201→// If preState is nil, newFiles will be nil but deletedFiles will still be computed
   202→// (deleted files don't depend on pre-prompt state).
   203→func ComputeFileChanges(preState *PrePromptState) (newFiles, deletedFiles []string, err error) {
   204→	repo, err := openRepository()
   205→	if err != nil {
   206→		return nil, nil, err
   207→	}
   208→
   209→	worktree, err := repo.Worktree()
   210→	if err != nil {
   211→		return nil, nil, fmt.Errorf("failed to get worktree: %w", err)
   212→	}
   213→
   214→	status, err := worktree.Status()
   215→	if err != nil {
   216→		return nil, nil, fmt.Errorf("failed to get status: %w", err)
   217→	}
   218→
   219→	// Build set of pre-existing untracked files for quick lookup (only if preState exists)
   220→	var preExisting map[string]bool
   221→	if preState != nil {
   222→		preExisting = make(map[string]bool, len(preState.UntrackedFiles))
   223→		for _, f := range preState.UntrackedFiles {
   224→			preExisting[f] = true
   225→		}
   226→	}
   227→
   228→	// Process all files from git status
   229→	for file, st := range status {
   230→		// Skip .entire directory
   231→		if paths.IsInfrastructurePath(file) {
   232→			continue
   233→		}
   234→
   235→		switch {
   236→		case st.Worktree == git.Untracked && preState != nil:
   237→			// New file if it wasn't untracked before the session
   238→			// (only compute if we have preState to compare against)
   239→			if !preExisting[file] {
   240→				newFiles = append(newFiles, file)
   241→			}
   242→		case st.Worktree == git.Deleted && st.Staging != git.Deleted:
   243→			// Deleted file (tracked file that was deleted, not yet staged)
   244→			deletedFiles = append(deletedFiles, file)
   245→		}
   246→	}
   247→
   248→	return newFiles, deletedFiles, nil
   249→}

   483→// FindActivePreTaskFile finds an active pre-task file in .entire/tmp/ and returns
   484→// the parent Task's tool_use_id. Returns ("", false) if no pre-task file exists.
   485→// When multiple pre-task files exist (nested subagents), returns the most recently

   488→func FindActivePreTaskFile() (taskToolUseID string, found bool) {
   489→	tmpDirAbs, err := paths.AbsPath(paths.EntireTmpDir)
   490→	if err != nil {
   491→		tmpDirAbs = paths.EntireTmpDir // Fallback to relative
   492→	}
   493→	entries, err := os.ReadDir(tmpDirAbs)
   494→	if err != nil {
   495→		return "", false
   496→	}
   497→
   498→	var latestFile string
   499→	var latestTime time.Time
   500→
   501→	for _, entry := range entries {
   502→		if entry.IsDir() {
   503→			continue
   504→		}
   505→		name := entry.Name()
   506→		if !strings.HasPrefix(name, preTaskFilePrefix) || !strings.HasSuffix(name, ".json") {
   507→			continue
   508→		}
   509→
   510→		// Check modification time for nested subagent handling
   511→		info, err := entry.Info()
   512→		if err != nil {
   513→			continue
   514→		}
   515→		if latestFile == "" || info.ModTime().After(latestTime) {
   516→			latestFile = name
   517→			latestTime = info.ModTime()
   518→		}
   519→	}
   520→
   521→	if latestFile == "" {
   522→		return "", false
   523→	}
   524→

   526→	toolUseID := strings.TrimPrefix(latestFile, preTaskFilePrefix)
   527→	toolUseID = strings.TrimSuffix(toolUseID, ".json")
   528→	return toolUseID, true
