   678→// shellCompletionTarget returns the rc file path and completion lines for the
   679→// user's current shell.
   680→func shellCompletionTarget() (shellName, rcFile, completionLine string, err error) {
   681→	home, err := os.UserHomeDir()
   682→	if err != nil {
   683→		return "", "", "", fmt.Errorf("cannot determine home directory: %w", err)
   684→	}
   685→
   686→	shell := os.Getenv("SHELL")
   687→	switch {
   688→	case strings.Contains(shell, "zsh"):
   689→		return "Zsh",
   690→			filepath.Join(home, ".zshrc"),
   691→			"source <(entire completion zsh)",
   692→			nil
   693→	case strings.Contains(shell, "bash"):
   694→		bashRC := filepath.Join(home, ".bashrc")
   695→		if _, err := os.Stat(filepath.Join(home, ".bash_profile")); err == nil {
   696→			bashRC = filepath.Join(home, ".bash_profile")
   697→		}
   698→		return "Bash",
   699→			bashRC,
   700→			"source <(entire completion bash)",
   701→			nil
   702→	default:
   703→		return "", "", "", errUnsupportedShell
   704→	}
   705→}

   707→// promptShellCompletion offers to add shell completion to the user's rc file.
   708→// Only prompts if completion is not already configured.
   709→func promptShellCompletion(w io.Writer) error {
   710→	shellName, rcFile, completionLine, err := shellCompletionTarget()
   711→	if err != nil {
   712→		if errors.Is(err, errUnsupportedShell) {
   713→			fmt.Fprintf(w, "Note: Shell completion not available for your shell. Supported: zsh, bash.\n")
   714→			return nil
   715→		}
   716→		return fmt.Errorf("shell completion: %w", err)
   717→	}
   718→
   719→	if isCompletionConfigured(rcFile) {
   720→		fmt.Fprintf(w, "✓ Shell completion already configured in %s\n", rcFile)
   721→		return nil
   722→	}
   723→
   724→	var selected string
   725→	form := NewAccessibleForm(
   726→		huh.NewGroup(
   727→			huh.NewSelect[string]().
   728→				Title(fmt.Sprintf("Enable shell completion? (detected: %s)", shellName)).
   729→				Options(
   730→					huh.NewOption("Yes", "yes"),
   731→					huh.NewOption("No", "no"),
   732→				).
   733→				Value(&selected),
   734→		),
   735→	)
   736→
   737→	if err := form.Run(); err != nil {
   738→		//nolint:nilerr // User cancelled - not a fatal error, just skip
   739→		return nil
   740→	}
   741→
   742→	if selected != "yes" {
   743→		return nil
   744→	}
   745→
   746→	if err := appendShellCompletion(rcFile, completionLine); err != nil {
   747→		return fmt.Errorf("failed to update %s: %w", rcFile, err)
   748→	}
   749→
   750→	fmt.Fprintf(w, "✓ Shell completion added to %s\n", rcFile)
   751→	fmt.Fprintln(w, "  Restart your shell to activate")
   752→
   753→	return nil
   754→}

   756→// isCompletionConfigured checks if shell completion is already in the rc file.
   757→func isCompletionConfigured(rcFile string) bool {
   758→	//nolint:gosec // G304: rcFile is constructed from home dir + known filename, not user input
   759→	content, err := os.ReadFile(rcFile)
   760→	if err != nil {
   761→		return false // File doesn't exist or can't read, treat as not configured
   762→	}
   763→	return strings.Contains(string(content), "entire completion")
   764→}

   766→// appendShellCompletion adds the completion line to the rc file.
   767→func appendShellCompletion(rcFile, completionLine string) error {
   768→	//nolint:gosec // G302: Shell rc files need 0644 for user readability
   769→	f, err := os.OpenFile(rcFile, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o644)
   770→	if err != nil {
   771→		return fmt.Errorf("opening file: %w", err)
   772→	}
   773→	defer f.Close()
   774→
   775→	_, err = f.WriteString("\n" + shellCompletionComment + "\n" + completionLine + "\n")
   776→	if err != nil {
   777→		return fmt.Errorf("writing completion: %w", err)
   778→	}
   779→	return nil
   780→}
