   461→	}

   463→	// Filter the transcript through configured output filter
   464→	filteredTranscript, err := filter.Content(ctx, transcript, paths.TranscriptFileName)
   465→	if err != nil {
   466→		return fmt.Errorf("failed to filter transcript: %w", err)
   467→	}
   468→
   469→	// Chunk the filtered transcript if it's too large
   470→	chunks, err := agent.ChunkTranscript(filteredTranscript, opts.Agent)
   471→	if err != nil {
   472→		return fmt.Errorf("failed to chunk transcript: %w", err)
   473→	}
   474→
   475→	// Write chunk files
   476→	for i, chunk := range chunks {
   477→		chunkPath := basePath + agent.ChunkFileName(paths.TranscriptFileName, i)
   478→		blobHash, err := CreateBlobFromContent(s.repo, chunk)
   479→		if err != nil {
   480→			return err
   481→		}
   482→		entries[chunkPath] = object.TreeEntry{
   483→			Name: chunkPath,
   484→			Mode: filemode.Regular,
   485→			Hash: blobHash,
   486→		}
   487→	}
   488→
   489→	// Content hash for deduplication (hash of filtered transcript)
   490→	contentHash := fmt.Sprintf("sha256:%x", sha256.Sum256(filteredTranscript))
   491→	hashBlob, err := CreateBlobFromContent(s.repo, []byte(contentHash))
   492→	if err != nil {
   493→		return err
   494→	}
   495→	entries[basePath+paths.ContentHashFileName] = object.TreeEntry{
   496→		Name: basePath + paths.ContentHashFileName,
   497→		Mode: filemode.Regular,
   498→		Hash: hashBlob,
   499→	}
   500→	return nil
   501→}

   503→// mergeFilesTouched combines two file lists, removing duplicates.
   504→func mergeFilesTouched(existing, additional []string) []string {
   505→	seen := make(map[string]bool)
   506→	var result []string
   507→
   508→	for _, f := range existing {
   509→		if !seen[f] {
   510→			seen[f] = true
   511→			result = append(result, f)
   512→		}
   513→	}
   514→	for _, f := range additional {
   515→		if !seen[f] {
   516→			seen[f] = true
   517→			result = append(result, f)
   518→		}
   519→	}
   520→
   521→	sort.Strings(result)
   522→	return result
   523→}

   576→// ReadCommitted reads a committed checkpoint's summary by ID from the entire/sessions branch.
   577→// Returns only the CheckpointSummary (paths + aggregated stats), not actual content.
   578→// Use ReadSessionContent to read actual transcript/prompts/context.
   579→// Returns nil, nil if the checkpoint doesn't exist.
   580→//
   581→// The storage format uses numbered subdirectories for each session (0-based):
   582→//
   583→//	<checkpoint-id>/
   584→//	├── metadata.json      # CheckpointSummary with sessions map
   585→//	├── 0/                 # First session
   586→//	│   ├── metadata.json  # Session-specific metadata
   587→//	│   └── full.jsonl     # Transcript
   588→//	├── 1/                 # Second session
   589→//	└── ...
   590→func (s *GitStore) ReadCommitted(ctx context.Context, checkpointID id.CheckpointID) (*CheckpointSummary, error) {
   591→	_ = ctx // Reserved for future use
   592→
   593→	tree, err := s.getSessionsBranchTree()
   594→	if err != nil {
   595→		return nil, nil //nolint:nilnil,nilerr // No sessions branch means no checkpoint exists
   596→	}
   597→
   598→	checkpointPath := checkpointID.Path()
   599→	checkpointTree, err := tree.Tree(checkpointPath)
   600→	if err != nil {
   601→		return nil, nil //nolint:nilnil,nilerr // Checkpoint directory not found
   602→	}
   603→
   604→	// Read root metadata.json as CheckpointSummary
   605→	metadataFile, err := checkpointTree.File(paths.MetadataFileName)
   606→	if err != nil {
   607→		return nil, nil //nolint:nilnil,nilerr // metadata.json not found
   608→	}
   609→
   610→	content, err := metadataFile.Contents()
   611→	if err != nil {
   612→		return nil, fmt.Errorf("failed to read metadata.json: %w", err)
   613→	}
   614→
   615→	var summary CheckpointSummary
   616→	if err := json.Unmarshal([]byte(content), &summary); err != nil {
   617→		return nil, fmt.Errorf("failed to parse metadata.json: %w", err)
   618→	}
   619→
   620→	return &summary, nil
   621→}

   623→// ReadSessionContent reads the actual content for a specific session within a checkpoint.
   624→// sessionIndex is 0-based (0 for first session, 1 for second, etc.).
   625→// Returns the session's metadata, transcript, prompts, and context.
   626→// Returns an error if the checkpoint or session doesn't exist.
   627→func (s *GitStore) ReadSessionContent(ctx context.Context, checkpointID id.CheckpointID, sessionIndex int) (*SessionContent, error) {
   628→	_ = ctx // Reserved for future use
   629→
   630→	tree, err := s.getSessionsBranchTree()
   631→	if err != nil {
   632→		return nil, ErrCheckpointNotFound
   633→	}
   634→
   635→	checkpointPath := checkpointID.Path()
   636→	checkpointTree, err := tree.Tree(checkpointPath)
   637→	if err != nil {
   638→		return nil, ErrCheckpointNotFound
   639→	}
   640→
   641→	// Get the session subdirectory
   642→	sessionDir := strconv.Itoa(sessionIndex)
   643→	sessionTree, err := checkpointTree.Tree(sessionDir)
   644→	if err != nil {
   645→		return nil, fmt.Errorf("session %d not found: %w", sessionIndex, err)
   646→	}
   647→
   648→	result := &SessionContent{}
   649→
   650→	// Read session-specific metadata
   651→	var agentType agent.AgentType
   652→	if metadataFile, fileErr := sessionTree.File(paths.MetadataFileName); fileErr == nil {
   653→		if content, contentErr := metadataFile.Contents(); contentErr == nil {
   654→			if jsonErr := json.Unmarshal([]byte(content), &result.Metadata); jsonErr == nil {
   655→				agentType = result.Metadata.Agent
   656→			}
   657→		}
   658→	}
   659→
   660→	// Read transcript
   661→	if transcript, transcriptErr := readTranscriptFromTree(sessionTree, agentType); transcriptErr == nil && transcript != nil {
   662→		result.Transcript = transcript
   663→	}
   664→
   665→	// Read prompts
   666→	if file, fileErr := sessionTree.File(paths.PromptFileName); fileErr == nil {
   667→		if content, contentErr := file.Contents(); contentErr == nil {
   668→			result.Prompts = content
   669→		}
   670→	}
   671→
   672→	// Read context
   673→	if file, fileErr := sessionTree.File(paths.ContextFileName); fileErr == nil {
   674→		if content, contentErr := file.Contents(); contentErr == nil {
   675→			result.Context = content
   676→		}
   677→	}
   678→
   679→	return result, nil
   680→}

   727→// ListCommitted lists all committed checkpoints from the entire/sessions branch.
   728→// Scans sharded paths: <id[:2]>/<id[2:]>/ directories containing metadata.json.
   729→//
   730→
   731→func (s *GitStore) ListCommitted(ctx context.Context) ([]CommittedInfo, error) {
   732→	_ = ctx // Reserved for future use
   733→
   734→	tree, err := s.getSessionsBranchTree()
   735→	if err != nil {
   736→		return []CommittedInfo{}, nil //nolint:nilerr // No sessions branch means empty list
   737→	}
   738→
   739→	var checkpoints []CommittedInfo
   740→
   741→	// Scan sharded structure: <2-char-prefix>/<remaining-id>/metadata.json
   742→	for _, bucketEntry := range tree.Entries {
   743→		if bucketEntry.Mode != filemode.Dir {
   744→			continue
   745→		}
   746→		// Bucket should be 2 hex chars
   747→		if len(bucketEntry.Name) != 2 {
   748→			continue
   749→		}
   750→
   751→		bucketTree, treeErr := s.repo.TreeObject(bucketEntry.Hash)
   752→		if treeErr != nil {
   753→			continue
   754→		}
   755→
   756→		// Each entry in the bucket is the remaining part of the checkpoint ID
   757→		for _, checkpointEntry := range bucketTree.Entries {
   758→			if checkpointEntry.Mode != filemode.Dir {
   759→				continue
   760→			}
   761→
   762→			checkpointTree, cpTreeErr := s.repo.TreeObject(checkpointEntry.Hash)
   763→			if cpTreeErr != nil {
   764→				continue
   765→			}
   766→
   767→			// Reconstruct checkpoint ID: <bucket><remaining>
   768→			checkpointIDStr := bucketEntry.Name + checkpointEntry.Name
   769→			checkpointID, cpIDErr := id.NewCheckpointID(checkpointIDStr)
   770→			if cpIDErr != nil {
   771→				// Skip invalid checkpoint IDs (shouldn't happen with our own data)
   772→				continue
   773→			}
   774→
   775→			info := CommittedInfo{
   776→				CheckpointID: checkpointID,
   777→			}
   778→
   779→			// Get details from root metadata file (CheckpointSummary format)
   780→			if metadataFile, fileErr := checkpointTree.File(paths.MetadataFileName); fileErr == nil {
   781→				if content, contentErr := metadataFile.Contents(); contentErr == nil {
   782→					var summary CheckpointSummary
   783→					if err := json.Unmarshal([]byte(content), &summary); err == nil {
   784→						info.CheckpointsCount = summary.CheckpointsCount
   785→						info.FilesTouched = summary.FilesTouched
   786→						info.SessionCount = len(summary.Sessions)
   787→
   788→						// Read session metadata from latest session to get Agent, SessionID, CreatedAt
   789→						if len(summary.Sessions) > 0 {
   790→							latestIndex := len(summary.Sessions) - 1
   791→							latestDir := strconv.Itoa(latestIndex)
   792→							if sessionTree, treeErr := checkpointTree.Tree(latestDir); treeErr == nil {
   793→								if sessionMetadataFile, smErr := sessionTree.File(paths.MetadataFileName); smErr == nil {
   794→									if sessionContent, scErr := sessionMetadataFile.Contents(); scErr == nil {
   795→										var sessionMetadata CommittedMetadata
   796→										if json.Unmarshal([]byte(sessionContent), &sessionMetadata) == nil {
   797→											info.Agent = sessionMetadata.Agent
   798→											info.SessionID = sessionMetadata.SessionID
   799→											info.CreatedAt = sessionMetadata.CreatedAt
   800→										}
   801→									}
   802→								}
   803→							}
   804→						}
   805→					}
   806→				}
   807→			}
   808→
   809→			checkpoints = append(checkpoints, info)
   810→		}
   811→	}
   812→
   813→	// Sort by time (most recent first)
   814→	sort.Slice(checkpoints, func(i, j int) bool {
   815→		return checkpoints[i].CreatedAt.After(checkpoints[j].CreatedAt)
   816→	})
   817→
   818→	return checkpoints, nil
   819→}

   866→// UpdateSummary updates the summary field in the latest session's metadata.
   867→// Returns ErrCheckpointNotFound if the checkpoint doesn't exist.
   868→func (s *GitStore) UpdateSummary(ctx context.Context, checkpointID id.CheckpointID, summary *Summary) error {
   869→	_ = ctx // Reserved for future use
   870→
   871→	// Ensure sessions branch exists
   872→	if err := s.ensureSessionsBranch(); err != nil {
   873→		return fmt.Errorf("failed to ensure sessions branch: %w", err)
   874→	}
   875→
   876→	// Get current branch tip and flatten tree
   877→	ref, entries, err := s.getSessionsBranchEntries()
   878→	if err != nil {
   879→		return err
   880→	}
   881→
   882→	// Read root CheckpointSummary to find the latest session
   883→	basePath := checkpointID.Path() + "/"
   884→	rootMetadataPath := basePath + paths.MetadataFileName
   885→	entry, exists := entries[rootMetadataPath]
   886→	if !exists {
   887→		return ErrCheckpointNotFound
   888→	}
   889→
   890→	checkpointSummary, err := s.readSummaryFromBlob(entry.Hash)
   891→	if err != nil {
   892→		return fmt.Errorf("failed to read checkpoint summary: %w", err)
   893→	}
   894→
   895→	// Find the latest session's metadata path (0-based indexing)
   896→	latestIndex := len(checkpointSummary.Sessions) - 1
   897→	sessionMetadataPath := fmt.Sprintf("%s%d/%s", basePath, latestIndex, paths.MetadataFileName)
   898→	sessionEntry, exists := entries[sessionMetadataPath]
   899→	if !exists {
   900→		return fmt.Errorf("session metadata not found at %s", sessionMetadataPath)
   901→	}
   902→
   903→	// Read and update session metadata
   904→	existingMetadata, err := s.readMetadataFromBlob(sessionEntry.Hash)
   905→	if err != nil {
   906→		return fmt.Errorf("failed to read session metadata: %w", err)
   907→	}
   908→
   909→	// Update the summary
   910→	existingMetadata.Summary = summary
   911→
   912→	// Write updated session metadata
   913→	metadataJSON, err := jsonutil.MarshalIndentWithNewline(existingMetadata, "", "  ")
   914→	if err != nil {
   915→		return fmt.Errorf("failed to marshal metadata: %w", err)
   916→	}
   917→	metadataHash, err := CreateBlobFromContent(s.repo, metadataJSON)
   918→	if err != nil {
   919→		return fmt.Errorf("failed to create metadata blob: %w", err)
   920→	}
   921→	entries[sessionMetadataPath] = object.TreeEntry{
   922→		Name: sessionMetadataPath,
   923→		Mode: filemode.Regular,
   924→		Hash: metadataHash,
   925→	}
   926→
   927→	// Build and commit
   928→	newTreeHash, err := BuildTreeFromEntries(s.repo, entries)
   929→	if err != nil {
   930→		return err
   931→	}
   932→
   933→	authorName, authorEmail := getGitAuthorFromRepo(s.repo)
   934→	commitMsg := fmt.Sprintf("Update summary for checkpoint %s (session: %s)", checkpointID, existingMetadata.SessionID)
   935→	newCommitHash, err := s.createCommit(newTreeHash, ref.Hash(), commitMsg, authorName, authorEmail)
   936→	if err != nil {
   937→		return err
   938→	}
   939→
   940→	refName := plumbing.NewBranchReferenceName(paths.MetadataBranchName)
   941→	newRef := plumbing.NewHashReference(refName, newCommitHash)
   942→	if err := s.repo.Storer.SetReference(newRef); err != nil {
   943→		return fmt.Errorf("failed to set branch reference: %w", err)
   944→	}
   945→
   946→	return nil
   947→}

   949→// ensureSessionsBranch ensures the entire/sessions branch exists.
   950→func (s *GitStore) ensureSessionsBranch() error {
   951→	refName := plumbing.NewBranchReferenceName(paths.MetadataBranchName)
   952→	_, err := s.repo.Reference(refName, true)
   953→	if err == nil {
   954→		return nil // Branch exists
   955→	}
   956→
   957→	// Create orphan branch with empty tree
   958→	emptyTreeHash, err := BuildTreeFromEntries(s.repo, make(map[string]object.TreeEntry))
   959→	if err != nil {
   960→		return err
   961→	}
   962→
   963→	authorName, authorEmail := getGitAuthorFromRepo(s.repo)
   964→	commitHash, err := s.createCommit(emptyTreeHash, plumbing.ZeroHash, "Initialize sessions branch", authorName, authorEmail)
   965→	if err != nil {
   966→		return err
   967→	}
   968→
   969→	newRef := plumbing.NewHashReference(refName, commitHash)
   970→	if err := s.repo.Storer.SetReference(newRef); err != nil {
   971→		return fmt.Errorf("failed to set branch reference: %w", err)
   972→	}
   973→	return nil
   974→}

   976→// getSessionsBranchTree returns the tree object for the entire/sessions branch.
   977→// Falls back to origin/entire/sessions if the local branch doesn't exist.
   978→func (s *GitStore) getSessionsBranchTree() (*object.Tree, error) {
   979→	refName := plumbing.NewBranchReferenceName(paths.MetadataBranchName)
   980→	ref, err := s.repo.Reference(refName, true)
   981→	if err != nil {
   982→		// Local branch doesn't exist, try remote-tracking branch
   983→		remoteRefName := plumbing.NewRemoteReferenceName("origin", paths.MetadataBranchName)
   984→		ref, err = s.repo.Reference(remoteRefName, true)
   985→		if err != nil {
   986→			return nil, fmt.Errorf("sessions branch not found: %w", err)
   987→		}
   988→	}
   989→
   990→	commit, err := s.repo.CommitObject(ref.Hash())
   991→	if err != nil {
   992→		return nil, fmt.Errorf("failed to get commit object: %w", err)
   993→	}
   994→
   995→	tree, err := commit.Tree()
   996→	if err != nil {
   997→		return nil, fmt.Errorf("failed to get commit tree: %w", err)
   998→	}
   999→
  1000→	return tree, nil
  1001→}

  1003→// CreateBlobFromContent creates a blob object from in-memory content.
  1004→// Exported for use by strategy package (session_test.go)
  1005→func CreateBlobFromContent(repo *git.Repository, content []byte) (plumbing.Hash, error) {
  1006→	obj := repo.Storer.NewEncodedObject()
  1007→	obj.SetType(plumbing.BlobObject)
  1008→	obj.SetSize(int64(len(content)))
  1009→
  1010→	writer, err := obj.Writer()
  1011→	if err != nil {
  1012→		return plumbing.ZeroHash, fmt.Errorf("failed to get object writer: %w", err)
  1013→	}
