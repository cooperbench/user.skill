   436→// (source="commit"). It preserves existing trailers or restores from PendingCheckpointID.
   437→func (s *ManualCommitStrategy) handleAmendCommitMsg(logCtx context.Context, commitMsgFile string) error {
   438→	// Read current commit message
   439→	content, err := os.ReadFile(commitMsgFile) //nolint:gosec // commitMsgFile is provided by git hook
   440→	if err != nil {
   441→		return nil //nolint:nilerr // Hook must be silent on failure
   442→	}
   443→
   444→	message := string(content)
   445→
   446→	// If message already has a trailer, keep it unchanged
   447→	if existingCpID, found := trailers.ParseCheckpoint(message); found {
   448→		logging.Debug(logCtx, "prepare-commit-msg: amend preserves existing trailer",
   449→			slog.String("strategy", "manual-commit"),
   450→			slog.String("checkpoint_id", existingCpID.String()),
   451→		)
   452→		return nil
   453→	}
   454→
   455→	// No trailer in message — check if any session has PendingCheckpointID to restore
   456→	worktreePath, err := GetWorktreePath()
   457→	if err != nil {
   458→		return nil //nolint:nilerr // Hook must be silent on failure
   459→	}
   460→
   461→	sessions, err := s.findSessionsForWorktree(worktreePath)
   462→	if err != nil || len(sessions) == 0 {
   463→		return nil //nolint:nilerr // No sessions - nothing to restore
   464→	}
   465→
   466→	// Find first session with PendingCheckpointID
   467→	for _, state := range sessions {
   468→		if state.PendingCheckpointID == "" {
   469→			continue
   470→		}
   471→
   472→		cpID, err := id.NewCheckpointID(state.PendingCheckpointID)
   473→		if err != nil {
   474→			continue // Invalid ID, skip
   475→		}
   476→
   477→		// Restore the trailer
   478→		message = addCheckpointTrailer(message, cpID)
   479→		if writeErr := os.WriteFile(commitMsgFile, []byte(message), 0o600); writeErr != nil {
   480→			return nil //nolint:nilerr // Hook must be silent on failure
   481→		}
   482→
   483→		logging.Info(logCtx, "prepare-commit-msg: restored trailer on amend from PendingCheckpointID",
   484→			slog.String("strategy", "manual-commit"),
   485→			slog.String("checkpoint_id", cpID.String()),
   486→			slog.String("session_id", state.SessionID),
   487→		)
   488→		return nil
   489→	}
   490→
   491→	// No PendingCheckpointID found - leave message unchanged
   492→	logging.Debug(logCtx, "prepare-commit-msg: amend with no pending checkpoint to restore",
   493→		slog.String("strategy", "manual-commit"),
   494→	)
   495→	return nil
   496→}

   498→// PostCommit is called by the git post-commit hook after a commit is created.
   499→// Uses the session state machine to determine what action to take per session:
   500→//   - ACTIVE → ACTIVE_COMMITTED: defer condensation (agent still working)
   501→//   - IDLE → condense immediately
   502→//   - ACTIVE_COMMITTED → migrate shadow branch (additional commit during same turn)
   503→//   - ENDED → condense if files touched, discard if empty
   504→//
   505→// Shadow branches are only deleted when ALL sessions sharing the branch are non-active.
   506→// During rebase/cherry-pick/revert operations, phase transitions are skipped entirely.
   507→//
   508→//nolint:unparam // error return required by interface but hooks must return nil
   509→func (s *ManualCommitStrategy) PostCommit() error {
   510→	logCtx := logging.WithComponent(context.Background(), "checkpoint")
   511→
   512→	repo, err := OpenRepository()
   513→	if err != nil {
   514→		return nil //nolint:nilerr // Hook must be silent on failure
   515→	}
   516→
   517→	// Get HEAD commit to check for trailer
   518→	head, err := repo.Head()
   519→	if err != nil {
   520→		return nil //nolint:nilerr // Hook must be silent on failure
   521→	}
   522→
   523→	commit, err := repo.CommitObject(head.Hash())
   524→	if err != nil {
   525→		return nil //nolint:nilerr // Hook must be silent on failure
   526→	}
   527→
   528→	// Check if commit has checkpoint trailer (ParseCheckpoint validates format)
   529→	checkpointID, found := trailers.ParseCheckpoint(commit.Message)
   530→	if !found {
   531→		// No trailer - user removed it, treat as manual commit
   532→		logging.Debug(logCtx, "post-commit: no checkpoint trailer",
   533→			slog.String("strategy", "manual-commit"),
   534→		)
   535→		return nil
   536→	}
   537→
   538→	worktreePath, err := GetWorktreePath()
   539→	if err != nil {
   540→		return nil //nolint:nilerr // Hook must be silent on failure
   541→	}
   542→
   543→	// Find all active sessions for this worktree
   544→	sessions, err := s.findSessionsForWorktree(worktreePath)
   545→	if err != nil || len(sessions) == 0 {
   546→		logging.Warn(logCtx, "post-commit: no active sessions despite trailer",
   547→			slog.String("strategy", "manual-commit"),
   548→			slog.String("checkpoint_id", checkpointID.String()),
   549→		)
   550→		return nil //nolint:nilerr // Intentional: hooks must be silent on failure
   551→	}
   552→
   553→	// Build transition context
   554→	isRebase := isGitSequenceOperation()
   555→	transitionCtx := session.TransitionContext{
   556→		IsRebaseInProgress: isRebase,
   557→	}
   558→
   559→	if isRebase {
   560→		logging.Debug(logCtx, "post-commit: rebase/sequence in progress, skipping phase transitions",
   561→			slog.String("strategy", "manual-commit"),
   562→		)
   563→	}
   564→
   565→	// Track shadow branch names and whether they can be deleted
   566→	shadowBranchesToDelete := make(map[string]struct{})
   567→	// Track sessions that are still active AFTER transitions
   568→	activeSessionsOnBranch := make(map[string]bool)
   569→
   570→	newHead := head.Hash().String()
   571→
   572→	// Two-pass processing: condensation first, migration second.
   573→	// This prevents a migration from renaming a shadow branch before another
   574→	// session sharing that branch has had a chance to condense from it.
   575→	type pendingMigration struct {
   576→		state *SessionState
   577→	}
   578→	var pendingMigrations []pendingMigration
   579→
   580→	// Pass 1: Run transitions and dispatch condensation/discard actions.
   581→	// Defer migration actions to pass 2.
   582→	for _, state := range sessions {
   583→		shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   584→
   585→		// Check for new content (needed for TransitionContext and condensation).
   586→		// Fail-open: if content check errors, assume new content exists so we
   587→		// don't silently skip data that should have been condensed.
   588→		hasNew, contentErr := s.sessionHasNewContent(repo, state)
   589→		if contentErr != nil {
   590→			hasNew = true
   591→			logging.Debug(logCtx, "post-commit: error checking session content, assuming new content",
   592→				slog.String("session_id", state.SessionID),
   593→				slog.String("error", contentErr.Error()),
   594→			)
   595→		}
   596→		transitionCtx.HasFilesTouched = len(state.FilesTouched) > 0
   597→
   598→		// Run the state machine transition
   599→		remaining := TransitionAndLog(state, session.EventGitCommit, transitionCtx)
   600→
   601→		// Dispatch strategy-specific actions.
   602→		// Each branch handles its own BaseCommit update so there is no
   603→		// fallthrough conditional at the end. On condensation failure,
   604→		// BaseCommit is intentionally NOT updated to preserve access to
   605→		// the shadow branch (which is named after the old BaseCommit).
   606→		for _, action := range remaining {
   607→			switch action {
   608→			case session.ActionCondense:
   609→				if hasNew {
   610→					s.condenseAndUpdateState(logCtx, repo, checkpointID, state, head, shadowBranchName, shadowBranchesToDelete)
   611→					// condenseAndUpdateState updates BaseCommit on success.
   612→					// On failure, BaseCommit is preserved so the shadow branch remains accessible.
   613→				} else {
   614→					// No new content to condense — just update BaseCommit
   615→					s.updateBaseCommitIfChanged(logCtx, state, newHead)
   616→				}
   617→			case session.ActionCondenseIfFilesTouched:
   618→				// The state machine already gates this action on HasFilesTouched,
   619→				// but hasNew is an additional content-level check (transcript has
   620→				// new content beyond what was previously condensed).
   621→				if len(state.FilesTouched) > 0 && hasNew {
   622→					s.condenseAndUpdateState(logCtx, repo, checkpointID, state, head, shadowBranchName, shadowBranchesToDelete)
   623→					// On failure, BaseCommit is preserved (same as ActionCondense).
   624→				} else {
   625→					s.updateBaseCommitIfChanged(logCtx, state, newHead)
   626→				}
   627→			case session.ActionDiscardIfNoFiles:
   628→				if len(state.FilesTouched) == 0 {
   629→					logging.Debug(logCtx, "post-commit: skipping empty ended session (no files to condense)",
   630→						slog.String("session_id", state.SessionID),
   631→					)
   632→				}
   633→				s.updateBaseCommitIfChanged(logCtx, state, newHead)
   634→			case session.ActionMigrateShadowBranch:
   635→				// Deferred to pass 2 so condensation reads the old shadow branch first.
   636→				// Migration updates BaseCommit as part of the rename.
   637→				// Store checkpointID so HandleTurnEnd can reuse it for deferred condensation.
   638→				state.PendingCheckpointID = checkpointID.String()
   639→				pendingMigrations = append(pendingMigrations, pendingMigration{state: state})
   640→			case session.ActionClearEndedAt, session.ActionUpdateLastInteraction:
   641→				// Handled by session.ApplyCommonActions above
   642→			case session.ActionWarnStaleSession:
   643→				// Not produced by EventGitCommit; listed for switch exhaustiveness
   644→			}
   645→		}
   646→
   647→		// Save the updated state
   648→		if err := s.saveSessionState(state); err != nil {
   649→			fmt.Fprintf(os.Stderr, "[entire] Warning: failed to update session state: %v\n", err)
   650→		}
   651→
   652→		// Track whether any session on this shadow branch is still active
   653→		if state.Phase.IsActive() {
   654→			activeSessionsOnBranch[shadowBranchName] = true
   655→		}
   656→	}
   657→
   658→	// Pass 2: Run deferred migrations now that all condensations are complete.
   659→	for _, pm := range pendingMigrations {
   660→		if _, migErr := s.migrateShadowBranchIfNeeded(repo, pm.state); migErr != nil {
   661→			logging.Warn(logCtx, "post-commit: shadow branch migration failed",
   662→				slog.String("session_id", pm.state.SessionID),
   663→				slog.String("error", migErr.Error()),
   664→			)
   665→		}
   666→		// Save the migrated state
   667→		if err := s.saveSessionState(pm.state); err != nil {
   668→			fmt.Fprintf(os.Stderr, "[entire] Warning: failed to update session state after migration: %v\n", err)
   669→		}
   670→	}
   671→
   672→	// Clean up shadow branches — only delete when ALL sessions on the branch are non-active
   673→	for shadowBranchName := range shadowBranchesToDelete {
   674→		if activeSessionsOnBranch[shadowBranchName] {
   675→			logging.Debug(logCtx, "post-commit: preserving shadow branch (active session exists)",
   676→				slog.String("shadow_branch", shadowBranchName),
   677→			)
   678→			continue
   679→		}
   680→		if err := deleteShadowBranch(repo, shadowBranchName); err != nil {
   681→			fmt.Fprintf(os.Stderr, "[entire] Warning: failed to clean up %s: %v\n", shadowBranchName, err)
   682→		} else {
   683→			fmt.Fprintf(os.Stderr, "[entire] Cleaned up shadow branch: %s\n", shadowBranchName)
   684→			logging.Info(logCtx, "shadow branch deleted",
   685→				slog.String("strategy", "manual-commit"),
   686→				slog.String("shadow_branch", shadowBranchName),
   687→			)
   688→		}
   689→	}
   690→
   691→	return nil
   692→}

   694→// condenseAndUpdateState runs condensation for a session and updates state afterward.
   695→// Returns true if condensation succeeded.
   696→func (s *ManualCommitStrategy) condenseAndUpdateState(
   697→	logCtx context.Context,
   698→	repo *git.Repository,
   699→	checkpointID id.CheckpointID,
   700→	state *SessionState,
   701→	head *plumbing.Reference,
   702→	shadowBranchName string,
   703→	shadowBranchesToDelete map[string]struct{},
   704→) {
   705→	result, err := s.CondenseSession(repo, checkpointID, state)
   706→	if err != nil {
   707→		fmt.Fprintf(os.Stderr, "[entire] Warning: condensation failed for session %s: %v\n",
   708→			state.SessionID, err)
   709→		logging.Warn(logCtx, "post-commit: condensation failed",
   710→			slog.String("session_id", state.SessionID),
   711→			slog.String("error", err.Error()),
   712→		)
   713→		return
   714→	}
   715→
   716→	// Track this shadow branch for cleanup
   717→	shadowBranchesToDelete[shadowBranchName] = struct{}{}
   718→
   719→	// Update session state for the new base commit
   720→	state.BaseCommit = head.Hash().String()
   721→	state.StepCount = 0
   722→	state.CheckpointTranscriptStart = result.TotalTranscriptLines
   723→
   724→	// Clear attribution tracking — condensation already used these values
   725→	state.PromptAttributions = nil
   726→	state.PendingPromptAttribution = nil
   727→
   728→	// Save checkpoint ID so subsequent commits can reuse it
   729→	state.LastCheckpointID = checkpointID
   730→	// Set PendingCheckpointID for amend trailer restoration
   731→	state.PendingCheckpointID = checkpointID.String()
   732→
   733→	shortID := state.SessionID
   734→	if len(shortID) > 8 {
   735→		shortID = shortID[:8]
   736→	}
   737→	fmt.Fprintf(os.Stderr, "[entire] Condensed session %s: %s (%d checkpoints)\n",
   738→		shortID, result.CheckpointID, result.CheckpointsCount)
   739→
   740→	logging.Info(logCtx, "session condensed",
   741→		slog.String("strategy", "manual-commit"),
   742→		slog.String("checkpoint_id", result.CheckpointID.String()),
   743→		slog.Int("checkpoints_condensed", result.CheckpointsCount),
   744→		slog.Int("transcript_lines", result.TotalTranscriptLines),
   745→	)
   746→}

   779→// sessionHasNewContent checks if a session has new transcript content
   780→// beyond what was already condensed.
   781→func (s *ManualCommitStrategy) sessionHasNewContent(repo *git.Repository, state *SessionState) (bool, error) {
   782→	// Get shadow branch
   783→	shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   784→	refName := plumbing.NewBranchReferenceName(shadowBranchName)
   785→	ref, err := repo.Reference(refName, true)
   786→	if err != nil {
   787→		// No shadow branch means no Stop has happened since the last condensation.
   788→		// However, the agent may have done work (including commits) without a Stop.
   789→		// Check the live transcript to detect this scenario.
   790→		return s.sessionHasNewContentFromLiveTranscript(repo, state)
   791→	}
   792→
   793→	commit, err := repo.CommitObject(ref.Hash())
   794→	if err != nil {
   795→		return false, fmt.Errorf("failed to get commit object: %w", err)
   796→	}
   797→
   798→	tree, err := commit.Tree()
   799→	if err != nil {
   800→		return false, fmt.Errorf("failed to get commit tree: %w", err)
   801→	}
   802→
   803→	// Look for transcript file
   804→	metadataDir := paths.EntireMetadataDir + "/" + state.SessionID
   805→	var transcriptLines int
   806→
   807→	if file, fileErr := tree.File(metadataDir + "/" + paths.TranscriptFileName); fileErr == nil {
   808→		if content, contentErr := file.Contents(); contentErr == nil {
   809→			transcriptLines = countTranscriptLines(content)
   810→		}
   811→	} else if file, fileErr := tree.File(metadataDir + "/" + paths.TranscriptFileNameLegacy); fileErr == nil {
   812→		if content, contentErr := file.Contents(); contentErr == nil {
   813→			transcriptLines = countTranscriptLines(content)
   814→		}
   815→	}
   816→
   817→	// Has new content if there are more lines than already condensed
   818→	return transcriptLines > state.CheckpointTranscriptStart, nil
   819→}
   820→
   821→// countTranscriptLines counts lines in a transcript, matching the counting method used
   822→// in extractSessionData for consistency. This trims trailing empty lines (from final \n
   823→// in JSONL) but includes empty lines in the middle of the file.
   824→func countTranscriptLines(content string) int {
   825→	lines := strings.Split(content, "\n")
   826→	// Trim trailing empty lines (from final \n in JSONL)
   827→	for len(lines) > 0 && strings.TrimSpace(lines[len(lines)-1]) == "" {
   828→		lines = lines[:len(lines)-1]
   829→	}
   830→	return len(lines)
   831→}

   833→// sessionHasNewContentFromLiveTranscript checks if a session has new content
   834→// by examining the live transcript file. This is used when no shadow branch exists
   835→// (i.e., no Stop has happened yet) but the agent may have done work.
   836→//
   837→// Returns true if:
   838→//  1. The transcript has grown since the last condensation, AND
   839→//  2. The new transcript portion contains file modifications, AND
   840→//  3. At least one modified file overlaps with the currently staged files
   841→//
   842→// The overlap check ensures we don't add checkpoint trailers to commits that are
   843→// unrelated to the agent's recent changes.
   844→//
   845→// This handles the scenario where the agent commits mid-session before Stop.
   846→func (s *ManualCommitStrategy) sessionHasNewContentFromLiveTranscript(repo *git.Repository, state *SessionState) (bool, error) {
   847→	// Need both transcript path and agent type to analyze
   848→	if state.TranscriptPath == "" || state.AgentType == "" {
   849→		return false, nil
   850→	}
   851→
   852→	// Get the agent for transcript analysis
   853→	ag, err := agent.GetByAgentType(state.AgentType)
   854→	if err != nil {
   855→		return false, nil //nolint:nilerr // Unknown agent type, fail gracefully
   856→	}
   857→
   858→	// Cast to TranscriptAnalyzer
   859→	analyzer, ok := ag.(agent.TranscriptAnalyzer)
   860→	if !ok {
   861→		return false, nil // Agent doesn't support transcript analysis
   862→	}
   863→
   864→	// Get current transcript position
   865→	currentPos, err := analyzer.GetTranscriptPosition(state.TranscriptPath)
   866→	if err != nil {
   867→		return false, nil //nolint:nilerr // Error reading transcript, fail gracefully
   868→	}
   869→
   870→	// Check if transcript has grown since last condensation
   871→	if currentPos <= state.CheckpointTranscriptStart {
   872→		return false, nil // No new content
   873→	}
   874→
   875→	// Transcript has grown - check if there are file modifications in the new portion
   876→	modifiedFiles, _, err := analyzer.ExtractModifiedFilesFromOffset(state.TranscriptPath, state.CheckpointTranscriptStart)
   877→	if err != nil {
   878→		return false, nil //nolint:nilerr // Error parsing transcript, fail gracefully
   879→	}
   880→
   881→	// No file modifications means no new content to checkpoint
   882→	if len(modifiedFiles) == 0 {
   883→		return false, nil
   884→	}
   885→
   886→	// Check if any modified files overlap with currently staged files
   887→	// This ensures we only add checkpoint trailers to commits that include
   888→	// files the agent actually modified
   889→	stagedFiles := getStagedFiles(repo) // TODO: does it work? Do we have staged files if this is called at post-commit hook ?
   890→	if !hasOverlappingFiles(stagedFiles, modifiedFiles) {
   891→		return false, nil // No overlap - staged files are unrelated to agent's work
   892→	}
   893→
   894→	return true, nil
   895→}

   897→// addCheckpointTrailer adds the Entire-Checkpoint trailer to a commit message.
   898→// Handles proper trailer formatting (blank line before trailers if needed).
   899→func addCheckpointTrailer(message string, checkpointID id.CheckpointID) string {
   900→	trailer := trailers.CheckpointTrailerKey + ": " + checkpointID.String()
   901→
   902→	// If message already ends with trailers (lines starting with key:), just append
   903→	// Otherwise, add a blank line first
   904→	lines := strings.Split(strings.TrimRight(message, "\n"), "\n")
   905→
   906→	// Check if last non-empty, non-comment line looks like a trailer
   907→	// Git comment lines start with # and may contain ": " (e.g., "# Changes to be committed:")
   908→	hasTrailers := false
   909→	for i := len(lines) - 1; i >= 0; i-- {
   910→		line := strings.TrimSpace(lines[i])
   911→		if line == "" {
   912→			break
   913→		}
   914→		// Skip git comment lines
   915→		if strings.HasPrefix(line, "#") {
   916→			continue
   917→		}
   918→		if strings.Contains(line, ": ") {
   919→			hasTrailers = true
   920→			break
   921→		}
   922→		// Non-comment, non-trailer line found - no existing trailers
   923→		break
   924→	}
   925→
   926→	if hasTrailers {
   927→		// Append trailer directly
   928→		return strings.TrimRight(message, "\n") + "\n" + trailer + "\n"
   929→	}
   930→
   931→	// Add blank line before trailer
   932→	return strings.TrimRight(message, "\n") + "\n\n" + trailer + "\n"
   933→}

   935→// addCheckpointTrailerWithComment adds the Entire-Checkpoint trailer with an explanatory comment.
   936→// The trailer is placed above the git comment block but below the user's message area,
   937→// with a comment explaining that the user can remove it if they don't want to link the commit
   938→// to the agent session. If prompt is non-empty, it's shown as context.
   939→func addCheckpointTrailerWithComment(message string, checkpointID id.CheckpointID, agentName, prompt string) string {
   940→	trailer := trailers.CheckpointTrailerKey + ": " + checkpointID.String()
   941→	commentLines := []string{
   942→		"# Remove the Entire-Checkpoint trailer above if you don't want to link this commit to " + agentName + " session context.",
   943→	}
   944→	if prompt != "" {
   945→		commentLines = append(commentLines, "# Last Prompt: "+prompt)
   946→	}
   947→	commentLines = append(commentLines, "# The trailer will be added to your next commit based on this branch.")
   948→	comment := strings.Join(commentLines, "\n")
   949→
   950→	lines := strings.Split(message, "\n")
   951→
   952→	// Find where the git comment block starts (first # line)
   953→	commentStart := -1
   954→	for i, line := range lines {
   955→		if strings.HasPrefix(line, "#") {

    70→		if cp.SessionID == sessionID || strings.HasPrefix(cp.SessionID, sessionID) {
    71→			result = append(result, cp)
    72→		}
    73→	}
    74→
    75→	if len(result) == 0 {
    76→		return nil, fmt.Errorf("no checkpoints for session: %s", sessionID)
    77→	}
    78→
    79→	return result, nil
    80→}

   104→// CondenseSession condenses a session's shadow branch to permanent storage.
   105→// checkpointID is the 12-hex-char value from the Entire-Checkpoint trailer.
   106→// Metadata is stored at sharded path: <checkpoint_id[:2]>/<checkpoint_id[2:]>/
   107→// Uses checkpoint.GitStore.WriteCommitted for the git operations.
   108→func (s *ManualCommitStrategy) CondenseSession(repo *git.Repository, checkpointID id.CheckpointID, state *SessionState) (*CondenseResult, error) {
   109→	// Get shadow branch
   110→	shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   111→	refName := plumbing.NewBranchReferenceName(shadowBranchName)
   112→	ref, err := repo.Reference(refName, true)
   113→	if err != nil {
   114→		return nil, fmt.Errorf("shadow branch not found: %w", err)
   115→	}
   116→
   117→	// Extract session data from the shadow branch (with live transcript fallback).
   118→	// Use tracked files from session state instead of collecting all files from tree.
   119→	// Pass agent type to handle different transcript formats (JSONL for Claude, JSON for Gemini).
   120→	// Pass live transcript path so condensation reads the current file rather than a
   121→	// potentially stale shadow branch copy (SaveChanges may have been skipped if the
   122→	// last turn had no code changes).
   123→	sessionData, err := s.extractSessionData(repo, ref.Hash(), state.SessionID, state.FilesTouched, state.AgentType, state.TranscriptPath)
   124→	if err != nil {
   125→		return nil, fmt.Errorf("failed to extract session data: %w", err)
   126→	}
   127→
   128→	// Get checkpoint store
   129→	store, err := s.getCheckpointStore()
   130→	if err != nil {
   131→		return nil, fmt.Errorf("failed to get checkpoint store: %w", err)
   132→	}
   133→
   134→	// Get author info
   135→	authorName, authorEmail := GetGitAuthorFromRepo(repo)
   136→	attribution := calculateSessionAttributions(repo, ref, sessionData, state)
   137→	// Get current branch name
   138→	branchName := GetCurrentBranchName(repo)
   139→
   140→	// Generate summary if enabled
   141→	var summary *cpkg.Summary
   142→	if settings.IsSummarizeEnabled() && len(sessionData.Transcript) > 0 {
   143→		logCtx := logging.WithComponent(context.Background(), "attribution")
   144→		summarizeCtx := logging.WithComponent(logCtx, "summarize")
   145→
   146→		// Scope transcript to this checkpoint's portion
   147→		scopedTranscript := transcript.SliceFromLine(sessionData.Transcript, state.CheckpointTranscriptStart)
   148→		if len(scopedTranscript) > 0 {
   149→			var err error
   150→			summary, err = summarize.GenerateFromTranscript(summarizeCtx, scopedTranscript, sessionData.FilesTouched, nil)
   151→			if err != nil {
   152→				logging.Warn(summarizeCtx, "summary generation failed",
   153→					slog.String("session_id", state.SessionID),
   154→					slog.String("error", err.Error()))
   155→				// Continue without summary - non-blocking
   156→			} else {
   157→				logging.Info(summarizeCtx, "summary generated",
   158→					slog.String("session_id", state.SessionID))
   159→			}
   160→		}
   161→	}
   162→
   163→	// Write checkpoint metadata using the checkpoint store
   164→	if err := store.WriteCommitted(context.Background(), cpkg.WriteCommittedOptions{
   165→		CheckpointID:                checkpointID,
   166→		SessionID:                   state.SessionID,
   167→		Strategy:                    StrategyNameManualCommit,
   168→		Branch:                      branchName,
   169→		Transcript:                  sessionData.Transcript,
   170→		Prompts:                     sessionData.Prompts,
   171→		Context:                     sessionData.Context,
   172→		FilesTouched:                sessionData.FilesTouched,
   173→		CheckpointsCount:            state.StepCount,
   174→		EphemeralBranch:             shadowBranchName,
   175→		AuthorName:                  authorName,
   176→		AuthorEmail:                 authorEmail,
   177→		Agent:                       state.AgentType,
   178→		TranscriptIdentifierAtStart: state.TranscriptIdentifierAtStart,
   179→		CheckpointTranscriptStart:   state.CheckpointTranscriptStart,
   180→		TokenUsage:                  sessionData.TokenUsage,
   181→		InitialAttribution:          attribution,
   182→		Summary:                     summary,
   183→	}); err != nil {
   184→		return nil, fmt.Errorf("failed to write checkpoint metadata: %w", err)
   185→	}
   186→
   187→	return &CondenseResult{
   188→		CheckpointID:         checkpointID,
   189→		SessionID:            state.SessionID,
   190→		CheckpointsCount:     state.StepCount,
   191→		FilesTouched:         sessionData.FilesTouched,
   192→		TotalTranscriptLines: sessionData.FullTranscriptLines,
   193→	}, nil
   194→}

   196→func calculateSessionAttributions(repo *git.Repository, shadowRef *plumbing.Reference, sessionData *ExtractedSessionData, state *SessionState) *cpkg.InitialAttribution {
   197→	// Calculate initial attribution using accumulated prompt attribution data.
   198→	// This uses user edits captured at each prompt start (before agent works),
   199→	// plus any user edits after the final checkpoint (shadow → head).
   200→	logCtx := logging.WithComponent(context.Background(), "attribution")
   201→	var attribution *cpkg.InitialAttribution
   202→	headRef, headErr := repo.Head()
   203→	if headErr != nil {
   204→		logging.Debug(logCtx, "attribution skipped: failed to get HEAD",
   205→			slog.String("error", headErr.Error()))
   206→	} else {
   207→		headCommit, commitErr := repo.CommitObject(headRef.Hash())
   208→		if commitErr != nil {
   209→			logging.Debug(logCtx, "attribution skipped: failed to get HEAD commit",
   210→				slog.String("error", commitErr.Error()))
   211→		} else {
   212→			headTree, treeErr := headCommit.Tree()
   213→			if treeErr != nil {
   214→				logging.Debug(logCtx, "attribution skipped: failed to get HEAD tree",
   215→					slog.String("error", treeErr.Error()))
   216→			} else {
   217→				// Get shadow branch tree (checkpoint tree - what the agent wrote)
   218→				shadowCommit, shadowErr := repo.CommitObject(shadowRef.Hash())
   219→				if shadowErr != nil {
   220→					logging.Debug(logCtx, "attribution skipped: failed to get shadow commit",
   221→						slog.String("error", shadowErr.Error()),
   222→						slog.String("shadow_ref", shadowRef.Hash().String()))
   223→				} else {
   224→					shadowTree, shadowTreeErr := shadowCommit.Tree()
   225→					if shadowTreeErr != nil {
   226→						logging.Debug(logCtx, "attribution skipped: failed to get shadow tree",
   227→							slog.String("error", shadowTreeErr.Error()))
   228→					} else {
   229→						// Get base tree (state before session started)
   230→						var baseTree *object.Tree
   231→						if baseCommit, baseErr := repo.CommitObject(plumbing.NewHash(state.BaseCommit)); baseErr == nil {
   232→							if tree, baseTErr := baseCommit.Tree(); baseTErr == nil {
   233→								baseTree = tree
   234→							} else {
   235→								logging.Debug(logCtx, "attribution: base tree unavailable",
   236→									slog.String("error", baseTErr.Error()))
   237→							}
   238→						} else {
   239→							logging.Debug(logCtx, "attribution: base commit unavailable",
   240→								slog.String("error", baseErr.Error()),
   241→								slog.String("base_commit", state.BaseCommit))
   242→						}
   243→
   244→						// Log accumulated prompt attributions for debugging
   245→						var totalUserAdded, totalUserRemoved int
   246→						for i, pa := range state.PromptAttributions {
   247→							totalUserAdded += pa.UserLinesAdded
   248→							totalUserRemoved += pa.UserLinesRemoved
   249→							logging.Debug(logCtx, "prompt attribution data",
   250→								slog.Int("checkpoint", pa.CheckpointNumber),
   251→								slog.Int("user_added", pa.UserLinesAdded),
   252→								slog.Int("user_removed", pa.UserLinesRemoved),
   253→								slog.Int("agent_added", pa.AgentLinesAdded),
   254→								slog.Int("agent_removed", pa.AgentLinesRemoved),
   255→								slog.Int("index", i))
   256→						}
   257→
   258→						attribution = CalculateAttributionWithAccumulated(
   259→							baseTree,
   260→							shadowTree,
   261→							headTree,
   262→							sessionData.FilesTouched,
   263→							state.PromptAttributions,
   264→						)
   265→
   266→						if attribution != nil {
   267→							logging.Info(logCtx, "attribution calculated",
   268→								slog.Int("agent_lines", attribution.AgentLines),
   269→								slog.Int("human_added", attribution.HumanAdded),
   270→								slog.Int("human_modified", attribution.HumanModified),
   271→								slog.Int("human_removed", attribution.HumanRemoved),
   272→								slog.Int("total_committed", attribution.TotalCommitted),
   273→								slog.Float64("agent_percentage", attribution.AgentPercentage),
   274→								slog.Int("accumulated_user_added", totalUserAdded),
   275→								slog.Int("accumulated_user_removed", totalUserRemoved),
   276→								slog.Int("files_touched", len(sessionData.FilesTouched)))
   277→						}
   278→					}
   279→				}
   280→			}
   281→		}
   282→	}
   283→	return attribution
   284→}

   286→// extractSessionData extracts session data from the shadow branch.
   287→// filesTouched is the list of files tracked during the session (from SessionState.FilesTouched).
   288→// agentType identifies the agent (e.g., "Gemini CLI", "Claude Code") to determine transcript format.
   289→// liveTranscriptPath, when non-empty and readable, is preferred over the shadow branch copy.
   290→// This handles the case where SaveChanges was skipped (no code changes) but the transcript
   291→// continued growing — the shadow branch copy would be stale.
   292→func (s *ManualCommitStrategy) extractSessionData(repo *git.Repository, shadowRef plumbing.Hash, sessionID string, filesTouched []string, agentType agent.AgentType, liveTranscriptPath string) (*ExtractedSessionData, error) {
   293→	commit, err := repo.CommitObject(shadowRef)
   294→	if err != nil {
   295→		return nil, fmt.Errorf("failed to get commit object: %w", err)
   296→	}
   297→
   298→	tree, err := commit.Tree()
   299→	if err != nil {
   300→		return nil, fmt.Errorf("failed to get commit tree: %w", err)
   301→	}
   302→
   303→	data := &ExtractedSessionData{}
   304→	// sessionID is already an "entire session ID" (with date prefix)
   305→	metadataDir := paths.SessionMetadataDirFromSessionID(sessionID)
   306→
   307→	// Extract transcript — prefer the live file when available, fall back to shadow branch.
   308→	// The shadow branch copy may be stale if the last turn ended without code changes
   309→	// (SaveChanges is only called when there are file modifications).
   310→	var fullTranscript string
   311→	if liveTranscriptPath != "" {
   312→		if liveData, readErr := os.ReadFile(liveTranscriptPath); readErr == nil && len(liveData) > 0 { //nolint:gosec // path from session state
   313→			fullTranscript = string(liveData)
   314→		}
   315→	}
   316→	if fullTranscript == "" {
   317→		// Fall back to shadow branch copy
   318→		if file, fileErr := tree.File(metadataDir + "/" + paths.TranscriptFileName); fileErr == nil {
   319→			if content, contentErr := file.Contents(); contentErr == nil {
   320→				fullTranscript = content
   321→			}
   322→		} else if file, fileErr := tree.File(metadataDir + "/" + paths.TranscriptFileNameLegacy); fileErr == nil {
   323→			if content, contentErr := file.Contents(); contentErr == nil {
   324→				fullTranscript = content
   325→			}
   326→		}
   327→	}
   328→
   329→	// Process transcript based on agent type
   330→	if fullTranscript != "" {
   331→		// Check if this is a Gemini CLI transcript (JSON format, not JSONL)
   332→		isGeminiFormat := agentType == agent.AgentTypeGemini || isGeminiJSONTranscript(fullTranscript)
   333→
   334→		if isGeminiFormat {
   335→			// Gemini uses JSON format with a "messages" array
   336→			data.Transcript = []byte(fullTranscript)
   337→			data.FullTranscriptLines = 1 // JSON is a single "line"
   338→			data.Prompts = extractUserPromptsFromGeminiJSON(fullTranscript)
   339→			data.Context = generateContextFromPrompts(data.Prompts)
   340→		} else {
   341→			// Claude Code and others use JSONL format (one JSON object per line)
   342→			allLines := strings.Split(fullTranscript, "\n")
   343→
   344→			// Trim trailing empty lines (from final \n in JSONL)
   345→			for len(allLines) > 0 && strings.TrimSpace(allLines[len(allLines)-1]) == "" {
   346→				allLines = allLines[:len(allLines)-1]
   347→			}
   348→
   349→			data.FullTranscriptLines = len(allLines)
   350→
   351→			// Always store the full transcript for complete session history
   352→			data.Transcript = []byte(strings.Join(allLines, "\n"))
   353→
   354→			// Extract prompts from the full transcript
   355→			data.Prompts = extractUserPromptsFromLines(allLines)
   356→
   357→			// Generate context from prompts
   358→			data.Context = generateContextFromPrompts(data.Prompts)
   359→		}
   360→	}
   361→
   362→	// Use tracked files from session state (not all files in tree)
   363→	data.FilesTouched = filesTouched
   364→
   365→	// Calculate token usage from the extracted transcript portion
   366→	// TODO: Missing Gemini token usage
   367→	if len(data.Transcript) > 0 {
   368→		// TODO: Calculate token usage per transcript slice (only checkpoint related)
   369→		transcriptLines, err := claudecode.ParseTranscript(data.Transcript)
   370→		if err == nil && len(transcriptLines) > 0 {
   371→			data.TokenUsage = claudecode.CalculateTokenUsage(transcriptLines)
   372→		}
   373→	}
   374→
   375→	return data, nil
   376→}

   378→// isGeminiJSONTranscript detects if the transcript is in Gemini's JSON format.
   379→// Gemini transcripts start with a JSON object containing a "messages" array.
   380→func isGeminiJSONTranscript(content string) bool {
   381→	content = strings.TrimSpace(content)
   382→	// Quick check: Gemini JSON starts with { and contains "messages"
   383→	if !strings.HasPrefix(content, "{") {
   384→		return false
   385→	}
   386→	// Try to parse as Gemini format
   387→	var transcript struct {
   388→		Messages []json.RawMessage `json:"messages"`
   389→	}
   390→	if err := json.Unmarshal([]byte(content), &transcript); err != nil {
   391→		return false
   392→	}
   393→	return len(transcript.Messages) > 0
   394→}

   396→// extractUserPromptsFromGeminiJSON extracts user prompts from Gemini's JSON transcript format.
   397→// Gemini transcripts are structured as: {"messages": [{"type": "user", "content": "..."}, ...]}
   398→func extractUserPromptsFromGeminiJSON(content string) []string {
   399→	var transcript struct {
   400→		Messages []struct {
   401→			Type    string `json:"type"`
   402→			Content string `json:"content"`
   403→		} `json:"messages"`
   404→	}
   405→
   406→	if err := json.Unmarshal([]byte(content), &transcript); err != nil {
   407→		return nil
   408→	}
   409→
   410→	var prompts []string
   411→	for _, msg := range transcript.Messages {
   412→		if msg.Type == "user" && msg.Content != "" {
   413→			// Strip IDE context tags for consistency with Claude Code handling
   414→			cleaned := textutil.StripIDEContextTags(msg.Content)
   415→			if cleaned != "" {
   416→				prompts = append(prompts, cleaned)
   417→			}
   418→		}
   419→	}
   420→
   421→	return prompts
   422→}

   424→// extractUserPromptsFromLines extracts user prompts from JSONL transcript lines.
   425→// IDE-injected context tags (like <ide_opened_file>) are stripped from the results.
   426→func extractUserPromptsFromLines(lines []string) []string {
   427→	var prompts []string
   428→	for _, line := range lines {
   429→		line = strings.TrimSpace(line)
   430→		if line == "" {
   431→			continue
   432→		}
   433→
   434→		var entry map[string]interface{}
   435→		if err := json.Unmarshal([]byte(line), &entry); err != nil {
   436→			continue
   437→		}
   438→
   439→		// Check for user message (supports both "human" and "user" types)
   440→		msgType, ok := entry["type"].(string)
   441→		if !ok || (msgType != "human" && msgType != "user") {
   442→			continue
   443→		}
   444→
   445→		// Extract message content
   446→		message, ok := entry["message"].(map[string]interface{})
   447→		if !ok {
   448→			continue
   449→		}
   450→
   451→		// Handle string content
   452→		if content, ok := message["content"].(string); ok && content != "" {
   453→			cleaned := textutil.StripIDEContextTags(content)
   454→			if cleaned != "" {
   455→				prompts = append(prompts, cleaned)
   456→			}
   457→			continue
   458→		}
   459→
   460→		// Handle array content (e.g., multiple text blocks from VSCode)
   461→		if arr, ok := message["content"].([]interface{}); ok {
   462→			var texts []string
   463→			for _, item := range arr {
   464→				if m, ok := item.(map[string]interface{}); ok {
   465→					if m["type"] == "text" {
   466→						if text, ok := m["text"].(string); ok {
   467→							texts = append(texts, text)
   468→						}
   469→					}
   470→				}
   471→			}
   472→			if len(texts) > 0 {
   473→				cleaned := textutil.StripIDEContextTags(strings.Join(texts, "\n\n"))
   474→				if cleaned != "" {
   475→					prompts = append(prompts, cleaned)
   476→				}
   477→			}
   478→		}
   479→	}
   480→	return prompts
   481→}

   483→// generateContextFromPrompts generates context.md content from a list of prompts.
   484→func generateContextFromPrompts(prompts []string) []byte {
   485→	if len(prompts) == 0 {
   486→		return nil
   487→	}
   488→
   489→	var buf strings.Builder
   490→	buf.WriteString("# Session Context\n\n")
   491→	buf.WriteString("## User Prompts\n\n")
   492→
   493→	for i, prompt := range prompts {
   494→		// Truncate very long prompts for readability
   495→		displayPrompt := prompt
   496→		if len(displayPrompt) > 500 {
   497→			displayPrompt = displayPrompt[:500] + "..."
   498→		}
   499→		buf.WriteString(fmt.Sprintf("### Prompt %d\n\n", i+1))
   500→		buf.WriteString(displayPrompt)
   501→		buf.WriteString("\n\n")
   502→	}
   503→
   504→	return []byte(buf.String())
   505→}

   507→// CondenseSessionByID force-condenses a session by its ID and cleans up.
   508→// This is used by "entire sessions fix" to salvage stuck sessions.
   509→func (s *ManualCommitStrategy) CondenseSessionByID(sessionID string) error {
   510→	ctx := logging.WithComponent(context.Background(), "condense-by-id")
   511→
   512→	// Load session state
   513→	state, err := s.loadSessionState(sessionID)
   514→	if err != nil {
   515→		return fmt.Errorf("failed to load session state: %w", err)
   516→	}
   517→	if state == nil {
   518→		return fmt.Errorf("session not found: %s", sessionID)
   519→	}
   520→
   521→	// Open repository
   522→	repo, err := OpenRepository()
   523→	if err != nil {
   524→		return fmt.Errorf("failed to open repository: %w", err)
   525→	}
   526→
   527→	// Generate a checkpoint ID
   528→	checkpointID, err := id.Generate()
   529→	if err != nil {
   530→		return fmt.Errorf("failed to generate checkpoint ID: %w", err)
   531→	}
   532→
   533→	// Check if shadow branch exists (required for condensation)
   534→	shadowBranchName := getShadowBranchNameForCommit(state.BaseCommit, state.WorktreeID)
   535→	refName := plumbing.NewBranchReferenceName(shadowBranchName)
   536→	_, refErr := repo.Reference(refName, true)
   537→	hasShadowBranch := refErr == nil
   538→
   539→	if !hasShadowBranch {
   540→		// No shadow branch means no checkpoint data to condense.
   541→		// Just clean up the state file.
   542→		logging.Info(ctx, "no shadow branch for session, clearing state only",
   543→			slog.String("session_id", sessionID),
   544→			slog.String("shadow_branch", shadowBranchName),
   545→		)
   546→		if err := s.clearSessionState(sessionID); err != nil {
   547→			return fmt.Errorf("failed to clear session state: %w", err)
   548→		}
   549→		return nil
   550→	}
   551→
   552→	// Condense the session
   553→	result, err := s.CondenseSession(repo, checkpointID, state)
   554→	if err != nil {
   555→		return fmt.Errorf("failed to condense session: %w", err)
   556→	}
   557→
   558→	logging.Info(ctx, "session condensed by ID",
   559→		slog.String("session_id", sessionID),
   560→		slog.String("checkpoint_id", result.CheckpointID.String()),
   561→		slog.Int("checkpoints_condensed", result.CheckpointsCount),
   562→	)
   563→
   564→	// Update session state: reset step count and transition to idle
   565→	state.StepCount = 0
   566→	state.CheckpointTranscriptStart = result.TotalTranscriptLines
   567→	state.Phase = session.PhaseIdle
   568→	state.LastCheckpointID = checkpointID
   569→	state.PendingCheckpointID = checkpointID.String()
   570→	state.PromptAttributions = nil
   571→	state.PendingPromptAttribution = nil
   572→
   573→	if err := s.saveSessionState(state); err != nil {
   574→		return fmt.Errorf("failed to save session state: %w", err)
   575→	}
   576→
   577→	// Clean up shadow branch if no other sessions need it
   578→	if err := s.cleanupShadowBranchIfUnused(repo, shadowBranchName, sessionID); err != nil {
   579→		logging.Warn(ctx, "failed to clean up shadow branch",
   580→			slog.String("shadow_branch", shadowBranchName),
   581→			slog.String("error", err.Error()),
   582→		)
   583→		// Non-fatal: condensation succeeded, shadow branch cleanup is best-effort
   584→	}
   585→
   586→	return nil
   587→}

   589→// cleanupShadowBranchIfUnused deletes a shadow branch if no other active sessions reference it.
   590→func (s *ManualCommitStrategy) cleanupShadowBranchIfUnused(repo *git.Repository, shadowBranchName, excludeSessionID string) error {
   591→	// List all session states to check if any other session uses this shadow branch
   592→	allStates, err := s.listAllSessionStates()
   593→	if err != nil {

2. **PostCommit**: `ACTIVE → ACTIVE_COMMITTED`, sets `PendingCheckpointID = A` (for deferred condensation at turn end)
