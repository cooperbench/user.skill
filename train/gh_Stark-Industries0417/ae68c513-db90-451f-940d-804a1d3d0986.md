   136→		slog.String("hook_type", "agent"),
   137→		slog.String("model_session_id", input.SessionID),
   138→		slog.String("transcript_path", input.SessionRef),
   139→	)
   140→
   141→	sessionID := input.SessionID
   142→	if sessionID == "" {
   143→		sessionID = unknownSessionID
   144→	}
   145→
   146→	transcriptPath := input.SessionRef
   147→	if transcriptPath == "" || !fileExists(transcriptPath) {
   148→		return fmt.Errorf("transcript file not found or empty: %s", transcriptPath)
   149→	}
   150→
   151→	// Early check: bail out quickly if the repo has no commits yet.
   152→	// Without this, the function does a lot of work (transcript copy, prompt extraction, etc.)
   153→	// before eventually failing deep in strategy.SaveChanges().
   154→	if repo, err := strategy.OpenRepository(); err == nil && strategy.IsEmptyRepository(repo) {
   155→		fmt.Fprintln(os.Stderr, "Entire: skipping checkpoint. Will activate after first commit.")
   156→		return NewSilentError(strategy.ErrEmptyRepository)
   157→	}
   158→
   159→	// Create session metadata folder using the entire session ID (preserves original date on resume)
   160→	// Use AbsPath to ensure we create at repo root, not relative to cwd
   161→	sessionDir := paths.SessionMetadataDirFromSessionID(sessionID)
   162→	sessionDirAbs, err := paths.AbsPath(sessionDir)
   163→	if err != nil {
   164→		sessionDirAbs = sessionDir // Fallback to relative
   165→	}
   166→	if err := os.MkdirAll(sessionDirAbs, 0o750); err != nil {
   167→		return fmt.Errorf("failed to create session directory: %w", err)
   168→	}
   169→
   170→	// Wait for Claude Code to flush the transcript file.
   171→	// The stop hook fires before the transcript is fully written to disk.
   172→	// We poll for our own hook_progress sentinel entry in the file tail,
   173→	// which guarantees all prior entries have been flushed.
   174→	waitForTranscriptFlush(transcriptPath, time.Now())
   175→
   176→	// Copy transcript
   177→	logFile := filepath.Join(sessionDirAbs, paths.TranscriptFileName)
   178→	if err := copyFile(transcriptPath, logFile); err != nil {
   179→		return fmt.Errorf("failed to copy transcript: %w", err)
   180→	}
   181→	fmt.Fprintf(os.Stderr, "Copied transcript to: %s\n", sessionDir+"/"+paths.TranscriptFileName)
   182→
   183→	// Load pre-prompt state (captured on UserPromptSubmit)
   184→	// Needed for transcript offset and file change detection
   185→	preState, err := LoadPrePromptState(sessionID)
   186→	if err != nil {
   187→		fmt.Fprintf(os.Stderr, "Warning: failed to load pre-prompt state: %v\n", err)
   188→	}
   189→
   190→	// Determine transcript offset: prefer pre-prompt state, fall back to session state.
   191→	// Pre-prompt state has the offset when the transcript path was available at prompt time.
   192→	// Session state has the offset updated after each successful checkpoint save (auto-commit).
   193→	var transcriptOffset int
   194→	if preState != nil && preState.StepTranscriptStart > 0 {
   195→		transcriptOffset = preState.StepTranscriptStart
   196→		fmt.Fprintf(os.Stderr, "Pre-prompt state found: parsing transcript from line %d\n", transcriptOffset)
   197→	} else {
   198→		// Fall back to session state (e.g., auto-commit strategy updates it after each save)
   199→		sessionState, loadErr := strategy.LoadSessionState(sessionID)
   200→		if loadErr != nil {
   201→			fmt.Fprintf(os.Stderr, "Warning: failed to load session state: %v\n", loadErr)
   202→		}
   203→		if sessionState != nil && sessionState.CheckpointTranscriptStart > 0 {
   204→			transcriptOffset = sessionState.CheckpointTranscriptStart
   205→			fmt.Fprintf(os.Stderr, "Session state found: parsing transcript from line %d\n", transcriptOffset)
   206→		}
   207→	}
   208→
   209→	// Parse transcript (optionally from offset for strategies that track transcript position)
   210→	// When transcriptOffset > 0, only parse NEW lines since the last checkpoint
   211→	var transcript []transcriptLine
   212→	var totalLines int
   213→	if transcriptOffset > 0 {
   214→		// Parse only NEW lines since last checkpoint
   215→		transcript, totalLines, err = parseTranscriptFromLine(transcriptPath, transcriptOffset)
   216→		if err != nil {
   217→			return fmt.Errorf("failed to parse transcript from line %d: %w", transcriptOffset, err)
   218→		}
   219→		fmt.Fprintf(os.Stderr, "Parsed %d new transcript lines (total: %d)\n", len(transcript), totalLines)
   220→	} else {
   221→		// First prompt or no session state - parse entire transcript
   222→		// Use parseTranscriptFromLine with offset 0 to also get totalLines
   223→		transcript, totalLines, err = parseTranscriptFromLine(transcriptPath, 0)
   224→		if err != nil {
   225→			return fmt.Errorf("failed to parse transcript: %w", err)
   226→		}
   227→	}
   228→
   229→	// Extract all prompts since last checkpoint for prompt file
   230→	allPrompts := extractUserPrompts(transcript)
   231→	promptFile := filepath.Join(sessionDirAbs, paths.PromptFileName)
   232→	promptContent := strings.Join(allPrompts, "\n\n---\n\n")
   233→	if err := os.WriteFile(promptFile, []byte(promptContent), 0o600); err != nil {
   234→		return fmt.Errorf("failed to write prompt file: %w", err)
   235→	}
   236→	fmt.Fprintf(os.Stderr, "Extracted %d prompt(s) to: %s\n", len(allPrompts), sessionDir+"/"+paths.PromptFileName)
   237→
   238→	// Extract summary
   239→	summaryFile := filepath.Join(sessionDirAbs, paths.SummaryFileName)
   240→	summary := extractLastAssistantMessage(transcript)
   241→	if err := os.WriteFile(summaryFile, []byte(summary), 0o600); err != nil {
   242→		return fmt.Errorf("failed to write summary file: %w", err)
   243→	}
   244→	fmt.Fprintf(os.Stderr, "Extracted summary to: %s\n", sessionDir+"/"+paths.SummaryFileName)
   245→
   246→	// Get modified files from transcript
   247→	modifiedFiles := extractModifiedFiles(transcript)
   248→
   249→	// Generate commit message from last user prompt
   250→	lastPrompt := ""
   251→	if len(allPrompts) > 0 {
   252→		lastPrompt = allPrompts[len(allPrompts)-1]
   253→	}
   254→	commitMessage := generateCommitMessage(lastPrompt)
   255→	fmt.Fprintf(os.Stderr, "Using commit message: %s\n", commitMessage)
   256→
   257→	// Get repo root for path conversion (not cwd, since Claude may be in a subdirectory)
   258→	// Using cwd would filter out files in sibling directories (paths starting with ..)
   259→	repoRoot, err := paths.RepoRoot()
   260→	if err != nil {
   261→		return fmt.Errorf("failed to get repo root: %w", err)
   262→	}
   263→
   264→	if preState != nil {
   265→		fmt.Fprintf(os.Stderr, "Pre-prompt state: %d pre-existing untracked files\n", len(preState.UntrackedFiles))
   266→	}
   267→

   269→	changes, err := DetectFileChanges(preState.PreUntrackedFiles())

   271→		fmt.Fprintf(os.Stderr, "Warning: failed to compute file changes: %v\n", err)
   272→	}

   275→	relModifiedFiles := FilterAndNormalizePaths(modifiedFiles, repoRoot)
   276→	var relNewFiles, relDeletedFiles []string
   277→	if changes != nil {
   278→		relNewFiles = FilterAndNormalizePaths(changes.New, repoRoot)
   279→		relDeletedFiles = FilterAndNormalizePaths(changes.Deleted, repoRoot)
   280→	}

   283→	totalChanges := len(relModifiedFiles) + len(relNewFiles) + len(relDeletedFiles)
   284→	if totalChanges == 0 {

   286→		fmt.Fprintf(os.Stderr, "Skipping commit\n")

   288→		transitionSessionTurnEnd(sessionID)
   289→		// Auto-apply pending wingman review even when no file changes this turn
   290→		triggerWingmanAutoApplyIfPending(repoRoot)

   292→		if err := CleanupPrePromptState(sessionID); err != nil {
   293→			fmt.Fprintf(os.Stderr, "Warning: failed to cleanup pre-prompt state: %v\n", err)
   294→		}
   295→		return nil
   296→	}
   297→
   298→	fmt.Fprintf(os.Stderr, "Files modified during session (%d):\n", len(relModifiedFiles))
   299→	for _, file := range relModifiedFiles {
   300→		fmt.Fprintf(os.Stderr, "  - %s\n", file)
   301→	}
   302→	if len(relNewFiles) > 0 {
   303→		fmt.Fprintf(os.Stderr, "New files created (%d):\n", len(relNewFiles))
   304→		for _, file := range relNewFiles {
   305→			fmt.Fprintf(os.Stderr, "  + %s\n", file)
   306→		}
   307→	}
   308→	if len(relDeletedFiles) > 0 {
   309→		fmt.Fprintf(os.Stderr, "Files deleted (%d):\n", len(relDeletedFiles))
   310→		for _, file := range relDeletedFiles {
   311→			fmt.Fprintf(os.Stderr, "  - %s\n", file)
   312→		}
   313→	}

   316→	contextFile := filepath.Join(sessionDirAbs, paths.ContextFileName)
   317→	if err := createContextFileMinimal(contextFile, commitMessage, sessionID, promptFile, summaryFile, transcript); err != nil {
   318→		return fmt.Errorf("failed to create context file: %w", err)
   319→	}
   320→	fmt.Fprintf(os.Stderr, "Created context file: %s\n", sessionDir+"/"+paths.ContextFileName)
   321→
   322→	// Get git author from local/global config
   323→	author, err := GetGitAuthor()
   324→	if err != nil {
   325→		return fmt.Errorf("failed to get git author: %w", err)
   326→	}
   327→
   328→	// Get the configured strategy
   329→	strat := GetStrategy()
   330→
   331→	// Ensure strategy setup is in place (auto-installs git hook, gitignore, etc. if needed)
   332→	if err := strat.EnsureSetup(); err != nil {
   333→		fmt.Fprintf(os.Stderr, "Warning: failed to ensure strategy setup: %v\n", err)
   334→	}
   335→
   336→	// Get agent type from the currently executing hook agent (authoritative source)
   337→	var agentType agent.AgentType
   338→	if hookAgent, agentErr := GetCurrentHookAgent(); agentErr == nil {
   339→		agentType = hookAgent.Type()
   340→	}
   341→
   342→	// Get transcript position from pre-prompt state (captured at step/turn start)
   343→	var transcriptIdentifierAtStart string
   344→	var transcriptLinesAtStart int
   345→	if preState != nil {
   346→		transcriptIdentifierAtStart = preState.LastTranscriptIdentifier
   347→		transcriptLinesAtStart = preState.StepTranscriptStart
   348→	}
   349→
   350→	// Calculate token usage for this checkpoint (Claude Code specific)
   351→	var tokenUsage *agent.TokenUsage
   352→	if transcriptPath != "" {
   353→		// Subagents are stored in a subagents/ directory next to the main transcript
   354→		subagentsDir := filepath.Join(filepath.Dir(transcriptPath), sessionID, "subagents")
   355→		usage, err := claudecode.CalculateTotalTokenUsage(transcriptPath, transcriptLinesAtStart, subagentsDir)
   356→		if err != nil {
   357→			fmt.Fprintf(os.Stderr, "Warning: failed to calculate token usage: %v\n", err)
   358→		} else {
   359→			tokenUsage = usage
   360→		}
   361→	}
   362→
   363→	// Build fully-populated save context and delegate to strategy
   364→	ctx := strategy.SaveContext{
   365→		SessionID:                sessionID,
   366→		ModifiedFiles:            relModifiedFiles,
   367→		NewFiles:                 relNewFiles,
   368→		DeletedFiles:             relDeletedFiles,
   369→		MetadataDir:              sessionDir,
   370→		MetadataDirAbs:           sessionDirAbs,
   371→		CommitMessage:            commitMessage,
   372→		TranscriptPath:           transcriptPath,
   373→		AuthorName:               author.Name,
   374→		AuthorEmail:              author.Email,
   375→		AgentType:                agentType,
   376→		StepTranscriptIdentifier: transcriptIdentifierAtStart,
   377→		StepTranscriptStart:      transcriptLinesAtStart,
   378→		TokenUsage:               tokenUsage,
   379→	}
   380→
   381→	if err := strat.SaveChanges(ctx); err != nil {
   382→		return fmt.Errorf("failed to save changes: %w", err)
   383→	}
   384→
   385→	// Update session state with new transcript position for strategies that create
   386→	// commits on the active branch (auto-commit strategy). This prevents parsing old transcript
   387→	// lines on subsequent checkpoints.
   388→	// Note: Shadow strategy tracks transcript position per-step via StepTranscriptStart in
   389→	// pre-prompt state, but doesn't advance CheckpointTranscriptStart in session state because
   390→	// its checkpoints accumulate all files touched across the entire session.
   391→	if strat.Name() == strategy.StrategyNameAutoCommit {
   392→		// Load session state for updating transcript position
   393→		sessionState, loadErr := strategy.LoadSessionState(sessionID)
   394→		if loadErr != nil {
   395→			fmt.Fprintf(os.Stderr, "Warning: failed to load session state: %v\n", loadErr)
   396→		}
   397→		// Create session state lazily if it doesn't exist (backward compat for resumed sessions
   398→		// or if InitializeSession was never called/failed)
   399→		if sessionState == nil {
   400→			sessionState = &strategy.SessionState{
   401→				SessionID: sessionID,
   402→			}
   403→		}
   404→		sessionState.CheckpointTranscriptStart = totalLines
   405→		sessionState.StepCount++
   406→		if updateErr := strategy.SaveSessionState(sessionState); updateErr != nil {
   407→			fmt.Fprintf(os.Stderr, "Warning: failed to update session state: %v\n", updateErr)
   408→		} else {
   409→			fmt.Fprintf(os.Stderr, "Updated session state: transcript position=%d, checkpoint=%d\n",
   410→				totalLines, sessionState.StepCount)
   411→		}
   412→	}
   413→
   414→	// Fire EventTurnEnd to transition session phase (all strategies).
   415→	// This moves ACTIVE → IDLE or ACTIVE_COMMITTED → IDLE.
   416→	// For ACTIVE_COMMITTED → IDLE, HandleTurnEnd dispatches ActionCondense.
   417→	transitionSessionTurnEnd(sessionID)
   418→
   419→	// Trigger wingman review for auto-commit strategy (commit already happened
   420→	// in SaveChanges). Manual-commit triggers wingman from the git post-commit hook
   421→	// instead, since the user commits manually.
   422→	if totalChanges > 0 && strat.Name() == strategy.StrategyNameAutoCommit && settings.IsWingmanEnabled() {
   423→		triggerWingmanReview(WingmanPayload{
   424→			SessionID:     sessionID,
   425→			RepoRoot:      repoRoot,
   426→			ModifiedFiles: relModifiedFiles,
   427→			NewFiles:      relNewFiles,
   428→			DeletedFiles:  relDeletedFiles,
   429→			Prompts:       allPrompts,
   430→			CommitMessage: commitMessage,
   431→		})
   432→	}
   433→
   434→	// Auto-apply pending wingman review on turn end
   435→	triggerWingmanAutoApplyIfPending(repoRoot)
   436→
   437→	// Clean up pre-prompt state (CLI responsibility)
   438→	if err := CleanupPrePromptState(sessionID); err != nil {
   439→		fmt.Fprintf(os.Stderr, "Warning: failed to cleanup pre-prompt state: %v\n", err)
   440→	}
   441→
   442→	return nil
   443→}

   445→// handleClaudeCodePostTodo handles the PostToolUse[TodoWrite] hook for subagent checkpoints.
   446→// Creates a checkpoint if we're in a subagent context (active pre-task file exists).

   448→func handleClaudeCodePostTodo() error {

   462→		slog.String("hook", "post-todo"),
   463→		slog.String("hook_type", "subagent"),
   464→		slog.String("model_session_id", input.SessionID),
   465→		slog.String("transcript_path", input.TranscriptPath),
   466→		slog.String("tool_use_id", input.ToolUseID),
   467→	)
   468→
   469→	// Check if we're in a subagent context by looking for an active pre-task file
   470→	taskToolUseID, found := FindActivePreTaskFile()
   471→	if !found {
   472→		// Not in subagent context - this is a main agent TodoWrite, skip
   473→		return nil
   474→	}
   475→
   476→	// Skip on default branch to avoid polluting main/master history
   477→	if skip, branchName := ShouldSkipOnDefaultBranch(); skip {
   478→		fmt.Fprintf(os.Stderr, "Entire: skipping incremental checkpoint on branch '%s'\n", branchName)
   479→		return nil
   480→	}
   481→
   482→	// Detect file changes since last checkpoint
   483→	changes, err := DetectFileChanges(nil)
   484→	if err != nil {
   485→		fmt.Fprintf(os.Stderr, "Warning: failed to detect changed files: %v\n", err)
   486→		return nil
   487→	}
   488→
   489→	// If no file changes, skip creating a checkpoint
   490→	if len(changes.Modified) == 0 && len(changes.New) == 0 && len(changes.Deleted) == 0 {
   491→		fmt.Fprintf(os.Stderr, "[entire] No file changes detected, skipping incremental checkpoint\n")
   492→		return nil
   493→	}
   494→
   495→	// Get git author
   496→	author, err := GetGitAuthor()
   497→	if err != nil {
   498→		fmt.Fprintf(os.Stderr, "Warning: failed to get git author: %v\n", err)
   499→		return nil
   500→	}
   501→
   502→	// Get the active strategy
   503→	strat := GetStrategy()
   504→
   505→	// Ensure strategy setup is complete
   506→	if err := strat.EnsureSetup(); err != nil {

   509→	}

   527→		todoCount := CountTodosFromToolInput(input.ToolInput)

   529→			// Use "Planning: N todos" format for the first TodoWrite
   530→			todoContent = fmt.Sprintf("Planning: %d todos", todoCount)
   531→		}
   532→		// If todoCount == 0, todoContent remains empty and FormatIncrementalMessage
   533→		// will fall back to "Checkpoint #N" format
   534→	}
   535→
   536→	// Get agent type from the currently executing hook agent (authoritative source)
   537→	var agentType agent.AgentType
   538→	if hookAgent, agentErr := GetCurrentHookAgent(); agentErr == nil {
   539→		agentType = hookAgent.Type()
   540→	}
   541→
   542→	// Build incremental checkpoint context
   543→	ctx := strategy.TaskCheckpointContext{
   544→		SessionID:           sessionID,
   545→		ToolUseID:           taskToolUseID,
   546→		ModifiedFiles:       changes.Modified,
   547→		NewFiles:            changes.New,
   548→		DeletedFiles:        changes.Deleted,
   549→		TranscriptPath:      input.TranscriptPath,
   550→		AuthorName:          author.Name,
   551→		AuthorEmail:         author.Email,
   552→		IsIncremental:       true,
   553→		IncrementalSequence: seq,
   554→		IncrementalType:     input.ToolName,
   555→		IncrementalData:     input.ToolInput,
   556→		TodoContent:         todoContent,
   557→		AgentType:           agentType,
   558→	}
   559→
   560→	// Save incremental checkpoint
   561→	if err := strat.SaveTaskCheckpoint(ctx); err != nil {
   562→		fmt.Fprintf(os.Stderr, "Warning: failed to save incremental checkpoint: %v\n", err)
   563→		return nil
   564→	}
   565→
   566→	fmt.Fprintf(os.Stderr, "[entire] Created incremental checkpoint #%d for %s (task: %s)\n",
   567→		seq, input.ToolName, taskToolUseID[:min(12, len(taskToolUseID))])
   568→	return nil
   569→}

   921→// waitForTranscriptFlush polls the transcript file tail for the hook_progress
   922→// sentinel entry that Claude Code writes when launching the stop hook.
   923→// Once this entry appears in the file, all prior entries (assistant replies,
   924→// tool results) are guaranteed to have been flushed.
   925→//
   926→// hookStartTime is the approximate time our process started, used to avoid
   927→// matching stale sentinel entries from previous stop hook invocations.
   928→//
   929→// Falls back silently after a timeout — the transcript copy will proceed
   930→// with whatever data is available.
   931→func waitForTranscriptFlush(transcriptPath string, hookStartTime time.Time) {
   932→	const (
   933→		maxWait      = 3 * time.Second
   934→		pollInterval = 50 * time.Millisecond
   935→		tailBytes    = 4096 // Read last 4KB — sentinel is near the end
   936→		maxSkew      = 2 * time.Second
   937→	)
   938→
   939→	logCtx := logging.WithComponent(context.Background(), "hooks")
   940→	deadline := time.Now().Add(maxWait)
   941→	for time.Now().Before(deadline) {
   942→		if checkStopSentinel(transcriptPath, tailBytes, hookStartTime, maxSkew) {
   943→			logging.Debug(logCtx, "transcript flush sentinel found",
   944→				slog.Duration("wait", time.Since(hookStartTime)),
   945→			)
   946→			return
   947→		}
   948→		time.Sleep(pollInterval)
   949→	}
   950→	// Timeout — proceed with whatever is on disk.
   951→	logging.Warn(logCtx, "transcript flush sentinel not found within timeout, proceeding",
   952→		slog.Duration("timeout", maxWait),
   953→	)
   954→}

   956→// checkStopSentinel reads the tail of the transcript file and looks for a
   957→// hook_progress entry containing the stop hook sentinel, with a timestamp
   958→// close to hookStartTime.
   959→func checkStopSentinel(path string, tailBytes int64, hookStartTime time.Time, maxSkew time.Duration) bool {
   960→	f, err := os.Open(path) //nolint:gosec // path comes from agent hook input
   961→	if err != nil {
   962→		return false
   963→	}
   964→	defer f.Close()
   965→
   966→	// Seek to tail
   967→	info, err := f.Stat()
   968→	if err != nil {
   969→		return false
   970→	}
   971→	offset := info.Size() - tailBytes
   972→	if offset < 0 {
   973→		offset = 0
   974→	}
   975→	buf := make([]byte, info.Size()-offset)
   976→	if _, err := f.ReadAt(buf, offset); err != nil {
   977→		return false
   978→	}
   979→
   980→	// Scan lines from the tail for the sentinel
   981→	lines := strings.Split(string(buf), "\n")
   982→	for _, line := range lines {
   983→		line = strings.TrimSpace(line)
   984→		if line == "" || !strings.Contains(line, stopHookSentinel) {
   985→			continue
   986→		}
   987→
   988→		// Parse timestamp to check recency
   989→		var entry struct {
   990→			Timestamp string `json:"timestamp"`
   991→		}
   992→		if json.Unmarshal([]byte(line), &entry) != nil || entry.Timestamp == "" {
   993→			continue
   994→		}
   995→		ts, err := time.Parse(time.RFC3339Nano, entry.Timestamp)
   996→		if err != nil {
   997→			ts, err = time.Parse(time.RFC3339, entry.Timestamp)
   998→			if err != nil {
   999→				continue
  1000→			}
  1001→		}
  1002→
  1003→		// Check timestamp is within skew window of our start time
  1004→		if ts.After(hookStartTime.Add(-maxSkew)) && ts.Before(hookStartTime.Add(maxSkew)) {
  1005→			return true
  1006→		}
  1007→	}
  1008→	return false
  1009→}
