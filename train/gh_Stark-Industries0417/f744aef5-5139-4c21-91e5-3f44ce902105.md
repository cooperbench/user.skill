```
   255→func findCheckpointInHistory(start *object.Commit, stopAt *plumbing.Hash) *branchCheckpointResult {
   256→	result := &branchCheckpointResult{}

   258→	const maxCommits = 100 // Limit search depth
   259→	totalChecked := 0
   260→
   261→	current := start
   262→	for current != nil && totalChecked < maxCommits {
   263→		// Stop if we've reached the boundary
   264→		if stopAt != nil && current.Hash == *stopAt {
   265→			break
   266→		}
   267→
   268→		// Check for checkpoint trailer
   269→		if cpID, found := trailers.ParseCheckpoint(current.Message); found {
   270→			result.checkpointID = cpID
   271→			result.commitHash = current.Hash.String()
   272→			result.commitMessage = current.Message
   273→			// Only warn about branch work commits, not merge commits
   274→			result.newerCommitsExist = branchWorkCommits > 0
   275→			result.newerCommitCount = branchWorkCommits
   276→			return result
   277→		}
   278→
   279→		// Only count regular commits (not merge commits) as "branch work"
   280→		if current.NumParents() <= 1 {
   281→			branchWorkCommits++
   282→		}
   283→
   284→		totalChecked++
   285→
   286→		// Move to parent (first parent for merge commits - follows the main line)
   287→		if current.NumParents() == 0 {
   288→			break
   289→		}
   290→		parent, err := current.Parent(0)
   291→		if err != nil {
   292→			// Can't get parent, treat as end of history
   293→			break
   294→		}
   295→		current = parent
   296→	}
   297→
   298→	// No checkpoint found
   299→	return result
```

## Potential Issues with HTTPS URLs in `entire resume`

### The Problem

Two functions in `git_operations.go` use go-git's `Fetch()` without authentication options:

1. **`FetchAndCheckoutRemoteBranch`** (line 346-350)
2. **`FetchMetadataBranch`** (line 387-391)

```go
err = remote.Fetch(&git.FetchOptions{
    RefSpecs: []config.RefSpec{config.RefSpec(refSpec)},
})
```

go-git doesn't automatically use the system's credential helpers (like `git credential-osxkeychain` on macOS or `git-credential-manager` on Windows). For HTTPS URLs that require authentication, these fetches will fail.

### When This Affects `entire resume`

The issue only manifests when:
1. The target branch doesn't exist locally **AND** needs to be fetched from remote
2. The `entire/sessions` metadata branch doesn't exist locally **AND** needs to be fetched

If both the branch and metadata exist locally, resume works fine regardless of URL type.

### The Codebase Already Acknowledges This

In `push_common.go:185`:
```go
// Use git CLI for fetch (go-git's fetch can be tricky with auth)
fetchCmd := exec.CommandContext(ctx, "git", "fetch", remote, branchName)
```

The push/sync operations use the git CLI instead of go-git for exactly this reason, but the same fix wasn't applied to `FetchAndCheckoutRemoteBranch` and `FetchMetadataBranch`.

### Impact

| Scenario | SSH URL | HTTPS (public) | HTTPS (private) |
|----------|---------|----------------|-----------------|
| Branch exists locally | ✅ Works | ✅ Works | ✅ Works |
| Branch needs fetch | ✅ Works | ✅ Works | ❌ Fails |
| Metadata needs fetch | ✅ Works | ✅ Works | ❌ Fails |

### Fix

Replace go-git's `remote.Fetch()` with git CLI calls in both functions, similar to how `push_common.go` does it:

```go
cmd := exec.CommandContext(ctx, "git", "fetch", "origin", 
    fmt.Sprintf("+refs/heads/%s:refs/remotes/origin/%s", branchName, branchName))
```

```go
// ValidateBranchName checks if a branch name is valid using git check-ref-format.
// Returns an error if the name is invalid or contains unsafe characters.
func ValidateBranchName(branchName string) error {
	ctx := context.Background()
	cmd := exec.CommandContext(ctx, "git", "check-ref-format", "--branch", branchName)
	if err := cmd.Run(); err != nil {
		return fmt.Errorf("invalid branch name %q", branchName)
	}
	return nil
}
```
