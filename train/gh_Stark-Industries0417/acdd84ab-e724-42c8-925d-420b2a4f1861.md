```go
// EnsureEntireGitignore ensures all required entries are in .entire/.gitignore
// Works correctly from any subdirectory within the repository.
func EnsureEntireGitignore() error {
	// Get absolute path for the gitignore file
	gitignoreAbs, err := paths.AbsPath(entireGitignore)
	if err != nil {
		gitignoreAbs = entireGitignore // Fallback to relative
	}

	// Read existing content
	var content string
	if data, err := os.ReadFile(gitignoreAbs); err == nil {
		content = string(data)
	}

	// All entries that should be in .entire/.gitignore
	requiredEntries := []string{
		"tmp/",
		"settings.local.json",
		"metadata/",
		"logs/",
	}

	// Track what needs to be added
	var toAdd []string
	for _, entry := range requiredEntries {
		if !strings.Contains(content, entry) {
			toAdd = append(toAdd, entry)
		}
	}

	// Nothing to add
	if len(toAdd) == 0 {
		return nil
	}

	// Ensure .entire directory exists
	if err := os.MkdirAll(filepath.Dir(gitignoreAbs), 0o750); err != nil {
		return fmt.Errorf("failed to create .entire directory: %w", err)
	}

	// Append missing entries to gitignore
	var sb strings.Builder
	for _, entry := range toAdd {
		sb.WriteString(entry + "\n")
	}
	content += sb.String()

	if err := os.WriteFile(gitignoreAbs, []byte(content), 0o644); err != nil {
		return fmt.Errorf("failed to write gitignore: %w", err)
	}
	return nil
}
```

```go
func acquireWingmanLock(lockPath, sessionID string) bool {
	f, err := os.OpenFile(lockPath, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0o644)
	if err != nil {
		if !errors.Is(err, os.ErrExist) {
			return false
		}
		// Lock exists — check if it's stale
		info, statErr := os.Stat(lockPath)
		if statErr != nil || time.Since(info.ModTime()) <= staleLockThreshold {
			return false
		}
		// Stale lock detected — remove and retry atomically
		_ = os.Remove(lockPath)
		f, err = os.OpenFile(lockPath, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0o644)
		if err != nil {
			return false
		}
	}
	_, _ = f.WriteString(sessionID)
	_ = f.Close()
	return true
}
```

**Stale Lock Threshold:** 30 minutes (`staleLockThreshold = 30 * time.Minute`, line 412)
