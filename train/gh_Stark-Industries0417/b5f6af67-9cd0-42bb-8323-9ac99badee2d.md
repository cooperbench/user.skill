   456→	commitMessage := generateCommitMessage(lastPrompt)
   457→	fmt.Fprintf(os.Stderr, "Using commit message: %s\n", commitMessage)
   458→
   459→	// Get repo root for path conversion (not cwd, since Claude may be in a subdirectory)
   460→	// Using cwd would filter out files in sibling directories (paths starting with ..)
   461→	repoRoot, err := paths.RepoRoot()
   462→	if err != nil {
   463→		return fmt.Errorf("failed to get repo root: %w", err)
   464→	}
   465→
   466→	// Load pre-prompt state (captured on UserPromptSubmit)
   467→	preState, err := LoadPrePromptState(entireSessionID)
   468→	if err != nil {
   469→		fmt.Fprintf(os.Stderr, "Warning: failed to load pre-prompt state: %v\n", err)
   470→	}
   471→	if preState != nil {
   472→		fmt.Fprintf(os.Stderr, "Loaded pre-prompt state: %d pre-existing untracked files\n", len(preState.UntrackedFiles))
   473→	}
   474→
   475→	// Compute new files (files created during session)
   476→	newFiles, err := ComputeNewFiles(preState)
   477→	if err != nil {
   478→		fmt.Fprintf(os.Stderr, "Warning: failed to compute new files: %v\n", err)
   479→	}
   480→
   481→	// Compute deleted files (tracked files that were deleted)
   482→	deletedFiles, err := ComputeDeletedFiles()
   483→	if err != nil {
   484→		fmt.Fprintf(os.Stderr, "Warning: failed to compute deleted files: %v\n", err)
   485→	}
   486→
   487→	// Filter and normalize all paths (CLI responsibility)
   488→	relModifiedFiles := FilterAndNormalizePaths(modifiedFiles, repoRoot)
   489→	relNewFiles := FilterAndNormalizePaths(newFiles, repoRoot)
   490→	relDeletedFiles := FilterAndNormalizePaths(deletedFiles, repoRoot)
   491→
   492→	// Check if there are any changes to commit
   493→	totalChanges := len(relModifiedFiles) + len(relNewFiles) + len(relDeletedFiles)
   494→	if totalChanges == 0 {
   495→		fmt.Fprintf(os.Stderr, "No files were modified during this session\n")
   496→		fmt.Fprintf(os.Stderr, "Skipping commit\n")
   497→		// Clean up state even when skipping
   498→		if err := CleanupPrePromptState(entireSessionID); err != nil {
   499→			fmt.Fprintf(os.Stderr, "Warning: failed to cleanup pre-prompt state: %v\n", err)
   500→		}
   501→		return nil
   502→	}
   503→
   504→	fmt.Fprintf(os.Stderr, "Files modified during session (%d):\n", len(relModifiedFiles))
   505→	for _, file := range relModifiedFiles {
   506→		fmt.Fprintf(os.Stderr, "  - %s\n", file)
   507→	}
   508→	if len(relNewFiles) > 0 {
   509→		fmt.Fprintf(os.Stderr, "New files created (%d):\n", len(relNewFiles))
   510→		for _, file := range relNewFiles {
   511→			fmt.Fprintf(os.Stderr, "  + %s\n", file)
   512→		}
   513→	}
   514→	if len(relDeletedFiles) > 0 {
   515→		fmt.Fprintf(os.Stderr, "Files deleted (%d):\n", len(relDeletedFiles))
   516→		for _, file := range relDeletedFiles {
   517→			fmt.Fprintf(os.Stderr, "  - %s\n", file)
   518→		}
   519→	}
   520→
   521→	// Create context file before saving changes
   522→	contextFile := filepath.Join(sessionDirAbs, paths.ContextFileName)
   523→	if err := createContextFileMinimal(contextFile, commitMessage, entireSessionID, promptFile, summaryFile, transcript); err != nil {
   524→		return fmt.Errorf("failed to create context file: %w", err)
   525→	}
   526→	fmt.Fprintf(os.Stderr, "Created context file: %s\n", sessionDir+"/"+paths.ContextFileName)
   527→
   528→	// Get git author from local/global config
   529→	author, err := GetGitAuthor()
   530→	if err != nil {
   531→		return fmt.Errorf("failed to get git author: %w", err)
   532→	}
   533→
   534→	// Get the configured strategy
   535→	strat := GetStrategy()
   536→
   537→	// Ensure strategy setup is in place (auto-installs git hook, gitignore, etc. if needed)
   538→	if err := strat.EnsureSetup(); err != nil {
   539→		fmt.Fprintf(os.Stderr, "Warning: failed to ensure strategy setup: %v\n", err)
   540→	}
   541→
   542→	// Get agent type from session state (set during InitializeSession)
   543→	var agentType agent.AgentType
   544→	if sessionState != nil {
   545→		agentType = sessionState.AgentType
   546→	}
   547→
   548→	// Get transcript position from pre-prompt state (captured at checkpoint start)
   549→	var transcriptUUIDAtStart string
   550→	var transcriptLinesAtStart int
   551→	if preState != nil {
   552→		transcriptUUIDAtStart = preState.LastTranscriptUUID
   553→		transcriptLinesAtStart = preState.LastTranscriptLineCount
   554→	}
   555→
   556→	// Calculate token usage for this checkpoint (Claude Code specific)
   557→	var tokenUsage *agent.TokenUsage
   558→	if transcriptPath != "" {
   559→		// Subagents are stored in a subagents/ directory next to the main transcript
   560→		subagentsDir := filepath.Join(filepath.Dir(transcriptPath), entireSessionID, "subagents")
   561→		usage, err := claudecode.CalculateTotalTokenUsage(transcriptPath, transcriptLinesAtStart, subagentsDir)
   562→		if err != nil {
   563→			fmt.Fprintf(os.Stderr, "Warning: failed to calculate token usage: %v\n", err)
   564→		} else {
   565→			tokenUsage = usage
   566→		}
   567→	}
   568→
   569→	// Build fully-populated save context and delegate to strategy
   570→	ctx := strategy.SaveContext{
   571→		SessionID:              entireSessionID,
   572→		ModifiedFiles:          relModifiedFiles,
   573→		NewFiles:               relNewFiles,
   574→		DeletedFiles:           relDeletedFiles,
   575→		MetadataDir:            sessionDir,
   576→		MetadataDirAbs:         sessionDirAbs,
   577→		CommitMessage:          commitMessage,
   578→		TranscriptPath:         transcriptPath,
   579→		AuthorName:             author.Name,
   580→		AuthorEmail:            author.Email,
   581→		AgentType:              agentType,
   582→		TranscriptUUIDAtStart:  transcriptUUIDAtStart,
   583→		TranscriptLinesAtStart: transcriptLinesAtStart,
   584→		TokenUsage:             tokenUsage,
   585→	}
   586→
   587→	if err := strat.SaveChanges(ctx); err != nil {
   588→		return fmt.Errorf("failed to save changes: %w", err)
   589→	}
   590→
   591→	// Update session state with new transcript position for strategies that create
   592→	// commits on the active branch (auto-commit strategy). This prevents parsing old transcript
   593→	// lines on subsequent checkpoints.
   594→	// Note: Shadow strategy doesn't create commits on the active branch, so its
   595→	// checkpoints don't "consume" the transcript in the same way. Shadow should
   596→	// continue parsing the full transcript to capture all files touched in the session.
   597→	if strat.Name() == strategy.StrategyNameAutoCommit {
   598→		// Create session state lazily if it doesn't exist (backward compat for resumed sessions
   599→		// or if InitializeSession was never called/failed)
   600→		if sessionState == nil {
   601→			sessionState = &strategy.SessionState{
   602→				SessionID: entireSessionID,
   603→			}
   604→		}
   605→		sessionState.CondensedTranscriptLines = totalLines
   606→		sessionState.CheckpointCount++
   607→		if updateErr := strategy.SaveSessionState(sessionState); updateErr != nil {
   608→			fmt.Fprintf(os.Stderr, "Warning: failed to update session state: %v\n", updateErr)
   609→		} else {
   610→			fmt.Fprintf(os.Stderr, "Updated session state: transcript position=%d, checkpoint=%d\n",
   611→				totalLines, sessionState.CheckpointCount)
   612→		}
   613→	}
   614→
   615→	// Clean up pre-prompt state (CLI responsibility)
   616→	if err := CleanupPrePromptState(entireSessionID); err != nil {
   617→		fmt.Fprintf(os.Stderr, "Warning: failed to cleanup pre-prompt state: %v\n", err)
   618→	}
   619→
   620→	return nil
   621→}

   623→// handlePostTodo handles the PostToolUse[TodoWrite] hook for subagent checkpoints.
   624→// Creates a checkpoint if we're in a subagent context (active pre-task file exists).
   625→// Skips silently if not in subagent context (main agent).
   626→func handlePostTodo() error {
   627→	input, err := parseSubagentCheckpointHookInput(os.Stdin)
   628→	if err != nil {
   629→		return fmt.Errorf("failed to parse PostToolUse[TodoWrite] input: %w", err)
   630→	}
   631→
   632→	// Get agent for logging context
   633→	ag, err := GetCurrentHookAgent()
   634→	if err != nil {
   635→		return fmt.Errorf("failed to get agent: %w", err)
   636→	}
   637→
   638→	logCtx := logging.WithAgent(logging.WithComponent(context.Background(), "hooks"), ag.Name())
   639→	logging.Info(logCtx, "post-todo",
   640→		slog.String("hook", "post-todo"),
   641→		slog.String("hook_type", "subagent"),
   642→		slog.String("model_session_id", input.SessionID),
   643→		slog.String("transcript_path", input.TranscriptPath),
   644→		slog.String("tool_use_id", input.ToolUseID),
   645→	)
   646→
   647→	// Check if we're in a subagent context by looking for an active pre-task file
   648→	taskToolUseID, found := FindActivePreTaskFile()
   649→	if !found {
   650→		// Not in subagent context - this is a main agent TodoWrite, skip
   651→		return nil
   652→	}
   653→
   654→	// Skip on default branch to avoid polluting main/master history
   655→	if skip, branchName := ShouldSkipOnDefaultBranch(); skip {
   656→		fmt.Fprintf(os.Stderr, "Entire: skipping incremental checkpoint on branch '%s'\n", branchName)
   657→		return nil
   658→	}
   659→
   660→	// Detect file changes since last checkpoint
   661→	modified, newFiles, deleted, err := DetectChangedFiles()
   662→	if err != nil {
   663→		fmt.Fprintf(os.Stderr, "Warning: failed to detect changed files: %v\n", err)
   664→		return nil
   665→	}
   666→
   667→	// If no file changes, skip creating a checkpoint
   668→	if len(modified) == 0 && len(newFiles) == 0 && len(deleted) == 0 {
   669→		fmt.Fprintf(os.Stderr, "[entire] No file changes detected, skipping incremental checkpoint\n")
   670→		return nil
   671→	}
   672→
   673→	// Get git author
   674→	author, err := GetGitAuthor()
   675→	if err != nil {
   676→		fmt.Fprintf(os.Stderr, "Warning: failed to get git author: %v\n", err)
   677→		return nil
   678→	}
   679→
   680→	// Get the active strategy
   681→	strat := GetStrategy()
   682→
   683→	// Ensure strategy setup is complete
   684→	if err := strat.EnsureSetup(); err != nil {
   685→		fmt.Fprintf(os.Stderr, "Warning: failed to ensure strategy setup: %v\n", err)
   686→		return nil
   687→	}
   688→
   689→	// Get the session ID from the transcript path or input, then transform to Entire session ID
   690→	sessionID := paths.ExtractSessionIDFromTranscriptPath(input.TranscriptPath)
   691→	if sessionID == "" {
   692→		sessionID = input.SessionID
   693→	}
   694→	entireSessionID := currentSessionIDWithFallback(sessionID)
   695→
   696→	// Get next checkpoint sequence
   697→	seq := GetNextCheckpointSequence(entireSessionID, taskToolUseID)
   698→
   699→	// Extract the todo content from the tool_input.
   700→	// PostToolUse receives the NEW todo list where the just-completed work is
   701→	// marked as "completed". The last completed item is the work that was just done.
   702→	todoContent := ExtractLastCompletedTodoFromToolInput(input.ToolInput)
   703→	if todoContent == "" {
   704→		// No completed items - this is likely the first TodoWrite (planning phase).
   705→		// Check if there are any todos at all to avoid duplicate messages.
   706→		todoCount := CountTodosFromToolInput(input.ToolInput)
   707→		if todoCount > 0 {
   708→			// Use "Planning: N todos" format for the first TodoWrite
   709→			todoContent = fmt.Sprintf("Planning: %d todos", todoCount)
   710→		}
   711→		// If todoCount == 0, todoContent remains empty and FormatIncrementalMessage
   712→		// will fall back to "Checkpoint #N" format
   713→	}
   714→
   715→	// Get agent type from session state
   716→	var agentType agent.AgentType
   717→	if sessionState, loadErr := strategy.LoadSessionState(entireSessionID); loadErr == nil && sessionState != nil {
   718→		agentType = sessionState.AgentType
   719→	}
   720→
   721→	// Build incremental checkpoint context
   722→	ctx := strategy.TaskCheckpointContext{
   723→		SessionID:           entireSessionID,
   724→		ToolUseID:           taskToolUseID,
   725→		ModifiedFiles:       modified,
   726→		NewFiles:            newFiles,
   727→		DeletedFiles:        deleted,
   728→		TranscriptPath:      input.TranscriptPath,
   729→		AuthorName:          author.Name,
   730→		AuthorEmail:         author.Email,
   731→		IsIncremental:       true,
   732→		IncrementalSequence: seq,
   733→		IncrementalType:     input.ToolName,
   734→		IncrementalData:     input.ToolInput,
   735→		TodoContent:         todoContent,
   736→		AgentType:           agentType,
   737→	}
   738→
   739→	// Save incremental checkpoint
   740→	if err := strat.SaveTaskCheckpoint(ctx); err != nil {
   741→		fmt.Fprintf(os.Stderr, "Warning: failed to save incremental checkpoint: %v\n", err)
   742→		return nil
   743→	}
   744→
   745→	fmt.Fprintf(os.Stderr, "[entire] Created incremental checkpoint #%d for %s (task: %s)\n",
   746→		seq, input.ToolName, taskToolUseID[:min(12, len(taskToolUseID))])
   747→	return nil
   748→}

   750→// handlePreTask handles the PreToolUse[Task] hook

   757→
   758→	input, err := parseTaskHookInput(os.Stdin)
   759→	if err != nil {

   762→
   763→	// Get agent for logging context

   765→	if err != nil {

   777→
   778→	// Log context to stdout
   779→	logPreTaskHookContext(os.Stdout, input)
   780→
   781→	// Capture pre-task state
   782→	if err := CapturePreTaskState(input.ToolUseID); err != nil {

   784→	}
   785→
   786→	// Create "Starting agent" checkpoint
   787→	// This allows rewinding to the state just before the subagent began
   788→	if err := createStartingAgentCheckpoint(input); err != nil {

   790→		fmt.Fprintf(os.Stderr, "Warning: failed to create starting checkpoint: %v\n", err)
   791→	}
   792→
   793→	return nil
   794→}

   796→// createStartingAgentCheckpoint creates a checkpoint commit marking the start of a subagent.

   935→	repoRoot, err := paths.RepoRoot()
   936→	if err != nil {
   937→		return fmt.Errorf("failed to get repo root: %w", err)
   938→	}

   942→	relNewFiles := FilterAndNormalizePaths(newFiles, repoRoot)
   943→
   944→	// Find checkpoint UUID from main transcript (best-effort, ignore errors)
   945→	transcript, _ := parseTranscript(input.TranscriptPath) //nolint:errcheck // best-effort extraction
   946→	checkpointUUID, _ := FindCheckpointUUID(transcript, input.ToolUseID)
   947→

   949→	author, err := GetGitAuthor()
   950→	if err != nil {
   951→		return fmt.Errorf("failed to get git author: %w", err)
   952→	}
   953→
   954→	// Get the configured strategy
   955→	strat := GetStrategy()
   956→
   957→	// Ensure strategy setup is in place
   958→	if err := strat.EnsureSetup(); err != nil {
   959→		fmt.Fprintf(os.Stderr, "Warning: failed to ensure strategy setup: %v\n", err)
   960→	}
   961→
   962→	entireSessionID := currentSessionIDWithFallback(input.SessionID)
   963→
   964→	// Get agent type from session state
   965→	var agentType agent.AgentType
   966→	if sessionState, loadErr := strategy.LoadSessionState(entireSessionID); loadErr == nil && sessionState != nil {
   967→		agentType = sessionState.AgentType
   968→	}
   969→
   970→	// Build task checkpoint context - strategy handles metadata creation
   971→	// Note: Incremental checkpoints are now created during task execution via handlePostTodo,
   972→	// so we don't need to collect/cleanup staging area here.
   973→	ctx := strategy.TaskCheckpointContext{
